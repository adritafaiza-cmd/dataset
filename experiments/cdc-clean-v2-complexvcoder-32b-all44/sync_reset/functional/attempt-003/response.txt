=== STAGE 2: VERILOG ===
```verilog
module sync_reset #
(
    parameter N = 2
)
(
    input  wire clk,
    input  wire rst,
    output wire out
);

    reg [N-1:0] sync_chain;

    always @(posedge clk) begin
        if (rst) begin
            sync_chain <= {N{1'b1}};
        end else begin
            sync_chain <= {sync_chain[N-2:0], 1'b0};
        end
    end

    assign out = sync_chain[N-1];

endmodule
```
