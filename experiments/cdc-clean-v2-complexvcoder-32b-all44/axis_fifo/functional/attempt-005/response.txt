=== STAGE 2: VERILOG ===
```verilog
module axis_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    input  wire                   clk,
    input  wire                   rst,
    /*
     * AXI input
     */
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   pause_req,
    output wire                   pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] status_depth,
    output wire [$clog2(DEPTH):0] status_depth_commit,
    output wire                   status_overflow,
    output wire                   status_bad_frame,
    output wire                   status_good_frame
);

localparam ADDR_WIDTH = $clog2(DEPTH);

reg [DATA_WIDTH-1:0]  mem_data [0:DEPTH-1];
reg [KEEP_WIDTH-1:0]  mem_keep [0:DEPTH-1];
reg                   mem_last [0:DEPTH-1];
reg [ID_WIDTH-1:0]    mem_id [0:DEPTH-1];
reg [DEST_WIDTH-1:0]  mem_dest [0:DEPTH-1];
reg [USER_WIDTH-1:0]  mem_user [0:DEPTH-1];

reg [ADDR_WIDTH:0] wr_ptr, rd_ptr, depth, depth_commit;
reg overflow, bad_frame, good_frame;

reg s_axis_tready_reg;
reg m_axis_tvalid_reg;
reg m_axis_tlast_reg;
reg [DATA_WIDTH-1:0] m_axis_tdata_reg;
reg [KEEP_WIDTH-1:0] m_axis_tkeep_reg;
reg [ID_WIDTH-1:0] m_axis_tid_reg;
reg [DEST_WIDTH-1:0] m_axis_tdest_reg;
reg [USER_WIDTH-1:0] m_axis_tuser_reg;

reg pause_ack_reg;

always @(posedge clk or posedge rst) begin
    if (rst) begin
        wr_ptr <= 0;
        rd_ptr <= 0;
        depth <= 0;
        depth_commit <= 0;
        overflow <= 0;
        bad_frame <= 0;
        good_frame <= 0;
        s_axis_tready_reg <= 0;
        m_axis_tvalid_reg <= 0;
        m_axis_tlast_reg <= 0;
        m_axis_tdata_reg <= 0;
        m_axis_tkeep_reg <= 0;
        m_axis_tid_reg <= 0;
        m_axis_tdest_reg <= 0;
        m_axis_tuser_reg <= 0;
        pause_ack_reg <= 0;
    end else begin
        // Write to FIFO
        if (s_axis_tvalid && s_axis_tready_reg) begin
            mem_data[wr_ptr] <= s_axis_tdata;
            mem_keep[wr_ptr] <= s_axis_tkeep;
            mem_last[wr_ptr] <= s_axis_tlast;
            mem_id[wr_ptr] <= s_axis_tid;
            mem_dest[wr_ptr] <= s_axis_tdest;
            mem_user[wr_ptr] <= s_axis_tuser;
            wr_ptr <= wr_ptr + 1;
            depth <= depth + 1;
            if (depth == DEPTH)
                overflow <= 1;
        end

        // Read from FIFO
        if (m_axis_tvalid_reg && m_axis_tready) begin
            rd_ptr <= rd_ptr + 1;
            depth <= depth - 1;
        end

        // Update depth_commit
        depth_commit <= depth;

        // Update status signals
        if (s_axis_tvalid && s_axis_tready_reg && s_axis_tlast)
            good_frame <= 1;
        if (s_axis_tvalid && s_axis_tready_reg && (s_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE)
            bad_frame <= 1;

        // Handshake signals
        s_axis_tready_reg <= (depth < DEPTH) && !pause_req;
        m_axis_tvalid_reg <= (depth > 0);
        m_axis_tlast_reg <= mem_last[rd_ptr];
        m_axis_tdata_reg <= mem_data[rd_ptr];
        m_axis_tkeep_reg <= mem_keep[rd_ptr];
        m_axis_tid_reg <= mem_id[rd_ptr];
        m_axis_tdest_reg <= mem_dest[rd_ptr];
        m_axis_tuser_reg <= mem_user[rd_ptr];

        // Pause control
        pause_ack_reg <= pause_req && (depth > 0);

        // Clear status signals
        if (s_axis_tready_reg && s_axis_tvalid)
            good_frame <= 0;
        if (s_axis_tready_reg && s_axis_tvalid && (s_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE)
            bad_frame <= 0;
    end
end

assign s_axis_tready = s_axis_tready_reg;
assign m_axis_tvalid = m_axis_tvalid_reg;
assign m_axis_tlast = m_axis_tlast_reg;
assign m_axis_tdata = m_axis_tdata_reg;
assign m_axis_tkeep = m_axis_tkeep_reg;
assign m_axis_tid = m_axis_tid_reg;
assign m_axis_tdest = m_axis_tdest_reg;
assign m_axis_tuser = m_axis_tuser_reg;
assign pause_ack = pause_ack_reg;
assign status_depth = depth;
assign status_depth_commit = depth_commit;
assign status_overflow = overflow;
assign status_bad_frame = bad_frame;
assign status_good_frame = good_frame;

endmodule
```
