=== STAGE 2: VERILOG ===
```verilog
module apbxclk #(
        parameter   C_APB_ADDR_WIDTH = 12,
        parameter   C_APB_DATA_WIDTH = 32,
        parameter [0:0]   OPT_REGISTERED = 1'b0,
        localparam  AW = C_APB_ADDR_WIDTH,
        localparam  DW = C_APB_DATA_WIDTH
    ) (
        input   wire            S_APB_PCLK, S_PRESETn,
        input   wire            S_APB_PSEL,
        input   wire            S_APB_PENABLE,
        output  reg             S_APB_PREADY,
        input   wire    [AW-1:0] S_APB_PADDR,
        input   wire            S_APB_PWRITE,
        input   wire    [DW-1:0] S_APB_PWDATA,
        input   wire    [DW/8-1:0] S_APB_PWSTRB,
        input   wire    [2:0]   S_APB_PPROT,
        output  wire    [DW-1:0] S_APB_PRDATA,
        output  wire            S_APB_PSLVERR,
        input   wire            M_APB_PCLK,
        output  reg             M_PRESETn,
        output  reg             M_APB_PSEL,
        output  reg             M_APB_PENABLE,
        input   wire            M_APB_PREADY,
        output  wire    [AW-1:0] M_APB_PADDR,
        output  wire            M_APB_PWRITE,
        output  wire    [DW-1:0] M_APB_PWDATA,
        output  wire    [DW/8-1:0] M_APB_PWSTRB,
        output  wire    [2:0]   M_APB_PPROT,
        input   wire    [DW-1:0] M_APB_PRDATA,
        input   wire            M_APB_PSLVERR
    );

    // Internal signals
    reg [AW-1:0]  s_addr, m_addr;
    reg           s_write, m_write;
    reg [DW-1:0]  s_wdata, m_wdata;
    reg [DW/8-1:0] s_wstrb, m_wstrb;
    reg [2:0]     s_prot, m_prot;
    reg [DW-1:0]  m_rdata, s_rdata;
    reg           m_slverr, s_slverr;
    reg           s_req_valid, s_req_ready;
    reg           m_req_valid, m_req_ready;
    reg           m_rdata_valid, m_rdata_ready;
    reg           s_rdata_valid, s_rdata_ready;

    // Clock-domain crossing for request from slave to master
    always @(posedge M_APB_PCLK or negedge S_PRESETn) begin
        if (!S_PRESETn) begin
            m_req_valid <= 1'b0;
            m_addr <= {AW{1'b0}};
            m_write <= 1'b0;
            m_wdata <= {DW{1'b0}};
            m_wstrb <= {DW/8{1'b0}};
            m_prot <= 3'b000;
        end else if (s_req_ready) begin
            m_req_valid <= s_req_valid;
            m_addr <= s_addr;
            m_write <= s_write;
            m_wdata <= s_wdata;
            m_wstrb <= s_wstrb;
            m_prot <= s_prot;
        end
    end

    // Clock-domain crossing for response from master to slave
    always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
        if (!S_PRESETn) begin
            s_rdata_valid <= 1'b0;
            s_rdata <= {DW{1'b0}};
            s_slverr <= 1'b0;
        end else if (m_rdata_ready) begin
            s_rdata_valid <= m_rdata_valid;
            s_rdata <= m_rdata;
            s_slverr <= m_slverr;
        end
    end

    // Master side control logic
    always @(posedge M_APB_PCLK or negedge S_PRESETn) begin
        if (!S_PRESETn) begin
            M_APB_PSEL <= 1'b0;
            M_APB_PENABLE <= 1'b0;
            m_req_ready <= 1'b0;
        end else begin
            if (m_req_valid && !m_req_ready) begin
                M_APB_PSEL <= 1'b1;
                M_APB_PENABLE <= 1'b1;
                m_req_ready <= 1'b1;
            end else if (M_APB_PREADY) begin
                M_APB_PSEL <= 1'b0;
                M_APB_PENABLE <= 1'b0;
                m_req_ready <= 1'b0;
            end
        end
    end

    // Slave side control logic
    always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
        if (!S_PRESETn) begin
            S_APB_PREADY <= 1'b0;
            s_req_valid <= 1'b0;
            m_rdata_ready <= 1'b0;
        end else begin
            if (S_APB_PSEL && S_APB_PENABLE && !s_req_valid) begin
                s_req_valid <= 1'b1;
                s_addr <= S_APB_PADDR;
                s_write <= S_APB_PWRITE;
                s_wdata <= S_APB_PWDATA;
                s_wstrb <= S_APB_PWSTRB;
                s_prot <= S_APB_PPROT;
            end else if (s_req_ready) begin
                s_req_valid <= 1'b0;
            end

            if (m_rdata_valid && !m_rdata_ready) begin
                m_rdata_ready <= 1'b1;
                S_APB_PREADY <= 1'b1;
            end else if (S_APB_PREADY) begin
                m_rdata_ready <= 1'b0;
                S_APB_PREADY <= 1'b0;
            end
        end
    end

    // Drive master signals
    assign M_PRESETn = S_PRESETn;
    assign M_APB_PADDR = m_addr;
    assign M_APB_PWRITE = m_write;
    assign M_APB_PWDATA = m_wdata;
    assign M_APB_PWSTRB = m_wstrb;
    assign M_APB_PPROT = m_prot;

    // Drive slave signals
    assign S_APB_PRDATA = s_rdata;
    assign S_APB_PSLVERR = s_slverr;

endmodule
```
