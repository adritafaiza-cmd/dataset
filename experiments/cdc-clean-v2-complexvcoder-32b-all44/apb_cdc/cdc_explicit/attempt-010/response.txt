=== STAGE 2: VERILOG ===
```verilog
module apb_cdc #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_pclk_i,
    input                      src_preset_ni,
    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,
    input  [ADDR_WIDTH-1:0]    src_paddr_i,
    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    // Internal signals for FIFOs
    wire [ADDR_WIDTH + DATA_WIDTH + DATA_WIDTH/8 + 3:0] fifo_wdata;
    wire fifo_wvalid;
    wire fifo_wready;
    wire [ADDR_WIDTH + DATA_WIDTH + DATA_WIDTH/8 + 3:0] fifo_rdata;
    wire fifo_rvalid;
    wire fifo_rready;

    wire [DATA_WIDTH:0] resp_fifo_wdata;
    wire resp_fifo_wvalid;
    wire resp_fifo_wready;
    wire [DATA_WIDTH:0] resp_fifo_rdata;
    wire resp_fifo_rvalid;
    wire resp_fifo_rready;

    // Request FIFO
    async_fifo #(
        .DATA_WIDTH(ADDR_WIDTH + DATA_WIDTH + DATA_WIDTH/8 + 4),
        .LOG_DEPTH(LOG_DEPTH)
    ) req_fifo_inst (
        .write_clk(src_pclk_i),
        .write_reset(src_preset_ni),
        .read_clk(dst_pclk_i),
        .read_reset(dst_preset_ni),
        .write_data(fifo_wdata),
        .write_valid(fifo_wvalid),
        .write_ready(fifo_wready),
        .read_data(fifo_rdata),
        .read_valid(fifo_rvalid),
        .read_ready(fifo_rready)
    );

    // Response FIFO
    async_fifo #(
        .DATA_WIDTH(DATA_WIDTH + 1),
        .LOG_DEPTH(LOG_DEPTH)
    ) resp_fifo_inst (
        .write_clk(dst_pclk_i),
        .write_reset(dst_preset_ni),
        .read_clk(src_pclk_i),
        .read_reset(src_preset_ni),
        .write_data(resp_fifo_wdata),
        .write_valid(resp_fifo_wvalid),
        .write_ready(resp_fifo_wready),
        .read_data(resp_fifo_rdata),
        .read_valid(resp_fifo_rvalid),
        .read_ready(resp_fifo_rready)
    );

    // Assignments for request FIFO
    assign fifo_wdata = {src_pprot_i, src_pwrite_i, src_pstrb_i, src_pwdata_i, src_paddr_i};
    assign fifo_wvalid = src_psel_i & src_penable_i;
    assign src_pready_o = fifo_wready;

    // Assignments for destination interface
    assign dst_paddr_o = fifo_rdata[ADDR_WIDTH-1:0];
    assign dst_pwdata_o = fifo_rdata[ADDR_WIDTH + DATA_WIDTH-1:ADDR_WIDTH];
    assign dst_pstrb_o = fifo_rdata[ADDR_WIDTH + DATA_WIDTH + DATA_WIDTH/8-1:ADDR_WIDTH + DATA_WIDTH];
    assign dst_pprot_o = fifo_rdata[ADDR_WIDTH + DATA_WIDTH + DATA_WIDTH/8 + 2:ADDR_WIDTH + DATA_WIDTH + DATA_WIDTH/8];
    assign dst_pwrite_o = fifo_rdata[ADDR_WIDTH + DATA_WIDTH + DATA_WIDTH/8 + 3];
    assign dst_psel_o = fifo_rvalid;
    assign dst_penable_o = dst_psel_o;

    // Assignments for response FIFO
    assign resp_fifo_wdata = {dst_pslverr_i, dst_prdata_i};
    assign resp_fifo_wvalid = dst_psel_o & dst_pready_i;
    assign src_prdata_o = resp_fifo_rdata[DATA_WIDTH-1:0];
    assign src_pslverr_o = resp_fifo_rdata[DATA_WIDTH];
    assign resp_fifo_rready = src_pready_o;

endmodule

module async_fifo #(
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      write_clk,
    input                      write_reset,
    input                      read_clk,
    input                      read_reset,
    input  [DATA_WIDTH-1:0]    write_data,
    input                      write_valid,
    output                     write_ready,
    output [DATA_WIDTH-1:0]    read_data,
    output                     read_valid,
    input                      read_ready
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    reg [DATA_WIDTH-1:0] mem [DEPTH-1:0];
    reg [ADDR_WIDTH-1:0] write_ptr;
    reg [ADDR_WIDTH-1:0] read_ptr;
    reg [ADDR_WIDTH:0] write_count;
    reg [ADDR_WIDTH:0] read_count;

    reg [ADDR_WIDTH-1:0] write_ptr_gray;
    reg [ADDR_WIDTH-1:0] read_ptr_gray;

    always @(posedge write_clk or posedge write_reset) begin
        if (write_reset) begin
            write_ptr <= 0;
            write_ptr_gray <= 0;
            write_count <= 0;
        end else if (write_valid && write_ready) begin
            mem[write_ptr] <= write_data;
            write_ptr <= write_ptr + 1;
            write_ptr_gray <= (write_ptr_gray & 1) ^ ((write_ptr_gray >> 1) ^ write_ptr);
            write_count <= write_count + 1;
        end
    end

    always @(posedge read_clk or posedge read_reset) begin
        if (read_reset) begin
            read_ptr <= 0;
            read_ptr_gray <= 0;
            read_count <= 0;
        end else if (read_valid && read_ready) begin
            read_ptr <= read_ptr + 1;
            read_ptr_gray <= (read_ptr_gray & 1) ^ ((read_ptr_gray >> 1) ^ read_ptr);
            read_count <= read_count + 1;
        end
    end

    always @(*) begin
        write_ready = (write_count < DEPTH);
        read_valid = (read_count > 0);
    end

    assign read_data = mem[read_ptr];

endmodule
```
