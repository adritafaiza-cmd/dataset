=== STAGE 2: VERILOG ===
```verilog
module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

    // Internal signals for clock-domain crossing
    reg [N-1:0] parallel_data_from_pclk;
    reg         parallel_valid_from_pclk;
    reg         parallel_ready_to_pclk;
    reg [N-1:0] parallel_data_to_pclk;
    reg         parallel_valid_to_pclk;

    // Instantiate sclk_domain_logic
    sclk_domain_logic #(
        .N(N),
        .SPI_2X_CLK_DIV(SPI_2X_CLK_DIV)
    ) sclk_logic (
        .sclk_i(sclk_i),
        .rst_i(rst_i),
        .spi_ssel_o(spi_ssel_o),
        .spi_sck_o(spi_sck_o),
        .spi_mosi_o(spi_mosi_o),
        .spi_miso_i(spi_miso_i),
        .parallel_data_i(parallel_data_from_pclk),
        .parallel_valid_i(parallel_valid_from_pclk),
        .parallel_ready_o(parallel_ready_to_pclk),
        .parallel_data_o(parallel_data_to_pclk),
        .parallel_valid_o(parallel_valid_to_pclk)
    );

    // Instantiate pclk_domain_logic
    pclk_domain_logic #(
        .N(N)
    ) pclk_logic (
        .pclk_i(pclk_i),
        .rst_i(rst_i),
        .di_req_o(di_req_o),
        .di_i(di_i),
        .wren_i(wren_i),
        .wr_ack_o(wr_ack_o),
        .do_valid_o(do_valid_o),
        .do_o(do_o),
        .parallel_data_o(parallel_data_from_pclk),
        .parallel_valid_o(parallel_valid_from_pclk),
        .parallel_ready_i(parallel_ready_to_pclk),
        .parallel_data_i(parallel_data_to_pclk),
        .parallel_valid_i(parallel_valid_to_pclk)
    );

endmodule

// sclk_domain_logic module
module sclk_domain_logic #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    input  [N-1:0]     parallel_data_i,
    input              parallel_valid_i,
    output reg         parallel_ready_o,
    output reg [N-1:0] parallel_data_o,
    output reg         parallel_valid_o
);

    reg [2:0] bit_count;
    reg [N-1:0] shift_register;
    reg [1:0] spi_state;
    reg [1:0] spi_clk_div;

    localparam IDLE = 2'b00;
    localparam TRANSFER = 2'b01;
    localparam COMPLETE = 2'b10;

    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            spi_ssel_o <= 1'b1;
            spi_sck_o <= 1'b0;
            spi_clk_div <= 2'b00;
            bit_count <= 3'b000;
            spi_state <= IDLE;
            shift_register <= {N{1'b0}};
            parallel_ready_o <= 1'b0;
            parallel_data_o <= {N{1'b0}};
            parallel_valid_o <= 1'b0;
        end else begin
            case (spi_state)
                IDLE: begin
                    if (parallel_valid_i) begin
                        spi_ssel_o <= 1'b0;
                        shift_register <= parallel_data_i;
                        bit_count <= 3'b000;
                        spi_state <= TRANSFER;
                        parallel_ready_o <= 1'b1;
                    end else begin
                        parallel_ready_o <= 1'b0;
                    end
                end
                TRANSFER: begin
                    if (spi_clk_div == SPI_2X_CLK_DIV - 1) begin
                        spi_clk_div <= 2'b00;
                        if (bit_count < N) begin
                            spi_sck_o <= ~spi_sck_o;
                            if (spi_sck_o) begin
                                spi_mosi_o <= shift_register[N-1];
                                shift_register <= {shift_register[N-2:0], spi_miso_i};
                                bit_count <= bit_count + 1;
                            end
                        end else begin
                            spi_state <= COMPLETE;
                        end
                    end else begin
                        spi_clk_div <= spi_clk_div + 1;
                    end
                end
                COMPLETE: begin
                    spi_ssel_o <= 1'b1;
                    parallel_data_o <= shift_register;
                    parallel_valid_o <= 1'b1;
                    spi_state <= IDLE;
                end
            endcase
        end
    end

endmodule

// pclk_domain_logic module
module pclk_domain_logic #(
    parameter N = 8
)(
    input              pclk_i,
    input              rst_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o,
    output reg [N-1:0] parallel_data_o,
    output reg         parallel_valid_o,
    input              parallel_ready_i,
    input  [N-1:0]     parallel_data_i,
    input              parallel_valid_i
);

    reg [1:0] parallel_state;

    localparam IDLE = 2'b00;
    localparam ACCEPT = 2'b01;
    localparam VALID = 2'b10;

    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            di_req_o <= 1'b1;
            wr_ack_o <= 1'b0;
            do_valid_o <= 1'b0;
            parallel_data_o <= {N{1'b0}};
            parallel_valid_o <= 1'b0;
            parallel_state <= IDLE;
        end else begin
            case (parallel_state)
                IDLE: begin
                    if (parallel_ready_i) begin
                        di_req_o <= 1'b1;
                        parallel_state <= ACCEPT;
                    end
                end
                ACCEPT: begin
                    if (wren_i && parallel_ready_i) begin
                        parallel_data_o <= di_i;
                        parallel_valid_o <= 1'b1;
                        wr_ack_o <= 1'b1;
                        parallel_state <= VALID;
                    end
                end
                VALID: begin
                    wr_ack_o <= 1'b0;
                    parallel_valid_o <= 1'b0;
                    if (parallel_valid_i) begin
                        do_o <= parallel_data_i;
                        do_valid_o <= 1'b1;
                        parallel_state <= IDLE;
                    end else begin
                        do_valid_o <= 1'b0;
                    end
                end
            endcase
        end
    end

endmodule
```
