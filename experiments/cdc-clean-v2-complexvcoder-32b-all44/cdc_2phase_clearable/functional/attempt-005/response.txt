```verilog
module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Internal signals
  logic sync_src_rst_ni;
  logic sync_dst_rst_ni;
  logic clear_src;
  logic clear_dst;

  // Reset synchronizers
  async_reset_synchronizer #(
    .SYNC_STAGES(SYNC_STAGES)
  ) src_async_reset_sync (
    .async_rst_ni(src_rst_ni),
    .target_clk_i(src_clk_i),
    .sync_rst_no(sync_src_rst_ni)
  );

  async_reset_synchronizer #(
    .SYNC_STAGES(SYNC_STAGES)
  ) dst_async_reset_sync (
    .async_rst_ni(dst_rst_ni),
    .target_clk_i(dst_clk_i),
    .sync_rst_no(sync_dst_rst_ni)
  );

  // Two-phase handshake
  async_handshake async_handshake_inst (
    .src_clk_i(src_clk_i),
    .src_rst_ni(sync_src_rst_ni),
    .src_valid_i(src_valid_i),
    .src_ready_o(src_ready_o),
    .dst_clk_i(dst_clk_i),
    .dst_rst_ni(sync_dst_rst_ni),
    .dst_ready_i(dst_ready_i),
    .dst_valid_o(dst_valid_o),
    .src_clear_i(src_clear_i),
    .dst_clear_i(dst_clear_i),
    .clear_src(clear_src),
    .clear_dst(clear_dst),
    .src_clear_pending_o(src_clear_pending_o),
    .dst_clear_pending_o(dst_clear_pending_o)
  );

  // FIFO buffer
  fifo_buffer #(
    .WIDTH(WIDTH)
  ) fifo_buffer_inst (
    .src_clk_i(src_clk_i),
    .src_rst_ni(sync_src_rst_ni),
    .src_data_i(src_data_i),
    .src_valid_i(src_valid_i),
    .src_ready_o(src_ready_o),
    .dst_clk_i(dst_clk_i),
    .dst_rst_ni(sync_dst_rst_ni),
    .dst_data_o(dst_data_o),
    .dst_valid_o(dst_valid_o),
    .dst_ready_i(dst_ready_i),
    .clear_src(clear_src),
    .clear_dst(clear_dst)
  );

endmodule

module async_reset_synchronizer #(
  parameter SYNC_STAGES = 3
)(
  input  async_rst_ni,
  input  target_clk_i,
  output sync_rst_no
);

  logic [SYNC_STAGES-1:0] sync_stages;

  always_ff @(posedge target_clk_i) begin
    sync_stages <= {sync_stages[SYNC_STAGES-2:0], ~async_rst_ni};
  end

  assign sync_rst_no = sync_stages[SYNC_STAGES-1];

endmodule

module async_handshake (
  input  src_clk_i,
  input  src_rst_ni,
  input  src_valid_i,
  output src_ready_o,
  input  dst_clk_i,
  input  dst_rst_ni,
  input  dst_ready_i,
  output dst_valid_o,
  input  src_clear_i,
  input  dst_clear_i,
  output clear_src,
  output clear_dst,
  output src_clear_pending_o,
  output dst_clear_pending_o
);

  logic src_req;
  logic src_ack;
  logic dst_req;
  logic dst_ack;
  logic src_clear_req;
  logic dst_clear_req;
  logic src_clear_ack;
  logic dst_clear_ack;

  // Source side
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_req <= 0;
      src_clear_req <= 0;
    end else begin
      if (src_clear_i) begin
        src_clear_req <= 1;
      end else if (!src_clear_req) begin
        src_req <= src_valid_i;
      end else if (src_clear_ack) begin
        src_clear_req <= 0;
      end
    end
  end

  // Destination side
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_ack <= 0;
      dst_clear_ack <= 0;
    end else begin
      if (dst_clear_i) begin
        dst_clear_ack <= 1;
      end else if (!dst_clear_ack) begin
        dst_ack <= dst_ready_i;
      end else if (dst_clear_req) begin
        dst_clear_ack <= 0;
      end
    end
  end

  // Cross-clock domain synchronization
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_req <= 0;
      dst_clear_req <= 0;
    end else begin
      dst_req <= src_req;
      dst_clear_req <= src_clear_req;
    end
  end

  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ack <= 0;
      src_clear_ack <= 0;
    end else begin
      src_ack <= dst_ack;
      src_clear_ack <= dst_clear_ack;
    end
  end

  // Outputs
  assign src_ready_o = src_ack & ~src_clear_req;
  assign dst_valid_o = dst_req & ~dst_clear_req;
  assign clear_src = src_clear_req;
  assign clear_dst = dst_clear_req;
  assign src_clear_pending_o = src_clear_req & ~src_clear_ack;
  assign dst_clear_pending_o = dst_clear_req & ~dst_clear_ack;

endmodule

module fifo_buffer #(
  parameter WIDTH = 1
)(
  input  src_clk_i,
  input  src_rst_ni,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  input  src_ready_o,
  input  dst_clk_i,
  input  dst_rst_ni,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i,
  input  clear_src,
  input  clear_dst
);

  logic [WIDTH-1:0] fifo_data;
  logic fifo_full;
  logic fifo_empty;

  // FIFO write side
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      fifo_full <= 0;
      fifo_data <= 0;
    end else if (clear_src) begin
      fifo_full <= 0;
    end else if (src_valid_i && src_ready_o && !fifo_full) begin
      fifo_data <= src_data_i;
      fifo_full <= 1;
    end else if (dst_ready_i && dst_valid_o) begin
      fifo_full <= 0;
    end
  end

  // FIFO read side
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      fifo_empty <= 1;
    end else if (clear_dst) begin
      fifo_empty <= 1;
    end else if (src_valid_i && src_ready_o && !fifo_full) begin
      fifo_empty <= 0;
    end else if (dst_ready_i && dst_valid_o) begin
      fifo_empty <= 1;
    end
  end

  // Outputs
  assign dst_data_o = fifo_data;
  assign dst_valid_o = fifo_full & ~clear_dst;

endmodule
```
