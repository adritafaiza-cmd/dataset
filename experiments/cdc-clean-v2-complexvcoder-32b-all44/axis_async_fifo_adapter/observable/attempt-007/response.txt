=== STAGE 2: VERILOG ===
```verilog
module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    // Internal signals
    reg [S_DATA_WIDTH-1:0] s_data_reg;
    reg [S_KEEP_WIDTH-1:0] s_keep_reg;
    reg                    s_last_reg;
    reg [ID_WIDTH-1:0]     s_id_reg;
    reg [DEST_WIDTH-1:0]   s_dest_reg;
    reg [USER_WIDTH-1:0]   s_user_reg;
    reg                    s_tready_reg;

    reg [M_DATA_WIDTH-1:0] m_data_reg;
    reg [M_KEEP_WIDTH-1:0] m_keep_reg;
    reg                    m_last_reg;
    reg [ID_WIDTH-1:0]     m_id_reg;
    reg [DEST_WIDTH-1:0]   m_dest_reg;
    reg [USER_WIDTH-1:0]   m_user_reg;
    reg                    m_tvalid_reg;

    // FIFO signals
    wire [$clog2(DEPTH):0] fifo_wptr, fifo_rptr;
    wire [$clog2(DEPTH):0] fifo_wptr_gray, fifo_rptr_gray;
    wire [$clog2(DEPTH):0] fifo_depth;
    wire                   fifo_full, fifo_empty;
    wire                   fifo_wren, fifo_rden;

    // FIFO status
    wire [$clog2(DEPTH):0] fifo_depth_sync;
    wire [$clog2(DEPTH):0] fifo_depth_commit_sync;

    // FIFO data width conversion
    wire [M_DATA_WIDTH-1:0] width_adapter_out_data;
    wire [M_KEEP_WIDTH-1:0] width_adapter_out_keep;
    wire                    width_adapter_out_last;
    wire [ID_WIDTH-1:0]     width_adapter_out_id;
    wire [DEST_WIDTH-1:0]   width_adapter_out_dest;
    wire [USER_WIDTH-1:0]   width_adapter_out_user;

    // FIFO module instantiation
    async_fifo #(
        .WIDTH(S_DATA_WIDTH + S_KEEP_WIDTH + 1 + ID_WIDTH + DEST_WIDTH + USER_WIDTH),
        .DEPTH(DEPTH)
    ) s_to_m_fifo (
        .w_clk(s_clk),
        .w_rst(s_rst),
        .w_data({s_axis_tdata, s_axis_tkeep, s_axis_tlast, s_axis_tid, s_axis_tdest, s_axis_tuser}),
        .w_en(fifo_wren),
        .w_full(fifo_full),
        .r_clk(m_clk),
        .r_rst(m_rst),
        .r_data({m_data_reg, m_keep_reg, m_last_reg, m_id_reg, m_dest_reg, m_user_reg}),
        .r_en(fifo_rden),
        .r_empty(fifo_empty),
        .wptr(fifo_wptr),
        .rptr(fifo_rptr),
        .wptr_gray(fifo_wptr_gray),
        .rptr_gray(fifo_rptr_gray),
        .depth(fifo_depth)
    );

    // FIFO depth synchronization
    sync_counter #(
        .WIDTH($clog2(DEPTH)+1)
    ) fifo_depth_sync_inst (
        .clk(m_clk),
        .rst(m_rst),
        .async_in(fifo_depth),
        .sync_out(fifo_depth_sync)
    );

    sync_counter #(
        .WIDTH($clog2(DEPTH)+1)
    ) fifo_depth_commit_sync_inst (
        .clk(m_clk),
        .rst(m_rst),
        .async_in(fifo_depth),
        .sync_out(fifo_depth_commit_sync)
    );

    // FIFO write enable
    assign fifo_wren = s_axis_tvalid & s_tready_reg;

    // FIFO read enable
    assign fifo_rden = m_axis_tready & m_tvalid_reg;

    // AXI-Stream output register
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_axis_tdata <= 0;
            m_axis_tkeep <= 0;
            m_axis_tlast <= 0;
            m_axis_tid <= 0;
            m_axis_tdest <= 0;
            m_axis_tuser <= 0;
            m_axis_tvalid <= 0;
        end else if (fifo_rden) begin
            m_axis_tdata <= m_data_reg;
            m_axis_tkeep <= m_keep_reg;
            m_axis_tlast <= m_last_reg;
            m_axis_tid <= m_id_reg;
            m_axis_tdest <= m_dest_reg;
            m_axis_tuser <= m_user_reg;
            m_axis_tvalid <= 1'b1;
        end else if (!m_axis_tready) begin
            m_axis_tvalid <= 1'b0;
        end
    end

    // AXI-Stream input register
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_data_reg <= 0;
            s_keep_reg <= 0;
            s_last_reg <= 0;
            s_id_reg <= 0;
            s_dest_reg <= 0;
            s_user_reg <= 0;
            s_tready_reg <= 0;
        end else if (s_axis_tvalid & s_tready_reg) begin
            s_data_reg <= s_axis_tdata;
            s_keep_reg <= s_axis_tkeep;
            s_last_reg <= s_axis_tlast;
            s_id_reg <= s_axis_tid;
            s_dest_reg <= s_axis_tdest;
            s_user_reg <= s_axis_tuser;
        end
        s_tready_reg <= !fifo_full;
    end

    // AXI-Stream input ready signal
    assign s_axis_tready = s_tready_reg;

    // Status outputs
    assign s_status_depth = fifo_depth;
    assign s_status_depth_commit = fifo_depth_commit_sync;
    assign s_status_overflow = fifo_full;
    assign s_status_bad_frame = 1'b0; // Placeholder for bad frame detection
    assign s_status_good_frame = 1'b0; // Placeholder for good frame detection

    assign m_status_depth = fifo_depth_sync;
    assign m_status_depth_commit = fifo_depth_commit_sync;
    assign m_status_overflow = fifo_full;
    assign m_status_bad_frame = 1'b0; // Placeholder for bad frame detection
    assign m_status_good_frame = 1'b0; // Placeholder for good frame detection

    // Pause signals (placeholders, no actual pause logic implemented)
    assign s_pause_ack = 1'b0;
    assign m_pause_ack = 1'b0;

endmodule

// Asynchronous FIFO module
module async_fifo #(
    parameter WIDTH = 8,
    parameter DEPTH = 16
)
(
    input  wire                     w_clk,
    input  wire                     w_rst,
    input  wire [WIDTH-1:0]         w_data,
    input  wire                     w_en,
    output wire                     w_full,
    input  wire                     r_clk,
    input  wire                     r_rst,
    output wire [WIDTH-1:0]         r_data,
    input  wire                     r_en,
    output wire                     r_empty,
    output reg [$clog2(DEPTH):0]    wptr,
    output reg [$clog2(DEPTH):0]    rptr,
    output wire [$clog2(DEPTH):0]   wptr_gray,
    output wire [$clog2(DEPTH):0]   rptr_gray,
    output wire [$clog2(DEPTH):0]   depth
);

    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [$clog2(DEPTH):0] wptr_gray_reg, rptr_gray_reg;

    // Write pointer logic
    always @(posedge w_clk or posedge w_rst) begin
        if (w_rst) begin
            wptr <= 0;
            wptr_gray_reg <= 0;
        end else if (w_en & !w_full) begin
            wptr <= wptr + 1;
            wptr_gray_reg <= {wptr_gray_reg[$clog2(DEPTH)-1:0], ~wptr[$clog2(DEPTH)] ^ wptr[$clog2(DEPTH)-1]};
        end
    end

    // Read pointer logic
    always @(posedge r_clk or posedge r_rst) begin
        if (r_rst) begin
            rptr <= 0;
            rptr_gray_reg <= 0;
        end else if (r_en & !r_empty) begin
            rptr <= rptr + 1;
            rptr_gray_reg <= {rptr_gray_reg[$clog2(DEPTH)-1:0], ~rptr[$clog2(DEPTH)] ^ rptr[$clog2(DEPTH)-1]};
        end
    end

    // Memory write
    always @(posedge w_clk) begin
        if (w_en & !w_full) begin
            mem[wptr[$clog2(DEPTH)-1:0]] <= w_data;
        end
    end

    // Memory read
    assign r_data = mem[rptr[$clog2(DEPTH)-1:0]];

    // Gray code conversion for depth calculation
    assign wptr_gray = wptr_gray_reg;
    assign rptr_gray = rptr_gray_reg;

    // Depth calculation
    assign depth = wptr_gray_reg - rptr_gray_reg;

    // Full and empty flags
    assign w_full = (wptr_gray_reg == {rptr_gray_reg[$clog2(DEPTH)-1:0], ~rptr_gray_reg[$clog2(DEPTH)]});
    assign r_empty = (wptr_gray_reg == rptr_gray_reg);

endmodule

// Synchronizer for counter
module sync_counter #(
    parameter WIDTH = 8
)
(
    input  wire                     clk,
    input  wire                     rst,
    input  wire [WIDTH-1:0]         async_in,
    output reg [WIDTH-1:0]          sync_out
);

    reg [WIDTH-1:0] sync_reg1, sync_reg2;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            sync_reg1 <= 0;
            sync_reg2 <= 0;
            sync_out <= 0;
        end else begin
            sync_reg1 <= async_in;
            sync_reg2 <= sync_reg1;
            sync_out <= sync_reg2;
        end
    end

endmodule
```
