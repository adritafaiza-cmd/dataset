// FIFO Core Module
module fifo_core #(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8
)
(
    input  wire                     wr_clk,
    input  wire                     wr_rst,
    input  wire [DATA_WIDTH-1:0]    wr_data,
    input  wire                     wr_en,
    output wire                     full,
    output wire                     almost_full,
    input  wire                     rd_clk,
    input  wire                     rd_rst,
    output wire [DATA_WIDTH-1:0]    rd_data,
    input  wire                     rd_en,
    output wire                     empty,
    output wire                     almost_empty,
    output wire [$clog2(DEPTH)-1:0] wr_ptr,
    output wire [$clog2(DEPTH)-1:0] rd_ptr
);

    // Memory array
    reg [DATA_WIDTH-1:0] mem [DEPTH-1:0];

    // Write pointer
    reg [$clog2(DEPTH)-1:0] wr_ptr_reg;
    reg [$clog2(DEPTH)-1:0] wr_ptr_next;

    // Read pointer
    reg [$clog2(DEPTH)-1:0] rd_ptr_reg;
    reg [$clog2(DEPTH)-1:0] rd_ptr_next;

    // Full and empty flags
    reg full_reg;
    reg empty_reg;

    // Write pointer logic
    always @(posedge wr_clk or posedge wr_rst) begin
        if (wr_rst) begin
            wr_ptr_reg <= 0;
        end else if (wr_en && !full_reg) begin
            wr_ptr_reg <= wr_ptr_next;
        end
    end

    // Read pointer logic
    always @(posedge rd_clk or posedge rd_rst) begin
        if (rd_rst) begin
            rd_ptr_reg <= 0;
        end else if (rd_en && !empty_reg) begin
            rd_ptr_reg <= rd_ptr_next;
        end
    end

    // Next pointer calculation
    assign wr_ptr_next = (wr_ptr_reg == DEPTH-1) ? 0 : wr_ptr_reg + 1;
    assign rd_ptr_next = (rd_ptr_reg == DEPTH-1) ? 0 : rd_ptr_reg + 1;

    // Full and empty flags
    assign full_reg = (wr_ptr_next == rd_ptr_reg);
    assign empty_reg = (wr_ptr_reg == rd_ptr_reg);

    // Almost full and almost empty flags
    assign almost_full = (wr_ptr_next == (rd_ptr_reg - 1)) || (wr_ptr_next == (DEPTH-1) && rd_ptr_reg == 0);
    assign almost_empty = (wr_ptr_reg == (rd_ptr_next - 1)) || (wr_ptr_reg == 0 && rd_ptr_next == DEPTH-1);

    // Assign outputs
    assign wr_ptr = wr_ptr_reg;
    assign rd_ptr = rd_ptr_reg;
    assign full = full_reg;
    assign empty = empty_reg;

    // Memory write and read
    always @(posedge wr_clk) begin
        if (wr_en && !full_reg) begin
            mem[wr_ptr_reg] <= wr_data;
        end
    end

    always @(posedge rd_clk) begin
        if (rd_en && !empty_reg) begin
            rd_data <= mem[rd_ptr_reg];
        end
    end

endmodule
