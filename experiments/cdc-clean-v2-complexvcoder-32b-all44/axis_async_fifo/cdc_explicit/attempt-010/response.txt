module async_fifo #(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8
)
(
    input  wire clk,
    input  wire rst,
    input  wire [DATA_WIDTH-1:0] din,
    input  wire din_valid,
    output wire din_ready,
    output wire [DATA_WIDTH-1:0] dout,
    output wire dout_valid,
    input  wire dout_ready,
    output wire [$clog2(DEPTH):0] depth,
    output wire [$clog2(DEPTH):0] depth_commit,
    output wire overflow
);

    /*
     * Internal signals
     */
    reg [$clog2(DEPTH):0] wr_ptr, rd_ptr;
    reg [$clog2(DEPTH):0] wr_ptr_gray, rd_ptr_gray;
    reg [$clog2(DEPTH):0] wr_ptr_gray_sync, rd_ptr_gray_sync;
    reg [DATA_WIDTH-1:0] mem [DEPTH-1:0];
    reg [$clog2(DEPTH):0] wr_ptr_gray_next, rd_ptr_gray_next;
    reg [$clog2(DEPTH):0] depth_gray, depth_gray_sync;

    /*
     * Binary to Gray code conversion
     */
    assign wr_ptr_gray_next = (wr_ptr >> 1) ^ wr_ptr;
    assign rd_ptr_gray_next = (rd_ptr >> 1) ^ rd_ptr;

    /*
     * FIFO read/write logic
     */
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_ptr <= 0;
            rd_ptr <= 0;
        end else begin
            if (din_valid && din_ready) begin
                mem[wr_ptr[0+:$clog2(DEPTH)]] <= din;
                wr_ptr <= wr_ptr + 1;
            end
            if (dout_valid && dout_ready) begin
                rd_ptr <= rd_ptr + 1;
            end
        end
    end

    /*
     * Gray code synchronization
     */
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_ptr_gray_sync <= 0;
            rd_ptr_gray_sync <= 0;
        end else begin
            wr_ptr_gray_sync <= wr_ptr_gray_next;
            rd_ptr_gray_sync <= rd_ptr_gray_next;
        end
    end

    /*
     * Gray to binary code conversion
     */
    wire [$clog2(DEPTH):0] wr_ptr_sync, rd_ptr_sync;
    assign wr_ptr_sync = (wr_ptr_gray_sync ^ (wr_ptr_gray_sync >> 1));
    assign rd_ptr_sync = (rd_ptr_gray_sync ^ (rd_ptr_gray_sync >> 1));

    /*
     * FIFO status logic
     */
    assign depth_gray = wr_ptr_gray_sync - rd_ptr_gray_sync;
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            depth_gray_sync <= 0;
        end else begin
            depth_gray_sync <= depth_gray;
        end
    end

    assign depth = depth_gray_sync;
    assign depth_commit = depth_gray_sync;
    assign overflow = (wr_ptr_gray_next == {1'b1, {($clog2(DEPTH)-1){1'b0}}}) && (rd_ptr_gray_next == {1'b0, {($clog2(DEPTH)-1){1'b0}}});

    /*
     * FIFO control logic
     */
    assign din_ready = (wr_ptr_gray_next != rd_ptr_gray_sync);
    assign dout_valid = (wr_ptr_gray_sync != rd_ptr_gray_next);
    assign dout = mem[rd_ptr_sync[0+:$clog2(DEPTH)]];

endmodule
