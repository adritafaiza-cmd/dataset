module cdc_fifo_axis_async_fifo #
(
    parameter DATA_WIDTH = 32
)
(
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire                   m_clk,
    input  wire                   m_rst,
    input  wire [DATA_WIDTH-1:0]  s_data,
    input  wire                   s_valid,
    output wire                   s_ready,
    output wire [DATA_WIDTH-1:0]  m_data,
    output wire                   m_valid,
    input  wire                   m_ready
);

    reg [DATA_WIDTH-1:0] s_reg_data;
    reg                  s_reg_valid;
    reg [DATA_WIDTH-1:0] m_reg_data;
    reg                  m_reg_valid;

    // Sync FIFO from s_clk to m_clk
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_reg_data <= 0;
            s_reg_valid <= 0;
        end else if (s_ready && s_valid) begin
            s_reg_data <= s_data;
            s_reg_valid <= 1;
        end else if (m_ready && m_valid) begin
            s_reg_valid <= 0;
        end
    end

    // Sync FIFO from m_clk to s_clk
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_reg_data <= 0;
            m_reg_valid <= 0;
        end else if (m_ready && m_valid) begin
            m_reg_valid <= 0;
        end else if (s_ready && s_valid) begin
            m_reg_data <= s_reg_data;
            m_reg_valid <= s_reg_valid;
        end
    end

    assign s_ready = ~s_reg_valid || (m_ready && m_valid);
    assign m_data = m_reg_data;
    assign m_valid = m_reg_valid;

endmodule

module cdc_fifo_axis_async_fifo_ready
(
    input  wire s_clk,
    input  wire s_rst,
    input  wire m_clk,
    input  wire m_rst,
    input  wire s_data,
    input  wire s_valid,
    output wire s_ready,
    output wire m_data,
    output wire m_valid,
    input  wire m_ready
);

    reg s_reg_data;
    reg s_reg_valid;
    reg m_reg_data;
    reg m_reg_valid;

    // Sync FIFO from s_clk to m_clk
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_reg_data <= 0;
            s_reg_valid <= 0;
        end else if (s_valid) begin
            s_reg_data <= s_data;
            s_reg_valid <= 1;
        end else if (m_ready && m_valid) begin
            s_reg_valid <= 0;
        end
    end

    // Sync FIFO from m_clk to s_clk
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_reg_data <= 0;
            m_reg_valid <= 0;
        end else if (m_ready && m_valid) begin
            m_reg_valid <= 0;
        end else if (s_valid) begin
            m_reg_data <= s_reg_data;
            m_reg_valid <= s_reg_valid;
        end
    end

    assign s_ready = ~s_reg_valid || (m_ready && m_valid);
    assign m_data = m_reg_data;
    assign m_valid = m_reg_valid;

endmodule
