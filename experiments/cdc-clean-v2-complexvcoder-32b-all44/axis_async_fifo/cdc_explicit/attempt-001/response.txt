module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,

    parameter DEST_ENABLE = 0,

    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,

    parameter USER_BAD_FRAME_MASK = 1'b1,

    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,

    parameter FRAME_PAUSE = FRAME_FIFO
)

     * AXI input
     */

    input  wire [USER_WIDTH-1:0]  s_axis_tuser,

     * AXI output
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,

    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,

            end else begin
                s_status_overflow_int <= 0;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_status_depth_int <= 0;
            m_status_depth_commit_int <= 0;
            m_status_overflow_int <= 0;
            m_status_bad_frame_int <= 0;
            m_status_good_frame_int <= 0;
        end else begin
            m_status_depth_int <= m_fifo_depth;
            if (m_frame_commit) begin
                m_status_depth_commit_int <= m_fifo_depth;
            end
            // Additional status logic can be added here
        end
    end

    always @(posedge s_clk or posedge s_rst) begin

            s_pause_ack_int <= 0;

// Asynchronous FIFO module
module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter DEPTH = 1024
)
(
    input wire clk_w,
    input wire rst_w,
    input wire wr_en,
    output wire full,
    input wire [DATA_WIDTH-1:0] din,
    input wire clk_r,
    input wire rst_r,
    input wire rd_en,
    output wire empty,
    output wire [DATA_WIDTH-1:0] dout,
    output wire valid
);

    localparam ADDR_WIDTH = $clog2(DEPTH);

    reg [ADDR_WIDTH-1:0] wr_ptr_w, wr_ptr_r;
    reg [ADDR_WIDTH-1:0] rd_ptr_w, rd_ptr_r;
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];
    reg [ADDR_WIDTH-1:0] wr_ptr_gray_w, wr_ptr_gray_r;
    reg [ADDR_WIDTH-1:0] rd_ptr_gray_w, rd_ptr_gray_r;

    // Write pointer logic
    always @(posedge clk_w or posedge rst_w) begin
        if (rst_w) begin
            wr_ptr_w <= 0;
        end else if (wr_en & ~full) begin
            wr_ptr_w <= wr_ptr_w + 1;
        end
    end

    // Read pointer logic
    always @(posedge clk_r or posedge rst_r) begin
        if (rst_r) begin
            rd_ptr_w <= 0;
        end else if (rd_en & ~empty) begin
            rd_ptr_w <= rd_ptr_w + 1;
        end
    end

    // Memory write
    always @(posedge clk_w) begin
        if (wr_en & ~full) begin
            mem[wr_ptr_w] <= din;
        end
    end

    // Memory read
    always @(posedge clk_r) begin
        if (rd_en & ~empty) begin
            dout <= mem[rd_ptr_r];
        end
    end

    // Gray code conversion
    always @(*) begin
        wr_ptr_gray_w = wr_ptr_w ^ (wr_ptr_w >> 1);
        rd_ptr_gray_w = rd_ptr_w ^ (rd_ptr_w >> 1);
    end

    // Gray code synchronization
    always @(posedge clk_r or posedge rst_r) begin
        if (rst_r) begin
            wr_ptr_gray_r <= 0;
            wr_ptr_r <= 0;
        end else begin
            wr_ptr_gray_r <= wr_ptr_gray_w;
            wr_ptr_r <= (wr_ptr_gray_r ^ (wr_ptr_gray_r >> 1));
        end
    end

    always @(posedge clk_w or posedge rst_w) begin
        if (rst_w) begin
            rd_ptr_gray_r <= 0;
            rd_ptr_r <= 0;
        end else begin
            rd_ptr_gray_r <= rd_ptr_gray_w;
            rd_ptr_r <= (rd_ptr_gray_r ^ (rd_ptr_gray_r >> 1));
        end
    end

    // Full flag
    assign full = (wr_ptr_gray_w == (rd_ptr_gray_r ^ (1 << ADDR_WIDTH)));

    // Empty flag
    assign empty = (wr_ptr_gray_r == rd_ptr_gray_w);

    // Valid flag
    assign valid = ~empty;

endmodule

// CDC counter module
module cdc_counter #(
    parameter DEPTH = 1024
)
(
    input wire clk,
    input wire rst,
    input wire wr_en,
    input wire rd_en,
    output reg [$clog2(DEPTH):0] depth
);

    reg [$clog2(DEPTH):0] wr_cnt;
    reg [$clog2(DEPTH):0] rd_cnt;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_cnt <= 0;
            rd_cnt <= 0;
        end else begin
            if (wr_en) begin
                wr_cnt <= wr_cnt + 1;
            end
            if (rd_en) begin
                rd_cnt <= rd_cnt + 1;
            end
        end
    end

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            depth <= 0;
        end else begin
            depth <= wr_cnt - rd_cnt;
        end
    end

endmodule
