=== STAGE 2: VERILOG ===
```verilog
module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   s_pause_req,
    output wire                   s_pause_ack,
    input  wire                   m_pause_req,
    output wire                   m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] s_status_depth,
    output wire [$clog2(DEPTH):0] s_status_depth_commit,
    output wire                   s_status_overflow,
    output wire                   s_status_bad_frame,
    output wire                   s_status_good_frame,
    output wire [$clog2(DEPTH):0] m_status_depth,
    output wire [$clog2(DEPTH):0] m_status_depth_commit,
    output wire                   m_status_overflow,
    output wire                   m_status_bad_frame,
    output wire                   m_status_good_frame
);

    // FIFO control signals
    wire [$clog2(DEPTH)-1:0] s_wr_ptr;
    wire [$clog2(DEPTH)-1:0] s_wr_ptr_gray;
    wire [$clog2(DEPTH)-1:0] s_rd_ptr_gray;
    wire [$clog2(DEPTH)-1:0] m_wr_ptr_gray;
    wire [$clog2(DEPTH)-1:0] m_rd_ptr;

    // FIFO memory
    reg [DATA_WIDTH-1:0] mem_data [0:DEPTH-1];
    reg [KEEP_WIDTH-1:0] mem_keep [0:DEPTH-1];
    reg [ID_WIDTH-1:0]   mem_id   [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] mem_dest [0:DEPTH-1];
    reg [USER_WIDTH-1:0] mem_user [0:DEPTH-1];
    reg                  mem_last [0:DEPTH-1];

    // FIFO control logic
    reg [$clog2(DEPTH)-1:0] s_wr_ptr_reg;
    reg [$clog2(DEPTH)-1:0] s_rd_ptr_reg;
    reg [$clog2(DEPTH)-1:0] m_wr_ptr_reg;
    reg [$clog2(DEPTH)-1:0] m_rd_ptr_reg;

    // FIFO status
    reg [$clog2(DEPTH):0] s_depth;
    reg [$clog2(DEPTH):0] m_depth;

    // FIFO flags
    reg s_full;
    reg s_empty;
    reg m_full;
    reg m_empty;

    // Gray code conversion
    function [$clog2(DEPTH)-1:0] bin2gray;
        input [$clog2(DEPTH)-1:0] bin;
        bin2gray = bin ^ (bin >> 1);
    endfunction

    // FIFO write side
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr_reg <= 0;
            s_full <= 0;
        end else if (s_axis_tvalid && s_axis_tready) begin
            s_wr_ptr_reg <= s_wr_ptr_reg + 1;
            s_full <= (s_wr_ptr_gray == {~m_rd_ptr_gray[$clog2(DEPTH)-1], m_rd_ptr_gray[$clog2(DEPTH)-2:0]});
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr_reg <= 0;
            m_empty <= 1;
        end else if (m_axis_tvalid && m_axis_tready) begin
            m_rd_ptr_reg <= m_rd_ptr_reg + 1;
            m_empty <= (m_wr_ptr_gray == m_rd_ptr_gray);
        end
    end

    // FIFO read side
    assign s_axis_tready = !s_full && !s_pause_req;
    assign m_axis_tvalid = !m_empty && !m_pause_req;

    // FIFO memory write
    always @(posedge s_clk) begin
        if (s_axis_tvalid && s_axis_tready) begin
            mem_data[s_wr_ptr_reg] <= s_axis_tdata;
            mem_keep[s_wr_ptr_reg] <= s_axis_tkeep;
            mem_last[s_wr_ptr_reg] <= s_axis_tlast;
            if (ID_ENABLE) mem_id[s_wr_ptr_reg] <= s_axis_tid;
            if (DEST_ENABLE) mem_dest[s_wr_ptr_reg] <= s_axis_tdest;
            if (USER_ENABLE) mem_user[s_wr_ptr_reg] <= s_axis_tuser;
        end
    end

    // FIFO memory read
    always @(posedge m_clk) begin
        m_axis_tdata <= mem_data[m_rd_ptr_reg];
        m_axis_tkeep <= mem_keep[m_rd_ptr_reg];
        m_axis_tlast <= mem_last[m_rd_ptr_reg];
        if (ID_ENABLE) m_axis_tid <= mem_id[m_rd_ptr_reg];
        if (DEST_ENABLE) m_axis_tdest <= mem_dest[m_rd_ptr_reg];
        if (USER_ENABLE) m_axis_tuser <= mem_user[m_rd_ptr_reg];
    end

    // Gray code pointers
    assign s_wr_ptr_gray = bin2gray(s_wr_ptr_reg);
    assign m_wr_ptr_gray = bin2gray(m_wr_ptr_reg);
    assign s_rd_ptr_gray = bin2gray(s_rd_ptr_reg);
    assign m_rd_ptr_gray = bin2gray(m_rd_ptr_reg);

    // FIFO depth calculation
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_depth <= 0;
        end else if (s_axis_tvalid && s_axis_tready) begin
            if (m_axis_tvalid && m_axis_tready) begin
                s_depth <= s_depth;
            end else begin
                s_depth <= s_depth + 1;
            end
        end else if (m_axis_tvalid && m_axis_tready) begin
            s_depth <= s_depth - 1;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_depth <= 0;
        end else if (s_axis_tvalid && s_axis_tready) begin
            if (m_axis_tvalid && m_axis_tready) begin
                m_depth <= m_depth;
            end else begin
                m_depth <= m_depth + 1;
            end
        end else if (m_axis_tvalid && m_axis_tready) begin
            m_depth <= m_depth - 1;
        end
    end

    // Status outputs
    assign s_status_depth = s_depth;
    assign s_status_depth_commit = s_depth;
    assign s_status_overflow = s_full;
    assign s_status_bad_frame = 1'b0;
    assign s_status_good_frame = 1'b0;
    assign m_status_depth = m_depth;
    assign m_status_depth_commit = m_depth;
    assign m_status_overflow = m_full;
    assign m_status_bad_frame = 1'b0;
    assign m_status_good_frame = 1'b0;

    // Pause signals
    assign s_pause_ack = s_pause_req && s_full;
    assign m_pause_ack = m_pause_req && m_empty;

endmodule
```
