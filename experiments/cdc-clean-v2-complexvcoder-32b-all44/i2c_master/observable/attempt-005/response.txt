```verilog
module i2c_master (
    input  wire        clk,
    input  wire        rst,
    /*
     * Host interface
     */
    input  wire [6:0]  s_axis_cmd_address,
    input  wire        s_axis_cmd_start,
    input  wire        s_axis_cmd_read,
    input  wire        s_axis_cmd_write,
    input  wire        s_axis_cmd_write_multiple,
    input  wire        s_axis_cmd_stop,
    input  wire        s_axis_cmd_valid,
    output wire        s_axis_cmd_ready,
    input  wire [7:0]  s_axis_data_tdata,
    input  wire        s_axis_data_tvalid,
    output wire        s_axis_data_tready,
    input  wire        s_axis_data_tlast,
    output wire [7:0]  m_axis_data_tdata,
    output wire        m_axis_data_tvalid,
    input  wire        m_axis_data_tready,
    output wire        m_axis_data_tlast,
    /*
     * I2C interface
     */
    input  wire        scl_i,
    output wire        scl_o,
    output wire        scl_t,
    input  wire        sda_i,
    output wire        sda_o,
    output wire        sda_t,
    /*
     * Status
     */
    output wire        busy,
    output wire        bus_control,
    output wire        bus_active,
    output wire        missed_ack,
    /*
     * Configuration
     */
    input  wire [15:0] prescale,
    input  wire        stop_on_idle
);

    // Internal signals
    reg [6:0]          cmd_address;
    reg                cmd_start;
    reg                cmd_read;
    reg                cmd_write;
    reg                cmd_write_multiple;
    reg                cmd_stop;
    reg                cmd_valid;
    reg                cmd_ready;
    reg [7:0]          data_tdata;
    reg                data_tvalid;
    reg                data_tready;
    reg                data_tlast;
    reg [7:0]          read_data_tdata;
    reg                read_data_tvalid;
    reg                read_data_tready;
    reg                read_data_tlast;
    reg [3:0]          current_state;
    reg [3:0]          next_state;
    reg [2:0]          bit_counter;
    reg [7:0]          byte_counter;
    reg [7:0]          data_shift_reg;
    reg                acknowledge;
    reg                bus_control_signal;
    reg                bus_active_signal;
    reg                missed_acknowledge;
    reg                bit_bang_sda_out;
    reg                bit_bang_scl_out;
    reg                bit_bang_sda_dir;
    reg                bit_bang_scl_dir;
    reg                bit_bang_start_bit;
    reg                bit_bang_stop_bit;
    reg                bit_bang_data_out;
    reg                bit_bang_data_in;
    reg                bit_bang_ack;
    reg                bit_bang_nack;
    reg                bit_bang_busy;
    reg [15:0]         prescale_counter;

    // State definitions
    parameter IDLE        = 4'b0000;
    parameter START       = 4'b0001;
    parameter ADDR        = 4'b0010;
    parameter DATA        = 4'b0011;
    parameter ACK         = 4'b0100;
    parameter STOP        = 4'b0101;
    parameter ERROR       = 4'b0110;

    // State machine
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            current_state <= IDLE;
        end else begin
            current_state <= next_state;
        end
    end

    always @(*) begin
        next_state = current_state;
        case (current_state)
            IDLE: begin
                if (cmd_valid) begin
                    next_state = START;
                end
            end
            START: begin
                if (bit_bang_busy == 1'b0) begin
                    next_state = ADDR;
                end
            end
            ADDR: begin
                if (bit_counter == 3'b111 && bit_bang_busy == 1'b0) begin
                    next_state = ACK;
                end
            end
            ACK: begin
                if (bit_bang_busy == 1'b0) begin
                    if (bit_bang_ack == 1'b0) begin
                        next_state = ERROR;
                    end else begin
                        if (cmd_read == 1'b1) begin
                            next_state = DATA;
                        end else if (cmd_write == 1'b1 || cmd_write_multiple == 1'b1) begin
                            next_state = DATA;
                        end else begin
                            next_state = STOP;
                        end
                    end
                end
            end
            DATA: begin
                if (bit_counter == 3'b111 && bit_bang_busy == 1'b0) begin
                    next_state = ACK;
                end
            end
            STOP: begin
                if (bit_bang_busy == 1'b0) begin
                    next_state = IDLE;
                end
            end
            ERROR: begin
                if (cmd_stop == 1'b1 && bit_bang_busy == 1'b0) begin
                    next_state = STOP;
                end
            end
            default: begin
                next_state = IDLE;
            end
        endcase
    end

    // Bit counter
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            bit_counter <= 3'b000;
        end else if (bit_bang_busy == 1'b0) begin
            if (current_state == ADDR || current_state == DATA) begin
                bit_counter <= bit_counter + 1'b1;
            end else begin
                bit_counter <= 3'b000;
            end
        end
    end

    // Byte counter
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            byte_counter <= 8'b00000000;
        end else if (current_state == IDLE) begin
            byte_counter <= 8'b00000000;
        end else if (bit_counter == 3'b111 && bit_bang_busy == 1'b0) begin
            if (current_state == ADDR || current_state == DATA) begin
                byte_counter <= byte_counter + 1'b1;
            end
        end
    end

    // Data shift register
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            data_shift_reg <= 8'b00000000;
        end else if (current_state == ADDR) begin
            if (bit_counter == 3'b000) begin
                data_shift_reg <= {cmd_address, cmd_read};
            end
        end else if (current_state == DATA && bit_counter == 3'b000) begin
            data_shift_reg <= data_tdata;
        end
    end

    // Acknowledge handling
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            acknowledge <= 1'b0;
            missed_acknowledge <= 1'b0;
        end else if (current_state == ACK && bit_bang_busy == 1'b0) begin
            acknowledge <= bit_bang_ack;
            missed_acknowledge <= bit_bang_nack;
        end
    end

    // Command handling
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            cmd_address <= 7'b0000000;
            cmd_start <= 1'b0;
            cmd_read <= 1'b0;
            cmd_write <= 1'b0;
            cmd_write_multiple <= 1'b0;
            cmd_stop <= 1'b0;
            cmd_valid <= 1'b0;
            cmd_ready <= 1'b0;
        end else if (current_state == IDLE && cmd_valid) begin
            cmd_address <= s_axis_cmd_address;
            cmd_start <= s_axis_cmd_start;
            cmd_read <= s_axis_cmd_read;
            cmd_write <= s_axis_cmd_write;
            cmd_write_multiple <= s_axis_cmd_write_multiple;
            cmd_stop <= s_axis_cmd_stop;
            cmd_valid <= 1'b0;
            cmd_ready <= 1'b1;
        end else begin
            cmd_ready <= 1'b0;
        end
    end

    // Data handling
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            data_tdata <= 8'b00000000;
            data_tvalid <= 1'b0;
            data_tready <= 1'b0;
            data_tlast <= 1'b0;
        end else if (current_state == DATA && bit_counter == 3'b000) begin
            data_tready <= 1'b1;
        end else begin
            data_tready <= 1'b0;
        end
    end

    // Read data handling
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            read_data_tdata <= 8'b00000000;
            read_data_tvalid <= 1'b0;
            read_data_tready <= 1'b0;
            read_data_tlast <= 1'b0;
        end else if (current_state == DATA && bit_counter == 3'b111 && bit_bang_busy == 1'b0) begin
            read_data_tdata <= {read_data_tdata[6:0], bit_bang_data_in};
            read_data_tvalid <= 1'b1;
            read_data_tready <= m_axis_data_tready;
            read_data_tlast <= byte_counter == 8'd1 && data_tlast;
        end else begin
            read_data_tready <= 1'b0;
        end
    end

    // Bit-bang control
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            bit_bang_start_bit <= 1'b0;
            bit_bang_stop_bit <= 1'b0;
            bit_bang_data_out <= 1'b0;
            bit_bang_sda_dir <= 1'b0;
            bit_bang_scl_dir <= 1'b0;
        end else begin
            case (current_state)
                START: begin
                    bit_bang_start_bit <= 1'b1;
                    bit_bang_stop_bit <= 1'b0;
                    bit_bang_data_out <= 1'b0;
                    bit_bang_sda_dir <= 1'b1;
                    bit_bang_scl_dir <= 1'b1;
                end
                ADDR: begin
                    bit_bang_start_bit <= 1'b0;
                    bit_bang_stop_bit <= 1'b0;
                    bit_bang_data_out <= data_shift_reg[7 - bit_counter];
                    bit_bang_sda_dir <= 1'b1;
                    bit_bang_scl_dir <= 1'b1;
                end
                DATA: begin
                    bit_bang_start_bit <= 1'b0;
                    bit_bang_stop_bit <= 1'b0;
                    bit_bang_data_out <= data_shift_reg[7 - bit_counter];
                    bit_bang_sda_dir <= 1'b1;
                    bit_bang_scl_dir <= 1'b1;
                end
                ACK: begin
                    bit_bang_start_bit <= 1'b0;
                    bit_bang_stop_bit <= 1'b0;
                    bit_bang_data_out <= 1'b0;
                    bit_bang_sda_dir <= 1'b0;
                    bit_bang_scl_dir <= 1'b1;
                end
                STOP: begin
                    bit_bang_start_bit <= 1'b0;
                    bit_bang_stop_bit <= 1'b1;
                    bit_bang_data_out <= 1'b0;
                    bit_bang_sda_dir <= 1'b1;
                    bit_bang_scl_dir <= 1'b1;
                end
                default: begin
                    bit_bang_start_bit <= 1'b0;
                    bit_bang_stop_bit <= 1'b0;
                    bit_bang_data_out <= 1'b0;
                    bit_bang_sda_dir <= 1'b0;
                    bit_bang_scl_dir <= 1'b0;
                end
            endcase
        end
    end

    // Prescale counter
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            prescale_counter <= 16'd0;
        end else begin
            if (prescale_counter == prescale) begin
                prescale_counter <= 16'd0;
            end else begin
                prescale_counter <= prescale_counter + 1'b1;
            end
        end
    end

    // Output assignments
    assign s_axis_cmd_ready = cmd_ready;
    assign s_axis_data_tready = data_tready;
    assign m_axis_data_tdata = read_data_tdata;
    assign m_axis_data_tvalid = read_data_tvalid;
    assign m_axis_data_tlast = read_data_tlast;
    assign busy = bit_bang_busy;
    assign bus_control = bus_control_signal;
    assign bus_active = bus_active_signal;
    assign missed_ack = missed_acknowledge;

    // Instantiate the i2c_bit_bang module
    i2c_bit_bang bit_bang_inst (
        .clk(clk),
        .rst(rst),
        .scl_i(scl_i),
        .scl_o(bit_bang_scl_out),
        .scl_t(bit_bang_scl_dir),
        .sda_i(sda_i),
        .sda_o(bit_bang_sda_out),
        .sda_t(bit_bang_sda_dir),
        .start_bit(bit_bang_start_bit),
        .stop_bit(bit_bang_stop_bit),
        .data_out(bit_bang_data_out),
        .data_in(bit_bang_data_in),
        .ack(bit_bang_ack),
        .nack(bit_bang_nack),
        .busy(bit_bang_busy)
    );

endmodule

    input  wire        rst,
    input  wire        scl_i,
    output wire        scl_o,
    output wire        scl_t,
    input  wire        sda_i,
    output wire        sda_o,
    output wire        sda_t,
    input  wire        start_bit,
    input  wire        stop_bit,
    input  wire        data_out,
    output wire        data_in,

    output wire        nack,
    output wire        busy

    // Internal signals
    reg                scl_reg;
    reg                scl_dir;
    reg                sda_reg;
    reg                sda_dir;
    reg [2:0]          bit_counter;
    reg [15:0]         prescale_counter;
    reg                busy_reg;
    reg                start_reg;
    reg                stop_reg;
    reg                data_out_reg;
    reg                data_in_reg;
    reg                ack_reg;
    reg                nack_reg;

    // Prescale counter
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            prescale_counter <= 16'd0;
        end else begin
            if (prescale_counter == 16'd15) begin
                prescale_counter <= 16'd0;
            end else begin
                prescale_counter <= prescale_counter + 1'b1;
            end
        end
    end

    // Bit counter
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            bit_counter <= 3'b000;
        end else if (busy_reg == 1'b1) begin
            if (prescale_counter == 16'd15) begin
                if (bit_counter == 3'b111) begin
                    bit_counter <= 3'b000;
                end else begin
                    bit_counter <= bit_counter + 1'b1;
                end
            end
        end
    end

    // Busy flag
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            busy_reg <= 1'b0;
        end else begin
            if (start_bit == 1'b1) begin
                busy_reg <= 1'b1;
            end else if (stop_bit == 1'b1 && bit_counter == 3'b111) begin
                busy_reg <= 1'b0;
            end
        end
    end

    // Start condition
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            start_reg <= 1'b0;
        end else begin
            if (start_bit == 1'b1) begin
                start_reg <= 1'b1;
            end else if (bit_counter == 3'b000) begin
                start_reg <= 1'b0;
            end
        end
    end

    // Stop condition
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            stop_reg <= 1'b0;
        end else begin
            if (stop_bit == 1'b1 && bit_counter == 3'b111) begin
                stop_reg <= 1'b1;
            end else if (prescale_counter == 16'd15) begin
                stop_reg <= 1'b0;
            end
        end
    end

    // Data out
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            data_out_reg <= 1'b0;
        end else begin
            if (bit_counter == 3'b000) begin
                data_out_reg <= data_out;
            end
        end
    end

    // Data in
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            data_in_reg <= 1'b0;
        end else begin
            if (bit_counter == 3'b111 && prescale_counter == 16'd15) begin
                data_in_reg <= sda_i;
            end
        end
    end

    // Acknowledge
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            ack_reg <= 1'b0;
            nack_reg <= 1'b0;
        end else begin
            if (bit_counter == 3'b111 && prescale_counter == 16'd15) begin
                if (sda_i == 1'b0) begin
                    ack_reg <= 1'b1;
                    nack_reg <= 1'b0;
                end else begin
                    ack_reg <= 1'b0;
                    nack_reg <= 1'b1;
                end
            end else begin
                ack_reg <= 1'b0;
                nack_reg <= 1'b0;
            end
        end
    end

    // SDA and SCL control
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            scl_reg <= 1'b1;
            scl_dir <= 1'b1;
            sda_reg <= 1'b1;
            sda_dir <= 1'b1;
        end else begin
            if (busy_reg == 1'b1) begin
                if (start_reg == 1'b1 && bit_counter == 3'b000) begin
                    scl_reg <= 1'b1;
                    scl_dir <= 1'b1;
                    sda_reg <= 1'b0;
                    sda_dir <= 1'b1;
                end else if (bit_counter == 3'b001) begin
                    scl_reg <= 1'b0;
                    scl_dir <= 1'b1;
                    sda_reg <= 1'b0;
                    sda_dir <= 1'b1;
                end else if (bit_counter >= 3'b010 && bit_counter <= 3'b110) begin
                    scl_reg <= 1'b0;
                    scl_dir <= 1'b1;
                    sda_reg <= data_out_reg;
                    sda_dir <= 1'b1;
                end else if (bit_counter == 3'b111) begin
                    scl_reg <= 1'b1;
                    scl_dir <= 1'b1;
                    sda_reg <= 1'bz;
                    sda_dir <= 1'b0;
                end
            end else if (stop_reg == 1'b1) begin
                scl_reg <= 1'b0;
                scl_dir <= 1'b1;
                sda_reg <= 1'b0;
                sda_dir <= 1'b1;
            end else begin
                scl_reg <= 1'b1;
                scl_dir <= 1'b1;
                sda_reg <= 1'b1;
                sda_dir <= 1'b1;
            end
        end
    end

    // Output assignments
    assign scl_o = scl_reg;
    assign scl_t = scl_dir;
    assign sda_o = sda_reg;
    assign sda_t = sda_dir;
    assign data_in = data_in_reg;
    assign ack = ack_reg;
    assign nack = nack_reg;
    assign busy = busy_reg;

endmodule
```
