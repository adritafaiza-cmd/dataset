=== STAGE 2: VERILOG ===
```verilog
module i2c_master (
    input  wire        clk,
    input  wire        rst,
    /*
     * Host interface
     */
    input  wire [6:0]  s_axis_cmd_address,
    input  wire        s_axis_cmd_start,
    input  wire        s_axis_cmd_read,
    input  wire        s_axis_cmd_write,
    input  wire        s_axis_cmd_write_multiple,
    input  wire        s_axis_cmd_stop,
    input  wire        s_axis_cmd_valid,
    output wire        s_axis_cmd_ready,
    input  wire [7:0]  s_axis_data_tdata,
    input  wire        s_axis_data_tvalid,
    output wire        s_axis_data_tready,
    input  wire        s_axis_data_tlast,
    output wire [7:0]  m_axis_data_tdata,
    output wire        m_axis_data_tvalid,
    input  wire        m_axis_data_tready,
    output wire        m_axis_data_tlast,
    /*
     * I2C interface
     */
    input  wire        scl_i,
    output wire        scl_o,
    output wire        scl_t,
    input  wire        sda_i,
    output wire        sda_o,
    output wire        sda_t,
    /*
     * Status
     */
    output wire        busy,
    output wire        bus_control,
    output wire        bus_active,
    output wire        missed_ack,
    /*
     * Configuration
     */
    input  wire [15:0] prescale,
    input  wire        stop_on_idle
);

    // State definitions
    localparam IDLE      = 4'b0000;
    localparam START     = 4'b0001;
    localparam ADDR      = 4'b0010;
    localparam ACK       = 4'b0011;
    localparam DATA      = 4'b0100;
    localparam STOP      = 4'b0101;

    // Internal signals
    reg [3:0]          state, next_state;
    reg [2:0]          bit_count;
    reg [7:0]          byte_count;
    reg [7:0]          shift_reg;
    reg [7:0]          data_reg;
    reg                start_condition;
    reg                stop_condition;
    reg                read_condition;
    reg                write_condition;
    reg                write_multiple_condition;
    reg                ack_received;
    reg                scl_reg;
    reg                sda_reg;
    reg                scl_t_reg;
    reg                sda_t_reg;
    reg [15:0]         timer;
    reg                cmd_ready;
    reg                data_ready;
    reg                data_valid;
    reg                data_last;
    reg                bus_control_reg;
    reg                bus_active_reg;
    reg                busy_reg;
    reg                missed_ack_reg;

    // State machine
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= IDLE;
        end else begin
            state <= next_state;
        end
    end

    always @(*) begin
        next_state = state;
        case (state)
            IDLE: begin
                if (start_condition) begin
                    next_state = START;
                end
            end
            START: begin
                if (timer == 0) begin
                    next_state = ADDR;
                end
            end
            ADDR: begin
                if (timer == 0) begin
                    if (bit_count < 7) begin
                        next_state = ADDR;
                    end else begin
                        next_state = ACK;
                    end
                end
            end
            ACK: begin
                if (timer == 0) begin
                    if (ack_received) begin
                        if (write_condition) begin
                            next_state = DATA;
                        end else if (read_condition) begin
                            next_state = DATA;
                        end else if (stop_condition) begin
                            next_state = STOP;
                        end
                    end else begin
                        next_state = IDLE;
                    end
                end
            end
            DATA: begin
                if (timer == 0) begin
                    if (bit_count < 7) begin
                        next_state = DATA;
                    end else begin
                        if (ack_received) begin
                            if (read_condition && !data_last) begin
                                next_state = DATA;
                            end else if (write_condition || write_multiple_condition) begin
                                next_state = DATA;
                            end else if (stop_condition) begin
                                next_state = STOP;
                            end
                        end else begin
                            next_state = IDLE;
                        end
                    end
                end
            end
            STOP: begin
                if (timer == 0) begin
                    next_state = IDLE;
                end
            end
            default: begin
                next_state = IDLE;
            end
        endcase
    end

    // Timer for SCL generation
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            timer <= prescale;
        end else if (timer == 0) begin
            timer <= prescale;
        end else if (state != IDLE) begin
            timer <= timer - 1;
        end
    end

    // Bit and byte counters
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            bit_count <= 0;
            byte_count <= 0;
        end else begin
            case (state)
                START, ADDR, ACK, DATA: begin
                    if (timer == 0) begin
                        bit_count <= bit_count + 1;
                    end
                end
                default: begin
                    bit_count <= 0;
                end
            endcase
            case (state)
                DATA: begin
                    if (timer == 0 && bit_count == 7) begin
                        byte_count <= byte_count + 1;
                    end
                end
                default: begin
                    byte_count <= 0;
                end
            endcase
        end
    end

    // Shift register for data transfer
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            shift_reg <= 0;
        end else begin
            case (state)
                ADDR: begin
                    if (timer == 0 && bit_count < 7) begin
                        shift_reg <= {shift_reg[6:0], s_axis_cmd_address[6-bit_count]};
                    end else if (timer == 0 && bit_count == 7) begin
                        shift_reg <= {shift_reg[6:0], read_condition};
                    end
                end
                DATA: begin
                    if (timer == 0 && bit_count < 7 && write_condition) begin
                        shift_reg <= {shift_reg[6:0], s_axis_data_tdata[7-bit_count]};
                    end else if (timer == 0 && bit_count < 7 && read_condition) begin
                        shift_reg <= {shift_reg[6:0], sda_i};
                    end
                end
                default: begin
                    shift_reg <= shift_reg;
                end
            endcase
        end
    end

    // Command and data interface handshakes
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            cmd_ready <= 0;
            data_ready <= 0;
            data_valid <= 0;
            data_last <= 0;
        end else begin
            case (state)
                IDLE: begin
                    if (s_axis_cmd_valid && (s_axis_cmd_start || s_axis_cmd_stop || s_axis_cmd_read || s_axis_cmd_write || s_axis_cmd_write_multiple)) begin
                        cmd_ready <= 1;
                        data_valid <= 0;
                        data_last <= 0;
                    end else begin
                        cmd_ready <= 0;
                    end
                end
                DATA: begin
                    if (timer == 0 && bit_count == 7 && write_condition && s_axis_data_tvalid) begin
                        data_ready <= 1;
                    end else if (timer == 0 && bit_count == 7 && read_condition && m_axis_data_tready) begin
                        data_valid <= 1;
                    end else begin
                        data_ready <= 0;
                        data_valid <= 0;
                    end
                    if (timer == 0 && bit_count == 7 && s_axis_data_tlast) begin
                        data_last <= 1;
                    end else begin
                        data_last <= 0;
                    end
                end
                default: begin
                    cmd_ready <= 0;
                    data_ready <= 0;
                    data_valid <= 0;
                    data_last <= 0;
                end
            endcase
        end
    end

    // Control signals
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            start_condition <= 0;
            stop_condition <= 0;
            read_condition <= 0;
            write_condition <= 0;
            write_multiple_condition <= 0;
            ack_received <= 0;
            scl_reg <= 1;
            sda_reg <= 1;
            scl_t_reg <= 1;
            sda_t_reg <= 1;
            bus_control_reg <= 0;
            bus_active_reg <= 0;
            busy_reg <= 0;
            missed_ack_reg <= 0;
        end else begin
            case (state)
                IDLE: begin
                    if (s_axis_cmd_valid && s_axis_cmd_start) begin
                        start_condition <= 1;
                    end else begin
                        start_condition <= 0;
                    end
                    stop_condition <= 0;
                    read_condition <= 0;
                    write_condition <= 0;
                    write_multiple_condition <= 0;
                    ack_received <= 0;
                    scl_reg <= 1;
                    sda_reg <= 1;
                    scl_t_reg <= 1;
                    sda_t_reg <= 1;
                    bus_control_reg <= 0;
                    bus_active_reg <= 0;
                    busy_reg <= 0;
                    missed_ack_reg <= 0;
                end
                START: begin
                    start_condition <= 0;
                    stop_condition <= 0;
                    read_condition <= 0;
                    write_condition <= 0;
                    write_multiple_condition <= 0;
                    ack_received <= 0;
                    scl_reg <= 0;
                    sda_reg <= 0;
                    scl_t_reg <= 0;
                    sda_t_reg <= 0;
                    bus_control_reg <= 1;
                    bus_active_reg <= 1;
                    busy_reg <= 1;
                    missed_ack_reg <= 0;
                end
                ADDR: begin
                    if (timer == 0 && bit_count == 7) begin
                        sda_reg <= 1;
                        scl_reg <= 1;
                        scl_t_reg <= 0;
                        sda_t_reg <= 0;
                    end else begin
                        sda_reg <= shift_reg[7];
                        scl_reg <= 1;
                        scl_t_reg <= 0;
                        sda_t_reg <= 0;
                    end
                    read_condition <= s_axis_cmd_read;
                    write_condition <= s_axis_cmd_write;
                    write_multiple_condition <= s_axis_cmd_write_multiple;
                    stop_condition <= s_axis_cmd_stop;
                    ack_received <= 0;
                    bus_control_reg <= 1;
                    bus_active_reg <= 1;
                    busy_reg <= 1;
                    missed_ack_reg <= 0;
                end
                ACK: begin
                    if (timer == 0) begin
                        scl_reg <= 1;
                        scl_t_reg <= 0;
                        sda_reg <= 1;
                        sda_t_reg <= 1;
                        if (sda_i) begin
                            ack_received <= 0;
                            missed_ack_reg <= 1;
                        end else begin
                            ack_received <= 1;
                            missed_ack_reg <= 0;
                        end
                    end
                    read_condition <= s_axis_cmd_read;
                    write_condition <= s_axis_cmd_write;
                    write_multiple_condition <= s_axis_cmd_write_multiple;
                    stop_condition <= s_axis_cmd_stop;
                    bus_control_reg <= 1;
                    bus_active_reg <= 1;
                    busy_reg <= 1;
                end
                DATA: begin
                    if (timer == 0) begin
                        if (bit_count == 7) begin
                            scl_reg <= 1;
                            scl_t_reg <= 0;
                            sda_reg <= 1;
                            sda_t_reg <= 1;
                            if (sda_i) begin
                                ack_received <= 0;
                                missed_ack_reg <= 1;
                            end else begin
                                ack_received <= 1;
                                missed_ack_reg <= 0;
                            end
                        end else begin
                            scl_reg <= 1;
                            scl_t_reg <= 0;
                            sda_reg <= shift_reg[7];
                            sda_t_reg <= 0;
                        end
                    end
                    read_condition <= s_axis_cmd_read;
                    write_condition <= s_axis_cmd_write;
                    write_multiple_condition <= s_axis_cmd_write_multiple;
                    stop_condition <= s_axis_cmd_stop;
                    bus_control_reg <= 1;
                    bus_active_reg <= 1;
                    busy_reg <= 1;
                end
                STOP: begin
                    if (timer == 0) begin
                        scl_reg <= 1;
                        scl_t_reg <= 0;
                        sda_reg <= 0;
                        sda_t_reg <= 0;
                    end else if (timer == prescale) begin
                        scl_reg <= 1;
                        scl_t_reg <= 0;
                        sda_reg <= 1;
                        sda_t_reg <= 0;
                    end else if (timer == 2*prescale) begin
                        scl_reg <= 1;
                        scl_t_reg <= 0;
                        sda_reg <= 1;
                        sda_t_reg <= 1;
                    end
                    start_condition <= 0;
                    stop_condition <= 0;
                    read_condition <= 0;
                    write_condition <= 0;
                    write_multiple_condition <= 0;
                    ack_received <= 0;
                    bus_control_reg <= 0;
                    bus_active_reg <= 0;
                    busy_reg <= 0;
                    missed_ack_reg <= 0;
                end
                default: begin
                    start_condition <= 0;
                    stop_condition <= 0;
                    read_condition <= 0;
                    write_condition <= 0;
                    write_multiple_condition <= 0;
                    ack_received <= 0;
                    scl_reg <= 1;
                    sda_reg <= 1;
                    scl_t_reg <= 1;
                    sda_t_reg <= 1;
                    bus_control_reg <= 0;
                    bus_active_reg <= 0;
                    busy_reg <= 0;
                    missed_ack_reg <= 0;
                end
            endcase
        end
    end

    // Output assignments
    assign s_axis_cmd_ready = cmd_ready;
    assign s_axis_data_tready = data_ready;
    assign m_axis_data_tdata = shift_reg;
    assign m_axis_data_tvalid = data_valid;
    assign m_axis_data_tlast = data_last;
    assign scl_o = scl_reg;
    assign scl_t = scl_t_reg;
    assign sda_o = sda_reg;
    assign sda_t = sda_t_reg;
    assign busy = busy_reg;
    assign bus_control = bus_control_reg;
    assign bus_active = bus_active_reg;
    assign missed_ack = missed_ack_reg;

endmodule
```
