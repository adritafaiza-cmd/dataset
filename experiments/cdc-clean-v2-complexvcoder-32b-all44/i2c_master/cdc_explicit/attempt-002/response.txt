=== STAGE 2: VERILOG ===
```verilog
module i2c_master (
    input  wire        clk,
    input  wire        rst,
    /*
     * Host interface
     */
    input  wire [6:0]  s_axis_cmd_address,
    input  wire        s_axis_cmd_start,
    input  wire        s_axis_cmd_read,
    input  wire        s_axis_cmd_write,
    input  wire        s_axis_cmd_write_multiple,
    input  wire        s_axis_cmd_stop,
    input  wire        s_axis_cmd_valid,
    output wire        s_axis_cmd_ready,
    input  wire [7:0]  s_axis_data_tdata,
    input  wire        s_axis_data_tvalid,
    output wire        s_axis_data_tready,
    input  wire        s_axis_data_tlast,
    output wire [7:0]  m_axis_data_tdata,
    output wire        m_axis_data_tvalid,
    input  wire        m_axis_data_tready,
    output wire        m_axis_data_tlast,
    /*
     * I2C interface
     */
    input  wire        scl_i,
    output wire        scl_o,
    output wire        scl_t,
    input  wire        sda_i,
    output wire        sda_o,
    output wire        sda_t,
    /*
     * Status
     */
    output wire        busy,
    output wire        bus_control,
    output wire        bus_active,
    output wire        missed_ack,
    /*
     * Configuration
     */
    input  wire [15:0] prescale,
    input  wire        stop_on_idle
);

    // Internal signals
    reg                scl_reg;
    reg                sda_reg;
    reg                scl_t_reg;
    reg                sda_t_reg;
    reg                scl_o_reg;
    reg                sda_o_reg;
    reg [1:0]          state;
    reg [1:0]          next_state;
    reg [3:0]          bit_cnt;
    reg [7:0]          data_reg;
    reg [7:0]          shift_reg;
    reg [15:0]         clk_cnt;
    reg                cmd_start;
    reg                cmd_read;
    reg                cmd_write;
    reg                cmd_write_multiple;
    reg                cmd_stop;
    reg                cmd_valid;
    reg                data_valid;
    reg                data_tlast;
    reg [7:0]          data_tdata;
    reg                missed_ack_reg;
    reg                busy_reg;
    reg                bus_control_reg;
    reg                bus_active_reg;

    // State definitions
    localparam IDLE       = 2'b00;
    localparam START      = 2'b01;
    localparam ADDR       = 2'b10;
    localparam ACK_ADDR   = 2'b11;
    localparam DATA_WRITE = 2'b00;
    localparam ACK_WRITE  = 2'b01;
    localparam DATA_READ  = 2'b10;
    localparam STOP       = 2'b11;

    // Clock divider
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            clk_cnt <= 16'd0;
        end else begin
            if (clk_cnt == prescale - 1'd1) begin
                clk_cnt <= 16'd0;
            end else begin
                clk_cnt <= clk_cnt + 1'd1;
            end
        end
    end

    // State machine
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= IDLE;
        end else begin
            state <= next_state;
        end
    end

    always @(*) begin
        next_state = state;
        case (state)
            IDLE: begin
                if (cmd_valid && cmd_start) begin
                    next_state = START;
                end
            end
            START: begin
                next_state = ADDR;
            end
            ADDR: begin
                if (bit_cnt == 3'd7) begin
                    next_state = ACK_ADDR;
                end
            end
            ACK_ADDR: begin
                next_state = (cmd_read) ? DATA_READ : DATA_WRITE;
            end
            DATA_WRITE: begin
                if (bit_cnt == 3'd7) begin
                    next_state = ACK_WRITE;
                end
            end
            ACK_WRITE: begin
                if (cmd_write_multiple && data_valid && !data_tlast) begin
                    next_state = DATA_WRITE;
                end else begin
                    next_state = STOP;
                end
            end
            DATA_READ: begin
                if (bit_cnt == 3'd7) begin
                    next_state = STOP;
                end
            end
            STOP: begin
                next_state = IDLE;
            end
            default: begin
                next_state = IDLE;
            end
        endcase
    end

    // Control logic
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            bit_cnt <= 4'd0;
            data_reg <= 8'd0;
            shift_reg <= 8'd0;
            cmd_start <= 1'b0;
            cmd_read <= 1'b0;
            cmd_write <= 1'b0;
            cmd_write_multiple <= 1'b0;
            cmd_stop <= 1'b0;
            cmd_valid <= 1'b0;
            data_valid <= 1'b0;
            data_tlast <= 1'b0;
            data_tdata <= 8'd0;
            missed_ack_reg <= 1'b0;
            busy_reg <= 1'b0;
            bus_control_reg <= 1'b0;
            bus_active_reg <= 1'b0;
        end else begin
            case (state)
                IDLE: begin
                    cmd_start <= s_axis_cmd_start;
                    cmd_read <= s_axis_cmd_read;
                    cmd_write <= s_axis_cmd_write;
                    cmd_write_multiple <= s_axis_cmd_write_multiple;
                    cmd_stop <= s_axis_cmd_stop;
                    cmd_valid <= s_axis_cmd_valid;
                    data_valid <= s_axis_data_tvalid;
                    data_tlast <= s_axis_data_tlast;
                    data_tdata <= s_axis_data_tdata;
                    if (cmd_valid && cmd_start) begin
                        bit_cnt <= 4'd0;
                        shift_reg <= {s_axis_cmd_address, 1'b0}; // Write
                        if (cmd_read) begin
                            shift_reg[0] <= 1'b1; // Read
                        end
                        busy_reg <= 1'b1;
                        bus_control_reg <= 1'b1;
                        bus_active_reg <= 1'b1;
                    end
                end
                START: begin
                    scl_reg <= 1'b1;
                    sda_reg <= 1'b1;
                    if (clk_cnt == prescale - 1'd1) begin
                        scl_reg <= 1'b0;
                        sda_reg <= 1'b0;
                    end
                end
                ADDR: begin
                    scl_reg <= (clk_cnt < prescale) ? 1'b0 : 1'b1;
                    sda_reg <= (clk_cnt < prescale) ? shift_reg[7-bit_cnt] : 1'bz;
                    if (clk_cnt == prescale - 1'd1) begin
                        if (bit_cnt < 3'd7) begin
                            bit_cnt <= bit_cnt + 1'd1;
                        end
                    end
                end
                ACK_ADDR: begin
                    scl_reg <= (clk_cnt < prescale) ? 1'b0 : 1'b1;
                    sda_reg <= 1'bz;
                    if (clk_cnt == prescale - 1'd1) begin
                        if (sda_i) begin
                            missed_ack_reg <= 1'b1;
                        end
                    end
                end
                DATA_WRITE: begin
                    scl_reg <= (clk_cnt < prescale) ? 1'b0 : 1'b1;
                    sda_reg <= (clk_cnt < prescale) ? shift_reg[7-bit_cnt] : 1'bz;
                    if (clk_cnt == prescale - 1'd1) begin
                        if (bit_cnt < 3'd7) begin
                            bit_cnt <= bit_cnt + 1'd1;
                        end
                    end
                end
                ACK_WRITE: begin
                    scl_reg <= (clk_cnt < prescale) ? 1'b0 : 1'b1;
                    sda_reg <= 1'bz;
                    if (clk_cnt == prescale - 1'd1) begin
                        if (sda_i) begin
                            missed_ack_reg <= 1'b1;
                        end else begin
                            if (cmd_write_multiple && data_valid && !data_tlast) begin
                                shift_reg <= data_tdata;
                                bit_cnt <= 4'd0;
                            end
                        end
                    end
                end
                DATA_READ: begin
                    scl_reg <= (clk_cnt < prescale) ? 1'b0 : 1'b1;
                    sda_reg <= 1'bz;
                    if (clk_cnt == prescale - 1'd1) begin
                        if (bit_cnt < 3'd7) begin
                            shift_reg <= {shift_reg[6:0], sda_i};
                            bit_cnt <= bit_cnt + 1'd1;
                        end else begin
                            shift_reg <= {shift_reg[6:0], sda_i};
                            data_reg <= shift_reg;
                            m_axis_data_tdata <= data_reg;
                            m_axis_data_tvalid <= 1'b1;
                            m_axis_data_tlast <= data_tlast;
                        end
                    end
                end
                STOP: begin
                    scl_reg <= (clk_cnt < prescale) ? 1'b0 : 1'b1;
                    sda_reg <= (clk_cnt < prescale) ? 1'b0 : 1'b1;
                    if (clk_cnt == prescale - 1'd1) begin
                        busy_reg <= 1'b0;
                        bus_control_reg <= 1'b0;
                        bus_active_reg <= 1'b0;
                        s_axis_cmd_ready <= 1'b1;
                        s_axis_data_tready <= 1'b1;
                        m_axis_data_tvalid <= 1'b0;
                    end
                end
                default: begin
                    busy_reg <= 1'b0;
                    bus_control_reg <= 1'b0;
                    bus_active_reg <= 1'b0;
                end
            endcase
        end
    end

    // Output assignments
    assign s_axis_cmd_ready = (state == IDLE) && !cmd_valid;
    assign s_axis_data_tready = (state == DATA_WRITE) && data_valid && (bit_cnt == 3'd7);
    assign m_axis_data_tdata = data_reg;
    assign m_axis_data_tvalid = (state == DATA_READ) && (bit_cnt == 3'd7);
    assign m_axis_data_tlast = data_tlast;
    assign scl_o = scl_reg;
    assign scl_t = scl_t_reg;
    assign sda_o = sda_reg;
    assign sda_t = sda_t_reg;
    assign busy = busy_reg;
    assign bus_control = bus_control_reg;
    assign bus_active = bus_active_reg;
    assign missed_ack = missed_ack_reg;

endmodule
```
