module i2c_master_command_handler (
    input  wire        clk,
    input  wire        rst,
    input  wire [6:0]  s_axis_cmd_address,
    input  wire        s_axis_cmd_start,
    input  wire        s_axis_cmd_read,
    input  wire        s_axis_cmd_write,
    input  wire        s_axis_cmd_write_multiple,
    input  wire        s_axis_cmd_stop,
    input  wire        s_axis_cmd_valid,
    output wire        s_axis_cmd_ready,
    input  wire [7:0]  s_axis_data_tdata,
    input  wire        s_axis_data_tvalid,
    output wire        s_axis_data_tready,
    input  wire        s_axis_data_tlast,
    output wire [7:0]  m_axis_data_tdata,
    output wire        m_axis_data_tvalid,
    input  wire        m_axis_data_tready,
    output wire        m_axis_data_tlast,
    input  wire [15:0] prescale,
    input  wire        stop_on_idle,
    output wire        i2c_scl_o,
    output wire        i2c_scl_t,
    output wire        i2c_sda_o,
    output wire        i2c_sda_t,
    output wire        i2c_busy,
    output wire        i2c_bus_control,
    output wire        i2c_bus_active,
    output wire        i2c_missed_ack,
    input  wire        i2c_scl_i,
    input  wire        i2c_sda_i
);

    // Internal state signals
    reg [15:0] scl_cnt;
    reg        scl;
    reg        sda;
    reg        scl_dir;
    reg        sda_dir;
    reg [3:0]  state;
    reg [3:0]  next_state;
    reg [7:0]  shift_reg;
    reg [2:0]  bit_cnt;
    reg        cmd_valid;
    reg        cmd_start;
    reg        cmd_read;
    reg        cmd_write;
    reg        cmd_write_multiple;
    reg        cmd_stop;
    reg        cmd_tlast;
    reg        data_valid;
    reg [7:0]  data_in;
    reg [7:0]  data_out;
    reg        data_tlast;
    reg        data_tvalid;
    reg        data_tready;
    reg        busy;
    reg        bus_control;
    reg        bus_active;
    reg        missed_ack;

    // State definitions
    localparam IDLE = 4'd0;
    localparam START = 4'd1;
    localparam ADDR = 4'd2;
    localparam ACK_ADDR = 4'd3;
    localparam WRITE = 4'd4;
    localparam ACK_WRITE = 4'd5;
    localparam READ = 4'd6;
    localparam ACK_READ = 4'd7;
    localparam STOP = 4'd8;

    // State machine
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= IDLE;
        end else begin
            state <= next_state;
        end
    end

    // Next state logic
    always @(*) begin
        next_state = state;
        case (state)
            IDLE: begin
                if (cmd_valid && cmd_start) begin
                    next_state = START;
                end
            end
            START: begin
                next_state = ADDR;
            end
            ADDR: begin
                if (bit_cnt == 3'd7) begin
                    next_state = ACK_ADDR;
                end
            end
            ACK_ADDR: begin
                next_state = cmd_read ? READ : WRITE;
            end
            WRITE: begin
                if (bit_cnt == 3'd7) begin
                    next_state = ACK_WRITE;
                end
            end
            ACK_WRITE: begin
                if (cmd_write_multiple && data_valid) begin
                    next_state = WRITE;
                end else if (cmd_stop) begin
                    next_state = STOP;
                end else begin
                    next_state = IDLE;
                end
            end
            READ: begin
                if (bit_cnt == 3'd7) begin
                    next_state = ACK_READ;
                end
            end
            ACK_READ: begin
                next_state = cmd_stop ? STOP : READ;
            end
            STOP: begin
                next_state = IDLE;
            end
        endcase
    end

    // Output logic
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            scl_cnt <= 16'd0;
            scl <= 1'b1;
            sda <= 1'b1;
            scl_dir <= 1'b0;
            sda_dir <= 1'b0;
            bit_cnt <= 3'd0;
            cmd_valid <= 1'b0;
            cmd_start <= 1'b0;
            cmd_read <= 1'b0;
            cmd_write <= 1'b0;
            cmd_write_multiple <= 1'b0;
            cmd_stop <= 1'b0;
            cmd_tlast <= 1'b0;
            data_valid <= 1'b0;
            data_in <= 8'd0;
            data_out <= 8'd0;
            data_tlast <= 1'b0;
            data_tvalid <= 1'b0;
            data_tready <= 1'b0;
            busy <= 1'b0;
            bus_control <= 1'b0;
            bus_active <= 1'b0;
            missed_ack <= 1'b0;
        end else begin
            // SCL and SDA control
            if (scl_cnt == prescale - 1'd1) begin
                scl_cnt <= 16'd0;
                scl <= ~scl;
            end else begin
                scl_cnt <= scl_cnt + 1'd1;
            end

            case (state)
                IDLE: begin
                    scl_dir <= 1'b0;
                    sda_dir <= 1'b0;
                    scl <= 1'b1;
                    sda <= 1'b1;
                    cmd_valid <= s_axis_cmd_valid;
                    cmd_start <= s_axis_cmd_start;
                    cmd_read <= s_axis_cmd_read;
                    cmd_write <= s_axis_cmd_write;
                    cmd_write_multiple <= s_axis_cmd_write_multiple;
                    cmd_stop <= s_axis_cmd_stop;
                    cmd_tlast <= s_axis_data_tlast;
                    data_valid <= s_axis_data_tvalid;
                    data_in <= s_axis_data_tdata;
                    busy <= 1'b0;
                    bus_control <= 1'b0;
                    bus_active <= 1'b0;
                    missed_ack <= 1'b0;
                    s_axis_cmd_ready <= 1'b1;
                    s_axis_data_tready <= 1'b1;
                    m_axis_data_tvalid <= 1'b0;
                    m_axis_data_tlast <= 1'b0;
                end
                START: begin
                    if (scl_cnt == prescale - 1'd1) begin
                        scl_dir <= 1'b1;
                        sda_dir <= 1'b1;
                        sda <= 1'b0;
                    end
                    busy <= 1'b1;
                    bus_control <= 1'b1;
                    bus_active <= 1'b1;
                    s_axis_cmd_ready <= 1'b0;
                    s_axis_data_tready <= 1'b0;
                    m_axis_data_tvalid <= 1'b0;
                    m_axis_data_tlast <= 1'b0;
                end
                ADDR: begin
                    if (scl_cnt == prescale - 1'd1) begin
                        if (scl) begin
                            if (bit_cnt == 3'd0) begin
                                shift_reg <= {s_axis_cmd_address, cmd_read};
                            end
                            sda <= shift_reg[7 - bit_cnt];
                            bit_cnt <= bit_cnt + 1'd1;
                        end
                    end
                    scl_dir <= 1'b1;
                    sda_dir <= 1'b1;
                    busy <= 1'b1;
                    bus_control <= 1'b1;
                    bus_active <= 1'b1;
                    s_axis_cmd_ready <= 1'b0;
                    s_axis_data_tready <= 1'b0;
                    m_axis_data_tvalid <= 1'b0;
                    m_axis_data_tlast <= 1'b0;
                end
                ACK_ADDR: begin
                    if (scl_cnt == prescale - 1'd1) begin
                        if (scl) begin
                            sda_dir <= 1'b0;
                            missed_ack <= sda_i;
                            bit_cnt <= 3'd0;
                        end
                    end
                    scl_dir <= 1'b1;
                    busy <= 1'b1;
                    bus_control <= 1'b1;
                    bus_active <= 1'b1;
                    s_axis_cmd_ready <= 1'b0;
                    s_axis_data_tready <= 1'b0;
                    m_axis_data_tvalid <= 1'b0;
                    m_axis_data_tlast <= 1'b0;
                end
                WRITE: begin
                    if (scl_cnt == prescale - 1'd1) begin
                        if (scl) begin
                            if (bit_cnt == 3'd0) begin
                                shift_reg <= data_in;
                            end
                            sda <= shift_reg[7 - bit_cnt];
                            bit_cnt <= bit_cnt + 1'd1;
                        end
                    end
                    scl_dir <= 1'b1;
                    sda_dir <= 1'b1;
                    busy <= 1'b1;
                    bus_control <= 1'b1;
                    bus_active <= 1'b1;
                    s_axis_cmd_ready <= 1'b0;
                    s_axis_data_tready <= 1'b1;
                    m_axis_data_tvalid <= 1'b0;
                    m_axis_data_tlast <= 1'b0;
                end
                ACK_WRITE: begin
                    if (scl_cnt == prescale - 1'd1) begin
                        if (scl) begin
                            sda_dir <= 1'b0;
                            missed_ack <= sda_i;
                            bit_cnt <= 3'd0;
                            if (cmd_write_multiple && data_valid) begin
                                data_tready <= 1'b1;
                            end else begin
                                data_tready <= 1'b0;
                            end
                        end
                    end
                    scl_dir <= 1'b1;
                    busy <= 1'b1;
                    bus_control <= 1'b1;
                    bus_active <= 1'b1;
                    s_axis_cmd_ready <= 1'b0;
                    s_axis_data_tready <= 1'b0;
                    m_axis_data_tvalid <= 1'b0;
                    m_axis_data_tlast <= 1'b0;
                end
                READ: begin
                    if (scl_cnt == prescale - 1'd1) begin
                        if (scl) begin
                            if (bit_cnt == 3'd0) begin
                                shift_reg <= 8'd0;
                            end
                            shift_reg[7 - bit_cnt] <= sda_i;
                            bit_cnt <= bit_cnt + 1'd1;
                        end
                    end
                    scl_dir <= 1'b1;
                    sda_dir <= 1'b0;
                    busy <= 1'b1;
                    bus_control <= 1'b1;
                    bus_active <= 1'b1;
                    s_axis_cmd_ready <= 1'b0;
                    s_axis_data_tready <= 1'b0;
                    m_axis_data_tvalid <= 1'b0;
                    m_axis_data_tlast <= 1'b0;
                end
                ACK_READ: begin
                    if (scl_cnt == prescale - 1'd1) begin
                        if (scl) begin
                            sda <= ~cmd_stop;
                            bit_cnt <= 3'd0;
                            data_out <= shift_reg;
                            data_tvalid <= 1'b1;
                            data_tlast <= cmd_tlast;
                            if (cmd_stop) begin
                                data_tready <= 1'b1;
                            end else begin
                                data_tready <= 1'b0;
                            end
                        end
                    end
                    scl_dir <= 1'b1;
                    sda_dir <= 1'b1;
                    busy <= 1'b1;
                    bus_control <= 1'b1;
                    bus_active <= 1'b1;
                    s_axis_cmd_ready <= 1'b0;
                    s_axis_data_tready <= 1'b0;
                    m_axis_data_tvalid <= 1'b1;
                    m_axis_data_tlast <= cmd_tlast;
                end
                STOP: begin
                    if (scl_cnt == prescale - 1'd1) begin
                        if (scl) begin
                            scl_dir <= 1'b1;
                            sda_dir <= 1'b1;
                            sda <= 1'b1;
                        end
                    end
                    busy <= 1'b0;
                    bus_control <= 1'b0;
                    bus_active <= 1'b0;
                    s_axis_cmd_ready <= 1'b1;
                    s_axis_data_tready <= 1'b1;
                    m_axis_data_tvalid <= 1'b0;
                    m_axis_data_tlast <= 1'b0;
                end
            endcase
        end
    end

    // Assign outputs
    assign i2c_scl_o = scl;
    assign i2c_scl_t = ~scl_dir;
    assign i2c_sda_o = sda;
    assign i2c_sda_t = ~sda_dir;
    assign i2c_busy = busy;
    assign i2c_bus_control = bus_control;
    assign i2c_bus_active = bus_active;
    assign i2c_missed_ack = missed_ack;
    assign s_axis_cmd_ready = ~busy & s_axis_cmd_valid;
    assign s_axis_data_tready = data_tready;
    assign m_axis_data_tdata = data_out;
    assign m_axis_data_tvalid = data_tvalid;
    assign m_axis_data_tlast = data_tlast;

endmodule
