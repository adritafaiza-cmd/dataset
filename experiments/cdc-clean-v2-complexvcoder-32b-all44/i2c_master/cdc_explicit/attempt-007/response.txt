=== STAGE 2: VERILOG ===
```verilog
module i2c_master (
    input  wire        clk,
    input  wire        rst,
    /*
     * Host interface
     */
    input  wire [6:0]  s_axis_cmd_address,
    input  wire        s_axis_cmd_start,
    input  wire        s_axis_cmd_read,
    input  wire        s_axis_cmd_write,
    input  wire        s_axis_cmd_write_multiple,
    input  wire        s_axis_cmd_stop,
    input  wire        s_axis_cmd_valid,
    output wire        s_axis_cmd_ready,
    input  wire [7:0]  s_axis_data_tdata,
    input  wire        s_axis_data_tvalid,
    output wire        s_axis_data_tready,
    input  wire        s_axis_data_tlast,
    output wire [7:0]  m_axis_data_tdata,
    output wire        m_axis_data_tvalid,
    input  wire        m_axis_data_tready,
    output wire        m_axis_data_tlast,
    /*
     * I2C interface
     */
    input  wire        scl_i,
    output wire        scl_o,
    output wire        scl_t,
    input  wire        sda_i,
    output wire        sda_o,
    output wire        sda_t,
    /*
     * Status
     */
    output wire        busy,
    output wire        bus_control,
    output wire        bus_active,
    output wire        missed_ack,
    /*
     * Configuration
     */
    input  wire [15:0] prescale,
    input  wire        stop_on_idle
);

    // Internal signals
    reg [15:0] clk_div;
    reg [2:0]  bit_count;
    reg [7:0]  byte_count;
    reg [7:0]  current_byte;
    reg        scl_reg;
    reg        sda_reg;
    reg        scl_reg_next;
    reg        sda_reg_next;
    reg        scl_t_next;
    reg        sda_t_next;
    reg [3:0]  state;
    reg [3:0]  next_state;
    reg        s_axis_cmd_ready_reg;
    reg        s_axis_data_tready_reg;
    reg        m_axis_data_tvalid_reg;
    reg        m_axis_data_tlast_reg;
    reg        busy_reg;
    reg        bus_control_reg;
    reg        bus_active_reg;
    reg        missed_ack_reg;

    // States
    localparam IDLE = 4'd0;
    localparam START = 4'd1;
    localparam ADDRESS = 4'd2;
    localparam ACK_ADDRESS = 4'd3;
    localparam WRITE_DATA = 4'd4;
    localparam ACK_WRITE = 4'd5;
    localparam READ_DATA = 4'd6;
    localparam ACK_READ = 4'd7;
    localparam STOP = 4'd8;

    // State machine
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= IDLE;
        end else begin
            state <= next_state;
        end
    end

    always @(*) begin
        next_state = state;
        s_axis_cmd_ready_reg = 0;
        s_axis_data_tready_reg = 0;
        m_axis_data_tvalid_reg = 0;
        m_axis_data_tlast_reg = 0;
        busy_reg = 1;
        bus_control_reg = 0;
        bus_active_reg = 0;
        missed_ack_reg = 0;
        scl_reg_next = scl_reg;
        sda_reg_next = sda_reg;
        scl_t_next = 1;
        sda_t_next = 1;

        case (state)
            IDLE: begin
                if (s_axis_cmd_valid && s_axis_cmd_start) begin
                    next_state = START;
                    s_axis_cmd_ready_reg = 1;
                    bus_control_reg = 1;
                end else begin
                    busy_reg = 0;
                    bus_control_reg = 0;
                    bus_active_reg = 0;
                end
            end

            START: begin
                scl_t_next = 0;
                sda_t_next = 0;
                sda_reg_next = 0;
                scl_reg_next = 1;
                next_state = ADDRESS;
            end

            ADDRESS: begin
                scl_t_next = 0;
                sda_t_next = 0;
                if (bit_count == 3'd7) begin
                    scl_reg_next = 0;
                    next_state = ACK_ADDRESS;
                end else begin
                    sda_reg_next = s_axis_cmd_address[6-bit_count];
                    scl_reg_next = 1;
                end
                bit_count = bit_count + 1;
            end

            ACK_ADDRESS: begin
                scl_t_next = 0;
                sda_t_next = 1;
                scl_reg_next = 1;
                next_state = (sda_i) ? ACK_ADDRESS : (s_axis_cmd_read) ? READ_DATA : WRITE_DATA;
                missed_ack_reg = sda_i;
            end

            WRITE_DATA: begin
                scl_t_next = 0;
                sda_t_next = 0;
                if (bit_count == 3'd7) begin
                    scl_reg_next = 0;
                    next_state = ACK_WRITE;
                end else begin
                    sda_reg_next = current_byte[7-bit_count];
                    scl_reg_next = 1;
                end
                bit_count = bit_count + 1;
            end

            ACK_WRITE: begin
                scl_t_next = 0;
                sda_t_next = 1;
                scl_reg_next = 1;
                next_state = (sda_i) ? ACK_WRITE : (s_axis_data_tlast) ? STOP : WRITE_DATA;
                missed_ack_reg = sda_i;
                if (s_axis_data_tvalid && !s_axis_data_tready_reg) begin
                    current_byte = s_axis_data_tdata;
                    s_axis_data_tready_reg = 1;
                end
            end

            READ_DATA: begin
                scl_t_next = 0;
                sda_t_next = 1;
                if (bit_count == 3'd7) begin
                    scl_reg_next = 0;
                    next_state = ACK_READ;
                end else begin
                    scl_reg_next = 1;
                    bit_count = bit_count + 1;
                end
                if (scl_reg && !scl_i) begin
                    current_byte = {current_byte[6:0], sda_i};
                end
            end

            ACK_READ: begin
                scl_t_next = 0;
                sda_t_next = 0;
                scl_reg_next = 1;
                next_state = (s_axis_data_tlast) ? STOP : READ_DATA;
                if (!s_axis_data_tvalid || s_axis_data_tready_reg) begin
                    m_axis_data_tvalid_reg = 1;
                    m_axis_data_tdata = current_byte;
                    m_axis_data_tlast_reg = s_axis_data_tlast;
                    s_axis_data_tready_reg = 1;
                end
            end

            STOP: begin
                scl_t_next = 0;
                sda_t_next = 0;
                sda_reg_next = 0;
                scl_reg_next = 0;
                next_state = IDLE;
                busy_reg = 0;
                bus_control_reg = 0;
                bus_active_reg = 0;
            end

            default: begin
                next_state = IDLE;
            end
        endcase
    end

    // Clock Divider
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            clk_div <= 16'd0;
        end else begin
            if (clk_div == prescale) begin
                clk_div <= 16'd0;
            end else begin
                clk_div <= clk_div + 1;
            end
        end
    end

    // Bit Counter
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            bit_count <= 3'd0;
        end else begin
            if (state == ADDRESS || state == WRITE_DATA || state == READ_DATA) begin
                if (clk_div == prescale && scl_reg && !scl_i) begin
                    bit_count <= bit_count + 1;
                end
            end else begin
                bit_count <= 3'd0;
            end
        end
    end

    // Byte Counter
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            byte_count <= 8'd0;
        end else begin
            if (state == WRITE_DATA || state == READ_DATA) begin
                if (s_axis_data_tvalid && s_axis_data_tready_reg) begin
                    byte_count <= byte_count + 1;
                end
            end else begin
                byte_count <= 8'd0;
            end
        end
    end

    // Current Byte
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            current_byte <= 8'd0;
        end else begin
            if (state == WRITE_DATA && s_axis_data_tvalid && s_axis_data_tready_reg) begin
                current_byte <= s_axis_data_tdata;
            end
        end
    end

    // SCL and SDA Registers
    always @(posedge clk) begin
        scl_reg <= scl_reg_next;
        sda_reg <= sda_reg_next;
        scl_t <= scl_t_next;
        sda_t <= sda_t_next;
    end

    // Outputs
    assign s_axis_cmd_ready = s_axis_cmd_ready_reg;
    assign s_axis_data_tready = s_axis_data_tready_reg;
    assign m_axis_data_tdata = m_axis_data_tdata;
    assign m_axis_data_tvalid = m_axis_data_tvalid_reg;
    assign m_axis_data_tlast = m_axis_data_tlast_reg;
    assign scl_o = scl_reg;
    assign scl_t = scl_t;
    assign sda_o = sda_reg;
    assign sda_t = sda_t;
    assign busy = busy_reg;
    assign bus_control = bus_control_reg;
    assign bus_active = bus_active_reg;
    assign missed_ack = missed_ack_reg;

endmodule
```
