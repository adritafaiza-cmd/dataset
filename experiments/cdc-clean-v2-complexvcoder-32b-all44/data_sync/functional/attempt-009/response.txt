=== STAGE 2: VERILOG ===
```verilog
module data_sync #(
    parameter STAGES = 2,
    parameter DWIDTH = 8
)(
    input                  clk,
    input                  rstn,
    input  [DWIDTH-1:0]    din,
    input                  dready_i,
    output reg [DWIDTH-1:0] dout,
    output reg             dready_o
);

    // Internal signals
    reg [DWIDTH-1:0] sync_stages [STAGES-1:0];
    reg                dready_o_reg;

    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dready_o <= 0;
            dready_o_reg <= 0;
            for (int i = 0; i < STAGES; i = i + 1) begin
                sync_stages[i] <= {DWIDTH{1'b0}};
            end
        end else begin
            // Shift the input through the synchronization stages
            sync_stages[0] <= din;
            for (int i = 1; i < STAGES; i = i + 1) begin
                sync_stages[i] <= sync_stages[i-1];
            end

            // Update dout with the last stage of the synchronization
            dout <= sync_stages[STAGES-1];

            // Pulse dready_o when dready_i is high and dready_o_reg is low
            if (dready_i && !dready_o_reg) begin
                dready_o <= 1;
            end else begin
                dready_o <= 0;
            end

            // Register the state of dready_o
            dready_o_reg <= dready_o;
        end
    end

endmodule
```
