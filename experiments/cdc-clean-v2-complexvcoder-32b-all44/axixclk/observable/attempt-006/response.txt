module axixclk #(
		parameter integer C_S_AXI_ID_WIDTH	= 2,
		parameter integer C_S_AXI_DATA_WIDTH	= 32,
		parameter integer C_S_AXI_ADDR_WIDTH	= 6,
		parameter [0:0]	OPT_WRITE_ONLY = 1'b0,

		parameter	LGFIFO = 5

		input	wire [(C_S_AXI_DATA_WIDTH/8)-1 : 0] S_AXI_WSTRB,
		input	wire				S_AXI_WLAST,
		input	wire				S_AXI_WVALID,
		output	wire				S_AXI_WREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_BID,
		output	wire [1 : 0]			S_AXI_BRESP,
		output	wire				S_AXI_BVALID,
		input	wire				S_AXI_BREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_ARID,
		input	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	S_AXI_ARADDR,
		input	wire [7 : 0]			S_AXI_ARLEN,
		input	wire [2 : 0]			S_AXI_ARSIZE,
		input	wire [1 : 0]			S_AXI_ARBURST,
		input	wire				S_AXI_ARLOCK,
		input	wire [3 : 0]			S_AXI_ARCACHE,

		input	wire				S_AXI_ARVALID,
		output	wire				S_AXI_ARREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_RID,
		output	wire [C_S_AXI_DATA_WIDTH-1 : 0]	S_AXI_RDATA,
		output	wire [1 : 0]			S_AXI_RRESP,
		output	wire				S_AXI_RLAST,
		output	wire				S_AXI_RVALID,
		input	wire				S_AXI_RREADY,
		input	wire				M_AXI_ACLK,
		output	wire				M_AXI_ARESETN,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_AWID,
		output	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	M_AXI_AWADDR,
		output	wire [7 : 0]			M_AXI_AWLEN,
		output	wire [2 : 0]			M_AXI_AWSIZE,
		output	wire [1 : 0]			M_AXI_AWBURST,
		output	wire				M_AXI_AWLOCK,
		output	wire [3 : 0]			M_AXI_AWCACHE,
		output	wire [2 : 0]			M_AXI_AWPROT,
		output	wire [3 : 0]			M_AXI_AWQOS,
		output	wire				M_AXI_AWVALID,
		input	wire				M_AXI_AWREADY,
		output	wire [C_S_AXI_DATA_WIDTH-1 : 0]	M_AXI_WDATA,

		output	wire				M_AXI_BREADY,

		output	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	M_AXI_ARADDR,
		output	wire [7 : 0]			M_AXI_ARLEN,
		output	wire [2 : 0]			M_AXI_ARSIZE,
		output	wire [1 : 0]			M_AXI_ARBURST,
		output	wire				M_AXI_ARLOCK,
		output	wire [3 : 0]			M_AXI_ARCACHE,

		input	wire [1 : 0]			M_AXI_RRESP,

		.DEPTH (1 << LGFIFO)
	) aw_fifo (
		.clk (S_AXI_ACLK),
		.rst (!s_reset_sync[XCLOCK_FFS-1]),
		.wr_en (S_AXI_AWVALID & S_AXI_AWREADY),
		.din ({S_AXI_AWID, S_AXI_AWADDR, S_AXI_AWLEN, S_AXI_AWSIZE, S_AXI_AWBURST, S_AXI_AWLOCK, S_AXI_AWCACHE, S_AXI_AWPROT, S_AXI_AWQOS, S_AXI_AWVALID}),
		.full (),
		.rd_en (awready_fifo_in),
		.dout ({awid_fifo_out, awaddr_fifo_out, awlen_fifo_out, awsize_fifo_out, awburst_fifo_out, awlock_fifo_out, awcache_fifo_out, awprot_fifo_out, awqos_fifo_out, awvalid_fifo_out}),

		.valid ()
	);

	fifo #(
		.WIDTH (C_S_AXI_DATA_WIDTH + (C_S_AXI_DATA_WIDTH/8) + 1 + 1),
		.DEPTH (1 << LGFIFO)
	) w_fifo (
		.clk (S_AXI_ACLK),
		.rst (!s_reset_sync[XCLOCK_FFS-1]),
		.wr_en (S_AXI_WVALID & S_AXI_WREADY),
		.din ({S_AXI_WDATA, S_AXI_WSTRB, S_AXI_WLAST, S_AXI_WVALID}),
		.full (),
		.rd_en (wready_fifo_in),
		.dout ({wdata_fifo_out, wstrb_fifo_out, wlast_fifo_out, wvalid_fifo_out}),
		.empty (),
		.valid ()
	);

	// M_AXI to S_AXI read FIFO
	fifo #(
		.WIDTH (C_S_AXI_ID_WIDTH + 2 + 1),
		.DEPTH (1 << LGFIFO)
	) b_fifo (
		.clk (M_AXI_ACLK),
		.rst (!m_reset_sync[XCLOCK_FFS-1]),
		.wr_en (M_AXI_BVALID & M_AXI_BREADY),
		.din ({M_AXI_BID, M_AXI_BRESP, M_AXI_BVALID}),
		.full (),
		.rd_en (bready_fifo_in),
		.dout ({bid_fifo_out, bresp_fifo_out, bvalid_fifo_out}),
		.empty (),
		.valid ()
	);

	fifo #(
		.WIDTH (C_S_AXI_ID_WIDTH + C_S_AXI_DATA_WIDTH + 2 + 1 + 1),
		.DEPTH (1 << LGFIFO)
	) r_fifo (
		.clk (M_AXI_ACLK),
		.rst (!m_reset_sync[XCLOCK_FFS-1]),
		.wr_en (M_AXI_RVALID & M_AXI_RREADY),

		.dout ({rid_fifo_out, rdata_fifo_out, rresp_fifo_out, rlast_fifo_out, rvalid_fifo_out}),
		.empty (),
		.valid ()

	assign M_AXI_AWPROT = awprot_fifo_out;

	assign M_AXI_ARQOS = S_AXI_ARQOS;
	assign M_AXI_ARVALID = S_AXI_ARVALID;

	assign S_AXI_BID = bid_fifo_out;
	assign S_AXI_BRESP = bresp_fifo_out;
	assign S_AXI_BVALID = bvalid_fifo_out;
	assign bready_fifo_in = S_AXI_BREADY;

	assign S_AXI_RID = rid_fifo_out;
	assign S_AXI_RDATA = rdata_fifo_out;
	assign S_AXI_RRESP = rresp_fifo_out;
	assign S_AXI_RLAST = rlast_fifo_out;
	assign S_AXI_RVALID = rvalid_fifo_out;
	assign rready_fifo_in = S_AXI_RREADY;

	// Clock-domain crossing for reset
	cdc_sync #(.STAGES(XCLOCK_FFS)) s_reset_cdc (
		.clk (S_AXI_ACLK),
		.rst (!S_AXI_ARESETN),
		.din (1'b1),
		.dout (s_reset_sync)
	);

	cdc_sync #(.STAGES(XCLOCK_FFS)) m_reset_cdc (
		.clk (M_AXI_ACLK),
		.rst (!S_AXI_ARESETN),
		.din (1'b1),
		.dout (m_reset_sync)
	);

	assign M_AXI_ARESETN = m_reset_sync[XCLOCK_FFS-1];

endmodule

module fifo #(
	parameter WIDTH = 8,
	parameter DEPTH = 16
) (
	input wire clk,
	input wire rst,
	input wire wr_en,
	input wire [WIDTH-1:0] din,
	output wire full,
	input wire rd_en,
	output wire [WIDTH-1:0] dout,
	output wire empty,
	output wire valid
);

	reg [WIDTH-1:0] mem [0:DEPTH-1];
	reg [$clog2(DEPTH)-1:0] wr_ptr, rd_ptr;
	reg [$clog2(DEPTH):0] count;

	always @(posedge clk or posedge rst) begin
		if (rst) begin
			wr_ptr <= 0;
			rd_ptr <= 0;
			count <= 0;
		end else begin
			if (wr_en && !full) begin
				mem[wr_ptr] <= din;
				wr_ptr <= wr_ptr + 1;
				count <= count + 1;
			end
			if (rd_en && !empty) begin
				rd_ptr <= rd_ptr + 1;
				count <= count - 1;
			end
		end
	end

	assign dout = mem[rd_ptr];
	assign full = (count == DEPTH);
	assign empty = (count == 0);
	assign valid = !empty;

endmodule

module cdc_sync #(
	parameter STAGES = 2
) (
	input wire clk,
	input wire rst,
	input wire din,
	output reg [STAGES-1:0] dout
);

	reg [STAGES-1:0] sync_reg;

	always @(posedge clk or posedge rst) begin
		if (rst) begin
			sync_reg <= 0;
		end else begin
			sync_reg <= {sync_reg[STAGES-2:0], din};
		end
	end

	assign dout = sync_reg;

endmodule
