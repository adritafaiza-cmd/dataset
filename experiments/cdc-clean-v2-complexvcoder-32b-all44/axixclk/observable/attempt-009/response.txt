module cdc_fifo #(
		parameter DATA_WIDTH = 32,
		parameter DEPTH = 5
	) (
		input	wire				src_clk,
		input	wire				src_rst,
		input	wire				dst_clk,
		input	wire				dst_rst,
		input	wire [DATA_WIDTH-1 : 0]	din,
		input	wire				wr_en,
		output	wire				full,
		output	wire [DATA_WIDTH-1 : 0]	dout,
		input	wire				rd_en,
		output	wire				empty
	);

	reg [DATA_WIDTH-1 : 0] mem [0:DEPTH-1];
	reg [$clog2(DEPTH)-1 : 0] wr_ptr, rd_ptr, wr_ptr_gray, rd_ptr_gray, rd_ptr_gray_sync;
	reg [$clog2(DEPTH)-1 : 0] wr_ptr_gray_sync;

	// Gray code conversion
	function automatic [$clog2(DEPTH)-1 : 0] bin2gray(input [$clog2(DEPTH)-1 : 0] bin);
		return bin ^ (bin >> 1);
	endfunction

	// Write pointer and FIFO memory update
	always @(posedge src_clk or posedge src_rst) begin
		if (src_rst) begin
			wr_ptr <= 0;
		end else if (wr_en & ~full) begin
			mem[wr_ptr] <= din;
			wr_ptr <= wr_ptr + 1;
		end
	end

	// Read pointer update
	always @(posedge dst_clk or posedge dst_rst) begin
		if (dst_rst) begin
			rd_ptr <= 0;
		end else if (rd_en & ~empty) begin
			rd_ptr <= rd_ptr + 1;
		end
	end

	// Gray code conversion and synchronization
	always @(posedge src_clk or posedge src_rst) begin
		if (src_rst) begin
			wr_ptr_gray <= 0;
		end else begin
			wr_ptr_gray <= bin2gray(wr_ptr);
		end
	end

	always @(posedge dst_clk or posedge dst_rst) begin
		if (dst_rst) begin
			wr_ptr_gray_sync <= 0;
		end else begin
			wr_ptr_gray_sync <= wr_ptr_gray;
		end
	end

	always @(posedge dst_clk or posedge dst_rst) begin
		if (dst_rst) begin
			rd_ptr_gray <= 0;
		end else begin
			rd_ptr_gray <= bin2gray(rd_ptr);
		end
	end

	always @(posedge src_clk or posedge src_rst) begin
		if (src_rst) begin
			rd_ptr_gray_sync <= 0;
		end else begin
			rd_ptr_gray_sync <= rd_ptr_gray;
		end
	end

	// FIFO status
	assign full = (wr_ptr_gray_sync == (rd_ptr_gray + 1));
	assign empty = (wr_ptr_gray_sync == rd_ptr_gray_sync);

	// FIFO output
	assign dout = mem[rd_ptr];

endmodule
