        .DEPTH(2**LGFIFO)
    ) w_cdc_fifo (
        .clk_wr(S_AXI_ACLK),
        .clk_rd(M_AXI_ACLK),
        .rst(s_axi_reset_sync[XCLOCK_FFS-1]),
        .data_in(w_fifo_data_in),
        .wr_en(S_AXI_WVALID & s_wready),
        .rd_en(m_wvalid & M_AXI_WREADY),
        .data_out(w_fifo_data_out),
        .full(~s_wready),
        .empty(~m_wvalid)
    );

    reg [1:0] w_delay_cnt;
    reg [C_S_AXI_DATA_WIDTH + (C_S_AXI_DATA_WIDTH/8) + 1 - 1 : 0] w_delayed_data;

    always @(posedge M_AXI_ACLK or posedge m_axi_reset_sync[XCLOCK_FFS-1]) begin
        if (m_axi_reset_sync[XCLOCK_FFS-1]) begin
            w_delay_cnt <= 2'b00;
            w_delayed_data <= 0;
        end else begin
            if (m_wvalid & M_AXI_WREADY) begin
                w_delay_cnt <= 2'b01;
                w_delayed_data <= w_fifo_data_out;
            end else if (w_delay_cnt == 2'b01) begin
                w_delay_cnt <= 2'b10;
            end else if (w_delay_cnt == 2'b10) begin
                w_delay_cnt <= 2'b00;
            end
        end
    end

    assign S_AXI_WREADY = s_wready;
    assign M_AXI_WDATA = w_delayed_data[C_S_AXI_DATA_WIDTH + (C_S_AXI_DATA_WIDTH/8) + 1 - 1 -: C_S_AXI_DATA_WIDTH];
    assign M_AXI_WSTRB = w_delayed_data[(C_S_AXI_DATA_WIDTH/8) + 1 - 1 -: (C_S_AXI_DATA_WIDTH/8)];
    assign M_AXI_WLAST = w_delayed_data[0];
    assign M_AXI_WVALID = w_delay_cnt == 2'b10;

    // B channel
    wire [C_S_AXI_ID_WIDTH + 2 - 1 : 0] b_fifo_data_in;
    wire [C_S_AXI_ID_WIDTH + 2 - 1 : 0] b_fifo_data_out;
    wire s_bvalid, m_bready;

    assign b_fifo_data_in = {M_AXI_BID, M_AXI_BRESP};

    cdc_fifo #(
        .WIDTH(C_S_AXI_ID_WIDTH + 2),
        .DEPTH(2**LGFIFO)
    ) b_cdc_fifo (
        .clk_wr(M_AXI_ACLK),
        .clk_rd(S_AXI_ACLK),
        .rst(m_axi_reset_sync[XCLOCK_FFS-1]),
        .data_in(b_fifo_data_in),
        .wr_en(M_AXI_BVALID & m_bready),
        .rd_en(s_bvalid & S_AXI_BREADY),
        .data_out(b_fifo_data_out),
        .full(~m_bready),
        .empty(~s_bvalid)
    );

    assign S_AXI_BID = b_fifo_data_out[C_S_AXI_ID_WIDTH + 2 - 1 -: C_S_AXI_ID_WIDTH];
    assign S_AXI_BRESP = b_fifo_data_out[2 - 1 -: 2];
    assign S_AXI_BVALID = s_bvalid;
    assign M_AXI_BREADY = m_bready;

    // AR channel
    wire [C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4 - 1 : 0] ar_fifo_data_in;
    wire [C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4 - 1 : 0] ar_fifo_data_out;
    wire s_arready, m_arvalid;

    assign ar_fifo_data_in = {S_AXI_ARID, S_AXI_ARADDR, S_AXI_ARLEN, S_AXI_ARSIZE, S_AXI_ARBURST, S_AXI_ARLOCK, S_AXI_ARCACHE, S_AXI_ARPROT, S_AXI_ARQOS};

    cdc_fifo #(
        .WIDTH(C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4),
        .DEPTH(2**LGFIFO)
    ) ar_cdc_fifo (

    assign S_AXI_ARREADY = s_arready;
    assign M_AXI_ARID = ar_fifo_data_out[C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4 + C_S_AXI_ID_WIDTH - 1 -: C_S_AXI_ID_WIDTH];
    assign M_AXI_ARADDR = ar_fifo_data_out[C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4 - 1 -: C_S_AXI_ADDR_WIDTH];
    assign M_AXI_ARLEN = ar_fifo_data_out[8 + 3 + 2 + 1 + 4 + 3 + 4 - 1 -: 8];
    assign M_AXI_ARSIZE = ar_fifo_data_out[3 + 2 + 1 + 4 + 3 + 4 - 1 -: 3];
    assign M_AXI_ARBURST = ar_fifo_data_out[2 + 1 + 4 + 3 + 4 - 1 -: 2];
    assign M_AXI_ARLOCK = ar_fifo_data_out[1 + 4 + 3 + 4 - 1];
    assign M_AXI_ARCACHE = ar_fifo_data_out[4 + 3 + 4 - 1 -: 4];
    assign M_AXI_ARPROT = ar_fifo_data_out[3 + 4 - 1 -: 3];
    assign M_AXI_ARQOS = ar_fifo_data_out[4 - 1 -: 4];
    assign M_AXI_ARVALID = m_arvalid;

    // R channel
    wire [C_S_AXI_ID_WIDTH + C_S_AXI_DATA_WIDTH + 2 + 1 - 1 : 0] r_fifo_data_in;
    wire [C_S_AXI_ID_WIDTH + C_S_AXI_DATA_WIDTH + 2 + 1 - 1 : 0] r_fifo_data_out;

    assign S_AXI_RRESP = r_fifo_data_out[2 + 1 - 1 -: 2];
    assign S_AXI_RLAST = r_fifo_data_out[0];
    assign S_AXI_RVALID = s_rvalid;
    assign M_AXI_RREADY = m_rready;

// Cross-domain FIFO module
module cdc_fifo #(
    parameter WIDTH = 8,
    parameter DEPTH = 16
) (
    input wire clk_wr,
    input wire clk_rd,
    input wire rst,
    input wire [WIDTH-1:0] data_in,
    input wire wr_en,
    input wire rd_en,
    output reg [WIDTH-1:0] data_out,
    output wire full,
    output wire empty
);

    reg [$clog2(DEPTH)-1:0] wr_ptr, rd_ptr, wr_ptr_gray, rd_ptr_gray, rd_ptr_gray_sync;
    reg [WIDTH-1:0] mem [DEPTH-1:0];

    // Write pointer and memory write
    always @(posedge clk_wr or posedge rst) begin
        if (rst) begin
            wr_ptr <= 0;
        end else if (wr_en && !full) begin
            mem[wr_ptr] <= data_in;
            wr_ptr <= wr_ptr + 1;
        end
    end

    // Binary to Gray conversion for write pointer
    assign wr_ptr_gray = wr_ptr ^ (wr_ptr >> 1);

    // Read pointer synchronization
    always @(posedge clk_rd or posedge rst) begin
        if (rst) begin
            rd_ptr_gray_sync <= 0;
        end else begin
            rd_ptr_gray_sync <= wr_ptr_gray;
        end
    end

    // Read pointer and memory read
    always @(posedge clk_rd or posedge rst) begin
        if (rst) begin
            rd_ptr <= 0;
        end else if (rd_en && !empty) begin
            rd_ptr <= rd_ptr + 1;
        end
    end

    // Gray to Binary conversion for synchronized read pointer
    wire [$clog2(DEPTH)-1:0] rd_ptr_sync;
    assign rd_ptr_sync[0] = rd_ptr_gray_sync[0];
    genvar i;
    generate
        for (i = 1; i < $clog2(DEPTH); i = i + 1) begin : gen_gray_to_bin
            assign rd_ptr_sync[i] = rd_ptr_gray_sync[i] ^ rd_ptr_sync[i-1];
        end
    endgenerate

    // Output data
    always @(posedge clk_rd or posedge rst) begin
        if (rst) begin
            data_out <= 0;
        end else if (rd_en && !empty) begin
            data_out <= mem[rd_ptr_sync];
        end
    end

    // Full and empty flags
    assign full = (wr_ptr_gray == (rd_ptr_gray_sync ^ {(WIDTH)'(1) << ($clog2(DEPTH) - 1)}));
    assign empty = (wr_ptr_gray == rd_ptr_gray_sync);

endmodule
