    // Clock domain crossing for write address channel
    reg [XCLOCK_FFS-1 : 0] awvalid_ffs;
    reg [XCLOCK_FFS-1 : 0] awready_ffs;

    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            awvalid_ffs <= {XCLOCK_FFS{1'b0}};
            awready_ffs <= {XCLOCK_FFS{1'b0}};
        end else begin
            awvalid_ffs <= {awvalid_ffs[XCLOCK_FFS-2:0], S_AXI_AWVALID};
            awready_ffs <= {awready_ffs[XCLOCK_FFS-2:0], M_AXI_AWREADY};
        end
    end

    assign S_AXI_AWREADY = awready_ffs[XCLOCK_FFS-1];
    assign M_AXI_AWVALID = awvalid_ffs[XCLOCK_FFS-1];

    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            awid_reg   <= {C_S_AXI_ID_WIDTH{1'b0}};
            awaddr_reg <= {C_S_AXI_ADDR_WIDTH{1'b0}};
            awlen_reg  <= 8'b0;
            awsize_reg <= 3'b0;
            awburst_reg <= 2'b0;
            awlock_reg <= 1'b0;
            awcache_reg <= 4'b0;
            awprot_reg <= 3'b0;
            awqos_reg <= 4'b0;

            awid_reg   <= S_AXI_AWID;
            awaddr_reg <= S_AXI_AWADDR;
            awlen_reg  <= S_AXI_AWLEN;
            awsize_reg <= S_AXI_AWSIZE;
            awburst_reg <= S_AXI_AWBURST;
            awlock_reg <= S_AXI_AWLOCK;
            awcache_reg <= S_AXI_AWCACHE;
            awprot_reg <= S_AXI_AWPROT;
            awqos_reg <= S_AXI_AWQOS;
        end
    end

    assign M_AXI_AWID   = awid_reg;
    assign M_AXI_AWADDR = awaddr_reg;
    assign M_AXI_AWLEN  = awlen_reg;
    assign M_AXI_AWSIZE = awsize_reg;
