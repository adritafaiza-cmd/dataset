module axixclk #(
		parameter integer C_S_AXI_ID_WIDTH	= 2,
		parameter integer C_S_AXI_DATA_WIDTH	= 32,
		parameter integer C_S_AXI_ADDR_WIDTH	= 6,
		parameter [0:0]	OPT_WRITE_ONLY = 1'b0,
		parameter [0:0]	OPT_READ_ONLY = 1'b0,
		parameter	XCLOCK_FFS = 2,
		parameter	LGFIFO = 5
	) (
		input	wire				S_AXI_ACLK,
		input	wire				S_AXI_ARESETN,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_AWID,
		input	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	S_AXI_AWADDR,
		input	wire [7 : 0]			S_AXI_AWLEN,
		input	wire [2 : 0]			S_AXI_AWSIZE,
		input	wire [1 : 0]			S_AXI_AWBURST,
		input	wire				S_AXI_AWLOCK,
		input	wire [3 : 0]			S_AXI_AWCACHE,
		input	wire [2 : 0]			S_AXI_AWPROT,
		input	wire [3 : 0]			S_AXI_AWQOS,
		input	wire				S_AXI_AWVALID,
		output	wire				S_AXI_AWREADY,
		input	wire [C_S_AXI_DATA_WIDTH-1 : 0]	S_AXI_WDATA,
		input	wire [(C_S_AXI_DATA_WIDTH/8)-1 : 0] S_AXI_WSTRB,
		input	wire				S_AXI_WLAST,
		input	wire				S_AXI_WVALID,
		output	wire				S_AXI_WREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_BID,
		output	wire [1 : 0]			S_AXI_BRESP,
		output	wire				S_AXI_BVALID,
		input	wire				S_AXI_BREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_ARID,
		input	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	S_AXI_ARADDR,
		input	wire [7 : 0]			S_AXI_ARLEN,
		input	wire [2 : 0]			S_AXI_ARSIZE,
		input	wire [1 : 0]			S_AXI_ARBURST,
		input	wire				S_AXI_ARLOCK,
		input	wire [3 : 0]			S_AXI_ARCACHE,
		input	wire [2 : 0]			S_AXI_ARPROT,
		input	wire [3 : 0]			S_AXI_ARQOS,
		input	wire				S_AXI_ARVALID,
		output	wire				S_AXI_ARREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_RID,
		output	wire [C_S_AXI_DATA_WIDTH-1 : 0]	S_AXI_RDATA,
		output	wire [1 : 0]			S_AXI_RRESP,
		output	wire				S_AXI_RLAST,
		output	wire				S_AXI_RVALID,
		input	wire				S_AXI_RREADY,
		input	wire				M_AXI_ACLK,
		output	wire				M_AXI_ARESETN,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_AWID,
		output	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	M_AXI_AWADDR,
		output	wire [7 : 0]			M_AXI_AWLEN,
		output	wire [2 : 0]			M_AXI_AWSIZE,
		output	wire [1 : 0]			M_AXI_AWBURST,
		output	wire				M_AXI_AWLOCK,
		output	wire [3 : 0]			M_AXI_AWCACHE,
		output	wire [2 : 0]			M_AXI_AWPROT,
		output	wire [3 : 0]			M_AXI_AWQOS,
		output	wire				M_AXI_AWVALID,
		input	wire				M_AXI_AWREADY,
		output	wire [C_S_AXI_DATA_WIDTH-1 : 0]	M_AXI_WDATA,
		output	wire [(C_S_AXI_DATA_WIDTH/8)-1 : 0] M_AXI_WSTRB,
		output	wire				M_AXI_WLAST,
		output	wire				M_AXI_WVALID,
		input	wire				M_AXI_WREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_BID,
		input	wire [1 : 0]			M_AXI_BRESP,
		input	wire				M_AXI_BVALID,
		output	wire				M_AXI_BREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_ARID,
		output	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	M_AXI_ARADDR,
		output	wire [7 : 0]			M_AXI_ARLEN,
		output	wire [2 : 0]			M_AXI_ARSIZE,
		output	wire [1 : 0]			M_AXI_ARBURST,
		output	wire				M_AXI_ARLOCK,
		output	wire [3 : 0]			M_AXI_ARCACHE,
		output	wire [2 : 0]			M_AXI_ARPROT,
		output	wire [3 : 0]			M_AXI_ARQOS,
		output	wire				M_AXI_ARVALID,
		input	wire				M_AXI_ARREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_RID,
		input	wire [C_S_AXI_DATA_WIDTH-1 : 0]	M_AXI_RDATA,
		input	wire [1 : 0]			M_AXI_RRESP,
		input	wire				M_AXI_RLAST,
		input	wire				M_AXI_RVALID,
		output	wire				M_AXI_RREADY
	);

	localparam FIFO_DEPTH = 1 << LGFIFO;

	// Write Address Channel
	reg [C_S_AXI_ID_WIDTH-1:0] awid_s, awid_m;
	reg [C_S_AXI_ADDR_WIDTH-1:0] awaddr_s, awaddr_m;
	reg [7:0] awlen_s, awlen_m;
	reg [2:0] awsize_s, awsize_m;
	reg [1:0] awburst_s, awburst_m;
	reg awlock_s, awlock_m;
	reg [3:0] awcache_s, awcache_m;
	reg [2:0] awprot_s, awprot_m;
	reg [3:0] awqos_s, awqos_m;
	reg awvalid_s, awvalid_m, awready_s, awready_m;

	// Write Data Channel
	reg [C_S_AXI_DATA_WIDTH-1:0] wdata_s, wdata_m;
	reg [(C_S_AXI_DATA_WIDTH/8)-1:0] wstrb_s, wstrb_m;
	reg wlast_s, wlast_m;
	reg wvalid_s, wvalid_m, wready_s, wready_m;

	// Write Response Channel
	reg [C_S_AXI_ID_WIDTH-1:0] bid_s, bid_m;
	reg [1:0] bresp_s, bresp_m;
	reg bvalid_s, bvalid_m, bready_s, bready_m;

	// Read Address Channel
	reg [C_S_AXI_ID_WIDTH-1:0] arid_s, arid_m;
	reg [C_S_AXI_ADDR_WIDTH-1:0] araddr_s, araddr_m;
	reg [7:0] arlen_s, arlen_m;
	reg [2:0] arsize_s, arsize_m;
	reg [1:0] arburst_s, arburst_m;
	reg arlock_s, arlock_m;
	reg [3:0] arcache_s, arcache_m;
	reg [2:0] arprot_s, arprot_m;
	reg [3:0] arqos_s, arqos_m;
	reg arvalid_s, arvalid_m, arready_s, arready_m;

	// Read Data Channel
	reg [C_S_AXI_ID_WIDTH-1:0] rid_s, rid_m;
	reg [C_S_AXI_DATA_WIDTH-1:0] rdata_s, rdata_m;
	reg [1:0] rresp_s, rresp_m;
	reg rlast_s, rlast_m;
	reg rvalid_s, rvalid_m, rready_s, rready_m;

	// FIFO for Write Address Channel
	fifo_async #(
		.WIDTH(C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4),
		.DEPTH(FIFO_DEPTH),
		.XCLOCK_FFS(XCLOCK_FFS)

module fifo_async #(
		parameter WIDTH = 32,
		parameter DEPTH = 16,
		parameter XCLOCK_FFS = 2
	) (
		input wire clk_wr,
		input wire rst_wr,
		input wire wr_en,
		input wire [WIDTH-1:0] din,
		input wire clk_rd,
		input wire rst_rd,
		input wire rd_en,
		output wire [WIDTH-1:0] dout,
		output wire full,
		output wire empty,
		output wire almost_full,
		output wire almost_empty
	);

	reg [WIDTH-1:0] mem [0:DEPTH-1];
	reg [$clog2(DEPTH)-1:0] wr_ptr, rd_ptr, wr_ptr_gray, rd_ptr_gray, rd_ptr_gray_synced, wr_ptr_gray_synced;

	// Gray code conversion
	function [WIDTH-1:0] bin2gray;
		input [WIDTH-1:0] bin;
		bin2gray = bin ^ (bin >> 1);
	endfunction

	// Write pointer logic
	always @(posedge clk_wr or posedge rst_wr) begin
		if (rst_wr) begin
			wr_ptr <= 0;
			wr_ptr_gray <= 0;
		end else if (wr_en && !full) begin
			wr_ptr <= wr_ptr + 1;
			wr_ptr_gray <= bin2gray(wr_ptr + 1);
		end
	end

	// Read pointer logic
	always @(posedge clk_rd or posedge rst_rd) begin
		if (rst_rd) begin
			rd_ptr <= 0;
			rd_ptr_gray <= 0;
		end else if (rd_en && !empty) begin
			rd_ptr <= rd_ptr + 1;
			rd_ptr_gray <= bin2gray(rd_ptr + 1);
		end
	end

	// Gray code synchronization
	generate
		genvar i;
		for (i = 0; i < XCLOCK_FFS; i = i + 1) begin : sync
			if (i == 0) begin
				always @(posedge clk_wr) rd_ptr_gray_synced <= rd_ptr_gray;
				always @(posedge clk_rd) wr_ptr_gray_synced <= wr_ptr_gray;
			end else begin
				always @(posedge clk_wr) rd_ptr_gray_synced <= rd_ptr_gray_synced;
				always @(posedge clk_rd) wr_ptr_gray_synced <= wr_ptr_gray_synced;
			end
		end
	endgenerate

	// Write operation
	always @(posedge clk_wr) begin
		if (wr_en && !full) mem[wr_ptr] <= din;
	end

	// Read operation
	always @(posedge clk_rd) begin
		if (rd_en && !empty) dout <= mem[rd_ptr];
	end

	// Full flag
	assign full = (wr_ptr_gray + 1) == rd_ptr_gray_synced;

	// Empty flag
	assign empty = wr_ptr_gray_synced == rd_ptr_gray;

	// Almost full flag
	assign almost_full = (wr_ptr_gray + 1 + 1) == rd_ptr_gray_synced;

	// Almost empty flag
	assign almost_empty = (wr_ptr_gray_synced + 1) == rd_ptr_gray;

endmodule
