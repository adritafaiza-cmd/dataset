```verilog
module axixclk #(
		parameter integer C_S_AXI_ID_WIDTH	= 2,
		parameter integer C_S_AXI_DATA_WIDTH	= 32,
		parameter integer C_S_AXI_ADDR_WIDTH	= 6,
		parameter [0:0]	OPT_WRITE_ONLY = 1'b0,
		parameter [0:0]	OPT_READ_ONLY = 1'b0,
		parameter	XCLOCK_FFS = 2,
		parameter	LGFIFO = 5
	) (
		input	wire				S_AXI_ACLK,
		input	wire				S_AXI_ARESETN,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_AWID,
		input	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	S_AXI_AWADDR,
		input	wire [7 : 0]			S_AXI_AWLEN,
		input	wire [2 : 0]			S_AXI_AWSIZE,
		input	wire [1 : 0]			S_AXI_AWBURST,
		input	wire				S_AXI_AWLOCK,
		input	wire [3 : 0]			S_AXI_AWCACHE,
		input	wire [2 : 0]			S_AXI_AWPROT,
		input	wire [3 : 0]			S_AXI_AWQOS,
		input	wire				S_AXI_AWVALID,
		output	wire				S_AXI_AWREADY,
		input	wire [C_S_AXI_DATA_WIDTH-1 : 0]	S_AXI_WDATA,
		input	wire [(C_S_AXI_DATA_WIDTH/8)-1 : 0] S_AXI_WSTRB,
		input	wire				S_AXI_WLAST,
		input	wire				S_AXI_WVALID,
		output	wire				S_AXI_WREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_BID,
		output	wire [1 : 0]			S_AXI_BRESP,
		output	wire				S_AXI_BVALID,
		input	wire				S_AXI_BREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_ARID,
		input	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	S_AXI_ARADDR,
		input	wire [7 : 0]			S_AXI_ARLEN,
		input	wire [2 : 0]			S_AXI_ARSIZE,
		input	wire [1 : 0]			S_AXI_ARBURST,
		input	wire				S_AXI_ARLOCK,
		input	wire [3 : 0]			S_AXI_ARCACHE,
		input	wire [2 : 0]			S_AXI_ARPROT,
		input	wire [3 : 0]			S_AXI_ARQOS,
		input	wire				S_AXI_ARVALID,
		output	wire				S_AXI_ARREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_RID,
		output	wire [C_S_AXI_DATA_WIDTH-1 : 0]	S_AXI_RDATA,
		output	wire [1 : 0]			S_AXI_RRESP,
		output	wire				S_AXI_RLAST,
		output	wire				S_AXI_RVALID,
		input	wire				S_AXI_RREADY,
		input	wire				M_AXI_ACLK,
		output	wire				M_AXI_ARESETN,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_AWID,
		output	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	M_AXI_AWADDR,
		output	wire [7 : 0]			M_AXI_AWLEN,
		output	wire [2 : 0]			M_AXI_AWSIZE,
		output	wire [1 : 0]			M_AXI_AWBURST,
		output	wire				M_AXI_AWLOCK,
		output	wire [3 : 0]			M_AXI_AWCACHE,
		output	wire [2 : 0]			M_AXI_AWPROT,
		output	wire [3 : 0]			M_AXI_AWQOS,
		output	wire				M_AXI_AWVALID,
		input	wire				M_AXI_AWREADY,
		output	wire [C_S_AXI_DATA_WIDTH-1 : 0]	M_AXI_WDATA,
		output	wire [(C_S_AXI_DATA_WIDTH/8)-1 : 0] M_AXI_WSTRB,
		output	wire				M_AXI_WLAST,
		output	wire				M_AXI_WVALID,
		input	wire				M_AXI_WREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_BID,
		input	wire [1 : 0]			M_AXI_BRESP,
		input	wire				M_AXI_BVALID,
		output	wire				M_AXI_BREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_ARID,
		output	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	M_AXI_ARADDR,
		output	wire [7 : 0]			M_AXI_ARLEN,
		output	wire [2 : 0]			M_AXI_ARSIZE,
		output	wire [1 : 0]			M_AXI_ARBURST,
		output	wire				M_AXI_ARLOCK,
		output	wire [3 : 0]			M_AXI_ARCACHE,
		output	wire [2 : 0]			M_AXI_ARPROT,
		output	wire [3 : 0]			M_AXI_ARQOS,
		output	wire				M_AXI_ARVALID,
		input	wire				M_AXI_ARREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_RID,
		input	wire [C_S_AXI_DATA_WIDTH-1 : 0]	M_AXI_RDATA,
		input	wire [1 : 0]			M_AXI_RRESP,
		input	wire				M_AXI_RLAST,
		input	wire				M_AXI_RVALID,
		output	wire				M_AXI_RREADY
	);

	// Internal signals
	reg [XCLOCK_FFS-1:0] s_axi_reset_sync;
	reg [XCLOCK_FFS-1:0] m_axi_reset_sync;
	reg s_axi_reset_sync_done;
	reg m_axi_reset_sync_done;

	// Write Address Channel
	reg [C_S_AXI_ID_WIDTH-1:0] awid_reg;
	reg [C_S_AXI_ADDR_WIDTH-1:0] awaddr_reg;
	reg [7:0] awlen_reg;
	reg [2:0] awsize_reg;
	reg [1:0] awburst_reg;
	reg awlock_reg;
	reg [3:0] awcache_reg;
	reg [2:0] awprot_reg;
	reg [3:0] awqos_reg;
	reg awvalid_reg;
	reg awready_reg;

	// Write Data Channel
	reg [C_S_AXI_DATA_WIDTH-1:0] wdata_reg;
	reg [(C_S_AXI_DATA_WIDTH/8)-1:0] wstrb_reg;
	reg wlast_reg;
	reg wvalid_reg;
	reg wready_reg;

	// Write Response Channel
	reg [C_S_AXI_ID_WIDTH-1:0] bid_reg;
	reg [1:0] bresp_reg;
	reg bvalid_reg;
	reg bready_reg;

	// Read Address Channel
	reg [C_S_AXI_ID_WIDTH-1:0] arid_reg;
	reg [C_S_AXI_ADDR_WIDTH-1:0] araddr_reg;
	reg [7:0] arlen_reg;
	reg [2:0] arsize_reg;
	reg [1:0] arburst_reg;
	reg arlock_reg;
	reg [3:0] arcache_reg;
	reg [2:0] arprot_reg;
	reg [3:0] arqos_reg;
	reg arvalid_reg;
	reg arready_reg;

	// Read Data Channel
	reg [C_S_AXI_ID_WIDTH-1:0] rid_reg;
	reg [C_S_AXI_DATA_WIDTH-1:0] rdata_reg;
	reg [1:0] rresp_reg;
	reg rlast_reg;
	reg rvalid_reg;
	reg rready_reg;

	// FIFOs for write address
	wire [C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 19:0] awfifo_out;
	wire awfifo_full;
	wire awfifo_empty;
	wire awfifo_almost_full;
	wire awfifo_almost_empty;

	fifo #(
		.WIDTH(C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 19),
		.DEPTH(1 << LGFIFO)
	) awfifo (
		.clk(S_AXI_ACLK),
		.reset(~s_axi_reset_sync_done),
		.wr_en(S_AXI_AWVALID & S_AXI_AWREADY),
		.rd_en(M_AXI_AWREADY),
		.din({S_AXI_AWID, S_AXI_AWADDR, S_AXI_AWLEN, S_AXI_AWSIZE, S_AXI_AWBURST, S_AXI_AWLOCK, S_AXI_AWCACHE, S_AXI_AWPROT, S_AXI_AWQOS}),
		.dout(awfifo_out),
		.full(awfifo_full),
		.empty(awfifo_empty),
		.almost_full(awfifo_almost_full),
		.almost_empty(awfifo_almost_empty)
	);

	assign S_AXI_AWREADY = ~awfifo_full;
	assign M_AXI_AWID = awfifo_out[C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 18:C_S_AXI_ADDR_WIDTH + 19];
	assign M_AXI_AWADDR = awfifo_out[C_S_AXI_ADDR_WIDTH + 18:19];
	assign M_AXI_AWLEN = awfifo_out[18:11];
	assign M_AXI_AWSIZE = awfifo_out[10:8];
	assign M_AXI_AWBURST = awfifo_out[7:6];
	assign M_AXI_AWLOCK = awfifo_out[5];
	assign M_AXI_AWCACHE = awfifo_out[4:1];
	assign M_AXI_AWPROT = awfifo_out[0];
	assign M_AXI_AWQOS = 0; // Not used in this implementation
	assign M_AXI_AWVALID = ~awfifo_empty;

	// FIFOs for write data
	wire [C_S_AXI_DATA_WIDTH + (C_S_AXI_DATA_WIDTH/8) + 1:0] wdfifo_out;
	wire wdfifo_full;
	wire wdfifo_empty;
	wire wdfifo_almost_full;
	wire wdfifo_almost_empty;

	fifo #(
		.WIDTH(C_S_AXI_DATA_WIDTH + (C_S_AXI_DATA_WIDTH/8) + 1),
		.DEPTH(1 << LGFIFO)
	) wdfifo (
		.clk(S_AXI_ACLK),
		.reset(~s_axi_reset_sync_done),
		.wr_en(S_AXI_WVALID & S_AXI_WREADY),
		.rd_en(M_AXI_WREADY),
		.din({S_AXI_WDATA, S_AXI_WSTRB, S_AXI_WLAST}),
		.dout(wdfifo_out),
		.full(wdfifo_full),
		.empty(wdfifo_empty),
		.almost_full(wdfifo_almost_full),
		.almost_empty(wdfifo_almost_empty)
	);

	assign S_AXI_WREADY = ~wdfifo_full;
	assign M_AXI_WDATA = wdfifo_out[C_S_AXI_DATA_WIDTH + (C_S_AXI_DATA_WIDTH/8):C_S_AXI_DATA_WIDTH/8 + 1];
	assign M_AXI_WSTRB = wdfifo_out[C_S_AXI_DATA_WIDTH/8:1];
	assign M_AXI_WLAST = wdfifo_out[0];
	assign M_AXI_WVALID = ~wdfifo_empty;

	// FIFOs for write response
	wire [C_S_AXI_ID_WIDTH + 2:0] brespfifo_out;
	wire brespfifo_full;
	wire brespfifo_empty;
	wire brespfifo_almost_full;
	wire brespfifo_almost_empty;

	fifo #(
		.WIDTH(C_S_AXI_ID_WIDTH + 2),
		.DEPTH(1 << LGFIFO)
	) brespfifo (
		.clk(M_AXI_ACLK),
		.reset(~m_axi_reset_sync_done),
		.wr_en(M_AXI_BVALID & M_AXI_BREADY),
		.rd_en(S_AXI_BREADY),
		.din({M_AXI_BID, M_AXI_BRESP}),
		.dout(brespfifo_out),
		.full(brespfifo_full),
		.empty(brespfifo_empty),
		.almost_full(brespfifo_almost_full),
		.almost_empty(brespfifo_almost_empty)
	);

	assign M_AXI_BREADY = ~brespfifo_full;
	assign S_AXI_BID = brespfifo_out[C_S_AXI_ID_WIDTH + 1:C_S_AXI_ID_WIDTH];
	assign S_AXI_BRESP = brespfifo_out[1:0];
	assign S_AXI_BVALID = ~brespfifo_empty;

	// FIFOs for read address
	wire [C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 19:0] arfifo_out;
	wire arfifo_full;
	wire arfifo_empty;
	wire arfifo_almost_full;
	wire arfifo_almost_empty;

	fifo #(
		.WIDTH(C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 19),
		.DEPTH(1 << LGFIFO)
	) arfifo (
		.clk(S_AXI_ACLK),
		.reset(~s_axi_reset_sync_done),
		.wr_en(S_AXI_ARVALID & S_AXI_ARREADY),
		.rd_en(M_AXI_ARREADY),
		.din({S_AXI_ARID, S_AXI_ARADDR, S_AXI_ARLEN, S_AXI_ARSIZE, S_AXI_ARBURST, S_AXI_ARLOCK, S_AXI_ARCACHE, S_AXI_ARPROT, S_AXI_ARQOS}),
		.dout(arfifo_out),
		.full(arfifo_full),
		.empty(arfifo_empty),
		.almost_full(arfifo_almost_full),
		.almost_empty(arfifo_almost_empty)
	);

	assign S_AXI_ARREADY = ~arfifo_full;
	assign M_AXI_ARID = arfifo_out[C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 18:C_S_AXI_ADDR_WIDTH + 19];
	assign M_AXI_ARADDR = arfifo_out[C_S_AXI_ADDR_WIDTH + 18:19];
	assign M_AXI_ARLEN = arfifo_out[18:11];
	assign M_AXI_ARSIZE = arfifo_out[10:8];
	assign M_AXI_ARBURST = arfifo_out[7:6];
	assign M_AXI_ARLOCK = arfifo_out[5];
	assign M_AXI_ARCACHE = arfifo_out[4:1];
	assign M_AXI_ARPROT = arfifo_out[0];
	assign M_AXI_ARQOS = 0; // Not used in this implementation
	assign M_AXI_ARVALID = ~arfifo_empty;

	// FIFOs for read data
	wire [C_S_AXI_ID_WIDTH + C_S_AXI_DATA_WIDTH + 2:0] rdfifo_out;
	wire rdfifo_full;
	wire rdfifo_empty;
	wire rdfifo_almost_full;
	wire rdfifo_almost_empty;

	fifo #(
		.WIDTH(C_S_AXI_ID_WIDTH + C_S_AXI_DATA_WIDTH + 2),
		.DEPTH(1 << LGFIFO)
	) rdfifo (
		.clk(M_AXI_ACLK),
		.reset(~m_axi_reset_sync_done),
		.wr_en(M_AXI_RVALID & M_AXI_RREADY),
		.rd_en(S_AXI_RREADY),
		.din({M_AXI_RID, M_AXI_RDATA, M_AXI_RRESP, M_AXI_RLAST}),
		.dout(rdfifo_out),
		.full(rdfifo_full),
		.empty(rdfifo_empty),
		.almost_full(rdfifo_almost_full),
		.almost_empty(rdfifo_almost_empty)
	);

	assign M_AXI_RREADY = ~rdfifo_full;
	assign S_AXI_RID = rdfifo_out[C_S_AXI_ID_WIDTH + C_S_AXI_DATA_WIDTH + 1:C_S_AXI_DATA_WIDTH + 2];
	assign S_AXI_RDATA = rdfifo_out[C_S_AXI_DATA_WIDTH + 1:C_S_AXI_DATA_WIDTH + 2 - C_S_AXI_DATA_WIDTH];
	assign S_AXI_RRESP = rdfifo_out[1:0];
	assign S_AXI_RLAST = rdfifo_out[0];
	assign S_AXI_RVALID = ~rdfifo_empty;

	// Reset synchronization
	generate
		if (XCLOCK_FFS == 2) begin
			always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
				if (!S_AXI_ARESETN) begin
					s_axi_reset_sync <= 0;
					s_axi_reset_sync_done <= 0;
				end else begin
					s_axi_reset_sync <= {s_axi_reset_sync[XCLOCK_FFS-2:0], 1'b1};
					s_axi_reset_sync_done <= &s_axi_reset_sync;
				end
			end

			always @(posedge M_AXI_ACLK or negedge S_AXI_ARESETN) begin
				if (!S_AXI_ARESETN) begin
					m_axi_reset_sync <= 0;
					m_axi_reset_sync_done <= 0;
				end else begin
					m_axi_reset_sync <= {m_axi_reset_sync[XCLOCK_FFS-2:0], 1'b1};
					m_axi_reset_sync_done <= &m_axi_reset_sync;
				end
			end
		end else begin
			always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
				if (!S_AXI_ARESETN) begin
					s_axi_reset_sync <= 0;
					s_axi_reset_sync_done <= 0;
				end else begin
					s_axi_reset_sync <= {s_axi_reset_sync[XCLOCK_FFS-2:0], 1'b1};
					s_axi_reset_sync_done <= s_axi_reset_sync[XCLOCK_FFS-1];
				end
			end

			always @(posedge M_AXI_ACLK or negedge S_AXI_ARESETN) begin
				if (!S_AXI_ARESETN) begin
					m_axi_reset_sync <= 0;
					m_axi_reset_sync_done <= 0;
				end else begin
					m_axi_reset_sync <= {m_axi_reset_sync[XCLOCK_FFS-2:0], 1'b1};
					m_axi_reset_sync_done <= m_axi_reset_sync[XCLOCK_FFS-1];
				end
			end
		end
	endgenerate

	assign M_AXI_ARESETN = s_axi_reset_sync_done;

endmodule

module fifo #(
		parameter WIDTH = 8,
		parameter DEPTH = 16
	) (
		input wire clk,
		input wire reset,
		input wire wr_en,
		input wire rd_en,
		input wire [WIDTH-1:0] din,
		output reg [WIDTH-1:0] dout,
		output reg full,
		output reg empty,
		output reg almost_full,
		output reg almost_empty
	);

	localparam ADDR_WIDTH = $clog2(DEPTH);
	reg [ADDR_WIDTH-1:0] wr_ptr;
	reg [ADDR_WIDTH-1:0] rd_ptr;
	reg [WIDTH-1:0] mem [0:DEPTH-1];

	always @(posedge clk or posedge reset) begin
		if (reset) begin
			wr_ptr <= 0;
			rd_ptr <= 0;
			full <= 0;
			empty <= 1;
			almost_full <= 0;
			almost_empty <= 1;
		end else begin
			if (wr_en && !full) begin
				mem[wr_ptr] <= din;
				wr_ptr <= wr_ptr + 1;
			end

			if (rd_en && !empty) begin
				dout <= mem[rd_ptr];
				rd_ptr <= rd_ptr + 1;
			end

			full <= (wr_ptr == rd_ptr + DEPTH - 1);
			empty <= (wr_ptr == rd_ptr);
			almost_full <= (wr_ptr == rd_ptr + DEPTH - 2);
			almost_empty <= (wr_ptr == rd_ptr + 1);
		end
	end

endmodule
```
