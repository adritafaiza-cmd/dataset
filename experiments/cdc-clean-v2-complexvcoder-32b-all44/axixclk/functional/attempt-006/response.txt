    output wire [3 : 0] M_AXI_ARQOS,

    reg [3:0] m_arcache_reg;

    reg [C_S_AXI_ID_WIDTH-1:0] m_rid_reg;

    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            aw_reset_sync <= {XCLOCK_FFS{1'b0}};
            w_reset_sync <= {XCLOCK_FFS{1'b0}};
            b_reset_sync <= {XCLOCK_FFS{1'b0}};
        end else begin
            aw_reset_sync <= {aw_reset_sync[XCLOCK_FFS-2:0], M_AXI_ARESETN};
            w_reset_sync <= {w_reset_sync[XCLOCK_FFS-2:0], M_AXI_ARESETN};

    end

    // Write Data Channel Cross Clock Domain
