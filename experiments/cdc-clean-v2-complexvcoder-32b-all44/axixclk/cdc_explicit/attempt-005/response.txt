module cdc_fifo #(
		parameter C_WIDTH = 32,
		parameter LGFIFO = 5,
		parameter XCLOCK_FFS = 2
	) (
		input wire write_clk,
		input wire write_rstn,
		input wire write_valid,
		output wire write_ready,
		input wire [C_WIDTH-1:0] write_data,
		input wire read_clk,
		input wire read_rstn,
		output wire read_valid,
		input wire read_ready,
		output wire [C_WIDTH-1:0] read_data
	);

	localparam DEPTH = 1 << LGFIFO;

	reg [C_WIDTH-1:0] mem [0:DEPTH-1];
	reg [$clog2(DEPTH)-1:0] wptr, rptr;
	reg [$clog2(DEPTH):0] diff;

	// Write pointer logic
	always @(posedge write_clk or negedge write_rstn) begin
		if (!write_rstn)
			wptr <= 0;
		else if (write_ready && write_valid)
			wptr <= wptr + 1;
	end

	// Read pointer logic
	always @(posedge read_clk or negedge read_rstn) begin
		if (!read_rstn)
			rptr <= 0;
		else if (read_ready && read_valid)
			rptr <= rptr + 1;
	end

	// FIFO difference calculation
	always @(posedge read_clk or negedge read_rstn) begin
		if (!read_rstn)
			diff <= 0;
		else
			diff <= {1'b0, wptr} - {1'b0, rptr};
	end

	// Write enable
	assign write_ready = diff < DEPTH;

	// Read enable
	assign read_valid = diff > 0;

	// Memory write
	always @(posedge write_clk) begin
		if (write_ready && write_valid)
			mem[wptr % DEPTH] <= write_data;
	end

	// Memory read
	always @(posedge read_clk) begin
		if (read_ready && read_valid)
			read_data <= mem[rptr % DEPTH];
	end

endmodule
