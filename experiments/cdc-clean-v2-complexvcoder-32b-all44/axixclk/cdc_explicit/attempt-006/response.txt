```verilog
module axixclk #(
		parameter integer C_S_AXI_ID_WIDTH	= 2,
		parameter integer C_S_AXI_DATA_WIDTH	= 32,
		parameter integer C_S_AXI_ADDR_WIDTH	= 6,
		parameter [0:0]	OPT_WRITE_ONLY = 1'b0,
		parameter [0:0]	OPT_READ_ONLY = 1'b0,
		parameter	XCLOCK_FFS = 2,
		parameter	LGFIFO = 5
	) (
		input	wire				S_AXI_ACLK,
		input	wire				S_AXI_ARESETN,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_AWID,
		input	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	S_AXI_AWADDR,
		input	wire [7 : 0]			S_AXI_AWLEN,
		input	wire [2 : 0]			S_AXI_AWSIZE,
		input	wire [1 : 0]			S_AXI_AWBURST,
		input	wire				S_AXI_AWLOCK,
		input	wire [3 : 0]			S_AXI_AWCACHE,
		input	wire [2 : 0]			S_AXI_AWPROT,
		input	wire [3 : 0]			S_AXI_AWQOS,
		input	wire				S_AXI_AWVALID,
		output	wire				S_AXI_AWREADY,
		input	wire [C_S_AXI_DATA_WIDTH-1 : 0]	S_AXI_WDATA,
		input	wire [(C_S_AXI_DATA_WIDTH/8)-1 : 0] S_AXI_WSTRB,
		input	wire				S_AXI_WLAST,
		input	wire				S_AXI_WVALID,
		output	wire				S_AXI_WREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_BID,
		output	wire [1 : 0]			S_AXI_BRESP,
		output	wire				S_AXI_BVALID,
		input	wire				S_AXI_BREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_ARID,
		input	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	S_AXI_ARADDR,
		input	wire [7 : 0]			S_AXI_ARLEN,
		input	wire [2 : 0]			S_AXI_ARSIZE,
		input	wire [1 : 0]			S_AXI_ARBURST,
		input	wire				S_AXI_ARLOCK,
		input	wire [3 : 0]			S_AXI_ARCACHE,
		input	wire [2 : 0]			S_AXI_ARPROT,
		input	wire [3 : 0]			S_AXI_ARQOS,
		input	wire				S_AXI_ARVALID,
		output	wire				S_AXI_ARREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_RID,
		output	wire [C_S_AXI_DATA_WIDTH-1 : 0]	S_AXI_RDATA,
		output	wire [1 : 0]			S_AXI_RRESP,
		output	wire				S_AXI_RLAST,
		output	wire				S_AXI_RVALID,
		input	wire				S_AXI_RREADY,
		input	wire				M_AXI_ACLK,
		output	wire				M_AXI_ARESETN,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_AWID,
		output	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	M_AXI_AWADDR,
		output	wire [7 : 0]			M_AXI_AWLEN,
		output	wire [2 : 0]			M_AXI_AWSIZE,
		output	wire [1 : 0]			M_AXI_AWBURST,
		output	wire				M_AXI_AWLOCK,
		output	wire [3 : 0]			M_AXI_AWCACHE,
		output	wire [2 : 0]			M_AXI_AWPROT,
		output	wire [3 : 0]			M_AXI_AWQOS,
		output	wire				M_AXI_AWVALID,
		input	wire				M_AXI_AWREADY,
		output	wire [C_S_AXI_DATA_WIDTH-1 : 0]	M_AXI_WDATA,
		output	wire [(C_S_AXI_DATA_WIDTH/8)-1 : 0] M_AXI_WSTRB,
		output	wire				M_AXI_WLAST,
		output	wire				M_AXI_WVALID,
		input	wire				M_AXI_WREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_BID,
		input	wire [1 : 0]			M_AXI_BRESP,
		input	wire				M_AXI_BVALID,
		output	wire				M_AXI_BREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_ARID,
		output	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	M_AXI_ARADDR,
		output	wire [7 : 0]			M_AXI_ARLEN,
		output	wire [2 : 0]			M_AXI_ARSIZE,
		output	wire [1 : 0]			M_AXI_ARBURST,
		output	wire				M_AXI_ARLOCK,
		output	wire [3 : 0]			M_AXI_ARCACHE,
		output	wire [2 : 0]			M_AXI_ARPROT,
		output	wire [3 : 0]			M_AXI_ARQOS,
		output	wire				M_AXI_ARVALID,
		input	wire				M_AXI_ARREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_RID,
		input	wire [C_S_AXI_DATA_WIDTH-1 : 0]	M_AXI_RDATA,
		input	wire [1 : 0]			M_AXI_RRESP,
		input	wire				M_AXI_RLAST,
		input	wire				M_AXI_RVALID,
		output	wire				M_AXI_RREADY
	);

	localparam integer FIFO_DEPTH = 1 << LGFIFO;

	// Write Address Channel
	reg [C_S_AXI_ID_WIDTH-1 : 0] awid_ff;
	reg [C_S_AXI_ADDR_WIDTH-1 : 0] awaddr_ff;
	reg [7 : 0] awlen_ff;
	reg [2 : 0] awsize_ff;
	reg [1 : 0] awburst_ff;
	reg awlock_ff;
	reg [3 : 0] awcache_ff;
	reg [2 : 0] awprot_ff;
	reg [3 : 0] awqos_ff;
	reg awvalid_ff;

	wire awready_cross;

	cdc_sync #(.WIDTH(1), .STAGES(XCLOCK_FFS)) awvalid_cross_inst (
		.clk_src(S_AXI_ACLK),
		.clk_dst(M_AXI_ACLK),
		.rst_src(~S_AXI_ARESETN),
		.rst_dst(~M_AXI_ARESETN),
		.d_src(S_AXI_AWVALID),
		.q_dst(awvalid_ff)
	);

	cdc_sync #(.WIDTH(1), .STAGES(XCLOCK_FFS)) awready_cross_inst (
		.clk_src(M_AXI_ACLK),
		.clk_dst(S_AXI_ACLK),
		.rst_src(~M_AXI_ARESETN),
		.rst_dst(~S_AXI_ARESETN),
		.d_src(M_AXI_AWREADY),
		.q_dst(awready_cross)
	);

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (~S_AXI_ARESETN) begin
			awid_ff <= 0;
			awaddr_ff <= 0;
			awlen_ff <= 0;
			awsize_ff <= 0;
			awburst_ff <= 0;
			awlock_ff <= 0;
			awcache_ff <= 0;
			awprot_ff <= 0;
			awqos_ff <= 0;
		end else if (S_AXI_AWVALID && S_AXI_AWREADY) begin
			awid_ff <= S_AXI_AWID;
			awaddr_ff <= S_AXI_AWADDR;
			awlen_ff <= S_AXI_AWLEN;
			awsize_ff <= S_AXI_AWSIZE;
			awburst_ff <= S_AXI_AWBURST;
			awlock_ff <= S_AXI_AWLOCK;
			awcache_ff <= S_AXI_AWCACHE;
			awprot_ff <= S_AXI_AWPROT;
			awqos_ff <= S_AXI_AWQOS;
		end
	end

	assign S_AXI_AWREADY = awready_cross;

	assign M_AXI_AWID = awid_ff;
	assign M_AXI_AWADDR = awaddr_ff;
	assign M_AXI_AWLEN = awlen_ff;
	assign M_AXI_AWSIZE = awsize_ff;
	assign M_AXI_AWBURST = awburst_ff;
	assign M_AXI_AWLOCK = awlock_ff;
	assign M_AXI_AWCACHE = awcache_ff;
	assign M_AXI_AWPROT = awprot_ff;
	assign M_AXI_AWQOS = awqos_ff;
	assign M_AXI_AWVALID = awvalid_ff;

	// Write Data Channel
	reg [C_S_AXI_DATA_WIDTH-1 : 0] wdata_ff;
	reg [(C_S_AXI_DATA_WIDTH/8)-1 : 0] wstrb_ff;
	reg wlast_ff;
	reg wvalid_ff;

		.clk_dst(M_AXI_ACLK),

		.rst_dst(~M_AXI_ARESETN),

		.clk_dst(S_AXI_ACLK),

			wvalid_delay <= 0;
		end else begin
			wvalid_delay <= {wvalid_delay[XCLOCK_FFS-2 : 0], wvalid_ff};
		end
	end

	assign M_AXI_WDATA = wdata_ff;
	assign M_AXI_WSTRB = wstrb_ff;
	assign M_AXI_WLAST = wlast_ff;
	assign M_AXI_WVALID = wvalid_delay[XCLOCK_FFS-1];

	// Write Response Channel

		.clk_dst(S_AXI_ACLK),
		.rst_src(~M_AXI_ARESETN),
		.rst_dst(~S_AXI_ARESETN),
		.d_src(M_AXI_BVALID),
		.q_dst(bvalid_cross)
	);

	assign M_AXI_BREADY = S_AXI_BREADY;

	assign S_AXI_BID = bid_cross;
	assign S_AXI_BRESP = bresp_cross;
	assign S_AXI_BVALID = bvalid_cross;

	// Read Address Channel
	reg [C_S_AXI_ID_WIDTH-1 : 0] arid_ff;
	reg [C_S_AXI_ADDR_WIDTH-1 : 0] araddr_ff;
	reg [7 : 0] arlen_ff;
	reg [2 : 0] arsize_ff;
	reg [1 : 0] arburst_ff;
	reg arlock_ff;
	reg [3 : 0] arcache_ff;
	reg [2 : 0] arprot_ff;
	reg [3 : 0] arqos_ff;
	reg arvalid_ff;

	wire arready_cross;

	cdc_sync #(.WIDTH(1), .STAGES(XCLOCK_FFS)) arvalid_cross_inst (
		.clk_src(S_AXI_ACLK),

module cdc_sync #(
	parameter integer WIDTH = 1,
	parameter integer STAGES = 2
) (
	input wire clk_src,
	input wire clk_dst,
	input wire rst_src,
	input wire rst_dst,
	input wire [WIDTH-1 : 0] d_src,
	output reg [WIDTH-1 : 0] q_dst
);

	reg [WIDTH-1 : 0] sync_reg [STAGES-1 : 0];

	always @(posedge clk_src or negedge rst_src) begin
		if (~rst_src) begin
			sync_reg[0] <= 0;
		end else begin
			sync_reg[0] <= d_src;
		end
	end

	generate
		if (STAGES > 1) begin
			genvar i;
			for (i = 1; i < STAGES; i = i + 1) begin : sync_stages
				always @(posedge clk_src or negedge rst_src) begin
					if (~rst_src) begin
						sync_reg[i] <= 0;
					end else begin
						sync_reg[i] <= sync_reg[i-1];
					end
				end
			end
		end
	endgenerate

	always @(posedge clk_dst or negedge rst_dst) begin
		if (~rst_dst) begin
			q_dst <= 0;
		end else begin
			q_dst <= sync_reg[STAGES-1];
		end
	end

endmodule
```
