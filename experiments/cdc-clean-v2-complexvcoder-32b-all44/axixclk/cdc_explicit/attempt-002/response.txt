// Synchronous FIFO module
module fifo_sync #(
	parameter DATA_WIDTH = 32,
	parameter LGFIFO = 4
) (
	input wire clk,
	input wire rst,
	input wire wr_en,
	input wire [DATA_WIDTH-1:0] wr_data,
	output reg wr_full,
	output reg wr_almost_full,
	input wire rd_en,
	output reg [DATA_WIDTH-1:0] rd_data,
	output reg rd_empty,
	output reg rd_almost_empty,
	output reg valid,
	output reg ready
);

	localparam FIFO_DEPTH = 1 << LGFIFO;
	reg [DATA_WIDTH-1:0] mem [0:FIFO_DEPTH-1];
	reg [$clog2(FIFO_DEPTH)-1:0] wr_ptr, rd_ptr;
	reg [$clog2(FIFO_DEPTH):0] count;

	always @(posedge clk or posedge rst) begin
		if (rst) begin
			wr_ptr <= 0;
			rd_ptr <= 0;
			count <= 0;
			wr_full <= 0;
			wr_almost_full <= 0;
			rd_empty <= 1;
			rd_almost_empty <= 1;
			valid <= 0;
			ready <= 1;
		end else begin
			if (wr_en && !wr_full) begin
				mem[wr_ptr] <= wr_data;
				wr_ptr <= wr_ptr + 1;
				count <= count + 1;
			end
			if (rd_en && !rd_empty) begin
				rd_ptr <= rd_ptr + 1;
				count <= count - 1;
			end
			wr_full <= (count == FIFO_DEPTH);
			wr_almost_full <= (count == FIFO_DEPTH - 1);
			rd_empty <= (count == 0);
			rd_almost_empty <= (count == 1);
			valid <= !rd_empty;
			ready <= !wr_full;
		end
	end

	always @(posedge clk or posedge rst) begin
		if (rst) begin
			rd_data <= 0;
		end else if (rd_en && !rd_empty) begin
			rd_data <= mem[rd_ptr];
		end
	end

endmodule

// Asynchronous FIFO module
module fifo_async #(
	parameter DATA_WIDTH = 32,
	parameter LGFIFO = 4,
	parameter XCLOCK_FFS = 2
) (
	input wire wr_clk,
	input wire wr_rst,
	input wire wr_en,
	input wire [DATA_WIDTH-1:0] wr_data,
	output reg wr_full,
	output reg wr_almost_full,
	input wire rd_clk,
	input wire rd_rst,
	input wire rd_en,
	output reg [DATA_WIDTH-1:0] rd_data,
	output reg rd_empty,
	output reg rd_almost_empty,
	output reg valid,
	output reg ready
);

	localparam FIFO_DEPTH = 1 << LGFIFO;
	reg [DATA_WIDTH-1:0] mem [0:FIFO_DEPTH-1];
	reg [$clog2(FIFO_DEPTH)-1:0] wr_ptr, rd_ptr;
	reg [$clog2(FIFO_DEPTH):0] count;
	reg [$clog2(FIFO_DEPTH)-1:0] wr_ptr_gray, rd_ptr_gray;
	reg [$clog2(FIFO_DEPTH)-1:0] wr_ptr_gray_sync, rd_ptr_gray_sync;

	// Gray code conversion
	function [$clog2(FIFO_DEPTH)-1:0] bin_to_gray;
		input [$clog2(FIFO_DEPTH)-1:0] bin;
		bin_to_gray = (bin >> 1) ^ bin;
	endfunction

	function [$clog2(FIFO_DEPTH)-1:0] gray_to_bin;
		input [$clog2(FIFO_DEPTH)-1:0] gray;
		integer i;
		gray_to_bin = gray;
		for (i = $clog2(FIFO_DEPTH)-2; i >= 0; i = i - 1)
			gray_to_bin[i] = gray_to_bin[i+1] ^ gray[i];
	endfunction

	// Write side
	always @(posedge wr_clk or posedge wr_rst) begin
		if (wr_rst) begin
			wr_ptr <= 0;
			wr_ptr_gray <= 0;
			count <= 0;
			wr_full <= 0;
			wr_almost_full <= 0;
		end else begin
			if (wr_en && !wr_full) begin
				mem[wr_ptr] <= wr_data;
				wr_ptr <= wr_ptr + 1;
				count <= count + 1;
			end
			wr_ptr_gray <= bin_to_gray(wr_ptr);
			wr_full <= (count == FIFO_DEPTH);
			wr_almost_full <= (count == FIFO_DEPTH - 1);
		end
	end

	// Read side
	always @(posedge rd_clk or posedge rd_rst) begin
		if (rd_rst) begin
			rd_ptr <= 0;
			rd_ptr_gray <= 0;
			count <= 0;
			rd_empty <= 1;
			rd_almost_empty <= 1;
			rd_ptr_gray_sync <= 0;
		end else begin
			if (rd_en && !rd_empty) begin
				rd_ptr <= rd_ptr + 1;
				count <= count - 1;
			end
			rd_ptr_gray <= bin_to_gray(rd_ptr);
			rd_empty <= (count == 0);
			rd_almost_empty <= (count == 1);
			rd_ptr_gray_sync <= wr_ptr_gray;
		end
	end

	// Cross-clock domain synchronization
	generate
		if (XCLOCK_FFS == 2) begin
			reg [$clog2(FIFO_DEPTH)-1:0] rd_ptr_gray_sync_ff1;

			always @(posedge rd_clk or posedge rd_rst) begin
				if (rd_rst) begin
					rd_ptr_gray_sync_ff1 <= 0;
				end else begin
					rd_ptr_gray_sync_ff1 <= rd_ptr_gray_sync;
				end
			end

			always @(posedge rd_clk or posedge rd_rst) begin
				if (rd_rst) begin
					rd_ptr_gray_sync <= 0;
				end else begin
					rd_ptr_gray_sync <= rd_ptr_gray_sync_ff1;
				end
			end
		end else if (XCLOCK_FFS == 3) begin
			reg [$clog2(FIFO_DEPTH)-1:0] rd_ptr_gray_sync_ff1;
			reg [$clog2(FIFO_DEPTH)-1:0] rd_ptr_gray_sync_ff2;

			always @(posedge rd_clk or posedge rd_rst) begin
				if (rd_rst) begin
					rd_ptr_gray_sync_ff1 <= 0;
				end else begin
					rd_ptr_gray_sync_ff1 <= rd_ptr_gray_sync;
				end
			end

			always @(posedge rd_clk or posedge rd_rst) begin
				if (rd_rst) begin
					rd_ptr_gray_sync_ff2 <= 0;
				end else begin
					rd_ptr_gray_sync_ff2 <= rd_ptr_gray_sync_ff1;
				end
			end

			always @(posedge rd_clk or posedge rd_rst) begin
				if (rd_rst) begin
					rd_ptr_gray_sync <= 0;
				end else begin
					rd_ptr_gray_sync <= rd_ptr_gray_sync_ff2;
				end
			end
		end else begin
			always @(posedge rd_clk or posedge rd_rst) begin
				if (rd_rst) begin
					rd_ptr_gray_sync <= 0;
				end else begin
					rd_ptr_gray_sync <= rd_ptr_gray_sync;
				end
			end
		end
	endgenerate

	always @(posedge rd_clk or posedge rd_rst) begin
		if (rd_rst) begin
			rd_data <= 0;
			valid <= 0;
			ready <= 1;
		end else if (rd_en && !rd_empty) begin
			rd_data <= mem[rd_ptr];
			valid <= 1;
			ready <= 1;
		end else begin
			valid <= 0;
			ready <= ~wr_full;
		end
	end

endmodule
