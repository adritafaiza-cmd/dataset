module cdc_fifo #(
		parameter integer WIDTH = 32,
		parameter integer DEPTH = 5,
		parameter integer XCLOCK_FFS = 2
	) (
		input	wire				src_clk,
		input	wire				src_rst,
		input	wire [WIDTH-1 : 0]		src_data,
		input	wire				src_valid,
		output	wire				src_ready,
		input	wire				dst_clk,
		input	wire				dst_rst,
		output	wire [WIDTH-1 : 0]		dst_data,
		output	wire				dst_valid,
		input	wire				dst_ready
	);

	// FIFO signals
	reg [WIDTH-1 : 0] fifo_mem [0 : (1 << DEPTH) - 1];
	reg [$clog2((1 << DEPTH)) - 1 : 0] wr_ptr, rd_ptr;
	reg [$clog2((1 << DEPTH)) : 0] fifo_count;

	// Gray code pointers for CDC
	function [XCLOCK_FFS-1 : 0] bin2gray;
		input [$clog2((1 << DEPTH)) - 1 : 0] bin;
		begin
			bin2gray = bin ^ (bin >> 1);
		end
	endfunction

	// Source clock domain
	reg [$clog2((1 << DEPTH)) - 1 : 0] wr_ptr_gray;
	reg [$clog2((1 << DEPTH)) - 1 : 0] rd_ptr_gray_sync;

	assign src_ready = (fifo_count < (1 << DEPTH));

	always @(posedge src_clk or posedge src_rst) begin
		if (src_rst) begin
			wr_ptr <= 0;
			wr_ptr_gray <= 0;
		end else if (src_valid && src_ready) begin
			fifo_mem[wr_ptr] <= src_data;
			wr_ptr <= wr_ptr + 1;
			wr_ptr_gray <= bin2gray(wr_ptr);
		end
	end

	always @(posedge dst_clk or posedge dst_rst) begin
		if (dst_rst)
			rd_ptr_gray_sync <= 0;
		else
			rd_ptr_gray_sync <= wr_ptr_gray;
	end

	// Destination clock domain
	reg [$clog2((1 << DEPTH)) - 1 : 0] rd_ptr_gray;
	reg [$clog2((1 << DEPTH)) - 1 : 0] wr_ptr_gray_sync;

	assign dst_valid = (fifo_count != 0);

	always @(posedge dst_clk or posedge dst_rst) begin
		if (dst_rst) begin
			rd_ptr <= 0;
			rd_ptr_gray <= 0;
		end else if (dst_valid && dst_ready) begin
			rd_ptr <= rd_ptr + 1;
			rd_ptr_gray <= bin2gray(rd_ptr);
		end
	end

	always @(posedge src_clk or posedge src_rst) begin
		if (src_rst)
			wr_ptr_gray_sync <= 0;
		else
			wr_ptr_gray_sync <= rd_ptr_gray;
	end

	// FIFO count
	always @(posedge src_clk or posedge src_rst) begin
		if (src_rst)
			fifo_count <= 0;
		else begin
			if (src_valid && src_ready && !(dst_valid && dst_ready))
				fifo_count <= fifo_count + 1;
			else if (!(src_valid && src_ready) && dst_valid && dst_ready)
				fifo_count <= fifo_count - 1;
		end
	end

	// Output data
	always @(posedge dst_clk or posedge dst_rst) begin
		if (dst_rst)
			dst_data <= 0;
		else if (dst_valid && dst_ready)
			dst_data <= fifo_mem[rd_ptr];
	end

endmodule
