=== STAGE 2: VERILOG ===
```verilog
module synchronizer #(
    parameter STAGES = 2
)(
    input  clk,
    input  rstn,
    input  async_sig_i,
    output sync_sig_o
);

    reg [STAGES-1:0] sync_stages;

    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            sync_stages <= {STAGES{1'b0}};
        end else begin
            sync_stages[0] <= async_sig_i;
            sync_stages[STAGES-1:1] <= sync_stages[STAGES-2:0];
        end
    end

    assign sync_sig_o = sync_stages[STAGES-1];

endmodule
```
