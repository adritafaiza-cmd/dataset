=== STAGE 2: VERILOG ===
```verilog
module async_bidir_ramif_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "FALSE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir,
       output wire             o_ram_a_clk,
       output wire [DSIZE-1:0] o_ram_a_wdata,
       input wire [DSIZE-1:0]  i_ram_a_rdata,
       output wire [ASIZE-1:0] o_ram_a_addr,
       output wire             o_ram_a_rinc,
       output wire             o_ram_a_winc,
       output wire             o_ram_b_clk,
       output wire [DSIZE-1:0] o_ram_b_wdata,
       input wire [DSIZE-1:0]  i_ram_b_rdata,
       output wire [ASIZE-1:0] o_ram_b_addr,
       output wire             o_ram_b_rinc,
       output wire             o_ram_b_winc
       );

    // Internal signals
    reg [ASIZE-1:0] a_wr_ptr, a_rd_ptr, b_wr_ptr, b_rd_ptr;
    reg [ASIZE-1:0] a_wr_ptr_gray, a_rd_ptr_gray, b_wr_ptr_gray, b_rd_ptr_gray;
    reg [ASIZE-1:0] a_wr_ptr_gray_sync, a_rd_ptr_gray_sync, b_wr_ptr_gray_sync, b_rd_ptr_gray_sync;
    wire [ASIZE-1:0] a_wr_ptr_gray_next, a_rd_ptr_gray_next, b_wr_ptr_gray_next, b_rd_ptr_gray_next;
    reg a_full_flag, a_afull_flag, a_empty_flag, a_aempty_flag;
    reg b_full_flag, b_afull_flag, b_empty_flag, b_aempty_flag;
    reg a_ram_we, b_ram_we;

    // Gray code conversion function
    function [ASIZE-1:0] bin2gray;
        input [ASIZE-1:0] bin;
        bin2gray = bin ^ (bin >> 1);
    endfunction

    // a_clk domain logic
    always @(posedge a_clk or negedge a_rst_n) begin
        if (!a_rst_n) begin
            a_wr_ptr <= 0;
            a_rd_ptr <= 0;
            a_wr_ptr_gray <= 0;
            a_rd_ptr_gray <= 0;
            a_wr_ptr_gray_sync <= 0;
            a_rd_ptr_gray_sync <= 0;
            a_full_flag <= 0;
            a_afull_flag <= 0;
            a_empty_flag <= 1;
            a_aempty_flag <= 1;
        end else begin
            a_wr_ptr_gray_next = bin2gray(a_wr_ptr + a_winc & ~a_full_flag);
            a_rd_ptr_gray_next = bin2gray(a_rd_ptr + a_rinc & ~a_empty_flag);

            // Synchronize gray-coded pointers
            a_wr_ptr_gray_sync <= b_rd_ptr_gray;
            a_rd_ptr_gray_sync <= b_wr_ptr_gray;

            // Update pointers
            if (a_winc & ~a_full_flag) a_wr_ptr <= a_wr_ptr + 1;
            if (a_rinc & ~a_empty_flag) a_rd_ptr <= a_rd_ptr + 1;

            // Update gray-coded pointers
            a_wr_ptr_gray <= a_wr_ptr_gray_next;
            a_rd_ptr_gray <= a_rd_ptr_gray_next;

            // Update flags
            a_full_flag <= (a_wr_ptr_gray_next == {1'b1, {ASIZE-2{1'b0}}} & a_rd_ptr_gray_sync);
            a_afull_flag <= (a_wr_ptr_gray_next == {2'b11, {ASIZE-3{1'b0}}} & a_rd_ptr_gray_sync);
            a_empty_flag <= (a_wr_ptr_gray == a_rd_ptr_gray_sync);
            a_aempty_flag <= (a_wr_ptr_gray == {1'b0, {ASIZE-2{1'b0}}} & a_rd_ptr_gray_sync);
        end
    end

    // b_clk domain logic
    always @(posedge b_clk or negedge b_rst_n) begin
        if (!b_rst_n) begin
            b_wr_ptr <= 0;
            b_rd_ptr <= 0;
            b_wr_ptr_gray <= 0;
            b_rd_ptr_gray <= 0;
            b_wr_ptr_gray_sync <= 0;
            b_rd_ptr_gray_sync <= 0;
            b_full_flag <= 0;
            b_afull_flag <= 0;
            b_empty_flag <= 1;
            b_aempty_flag <= 1;
        end else begin
            b_wr_ptr_gray_next = bin2gray(b_wr_ptr + b_winc & ~b_full_flag);
            b_rd_ptr_gray_next = bin2gray(b_rd_ptr + b_rinc & ~b_empty_flag);

            // Synchronize gray-coded pointers
            b_wr_ptr_gray_sync <= a_rd_ptr_gray;
            b_rd_ptr_gray_sync <= a_wr_ptr_gray;

            // Update pointers
            if (b_winc & ~b_full_flag) b_wr_ptr <= b_wr_ptr + 1;
            if (b_rinc & ~b_empty_flag) b_rd_ptr <= b_rd_ptr + 1;

            // Update gray-coded pointers
            b_wr_ptr_gray <= b_wr_ptr_gray_next;
            b_rd_ptr_gray <= b_rd_ptr_gray_next;

            // Update flags
            b_full_flag <= (b_wr_ptr_gray_next == {1'b1, {ASIZE-2{1'b0}}} & b_rd_ptr_gray_sync);
            b_afull_flag <= (b_wr_ptr_gray_next == {2'b11, {ASIZE-3{1'b0}}} & b_rd_ptr_gray_sync);
            b_empty_flag <= (b_wr_ptr_gray == b_rd_ptr_gray_sync);
            b_aempty_flag <= (b_wr_ptr_gray == {1'b0, {ASIZE-2{1'b0}}} & b_rd_ptr_gray_sync);
        end
    end

    // RAM write enables
    assign a_ram_we = a_winc & ~a_full_flag;
    assign b_ram_we = b_winc & ~b_full_flag;

    // RAM addresses and write data
    assign o_ram_a_addr = a_wr_ptr;
    assign o_ram_a_wdata = a_wdata;
    assign o_ram_b_addr = b_wr_ptr;
    assign o_ram_b_wdata = b_wdata;

    // RAM read enables and FIFO read data
    assign o_ram_a_rinc = a_rinc & ~a_empty_flag;
    assign o_ram_b_rinc = b_rinc & ~b_empty_flag;
    assign a_rdata = i_ram_a_rdata;
    assign b_rdata = i_ram_b_rdata;

    // FIFO flags
    assign a_full = a_full_flag;
    assign a_afull = a_afull_flag;
    assign a_empty = a_empty_flag;
    assign a_aempty = a_aempty_flag;
    assign b_full = b_full_flag;
    assign b_afull = b_afull_flag;
    assign b_empty = b_empty_flag;
    assign b_aempty = b_aempty_flag;

    // RAM clock outputs
    assign o_ram_a_clk = a_clk;
    assign o_ram_b_clk = b_clk;

endmodule
```
