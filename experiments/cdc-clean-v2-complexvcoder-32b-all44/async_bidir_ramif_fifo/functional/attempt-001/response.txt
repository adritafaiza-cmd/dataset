module async_fifo_side
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "FALSE"
    ) (
       input wire              clk,
       input wire              rst_n,
       input wire              winc,
       input wire [DSIZE-1:0]  wdata,
       input wire              rinc,
       input wire              dir,
       input wire [ASIZE-1:0]  rd_addr,
       output wire [ASIZE-1:0] wr_addr,
       output wire [ASIZE-1:0] wr_addr_synced,
       output wire [DSIZE-1:0] rdata,
       input wire [DSIZE-1:0]  rdata_from_ram,
       output wire             full,
       output wire             afull,
       output wire             empty,
       output wire             aempty,
       input wire              other_clk,
       input wire [ASIZE-1:0]  other_wr_addr
       );

  reg [ASIZE-1:0] wr_ptr;
  reg [ASIZE-1:0] rd_ptr;
  reg [ASIZE-1:0] rd_ptr_synced;
  reg full_flag;
  reg afull_flag;
  reg empty_flag;
  reg aempty_flag;

  binary_sync #(.ASIZE(ASIZE))
  wr_addr_sync (
    .clk(clk),
    .rst_n(rst_n),
    .data_in(wr_ptr),
    .data_out(wr_addr_synced),
    .other_clk(other_clk)
  );

  binary_sync #(.ASIZE(ASIZE))
  rd_addr_sync (
    .clk(clk),
    .rst_n(rst_n),
    .data_in(other_wr_addr),
    .data_out(rd_ptr_synced),
    .other_clk(other_clk)
  );

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      wr_ptr <= 0;
      rd_ptr <= 0;
      full_flag <= 0;
      afull_flag <= 0;
      empty_flag <= 1;
      aempty_flag <= 1;
    end else begin
      if (winc && !full_flag) begin
        wr_ptr <= wr_ptr + 1;
        if (wr_ptr + 1 == rd_ptr_synced)
          full_flag <= 1;
        else
          full_flag <= 0;

        if (wr_ptr + 1 == rd_ptr_synced + 1)
          afull_flag <= 1;
        else
          afull_flag <= 0;
      end

      if (rinc && !empty_flag) begin
        rd_ptr <= rd_ptr + 1;
        if (wr_ptr == rd_ptr + 1)
          empty_flag <= 1;
        else
          empty_flag <= 0;

        if (wr_ptr == rd_ptr)
          aempty_flag <= 1;
        else
          aempty_flag <= 0;
      end
    end
  end

  assign wr_addr = wr_ptr;
  assign rdata = rdata_from_ram;
  assign full = full_flag;
  assign afull = afull_flag;
  assign empty = empty_flag;
  assign aempty = aempty_flag;

endmodule

module binary_sync
  #(
    parameter ASIZE = 4
    ) (
       input wire              clk,
       input wire              rst_n,
       input wire [ASIZE-1:0]  data_in,
       output wire [ASIZE-1:0] data_out,
       input wire              other_clk
       );

  reg [ASIZE-1:0] sync_reg1;
  reg [ASIZE-1:0] sync_reg2;

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      sync_reg1 <= 0;
      sync_reg2 <= 0;
    end else begin
      sync_reg1 <= data_in;
      sync_reg2 <= sync_reg1;
    end
  end

  always @(posedge other_clk or negedge rst_n) begin
    if (!rst_n) begin
      data_out <= 0;
    end else begin
      data_out <= sync_reg2;
    end
  end

endmodule
