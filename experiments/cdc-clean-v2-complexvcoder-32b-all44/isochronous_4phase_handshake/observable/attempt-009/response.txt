=== STAGE 2: VERILOG ===
```verilog
module isochronous_4phase_handshake (
    input  src_clk_i,
    input  src_rst_ni,
    input  src_valid_i,
    output src_ready_o,
    input  dst_clk_i,
    input  dst_rst_ni,
    output dst_valid_o,
    input  dst_ready_i
);

    // Internal signals
    reg [1:0] src_handshake_phase;
    reg [1:0] dst_handshake_phase;
    reg [1:0] src_handshake_phase_sync;
    reg [1:0] dst_handshake_phase_sync;

    // Synchronize src_handshake_phase to dst_clk_i domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_handshake_phase_sync <= 2'b00;
        end else begin
            src_handshake_phase_sync <= src_handshake_phase;
        end
    end

    // Synchronize dst_handshake_phase to src_clk_i domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_handshake_phase_sync <= 2'b00;
        end else begin
            dst_handshake_phase_sync <= dst_handshake_phase;
        end
    end

    // Generate src_ready_o based on the current src_handshake_phase and dst_handshake_phase_sync
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_ready_o <= 1'b0;
        end else begin
            case (src_handshake_phase)
                2'b00: src_ready_o <= (dst_handshake_phase_sync == 2'b10);
                2'b01: src_ready_o <= (dst_handshake_phase_sync == 2'b11);
                2'b10: src_ready_o <= (dst_handshake_phase_sync == 2'b00);
                2'b11: src_ready_o <= (dst_handshake_phase_sync == 2'b01);
                default: src_ready_o <= 1'b0;
            endcase
        end
    end

    // Generate dst_valid_o based on the current dst_handshake_phase and src_handshake_phase_sync
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_valid_o <= 1'b0;
        end else begin
            case (dst_handshake_phase)
                2'b00: dst_valid_o <= (src_handshake_phase_sync == 2'b10);
                2'b01: dst_valid_o <= (src_handshake_phase_sync == 2'b11);
                2'b10: dst_valid_o <= (src_handshake_phase_sync == 2'b00);
                2'b11: dst_valid_o <= (src_handshake_phase_sync == 2'b01);
                default: dst_valid_o <= 1'b0;
            endcase
        end
    end

    // Update src_handshake_phase on src_clk_i based on src_valid_i and src_ready_o
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_handshake_phase <= 2'b00;
        end else if (src_valid_i && src_ready_o) begin
            src_handshake_phase <= src_handshake_phase + 1'b1;
        end
    end

    // Update dst_handshake_phase on dst_clk_i based on dst_ready_i and dst_valid_o
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_handshake_phase <= 2'b00;
        end else if (dst_ready_i && dst_valid_o) begin
            dst_handshake_phase <= dst_handshake_phase + 1'b1;
        end
    end

endmodule
```
