=== STAGE 2: VERILOG ===
```verilog
module afifo #(
    parameter LGFIFO = 3,
    parameter WIDTH = 16,
    parameter NFF = 2,
    parameter [0:0] WRITE_ON_POSEDGE = 1'b1,
    parameter [0:0] OPT_REGISTER_READS = 1'b1
) (
    input wire i_wclk, i_wr_reset_n, i_wr,
    input wire [WIDTH-1:0] i_wr_data,
    output reg o_wr_full,
    input wire i_rclk, i_rd_reset_n, i_rd,
    output reg [WIDTH-1:0] o_rd_data,
    output reg o_rd_empty
);

    // Internal signals
    reg [LGFIFO-1:0] write_count, read_count;
    reg [LGFIFO-1:0] write_ptr_sync, read_ptr_sync;
    reg [WIDTH-1:0] rd_data_sync;
    reg [LGFIFO:0] full_flag, empty_flag;

    // Write side
    async_counter #(.WIDTH(LGFIFO)) write_counter (
        .clk(i_wclk),
        .reset_n(i_wr_reset_n),
        .enable(i_wr & ~o_wr_full),
        .count(write_count)
    );

    // Read side
    async_counter #(.WIDTH(LGFIFO)) read_counter (
        .clk(i_rclk),
        .reset_n(i_rd_reset_n),
        .enable(i_rd & ~o_rd_empty),
        .count(read_count)
    );

    // Memory
    fifo_memory #(.WIDTH(WIDTH), .LGFIFO(LGFIFO)) memory (
        .wclk(i_wclk),
        .wr_reset_n(i_wr_reset_n),
        .wr_en(i_wr & ~o_wr_full),
        .wr_addr(write_count),
        .wr_data(i_wr_data),
        .wr_addr_sync(read_ptr_sync),
        .rclk(i_rclk),
        .rd_reset_n(i_rd_reset_n),
        .rd_addr(read_count),
        .rd_data(rd_data_sync),
        .rd_addr_sync(write_ptr_sync)
    );

    // Synchronizers
    synchronizer #(.WIDTH(LGFIFO), .NFF(NFF)) write_pointer_sync (
        .clk(i_rclk),
        .data_in(write_count),
        .data_out(write_ptr_sync)
    );

    synchronizer #(.WIDTH(LGFIFO), .NFF(NFF)) read_pointer_sync (
        .clk(i_wclk),
        .data_in(read_count),
        .data_out(read_ptr_sync)
    );

    // Full and empty flags
    assign full_flag = {1'b0, write_ptr_sync} - {1'b0, read_count} >= (1 << LGFIFO);
    assign empty_flag = {1'b0, read_ptr_sync} - {1'b0, write_ptr_sync} == 0;

    // Output full and empty flags
    always @(posedge i_wclk or negedge i_wr_reset_n) begin
        if (!i_wr_reset_n)
            o_wr_full <= 1'b0;
        else
            o_wr_full <= full_flag[LGFIFO];
    end

    always @(posedge i_rclk or negedge i_rd_reset_n) begin
        if (!i_rd_reset_n)
            o_rd_empty <= 1'b1;
        else
            o_rd_empty <= empty_flag[LGFIFO];
    end

    // Register read data for output
    register #(.WIDTH(WIDTH)) rd_data_reg (
        .clk(i_rclk),
        .reset_n(i_rd_reset_n),
        .data_in(rd_data_sync),
        .data_out(o_rd_data)
    );

endmodule

module async_counter #(
    parameter WIDTH = 3
) (
    input wire clk,
    input wire reset_n,
    input wire enable,
    output reg [WIDTH-1:0] count
);

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n)
            count <= {WIDTH{1'b0}};
        else if (enable)
            count <= count + 1;
    end

endmodule

module fifo_memory #(
    parameter WIDTH = 16,
    parameter LGFIFO = 3
) (
    input wire wclk,
    input wire wr_reset_n,
    input wire wr_en,
    input wire [LGFIFO-1:0] wr_addr,
    input wire [WIDTH-1:0] wr_data,
    input wire [LGFIFO-1:0] wr_addr_sync,
    input wire rclk,
    input wire rd_reset_n,
    input wire [LGFIFO-1:0] rd_addr,
    output reg [WIDTH-1:0] rd_data,
    input wire [LGFIFO-1:0] rd_addr_sync
);

    reg [WIDTH-1:0] memory [0:(1 << LGFIFO) - 1];

    always @(posedge wclk or negedge wr_reset_n) begin
        if (!wr_reset_n) begin
            for (integer i = 0; i < (1 << LGFIFO); i = i + 1)
                memory[i] <= {WIDTH{1'b0}};
        end else if (wr_en) begin
            memory[wr_addr] <= wr_data;
        end
    end

    always @(posedge rclk or negedge rd_reset_n) begin
        if (!rd_reset_n)
            rd_data <= {WIDTH{1'b0}};
        else
            rd_data <= memory[rd_addr];
    end

endmodule

module synchronizer #(
    parameter WIDTH = 3,
    parameter NFF = 2
) (
    input wire clk,
    input wire [WIDTH-1:0] data_in,
    output reg [WIDTH-1:0] data_out
);

    reg [WIDTH-1:0] sync_stages [0:NFF-1];

    always @(posedge clk) begin
        sync_stages[0] <= data_in;
        for (integer i = 1; i < NFF; i = i + 1)
            sync_stages[i] <= sync_stages[i-1];
    end

    assign data_out = sync_stages[NFF-1];

endmodule

module register #(
    parameter WIDTH = 16
) (
    input wire clk,
    input wire reset_n,
    input wire [WIDTH-1:0] data_in,
    output reg [WIDTH-1:0] data_out
);

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n)
            data_out <= {WIDTH{1'b0}};
        else
            data_out <= data_in;
    end

endmodule
```
