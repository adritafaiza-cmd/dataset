=== STAGE 2: VERILOG ===
```verilog
module axidma #(
        parameter   C_AXI_ID_WIDTH = 1,
        parameter   C_AXI_ADDR_WIDTH = 32,
        parameter   C_AXI_DATA_WIDTH = 32,
        localparam  C_AXIL_ADDR_WIDTH = 5,
        localparam  C_AXIL_DATA_WIDTH = 32,
        parameter [0:0]   OPT_UNALIGNED = 1'b1,
        parameter [0:0]   OPT_WRAPMEM = 1'b1,
        parameter   LGFIFO = LGMAXBURST+1,
        parameter   LGLEN = C_AXI_ADDR_WIDTH,
        parameter [0:0]   OPT_LOWPOWER = 1'b0,
        parameter [0:0]   OPT_CLKGATE = OPT_LOWPOWER,
        parameter   [C_AXI_ID_WIDTH-1:0]  AXI_READ_ID = 0,
        parameter   [C_AXI_ID_WIDTH-1:0]  AXI_WRITE_ID = 0,
        parameter   [7:0]             ABORT_KEY  = 8'h6d,
        localparam  ADDRLSB= $clog2(C_AXI_DATA_WIDTH)-3,
        localparam  AXILLSB= $clog2(C_AXIL_DATA_WIDTH)-3,
        localparam  LGLENW= LGLEN-ADDRLSB
    ) (
        input   wire    S_AXI_ACLK,
        input   wire    S_AXI_ARESETN,
        input   wire                S_AXIL_AWVALID,
        output  wire                S_AXIL_AWREADY,
        input   wire [C_AXIL_ADDR_WIDTH-1:0]  S_AXIL_AWADDR,
        input   wire  [2:0]           S_AXIL_AWPROT,
        input   wire                S_AXIL_WVALID,
        output  wire                S_AXIL_WREADY,
        input   wire [C_AXIL_DATA_WIDTH-1:0]  S_AXIL_WDATA,
        input   wire [C_AXIL_DATA_WIDTH/8-1:0]  S_AXIL_WSTRB,
        output  reg                 S_AXIL_BVALID,
        input   wire                S_AXIL_BREADY,
        output  wire    [1:0]           S_AXIL_BRESP,
        input   wire                S_AXIL_ARVALID,
        output  wire                S_AXIL_ARREADY,
        input   wire [C_AXIL_ADDR_WIDTH-1:0]  S_AXIL_ARADDR,
        input   wire  [2:0]           S_AXIL_ARPROT,
        output  reg                 S_AXIL_RVALID,
        input   wire                S_AXIL_RREADY,
        output  reg [C_AXIL_DATA_WIDTH-1:0]  S_AXIL_RDATA,
        output  wire    [1:0]           S_AXIL_RRESP,
        output  reg                 M_AXI_AWVALID,
        input   wire                M_AXI_AWREADY,
        output  reg [C_AXI_ID_WIDTH-1:0]  M_AXI_AWID,
        output  reg [C_AXI_ADDR_WIDTH-1:0]  M_AXI_AWADDR,
        output  reg [2:0]           M_AXI_AWSIZE,
        output  reg [1:0]           M_AXI_AWBURST,
        output  reg                 M_AXI_AWLOCK,
        output  reg [3:0]           M_AXI_AWCACHE,
        output  reg [2:0]           M_AXI_AWPROT,
        output  reg [3:0]           M_AXI_AWQOS,
        output  reg                 M_AXI_WVALID,
        input   wire                M_AXI_WREADY,
        output  reg [C_AXI_DATA_WIDTH-1:0]  M_AXI_WDATA,
        output  reg [C_AXI_DATA_WIDTH/8-1:0]  M_AXI_WSTRB,
        output  reg                 M_AXI_WLAST,
        input   wire                M_AXI_BVALID,
        output  reg                 M_AXI_BREADY,
        input   wire [C_AXI_ID_WIDTH-1:0]  M_AXI_BID,
        input   wire [1:0]            M_AXI_BRESP,
        output  reg                 M_AXI_ARVALID,
        input   wire                M_AXI_ARREADY,
        output  wire [C_AXI_ID_WIDTH-1:0]  M_AXI_ARID,
        output  reg [C_AXI_ADDR_WIDTH-1:0]  M_AXI_ARADDR,
        output  wire [2:0]            M_AXI_ARSIZE,
        output  wire [1:0]            M_AXI_ARBURST,
        output  wire                M_AXI_ARLOCK,
        output  wire [3:0]            M_AXI_ARCACHE,
        output  wire [2:0]            M_AXI_ARPROT,
        output  wire [3:0]            M_AXI_ARQOS,
        input   wire                M_AXI_RVALID,
        output  wire                M_AXI_RREADY,
        input   wire [C_AXI_ID_WIDTH-1:0]  M_AXI_RID,
        input   wire [C_AXI_DATA_WIDTH-1:0]  M_AXI_RDATA,
        input   wire                M_AXI_RLAST,
        input   wire [1:0]            M_AXI_RRESP,
        output  reg                 o_int
    );

    // Internal signals
    reg [C_AXIL_ADDR_WIDTH-1:0]     axil_awaddr_reg;
    reg [C_AXIL_DATA_WIDTH-1:0]     axil_wdata_reg;
    reg [C_AXIL_DATA_WIDTH/8-1:0]   axil_wstrb_reg;
    reg [C_AXI_ADDR_WIDTH-1:0]      src_addr_reg;
    reg [C_AXI_ADDR_WIDTH-1:0]      dst_addr_reg;
    reg [LGLEN-1:0]                 length_reg;
    reg [1:0]                       state;
    reg                             start_copy;
    reg                             copy_done;
    reg [LGLEN-1:0]                 remaining_length;

    // State definitions
    localparam IDLE = 2'b00;
    localparam READ = 2'b01;
    localparam WRITE = 2'b10;

    // AXI-Lite control register offsets
    localparam AXIL_REG_SRC_ADDR = 5'h00;
    localparam AXIL_REG_DST_ADDR = 5'h01;
    localparam AXIL_REG_LENGTH = 5'h02;
    localparam AXIL_REG_CONTROL = 5'h03;
    localparam AXIL_REG_STATUS = 5'h04;

    // AXI-Lite control register fields
    localparam CONTROL_START = 1'b1;

    // AXI-Lite control register signals
    reg [C_AXIL_DATA_WIDTH-1:0]     axil_reg_rdata;
    reg                             axil_reg_awready;
    reg                             axil_reg_wready;
    reg                             axil_reg_bvalid;
    reg                             axil_reg_arready;
    reg                             axil_reg_rvalid;

    // AXI-Lite control register logic
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axil_reg_awready <= 1'b0;
            axil_reg_wready <= 1'b0;
            axil_reg_bvalid <= 1'b0;
            axil_reg_arready <= 1'b0;
            axil_reg_rvalid <= 1'b0;
        end else begin
            axil_reg_awready <= S_AXIL_AWVALID;
            axil_reg_wready <= S_AXIL_WVALID & axil_reg_awready;
            axil_reg_bvalid <= axil_reg_wready & S_AXIL_WVALID & S_AXIL_BREADY;
            axil_reg_arready <= S_AXIL_ARVALID;
            axil_reg_rvalid <= axil_reg_arready & S_AXIL_ARVALID & S_AXIL_RREADY;
        end
    end

    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axil_awaddr_reg <= {C_AXIL_ADDR_WIDTH{1'b0}};
            axil_wdata_reg <= {C_AXIL_DATA_WIDTH{1'b0}};
            axil_wstrb_reg <= {C_AXIL_DATA_WIDTH/8{1'b0}};
        end else if (axil_reg_awready & S_AXIL_AWVALID) begin
            axil_awaddr_reg <= S_AXIL_AWADDR;
        end else if (axil_reg_wready & S_AXIL_WVALID) begin
            axil_wdata_reg <= S_AXIL_WDATA;
            axil_wstrb_reg <= S_AXIL_WSTRB;
        end
    end

    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            S_AXIL_BRESP <= 2'b00;
            S_AXIL_RRESP <= 2'b00;
        end else if (axil_reg_bvalid) begin
            S_AXIL_BRESP <= 2'b00;
        end else if (axil_reg_rvalid) begin
            S_AXIL_RRESP <= 2'b00;
        end
    end

    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axil_reg_rdata <= {C_AXIL_DATA_WIDTH{1'b0}};
        end else if (axil_reg_arready & S_AXIL_ARVALID) begin
            case (S_AXIL_ARADDR[4:0])
                AXIL_REG_SRC_ADDR: axil_reg_rdata <= src_addr_reg;
                AXIL_REG_DST_ADDR: axil_reg_rdata <= dst_addr_reg;
                AXIL_REG_LENGTH: axil_reg_rdata <= length_reg;
                AXIL_REG_CONTROL: axil_reg_rdata <= {C_AXIL_DATA_WIDTH-1{1'b0}};
                AXIL_REG_STATUS: axil_reg_rdata <= {C_AXIL_DATA_WIDTH-2{1'b0}, copy_done};
                default: axil_reg_rdata <= {C_AXIL_DATA_WIDTH{1'b0}};
            endcase
        end
    end

    // Control logic
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            state <= IDLE;
            start_copy <= 1'b0;
            copy_done <= 1'b0;
        end else begin
            case (state)
                IDLE: begin
                    if (axil_reg_wready & S_AXIL_WVALID & (axil_awaddr_reg[4:0] == AXIL_REG_CONTROL)) begin
                        if (axil_wdata_reg[0] == CONTROL_START) begin
                            start_copy <= 1'b1;
                        end
                    end else begin
                        start_copy <= 1'b0;
                    end
                    if (start_copy) begin
                        src_addr_reg <= axil_wdata_reg;
                        dst_addr_reg <= axil_wdata_reg;
                        length_reg <= axil_wdata_reg;
                        remaining_length <= axil_wdata_reg;
                        state <= READ;
                    end
                end
                READ: begin
                    if (M_AXI_RVALID & M_AXI_RREADY & M_AXI_RLAST) begin
                        state <= WRITE;
                    end
                end
                WRITE: begin
                    if (M_AXI_BVALID & M_AXI_BREADY & M_AXI_BRESP == 2'b00) begin
                        if (remaining_length == 0) begin
                            copy_done <= 1'b1;
                            state <= IDLE;
                        end else begin
                            remaining_length <= remaining_length - C_AXI_DATA_WIDTH/8;
                            state <= READ;
                        end
                    end
                end
                default: state <= IDLE;
            endcase
        end
    end

    // AXI Master signals
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            M_AXI_AWVALID <= 1'b0;
            M_AXI_AWADDR <= {C_AXI_ADDR_WIDTH{1'b0}};
            M_AXI_AWSIZE <= 3'b010; // 4 bytes
            M_AXI_AWBURST <= 2'b01; // INCR
            M_AXI_AWLOCK <= 1'b0;
            M_AXI_AWCACHE <= 4'b0011; // Non-cacheable, non-bufferable, no allocate
            M_AXI_AWPROT <= 3'b000; // Unprivileged, secure, data
            M_AXI_AWQOS <= 4'b0000;
            M_AXI_WVALID <= 1'b0;
            M_AXI_WDATA <= {C_AXI_DATA_WIDTH{1'b0}};
            M_AXI_WSTRB <= {C_AXI_DATA_WIDTH/8{1'b1}};
            M_AXI_WLAST <= 1'b0;
            M_AXI_BREADY <= 1'b0;
            M_AXI_ARVALID <= 1'b0;
            M_AXI_ARADDR <= {C_AXI_ADDR_WIDTH{1'b0}};
            M_AXI_ARSIZE <= 3'b010; // 4 bytes
            M_AXI_ARBURST <= 2'b01; // INCR
            M_AXI_ARLOCK <= 1'b0;
            M_AXI_ARCACHE <= 4'b0011; // Non-cacheable, non-bufferable, no allocate
            M_AXI_ARPROT <= 3'b000; // Unprivileged, secure, data
            M_AXI_ARQOS <= 4'b0000;
            M_AXI_RREADY <= 1'b0;
        end else begin
            case (state)
                IDLE: begin
                    M_AXI_AWVALID <= 1'b0;
                    M_AXI_WVALID <= 1'b0;
                    M_AXI_BREADY <= 1'b0;
                    M_AXI_ARVALID <= 1'b0;
                    M_AXI_RREADY <= 1'b0;
                end
                READ: begin
                    M_AXI_ARVALID <= 1'b1;
                    M_AXI_ARADDR <= src_addr_reg;
                    M_AXI_ARID <= AXI_READ_ID;
                    if (M_AXI_ARREADY) begin
                        M_AXI_ARVALID <= 1'b0;
                        M_AXI_RREADY <= 1'b1;
                    end
                end
                WRITE: begin
                    M_AXI_AWVALID <= 1'b1;
                    M_AXI_AWADDR <= dst_addr_reg;
                    M_AXI_AWID <= AXI_WRITE_ID;
                    if (M_AXI_AWREADY) begin
                        M_AXI_AWVALID <= 1'b0;
                        M_AXI_WVALID <= 1'b1;
                        M_AXI_WDATA <= M_AXI_RDATA;
                        M_AXI_WSTRB <= {C_AXI_DATA_WIDTH/8{1'b1}};
                        M_AXI_WLAST <= (remaining_length == C_AXI_DATA_WIDTH/8);
                        if (M_AXI_WREADY) begin
                            M_AXI_WVALID <= 1'b0;
                            M_AXI_BREADY <= 1'b1;
                        end
                    end
                end
                default: begin
                    M_AXI_AWVALID <= 1'b0;
                    M_AXI_WVALID <= 1'b0;
                    M_AXI_BREADY <= 1'b0;
                    M_AXI_ARVALID <= 1'b0;
                    M_AXI_RREADY <= 1'b0;
                end
            endcase
        end
    end

    // AXI-Lite signals
    assign S_AXIL_AWREADY = axil_reg_awready;
    assign S_AXIL_WREADY = axil_reg_wready;
    assign S_AXIL_BVALID = axil_reg_bvalid;
    assign S_AXIL_ARREADY = axil_reg_arready;
    assign S_AXIL_RVALID = axil_reg_rvalid;
    assign S_AXIL_RDATA = axil_reg_rdata;
    assign S_AXIL_BRESP = 2'b00;
    assign S_AXIL_RRESP = 2'b00;

    // Output interrupt signal
    assign o_int = copy_done;

endmodule
```
