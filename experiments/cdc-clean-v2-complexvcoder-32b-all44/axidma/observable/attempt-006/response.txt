```verilog
module axidma #(
		parameter	C_AXI_ID_WIDTH = 1,
		parameter	C_AXI_ADDR_WIDTH = 32,
		parameter	C_AXI_DATA_WIDTH = 32,
		localparam	C_AXIL_ADDR_WIDTH = 5,
		localparam	C_AXIL_DATA_WIDTH = 32,
		parameter [0:0]	OPT_UNALIGNED = 1'b1,
		parameter [0:0]	OPT_WRAPMEM = 1'b1,
		parameter	LGFIFO = LGMAXBURST+1,
		parameter	LGLEN = C_AXI_ADDR_WIDTH,
		parameter [0:0]	OPT_LOWPOWER = 1'b0,
		parameter [0:0]	OPT_CLKGATE = OPT_LOWPOWER,
		parameter	[C_AXI_ID_WIDTH-1:0]	AXI_READ_ID = 0,
		parameter	[C_AXI_ID_WIDTH-1:0]	AXI_WRITE_ID = 0,
		parameter	[7:0]			ABORT_KEY  = 8'h6d,
		localparam	ADDRLSB= $clog2(C_AXI_DATA_WIDTH)-3,
		localparam	AXILLSB= $clog2(C_AXIL_DATA_WIDTH)-3,
		localparam	LGLENW= LGLEN-ADDRLSB
	) (
		input	wire	S_AXI_ACLK,
		input	wire	S_AXI_ARESETN,
		input	wire				S_AXIL_AWVALID,
		output	wire				S_AXIL_AWREADY,
		input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_AWADDR,
		input	wire 	[2:0]			S_AXIL_AWPROT,
		input	wire				S_AXIL_WVALID,
		output	wire				S_AXIL_WREADY,
		input	wire [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_WDATA,
		input	wire [C_AXIL_DATA_WIDTH/8-1:0]	S_AXIL_WSTRB,
		output	reg				S_AXIL_BVALID,
		input	wire				S_AXIL_BREADY,
		output	wire	[1:0]			S_AXIL_BRESP,
		input	wire				S_AXIL_ARVALID,
		output	wire				S_AXIL_ARREADY,
		input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_ARADDR,
		input	wire 	[2:0]			S_AXIL_ARPROT,
		output	reg				S_AXIL_RVALID,
		input	wire				S_AXIL_RREADY,
		output	reg [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_RDATA,
		output	wire	[1:0]			S_AXIL_RRESP,
		output	reg				M_AXI_AWVALID,
		input	wire				M_AXI_AWREADY,
		output	reg	[C_AXI_ID_WIDTH-1:0]	M_AXI_AWID,
		output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_AWADDR,
		output	reg	[2:0]			M_AXI_AWSIZE,
		output	reg	[1:0]			M_AXI_AWBURST,
		output	reg				M_AXI_AWLOCK,
		output	reg	[3:0]			M_AXI_AWCACHE,
		output	reg	[2:0]			M_AXI_AWPROT,
		output	reg	[3:0]			M_AXI_AWQOS,
		output	reg				M_AXI_WVALID,
		input	wire				M_AXI_WREADY,
		output	reg	[C_AXI_DATA_WIDTH-1:0]	M_AXI_WDATA,
		output	reg [C_AXI_DATA_WIDTH/8-1:0]	M_AXI_WSTRB,
		output	reg				M_AXI_WLAST,
		input	wire				M_AXI_BVALID,
		output	reg				M_AXI_BREADY,
		input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_BID,
		input	wire	[1:0]			M_AXI_BRESP,
		output	reg				M_AXI_ARVALID,
		input	wire				M_AXI_ARREADY,
		output	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_ARID,
		output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_ARADDR,
		output	wire	[2:0]			M_AXI_ARSIZE,
		output	wire	[1:0]			M_AXI_ARBURST,
		output	wire				M_AXI_ARLOCK,
		output	wire	[3:0]			M_AXI_ARCACHE,
		output	wire	[2:0]			M_AXI_ARPROT,
		output	wire	[3:0]			M_AXI_ARQOS,
		input	wire				M_AXI_RVALID,
		output	wire				M_AXI_RREADY,
		input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_RID,
		input	wire	[C_AXI_DATA_WIDTH-1:0]	M_AXI_RDATA,
		input	wire				M_AXI_RLAST,
		input	wire	[1:0]			M_AXI_RRESP,
		output	reg				o_int
	);

	// Internal signals for AXI-Lite
	reg [C_AXIL_ADDR_WIDTH-1:0] axil_awaddr_reg;
	reg [C_AXIL_DATA_WIDTH-1:0] axil_wdata_reg;
	reg [C_AXIL_DATA_WIDTH/8-1:0] axil_wstrb_reg;
	reg [C_AXI_ADDR_WIDTH-1:0] src_addr_reg;
	reg [C_AXI_ADDR_WIDTH-1:0] dst_addr_reg;
	reg [LGLEN-1:0] len_reg;
	reg start_transfer;
	reg transfer_active;

	// Internal signals for AXI-MM
	reg [LGLEN-1:0] transfer_len;
	reg [C_AXI_ADDR_WIDTH-1:0] current_addr;
	reg read_phase;
	reg write_phase;

	// AXI-Lite FSM states
	typedef enum reg [1:0] {
		IDLE,
		WRITE_REG,
		READ_REG
	} axil_state_t;

	axil_state_t axil_state, axil_next_state;

	// AXI-MM FSM states
	typedef enum reg [1:0] {
		MM_IDLE,
		READ_DATA,
		WRITE_DATA
	} axi_mm_state_t;

	axi_mm_state_t axi_mm_state, axi_mm_next_state;

	// AXI-Lite FSM
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			axil_state <= IDLE;
			axil_awaddr_reg <= 0;
			axil_wdata_reg <= 0;
			axil_wstrb_reg <= 0;
			src_addr_reg <= 0;
			dst_addr_reg <= 0;
			len_reg <= 0;
			start_transfer <= 0;
		end else begin
			axil_state <= axil_next_state;

			case (axil_state)
				IDLE: begin
					if (S_AXIL_AWVALID && S_AXIL_WVALID) begin
						axil_awaddr_reg <= S_AXIL_AWADDR;
						axil_wdata_reg <= S_AXIL_WDATA;
						axil_wstrb_reg <= S_AXIL_WSTRB;
					end
				end
				WRITE_REG: begin
					if (axil_awaddr_reg == 0) begin
						src_addr_reg <= axil_wdata_reg;
					end else if (axil_awaddr_reg == 4) begin
						dst_addr_reg <= axil_wdata_reg;
					end else if (axil_awaddr_reg == 8) begin
						len_reg <= axil_wdata_reg;
					end else if (axil_awaddr_reg == 12) begin
						start_transfer <= 1;
					end
				end
				READ_REG: begin
					// Placeholder for read operations
				end
			endcase
		end
	end

	// AXI-Lite FSM next state logic
	always @(*) begin
		axil_next_state = axil_state;

		case (axil_state)
			IDLE: begin
				if (S_AXIL_AWVALID && S_AXIL_WVALID) begin
					axil_next_state = WRITE_REG;
				end else if (S_AXIL_ARVALID) begin
					axil_next_state = READ_REG;
				end
			end
			WRITE_REG: begin
				if (axil_awaddr_reg == 12) begin
					axil_next_state = IDLE;
				end
			end
			READ_REG: begin
				if (S_AXIL_RREADY) begin
					axil_next_state = IDLE;
				end
			end
		endcase
	end

	// AXI-Lite ready signals
	assign S_AXIL_AWREADY = (axil_state == IDLE) && S_AXIL_WVALID;
	assign S_AXIL_WREADY = (axil_state == IDLE) && S_AXIL_AWVALID;
	assign S_AXIL_BVALID = (axil_state == WRITE_REG) && (axil_awaddr_reg == 12);
	assign S_AXIL_BRESP = 2'b00; // OKAY
	assign S_AXIL_ARREADY = (axil_state == IDLE);
	assign S_AXIL_RVALID = (axil_state == READ_REG);
	assign S_AXIL_RRESP = 2'b00; // OKAY

	// AXI-MM FSM
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			axi_mm_state <= MM_IDLE;
			transfer_active <= 0;
			transfer_len <= 0;
			current_addr <= 0;
			read_phase <= 0;
			write_phase <= 0;
		end else begin
			axi_mm_state <= axi_mm_next_state;

			case (axi_mm_state)
				MM_IDLE: begin
					if (start_transfer) begin
						start_transfer <= 0;
						transfer_active <= 1;
						transfer_len <= len_reg;
						current_addr <= src_addr_reg;
						read_phase <= 1;
						write_phase <= 0;
					end
				end
				READ_DATA: begin
					if (M_AXI_RVALID && M_AXI_RREADY) begin
						transfer_len <= transfer_len - 1;
						current_addr <= current_addr + C_AXI_DATA_WIDTH/8;
						read_phase <= 0;
						write_phase <= 1;
					end
				end
				WRITE_DATA: begin
					if (M_AXI_WREADY && M_AXI_WVALID) begin
						transfer_len <= transfer_len - 1;
						current_addr <= current_addr + C_AXI_DATA_WIDTH/8;
						if (transfer_len == 0) begin
							transfer_active <= 0;
							read_phase <= 0;
							write_phase <= 0;
						end else begin
							read_phase <= 1;
							write_phase <= 0;
						end
					end
				end
			endcase
		end
	end

	// AXI-MM FSM next state logic
	always @(*) begin
		axi_mm_next_state = axi_mm_state;

		case (axi_mm_state)
			MM_IDLE: begin
				if (start_transfer) begin
					axi_mm_next_state = READ_DATA;
				end
			end
			READ_DATA: begin
				if (M_AXI_RVALID && M_AXI_RREADY) begin
					if (transfer_len == 0) begin
						axi_mm_next_state = MM_IDLE;
					end else begin
						axi_mm_next_state = WRITE_DATA;
					end
				end
			end
			WRITE_DATA: begin
				if (M_AXI_WREADY && M_AXI_WVALID) begin
					if (transfer_len == 0) begin
						axi_mm_next_state = MM_IDLE;
					end else begin
						axi_mm_next_state = READ_DATA;
					end
				end
			end
		endcase
	end

	// AXI-MM control signals
	assign M_AXI_AWVALID = (axi_mm_state == WRITE_DATA) && !transfer_active;
	assign M_AXI_AWID = AXI_WRITE_ID;
	assign M_AXI_AWADDR = dst_addr_reg;
	assign M_AXI_AWSIZE = 3'b010; // 32-bit data width
	assign M_AXI_AWBURST = 2'b01; // INCR
	assign M_AXI_AWLOCK = 1'b0;
	assign M_AXI_AWCACHE = 4'b0011; // Normal, Non-cacheable, Bufferable
	assign M_AXI_AWPROT = 3'b000; // Normal access
	assign M_AXI_AWQOS = 4'b0000;

	assign M_AXI_WVALID = (axi_mm_state == WRITE_DATA);
	assign M_AXI_WDATA = M_AXI_RDATA;
	assign M_AXI_WSTRB = {C_AXI_DATA_WIDTH/8{1'b1}};
	assign M_AXI_WLAST = (transfer_len == 1);

	assign M_AXI_BREADY = (axi_mm_state == WRITE_DATA);

	assign M_AXI_ARVALID = (axi_mm_state == READ_DATA) && !transfer_active;
	assign M_AXI_ARID = AXI_READ_ID;
	assign M_AXI_ARADDR = current_addr;
	assign M_AXI_ARSIZE = 3'b010; // 32-bit data width
	assign M_AXI_ARBURST = 2'b01; // INCR
	assign M_AXI_ARLOCK = 1'b0;
	assign M_AXI_ARCACHE = 4'b0011; // Normal, Non-cacheable, Bufferable
	assign M_AXI_ARPROT = 3'b000; // Normal access
	assign M_AXI_ARQOS = 4'b0000;

	assign M_AXI_RREADY = (axi_mm_state == READ_DATA);

	// Interrupt signal
	assign o_int = (axi_mm_state == MM_IDLE) && transfer_active;

endmodule
```
