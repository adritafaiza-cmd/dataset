=== STAGE 2: VERILOG ===
```verilog
module axidma #(
    parameter C_AXI_ID_WIDTH = 1,
    parameter C_AXI_ADDR_WIDTH = 32,
    parameter C_AXI_DATA_WIDTH = 32,
    localparam C_AXIL_ADDR_WIDTH = 5,
    localparam C_AXIL_DATA_WIDTH = 32,
    parameter [0:0] OPT_UNALIGNED = 1'b1,
    parameter [0:0] OPT_WRAPMEM = 1'b1,
    parameter LGFIFO = LGMAXBURST+1,
    parameter LGLEN = C_AXI_ADDR_WIDTH,
    parameter [0:0] OPT_LOWPOWER = 1'b0,
    parameter [0:0] OPT_CLKGATE = OPT_LOWPOWER,
    parameter [C_AXI_ID_WIDTH-1:0] AXI_READ_ID = 0,
    parameter [C_AXI_ID_WIDTH-1:0] AXI_WRITE_ID = 0,
    parameter [7:0] ABORT_KEY = 8'h6d,
    localparam ADDRLSB = $clog2(C_AXI_DATA_WIDTH)-3,
    localparam AXILLSB = $clog2(C_AXIL_DATA_WIDTH)-3,
    localparam LGLENW = LGLEN-ADDRLSB
) (
    input wire S_AXI_ACLK,
    input wire S_AXI_ARESETN,
    input wire S_AXIL_AWVALID,
    output wire S_AXIL_AWREADY,
    input wire [C_AXIL_ADDR_WIDTH-1:0] S_AXIL_AWADDR,
    input wire [2:0] S_AXIL_AWPROT,
    input wire S_AXIL_WVALID,
    output wire S_AXIL_WREADY,
    input wire [C_AXIL_DATA_WIDTH-1:0] S_AXIL_WDATA,
    input wire [C_AXIL_DATA_WIDTH/8-1:0] S_AXIL_WSTRB,
    output reg S_AXIL_BVALID,
    input wire S_AXIL_BREADY,
    output wire [1:0] S_AXIL_BRESP,
    input wire S_AXIL_ARVALID,
    output wire S_AXIL_ARREADY,
    input wire [C_AXIL_ADDR_WIDTH-1:0] S_AXIL_ARADDR,
    input wire [2:0] S_AXIL_ARPROT,
    output reg S_AXIL_RVALID,
    input wire S_AXIL_RREADY,
    output reg [C_AXIL_DATA_WIDTH-1:0] S_AXIL_RDATA,
    output wire [1:0] S_AXIL_RRESP,
    output reg M_AXI_AWVALID,
    input wire M_AXI_AWREADY,
    output reg [C_AXI_ID_WIDTH-1:0] M_AXI_AWID,
    output reg [C_AXI_ADDR_WIDTH-1:0] M_AXI_AWADDR,
    output reg [2:0] M_AXI_AWSIZE,
    output reg [1:0] M_AXI_AWBURST,
    output reg M_AXI_AWLOCK,
    output reg [3:0] M_AXI_AWCACHE,
    output reg [2:0] M_AXI_AWPROT,
    output reg [3:0] M_AXI_AWQOS,
    output reg M_AXI_WVALID,
    input wire M_AXI_WREADY,
    output reg [C_AXI_DATA_WIDTH-1:0] M_AXI_WDATA,
    output reg [C_AXI_DATA_WIDTH/8-1:0] M_AXI_WSTRB,
    output reg M_AXI_WLAST,
    input wire M_AXI_BVALID,
    output reg M_AXI_BREADY,
    input wire [C_AXI_ID_WIDTH-1:0] M_AXI_BID,
    input wire [1:0] M_AXI_BRESP,
    output reg M_AXI_ARVALID,
    input wire M_AXI_ARREADY,
    output wire [C_AXI_ID_WIDTH-1:0] M_AXI_ARID,
    output reg [C_AXI_ADDR_WIDTH-1:0] M_AXI_ARADDR,
    output wire [2:0] M_AXI_ARSIZE,
    output wire [1:0] M_AXI_ARBURST,
    output wire M_AXI_ARLOCK,
    output wire [3:0] M_AXI_ARCACHE,
    output wire [2:0] M_AXI_ARPROT,
    output wire [3:0] M_AXI_ARQOS,
    input wire M_AXI_RVALID,
    output wire M_AXI_RREADY,
    input wire [C_AXI_ID_WIDTH-1:0] M_AXI_RID,
    input wire [C_AXI_DATA_WIDTH-1:0] M_AXI_RDATA,
    input wire M_AXI_RLAST,
    input wire [1:0] M_AXI_RRESP,
    output reg o_int
);

    // Internal signals
    reg [C_AXI_ADDR_WIDTH-1:0] src_addr;
    reg [C_AXI_ADDR_WIDTH-1:0] dst_addr;
    reg [LGLEN-1:0] len;
    reg [LGLENW-1:0] len_cnt;
    reg [C_AXI_DATA_WIDTH-1:0] axil_reg [0:(1<<C_AXIL_ADDR_WIDTH)-1];
    reg axil_awready_reg;
    reg axil_wready_reg;
    reg axil_arready_reg;
    reg axil_rvalid_reg;
    reg [C_AXIL_DATA_WIDTH-1:0] axil_rdata_reg;
    reg [1:0] axil_bresp_reg;
    reg [1:0] axil_rresp_reg;
    reg axi_awvalid_reg;
    reg axi_wvalid_reg;
    reg axi_wlast_reg;
    reg axi_arvalid_reg;
    reg axi_rready_reg;
    reg [C_AXI_DATA_WIDTH-1:0] axi_wdata_reg;
    reg [C_AXI_DATA_WIDTH/8-1:0] axi_wstrb_reg;
    reg [C_AXI_ADDR_WIDTH-1:0] axi_araddr_reg;
    reg dma_state;
    localparam IDLE = 2'b00;
    localparam READ = 2'b01;
    localparam WRITE = 2'b10;

    // AXILite Write Address Channel
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axil_awready_reg <= 0;
        end else if (S_AXIL_AWVALID && axil_awready_reg) begin
            axil_awready_reg <= 0;
        end else if (!axil_awready_reg && !dma_state) begin
            axil_awready_reg <= 1;
        end
    end

    // AXILite Write Data Channel
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axil_wready_reg <= 0;
        end else if (S_AXIL_WVALID && axil_wready_reg) begin
            axil_wready_reg <= 0;
        end else if (!axil_wready_reg && !dma_state) begin
            axil_wready_reg <= 1;
        end
    end

    // AXILite Write Response Channel
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            S_AXIL_BVALID <= 0;
            axil_bresp_reg <= 0;
        end else if (S_AXIL_BREADY && S_AXIL_BVALID) begin
            S_AXIL_BVALID <= 0;
        end else if (dma_state == IDLE && axil_wready_reg && S_AXIL_WVALID) begin
            S_AXIL_BVALID <= 1;
            axil_bresp_reg <= 0;
        end
    end

    // AXILite Read Address Channel
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axil_arready_reg <= 0;
        end else if (S_AXIL_ARVALID && axil_arready_reg) begin
            axil_arready_reg <= 0;
        end else if (!axil_arready_reg && !dma_state) begin
            axil_arready_reg <= 1;
        end
    end

    // AXILite Read Data Channel
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axil_rvalid_reg <= 0;
            axil_rdata_reg <= 0;
            axil_rresp_reg <= 0;
        end else if (S_AXIL_RREADY && axil_rvalid_reg) begin
            axil_rvalid_reg <= 0;
        end else if (dma_state == IDLE && axil_arready_reg && S_AXIL_ARVALID) begin
            axil_rvalid_reg <= 1;
            axil_rdata_reg <= axil_reg[S_AXIL_ARADDR];
            axil_rresp_reg <= 0;
        end
    end

    // AXI Write Address Channel
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axi_awvalid_reg <= 0;
            M_AXI_AWID <= AXI_WRITE_ID;
            M_AXI_AWADDR <= 0;
            M_AXI_AWSIZE <= $clog2(C_AXI_DATA_WIDTH/8);
            M_AXI_AWBURST <= 2'b01; // INCR
            M_AXI_AWLOCK <= 0;
            M_AXI_AWCACHE <= 4'b0011; // Normal Non-cacheable
            M_AXI_AWPROT <= 3'b000;
            M_AXI_AWQOS <= 4'b0000;
        end else if (dma_state == WRITE && !axi_awvalid_reg && M_AXI_AWREADY) begin
            axi_awvalid_reg <= 1;
            M_AXI_AWADDR <= dst_addr;
        end else if (axi_awvalid_reg && M_AXI_AWREADY) begin
            axi_awvalid_reg <= 0;
        end
    end

    // AXI Write Data Channel
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axi_wvalid_reg <= 0;
            axi_wlast_reg <= 0;
            M_AXI_WDATA <= 0;
            M_AXI_WSTRB <= {C_AXI_DATA_WIDTH/8{1'b1}};
        end else if (dma_state == WRITE && !axi_wvalid_reg && M_AXI_WREADY) begin
            axi_wvalid_reg <= 1;
            M_AXI_WDATA <= axil_reg[src_addr];
            axi_wlast_reg <= (len_cnt == 1);
        end else if (axi_wvalid_reg && M_AXI_WREADY) begin
            axi_wvalid_reg <= 0;
            axi_wlast_reg <= 0;
            src_addr <= src_addr + C_AXI_DATA_WIDTH/8;
            len_cnt <= len_cnt - 1;
        end
    end

    // AXI Write Response Channel
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            M_AXI_BREADY <= 0;
        end else if (dma_state == WRITE && M_AXI_BVALID && !M_AXI_BREADY) begin
            M_AXI_BREADY <= 1;
        end else if (M_AXI_BVALID && M_AXI_BREADY) begin
            M_AXI_BREADY <= 0;
            if (len_cnt == 0) begin
                dma_state <= IDLE;
            end
        end
    end

    // AXI Read Address Channel
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axi_arvalid_reg <= 0;
            M_AXI_ARID <= AXI_READ_ID;
            M_AXI_ARADDR <= 0;
            M_AXI_ARSIZE <= $clog2(C_AXI_DATA_WIDTH/8);
            M_AXI_ARBURST <= 2'b01; // INCR
            M_AXI_ARLOCK <= 0;
            M_AXI_ARCACHE <= 4'b0011; // Normal Non-cacheable
            M_AXI_ARPROT <= 3'b000;
            M_AXI_ARQOS <= 4'b0000;
        end else if (dma_state == READ && !axi_arvalid_reg && M_AXI_ARREADY) begin
            axi_arvalid_reg <= 1;
            M_AXI_ARADDR <= src_addr;
        end else if (axi_arvalid_reg && M_AXI_ARREADY) begin
            axi_arvalid_reg <= 0;
        end
    end

    // AXI Read Data Channel
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axi_rready_reg <= 0;
        end else if (dma_state == READ && M_AXI_RVALID && !axi_rready_reg) begin
            axi_rready_reg <= 1;
            axil_reg[dst_addr] <= M_AXI_RDATA;
            dst_addr <= dst_addr + C_AXI_DATA_WIDTH/8;
            len_cnt <= len_cnt - 1;
        end else if (M_AXI_RVALID && axi_rready_reg) begin
            axi_rready_reg <= 0;
            if (len_cnt == 0) begin
                dma_state <= IDLE;
            end
        end
    end

    // Control Logic
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            dma_state <= IDLE;
            src_addr <= 0;
            dst_addr <= 0;
            len <= 0;
            len_cnt <= 0;
            o_int <= 0;
        end else if (dma_state == IDLE && axil_awready_reg && S_AXIL_AWVALID && axil_wready_reg && S_AXIL_WVALID) begin
            if (S_AXIL_AWADDR == 0) begin
                src_addr <= S_AXIL_WDATA;
            end else if (S_AXIL_AWADDR == 4) begin
                dst_addr <= S_AXIL_WDATA;
            end else if (S_AXIL_AWADDR == 8) begin
                len <= S_AXIL_WDATA;
                len_cnt <= S_AXIL_WDATA;
                if (S_AXIL_WDATA != 0) begin
                    if (S_AXIL_WDATA[1:0] == 2'b01) begin // READ
                        dma_state <= READ;
                    end else if (S_AXIL_WDATA[1:0] == 2'b10) begin // WRITE
                        dma_state <= WRITE;
                    end
                end
            end
        end else if (dma_state == READ && M_AXI_RVALID && axi_rready_reg) begin
            if (len_cnt == 0) begin
                dma_state <= IDLE;
                o_int <= 1;
            end
        end else if (dma_state == WRITE && M_AXI_BVALID && M_AXI_BREADY) begin
            if (len_cnt == 0) begin
                dma_state <= IDLE;
                o_int <= 1;
            end
        end
    end

    // AXILite Responses
    assign S_AXIL_AWREADY = axil_awready_reg;
    assign S_AXIL_WREADY = axil_wready_reg;
    assign S_AXIL_ARREADY = axil_arready_reg;
    assign S_AXIL_RVALID = axil_rvalid_reg;
    assign S_AXIL_RDATA = axil_rdata_reg;
    assign S_AXIL_BRESP = axil_bresp_reg;
    assign S_AXIL_RRESP = axil_rresp_reg;
    assign M_AXI_ARID = AXI_READ_ID;
    assign M_AXI_ARLOCK = 0;
    assign M_AXI_ARCACHE = 4'b0011;
    assign M_AXI_ARPROT = 3'b000;
    assign M_AXI_ARQOS = 4'b0000;
    assign M_AXI_RREADY = axi_rready_reg;

endmodule
```
