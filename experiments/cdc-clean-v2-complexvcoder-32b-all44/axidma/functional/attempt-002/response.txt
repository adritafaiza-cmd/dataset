=== STAGE 2: VERILOG ===
```verilog
module axidma #(
        parameter C_AXI_ID_WIDTH = 1,
        parameter C_AXI_ADDR_WIDTH = 32,
        parameter C_AXI_DATA_WIDTH = 32,
        localparam C_AXIL_ADDR_WIDTH = 5,
        localparam C_AXIL_DATA_WIDTH = 32,
        parameter [0:0] OPT_UNALIGNED = 1'b1,
        parameter [0:0] OPT_WRAPMEM = 1'b1,
        parameter LGFIFO = LGMAXBURST+1,
        parameter LGLEN = C_AXI_ADDR_WIDTH,
        parameter [0:0] OPT_LOWPOWER = 1'b0,
        parameter [0:0] OPT_CLKGATE = OPT_LOWPOWER,
        parameter [C_AXI_ID_WIDTH-1:0] AXI_READ_ID = 0,
        parameter [C_AXI_ID_WIDTH-1:0] AXI_WRITE_ID = 0,
        parameter [7:0] ABORT_KEY = 8'h6d,
        localparam ADDRLSB= $clog2(C_AXI_DATA_WIDTH)-3,
        localparam AXILLSB= $clog2(C_AXIL_DATA_WIDTH)-3,
        localparam LGLENW= LGLEN-ADDRLSB
    ) (
        input wire S_AXI_ACLK,
        input wire S_AXI_ARESETN,
        input wire S_AXIL_AWVALID,
        output wire S_AXIL_AWREADY,
        input wire [C_AXIL_ADDR_WIDTH-1:0] S_AXIL_AWADDR,
        input wire [2:0] S_AXIL_AWPROT,
        input wire S_AXIL_WVALID,
        output wire S_AXIL_WREADY,
        input wire [C_AXIL_DATA_WIDTH-1:0] S_AXIL_WDATA,
        input wire [C_AXIL_DATA_WIDTH/8-1:0] S_AXIL_WSTRB,
        output reg S_AXIL_BVALID,
        input wire S_AXIL_BREADY,
        output wire [1:0] S_AXIL_BRESP,
        input wire S_AXIL_ARVALID,
        output wire S_AXIL_ARREADY,
        input wire [C_AXIL_ADDR_WIDTH-1:0] S_AXIL_ARADDR,
        input wire [2:0] S_AXIL_ARPROT,
        output reg S_AXIL_RVALID,
        input wire S_AXIL_RREADY,
        output reg [C_AXIL_DATA_WIDTH-1:0] S_AXIL_RDATA,
        output wire [1:0] S_AXIL_RRESP,
        output reg M_AXI_AWVALID,
        input wire M_AXI_AWREADY,
        output reg [C_AXI_ID_WIDTH-1:0] M_AXI_AWID,
        output reg [C_AXI_ADDR_WIDTH-1:0] M_AXI_AWADDR,
        output reg [2:0] M_AXI_AWSIZE,
        output reg [1:0] M_AXI_AWBURST,
        output reg M_AXI_AWLOCK,
        output reg [3:0] M_AXI_AWCACHE,
        output reg [2:0] M_AXI_AWPROT,
        output reg [3:0] M_AXI_AWQOS,
        output reg M_AXI_WVALID,
        input wire M_AXI_WREADY,
        output reg [C_AXI_DATA_WIDTH-1:0] M_AXI_WDATA,
        output reg [C_AXI_DATA_WIDTH/8-1:0] M_AXI_WSTRB,
        output reg M_AXI_WLAST,
        input wire M_AXI_BVALID,
        output reg M_AXI_BREADY,
        input wire [C_AXI_ID_WIDTH-1:0] M_AXI_BID,
        input wire [1:0] M_AXI_BRESP,
        output reg M_AXI_ARVALID,
        input wire M_AXI_ARREADY,
        output wire [C_AXI_ID_WIDTH-1:0] M_AXI_ARID,
        output reg [C_AXI_ADDR_WIDTH-1:0] M_AXI_ARADDR,
        output wire [2:0] M_AXI_ARSIZE,
        output wire [1:0] M_AXI_ARBURST,
        output wire M_AXI_ARLOCK,
        output wire [3:0] M_AXI_ARCACHE,
        output wire [2:0] M_AXI_ARPROT,
        output wire [3:0] M_AXI_ARQOS,
        input wire M_AXI_RVALID,
        output wire M_AXI_RREADY,
        input wire [C_AXI_ID_WIDTH-1:0] M_AXI_RID,
        input wire [C_AXI_DATA_WIDTH-1:0] M_AXI_RDATA,
        input wire M_AXI_RLAST,
        input wire [1:0] M_AXI_RRESP,
        output reg o_int
    );

    // Internal registers for control and status
    reg [C_AXI_ADDR_WIDTH-1:0] src_addr;
    reg [C_AXI_ADDR_WIDTH-1:0] dest_addr;
    reg [LGLEN-1:0] length;
    reg [C_AXIL_DATA_WIDTH-1:0] control_reg;
    reg [C_AXIL_DATA_WIDTH-1:0] status_reg;

    // State machine states
    typedef enum reg [2:0] {
        IDLE,
        READ_ADDR,
        WRITE_ADDR,
        TRANSFER,
        DONE
    } state_t;

    state_t state, next_state;

    // AXI Lite FSM
    reg axil_awready_reg;
    reg axil_wready_reg;
    reg axil_bvalid_reg;
    reg axil_arready_reg;
    reg axil_rvalid_reg;

    // AXI Master FSM
    reg axi_awvalid_reg;
    reg [C_AXI_ADDR_WIDTH-1:0] axi_awaddr_reg;
    reg [2:0] axi_awsize_reg;
    reg [1:0] axi_awburst_reg;
    reg axi_wvalid_reg;
    reg [C_AXI_DATA_WIDTH-1:0] axi_wdata_reg;
    reg [C_AXI_DATA_WIDTH/8-1:0] axi_wstrb_reg;
    reg axi_wlast_reg;
    reg axi_arvalid_reg;
    reg [C_AXI_ADDR_WIDTH-1:0] axi_araddr_reg;

    // Control signals
    reg start_transfer;
    reg transfer_done;

    // AXI Lite Write FSM
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            axil_awready_reg <= 0;
            axil_wready_reg <= 0;
            axil_bvalid_reg <= 0;
            axil_arready_reg <= 0;
            axil_rvalid_reg <= 0;
            control_reg <= 0;
            status_reg <= 0;
        end else begin
            axil_awready_reg <= S_AXIL_AWVALID & !axil_awready_reg;
            axil_wready_reg <= S_AXIL_WVALID & !axil_wready_reg & axil_awready_reg;
            axil_bvalid_reg <= axil_wready_reg & !axil_bvalid_reg & S_AXIL_BREADY;
            axil_arready_reg <= S_AXIL_ARVALID & !axil_arready_reg;
            axil_rvalid_reg <= axil_arready_reg & !axil_rvalid_reg & S_AXIL_RREADY;

            if (axil_awready_reg & S_AXIL_AWVALID) begin
                case (S_AXIL_AWADDR)
                    5'd0: src_addr <= S_AXIL_WDATA;
                    5'd4: dest_addr <= S_AXIL_WDATA;
                    5'd8: length <= S_AXIL_WDATA;
                    5'd12: control_reg <= S_AXIL_WDATA;
                endcase
            end

            if (axil_wready_reg & S_AXIL_WVALID) begin
                axil_wready_reg <= 0;
            end

            if (axil_bvalid_reg & S_AXIL_BREADY) begin
                axil_bvalid_reg <= 0;
            end

            if (axil_arready_reg & S_AXIL_ARVALID) begin
                case (S_AXIL_ARADDR)
                    5'd16: status_reg <= transfer_done ? 32'h00000001 : 32'h00000000;
                    default: status_reg <= 32'h00000000;
                endcase
                axil_arready_reg <= 0;
            end

            if (axil_rvalid_reg & S_AXIL_RREADY) begin
                S_AXIL_RDATA <= status_reg;
                axil_rvalid_reg <= 0;
            end
        end
    end

    // AXI Lite signals
    assign S_AXIL_AWREADY = axil_awready_reg;
    assign S_AXIL_WREADY = axil_wready_reg;
    assign S_AXIL_BVALID = axil_bvalid_reg;
    assign S_AXIL_BRESP = 2'b00; // OKAY
    assign S_AXIL_ARREADY = axil_arready_reg;
    assign S_AXIL_RVALID = axil_rvalid_reg;
    assign S_AXIL_RRESP = 2'b00; // OKAY

    // AXI Master FSM
    always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
        if (!S_AXI_ARESETN) begin
            state <= IDLE;
            axi_awvalid_reg <= 0;
            axi_awaddr_reg <= 0;
            axi_awsize_reg <= 2'b10; // 4 bytes
            axi_awburst_reg <= 2'b01; // INCR
            axi_wvalid_reg <= 0;
            axi_wdata_reg <= 0;
            axi_wstrb_reg <= 0;
            axi_wlast_reg <= 0;
            axi_arvalid_reg <= 0;
            axi_araddr_reg <= 0;
            M_AXI_RREADY <= 0;
            M_AXI_BREADY <= 0;
            start_transfer <= 0;
            transfer_done <= 0;
        end else begin
            state <= next_state;

            case (state)
                IDLE: begin
                    if (control_reg[0]) begin
                        axi_awaddr_reg <= src_addr;
                        axi_araddr_reg <= dest_addr;
                        axi_awvalid_reg <= 1;
                        start_transfer <= 1;
                    end
                end

                READ_ADDR: begin
                    if (M_AXI_ARREADY) begin
                        axi_arvalid_reg <= 0;
                        M_AXI_RREADY <= 1;
                    end
                end

                WRITE_ADDR: begin
                    if (M_AXI_AWREADY) begin
                        axi_awvalid_reg <= 0;
                        axi_wvalid_reg <= 1;
                    end
                end

                TRANSFER: begin
                    if (M_AXI_RVALID) begin
                        axi_wdata_reg <= M_AXI_RDATA;
                        axi_wstrb_reg <= {C_AXI_DATA_WIDTH/8{1'b1}};
                        if (length == 32'd1) begin
                            axi_wlast_reg <= 1;
                        end
                    end

                    if (M_AXI_WREADY) begin
                        axi_wvalid_reg <= 0;
                        axi_wlast_reg <= 0;
                        if (length > 32'd1) begin
                            length <= length - 32'd1;
                            axi_awaddr_reg <= axi_awaddr_reg + C_AXI_DATA_WIDTH/8;
                            axi_araddr_reg <= axi_araddr_reg + C_AXI_DATA_WIDTH/8;
                            axi_arvalid_reg <= 1;
                            axi_awvalid_reg <= 1;
                        end else begin
                            next_state <= DONE;
                        end
                    end
                end

                DONE: begin
                    M_AXI_BREADY <= 1;
                    transfer_done <= 1;
                    if (M_AXI_BVALID) begin
                        M_AXI_BREADY <= 0;
                        transfer_done <= 0;
                        start_transfer <= 0;
                        next_state <= IDLE;
                    end
                end
            endcase

            if (start_transfer) begin
                next_state <= READ_ADDR;
            end
        end
    end

    // AXI Master signals
    assign M_AXI_AWVALID = axi_awvalid_reg;
    assign M_AXI_AWID = AXI_WRITE_ID;
    assign M_AXI_AWADDR = axi_awaddr_reg;
    assign M_AXI_AWSIZE = axi_awsize_reg;
    assign M_AXI_AWBURST = axi_awburst_reg;
    assign M_AXI_AWLOCK = 1'b0;
    assign M_AXI_AWCACHE = 4'b0011; // Non-cacheable, buffered, modified, write-back
    assign M_AXI_AWPROT = 3'b000;
    assign M_AXI_AWQOS = 4'b0000;

    assign M_AXI_WVALID = axi_wvalid_reg;
    assign M_AXI_WDATA = axi_wdata_reg;
    assign M_AXI_WSTRB = axi_wstrb_reg;
    assign M_AXI_WLAST = axi_wlast_reg;

    assign M_AXI_ARVALID = axi_arvalid_reg;
    assign M_AXI_ARID = AXI_READ_ID;
    assign M_AXI_ARADDR = axi_araddr_reg;
    assign M_AXI_ARSIZE = axi_awsize_reg;
    assign M_AXI_ARBURST = axi_awburst_reg;
    assign M_AXI_ARLOCK = 1'b0;
    assign M_AXI_ARCACHE = 4'b0011; // Non-cacheable, buffered, modified, write-back
    assign M_AXI_ARPROT = 3'b000;
    assign M_AXI_ARQOS = 4'b0000;

    assign o_int = transfer_done;

endmodule
```
