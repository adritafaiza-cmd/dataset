=== STAGE 2: VERILOG ===
```verilog
module axis_switch #
(
    parameter S_COUNT = 4,
    parameter M_COUNT = 4,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter S_ID_WIDTH = 8,
    parameter M_ID_WIDTH = S_ID_WIDTH+$clog2(S_COUNT),
    parameter M_DEST_WIDTH = 1,
    parameter S_DEST_WIDTH = M_DEST_WIDTH+$clog2(M_COUNT),
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter M_BASE = 0,
    parameter M_TOP = 0,
    parameter M_CONNECT = {M_COUNT{{S_COUNT{1'b1}}}},
    parameter UPDATE_TID = 0,
    parameter S_REG_TYPE = 0,
    parameter M_REG_TYPE = 2,
    parameter ARB_TYPE_ROUND_ROBIN = 1,
    parameter ARB_LSB_HIGH_PRIORITY = 1
)
(
    input  wire                             clk,
    input  wire                             rst,
    /*
     * AXI Stream inputs
     */
    input  wire [S_COUNT*DATA_WIDTH-1:0]    s_axis_tdata,
    input  wire [S_COUNT*KEEP_WIDTH-1:0]    s_axis_tkeep,
    input  wire [S_COUNT-1:0]               s_axis_tvalid,
    output wire [S_COUNT-1:0]               s_axis_tready,
    input  wire [S_COUNT-1:0]               s_axis_tlast,
    input  wire [S_COUNT*S_ID_WIDTH-1:0]    s_axis_tid,
    input  wire [S_COUNT*S_DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [S_COUNT*USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI Stream outputs
     */
    output wire [M_COUNT*DATA_WIDTH-1:0]    m_axis_tdata,
    output wire [M_COUNT*KEEP_WIDTH-1:0]    m_axis_tkeep,
    output wire [M_COUNT-1:0]               m_axis_tvalid,
    input  wire [M_COUNT-1:0]               m_axis_tready,
    output wire [M_COUNT-1:0]               m_axis_tlast,
    output wire [M_COUNT*M_ID_WIDTH-1:0]    m_axis_tid,
    output wire [M_COUNT*M_DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [M_COUNT*USER_WIDTH-1:0]    m_axis_tuser
);

    // Internal signals
    reg [M_COUNT*S_COUNT-1:0] grant;
    reg [S_COUNT-1:0]         s_axis_tready_int;
    reg [M_COUNT-1:0]         m_axis_tvalid_int;
    reg [M_COUNT-1:0]         m_axis_tlast_int;
    reg [M_COUNT*M_ID_WIDTH-1:0] m_axis_tid_int;
    reg [M_COUNT*M_DEST_WIDTH-1:0] m_axis_tdest_int;
    reg [M_COUNT*USER_WIDTH-1:0] m_axis_tuser_int;
    reg [M_COUNT*DATA_WIDTH-1:0] m_axis_tdata_int;
    reg [M_COUNT*KEEP_WIDTH-1:0] m_axis_tkeep_int;

    // Grant logic for round-robin arbitration
    genvar i, j;
    generate
        if (ARB_TYPE_ROUND_ROBIN) begin : round_robin_arb
            reg [S_COUNT-1:0] request;
            reg [$clog2(S_COUNT)-1:0] arb_ptr;

            always @(posedge clk or posedge rst) begin
                if (rst) begin
                    arb_ptr <= 0;
                end else begin
                    for (i = 0; i < S_COUNT; i = i + 1) begin
                        request[i] <= s_axis_tvalid[i];
                    end
                    if (|request) begin
                        for (i = 0; i < S_COUNT; i = i + 1) begin
                            if (request[(arb_ptr + i) % S_COUNT]) begin
                                arb_ptr <= (arb_ptr + i) % S_COUNT;
                                break;
                            end
                        end
                    end
                end
            end

            always @(*) begin
                grant = 0;
                for (i = 0; i < M_COUNT; i = i + 1) begin
                    for (j = 0; j < S_COUNT; j = j + 1) begin
                        if (M_CONNECT[i][j] && s_axis_tvalid[j] && (j == arb_ptr)) begin
                            grant[i*S_COUNT + j] = 1;
                        end
                    end
                end
            end
        end else begin : no_arb
            always @(*) begin
                grant = 0;
                for (i = 0; i < M_COUNT; i = i + 1) begin
                    for (j = 0; j < S_COUNT; j = j + 1) begin
                        if (M_CONNECT[i][j] && s_axis_tvalid[j]) begin
                            grant[i*S_COUNT + j] = 1;
                        end
                    end
                end
            end
        end
    endgenerate

    // Output multiplexers
    always @(*) begin
        m_axis_tdata_int = 0;
        m_axis_tkeep_int = 0;
        m_axis_tvalid_int = 0;
        m_axis_tlast_int = 0;
        m_axis_tid_int = 0;
        m_axis_tdest_int = 0;
        m_axis_tuser_int = 0;
        for (i = 0; i < M_COUNT; i = i + 1) begin
            for (j = 0; j < S_COUNT; j = j + 1) begin
                if (grant[i*S_COUNT + j]) begin
                    m_axis_tdata_int[i*DATA_WIDTH +: DATA_WIDTH] = s_axis_tdata[j*DATA_WIDTH +: DATA_WIDTH];
                    m_axis_tkeep_int[i*KEEP_WIDTH +: KEEP_WIDTH] = s_axis_tkeep[j*KEEP_WIDTH +: KEEP_WIDTH];
                    m_axis_tvalid_int[i] = 1;
                    m_axis_tlast_int[i] = s_axis_tlast[j];
                    m_axis_tid_int[i*M_ID_WIDTH +: M_ID_WIDTH] = s_axis_tid[j*S_ID_WIDTH +: S_ID_WIDTH];
                    m_axis_tdest_int[i*M_DEST_WIDTH +: M_DEST_WIDTH] = s_axis_tdest[j*S_DEST_WIDTH +: S_DEST_WIDTH];
                    m_axis_tuser_int[i*USER_WIDTH +: USER_WIDTH] = s_axis_tuser[j*USER_WIDTH +: USER_WIDTH];
                end
            end
        end
    end

    // Input ready signals
    always @(*) begin
        s_axis_tready_int = 0;
        for (i = 0; i < S_COUNT; i = i + 1) begin
            for (j = 0; j < M_COUNT; j = j + 1) begin
                if (grant[j*S_COUNT + i] && m_axis_tready[j]) begin
                    s_axis_tready_int[i] = 1;
                end
            end
        end
    end

    // Output assignments
    assign s_axis_tready = s_axis_tready_int;
    assign m_axis_tdata = m_axis_tdata_int;
    assign m_axis_tkeep = m_axis_tkeep_int;
    assign m_axis_tvalid = m_axis_tvalid_int;
    assign m_axis_tlast = m_axis_tlast_int;
    assign m_axis_tid = m_axis_tid_int;
    assign m_axis_tdest = m_axis_tdest_int;
    assign m_axis_tuser = m_axis_tuser_int;

endmodule
```
