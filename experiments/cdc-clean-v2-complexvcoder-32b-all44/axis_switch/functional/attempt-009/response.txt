=== STAGE 2: VERILOG ===
```verilog
module axis_switch #
(
    parameter S_COUNT = 4,
    parameter M_COUNT = 4,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter S_ID_WIDTH = 8,
    parameter M_ID_WIDTH = S_ID_WIDTH+$clog2(S_COUNT),
    parameter M_DEST_WIDTH = 1,
    parameter S_DEST_WIDTH = M_DEST_WIDTH+$clog2(M_COUNT),
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter M_BASE = 0,
    parameter M_TOP = 0,
    parameter M_CONNECT = {M_COUNT{{S_COUNT{1'b1}}}},
    parameter UPDATE_TID = 0,
    parameter S_REG_TYPE = 0,
    parameter M_REG_TYPE = 2,
    parameter ARB_TYPE_ROUND_ROBIN = 1,
    parameter ARB_LSB_HIGH_PRIORITY = 1
)
(
    input  wire                             clk,
    input  wire                             rst,
    /*
     * AXI Stream inputs
     */
    input  wire [S_COUNT*DATA_WIDTH-1:0]    s_axis_tdata,
    input  wire [S_COUNT*KEEP_WIDTH-1:0]    s_axis_tkeep,
    input  wire [S_COUNT-1:0]               s_axis_tvalid,
    output wire [S_COUNT-1:0]               s_axis_tready,
    input  wire [S_COUNT-1:0]               s_axis_tlast,
    input  wire [S_COUNT*S_ID_WIDTH-1:0]    s_axis_tid,
    input  wire [S_COUNT*S_DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [S_COUNT*USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI Stream outputs
     */
    output wire [M_COUNT*DATA_WIDTH-1:0]    m_axis_tdata,
    output wire [M_COUNT*KEEP_WIDTH-1:0]    m_axis_tkeep,
    output wire [M_COUNT-1:0]               m_axis_tvalid,
    input  wire [M_COUNT-1:0]               m_axis_tready,
    output wire [M_COUNT-1:0]               m_axis_tlast,
    output wire [M_COUNT*M_ID_WIDTH-1:0]    m_axis_tid,
    output wire [M_COUNT*M_DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [M_COUNT*USER_WIDTH-1:0]    m_axis_tuser
);

    // Internal signals
    wire [S_COUNT-1:0] [DATA_WIDTH-1:0]    s_axis_tdata_int;
    wire [S_COUNT-1:0] [KEEP_WIDTH-1:0]    s_axis_tkeep_int;
    wire [S_COUNT-1:0]                     s_axis_tvalid_int;
    wire [S_COUNT-1:0]                     s_axis_tready_int;
    wire [S_COUNT-1:0]                     s_axis_tlast_int;
    wire [S_COUNT-1:0] [S_ID_WIDTH-1:0]    s_axis_tid_int;
    wire [S_COUNT-1:0] [S_DEST_WIDTH-1:0]  s_axis_tdest_int;
    wire [S_COUNT-1:0] [USER_WIDTH-1:0]    s_axis_tuser_int;

    wire [M_COUNT-1:0] [DATA_WIDTH-1:0]    m_axis_tdata_int;
    wire [M_COUNT-1:0] [KEEP_WIDTH-1:0]    m_axis_tkeep_int;
    wire [M_COUNT-1:0]                     m_axis_tvalid_int;
    wire [M_COUNT-1:0]                     m_axis_tready_int;
    wire [M_COUNT-1:0]                     m_axis_tlast_int;
    wire [M_COUNT-1:0] [M_ID_WIDTH-1:0]    m_axis_tid_int;
    wire [M_COUNT-1:0] [M_DEST_WIDTH-1:0]  m_axis_tdest_int;
    wire [M_COUNT-1:0] [USER_WIDTH-1:0]    m_axis_tuser_int;

    wire [S_COUNT-1:0] [M_COUNT-1:0]       request_matrix;
    wire [M_COUNT-1:0] [S_COUNT-1:0]       grant_matrix;

    // Assign inputs to internal signals
    genvar i;
    generate
        for (i = 0; i < S_COUNT; i = i + 1) begin : assign_inputs
            assign s_axis_tdata_int[i] = s_axis_tdata[(i+1)*DATA_WIDTH-1:i*DATA_WIDTH];
            assign s_axis_tkeep_int[i] = s_axis_tkeep[(i+1)*KEEP_WIDTH-1:i*KEEP_WIDTH];
            assign s_axis_tvalid_int[i] = s_axis_tvalid[i];
            assign s_axis_tlast_int[i] = s_axis_tlast[i];
            assign s_axis_tid_int[i] = s_axis_tid[(i+1)*S_ID_WIDTH-1:i*S_ID_WIDTH];
            assign s_axis_tdest_int[i] = s_axis_tdest[(i+1)*S_DEST_WIDTH-1:i*S_DEST_WIDTH];
            assign s_axis_tuser_int[i] = s_axis_tuser[(i+1)*USER_WIDTH-1:i*USER_WIDTH];
        end
    endgenerate

    // Assign internal signals to outputs
    generate
        for (i = 0; i < M_COUNT; i = i + 1) begin : assign_outputs
            assign m_axis_tdata[(i+1)*DATA_WIDTH-1:i*DATA_WIDTH] = m_axis_tdata_int[i];
            assign m_axis_tkeep[(i+1)*KEEP_WIDTH-1:i*KEEP_WIDTH] = m_axis_tkeep_int[i];
            assign m_axis_tvalid[i] = m_axis_tvalid_int[i];
            assign m_axis_tlast[i] = m_axis_tlast_int[i];
            assign m_axis_tid[(i+1)*M_ID_WIDTH-1:i*M_ID_WIDTH] = m_axis_tid_int[i];
            assign m_axis_tdest[(i+1)*M_DEST_WIDTH-1:i*M_DEST_WIDTH] = m_axis_tdest_int[i];
            assign m_axis_tuser[(i+1)*USER_WIDTH-1:i*USER_WIDTH] = m_axis_tuser_int[i];
        end
    endgenerate

    // Generate request matrix
    generate
        for (i = 0; i < S_COUNT; i = i + 1) begin : gen_request_matrix
            genvar j;
            generate
                for (j = 0; j < M_COUNT; j = j + 1) begin : gen_request_matrix_inner
                    assign request_matrix[i][j] = s_axis_tvalid_int[i] & (s_axis_tdest_int[i][M_DEST_WIDTH+$clog2(M_COUNT)-1:$clog2(M_COUNT)] == j);
                end
            endgenerate
        end
    endgenerate

    // Round-robin arbiter
    wire [S_COUNT-1:0] [M_COUNT-1:0] grant_matrix_next;
    reg  [S_COUNT-1:0] [M_COUNT-1:0] grant_matrix_reg;

    generate
        for (i = 0; i < M_COUNT; i = i + 1) begin : gen_arbiter
            reg [S_COUNT-1:0] request_vector;
            reg [S_COUNT-1:0] grant_vector;
            reg [$clog2(S_COUNT)-1:0] arb_ptr;

            always @(posedge clk or posedge rst) begin
                if (rst) begin
                    arb_ptr <= 0;
                    grant_vector <= 0;
                end else begin
                    request_vector <= request_matrix[0][i] ? {S_COUNT-1'b0, 1'b1} :
                                      request_matrix[1][i] ? {S_COUNT-2'b0, 1'b1, 1'b0} :
                                      request_matrix[2][i] ? {S_COUNT-3'b0, 1'b1, 2'b0} :
                                      request_matrix[3][i] ? {S_COUNT-4'b0, 1'b1, 3'b0} :
                                      0;
                    if (|request_vector) begin
                        if (request_vector[arb_ptr]) begin
                            grant_vector <= request_vector;
                        end else begin
                            grant_vector <= grant_vector << 1;
                            if (grant_vector[0]) begin
                                grant_vector[0] <= 0;
                            end
                            arb_ptr <= arb_ptr + 1;
                        end
                    end else begin
                        grant_vector <= 0;
                    end
                end
            end
            assign grant_matrix_next[0][i] = grant_vector[0];
            assign grant_matrix_next[1][i] = grant_vector[1];
            assign grant_matrix_next[2][i] = grant_vector[2];
            assign grant_matrix_next[3][i] = grant_vector[3];
        end
    endgenerate

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            grant_matrix_reg <= 0;
        end else begin
            grant_matrix_reg <= grant_matrix_next;
        end
    end

    assign grant_matrix = grant_matrix_reg;

    // Assign grants to ready signals
    generate
        for (i = 0; i < S_COUNT; i = i + 1) begin : assign_ready
            assign s_axis_tready_int[i] = |grant_matrix[i];
        end
    endgenerate

    // Assign data to outputs based on grants
    generate
        for (i = 0; i < M_COUNT; i = i + 1) begin : assign_data
            reg [S_COUNT-1:0] [DATA_WIDTH-1:0] tdata_reg;
            reg [S_COUNT-1:0] [KEEP_WIDTH-1:0] tkeep_reg;
            reg [S_COUNT-1:0]                  tvalid_reg;
            reg [S_COUNT-1:0]                  tlast_reg;
            reg [S_COUNT-1:0] [S_ID_WIDTH-1:0] tid_reg;
            reg [S_COUNT-1:0] [USER_WIDTH-1:0] tuser_reg;

            always @(posedge clk or posedge rst) begin
                if (rst) begin
                    tdata_reg <= 0;
                    tkeep_reg <= 0;
                    tvalid_reg <= 0;
                    tlast_reg <= 0;
                    tid_reg <= 0;
                    tuser_reg <= 0;
                end else begin
                    for (int j = 0; j < S_COUNT; j = j + 1) begin
                        if (grant_matrix[j][i]) begin
                            tdata_reg[i] <= s_axis_tdata_int[j];
                            tkeep_reg[i] <= s_axis_tkeep_int[j];
                            tvalid_reg[i] <= s_axis_tvalid_int[j];
                            tlast_reg[i] <= s_axis_tlast_int[j];
                            tid_reg[i] <= s_axis_tid_int[j];
                            tuser_reg[i] <= s_axis_tuser_int[j];
                        end
                    end
                end
            end

            assign m_axis_tdata_int[i] = tdata_reg[i];
            assign m_axis_tkeep_int[i] = tkeep_reg[i];
            assign m_axis_tvalid_int[i] = tvalid_reg[i] & m_axis_tready[i];
            assign m_axis_tlast_int[i] = tlast_reg[i];
            assign m_axis_tid_int[i] = tid_reg[i];
            assign m_axis_tdest_int[i] = s_axis_tdest_int[i][M_DEST_WIDTH+$clog2(M_COUNT)-1:$clog2(M_COUNT)];
            assign m_axis_tuser_int[i] = tuser_reg[i];
        end
    endgenerate

    assign s_axis_tready = s_axis_tready_int;

endmodule
```
