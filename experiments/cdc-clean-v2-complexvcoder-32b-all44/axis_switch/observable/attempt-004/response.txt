=== STAGE 2: VERILOG ===
```verilog
module axis_switch #
(
    parameter S_COUNT = 4,
    parameter M_COUNT = 4,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter S_ID_WIDTH = 8,
    parameter M_ID_WIDTH = S_ID_WIDTH+$clog2(S_COUNT),
    parameter M_DEST_WIDTH = 1,
    parameter S_DEST_WIDTH = M_DEST_WIDTH+$clog2(M_COUNT),
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter M_BASE = 0,
    parameter M_TOP = 0,
    parameter M_CONNECT = {M_COUNT{{S_COUNT{1'b1}}}},
    parameter UPDATE_TID = 0,
    parameter S_REG_TYPE = 0,
    parameter M_REG_TYPE = 2,
    parameter ARB_TYPE_ROUND_ROBIN = 1,
    parameter ARB_LSB_HIGH_PRIORITY = 1
)
(
    input  wire                             clk,
    input  wire                             rst,
    /*
     * AXI Stream inputs
     */
    input  wire [S_COUNT*DATA_WIDTH-1:0]    s_axis_tdata,
    input  wire [S_COUNT*KEEP_WIDTH-1:0]    s_axis_tkeep,
    input  wire [S_COUNT-1:0]               s_axis_tvalid,
    output wire [S_COUNT-1:0]               s_axis_tready,
    input  wire [S_COUNT-1:0]               s_axis_tlast,
    input  wire [S_COUNT*S_ID_WIDTH-1:0]    s_axis_tid,
    input  wire [S_COUNT*S_DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [S_COUNT*USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI Stream outputs
     */
    output wire [M_COUNT*DATA_WIDTH-1:0]    m_axis_tdata,
    output wire [M_COUNT*KEEP_WIDTH-1:0]    m_axis_tkeep,
    output wire [M_COUNT-1:0]               m_axis_tvalid,
    input  wire [M_COUNT-1:0]               m_axis_tready,
    output wire [M_COUNT-1:0]               m_axis_tlast,
    output wire [M_COUNT*M_ID_WIDTH-1:0]    m_axis_tid,
    output wire [M_COUNT*M_DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [M_COUNT*USER_WIDTH-1:0]    m_axis_tuser
);

    reg [S_COUNT-1:0] valid_reg;
    reg [S_COUNT-1:0] ready_reg;
    reg [S_COUNT-1:0] last_reg;
    reg [S_COUNT*S_ID_WIDTH-1:0] tid_reg;
    reg [S_COUNT*S_DEST_WIDTH-1:0] tdest_reg;
    reg [S_COUNT*USER_WIDTH-1:0] tuser_reg;
    reg [S_COUNT*S_DEST_WIDTH-1:0] dest_bits;
    reg [S_COUNT-1:0] grant;
    reg [$clog2(S_COUNT)-1:0] arb_ptr;

    wire [S_COUNT-1:0] valid;
    wire [S_COUNT-1:0] ready;
    wire [S_COUNT-1:0] last;
    wire [S_COUNT*S_ID_WIDTH-1:0] tid;
    wire [S_COUNT*S_DEST_WIDTH-1:0] tdest;
    wire [S_COUNT*USER_WIDTH-1:0] tuser;

    assign valid = s_axis_tvalid;
    assign last = s_axis_tlast;
    assign tid = s_axis_tid;
    assign tdest = s_axis_tdest;
    assign tuser = s_axis_tuser;

    genvar i;
    generate
        for (i = 0; i < S_COUNT; i = i + 1) begin : gen_valid_reg
            always @(posedge clk or posedge rst) begin
                if (rst) begin
                    valid_reg[i] <= 1'b0;
                end else if (ready[i]) begin
                    valid_reg[i] <= 1'b0;
                end else if (valid[i]) begin
                    valid_reg[i] <= 1'b1;
                end
            end
        end

        for (i = 0; i < S_COUNT; i = i + 1) begin : gen_ready_reg
            assign ready[i] = m_axis_tready[dest_bits[i*2+:2]] & ~ready_reg[i];
            always @(posedge clk or posedge rst) begin
                if (rst) begin
                    ready_reg[i] <= 1'b0;
                end else if (ready[i]) begin
                    ready_reg[i] <= 1'b1;
                end else if (valid[i] & ~ready_reg[i]) begin
                    ready_reg[i] <= 1'b0;
                end
            end
        end

        for (i = 0; i < S_COUNT; i = i + 1) begin : gen_last_reg
            always @(posedge clk or posedge rst) begin
                if (rst) begin
                    last_reg[i] <= 1'b0;
                end else if (ready[i]) begin
                    last_reg[i] <= 1'b0;
                end else if (valid[i]) begin
                    last_reg[i] <= last[i];
                end
            end
        end

        for (i = 0; i < S_COUNT; i = i + 1) begin : gen_tid_reg
            always @(posedge clk or posedge rst) begin
                if (rst) begin
                    tid_reg[i*S_ID_WIDTH+:S_ID_WIDTH] <= {S_ID_WIDTH{1'b0}};
                end else if (ready[i]) begin
                    tid_reg[i*S_ID_WIDTH+:S_ID_WIDTH] <= {S_ID_WIDTH{1'b0}};
                end else if (valid[i]) begin
                    tid_reg[i*S_ID_WIDTH+:S_ID_WIDTH] <= tid[i*S_ID_WIDTH+:S_ID_WIDTH];
                end
            end
        end

        for (i = 0; i < S_COUNT; i = i + 1) begin : gen_tdest_reg
            always @(posedge clk or posedge rst) begin
                if (rst) begin
                    tdest_reg[i*S_DEST_WIDTH+:S_DEST_WIDTH] <= {S_DEST_WIDTH{1'b0}};
                end else if (ready[i]) begin
                    tdest_reg[i*S_DEST_WIDTH+:S_DEST_WIDTH] <= {S_DEST_WIDTH{1'b0}};
                end else if (valid[i]) begin
                    tdest_reg[i*S_DEST_WIDTH+:S_DEST_WIDTH] <= tdest[i*S_DEST_WIDTH+:S_DEST_WIDTH];
                end
            end
        end

        for (i = 0; i < S_COUNT; i = i + 1) begin : gen_tuser_reg
            always @(posedge clk or posedge rst) begin
                if (rst) begin
                    tuser_reg[i*USER_WIDTH+:USER_WIDTH] <= {USER_WIDTH{1'b0}};
                end else if (ready[i]) begin
                    tuser_reg[i*USER_WIDTH+:USER_WIDTH] <= {USER_WIDTH{1'b0}};
                end else if (valid[i]) begin
                    tuser_reg[i*USER_WIDTH+:USER_WIDTH] <= tuser[i*USER_WIDTH+:USER_WIDTH];
                end
            end
        end
    endgenerate

    assign dest_bits = tdest_reg;

    generate
        for (i = 0; i < M_COUNT; i = i + 1) begin : gen_m_axis
            assign m_axis_tdata[i*DATA_WIDTH+:DATA_WIDTH] = (grant == (1 << i)) ? s_axis_tdata[valid_reg*DATA_WIDTH+:DATA_WIDTH] : {DATA_WIDTH{1'b0}};
            assign m_axis_tkeep[i*KEEP_WIDTH+:KEEP_WIDTH] = (grant == (1 << i)) ? s_axis_tkeep[valid_reg*KEEP_WIDTH+:KEEP_WIDTH] : {KEEP_WIDTH{1'b0}};
            assign m_axis_tvalid[i] = |(grant & valid_reg);
            assign m_axis_tlast[i] = (grant == (1 << i)) ? last_reg[valid_reg] : 1'b0;
            assign m_axis_tid[i*M_ID_WIDTH+:M_ID_WIDTH] = (grant == (1 << i)) ? tid_reg[valid_reg*S_ID_WIDTH+:S_ID_WIDTH] : {M_ID_WIDTH{1'b0}};
            assign m_axis_tdest[i*M_DEST_WIDTH+:M_DEST_WIDTH] = (grant == (1 << i)) ? tdest_reg[valid_reg*S_DEST_WIDTH+:S_DEST_WIDTH] : {M_DEST_WIDTH{1'b0}};
            assign m_axis_tuser[i*USER_WIDTH+:USER_WIDTH] = (grant == (1 << i)) ? tuser_reg[valid_reg*USER_WIDTH+:USER_WIDTH] : {USER_WIDTH{1'b0}};
        end
    endgenerate

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            arb_ptr <= {$clog2(S_COUNT){1'b0}};
            grant <= {S_COUNT{1'b0}};
        end else begin
            if (|valid_reg) begin
                grant <= 1 << arb_ptr;
                if (grant[arb_ptr]) begin
                    arb_ptr <= (arb_ptr == S_COUNT-1) ? {$clog2(S_COUNT){1'b0}} : arb_ptr + 1;
                end
            end else begin
                grant <= {S_COUNT{1'b0}};
            end
        end
    end

    assign s_axis_tready = grant;

endmodule
```
