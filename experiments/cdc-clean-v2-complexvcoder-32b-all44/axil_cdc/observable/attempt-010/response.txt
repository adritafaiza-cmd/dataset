    input  wire                   s_rd_en,
    input  wire [DATA_WIDTH-1:0]  s_data,
    output wire                   s_full,
    output wire                   s_empty,
    input  wire                   m_clk,
    input  wire                   m_rst,
    input  wire                   m_wr_en,
    input  wire                   m_rd_en,
    output wire [DATA_WIDTH-1:0]  m_data,
    output wire                   m_full,
    output wire                   m_empty
);

    reg [DATA_WIDTH-1:0] fifo [0:15]; // 16-entry FIFO
    reg [3:0] s_wptr, s_rptr;
    reg [3:0] m_wptr, m_rptr;
    reg [3:0] s_wptr_gray, s_rptr_gray;
    reg [3:0] m_wptr_gray, m_rptr_gray;
    reg [3:0] s_wptr_sync, s_rptr_sync;
    reg [3:0] m_wptr_sync, m_rptr_sync;

    // Binary to Gray conversion
    function [3:0] bin2gray;
        input [3:0] bin;
        bin2gray = bin ^ (bin >> 1);
    endfunction

    // Gray to Binary conversion
    function [3:0] gray2bin;
        input [3:0] gray;
        reg [3:0] bin;
        bin[3] = gray[3];
        bin[2] = gray[3] ^ gray[2];
        bin[1] = gray[2] ^ gray[1];
        bin[0] = gray[1] ^ gray[0];
        gray2bin = bin;
    endfunction

    // Write pointer logic on s_clk
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wptr <= 4'b0;
        end else if (s_wr_en & ~s_full) begin
            s_wptr <= s_wptr + 1;
        end
    end

    // Read pointer logic on s_clk
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_rptr <= 4'b0;
        end else if (s_rd_en & ~s_empty) begin
            s_rptr <= s_rptr + 1;
        end
    end

    // Write pointer logic on m_clk
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_wptr <= 4'b0;
        end else if (m_wr_en & ~m_full) begin
            m_wptr <= m_wptr + 1;
        end
    end

    // Read pointer logic on m_clk
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rptr <= 4'b0;
        end else if (m_rd_en & ~m_empty) begin
            m_rptr <= m_rptr + 1;
        end
    end

    // Gray code conversion and synchronization
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wptr_gray <= 4'b0;
            s_rptr_gray <= 4'b0;
        end else begin
            s_wptr_gray <= bin2gray(s_wptr);
            s_rptr_gray <= bin2gray(s_rptr);
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_wptr_gray <= 4'b0;
            m_rptr_gray <= 4'b0;
        end else begin
            m_wptr_gray <= bin2gray(m_wptr);
            m_rptr_gray <= bin2gray(m_rptr);
        end
    end

    // Synchronize pointers across clock domains
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_wptr_sync <= 4'b0;
            s_rptr_sync <= 4'b0;
        end else begin
            s_wptr_sync <= s_wptr_gray;
            s_rptr_sync <= s_rptr_gray;
        end
    end

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_wptr_sync <= 4'b0;
            m_rptr_sync <= 4'b0;
        end else begin
            m_wptr_sync <= m_wptr_gray;
            m_rptr_sync <= m_rptr_gray;
        end
    end

    // FIFO full and empty flags
    assign s_full = (s_wptr_gray == bin2gray(s_rptr + 1));
    assign s_empty = (s_wptr_gray == s_rptr_gray);

    assign m_full = (m_wptr_gray == bin2gray(m_rptr + 1));
    assign m_empty = (m_wptr_gray == m_rptr_gray);

    // FIFO write and read
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            fifo <= '{default:0};
        end else if (s_wr_en & ~s_full) begin
            fifo[s_wptr] <= s_data;
        end
    end

    assign m_data = fifo[m_rptr];

endmodule
