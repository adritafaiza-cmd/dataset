    input  wire [2:0]             s_axil_awprot,
    input  wire                   s_axil_awvalid,
    output wire                   s_axil_awready,
    input  wire [DATA_WIDTH-1:0]  s_axil_wdata,
    input  wire [STRB_WIDTH-1:0]  s_axil_wstrb,

    output wire [1:0]             s_axil_bresp,
    output wire                   s_axil_bvalid,
    input  wire                   s_axil_bready,
    input  wire [ADDR_WIDTH-1:0]  s_axil_araddr,
    input  wire [2:0]             s_axil_arprot,
    input  wire                   s_axil_arvalid,
    output wire                   s_axil_arready,
    output wire [DATA_WIDTH-1:0]  s_axil_rdata,
    output wire [1:0]             s_axil_rresp,
    output wire                   s_axil_rvalid,
    input  wire                   s_axil_rready,
    /*
     * AXI lite master interface
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [ADDR_WIDTH-1:0]  m_axil_awaddr,
    output wire [2:0]             m_axil_awprot,
    output wire                   m_axil_awvalid,
    input  wire                   m_axil_awready,

module cdc_fifo #
(
    parameter DATA_WIDTH = 32
)
(
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  in_data,
    input  wire                   in_valid,
    output wire                   in_ready,
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  out_data,
    output wire                   out_valid,
    input  wire                   out_ready
);

    reg [DATA_WIDTH-1:0] s_data_reg;
    reg                  s_valid_reg;
    reg [DATA_WIDTH-1:0] m_data_reg;
    reg                  m_valid_reg;
    reg                  m_ready_reg;

    // Synchronous domain (s_clk)
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_data_reg  <= {DATA_WIDTH{1'b0}};
            s_valid_reg <= 1'b0;
        end else begin
            if (in_valid && !s_valid_reg) begin
                s_data_reg  <= in_data;
                s_valid_reg <= 1'b1;
            end else if (m_ready_reg) begin
                s_valid_reg <= 1'b0;
            end
        end
    end

    assign in_ready = !s_valid_reg;

    // Cross-domain synchronization
    reg [1:0] sync_valid;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            sync_valid <= 2'b00;
            m_data_reg <= {DATA_WIDTH{1'b0}};
            m_valid_reg <= 1'b0;
        end else begin
            sync_valid[0] <= s_valid_reg;
            sync_valid[1] <= sync_valid[0];

            if (!m_valid_reg && sync_valid[1]) begin
                m_data_reg  <= s_data_reg;
                m_valid_reg <= 1'b1;
            end else if (out_ready) begin
                m_valid_reg <= 1'b0;
            end
        end
    end

    assign out_data = m_data_reg;
    assign out_valid = m_valid_reg;

    // Back-pressure signal
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_ready_reg <= 1'b0;
        end else begin
            m_ready_reg <= out_ready;
        end
    end

endmodule
