module cdc_fifo #
(
    parameter DATA_WIDTH = 32
)
(
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  s_data,
    input  wire                   s_valid,
    output wire                   s_ready,
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_data,
    output wire                   m_valid,
    input  wire                   m_ready
);

    /*
     * FIFO depth
     */
    localparam DEPTH = 16;
    localparam ADDR_WIDTH = $clog2(DEPTH);

    /*
     * FIFO pointers and flags
     */
    reg [ADDR_WIDTH:0] s_wr_ptr;
    reg [ADDR_WIDTH:0] s_rd_ptr;
    reg [ADDR_WIDTH:0] m_wr_ptr;
    reg [ADDR_WIDTH:0] m_rd_ptr;
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];
    reg [ADDR_WIDTH:0] s_count;
    reg [ADDR_WIDTH:0] m_count;

    /*
     * FIFO flags
     */
    wire full;
    wire empty;

    /*
     * FIFO flags logic
     */
    assign full  = (s_count == DEPTH);
    assign empty = (m_count == 0);

    /*
     * FIFO ready signals
     */
    assign s_ready = ~full;
    assign m_valid = ~empty;

    /*
     * FIFO write logic
     */
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr  <= 0;
            s_rd_ptr  <= 0;
            s_count   <= 0;
        end else if (s_valid && s_ready) begin
            mem[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_data;
            s_wr_ptr <= s_wr_ptr + 1;
            s_count  <= s_count + 1;
        end
    end

    /*
     * FIFO read logic
     */
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_wr_ptr  <= 0;
            m_rd_ptr  <= 0;
            m_count   <= 0;
        end else if (m_valid && m_ready) begin
            m_rd_ptr <= m_rd_ptr + 1;
            m_count  <= m_count - 1;
        end
    end

    /*
     * FIFO data output
     */
    assign m_data = mem[m_rd_ptr[ADDR_WIDTH-1:0]];

endmodule
