module cdc_sync #
(
    parameter WIDTH = 1
)
(
    input wire s_clk,
    input wire s_rst,
    input wire m_clk,
    input wire m_rst,
    input wire [WIDTH-1:0] din,
    output reg [WIDTH-1:0] dout
);

    reg [WIDTH-1:0] dff1;
    reg [WIDTH-1:0] dff2;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            dff1 <= 0;
        end else begin
            dff1 <= din;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            dff2 <= 0;
            dout <= 0;
        end else begin
            dff2 <= dff1;
            dout <= dff2;
        end
    end

endmodule
