// Asynchronous FIFO module
module async_fifo #
(
    parameter DATA_WIDTH = 32
)
(
    input wire                  clk_wr,
    input wire                  rst_wr,
    input wire [DATA_WIDTH-1:0] din,
    input wire                  wr_en,
    output wire                 full,
    output wire                 empty,
    input wire                  clk_rd,
    input wire                  rst_rd,
    output wire [DATA_WIDTH-1:0] dout,
    input wire                  rd_en
);

    localparam DEPTH = 4;
    localparam ADDR_WIDTH = $clog2(DEPTH);

    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];
    reg [ADDR_WIDTH-1:0] wr_ptr;
    reg [ADDR_WIDTH-1:0] rd_ptr;
    reg [ADDR_WIDTH:0]   wr_count;
    reg [ADDR_WIDTH:0]   rd_count;

    // Write pointer and count
    always @(posedge clk_wr or posedge rst_wr) begin
        if (rst_wr) begin
            wr_ptr <= 0;
            wr_count <= 0;
        end else if (wr_en && !full) begin
            mem[wr_ptr] <= din;
            wr_ptr <= wr_ptr + 1;
            wr_count <= wr_count + 1;
        end
    end

    // Read pointer and count
    always @(posedge clk_rd or posedge rst_rd) begin
        if (rst_rd) begin
            rd_ptr <= 0;
            rd_count <= 0;
        end else if (rd_en && !empty) begin
            rd_ptr <= rd_ptr + 1;
            rd_count <= rd_count + 1;
        end
    end

    // Full flag
    assign full = (wr_count == DEPTH);

    // Empty flag
    assign empty = (wr_count == rd_count);

    // Data output
    reg [DATA_WIDTH-1:0] dout_reg;
    always @(posedge clk_rd or posedge rst_rd) begin
        if (rst_rd) begin
            dout_reg <= 0;
        end else if (rd_en && !empty) begin
            dout_reg <= mem[rd_ptr];
        end
    end

    assign dout = dout_reg;

endmodule
