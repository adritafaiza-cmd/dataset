module cdc_handshake #(
    parameter WIDTH = 1
)
(
    input  wire s_clk,
    input  wire s_rst,
    input  wire m_clk,
    input  wire m_rst,
    input  wire s_valid,
    output wire s_ready,
    output wire m_valid,
    input  wire m_ready
);

    reg [WIDTH-1:0] s_valid_reg;
    reg [WIDTH-1:0] m_valid_reg;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_valid_reg <= 0;
        else
            s_valid_reg <= s_valid;
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_valid_reg <= 0;
        else
            m_valid_reg <= s_valid_reg;
    end

    assign s_ready = m_valid_reg & m_ready;
    assign m_valid = s_valid_reg;

endmodule

module cdc_data #(
    parameter WIDTH = 32
)
(
    input  wire s_clk,
    input  wire s_rst,
    input  wire m_clk,
    input  wire m_rst,
    input  wire [WIDTH-1:0] s_data,
    output wire [WIDTH-1:0] m_data
);

    reg [WIDTH-1:0] s_data_reg1;
    reg [WIDTH-1:0] s_data_reg2;
    reg [WIDTH-1:0] m_data_reg1;
    reg [WIDTH-1:0] m_data_reg2;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_data_reg1 <= 0;
            s_data_reg2 <= 0;
        end else begin
            s_data_reg1 <= s_data;
            s_data_reg2 <= s_data_reg1;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_data_reg1 <= 0;
            m_data_reg2 <= 0;
        end else begin
            m_data_reg1 <= s_data_reg2;
            m_data_reg2 <= m_data_reg1;
        end
    end

    assign m_data = m_data_reg2;

endmodule
