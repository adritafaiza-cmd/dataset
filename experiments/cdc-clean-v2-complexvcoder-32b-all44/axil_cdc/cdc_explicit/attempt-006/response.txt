module cdc_handshake
(
    input  wire s_clk,
    input  wire s_rst,
    input  wire s_valid,
    output wire s_ready,
    input  wire m_clk,
    input  wire m_rst,
    input  wire m_ready,
    output wire m_valid
);

    reg s_valid_r, s_ready_r;
    reg m_valid_r, m_ready_r;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_valid_r <= 0;
        else
            s_valid_r <= s_valid;
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_ready_r <= 0;
        else
            m_ready_r <= m_ready;
    end

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_ready_r <= 0;
        else
            s_ready_r <= m_ready_r & ~s_valid_r;
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_valid_r <= 0;
        else
            m_valid_r <= s_valid_r & m_ready_r;
    end

    assign s_ready = s_ready_r;
    assign m_valid = m_valid_r;

endmodule

module cdc_sync #(
    parameter WIDTH = 1
)
(
    input  wire s_clk,
    input  wire s_rst,
    input  wire [WIDTH-1:0] s_in,
    input  wire m_clk,
    input  wire m_rst,
    output wire [WIDTH-1:0] m_out
);

    reg [WIDTH-1:0] s_in_r1, s_in_r2;
    reg [WIDTH-1:0] m_out_r1, m_out_r2;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_in_r1 <= 0;
            s_in_r2 <= 0;
        end else begin
            s_in_r1 <= s_in;
            s_in_r2 <= s_in_r1;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_out_r1 <= 0;
            m_out_r2 <= 0;
        end else begin
            m_out_r1 <= s_in_r2;
            m_out_r2 <= m_out_r1;
        end
    end

    assign m_out = m_out_r2;

endmodule
