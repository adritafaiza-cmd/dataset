module async_fifo #(
    parameter WIDTH = 32,
    parameter DEPTH = 2
)
(
    input  wire wr_clk,
    input  wire wr_rst,
    input  wire [WIDTH-1:0] wr_data,
    input  wire wr_en,
    output wire wr_full,
    input  wire rd_clk,
    input  wire rd_rst,
    output wire [WIDTH-1:0] rd_data,
    input  wire rd_en,
    output wire rd_empty
);

    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [$clog2(DEPTH)-1:0] wr_ptr;
    reg [$clog2(DEPTH)-1:0] rd_ptr;
    reg [$clog2(DEPTH):0] count;

    reg [WIDTH-1:0] wr_data_reg;
    reg [WIDTH-1:0] rd_data_reg;

    always @(posedge wr_clk or posedge wr_rst) begin
        if (wr_rst) begin
            wr_ptr <= 0;
            count <= 0;
        end else if (wr_en && (count < DEPTH)) begin
            mem[wr_ptr] <= wr_data;
            wr_ptr <= wr_ptr + 1;
            count <= count + 1;
        end
    end

    always @(posedge rd_clk or posedge rd_rst) begin
        if (rd_rst) begin
            rd_ptr <= 0;
            count <= 0;
        end else if (rd_en && (count > 0)) begin
            rd_ptr <= rd_ptr + 1;
            count <= count - 1;
        end
    end

    always @(posedge wr_clk or posedge wr_rst) begin
        if (wr_rst)
            wr_data_reg <= 0;
        else if (wr_en && (count < DEPTH))
            wr_data_reg <= wr_data;
    end

    always @(posedge rd_clk or posedge rd_rst) begin
        if (rd_rst)
            rd_data_reg <= 0;
        else if (rd_en && (count > 0))
            rd_data_reg <= mem[rd_ptr];
    end

    assign wr_full = (count == DEPTH);
    assign rd_empty = (count == 0);
    assign rd_data = rd_data_reg;

endmodule
