```verilog
module axi_dma #
(
    parameter AXI_DATA_WIDTH = 32,
    parameter AXI_ADDR_WIDTH = 16,
    parameter AXI_STRB_WIDTH = (AXI_DATA_WIDTH/8),
    parameter AXI_ID_WIDTH = 8,
    parameter AXI_MAX_BURST_LEN = 16,
    parameter AXIS_DATA_WIDTH = AXI_DATA_WIDTH,
    parameter AXIS_KEEP_ENABLE = (AXIS_DATA_WIDTH>8),
    parameter AXIS_KEEP_WIDTH = (AXIS_DATA_WIDTH/8),
    parameter AXIS_LAST_ENABLE = 1,
    parameter AXIS_ID_ENABLE = 0,
    parameter AXIS_ID_WIDTH = 8,
    parameter AXIS_DEST_ENABLE = 0,
    parameter AXIS_DEST_WIDTH = 8,
    parameter AXIS_USER_ENABLE = 1,
    parameter AXIS_USER_WIDTH = 1,
    parameter LEN_WIDTH = 20,
    parameter TAG_WIDTH = 8,
    parameter ENABLE_SG = 0,
    parameter ENABLE_UNALIGNED = 0
)
(
    input  wire                       clk,
    input  wire                       rst,
    /*
     * AXI read descriptor input
     */
    input  wire [AXI_ADDR_WIDTH-1:0]  s_axis_read_desc_addr,
    input  wire [LEN_WIDTH-1:0]       s_axis_read_desc_len,
    input  wire [TAG_WIDTH-1:0]       s_axis_read_desc_tag,
    input  wire [AXIS_ID_WIDTH-1:0]   s_axis_read_desc_id,
    input  wire [AXIS_DEST_WIDTH-1:0] s_axis_read_desc_dest,
    input  wire [AXIS_USER_WIDTH-1:0] s_axis_read_desc_user,
    input  wire                       s_axis_read_desc_valid,
    output wire                       s_axis_read_desc_ready,
    /*
     * AXI read descriptor status output
     */
    output wire [TAG_WIDTH-1:0]       m_axis_read_desc_status_tag,
    output wire [3:0]                 m_axis_read_desc_status_error,
    output wire                       m_axis_read_desc_status_valid,
    /*
     * AXI stream read data output
     */
    output wire [AXIS_DATA_WIDTH-1:0] m_axis_read_data_tdata,
    output wire [AXIS_KEEP_WIDTH-1:0] m_axis_read_data_tkeep,
    output wire                       m_axis_read_data_tvalid,
    input  wire                       m_axis_read_data_tready,
    output wire                       m_axis_read_data_tlast,
    output wire [AXIS_ID_WIDTH-1:0]   m_axis_read_data_tid,
    output wire [AXIS_DEST_WIDTH-1:0] m_axis_read_data_tdest,
    output wire [AXIS_USER_WIDTH-1:0] m_axis_read_data_tuser,
    /*
     * AXI write descriptor input
     */
    input  wire [AXI_ADDR_WIDTH-1:0]  s_axis_write_desc_addr,
    input  wire [LEN_WIDTH-1:0]       s_axis_write_desc_len,
    input  wire [TAG_WIDTH-1:0]       s_axis_write_desc_tag,
    input  wire                       s_axis_write_desc_valid,
    output wire                       s_axis_write_desc_ready,
    /*
     * AXI write descriptor status output
     */
    output wire [LEN_WIDTH-1:0]       m_axis_write_desc_status_len,
    output wire [TAG_WIDTH-1:0]       m_axis_write_desc_status_tag,
    output wire [AXIS_ID_WIDTH-1:0]   m_axis_write_desc_status_id,
    output wire [AXIS_DEST_WIDTH-1:0] m_axis_write_desc_status_dest,
    output wire [AXIS_USER_WIDTH-1:0] m_axis_write_desc_status_user,
    output wire [3:0]                 m_axis_write_desc_status_error,
    output wire                       m_axis_write_desc_status_valid,
    /*
     * AXI stream write data input
     */
    input  wire [AXIS_DATA_WIDTH-1:0] s_axis_write_data_tdata,
    input  wire [AXIS_KEEP_WIDTH-1:0] s_axis_write_data_tkeep,
    input  wire                       s_axis_write_data_tvalid,
    output wire                       s_axis_write_data_tready,
    input  wire                       s_axis_write_data_tlast,
    input  wire [AXIS_ID_WIDTH-1:0]   s_axis_write_data_tid,
    input  wire [AXIS_DEST_WIDTH-1:0] s_axis_write_data_tdest,
    input  wire [AXIS_USER_WIDTH-1:0] s_axis_write_data_tuser,
    /*
     * AXI master interface
     */
    output wire [AXI_ID_WIDTH-1:0]    m_axi_awid,
    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_awaddr,
    output wire [7:0]                 m_axi_awlen,
    output wire [2:0]                 m_axi_awsize,
    output wire [1:0]                 m_axi_awburst,
    output wire                       m_axi_awlock,
    output wire [3:0]                 m_axi_awcache,
    output wire [2:0]                 m_axi_awprot,
    output wire                       m_axi_awvalid,
    input  wire                       m_axi_awready,
    output wire [AXI_DATA_WIDTH-1:0]  m_axi_wdata,
    output wire [AXI_STRB_WIDTH-1:0]  m_axi_wstrb,
    output wire                       m_axi_wlast,
    output wire                       m_axi_wvalid,
    input  wire                       m_axi_wready,
    input  wire [AXI_ID_WIDTH-1:0]    m_axi_bid,
    input  wire [1:0]                 m_axi_bresp,
    input  wire                       m_axi_bvalid,
    output wire                       m_axi_bready,
    output wire [AXI_ID_WIDTH-1:0]    m_axi_arid,
    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_araddr,
    output wire [7:0]                 m_axi_arlen,
    output wire [2:0]                 m_axi_arsize,
    output wire [1:0]                 m_axi_arburst,
    output wire                       m_axi_arlock,
    output wire [3:0]                 m_axi_arcache,
    output wire [2:0]                 m_axi_arprot,
    output wire                       m_axi_arvalid,
    input  wire                       m_axi_arready,
    input  wire [AXI_ID_WIDTH-1:0]    m_axi_rid,
    input  wire [AXI_DATA_WIDTH-1:0]  m_axi_rdata,
    input  wire [1:0]                 m_axi_rresp,
    input  wire                       m_axi_rlast,
    input  wire                       m_axi_rvalid,
    output wire                       m_axi_rready,
    /*
     * Configuration
     */
    input  wire                       read_enable,
    input  wire                       write_enable,
    input  wire                       write_abort
);

    // Internal signals
    reg [AXI_ADDR_WIDTH-1:0]          read_addr_reg;
    reg [LEN_WIDTH-1:0]               read_len_reg;
    reg [TAG_WIDTH-1:0]               read_tag_reg;
    reg [AXIS_ID_WIDTH-1:0]           read_id_reg;
    reg [AXIS_DEST_WIDTH-1:0]         read_dest_reg;
    reg [AXIS_USER_WIDTH-1:0]         read_user_reg;
    reg [AXI_ADDR_WIDTH-1:0]          write_addr_reg;
    reg [LEN_WIDTH-1:0]               write_len_reg;
    reg [TAG_WIDTH-1:0]               write_tag_reg;
    reg [AXIS_ID_WIDTH-1:0]           write_id_reg;
    reg [AXIS_DEST_WIDTH-1:0]         write_dest_reg;
    reg [AXIS_USER_WIDTH-1:0]         write_user_reg;
    reg [AXI_ADDR_WIDTH-1:0]          read_current_addr;
    reg [LEN_WIDTH-1:0]               read_current_len;
    reg [AXI_ADDR_WIDTH-1:0]          write_current_addr;
    reg [LEN_WIDTH-1:0]               write_current_len;
    reg [7:0]                         read_burst_len;
    reg [7:0]                         write_burst_len;
    reg                               read_active;
    reg                               write_active;
    reg [3:0]                         read_error;
    reg [3:0]                         write_error;
    reg                               read_desc_ready;
    reg                               write_desc_ready;
    reg                               read_status_valid;
    reg                               write_status_valid;
    reg [AXIS_DATA_WIDTH-1:0]         read_data;
    reg [AXIS_KEEP_WIDTH-1:0]         read_keep;
    reg                               read_last;
    reg [AXIS_ID_WIDTH-1:0]           read_id;
    reg [AXIS_DEST_WIDTH-1:0]         read_dest;
    reg [AXIS_USER_WIDTH-1:0]         read_user;
    reg                               read_data_valid;
    reg [AXIS_DATA_WIDTH-1:0]         write_data;
    reg [AXIS_KEEP_WIDTH-1:0]         write_keep;
    reg                               write_last;
    reg                               write_data_valid;
    reg                               write_data_ready;
    reg [AXIS_ID_WIDTH-1:0]           write_id;
    reg [AXIS_DEST_WIDTH-1:0]         write_dest;
    reg [AXIS_USER_WIDTH-1:0]         write_user;

    // AXI read descriptor FIFO
    reg [AXI_ADDR_WIDTH-1:0]          read_desc_fifo_addr [0:AXI_MAX_BURST_LEN-1];
    reg [LEN_WIDTH-1:0]               read_desc_fifo_len [0:AXI_MAX_BURST_LEN-1];
    reg [TAG_WIDTH-1:0]               read_desc_fifo_tag [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_ID_WIDTH-1:0]           read_desc_fifo_id [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_DEST_WIDTH-1:0]         read_desc_fifo_dest [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_USER_WIDTH-1:0]         read_desc_fifo_user [0:AXI_MAX_BURST_LEN-1];
    reg [7:0]                         read_desc_fifo_wptr;
    reg [7:0]                         read_desc_fifo_rptr;
    reg [7:0]                         read_desc_fifo_count;

    // AXI write descriptor FIFO
    reg [AXI_ADDR_WIDTH-1:0]          write_desc_fifo_addr [0:AXI_MAX_BURST_LEN-1];
    reg [LEN_WIDTH-1:0]               write_desc_fifo_len [0:AXI_MAX_BURST_LEN-1];
    reg [TAG_WIDTH-1:0]               write_desc_fifo_tag [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_ID_WIDTH-1:0]           write_desc_fifo_id [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_DEST_WIDTH-1:0]         write_desc_fifo_dest [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_USER_WIDTH-1:0]         write_desc_fifo_user [0:AXI_MAX_BURST_LEN-1];
    reg [7:0]                         write_desc_fifo_wptr;
    reg [7:0]                         write_desc_fifo_rptr;
    reg [7:0]                         write_desc_fifo_count;

    // AXI read data FIFO
    reg [AXIS_DATA_WIDTH-1:0]         read_data_fifo [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_KEEP_WIDTH-1:0]         read_keep_fifo [0:AXI_MAX_BURST_LEN-1];
    reg                               read_last_fifo [0:AXI_MAX_BURST_LEN-1];
    reg [7:0]                         read_data_fifo_wptr;
    reg [7:0]                         read_data_fifo_rptr;
    reg [7:0]                         read_data_fifo_count;

    // AXI write data FIFO
    reg [AXIS_DATA_WIDTH-1:0]         write_data_fifo [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_KEEP_WIDTH-1:0]         write_keep_fifo [0:AXI_MAX_BURST_LEN-1];
    reg                               write_last_fifo [0:AXI_MAX_BURST_LEN-1];
    reg [7:0]                         write_data_fifo_wptr;
    reg [7:0]                         write_data_fifo_rptr;
    reg [7:0]                         write_data_fifo_count;

    // AXI read descriptor FIFO control
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            read_desc_fifo_wptr <= 8'h0;
            read_desc_fifo_rptr <= 8'h0;
            read_desc_fifo_count <= 8'h0;
        end else begin
            if (s_axis_read_desc_valid && read_desc_ready) begin
                read_desc_fifo_addr[read_desc_fifo_wptr] <= s_axis_read_desc_addr;
                read_desc_fifo_len[read_desc_fifo_wptr] <= s_axis_read_desc_len;
                read_desc_fifo_tag[read_desc_fifo_wptr] <= s_axis_read_desc_tag;
                read_desc_fifo_id[read_desc_fifo_wptr] <= s_axis_read_desc_id;
                read_desc_fifo_dest[read_desc_fifo_wptr] <= s_axis_read_desc_dest;
                read_desc_fifo_user[read_desc_fifo_wptr] <= s_axis_read_desc_user;
                read_desc_fifo_wptr <= read_desc_fifo_wptr + 1;
                read_desc_fifo_count <= read_desc_fifo_count + 1;
            end else if (read_active && m_axi_arready) begin
                read_desc_fifo_rptr <= read_desc_fifo_rptr + 1;
                read_desc_fifo_count <= read_desc_fifo_count - 1;
            end
        end
    end

    // AXI write descriptor FIFO control
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            write_desc_fifo_wptr <= 8'h0;
            write_desc_fifo_rptr <= 8'h0;
            write_desc_fifo_count <= 8'h0;
        end else begin
            if (s_axis_write_desc_valid && write_desc_ready) begin
                write_desc_fifo_addr[write_desc_fifo_wptr] <= s_axis_write_desc_addr;
                write_desc_fifo_len[write_desc_fifo_wptr] <= s_axis_write_desc_len;
                write_desc_fifo_tag[write_desc_fifo_wptr] <= s_axis_write_desc_tag;
                write_desc_fifo_id[write_desc_fifo_wptr] <= s_axis_write_desc_id;
                write_desc_fifo_dest[write_desc_fifo_wptr] <= s_axis_write_desc_dest;
                write_desc_fifo_user[write_desc_fifo_wptr] <= s_axis_write_desc_user;
                write_desc_fifo_wptr <= write_desc_fifo_wptr + 1;
                write_desc_fifo_count <= write_desc_fifo_count + 1;
            end else if (write_active && m_axi_awready) begin
                write_desc_fifo_rptr <= write_desc_fifo_rptr + 1;
                write_desc_fifo_count <= write_desc_fifo_count - 1;
            end
        end
    end

    // AXI read data FIFO control
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            read_data_fifo_wptr <= 8'h0;
            read_data_fifo_rptr <= 8'h0;
            read_data_fifo_count <= 8'h0;
        end else begin
            if (m_axi_rvalid && m_axi_rready) begin
                read_data_fifo[read_data_fifo_wptr] <= m_axi_rdata;
                read_keep_fifo[read_data_fifo_wptr] <= {AXIS_KEEP_WIDTH{1'b1}};
                read_last_fifo[read_data_fifo_wptr] <= m_axi_rlast;
                read_data_fifo_wptr <= read_data_fifo_wptr + 1;
                read_data_fifo_count <= read_data_fifo_count + 1;
            end else if (m_axis_read_data_tready && read_data_valid) begin
                read_data_fifo_rptr <= read_data_fifo_rptr + 1;
                read_data_fifo_count <= read_data_fifo_count - 1;
            end
        end
    end

    // AXI write data FIFO control
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            write_data_fifo_wptr <= 8'h0;
            write_data_fifo_rptr <= 8'h0;
            write_data_fifo_count <= 8'h0;
        end else begin
            if (s_axis_write_data_tvalid && write_data_ready) begin
                write_data_fifo[write_data_fifo_wptr] <= s_axis_write_data_tdata;
                write_keep_fifo[write_data_fifo_wptr] <= s_axis_write_data_tkeep;
                write_last_fifo[write_data_fifo_wptr] <= s_axis_write_data_tlast;
                write_data_fifo_wptr <= write_data_fifo_wptr + 1;
                write_data_fifo_count <= write_data_fifo_count + 1;
            end else if (m_axi_wready && write_data_valid) begin
                write_data_fifo_rptr <= write_data_fifo_rptr + 1;
                write_data_fifo_count <= write_data_fifo_count - 1;
            end
        end
    end

    // AXI read descriptor FIFO read logic
    assign read_addr_reg = read_desc_fifo_addr[read_desc_fifo_rptr];
    assign read_len_reg = read_desc_fifo_len[read_desc_fifo_rptr];
    assign read_tag_reg = read_desc_fifo_tag[read_desc_fifo_rptr];
    assign read_id_reg = read_desc_fifo_id[read_desc_fifo_rptr];
    assign read_dest_reg = read_desc_fifo_dest[read_desc_fifo_rptr];
    assign read_user_reg = read_desc_fifo_user[read_desc_fifo_rptr];

    // AXI write descriptor FIFO read logic
    assign write_addr_reg = write_desc_fifo_addr[write_desc_fifo_rptr];
    assign write_len_reg = write_desc_fifo_len[write_desc_fifo_rptr];
    assign write_tag_reg = write_desc_fifo_tag[write_desc_fifo_rptr];
    assign write_id_reg = write_desc_fifo_id[write_desc_fifo_rptr];
    assign write_dest_reg = write_desc_fifo_dest[write_desc_fifo_rptr];
    assign write_user_reg = write_desc_fifo_user[write_desc_fifo_rptr];

    // AXI read data FIFO read logic
    assign read_data = read_data_fifo[read_data_fifo_rptr];
    assign read_keep = read_keep_fifo[read_data_fifo_rptr];
    assign read_last = read_last_fifo[read_data_fifo_rptr];

    // AXI write data FIFO read logic
    assign write_data = write_data_fifo[write_data_fifo_rptr];
    assign write_keep = write_keep_fifo[write_data_fifo_rptr];
    assign write_last = write_last_fifo[write_data_fifo_rptr];

    // AXI read descriptor FIFO ready logic
    assign read_desc_ready = (read_desc_fifo_count < AXI_MAX_BURST_LEN) && read_enable;

    // AXI write descriptor FIFO ready logic
    assign write_desc_ready = (write_desc_fifo_count < AXI_MAX_BURST_LEN) && write_enable;

    // AXI read data FIFO valid logic
    assign read_data_valid = (read_data_fifo_count > 0);

    // AXI write data FIFO valid logic
    assign write_data_valid = (write_data_fifo_count > 0);

    // AXI read descriptor ready output
    assign s_axis_read_desc_ready = read_desc_ready;

    // AXI write descriptor ready output
    assign s_axis_write_desc_ready = write_desc_ready;

    // AXI read data valid output
    assign m_axis_read_data_tvalid = read_data_valid;

    // AXI write data ready output
    assign s_axis_write_data_tready = write_data_ready;

    // AXI read descriptor status output
    assign m_axis_read_desc_status_tag = read_tag_reg;
    assign m_axis_read_desc_status_error = read_error;
    assign m_axis_read_desc_status_valid = read_status_valid;

    // AXI write descriptor status output
    assign m_axis_write_desc_status_len = write_len_reg;
    assign m_axis_write_desc_status_tag = write_tag_reg;
    assign m_axis_write_desc_status_id = write_id_reg;
    assign m_axis_write_desc_status_dest = write_dest_reg;
    assign m_axis_write_desc_status_user = write_user_reg;
    assign m_axis_write_desc_status_error = write_error;
    assign m_axis_write_desc_status_valid = write_status_valid;

    // AXI read data output
    assign m_axis_read_data_tdata = read_data;
    assign m_axis_read_data_tkeep = read_keep;
    assign m_axis_read_data_tlast = read_last;
    assign m_axis_read_data_tid = read_id_reg;
    assign m_axis_read_data_tdest = read_dest_reg;
    assign m_axis_read_data_tuser = read_user_reg;

    // AXI write data output
    assign m_axi_wdata = write_data;
    assign m_axi_wstrb = write_keep;
    assign m_axi_wlast = write_last;
    assign m_axi_wvalid = write_data_valid;

    // AXI read address channel output
    assign m_axi_arid = read_id_reg;
    assign m_axi_araddr = read_current_addr;
    assign m_axi_arlen = read_burst_len;
    assign m_axi_arsize = 3'b010; // 4 bytes
    assign m_axi_arburst = 2'b01; // INCR
    assign m_axi_arlock = 1'b0;
    assign m_axi_arcache = 4'b0011; // Non-cacheable, buffered
    assign m_axi_arprot = 3'b000; // Normal, secure, data
    assign m_axi_arvalid = read_active && (read_desc_fifo_count > 0);

    // AXI write address channel output
    assign m_axi_awid = write_id_reg;
    assign m_axi_awaddr = write_current_addr;
    assign m_axi_awlen = write_burst_len;
    assign m_axi_awsize = 3'b010; // 4 bytes
    assign m_axi_awburst = 2'b01; // INCR
    assign m_axi_awlock = 1'b0;
    assign m_axi_awcache = 4'b0011; // Non-cacheable, buffered
    assign m_axi_awprot = 3'b000; // Normal, secure, data
    assign m_axi_awvalid = write_active && (write_desc_fifo_count > 0);

    // AXI read data channel ready
    assign m_axi_rready = read_active && (read_data_fifo_count < AXI_MAX_BURST_LEN);

    // AXI write data channel ready
    assign write_data_ready = write_active && (write_data_fifo_count < AXI_MAX_BURST_LEN);

    // AXI read response channel ready
    assign m_axi_bready = read_active;

    // AXI write response channel ready
    assign m_axi_bready = write_active;

    // Read state machine
    reg [2:0] read_state;
    localparam READ_IDLE = 3'b000;
    localparam READ_ADDR = 3'b001;
    localparam READ_DATA = 3'b010;
    localparam READ_STATUS = 3'b011;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            read_state <= READ_IDLE;
            read_active <= 1'b0;
            read_current_addr <= {AXI_ADDR_WIDTH{1'b0}};
            read_current_len <= {LEN_WIDTH{1'b0}};
            read_burst_len <= 8'h0;
            read_error <= 4'h0;
            read_status_valid <= 1'b0;
        end else begin
            case (read_state)
                READ_IDLE: begin
                    if (read_desc_fifo_count > 0 && read_enable) begin
                        read_active <= 1'b1;
                        read_current_addr <= read_addr_reg;
                        read_current_len <= read_len_reg;
                        read_burst_len <= (read_len_reg < AXI_MAX_BURST_LEN) ? read_len_reg[7:0] : AXI_MAX_BURST_LEN-1;
                        read_state <= READ_ADDR;
                    end
                end
                READ_ADDR: begin
                    if (m_axi_arready) begin
                        read_state <= READ_DATA;
                    end
                end
                READ_DATA: begin
                    if (m_axi_rvalid) begin
                        read_current_addr <= read_current_addr + AXI_DATA_WIDTH/8;
                        read_current_len <= read_current_len - AXI_DATA_WIDTH/8;
                        if (read_current_len == AXI_DATA_WIDTH/8) begin
                            read_state <= READ_STATUS;
                        end
                    end
                end
                READ_STATUS: begin
                    if (m_axi_rlast) begin
                        read_error <= (m_axi_rresp != 2'b00) ? 4'h1 : 4'h0;
                        read_status_valid <= 1'b1;
                        read_state <= READ_IDLE;
                        read_active <= 1'b0;
                    end
                end
            endcase
        end
    end

    // Write state machine
    reg [2:0] write_state;
    localparam WRITE_IDLE = 3'b000;
    localparam WRITE_ADDR = 3'b001;
    localparam WRITE_DATA = 3'b010;
    localparam WRITE_STATUS = 3'b011;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            write_state <= WRITE_IDLE;
            write_active <= 1'b0;
            write_current_addr <= {AXI_ADDR_WIDTH{1'b0}};
            write_current_len <= {LEN_WIDTH{1'b0}};
            write_burst_len <= 8'h0;
            write_error <= 4'h0;
            write_status_valid <= 1'b0;
        end else begin
            case (write_state)
                WRITE_IDLE: begin
                    if (write_desc_fifo_count > 0 && write_enable) begin
                        write_active <= 1'b1;
                        write_current_addr <= write_addr_reg;
                        write_current_len <= write_len_reg;
                        write_burst_len <= (write_len_reg < AXI_MAX_BURST_LEN) ? write_len_reg[7:0] : AXI_MAX_BURST_LEN-1;
                        write_state <= WRITE_ADDR;
                    end
                end
                WRITE_ADDR: begin
                    if (m_axi_awready) begin
                        write_state <= WRITE_DATA;
                    end
                end
                WRITE_DATA: begin
                    if (m_axi_wready) begin
                        write_current_addr <= write_current_addr + AXI_DATA_WIDTH/8;
                        write_current_len <= write_current_len - AXI_DATA_WIDTH/8;
                        if (write_current_len == AXI_DATA_WIDTH/8) begin
                            write_state <= WRITE_STATUS;
                        end
                    end
                end
                WRITE_STATUS: begin
                    if (m_axi_bvalid) begin
                        write_error <= (m_axi_bresp != 2'b00) ? 4'h1 : 4'h0;
                        write_status_valid <= 1'b1;
                        write_state <= WRITE_IDLE;
                        write_active <= 1'b0;
                    end
                end
            endcase
        end
    end

    // Read status valid clear
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            read_status_valid <= 1'b0;
        end else begin
            if (read_status_valid && m_axis_read_desc_status_valid) begin
                read_status_valid <= 1'b0;
            end
        end
    end

    // Write status valid clear
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            write_status_valid <= 1'b0;
        end else begin
            if (write_status_valid && m_axis_write_desc_status_valid) begin
                write_status_valid <= 1'b0;
            end
        end
    end

endmodule
```
