module read_desc_handler #(
    parameter AXI_ADDR_WIDTH = 16,
    parameter LEN_WIDTH = 20,
    parameter TAG_WIDTH = 8,
    parameter AXIS_ID_WIDTH = 8,
    parameter AXIS_DEST_WIDTH = 8,
    parameter AXIS_USER_WIDTH = 1
)
(
    input  wire                       clk,
    input  wire                       rst,
    input  wire [AXI_ADDR_WIDTH-1:0]  s_axis_read_desc_addr,
    input  wire [LEN_WIDTH-1:0]       s_axis_read_desc_len,
    input  wire [TAG_WIDTH-1:0]       s_axis_read_desc_tag,
    input  wire [AXIS_ID_WIDTH-1:0]   s_axis_read_desc_id,
    input  wire [AXIS_DEST_WIDTH-1:0] s_axis_read_desc_dest,
    input  wire [AXIS_USER_WIDTH-1:0] s_axis_read_desc_user,
    input  wire                       s_axis_read_desc_valid,
    output wire                       s_axis_read_desc_ready,
    output wire [AXI_ADDR_WIDTH-1:0]  read_desc_addr,
    output wire [LEN_WIDTH-1:0]       read_desc_len,
    output wire [TAG_WIDTH-1:0]       read_desc_tag,
    output wire [AXIS_ID_WIDTH-1:0]   read_desc_id,
    output wire [AXIS_DEST_WIDTH-1:0] read_desc_dest,
    output wire [AXIS_USER_WIDTH-1:0] read_desc_user,
    output wire                       read_desc_valid,
    input  wire                       read_desc_ready,
    output wire [3:0]                 read_desc_error,
    output wire                       read_desc_status_valid,
    input  wire                       read_enable
);

    reg [AXI_ADDR_WIDTH-1:0]  desc_addr_reg;
    reg [LEN_WIDTH-1:0]       desc_len_reg;
    reg [TAG_WIDTH-1:0]       desc_tag_reg;
    reg [AXIS_ID_WIDTH-1:0]   desc_id_reg;
    reg [AXIS_DEST_WIDTH-1:0] desc_dest_reg;
    reg [AXIS_USER_WIDTH-1:0] desc_user_reg;
    reg                       desc_valid_reg;
    reg                       desc_status_valid_reg;
    reg [3:0]                 desc_error_reg;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            desc_addr_reg <= 0;
            desc_len_reg <= 0;
            desc_tag_reg <= 0;
            desc_id_reg <= 0;
            desc_dest_reg <= 0;
            desc_user_reg <= 0;
            desc_valid_reg <= 0;
            desc_status_valid_reg <= 0;
            desc_error_reg <= 0;
        end else if (read_enable) begin
            if (s_axis_read_desc_valid && read_desc_ready) begin
                desc_addr_reg <= s_axis_read_desc_addr;
                desc_len_reg <= s_axis_read_desc_len;
                desc_tag_reg <= s_axis_read_desc_tag;
                desc_id_reg <= s_axis_read_desc_id;
                desc_dest_reg <= s_axis_read_desc_dest;
                desc_user_reg <= s_axis_read_desc_user;
                desc_valid_reg <= 1;
                desc_status_valid_reg <= 0;
                desc_error_reg <= 0;
            end else if (read_desc_ready) begin
                desc_valid_reg <= 0;
                desc_status_valid_reg <= 1;
            end
        end
    end

    assign s_axis_read_desc_ready = read_enable && !desc_valid_reg;
    assign read_desc_addr = desc_addr_reg;
    assign read_desc_len = desc_len_reg;
    assign read_desc_tag = desc_tag_reg;
    assign read_desc_id = desc_id_reg;
    assign read_desc_dest = desc_dest_reg;
    assign read_desc_user = desc_user_reg;
    assign read_desc_valid = desc_valid_reg && read_enable;
    assign read_desc_error = desc_error_reg;
    assign read_desc_status_valid = desc_status_valid_reg;

endmodule

    reg [AXIS_KEEP_WIDTH-1:0] rkeep_reg;
    reg                       rvalid_reg;
    reg                       rlast_reg;

    reg [TAG_WIDTH-1:0]       tag_reg;
    reg [3:0]                 error_reg;
    reg                       desc_status_valid_reg;
    reg                       desc_ready_reg;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            araddr_reg <= 0;
            arlen_reg <= 0;
            arid_reg <= 0;
            arvalid_reg <= 0;
            rdata_reg <= 0;
            rkeep_reg <= 0;
            rvalid_reg <= 0;
            rlast_reg <= 0;
            rid_reg <= 0;
            tag_reg <= 0;
            error_reg <= 0;
            desc_status_valid_reg <= 0;
            desc_ready_reg <= 0;

        end

    assign read_desc_error = error_reg;
    assign read_desc_status_valid = desc_status_valid_reg;

    assign m_axi_arid = arid_reg;
    assign m_axi_araddr = araddr_reg;
    assign m_axi_arlen = arlen_reg;
    assign m_axi_arsize = 3'b010; // 32-bit
    assign m_axi_arburst = 2'b01; // INCR
    assign m_axi_arlock = 1'b0;
    assign m_axi_arcache = 4'b0011; // Non-cacheable, bufferable
    assign m_axi_arprot = 3'b000;
    assign m_axi_arvalid = arvalid_reg;
