            end

                m_axi_arvalid_reg <= 0;

                m_axi_awvalid_reg <= 0;
            end

                m_axi_wvalid_reg <= 0;
            end

            if (m_axi_bvalid && m_axi_bready) begin
                m_axi_bready_reg <= 0;
            end
        end
