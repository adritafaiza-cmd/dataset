module axi_dma #
(
    parameter AXI_DATA_WIDTH = 32,
    parameter AXI_ADDR_WIDTH = 16,
    parameter AXI_STRB_WIDTH = (AXI_DATA_WIDTH/8),
    parameter AXI_ID_WIDTH = 8,
    parameter AXI_MAX_BURST_LEN = 16,

    parameter AXIS_ID_ENABLE = 0,

    parameter AXIS_DEST_WIDTH = 8,

    input  wire [TAG_WIDTH-1:0]       s_axis_read_desc_tag,

    /*
     * AXI read descriptor status output

    output wire [AXIS_KEEP_WIDTH-1:0] m_axis_read_data_tkeep,

    input  wire                       m_axis_read_data_tready,
    output wire                       m_axis_read_data_tlast,
    output wire [AXIS_ID_WIDTH-1:0]   m_axis_read_data_tid,
    output wire [AXIS_DEST_WIDTH-1:0] m_axis_read_data_tdest,
    output wire [AXIS_USER_WIDTH-1:0] m_axis_read_data_tuser,
    /*
     * AXI write descriptor input
     */
    input  wire [AXI_ADDR_WIDTH-1:0]  s_axis_write_desc_addr,
    input  wire [LEN_WIDTH-1:0]       s_axis_write_desc_len,

    input  wire                       s_axis_write_desc_valid,
    output wire                       s_axis_write_desc_ready,
    /*
     * AXI write descriptor status output

    output wire [TAG_WIDTH-1:0]       m_axis_write_desc_status_tag,
    output wire [AXIS_ID_WIDTH-1:0]   m_axis_write_desc_status_id,
    output wire [AXIS_DEST_WIDTH-1:0] m_axis_write_desc_status_dest,
    output wire [AXIS_USER_WIDTH-1:0] m_axis_write_desc_status_user,
    output wire [3:0]                 m_axis_write_desc_status_error,
    output wire                       m_axis_write_desc_status_valid,
    /*
     * AXI stream write data input
     */
    input  wire [AXIS_DATA_WIDTH-1:0] s_axis_write_data_tdata,
    input  wire [AXIS_KEEP_WIDTH-1:0] s_axis_write_data_tkeep,

    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_awaddr,
    output wire [7:0]                 m_axi_awlen,
    output wire [2:0]                 m_axi_awsize,
    output wire [1:0]                 m_axi_awburst,
    output wire                       m_axi_awlock,
    output wire [3:0]                 m_axi_awcache,
    output wire [2:0]                 m_axi_awprot,
    output wire                       m_axi_awvalid,
    input  wire                       m_axi_awready,
    output wire [AXI_DATA_WIDTH-1:0]  m_axi_wdata,
    output wire [AXI_STRB_WIDTH-1:0]  m_axi_wstrb,

    output wire                       m_axi_wvalid,

    output wire                       m_axi_bready,
    output wire [AXI_ID_WIDTH-1:0]    m_axi_arid,
    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_araddr,
    output wire [7:0]                 m_axi_arlen,
    output wire [2:0]                 m_axi_arsize,

    // Internal state and signals
    reg [AXI_ADDR_WIDTH-1:0] read_addr_reg;
    reg [LEN_WIDTH-1:0] read_len_reg;
    reg [TAG_WIDTH-1:0] read_tag_reg;
    reg [AXIS_ID_WIDTH-1:0] read_id_reg;
    reg [AXIS_DEST_WIDTH-1:0] read_dest_reg;
    reg [AXIS_USER_WIDTH-1:0] read_user_reg;
    reg read_desc_ready_reg;
    reg [TAG_WIDTH-1:0] read_desc_status_tag_reg;
    reg [3:0] read_desc_status_error_reg;
    reg read_desc_status_valid_reg;
    reg [AXIS_DATA_WIDTH-1:0] read_data_tdata_reg;
    reg [AXIS_KEEP_WIDTH-1:0] read_data_tkeep_reg;
    reg read_data_tvalid_reg;
    reg read_data_tlast_reg;
    reg [AXIS_ID_WIDTH-1:0] read_data_tid_reg;
    reg [AXIS_DEST_WIDTH-1:0] read_data_tdest_reg;
    reg [AXIS_USER_WIDTH-1:0] read_data_tuser_reg;
    reg [AXI_ADDR_WIDTH-1:0] write_addr_reg;
    reg [LEN_WIDTH-1:0] write_len_reg;
    reg [TAG_WIDTH-1:0] write_tag_reg;
    reg [AXIS_ID_WIDTH-1:0] write_id_reg;
    reg [AXIS_DEST_WIDTH-1:0] write_dest_reg;
    reg [AXIS_USER_WIDTH-1:0] write_user_reg;
    reg write_desc_ready_reg;
    reg [LEN_WIDTH-1:0] write_desc_status_len_reg;
    reg [TAG_WIDTH-1:0] write_desc_status_tag_reg;
    reg [AXIS_ID_WIDTH-1:0] write_desc_status_id_reg;
    reg [AXIS_DEST_WIDTH-1:0] write_desc_status_dest_reg;
    reg [AXIS_USER_WIDTH-1:0] write_desc_status_user_reg;
    reg [3:0] write_desc_status_error_reg;
    reg write_desc_status_valid_reg;
    reg [AXI_ADDR_WIDTH-1:0] read_addr;
    reg [LEN_WIDTH-1:0] read_len;
    reg [AXI_ADDR_WIDTH-1:0] write_addr;
    reg [LEN_WIDTH-1:0] write_len;
    reg read_burst;
    reg write_burst;
    reg [7:0] read_burst_counter;
    reg [7:0] write_burst_counter;

    // AXI Read Descriptor
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            read_addr_reg <= 0;
            read_len_reg <= 0;
            read_tag_reg <= 0;
            read_id_reg <= 0;
            read_dest_reg <= 0;
            read_user_reg <= 0;
            read_desc_ready_reg <= 0;
            read_desc_status_valid_reg <= 0;
        end else begin
            if (s_axis_read_desc_valid && read_desc_ready_reg) begin
                read_addr_reg <= s_axis_read_desc_addr;
                read_len_reg <= s_axis_read_desc_len;
                read_tag_reg <= s_axis_read_desc_tag;
                read_id_reg <= s_axis_read_desc_id;
                read_dest_reg <= s_axis_read_desc_dest;
                read_user_reg <= s_axis_read_desc_user;
                read_desc_ready_reg <= 0;
                read_desc_status_valid_reg <= 1;
            end else if (m_axi_arready && read_len_reg > 0) begin
                read_desc_status_valid_reg <= 0;
            end
        end
    end

    // AXI Write Descriptor
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            write_addr_reg <= 0;
            write_len_reg <= 0;
            write_tag_reg <= 0;
            write_id_reg <= 0;
            write_dest_reg <= 0;
            write_user_reg <= 0;
            write_desc_ready_reg <= 0;
            write_desc_status_valid_reg <= 0;
        end else begin
            if (s_axis_write_desc_valid && write_desc_ready_reg) begin
                write_addr_reg <= s_axis_write_desc_addr;
                write_len_reg <= s_axis_write_desc_len;
                write_tag_reg <= s_axis_write_desc_tag;
                write_id_reg <= s_axis_write_desc_id;
                write_dest_reg <= s_axis_write_desc_dest;
                write_user_reg <= s_axis_write_desc_user;
                write_desc_ready_reg <= 0;
                write_desc_status_valid_reg <= 1;
            end else if (m_axi_awready && write_len_reg > 0) begin
                write_desc_status_valid_reg <= 0;
            end
        end
    end

    // AXI Read Data
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            read_data_tvalid_reg <= 0;
            read_data_tlast_reg <= 0;
        end else begin
            if (m_axi_rvalid && m_axis_read_data_tready) begin
                read_data_tvalid_reg <= 1;
                read_data_tlast_reg <= m_axi_rlast;
            end else if (m_axis_read_data_tready) begin
                read_data_tvalid_reg <= 0;
            end
        end
    end

    // AXI Write Data
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            s_axis_write_data_tready <= 0;
        end else begin
            if (m_axi_wready && s_axis_write_data_valid) begin
                s_axis_write_data_tready <= 1;
            end else begin
                s_axis_write_data_tready <= 0;
            end
        end
    end

    // AXI Read Address Channel
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            m_axi_arvalid <= 0;
            read_addr <= 0;
            read_len <= 0;
            read_burst_counter <= 0;
        end else begin
            if (read_len_reg > 0 && !m_axi_arvalid) begin
                m_axi_arvalid <= 1;
                read_addr <= read_addr_reg;
                read_len <= read_len_reg;
                read_burst_counter <= read_len_reg[7:0];
            end else if (m_axi_arready) begin
                m_axi_arvalid <= 0;
            end
        end
    end

    // AXI Write Address Channel
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            m_axi_awvalid <= 0;
            write_addr <= 0;
            write_len <= 0;
            write_burst_counter <= 0;
        end else begin
            if (write_len_reg > 0 && !m_axi_awvalid) begin
                m_axi_awvalid <= 1;
                write_addr <= write_addr_reg;
                write_len <= write_len_reg;
                write_burst_counter <= write_len_reg[7:0];
            end else if (m_axi_awready) begin
                m_axi_awvalid <= 0;
            end
        end
    end

    // AXI Read Data Channel
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            read_data_tdata_reg <= 0;
            read_data_tkeep_reg <= 0;
            read_data_tid_reg <= 0;
            read_data_tdest_reg <= 0;
            read_data_tuser_reg <= 0;
        end else begin
            if (m_axi_rvalid && m_axis_read_data_tready) begin
                read_data_tdata_reg <= m_axi_rdata;
                read_data_tkeep_reg <= {AXIS_KEEP_WIDTH{1'b1}};
                read_data_tid_reg <= read_id_reg;
                read_data_tdest_reg <= read_dest_reg;
                read_data_tuser_reg <= read_user_reg;
            end
        end
    end

    // AXI Write Data Channel
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            m_axi_wdata <= 0;
            m_axi_wstrb <= 0;
            m_axi_wlast <= 0;
            m_axi_wvalid <= 0;
        end else begin
            if (s_axis_write_data_valid && s_axis_write_data_tready) begin
                m_axi_wdata <= s_axis_write_data_tdata;
                m_axi_wstrb <= s_axis_write_data_tkeep;
                m_axi_wlast <= s_axis_write_data_tlast;
                m_axi_wvalid <= 1;
            end else if (m_axi_wready) begin
                m_axi_wvalid <= 0;
            end
        end
    end

    // AXI Read Response Channel
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            read_desc_status_error_reg <= 0;
        end else begin
            if (m_axi_rvalid && m_axis_read_data_tready && m_axi_rresp != 2'b00) begin
                read_desc_status_error_reg <= 4'b0001;
            end else if (m_axi_rvalid && m_axis_read_data_tready) begin
                read_desc_status_error_reg <= 0;
            end
        end
    end

    // AXI Write Response Channel
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            write_desc_status_error_reg <= 0;
        end else begin
            if (m_axi_bvalid && m_axi_bresp != 2'b00) begin
                write_desc_status_error_reg <= 4'b0001;
            end else if (m_axi_bvalid) begin
                write_desc_status_error_reg <= 0;
            end
        end
    end

    // AXI Read Descriptor Ready
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            s_axis_read_desc_ready <= 0;
        end else begin
            if (!m_axi_arvalid) begin
                s_axis_read_desc_ready <= 1;
            end else begin
                s_axis_read_desc_ready <= 0;
            end
        end
    end

    // AXI Write Descriptor Ready
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            s_axis_write_desc_ready <= 0;
        end else begin
            if (!m_axi_awvalid) begin
                s_axis_write_desc_ready <= 1;
            end else begin
                s_axis_write_desc_ready <= 0;
            end
        end
    end

    // AXI Read Descriptor Status
    assign m_axis_read_desc_status_tag = read_tag_reg;
    assign m_axis_read_desc_status_error = read_desc_status_error_reg;
    assign m_axis_read_desc_status_valid = read_desc_status_valid_reg;

    // AXI Read Data
    assign m_axis_read_data_tdata = read_data_tdata_reg;
    assign m_axis_read_data_tkeep = read_data_tkeep_reg;
    assign m_axis_read_data_tvalid = read_data_tvalid_reg;
    assign m_axis_read_data_tlast = read_data_tlast_reg;
    assign m_axis_read_data_tid = read_data_tid_reg;
    assign m_axis_read_data_tdest = read_data_tdest_reg;
    assign m_axis_read_data_tuser = read_data_tuser_reg;

    // AXI Write Descriptor Status
    assign m_axis_write_desc_status_len = write_len_reg;
    assign m_axis_write_desc_status_tag = write_tag_reg;
    assign m_axis_write_desc_status_id = write_id_reg;
    assign m_axis_write_desc_status_dest = write_dest_reg;
    assign m_axis_write_desc_status_user = write_user_reg;
    assign m_axis_write_desc_status_error = write_desc_status_error_reg;
    assign m_axis_write_desc_status_valid = write_desc_status_valid_reg;

    // AXI Read Address Channel
    assign m_axi_arid = read_id_reg;
    assign m_axi_araddr = read_addr;
    assign m_axi_arlen = read_len[7:0];
    assign m_axi_arsize = 3'b010; // 32-bit data width
    assign m_axi_arburst = 2'b01; // INCR burst
    assign m_axi_arlock = 1'b0;
    assign m_axi_arcache = 4'b0011; // Normal Non-cacheable Bufferable
    assign m_axi_arprot = 3'b000;

    // AXI Write Address Channel
    assign m_axi_awid = write_id_reg;
    assign m_axi_awaddr = write_addr;
    assign m_axi_awlen = write_len[7:0];
    assign m_axi_awsize = 3'b010; // 32-bit data width
    assign m_axi_awburst = 2'b01; // INCR burst
    assign m_axi_awlock = 1'b0;
    assign m_axi_awcache = 4'b0011; // Normal Non-cacheable Bufferable
    assign m_axi_awprot = 3'b000;

    // AXI Read Data Channel
    assign m_axi_rready = m_axis_read_data_tready;

    // AXI Write Data Channel
    assign m_axi_bready = 1'b1;

endmodule
