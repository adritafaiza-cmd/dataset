    output wire [AXIS_KEEP_WIDTH-1:0] m_axis_read_data_tkeep,
    output wire                       m_axis_read_data_tvalid,

// FIFO generic module
module fifo #(
    parameter DATA_WIDTH = 32,
    parameter DEPTH = 16
)
(
    input  wire                       clk,
    input  wire                       rst,
    input  wire [DATA_WIDTH-1:0]      din,
    input  wire                       wr_en,
    input  wire                       rd_en,
    output wire [DATA_WIDTH-1:0]      dout,
    output wire                       full,
    output wire                       empty,
    output wire                       valid
);

    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];
    reg [$clog2(DEPTH)-1:0] wr_ptr;
    reg [$clog2(DEPTH)-1:0] rd_ptr;
    reg [3:0] count;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_ptr <= 0;
            rd_ptr <= 0;
            count <= 0;
        end else begin
            if (wr_en && !full) begin
                mem[wr_ptr] <= din;
                wr_ptr <= wr_ptr + 1;
                count <= count + 1;
            end
            if (rd_en && !empty) begin
                rd_ptr <= rd_ptr + 1;
                count <= count - 1;
            end
        end
    end

    assign dout = mem[rd_ptr];
    assign full = (count == DEPTH);
    assign empty = (count == 0);
    assign valid = !empty;

endmodule

// State machine for read operations
module read_state_machine #(
    parameter AXI_ADDR_WIDTH = 16,
    parameter LEN_WIDTH = 20,
    parameter TAG_WIDTH = 8,
    parameter AXI_MAX_BURST_LEN = 16
)
(
    input  wire                       clk,
    input  wire                       rst,
    input  wire                       start,
    input  wire [AXI_ADDR_WIDTH-1:0]  addr,
    input  wire [LEN_WIDTH-1:0]       len,
    input  wire [TAG_WIDTH-1:0]       tag,
    output reg                        done,
    output reg [3:0]                  error,
    output reg [AXI_ID_WIDTH-1:0]     m_axi_arid,
    output reg [AXI_ADDR_WIDTH-1:0]   m_axi_araddr,
    output reg [7:0]                  m_axi_arlen,
    output reg [2:0]                  m_axi_arsize,
    output reg [1:0]                  m_axi_arburst,
    output reg                        m_axi_arlock,
    output reg [3:0]                  m_axi_arcache,
    output reg [2:0]                  m_axi_arprot,
    output reg                        m_axi_arvalid,
    input  wire                       m_axi_arready,
    input  wire [AXI_ID_WIDTH-1:0]    m_axi_rid,
    input  wire [AXI_DATA_WIDTH-1:0]  m_axi_rdata,
    input  wire [1:0]                 m_axi_rresp,
    input  wire                       m_axi_rlast,
    input  wire                       m_axi_rvalid,
    output reg                        m_axi_rready
);

    reg [LEN_WIDTH-1:0] remaining_len;
    reg [AXI_ADDR_WIDTH-1:0] current_addr;
    reg [AXI_ID_WIDTH-1:0] current_id;
    reg [TAG_WIDTH-1:0] current_tag;

    reg [2:0] state;

    localparam IDLE = 3'b000;
    localparam START = 3'b001;
    localparam WAIT_ARREADY = 3'b010;
    localparam WAIT_RVALID = 3'b011;
    localparam DONE = 3'b100;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= IDLE;
            done <= 0;
            error <= 4'b0000;
            m_axi_arid <= 0;
            m_axi_araddr <= 0;
            m_axi_arlen <= 0;
            m_axi_arsize <= 3'b010; // 4 bytes
            m_axi_arburst <= 2'b01; // INCR
            m_axi_arlock <= 0;
            m_axi_arcache <= 4'b0011; // Non-cacheable
            m_axi_arprot <= 3'b000; // Normal
            m_axi_arvalid <= 0;
            m_axi_rready <= 0;
            remaining_len <= 0;
            current_addr <= 0;
            current_id <= 0;
            current_tag <= 0;
        end else begin
            case (state)
                IDLE: begin
                    if (start) begin
                        state <= START;
                        current_addr <= addr;
                        current_id <= addr[AXI_ID_WIDTH-1:0];
                        current_tag <= tag;
                        remaining_len <= len;
                    end
                end
                START: begin
                    m_axi_arvalid <= 1;
                    m_axi_arid <= current_id;
                    m_axi_araddr <= current_addr;
                    m_axi_arlen <= remaining_len < AXI_MAX_BURST_LEN ? remaining_len - 1 : AXI_MAX_BURST_LEN - 1;
                    state <= WAIT_ARREADY;
                end
                WAIT_ARREADY: begin
                    if (m_axi_arready) begin
                        m_axi_arvalid <= 0;
                        state <= WAIT_RVALID;
                    end
                end
                WAIT_RVALID: begin
                    m_axi_rready <= 1;
                    if (m_axi_rvalid) begin
                        if (m_axi_rresp != 2'b00) begin
                            state <= DONE;
                            error <= 4'b0001; // Error
                        end else begin
                            remaining_len <= remaining_len - (m_axi_rlast ? 1 : AXI_MAX_BURST_LEN);
                            current_addr <= current_addr + (m_axi_rlast ? 0 : AXI_MAX_BURST_LEN * (AXI_DATA_WIDTH / 8));
                            if (remaining_len == 0 || m_axi_rlast) begin
                                state <= DONE;
                                done <= 1;
                            end
                        end
                    end
                end
                DONE: begin
                    done <= 0;
                    error <= 4'b0000;
                    state <= IDLE;
                end
                default: begin
                    state <= IDLE;
                end
            endcase
        end
    end

endmodule
