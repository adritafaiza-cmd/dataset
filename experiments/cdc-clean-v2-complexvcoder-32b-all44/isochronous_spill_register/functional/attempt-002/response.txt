=== STAGE 2: VERILOG ===
```verilog
module isochronous_spill_register #(
    parameter WIDTH = 8
)(
    input              src_clk_i,
    input              src_rst_ni,
    input              src_valid_i,
    output             src_ready_o,
    input  [WIDTH-1:0] src_data_i,
    input              dst_clk_i,
    input              dst_rst_ni,
    output             dst_valid_o,
    input              dst_ready_i,
    output [WIDTH-1:0] dst_data_o
);

    // Internal signals for source FIFO
    reg                src_fifo_full;
    reg                src_fifo_empty;
    reg  [WIDTH-1:0] src_fifo_wdata;
    wire [WIDTH-1:0] src_fifo_rdata;
    reg                src_fifo_wen;
    reg                src_fifo_ren;

    // Internal signals for destination FIFO
    reg                dst_fifo_full;
    reg                dst_fifo_empty;
    reg  [WIDTH-1:0] dst_fifo_wdata;
    wire [WIDTH-1:0] dst_fifo_rdata;
    reg                dst_fifo_wen;
    reg                dst_fifo_ren;

    // Source FIFO instantiation
    fifo_2_deep #(
        .WIDTH(WIDTH)
    ) src_fifo_inst (
        .clk_i    (src_clk_i),
        .rst_ni   (src_rst_ni),
        .wen_i    (src_fifo_wen),
        .ren_i    (src_fifo_ren),
        .wdata_i  (src_fifo_wdata),
        .rdata_o  (src_fifo_rdata),
        .full_o   (src_fifo_full),
        .empty_o  (src_fifo_empty)
    );

    // Destination FIFO instantiation
    fifo_2_deep #(
        .WIDTH(WIDTH)
    ) dst_fifo_inst (
        .clk_i    (dst_clk_i),
        .rst_ni   (dst_rst_ni),
        .wen_i    (dst_fifo_wen),
        .ren_i    (dst_fifo_ren),
        .wdata_i  (dst_fifo_wdata),
        .rdata_o  (dst_fifo_rdata),
        .full_o   (dst_fifo_full),
        .empty_o  (dst_fifo_empty)
    );

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_fifo_wen <= 1'b0;
        end else begin
            src_fifo_wen <= src_valid_i && !src_fifo_full;
        end
    end

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_fifo_wdata <= {WIDTH{1'b0}};
        end else if (src_fifo_wen) begin
            src_fifo_wdata <= src_data_i;
        end
    end

    assign src_ready_o = !src_fifo_full;

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_fifo_wen <= 1'b0;
        end else begin
            dst_fifo_wen <= !src_fifo_empty && !dst_fifo_full;
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_fifo_ren <= 1'b0;
        end else begin
            dst_fifo_ren <= dst_ready_i && !dst_fifo_empty;
        end
    end

    assign dst_fifo_wdata = src_fifo_rdata;
    assign dst_data_o     = dst_fifo_rdata;
    assign dst_valid_o    = !dst_fifo_empty;

endmodule

// 2-deep FIFO module
module fifo_2_deep #(
    parameter WIDTH = 8
)(
    input              clk_i,
    input              rst_ni,
    input              wen_i,
    input              ren_i,
    input  [WIDTH-1:0] wdata_i,
    output [WIDTH-1:0] rdata_o,
    output             full_o,
    output             empty_o
);

    reg [WIDTH-1:0] mem [0:1];
    reg [1:0]         wr_ptr;
    reg [1:0]         rd_ptr;
    reg [1:0]         count;

    // Write operation
    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            wr_ptr  <= 2'b00;
            count   <= 2'b00;
        end else if (wen_i && !full_o) begin
            mem[wr_ptr] <= wdata_i;
            wr_ptr      <= wr_ptr + 1;
            count       <= count + 1;
        end
    end

    // Read operation
    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            rd_ptr  <= 2'b00;
        end else if (ren_i && !empty_o) begin
            rd_ptr  <= rd_ptr + 1;
            count   <= count - 1;
        end
    end

    assign rdata_o = mem[rd_ptr];
    assign full_o  = (count == 2'b10);
    assign empty_o = (count == 2'b00);

endmodule
```
