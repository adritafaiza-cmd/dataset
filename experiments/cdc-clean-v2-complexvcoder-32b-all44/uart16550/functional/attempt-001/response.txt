module uart_top (
    input wb_clk_i,
    input wb_rst_i,
    input [7:0] wb_adr_i,
    input [31:0] wb_dat_i,
    output [31:0] wb_dat_o,

    input wb_stb_i,

    input [3:0] wb_sel_i,
    output int_o,
    output stx_pad_o,
    input srx_pad_i,
    output rts_pad_o,
    input cts_pad_i,
    output dtr_pad_o,
    input dsr_pad_i,
    input ri_pad_i,
    input dcd_pad_i
);

    // Internal signals
    reg [31:0] registers;
    reg [15:0] baud_rate;
    reg [7:0] fifo_control;
    reg [7:0] modem_signals;

    // Instantiate wishbone_slave module
    wishbone_slave wb_slave (
        .wb_clk_i(wb_clk_i),

// Wishbone slave module
module wishbone_slave (
    input wb_clk_i,
    input wb_rst_i,
    input [7:0] wb_adr_i,
    input [31:0] wb_dat_i,
    output reg [31:0] wb_dat_o,
    input wb_we_i,
    input wb_stb_i,
    input wb_cyc_i,
    output reg wb_ack_o,
    input [3:0] wb_sel_i,
    output reg [31:0] registers,
    output reg [15:0] baud_rate,
    output reg [7:0] fifo_control,
    output reg [7:0] modem_signals
);

    reg [2:0] reg_addr;

    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            wb_ack_o <= 0;
            registers <= 0;
            baud_rate <= 0;
            fifo_control <= 0;
            modem_signals <= 0;
        end else if (wb_cyc_i && wb_stb_i) begin
            reg_addr <= wb_adr_i[4:2];
            wb_ack_o <= 1;
            if (wb_we_i) begin
                case (reg_addr)
                    0: begin // Transmit Holding Register (THR) / Receiver Buffer Register (RBR)
                        if (wb_sel_i[0]) registers[7:0] <= wb_dat_i[7:0];
                    end
                    1: begin // Interrupt Enable Register (IER)
                        if (wb_sel_i[0]) registers[15:8] <= wb_dat_i[15:8];
                    end
                    2: begin // FIFO Control Register (FCR)
                        if (wb_sel_i[0]) fifo_control <= wb_dat_i[7:0];
                    end
                    3: begin // Line Control Register (LCR)
                        if (wb_sel_i[0]) registers[23:16] <= wb_dat_i[23:16];
                    end
                    4: begin // Modem Control Register (MCR)
                        if (wb_sel_i[0]) modem_signals <= wb_dat_i[7:0];
                    end
                    5: begin // Line Status Register (LSR)
                        // Read-only register, do nothing on write
                    end
                    6: begin // Modem Status Register (MSR)
                        // Read-only register, do nothing on write
                    end
                    7: begin // Divisor Latch (LSB) (DLL)
                        if (registers[7]) baud_rate[7:0] <= wb_dat_i[7:0];
                    end
                    8: begin // Divisor Latch (MSB) (DLM)
                        if (registers[7]) baud_rate[15:8] <= wb_dat_i[7:0];
                    end
                    default: begin
                        // Reserved registers, do nothing
                    end
                endcase
            end else begin
                case (reg_addr)
                    0: wb_dat_o <= {24'b0, registers[7:0]}; // RBR
                    1: wb_dat_o <= {24'b0, registers[15:8]}; // IER
                    2: wb_dat_o <= {24'b0, fifo_control}; // FCR
                    3: wb_dat_o <= {24'b0, registers[23:16]}; // LCR
                    4: wb_dat_o <= {24'b0, modem_signals}; // MCR
                    5: wb_dat_o <= {24'b0, 8'b0}; // LSR (read-only, value is 0 for simplicity)
                    6: wb_dat_o <= {24'b0, 8'b0}; // MSR (read-only, value is 0 for simplicity)
                    7: wb_dat_o <= {16'b0, baud_rate[7:0]}; // DLL
                    8: wb_dat_o <= {16'b0, baud_rate[15:8]}; // DLM
                    default: wb_dat_o <= 32'b0; // Reserved registers
                endcase
            end
        end else begin
            wb_ack_o <= 0;
        end
    end

endmodule

// UART core module
module uart_core (
    input clk,
    input rst,
    input srx,
    output reg stx,
    input [15:0] baud_rate,
    input [7:0] fifo_control,
    input [7:0] modem_signals,
    output int_o
);

    reg [7:0] tx_fifo [0:15];
    reg [3:0] tx_fifo_head, tx_fifo_tail, tx_fifo_count;
    reg [7:0] rx_fifo [0:15];
    reg [3:0] rx_fifo_head, rx_fifo_tail, rx_fifo_count;
    reg [15:0] bit_counter;
    reg [3:0] bit_position;
    reg tx_busy, rx_busy;
    reg [7:0] tx_data, rx_data;
    reg tx_shift_reg [10:0];
    reg rx_shift_reg [10:0];
    reg [1:0] rx_state;
    reg [1:0] tx_state;
    reg [1:0] int_state;

    // Baud rate generation
    reg [15:0] baud_counter;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            baud_counter <= 0;
            bit_counter <= 0;
            bit_position <= 0;
            tx_busy <= 0;
            rx_busy <= 0;
            tx_shift_reg <= 0;
            rx_shift_reg <= 0;
            tx_data <= 0;
            rx_data <= 0;
            tx_fifo_head <= 0;
            tx_fifo_tail <= 0;
            tx_fifo_count <= 0;
            rx_fifo_head <= 0;
            rx_fifo_tail <= 0;
            rx_fifo_count <= 0;
            int_state <= 0;
        end else begin
            // Baud rate generation
            if (baud_counter == baud_rate) begin
                baud_counter <= 0;
                bit_counter <= bit_counter + 1;
            end else begin
                baud_counter <= baud_counter + 1;
            end

            // Transmit state machine
            case (tx_state)
                0: begin // Idle
                    if (tx_fifo_count > 0 && !tx_busy) begin
                        tx_data <= tx_fifo[tx_fifo_tail];
                        tx_fifo_tail <= tx_fifo_tail + 1;
                        tx_fifo_count <= tx_fifo_count - 1;
                        tx_busy <= 1;
                        tx_shift_reg <= {1'b1, tx_data, 1'b0}; // Start bit + data + stop bit
                        bit_position <= 0;
                        tx_state <= 1;
                    end
                end
                1: begin // Transmitting
                    if (bit_counter == baud_rate) begin
                        bit_counter <= 0;
                        if (bit_position < 11) begin
                            stx <= tx_shift_reg[bit_position];
                            bit_position <= bit_position + 1;
                        end else begin
                            tx_busy <= 0;
                            tx_state <= 0;
                            int_state[0] <= 1; // Transmit interrupt
                        end
                    end
                end
            endcase

            // Receive state machine
            case (rx_state)
                0: begin // Idle
                    if (srx == 0 && !rx_busy) begin
                        rx_busy <= 1;
                        bit_position <= 0;
                        rx_state <= 1;
                    end
                end
                1: begin // Receiving
                    if (bit_counter == baud_rate) begin
                        bit_counter <= 0;
                        if (bit_position < 11) begin
                            rx_shift_reg[bit_position] <= srx;
                            bit_position <= bit_position + 1;
                        end else begin
                            if (rx_shift_reg[0] == 0 && rx_shift_reg[10] == 1) begin
                                rx_data <= rx_shift_reg[9:2];
                                rx_fifo[rx_fifo_head] <= rx_data;
                                rx_fifo_head <= rx_fifo_head + 1;
                                rx_fifo_count <= rx_fifo_count + 1;
                                int_state[1] <= 1; // Receive interrupt
                            end
                            rx_busy <= 0;
                            rx_state <= 0;
                        end
                    end
                end
            endcase

            // FIFO control (simplified)
            if (fifo_control[0] && tx_fifo_count < 16) begin
                if (registers[7:0] != 0) begin // Transmit Holding Register (THR)
                    tx_fifo[tx_fifo_head] <= registers[7:0];
                    tx_fifo_head <= tx_fifo_head + 1;
                    tx_fifo_count <= tx_fifo_count + 1;
                end
            end

            // Interrupt generation
            int_o <= |int_state;
        end
    end

endmodule

// Clock divider module
module clock_divider (
    input clk_in,
    input rst,
    input [15:0] baud_rate,
    output reg clk_out
);

    reg [15:0] counter;

    always @(posedge clk_in or posedge rst) begin
        if (rst) begin
            counter <= 0;
            clk_out <= 0;
        end else begin
            if (counter == baud_rate) begin
                counter <= 0;
                clk_out <= ~clk_out;
            end else begin
                counter <= counter + 1;
            end
        end
    end

endmodule
