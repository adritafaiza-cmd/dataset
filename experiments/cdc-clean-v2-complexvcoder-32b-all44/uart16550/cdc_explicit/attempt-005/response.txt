```verilog
module uart_top (
    input wire wb_clk_i,
    input wire wb_rst_i,
    input wire [7:0] wb_adr_i,
    input wire [31:0] wb_dat_i,
    output reg [31:0] wb_dat_o,
    input wire wb_we_i,
    input wire wb_stb_i,
    input wire wb_cyc_i,
    output reg wb_ack_o,
    input wire [3:0] wb_sel_i,
    output reg int_o,
    output reg stx_pad_o,
    input wire srx_pad_i,
    output reg rts_pad_o,
    input wire cts_pad_i,
    output reg dtr_pad_o,
    input wire dsr_pad_i,
    input wire ri_pad_i,
    input wire dcd_pad_i
);

    // Internal signals
    reg [31:0] reg_write_data;
    reg reg_write_enable;
    reg [7:0] reg_address;
    wire [31:0] uart_core_reg_read_data;
    wire [31:0] modem_ctrl_reg_read_data;

    // Instantiate Wishbone Slave
    wishbone_slave wb_slave_inst (
        .clk(wb_clk_i),
        .rst(wb_rst_i),
        .adr_i(wb_adr_i),
        .dat_i(wb_dat_i),
        .dat_o(wb_dat_o),
        .we_i(wb_we_i),
        .stb_i(wb_stb_i),
        .cyc_i(wb_cyc_i),
        .ack_o(wb_ack_o),
        .sel_i(wb_sel_i),
        .reg_write_data(reg_write_data),
        .reg_write_enable(reg_write_enable),
        .reg_address(reg_address)
    );

    // Instantiate UART Core
    uart_core uart_core_inst (
        .clk(wb_clk_i),
        .rst(wb_rst_i),
        .tx_o(stx_pad_o),
        .rx_i(srx_pad_i),
        .int_o(int_o),
        .reg_write_data(reg_write_data),
        .reg_write_enable(reg_write_enable),
        .reg_address(reg_address),
        .reg_read_data(uart_core_reg_read_data)
    );

    // Instantiate Modem Control
    modem_control modem_ctrl_inst (
        .clk(wb_clk_i),
        .rst(wb_rst_i),
        .rts_o(rts_pad_o),
        .cts_i(cts_pad_i),
        .dtr_o(dtr_pad_o),
        .dsr_i(dsr_pad_i),
        .ri_i(ri_pad_i),
        .dcd_i(dcd_pad_i),
        .reg_write_data(reg_write_data),
        .reg_write_enable(reg_write_enable),
        .reg_address(reg_address),
        .reg_read_data(modem_ctrl_reg_read_data)
    );

    // Mux read data from UART Core or Modem Control
    always @(*) begin
        case (reg_address[7:5])
            3'b000: wb_dat_o = uart_core_reg_read_data; // UART registers
            3'b001: wb_dat_o = modem_ctrl_reg_read_data; // Modem Control registers
            default: wb_dat_o = 32'h00000000;
        endcase
    end

endmodule

// Wishbone Slave
module wishbone_slave (
    input wire clk,
    input wire rst,
    input wire [7:0] adr_i,
    input wire [31:0] dat_i,
    output reg [31:0] dat_o,
    input wire we_i,
    input wire stb_i,
    input wire cyc_i,
    output reg ack_o,
    input wire [3:0] sel_i,
    output reg [31:0] reg_write_data,
    output reg reg_write_enable,
    output reg [7:0] reg_address
);

    reg [31:0] registers [0:7];
    reg [3:0] sel;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            ack_o <= 0;
            reg_write_enable <= 0;
        end else begin
            if (cyc_i && stb_i) begin
                ack_o <= 1;
                reg_address <= adr_i;
                sel <= sel_i;
                if (we_i) begin
                    reg_write_enable <= 1;
                    reg_write_data <= dat_i;
                end else begin
                    reg_write_enable <= 0;
                end
            end else begin
                ack_o <= 0;
                reg_write_enable <= 0;
            end
        end
    end

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            registers[0] <= 32'h00000000; // RBR/THR
            registers[1] <= 32'h00000000; // IER
            registers[2] <= 32'h00000000; // IIR/FIFO
            registers[3] <= 32'h00000000; // LCR
            registers[4] <= 32'h00000000; // MCR
            registers[5] <= 32'h00000000; // LSR
            registers[6] <= 32'h00000000; // MSR
            registers[7] <= 32'h00000000; // SCR
        end else begin
            if (reg_write_enable) begin
                case (reg_address[2:0])
                    3'b000: if (sel[0]) registers[0][7:0] <= reg_write_data[7:0];
                    3'b001: if (sel[1]) registers[1][7:0] <= reg_write_data[7:0];
                    3'b010: if (sel[2]) registers[2][7:0] <= reg_write_data[7:0];
                    3'b011: if (sel[3]) registers[3][7:0] <= reg_write_data[7:0];
                    3'b100: if (sel[0]) registers[4][7:0] <= reg_write_data[7:0];
                    3'b101: if (sel[1]) registers[5][7:0] <= reg_write_data[7:0];
                    3'b110: if (sel[2]) registers[6][7:0] <= reg_write_data[7:0];
                    3'b111: if (sel[3]) registers[7][7:0] <= reg_write_data[7:0];
                endcase
            end
        end
    end

    always @(*) begin
        dat_o = 32'h00000000;
        case (reg_address[2:0])
            3'b000: dat_o[7:0] = registers[0][7:0]; // RBR/THR
            3'b001: dat_o[7:0] = registers[1][7:0]; // IER
            3'b010: dat_o[7:0] = registers[2][7:0]; // IIR/FIFO
            3'b011: dat_o[7:0] = registers[3][7:0]; // LCR
            3'b100: dat_o[7:0] = registers[4][7:0]; // MCR
            3'b101: dat_o[7:0] = registers[5][7:0]; // LSR
            3'b110: dat_o[7:0] = registers[6][7:0]; // MSR
            3'b111: dat_o[7:0] = registers[7][7:0]; // SCR
        endcase
    end

endmodule

// UART Core
module uart_core (
    input wire clk,
    input wire rst,
    output reg tx_o,
    input wire rx_i,
    output reg int_o,
    input wire [31:0] reg_write_data,
    input wire reg_write_enable,
    input wire [7:0] reg_address,
    output reg [31:0] reg_read_data
);

    // UART registers
    reg [7:0] rbr; // Receiver Buffer Register
    reg [7:0] thr; // Transmitter Holding Register
    reg [7:0] ier; // Interrupt Enable Register
    reg [7:0] iir; // Interrupt Identification Register
    reg [7:0] fcr; // FIFO Control Register
    reg [7:0] lcr; // Line Control Register
    reg [7:0] mcr; // Modem Control Register
    reg [7:0] lsr; // Line Status Register
    reg [7:0] msr; // Modem Status Register
    reg [7:0] scr; // Scratch Register

    // FIFOs
    reg [7:0] tx_fifo [0:15];
    reg [3:0] tx_fifo_wptr;
    reg [3:0] tx_fifo_rptr;
    reg [3:0] tx_fifo_count;

    reg [7:0] rx_fifo [0:15];
    reg [3:0] rx_fifo_wptr;
    reg [3:0] rx_fifo_rptr;
    reg [3:0] rx_fifo_count;

    // UART state machine
    reg [3:0] baud_div;
    reg [3:0] bit_cnt;
    reg [7:0] tx_data;
    reg tx_busy;

    // FIFO control
    wire fifo_enable = fcr[0];
    wire tx_fifo_full = (tx_fifo_count == 16);
    wire tx_fifo_empty = (tx_fifo_count == 0);
    wire rx_fifo_full = (rx_fifo_count == 16);
    wire rx_fifo_empty = (rx_fifo_count == 0);

    // Register read/write
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            reg_read_data <= 32'h00000000;
        end else begin
            case (reg_address[2:0])
                3'b000: reg_read_data <= {24'h000000, rbr}; // RBR
                3'b001: reg_read_data <= {24'h000000, ier}; // IER
                3'b010: reg_read_data <= {24'h000000, iir}; // IIR
                3'b011: reg_read_data <= {24'h000000, lcr}; // LCR
                3'b100: reg_read_data <= {24'h000000, mcr}; // MCR
                3'b101: reg_read_data <= {24'h000000, lsr}; // LSR
                3'b110: reg_read_data <= {24'h000000, msr}; // MSR
                3'b111: reg_read_data <= {24'h000000, scr}; // SCR
            endcase
        end
    end

    // Register write
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rbr <= 8'h00;
            thr <= 8'h00;
            ier <= 8'h00;
            iir <= 8'h01;
            fcr <= 8'h00;
            lcr <= 8'h00;
            mcr <= 8'h00;
            lsr <= 8'h00;
            msr <= 8'h00;
            scr <= 8'h00;
        end else if (reg_write_enable) begin
            case (reg_address[2:0])
                3'b000: thr <= reg_write_data[7:0]; // THR
                3'b001: ier <= reg_write_data[7:0]; // IER
                3'b010: fcr <= reg_write_data[7:0]; // FCR
                3'b011: lcr <= reg_write_data[7:0]; // LCR
                3'b100: mcr <= reg_write_data[7:0]; // MCR
                3'b111: scr <= reg_write_data[7:0]; // SCR
            endcase
        end
    end

    // Line Status Register
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            lsr <= 8'h00;
        end else begin
            lsr[0] <= ~tx_fifo_empty; // Data Ready
            lsr[1] <= tx_fifo_full; // Overrun Error
            lsr[5] <= tx_fifo_empty; // Transmitter Holding Register Empty
        end
    end

    // Modem Status Register
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            msr <= 8'h00;
        end else begin
            // Update modem status based on external signals
        end
    end

    // Transmitter
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            tx_fifo_wptr <= 4'h0;
            tx_fifo_rptr <= 4'h0;
            tx_fifo_count <= 4'h0;
            tx_data <= 8'h00;
            tx_busy <= 0;
            bit_cnt <= 4'h0;
            baud_div <= 4'h0;
            tx_o <= 1'b1; // Idle state
        end else begin
            if (fifo_enable) begin
                // FIFO mode
                if (reg_write_enable && (reg_address[2:0] == 3'b000)) begin
                    if (!tx_fifo_full) begin
                        tx_fifo[tx_fifo_wptr] <= reg_write_data[7:0];
                        tx_fifo_wptr <= tx_fifo_wptr + 1;
                        tx_fifo_count <= tx_fifo_count + 1;
                    end
                end
                if (!tx_busy && !tx_fifo_empty) begin
                    tx_data <= tx_fifo[tx_fifo_rptr];
                    tx_fifo_rptr <= tx_fifo_rptr + 1;
                    tx_fifo_count <= tx_fifo_count - 1;
                    tx_busy <= 1;
                    bit_cnt <= 4'h0;
                    baud_div <= 4'h0;
                    tx_o <= 1'b0; // Start bit
                end
            end else begin
                // Non-FIFO mode
                if (reg_write_enable && (reg_address[2:0] == 3'b000)) begin
                    tx_data <= reg_write_data[7:0];
                    tx_busy <= 1;
                    bit_cnt <= 4'h0;
                    baud_div <= 4'h0;
                    tx_o <= 1'b0; // Start bit
                end
            end
            if (tx_busy) begin
                if (baud_div == 4'hF) begin
                    baud_div <= 4'h0;
                    if (bit_cnt < 8) begin
                        tx_o <= tx_data[bit_cnt];
                        bit_cnt <= bit_cnt + 1;
                    end else if (bit_cnt == 8) begin
                        tx_o <= 1'b1; // Stop bit
                        bit_cnt <= bit_cnt + 1;
                    end else begin
                        tx_busy <= 0;
                    end
                end else begin
                    baud_div <= baud_div + 1;
                end
            end
        end
    end

    always @(posedge clk or posedge rst) begin

module modem_control (
    input wire clk,
    input wire rst,
    output reg rts_o,
    input wire cts_i,
    output reg dtr_o,
    input wire dsr_i,
    input wire ri_i,
    input wire dcd_i,
    input wire [31:0] reg_write_data,
    input wire reg_write_enable,
    input wire [7:0] reg_address,
    output reg [31:0] reg_read_data
);

    // Modem Control registers
    reg [7:0] mcr; // Modem Control Register
    reg [7:0] msr; // Modem Status Register

    // Register read/write
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            reg_read_data <= 32'h00000000;
        end else begin
            case (reg_address[2:0])
                3'b100: reg_read_data <= {24'h000000, mcr}; // MCR
                3'b110: reg_read_data <= {24'h000000, msr}; // MSR
            endcase
        end
    end

    // Register write
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            mcr <= 8'h00;
        end else if (reg_write_enable && (reg_address[2:0] == 3'b100)) begin
            mcr <= reg_write_data[7:0]; // MCR
        end
    end

    // Modem Status Register
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            msr <= 8'h00;
        end else begin
            msr[4] <= cts_i;
            msr[5] <= dsr_i;
            msr[6] <= ri_i;
            msr[7] <= dcd_i;
        end
    end

    // Modem Control Outputs
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rts_o <= 1'b0;
            dtr_o <= 1'b0;
        end else begin
            rts_o <= mcr[1];
            dtr_o <= mcr[0];
        end
    end

endmodule
```
