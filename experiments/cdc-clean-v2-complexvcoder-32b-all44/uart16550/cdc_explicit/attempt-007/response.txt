// Wishbone Slave module
module wishbone_slave (
    input  wire wb_clk_i,
    input  wire wb_rst_i,
    input  wire [2:0] wb_adr_i,
    input  wire [7:0] wb_dat_i,
    output reg [7:0] wb_dat_o,
    input  wire wb_we_i,
    input  wire wb_stb_i,
    input  wire wb_cyc_i,
    output reg wb_ack_o,
    input  wire wb_sel_i,

    // UART core connections
    output reg [7:0] uart_wb_dat_o,
    output reg uart_wb_we_i,
    output reg uart_wb_stb_i,
    output reg uart_wb_cyc_i,
    output reg [2:0] uart_wb_adr_i,
    input  wire uart_int_o,

    // Modem control connections
    output reg [7:0] modem_wb_dat_o,
    output reg modem_wb_we_i,
    output reg modem_wb_stb_i,
    output reg modem_wb_cyc_i,
    output reg [2:0] modem_wb_adr_i
);

    reg [7:0] registers [7:0];

    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            wb_dat_o <= 8'b0;
            wb_ack_o <= 1'b0;
            uart_wb_dat_o <= 8'b0;
            uart_wb_we_i <= 1'b0;
            uart_wb_stb_i <= 1'b0;
            uart_wb_cyc_i <= 1'b0;
            uart_wb_adr_i <= 3'b0;
            modem_wb_dat_o <= 8'b0;
            modem_wb_we_i <= 1'b0;
            modem_wb_stb_i <= 1'b0;
            modem_wb_cyc_i <= 1'b0;
            modem_wb_adr_i <= 3'b0;
        end else begin
            if (wb_cyc_i && wb_stb_i && wb_sel_i) begin
                wb_ack_o <= 1'b1;
                if (wb_adr_i < 3'b100) begin // UART registers
                    uart_wb_dat_o <= wb_dat_i;
                    uart_wb_we_i <= wb_we_i;
                    uart_wb_stb_i <= wb_stb_i;
                    uart_wb_cyc_i <= wb_cyc_i;
                    uart_wb_adr_i <= wb_adr_i;
                    if (wb_we_i) begin
                        registers[wb_adr_i] <= wb_dat_i;
                    end else begin
                        wb_dat_o <= registers[wb_adr_i];
                    end
                end else begin // Modem control registers
                    modem_wb_dat_o <= wb_dat_i;
                    modem_wb_we_i <= wb_we_i;
                    modem_wb_stb_i <= wb_stb_i;
                    modem_wb_cyc_i <= wb_cyc_i;
                    modem_wb_adr_i <= wb_adr_i;
                    if (wb_we_i) begin
                        registers[wb_adr_i] <= wb_dat_i;
                    end else begin
                        wb_dat_o <= registers[wb_adr_i];
                    end
                end
            end else begin
                wb_ack_o <= 1'b0;
                uart_wb_we_i <= 1'b0;
                uart_wb_stb_i <= 1'b0;
                uart_wb_cyc_i <= 1'b0;
                modem_wb_we_i <= 1'b0;
                modem_wb_stb_i <= 1'b0;
                modem_wb_cyc_i <= 1'b0;
            end
        end
    end

endmodule

// UART Core module
module uart_core (
    input  wire wb_clk_i,
    input  wire wb_rst_i,
    output reg stx_pad_o,
    input  wire srx_pad_i,
    output reg [7:0] wb_dat_o,
    input  wire wb_we_i,
    input  wire wb_stb_i,
    input  wire wb_cyc_i,
    input  wire [2:0] wb_adr_i,
    output reg int_o
);

    // UART registers
    reg [7:0] rbr; // Receive Buffer Register
    reg [7:0] thr; // Transmit Holding Register
    reg [7:0] ier; // Interrupt Enable Register
    reg [7:0] iir; // Interrupt Identification Register
    reg [7:0] lcr; // Line Control Register
    reg [7:0] mcr; // Modem Control Register
    reg [7:0] lsr; // Line Status Register
    reg [7:0] msr; // Modem Status Register
    reg [7:0] scr; // Scratch Register

    // FIFOs
    reg [7:0] tx_fifo [0:15];
    reg [3:0] tx_fifo_wr_ptr;
    reg [3:0] tx_fifo_rd_ptr;
    reg [3:0] tx_fifo_count;

    reg [7:0] rx_fifo [0:15];
    reg [3:0] rx_fifo_wr_ptr;
    reg [3:0] rx_fifo_rd_ptr;
    reg [3:0] rx_fifo_count;

    // Baud rate generation
    reg [15:0] baud_divisor;
    reg [15:0] baud_counter;

    // State machine for transmitter
    reg [3:0] tx_state;
    reg [2:0] tx_bit_count;
    reg tx_shift_reg;

    // State machine for receiver
    reg [3:0] rx_state;
    reg [2:0] rx_bit_count;
    reg rx_shift_reg;

    // Interrupt handling
    reg line_status_int;
    reg received_data_int;

    // Internal signals
    reg [7:0] tx_data;
    reg [7:0] rx_data;

    // UART core logic
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            rbr <= 8'b0;
            thr <= 8'b0;
            ier <= 8'b0;
            iir <= 8'b0;
            lcr <= 8'b0;
            mcr <= 8'b0;
            lsr <= 8'b0;
            msr <= 8'b0;
            scr <= 8'b0;

            tx_fifo_wr_ptr <= 4'b0;
            tx_fifo_rd_ptr <= 4'b0;
            tx_fifo_count <= 4'b0;

            rx_fifo_wr_ptr <= 4'b0;
            rx_fifo_rd_ptr <= 4'b0;
            rx_fifo_count <= 4'b0;

            baud_divisor <= 16'd0;
            baud_counter <= 16'd0;

            tx_state <= 4'b0;
            tx_bit_count <= 3'b0;
            tx_shift_reg <= 1'b0;

            rx_state <= 4'b0;
            rx_bit_count <= 3'b0;
            rx_shift_reg <= 1'b0;

            line_status_int <= 1'b0;
            received_data_int <= 1'b0;

            tx_data <= 8'b0;
            rx_data <= 8'b0;

            wb_dat_o <= 8'b0;
            int_o <= 1'b0;
            stx_pad_o <= 1'b1;
        end else begin
            // Wishbone read/write handling
            if (wb_cyc_i && wb_stb_i) begin
                if (wb_we_i) begin
                    case (wb_adr_i)
                        3'b000: thr <= wb_dat_i;
                        3'b001: ier <= wb_dat_i;
                        3'b010: fcr <= wb_dat_i;
                        3'b011: lcr <= wb_dat_i;
                        3'b100: mcr <= wb_dat_i;
                        3'b101: lsr <= wb_dat_i;
                        3'b110: msr <= wb_dat_i;
                        3'b111: scr <= wb_dat_i;
                    endcase
                end else begin
                    case (wb_adr_i)
                        3'b000: wb_dat_o <= rbr;
                        3'b001: wb_dat_o <= iir;
                        3'b010: wb_dat_o <= fcr;
                        3'b011: wb_dat_o <= lcr;
                        3'b100: wb_dat_o <= mcr;
                        3'b101: wb_dat_o <= lsr;
                        3'b110: wb_dat_o <= msr;
                        3'b111: wb_dat_o <= scr;
                    endcase
                end
            end

            // Baud rate generation
            if (baud_counter == baud_divisor) begin
                baud_counter <= 16'd0;
            end else begin
                baud_counter <= baud_counter + 1'b1;
            end

            // Transmitter state machine
            case (tx_state)
                4'b0000: begin // Idle
                    if (tx_fifo_count > 4'b0 && baud_counter == 16'd0) begin
                        tx_state <= 4'b0001;
                        tx_shift_reg <= 1'b0; // Start bit
                        tx_bit_count <= 3'b0;
                        tx_data <= tx_fifo[tx_fifo_rd_ptr];
                        tx_fifo_rd_ptr <= tx_fifo_rd_ptr + 1'b1;
                        tx_fifo_count <= tx_fifo_count - 1'b1;
                    end
                end
                4'b0001: begin // Transmitting
                    if (baud_counter == 16'd0) begin
                        if (tx_bit_count < 3'b101) begin
                            tx_bit_count <= tx_bit_count + 1'b1;
                            tx_shift_reg <= tx_data[tx_bit_count];
                        end else begin
                            tx_shift_reg <= 1'b1; // Stop bit
                            tx_state <= 4'b0010;
                        end
                    end
                end
                4'b0010: begin // Stop bit
                    if (baud_counter == 16'd0) begin
                        tx_state <= 4'b0000;
                    end
                end
                default: tx_state <= 4'b0000;
            endcase

            // Receiver state machine
            case (rx_state)
                4'b0000: begin // Idle
                    if (srx_pad_i == 1'b0) begin
                        rx_state <= 4'b0001;
                        rx_bit_count <= 3'b0;
                        rx_shift_reg <= 1'b0;
                    end
                end
                4'b0001: begin // Receiving
                    if (baud_counter == 16'd0) begin
                        if (rx_bit_count < 3'b101) begin
                            rx_bit_count <= rx_bit_count + 1'b1;
                            rx_shift_reg <= srx_pad_i;
                        end else begin
                            rx_data <= {rx_shift_reg, srx_pad_i};
                            rx_state <= 4'b0010;
                        end
                    end
                end
                4'b0010: begin // Stop bit
                    if (baud_counter == 16'd0) begin
                        rx_fifo[rx_fifo_wr_ptr] <= rx_data;
                        rx_fifo_wr_ptr <= rx_fifo_wr_ptr + 1'b1;
                        rx_fifo_count <= rx_fifo_count + 1'b1;
                        rx_state <= 4'b0000;
                    end
                end
                default: rx_state <= 4'b0000;
            endcase

            // Update transmit shift register
            stx_pad_o <= tx_shift_reg;

            // Update receive FIFO
            if (rx_fifo_count < 4'b1000) begin
                rx_fifo[rx_fifo_wr_ptr] <= rx_data;
                rx_fifo_wr_ptr <= rx_fifo_wr_ptr + 1'b1;
                rx_fifo_count <= rx_fifo_count + 1'b1;
            end

            // Update line status register
            if (rx_fifo_count == 4'b1000) begin
                lsr[5] <= 1'b1; // Overrun error
            end else begin
                lsr[5] <= 1'b0;
            end

            // Update interrupt signals
            if (rx_fifo_count > 4'b0) begin
                received_data_int <= 1'b1;
            end else begin
                received_data_int <= 1'b0;
            end

            int_o <= line_status_int | received_data_int;
        end
    end

endmodule

// Modem Control module
module modem_control (
    input  wire wb_clk_i,
    input  wire wb_rst_i,
    output reg rts_pad_o,
    input  wire cts_pad_i,
    output reg dtr_pad_o,
    input  wire dsr_pad_i,
    input  wire ri_pad_i,
    input  wire dcd_pad_i,
    output reg [7:0] wb_dat_o,
    input  wire wb_we_i,
    input  wire wb_stb_i,
    input  wire wb_cyc_i,
    input  wire [2:0] wb_adr_i
);

    // Modem control registers
    reg [7:0] mcr; // Modem Control Register
    reg [7:0] msr; // Modem Status Register

    // Internal signals
    reg cts_reg;
    reg dsr_reg;
    reg ri_reg;
    reg dcd_reg;

    // Modem control logic
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            mcr <= 8'b0;
            msr <= 8'b0;
            rts_pad_o <= 1'b0;
            dtr_pad_o <= 1'b0;
            cts_reg <= 1'b0;
            dsr_reg <= 1'b0;
            ri_reg <= 1'b0;
            dcd_reg <= 1'b0;
            wb_dat_o <= 8'b0;
        end else begin
            // Wishbone read/write handling
            if (wb_cyc_i && wb_stb_i) begin
                if (wb_we_i) begin
                    case (wb_adr_i)
                        3'b100: mcr <= wb_dat_i;
                    endcase
                end else begin
                    case (wb_adr_i)
                        3'b101: wb_dat_o <= msr;
                    endcase
                end
            end

            // Update modem control signals
            rts_pad_o <= mcr[1];
            dtr_pad_o <= mcr[0];

            // Update modem status register
            cts_reg <= cts_pad_i;
            dsr_reg <= dsr_pad_i;
            ri_reg <= ri_pad_i;
            dcd_reg <= dcd_pad_i;

            msr <= {1'b0, ri_reg, 1'b0, dcd_reg, dsr_reg, cts_reg, 1'b0, 1'b0};
        end
    end

endmodule
