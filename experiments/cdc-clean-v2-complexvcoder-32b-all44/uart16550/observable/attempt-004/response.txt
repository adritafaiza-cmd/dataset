```verilog
module uart_top (
	input wb_clk_i,
	input wb_rst_i,
	input [7:0] wb_adr_i,
	input [31:0] wb_dat_i,
	output [31:0] wb_dat_o,
	input wb_we_i,
	input wb_stb_i,
	input wb_cyc_i,
	output wb_ack_o,
	input [3:0] wb_sel_i,
	output int_o,
	output stx_pad_o,
	input srx_pad_i,
	output rts_pad_o,
	input cts_pad_i,
	output dtr_pad_o,
	input dsr_pad_i,
	input ri_pad_i,
	input dcd_pad_i
);

	// Internal signals for handshaking and data transfer
	wire [7:0] tx_data;
	wire tx_ready;
	wire [7:0] rx_data;
	wire rx_valid;

	// Wishbone Slave module
	wishbone_slave wb_slave_inst (
		.wb_clk_i(wb_clk_i),
		.wb_rst_i(wb_rst_i),
		.wb_adr_i(wb_adr_i),
		.wb_dat_i(wb_dat_i),
		.wb_dat_o(wb_dat_o),
		.wb_we_i(wb_we_i),
		.wb_stb_i(wb_stb_i),
		.wb_cyc_i(wb_cyc_i),
		.wb_ack_o(wb_ack_o),
		.wb_sel_i(wb_sel_i),
		.tx_data(tx_data),
		.tx_ready(tx_ready),
		.rx_data(rx_data),
		.rx_valid(rx_valid)
	);

	// UART Core module
	uart_core uart_core_inst (
		.wb_clk_i(wb_clk_i),
		.wb_rst_i(wb_rst_i),
		.stx_pad_o(stx_pad_o),
		.srx_pad_i(srx_pad_i),
		.int_o(int_o),
		.tx_data(tx_data),
		.tx_ready(tx_ready),
		.rx_data(rx_data),
		.rx_valid(rx_valid)
	);

	// Modem Control module
	modem_control modem_control_inst (
		.wb_clk_i(wb_clk_i),
		.wb_rst_i(wb_rst_i),
		.rts_pad_o(rts_pad_o),
		.cts_pad_i(cts_pad_i),
		.dtr_pad_o(dtr_pad_o),
		.dsr_pad_i(dsr_pad_i),
		.ri_pad_i(ri_pad_i),
		.dcd_pad_i(dcd_pad_i)
	);

endmodule

// Wishbone Slave module
module wishbone_slave (
	input wb_clk_i,
	input wb_rst_i,
	input [7:0] wb_adr_i,
	input [31:0] wb_dat_i,
	output [31:0] wb_dat_o,
	input wb_we_i,
	input wb_stb_i,
	input wb_cyc_i,
	output wb_ack_o,
	input [3:0] wb_sel_i,
	output [7:0] tx_data,
	input tx_ready,
	input [7:0] rx_data,
	input rx_valid
);

	reg [31:0] wb_dat_o_reg;
	reg wb_ack_o_reg;
	reg [7:0] tx_data_reg;
	reg tx_ready_reg;
	reg [7:0] rx_data_reg;
	reg rx_valid_reg;

	assign wb_dat_o = wb_dat_o_reg;
	assign wb_ack_o = wb_ack_o_reg;
	assign tx_data = tx_data_reg;
	assign tx_ready = tx_ready_reg;
	assign rx_data_reg = rx_data;
	assign rx_valid_reg = rx_valid;

	always @(posedge wb_clk_i or posedge wb_rst_i) begin
		if (wb_rst_i) begin
			wb_dat_o_reg <= 32'b0;
			wb_ack_o_reg <= 1'b0;
			tx_data_reg <= 8'b0;
			tx_ready_reg <= 1'b0;
		end else begin
			if (wb_cyc_i && wb_stb_i) begin
				wb_ack_o_reg <= 1'b1;
				if (wb_we_i && wb_sel_i[0]) begin
					tx_data_reg <= wb_dat_i[7:0];
				end
				if (wb_we_i && wb_sel_i[1]) begin
					tx_ready_reg <= wb_dat_i[8];
				end
				if (!wb_we_i) begin
					case (wb_adr_i)
						8'h00: wb_dat_o_reg <= {24'b0, tx_data_reg};
						8'h01: wb_dat_o_reg <= {31'b0, tx_ready_reg};
						8'h02: wb_dat_o_reg <= {24'b0, rx_data_reg};
						8'h03: wb_dat_o_reg <= {31'b0, rx_valid_reg};
						default: wb_dat_o_reg <= 32'b0;
					endcase
				end
			end else begin
				wb_ack_o_reg <= 1'b0;
			end
		end
	end

endmodule

// UART Core module
module uart_core (
	input wb_clk_i,
	input wb_rst_i,
	output stx_pad_o,
	input srx_pad_i,
	output int_o,
	input [7:0] tx_data,
	input tx_ready,
	output [7:0] rx_data,
	output rx_valid
);

	reg [7:0] tx_data_reg;
	reg tx_ready_reg;
	reg [7:0] rx_data_reg;
	reg rx_valid_reg;
	reg int_o_reg;

	assign stx_pad_o = tx_data_reg[0];
	assign tx_data_reg = {tx_data_reg[6:0], tx_ready};
	assign tx_ready_reg = tx_ready;
	assign rx_data = rx_data_reg;
	assign rx_valid = rx_valid_reg;
	assign int_o = int_o_reg;

	always @(posedge wb_clk_i or posedge wb_rst_i) begin
		if (wb_rst_i) begin
			tx_data_reg <= 8'b0;
			tx_ready_reg <= 1'b0;
			rx_data_reg <= 8'b0;
			rx_valid_reg <= 1'b0;
			int_o_reg <= 1'b0;
		end else begin
			if (tx_ready) begin
				tx_data_reg <= tx_data;
			end
			if (srx_pad_i) begin
				rx_data_reg <= {rx_data_reg[6:0], srx_pad_i};
				rx_valid_reg <= 1'b1;
			end else begin
				rx_valid_reg <= 1'b0;
			end
			// Placeholder for interrupt generation logic
			int_o_reg <= rx_valid_reg;
		end
	end

endmodule

// Modem Control module
module modem_control (
	input wb_clk_i,
	input wb_rst_i,
	output rts_pad_o,
	input cts_pad_i,
	output dtr_pad_o,
	input dsr_pad_i,
	input ri_pad_i,
	input dcd_pad_i
);

	reg rts_pad_o_reg;
	reg dtr_pad_o_reg;

	assign rts_pad_o = rts_pad_o_reg;
	assign dtr_pad_o = dtr_pad_o_reg;

	always @(posedge wb_clk_i or posedge wb_rst_i) begin
		if (wb_rst_i) begin
			rts_pad_o_reg <= 1'b0;
			dtr_pad_o_reg <= 1'b0;
		end else begin
			// Placeholder for modem control logic
			rts_pad_o_reg <= ~cts_pad_i; // Simple example: RTS active when CTS is inactive
			dtr_pad_o_reg <= 1'b1; // DTR always active
		end
	end

endmodule
```
