module uart_wishbone (
	input wb_clk_i,
	input wb_rst_i,
	input [3:0] wb_adr_i,
	input [7:0] wb_dat_i,
	output [7:0] wb_dat_o,
	input wb_we_i,
	input wb_stb_i,
	input wb_cyc_i,
	output wb_ack_o,
	input wb_sel_i,
	output [7:0] tx_data,
	output tx_start,
	input [7:0] rx_data,
	output rx_ready,
	input [7:0] interrupt_request,
	input [7:0] interrupt_mask,
	output cts,
	output dsr,
	output ri,
	output dcd,
	input rts,
	input dtr
);

	reg [7:0] registers [15:0];
	reg [7:0] wb_dat_o_reg;
	reg wb_ack_o_reg;

	assign wb_dat_o = wb_dat_o_reg;
	assign wb_ack_o = wb_ack_o_reg;
	assign tx_data = registers[0];
	assign tx_start = wb_we_i & wb_stb_i & wb_cyc_i & wb_sel_i & (wb_adr_i == 0);
	assign rx_ready = 1'b1;
	assign cts = registers[6][4];
	assign dsr = registers[6][5];
	assign ri = registers[6][6];
	assign dcd = registers[6][7];

	always @(posedge wb_clk_i or posedge wb_rst_i) begin
		if (wb_rst_i) begin
			wb_dat_o_reg <= 8'b0;
			wb_ack_o_reg <= 1'b0;
			for (int i = 0; i < 16; i = i + 1)
				registers[i] <= 8'b0;
		end else begin
			wb_ack_o_reg <= wb_stb_i & wb_cyc_i & wb_sel_i;
			if (wb_stb_i & wb_cyc_i & wb_sel_i) begin
				if (wb_we_i) begin
					if (wb_adr_i < 16)
						registers[wb_adr_i] <= wb_dat_i;
				end else begin
					if (wb_adr_i < 16)
						wb_dat_o_reg <= registers[wb_adr_i];
				end
			end
		end
	end

	always @(posedge wb_clk_i or posedge wb_rst_i) begin
		if (wb_rst_i)
			registers[1] <= rx_data;
		else if (rx_ready)
			registers[1] <= rx_data;
	end

	always @(posedge wb_clk_i or posedge wb_rst_i) begin
		if (wb_rst_i)
			registers[2] <= interrupt_request & interrupt_mask;
		else
			registers[2] <= interrupt_request & interrupt_mask;
	end

	always @(posedge wb_clk_i or posedge wb_rst_i) begin
		if (wb_rst_i)
			registers[6] <= {dcd_pad_i, dsr_pad_i, ri_pad_i, cts_pad_i, 4'b0};
		else
			registers[6] <= {dcd_pad_i, dsr_pad_i, ri_pad_i, cts_pad_i, 4'b0};
	end

endmodule

module uart_transmitter (
	input baud_clk,
	input wb_rst_i,
	input [7:0] tx_data,
	input tx_start,
	output stx_pad_o
);

	reg [7:0] shift_register;
	reg [3:0] bit_count;
	reg [1:0] tx_state;
	reg tx_shift;

	parameter IDLE = 2'b00;
	parameter START = 2'b01;
	parameter DATA = 2'b10;
	parameter STOP = 2'b11;

	assign stx_pad_o = (tx_state == IDLE) ? 1'b1 : (tx_state == STOP) ? 1'b1 : shift_register[0];

	always @(posedge baud_clk or posedge wb_rst_i) begin
		if (wb_rst_i) begin
			shift_register <= 8'b0;
			bit_count <= 4'b0;
			tx_state <= IDLE;
			tx_shift <= 1'b0;
		end else begin
			case (tx_state)
				IDLE: begin
					if (tx_start) begin
						shift_register <= {tx_data, 1'b0};
						bit_count <= 4'd0;
						tx_state <= START;
					end
				end
				START: begin
					tx_state <= DATA;
				end
				DATA: begin
					if (tx_shift) begin
						shift_register <= {1'b1, shift_register[7:1]};
						if (bit_count == 4'd7)
							tx_state <= STOP;
						else
							bit_count <= bit_count + 1;
					end
				end
				STOP: begin
					tx_state <= IDLE;
				end
			endcase
		end
	end

	always @(posedge baud_clk) begin
		tx_shift <= (tx_state == DATA) ? 1'b1 : 1'b0;
	end

endmodule

module uart_receiver (
	input baud_clk,
	input wb_rst_i,
	output reg [7:0] rx_data,
	input rx_ready,
	input srx_pad_i
);

	reg [7:0] shift_register;
	reg [3:0] bit_count;
	reg [1:0] rx_state;
	reg rx_shift;

	parameter IDLE = 2'b00;
	parameter START = 2'b01;
	parameter DATA = 2'b10;
	parameter STOP = 2'b11;

	always @(posedge baud_clk or posedge wb_rst_i) begin
		if (wb_rst_i) begin
			shift_register <= 8'b0;
			bit_count <= 4'b0;
			rx_state <= IDLE;
			rx_shift <= 1'b0;
			rx_data <= 8'b0;
		end else begin
			case (rx_state)
				IDLE: begin
					if (!srx_pad_i) begin
						rx_state <= START;
					end
				end
				START: begin
					rx_state <= DATA;
				end
				DATA: begin
					if (rx_shift) begin
						shift_register <= {srx_pad_i, shift_register[7:1]};
						if (bit_count == 4'd7) begin
							rx_state <= STOP;
							rx_data <= shift_register;
						end else begin
							bit_count <= bit_count + 1;
						end
					end
				end
				STOP: begin
					rx_state <= IDLE;
					bit_count <= 4'b0;
				end
			endcase
		end
	end

	always @(posedge baud_clk) begin
		rx_shift <= (rx_state == DATA) ? 1'b1 : 1'b0;
	end

endmodule

module uart_interrupt (
	input wb_clk_i,
	input wb_rst_i,
	input [7:0] interrupt_request,
	input [7:0] interrupt_mask,
	output int_o
);

	wire [7:0] masked_interrupts;

	assign masked_interrupts = interrupt_request & interrupt_mask;
	assign int_o = |masked_interrupts;

endmodule

module uart_modem (
	input wb_clk_i,
	input wb_rst_i,
	input cts_pad_i,
	input dsr_pad_i,
	input ri_pad_i,
	input dcd_pad_i,
	output rts_pad_o,
	output dtr_pad_o,
	output cts,
	output dsr,
	output ri,
	output dcd,
	input rts,
	input dtr
);

	assign rts_pad_o = rts;
	assign dtr_pad_o = dtr;
	assign cts = cts_pad_i;
	assign dsr = dsr_pad_i;
	assign ri = ri_pad_i;
	assign dcd = dcd_pad_i;

endmodule

module uart_baud_gen (
	input wb_clk_i,
	input wb_rst_i,
	output baud_clk
);

	reg [15:0] baud_divisor;
	reg [15:0] baud_counter;
	reg baud_clk_reg;

	always @(posedge wb_clk_i or posedge wb_rst_i) begin
		if (wb_rst_i) begin
			baud_divisor <= 16'd104; // 9600 baud rate assuming 100MHz wb_clk_i
			baud_counter <= 16'd0;
			baud_clk_reg <= 1'b0;
		end else begin
			if (baud_counter == baud_divisor) begin
				baud_counter <= 16'd0;
				baud_clk_reg <= ~baud_clk_reg;
			end else begin
				baud_counter <= baud_counter + 1;
			end
		end
	end

	assign baud_clk = baud_clk_reg;

endmodule
