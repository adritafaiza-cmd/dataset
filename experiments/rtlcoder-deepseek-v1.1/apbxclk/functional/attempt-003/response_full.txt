  output wire [C_APB_DATA_WIDTH-1:0]  M_APB_PWDATA,
  output wire [C_APB_DATA_WIDTH/8-1:0] M_APB_PWSTRB,
  output wire [2:0]                   M_APB_PPROT,

reg [C_APB_ADDR_WIDTH-1:0] M_APB_PADDR_reg;
reg [C_APB_DATA_WIDTH-1:0] M_APB_PWDATA_reg;
reg [C_APB_DATA_WIDTH/8-1:0] M_APB_PWSTRB_reg;
reg [2:0] M_APB_PPROT_reg;

reg [C_APB_DATA_WIDTH-1:0] S_APB_PRDATA_reg;
reg S_APB_PSLVERR_reg;

reg [1:0] state;
parameter IDLE = 2'b00;
parameter SETUP = 2'b01;
parameter ACCESS = 2'b10;

always @(posedge S_APB_PCLK) begin
  if (!S_PRESETn) begin
    S_APB_PREADY <= 1'b0;
    M_PRESETn <= 1'b0;
    M_APB_PSEL <= 1'b0;
    M_APB_PENABLE <= 1'b0;
    M_APB_PADDR_reg <= 0;
    M_APB_PWDATA_reg <= 0;
    M_APB_PWSTRB_reg <= 0;
    M_APB_PPROT_reg <= 0;
    state <= IDLE;
  end else begin
    case (state)
      IDLE: begin
        if (S_APB_PSEL && S_APB_PENABLE) begin
          S_APB_PREADY <= 1'b1;
          M_PRESETn <= 1'b1;
          M_APB_PSEL <= 1'b1;
          M_APB_PENABLE <= 1'b1;
          M_APB_PADDR_reg <= S_APB_PADDR;
          M_APB_PWDATA_reg <= S_APB_PWDATA;
          M_APB_PWSTRB_reg <= S_APB_PWSTRB;
          M_APB_PPROT_reg <= S_APB_PPROT;
          state <= SETUP;
        end else begin
          S_APB_PREADY <= 1'b0;
          M_PRESETn <= 1'b0;
          M_APB_PSEL <= 1'b0;
          M_APB_PENABLE <= 1'b0;
          M_APB_PADDR_reg <= 0;
          M_APB_PWDATA_reg <= 0;
          M_APB_PWSTRB_reg <= 0;
          M_APB_PPROT_reg <= 0;
          state <= IDLE;
        end
      end
      SETUP: begin
        if (M_APB_PREADY) begin
          M_APB_PSEL <= 1'b0;
          M_APB_PENABLE <= 1'b0;
          state <= ACCESS;
        end
      end
      ACCESS: begin
        if (M_APB_PREADY) begin
          S_APB_PRDATA_reg <= M_APB_PRDATA;
          S_APB_PSLVERR_reg <= M_APB_PSLVERR;
          S_APB_PREADY <= 1'b0;
          state <= IDLE;
        end
      end
    endcase
  end
end
