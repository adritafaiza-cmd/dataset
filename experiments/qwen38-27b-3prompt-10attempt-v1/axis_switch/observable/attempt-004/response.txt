    // s_tready[s] = 1 if there exists an output m such that:
    // - can_drive[m*S_COUNT + s] == 1
    // - m_axis_tready[m] == 1
    // - src_valid[m] == 1 && src_sel[m] == s

        for (gs = 0; gs < S_COUNT; gs = gs + 1) begin : gen_sready

            endgenerate
            assign s_ready_comb[gs] = |ready_mask;
        end
    endgenerate
