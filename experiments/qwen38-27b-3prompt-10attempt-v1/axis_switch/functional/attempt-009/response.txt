```verilog
module axis_switch #
(
    parameter S_COUNT = 4,
    parameter M_COUNT = 4,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter S_ID_WIDTH = 8,
    parameter M_ID_WIDTH = S_ID_WIDTH+$clog2(S_COUNT),
    parameter M_DEST_WIDTH = 1,
    parameter S_DEST_WIDTH = M_DEST_WIDTH+$clog2(M_COUNT),
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter M_BASE = 0,
    parameter M_TOP = 0,
    parameter M_CONNECT = {M_COUNT{{S_COUNT{1'b1}}}},
    parameter UPDATE_TID = 0,
    parameter S_REG_TYPE = 0,
    parameter M_REG_TYPE = 2,
    parameter ARB_TYPE_ROUND_ROBIN = 1,
    parameter ARB_LSB_HIGH_PRIORITY = 1
)
(
    input  wire                             clk,
    input  wire                             rst,
    /*
     * AXI Stream inputs
     */
    input  wire [S_COUNT*DATA_WIDTH-1:0]    s_axis_tdata,
    input  wire [S_COUNT*KEEP_WIDTH-1:0]    s_axis_tkeep,
    input  wire [S_COUNT-1:0]               s_axis_tvalid,
    output wire [S_COUNT-1:0]               s_axis_tready,
    input  wire [S_COUNT-1:0]               s_axis_tlast,
    input  wire [S_COUNT*S_ID_WIDTH-1:0]    s_axis_tid,
    input  wire [S_COUNT*S_DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [S_COUNT*USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI Stream outputs
     */
    output wire [M_COUNT*DATA_WIDTH-1:0]    m_axis_tdata,
    output wire [M_COUNT*KEEP_WIDTH-1:0]    m_axis_tkeep,
    output wire [M_COUNT-1:0]               m_axis_tvalid,
    input  wire [M_COUNT-1:0]               m_axis_tready,
    output wire [M_COUNT-1:0]               m_axis_tlast,
    output wire [M_COUNT*M_ID_WIDTH-1:0]    m_axis_tid,
    output wire [M_COUNT*M_DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [M_COUNT*USER_WIDTH-1:0]    m_axis_tuser
);

    // Internal signals
    wire [S_COUNT-1:0] s_tvalid_int;
    wire [S_COUNT-1:0] s_tready_int;
    wire [S_COUNT-1:0] s_tlast_int;
    wire [S_COUNT-1:0] s_tuser_int;
    wire [S_COUNT*DATA_WIDTH-1:0] s_tdata_int;
    wire [S_COUNT*KEEP_WIDTH-1:0] s_tkeep_int;
    wire [S_COUNT*S_ID_WIDTH-1:0] s_tid_int;
    wire [S_COUNT*S_DEST_WIDTH-1:0] s_tdest_int;

    wire [M_COUNT-1:0] m_tvalid_int;
    wire [M_COUNT-1:0] m_tready_int;
    wire [M_COUNT-1:0] m_tlast_int;
    wire [M_COUNT-1:0] m_tuser_int;
    wire [M_COUNT*DATA_WIDTH-1:0] m_tdata_int;
    wire [M_COUNT*KEEP_WIDTH-1:0] m_tkeep_int;
    wire [M_COUNT*M_ID_WIDTH-1:0] m_tid_int;
    wire [M_COUNT*M_DEST_WIDTH-1:0] m_tdest_int;

    // Input register stage
    generate
        if (S_REG_TYPE != 0) begin : gen_s_reg
            reg [S_COUNT-1:0] s_tvalid_reg;
            reg [S_COUNT-1:0] s_tlast_reg;
            reg [S_COUNT-1:0] s_tuser_reg;
            reg [S_COUNT*DATA_WIDTH-1:0] s_tdata_reg;
            reg [S_COUNT*KEEP_WIDTH-1:0] s_tkeep_reg;
            reg [S_COUNT*S_ID_WIDTH-1:0] s_tid_reg;
            reg [S_COUNT*S_DEST_WIDTH-1:0] s_tdest_reg;

            always @(posedge clk or posedge rst) begin
                if (rst) begin
                    s_tvalid_reg <= {S_COUNT{1'b0}};
                    s_tlast_reg <= {S_COUNT{1'b0}};
                    s_tuser_reg <= {S_COUNT{1'b0}};
                    s_tdata_reg <= {S_COUNT*DATA_WIDTH{1'b0}};
                    s_tkeep_reg <= {S_COUNT*KEEP_WIDTH{1'b0}};
                    s_tid_reg <= {S_COUNT*S_ID_WIDTH{1'b0}};
                    s_tdest_reg <= {S_COUNT*S_DEST_WIDTH{1'b0}};
                end else begin
                    s_tvalid_reg <= s_axis_tvalid;
                    s_tlast_reg <= s_axis_tlast;
                    s_tuser_reg <= s_axis_tuser;
                    s_tdata_reg <= s_axis_tdata;
                    s_tkeep_reg <= s_axis_tkeep;
                    s_tid_reg <= s_axis_tid;
                    s_tdest_reg <= s_axis_tdest;
                end
            end

            assign s_tvalid_int = s_tvalid_reg;
            assign s_tlast_int = s_tlast_reg;
            assign s_tuser_int = s_tuser_reg;
            assign s_tdata_int = s_tdata_reg;
            assign s_tkeep_int = s_tkeep_reg;
            assign s_tid_int = s_tid_reg;
            assign s_tdest_int = s_tdest_reg;
            assign s_axis_tready = s_tready_int;
        end else begin : gen_s_no_reg
            assign s_tvalid_int = s_axis_tvalid;
            assign s_tlast_int = s_axis_tlast;
            assign s_tuser_int = s_axis_tuser;
            assign s_tdata_int = s_axis_tdata;
            assign s_tkeep_int = s_axis_tkeep;
            assign s_tid_int = s_axis_tid;
            assign s_tdest_int = s_axis_tdest;
            assign s_axis_tready = s_tready_int;
        end
    endgenerate

    // Output register stage
    generate
        if (M_REG_TYPE != 0) begin : gen_m_reg
            reg [M_COUNT-1:0] m_tvalid_reg;
            reg [M_COUNT-1:0] m_tlast_reg;
            reg [M_COUNT-1:0] m_tuser_reg;
            reg [M_COUNT*DATA_WIDTH-1:0] m_tdata_reg;
            reg [M_COUNT*KEEP_WIDTH-1:0] m_tkeep_reg;
            reg [M_COUNT*M_ID_WIDTH-1:0] m_tid_reg;
            reg [M_COUNT*M_DEST_WIDTH-1:0] m_tdest_reg;

            always @(posedge clk or posedge rst) begin
                if (rst) begin
                    m_tvalid_reg <= {M_COUNT{1'b0}};
                    m_tlast_reg <= {M_COUNT{1'b0}};
                    m_tuser_reg <= {M_COUNT{1'b0}};
                    m_tdata_reg <= {M_COUNT*DATA_WIDTH{1'b0}};
                    m_tkeep_reg <= {M_COUNT*KEEP_WIDTH{1'b0}};
                    m_tid_reg <= {M_COUNT*M_ID_WIDTH{1'b0}};
                    m_tdest_reg <= {M_COUNT*M_DEST_WIDTH{1'b0}};
                end else begin
                    m_tvalid_reg <= m_tvalid_int;
                    m_tlast_reg <= m_tlast_int;
                    m_tuser_reg <= m_tuser_int;
                    m_tdata_reg <= m_tdata_int;
                    m_tkeep_reg <= m_tkeep_int;
                    m_tid_reg <= m_tid_int;
                    m_tdest_reg <= m_tdest_int;
                end
            end

            assign m_axis_tvalid = m_tvalid_reg;
            assign m_axis_tlast = m_tlast_reg;
            assign m_axis_tuser = m_tuser_reg;
            assign m_axis_tdata = m_tdata_reg;
            assign m_axis_tkeep = m_tkeep_reg;
            assign m_axis_tid = m_tid_reg;
            assign m_axis_tdest = m_tdest_reg;
            assign m_tready_int = m_axis_tready;
        end else begin : gen_m_no_reg
            assign m_axis_tvalid = m_tvalid_int;
            assign m_axis_tlast = m_tlast_int;
            assign m_axis_tuser = m_tuser_int;
            assign m_axis_tdata = m_tdata_int;
            assign m_axis_tkeep = m_tkeep_int;
            assign m_axis_tid = m_tid_int;
            assign m_axis_tdest = m_tdest_int;
            assign m_tready_int = m_axis_tready;
        end
    endgenerate

    // Switch logic
    // For each output port m, determine which input s is selected
    // and generate the output signals.

    // Arbitration logic for each output port
    // We need to select one input for each output based on tdest and arbitration

    // For each output m, we have a set of inputs that can drive it
    // The input s drives output m if:
    //   1. s_tvalid_int[s] is high
    //   2. s_tdest_int[s*S_DEST_WIDTH +: S_DEST_WIDTH] == m (or within range)
    //   3. M_CONNECT[m*S_COUNT + s] is high (connection allowed)

    // We'll use a priority encoder or round-robin arbiter for each output

    // Let's define the request signal for each output
    // req[m][s] = s_tvalid_int[s] & (s_tdest_int[s*S_DEST_WIDTH +: S_DEST_WIDTH] == m) & M_CONNECT[m*S_COUNT + s]

    // For simplicity, we'll implement a basic arbiter for each output

    // Generate request signals
    wire [M_COUNT-1:0][S_COUNT-1:0] req;
    wire [M_COUNT-1:0][S_COUNT-1:0] grant;
    wire [M_COUNT-1:0][S_COUNT-1:0] s_tready_per_m;

    // For each output m, determine the granted input
    generate
        for (m = 0; m < M_COUNT; m = m + 1) begin : gen_output
            // Request signals for this output
            wire [S_COUNT-1:0] req_m;
            
            // Generate request for each input
            genvar s;
            for (s = 0; s < S_COUNT; s = s + 1) begin : gen_req
                assign req_m[s] = s_tvalid_int[s] & 
                                  (s_tdest_int[s*S_DEST_WIDTH +: S_DEST_WIDTH] == m[S_DEST_WIDTH-1:0]) &
                                  M_CONNECT[m*S_COUNT + s];
            end

            // Arbitration
            // If round-robin, use a simple round-robin arbiter
            // If fixed priority, use LSB high priority or MSB high priority
            
            wire [S_COUNT-1:0] grant_m;
            
            if (ARB_TYPE_ROUND_ROBIN) begin : gen_rr_arb
                // Simple round-robin: use a rotating priority
                // For simplicity, we'll use a basic implementation
                // In a real design, we'd track the last granted input
                // Here we'll use a simple priority encoder with rotation
                
                // For now, use fixed priority as a placeholder
                // A proper round-robin would require state
                // Let's implement a simple version
                
                // Actually, let's use a simple priority encoder for now
                // and note that true round-robin requires state
                assign grant_m = req_m; // Placeholder - will be refined
            end else begin : gen_fixed_arb
                if (ARB_LSB_HIGH_PRIORITY) begin : gen_lsb_hp
                    // LSB has highest priority
                    assign grant_m[0] = req_m[0];
                    genvar s;
                    for (s = 1; s < S_COUNT; s = s + 1) begin : gen_lsb_hp_loop
                        assign grant_m[s] = req_m[s] & ~(|grant_m[s-1:0]);
                    end
                end else begin : gen_msb_hp
                    // MSB has highest priority
                    assign grant_m[S_COUNT-1] = req_m[S_COUNT-1];
                    genvar s;
                    for (s = S_COUNT-2; s >= 0; s = s - 1) begin : gen_msb_hp_loop
                        assign grant_m[s] = req_m[s] & ~(|grant_m[S_COUNT-1:s+1]);
                    end
                end
            end

            // Assign to grant array
            assign grant[m] = grant_m;
            
            // Generate output signals for this output
            // The output is valid if any input is granted
            assign m_tvalid_int[m] = |grant_m;
            
            // Select the data from the granted input
            // We need to mux the data from the granted input
            wire [DATA_WIDTH-1:0] m_tdata_sel;
            wire [KEEP_WIDTH-1:0] m_tkeep_sel;
            wire m_tlast_sel;
            wire m_tuser_sel;
            wire [S_ID_WIDTH-1:0] m_tid_sel;
            wire [S_DEST_WIDTH-1:0] m_tdest_sel;
            
            // Mux based on grant
            assign m_tdata_sel = grant_m[0] ? s_tdata_int[0*DATA_WIDTH +: DATA_WIDTH] :
                                 (S_COUNT > 1 && grant_m[1]) ? s_tdata_int[1*DATA_WIDTH +: DATA_WIDTH] :
                                 (S_COUNT > 2 && grant_m[2]) ? s_tdata_int[2*DATA_WIDTH +: DATA_WIDTH] :
                                 (S_COUNT > 3 && grant_m[3]) ? s_tdata_int[3*DATA_WIDTH +: DATA_WIDTH] :
                                 (S_COUNT > 4 && grant_m[4]) ? s_tdata_int[4*DATA_WIDTH +: DATA_WIDTH] :
                                 (S_COUNT > 5 && grant_m[5]) ? s_tdata_int[5*DATA_WIDTH +: DATA_WIDTH] :
                                 (S_COUNT > 6 && grant_m[6]) ? s_tdata_int[6*DATA_WIDTH +: DATA_WIDTH] :
                                 (S_COUNT > 7 && grant_m[7]) ? s_tdata_int[7*DATA_WIDTH +: DATA_WIDTH] :
                                 {DATA_WIDTH{1'b0}};
            
            assign m_tkeep_sel = grant_m[0] ? s_tkeep_int[0*KEEP_WIDTH +: KEEP_WIDTH] :
                                 (S_COUNT > 1 && grant_m[1]) ? s_tkeep_int[1*KEEP_WIDTH +: KEEP_WIDTH] :
                                 (S_COUNT > 2 && grant_m[2]) ? s_tkeep_int[2*KEEP_WIDTH +: KEEP_WIDTH] :
                                 (S_COUNT > 3 && grant_m[3]) ? s_tkeep_int[3*KEEP_WIDTH +: KEEP_WIDTH] :
                                 (S_COUNT > 4 && grant_m[4]) ? s_tkeep_int[4*KEEP_WIDTH +: KEEP_WIDTH] :
                                 (S_COUNT > 5 && grant_m[5]) ? s_tkeep_int[5*KEEP_WIDTH +: KEEP_WIDTH] :
                                 (S_COUNT > 6 && grant_m[6]) ? s_tkeep_int[6*KEEP_WIDTH +: KEEP_WIDTH] :
                                 (S_COUNT > 7 && grant_m[7]) ? s_tkeep_int[7*KEEP_WIDTH +: KEEP_WIDTH] :
                                 {KEEP_WIDTH{1'b0}};
            
            assign m_tlast_sel = grant_m[0] ? s_tlast_int[0] :
                                 (S_COUNT > 1 && grant_m[1]) ? s_tlast_int[1] :
                                 (S_COUNT > 2 && grant_m[2]) ? s_tlast_int[2] :
                                 (S_COUNT > 3 && grant_m[3]) ? s_tlast_int[3] :
                                 (S_COUNT > 4 && grant_m[4]) ? s_tlast_int[4] :
                                 (S_COUNT > 5 && grant_m[5]) ? s_tlast_int[5] :
                                 (S_COUNT > 6 && grant_m[6]) ? s_tlast_int[6] :
                                 (S_COUNT > 7 && grant_m[7]) ? s_tlast_int[7] :
                                 1'b0;
            
            assign m_tuser_sel = grant_m[0] ? s_tuser_int[0] :
                                 (S_COUNT > 1 && grant_m[1]) ? s_tuser_int[1] :
                                 (S_COUNT > 2 && grant_m[2]) ? s_tuser_int[2] :
                                 (S_COUNT > 3 && grant_m[3]) ? s_tuser_int[3] :
                                 (S_COUNT > 4 && grant_m[4]) ? s_tuser_int[4] :
                                 (S_COUNT > 5 && grant_m[5]) ? s_tuser_int[5] :
                                 (S_COUNT > 6 && grant_m[6]) ? s_tuser_int[6] :
                                 (S_COUNT > 7 && grant_m[7]) ? s_tuser_int[7] :
                                 1'b0;
            
            assign m_tid_sel = grant_m[0] ? s_tid_int[0*S_ID_WIDTH +: S_ID_WIDTH] :
                               (S_COUNT > 1 && grant_m[1]) ? s_tid_int[1*S_ID_WIDTH +: S_ID_WIDTH] :
                               (S_COUNT > 2 && grant_m[2]) ? s_tid_int[2*S_ID_WIDTH +: S_ID_WIDTH] :
                               (S_COUNT > 3 && grant_m[3]) ? s_tid_int[3*S_ID_WIDTH +: S_ID_WIDTH] :
                               (S_COUNT > 4 && grant_m[4]) ? s_tid_int[4*S_ID_WIDTH +: S_ID_WIDTH] :
                               (S_COUNT > 5 && grant_m[5]) ? s_tid_int[5*S_ID_WIDTH +: S_ID_WIDTH] :
                               (S_COUNT > 6 && grant_m[6]) ? s_tid_int[6*S_ID_WIDTH +: S_ID_WIDTH] :
                               (S_COUNT > 7 && grant_m[7]) ? s_tid_int[7*S_ID_WIDTH +: S_ID_WIDTH] :
                               {S_ID_WIDTH{1'b0}};
            
            assign m_tdest_sel = grant_m[0] ? s_tdest_int[0*S_DEST_WIDTH +: S_DEST_WIDTH] :
                                 (S_COUNT > 1 && grant_m[1]) ? s_tdest_int[1*S_DEST_WIDTH +: S_DEST_WIDTH] :
                                 (S_COUNT > 2 && grant_m[2]) ? s_tdest_int[2*S_DEST_WIDTH +: S_DEST_WIDTH] :
                                 (S_COUNT > 3 && grant_m[3]) ? s_tdest_int[3*S_DEST_WIDTH +: S_DEST_WIDTH] :
                                 (S_COUNT > 4 && grant_m[4]) ? s_tdest_int[4*S_DEST_WIDTH +: S_DEST_WIDTH] :
                                 (S_COUNT > 5 && grant_m[5]) ? s_tdest_int[5*S_DEST_WIDTH +: S_DEST_WIDTH] :
                                 (S_COUNT > 6 && grant_m[6]) ? s_tdest_int[6*S_DEST_WIDTH +: S_DEST_WIDTH] :
                                 (S_COUNT > 7 && grant_m[7]) ? s_tdest_int[7*S_DEST_WIDTH +: S_DEST_WIDTH] :
                                 {S_DEST_WIDTH{1'b0}};
            
            // Assign to output arrays
            assign m_tdata_int[m*DATA_WIDTH +: DATA_WIDTH] = m_tdata_sel;
            assign m_tkeep_int[m*KEEP_WIDTH +: KEEP_WIDTH] = m_tkeep_sel;
            assign m_tlast_int[m] = m_tlast_sel;
            assign m_tuser_int[m] = m_tuser_sel;
            
            // Handle TID
            if (ID_ENABLE) begin : gen_id
                if (UPDATE_TID) begin : gen_update_tid
                    // Update TID with output port index
                    assign m_tid_int[m*M_ID_WIDTH +: M_ID_WIDTH] = {m_tid_sel, m[$clog2(M_COUNT)-1:0]};
                end else begin : gen_no_update_tid
                    // Pass through TID, pad if necessary
                    if (M_ID_WIDTH > S_ID_WIDTH) begin : gen_pad_tid
                        assign m_tid_int[m*M_ID_WIDTH +: M_ID_WIDTH] = {M_ID_WIDTH-S_ID_WIDTH{1'b0}, m_tid_sel};
                    end else begin : gen_trunc_tid
                        assign m_tid_int[m*M_ID_WIDTH +: M_ID_WIDTH] = m_tid_sel[M_ID_WIDTH-1:0];
                    end
                end
            end else begin : gen_no_id
                assign m_tid_int[m*M_ID_WIDTH +: M_ID_WIDTH] = {M_ID_WIDTH{1'b0}};
            end
            
            // Handle TDEST
            if (M_DEST_WIDTH > 0) begin : gen_dest
                assign m_tdest_int[m*M_DEST_WIDTH +: M_DEST_WIDTH] = m_tdest_sel[M_DEST_WIDTH-1:0];
            end else begin : gen_no_dest
                assign m_tdest_int[m*M_DEST_WIDTH +: M_DEST_WIDTH] = {M_DEST_WIDTH{1'b0}};
            end
        end
    endgenerate

    // Generate s_axis_tready
    // An input s is ready if it is granted by any output that it is requesting
    // And the output is ready
    generate
        for (s = 0; s < S_COUNT; s = s + 1) begin : gen_s_ready
            wire [M_COUNT-1:0] s_granted_by;
            
            genvar m;
            for (m = 0; m < M_COUNT; m = m + 1) begin : gen_s_granted
                assign s_granted_by[m] = grant[m][s] & m_tready_int[m];
            end
            
            assign s_tready_int[s] = |s_granted_by;
        end
    endgenerate

endmodule
