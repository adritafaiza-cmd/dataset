```verilog
module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   s_pause_req,
    output wire                   s_pause_ack,
    input  wire                   m_pause_req,
    output wire                   m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] s_status_depth,
    output wire [$clog2(DEPTH):0] s_status_depth_commit,
    output wire                   s_status_overflow,
    output wire                   s_status_bad_frame,
    output wire                   s_status_good_frame,
    output wire [$clog2(DEPTH):0] m_status_depth,
    output wire [$clog2(DEPTH):0] m_status_depth_commit,
    output wire                   m_status_overflow,
    output wire                   m_status_bad_frame,
    output wire                   m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Total data width for storage
    localparam TOTAL_DATA_WIDTH = DATA_WIDTH + 
        (KEEP_ENABLE ? KEEP_WIDTH : 0) +
        (LAST_ENABLE ? 1 : 0) +
        (ID_ENABLE ? ID_WIDTH : 0) +
        (DEST_ENABLE ? DEST_WIDTH : 0) +
        (USER_ENABLE ? USER_WIDTH : 0);

    // Internal signals
    reg [TOTAL_DATA_WIDTH-1:0] mem [0:DEPTH-1];
    
    // Write pointer (s_clk domain)
    reg [PTR_WIDTH-1:0] w_ptr;
    reg [ADDR_WIDTH-1:0] w_addr;
    
    // Read pointer (m_clk domain)
    reg [PTR_WIDTH-1:0] r_ptr;
    reg [ADDR_WIDTH-1:0] r_addr;
    
    // Synchronizers for write pointer to m_clk domain
    reg [PTR_WIDTH-1:0] w_ptr_sync1, w_ptr_sync2, w_ptr_sync3;
    
    // Synchronizers for read pointer to s_clk domain
    reg [PTR_WIDTH-1:0] r_ptr_sync1, r_ptr_sync2, r_ptr_sync3;
    
    // Depth calculations
    wire [PTR_WIDTH-1:0] w_ptr_m = w_ptr_sync3;
    wire [PTR_WIDTH-1:0] r_ptr_s = r_ptr_sync3;
    
    wire [PTR_WIDTH-1:0] depth_s = w_ptr - r_ptr_s;
    wire [PTR_WIDTH-1:0] depth_m = w_ptr_m - r_ptr;
    
    // Full and empty conditions
    wire s_full = (depth_s >= DEPTH);
    wire m_empty = (depth_m == 0);
    
    // Pause logic
    reg s_paused, m_paused;
    wire s_pause_active = PAUSE_ENABLE ? (s_pause_req || s_paused) : 1'b0;
    wire m_pause_active = PAUSE_ENABLE ? (m_pause_req || m_paused) : 1'b0;
    
    // Frame tracking
    reg s_in_frame;
    reg m_in_frame;
    
    // Drop/mark logic
    wire s_drop = (DROP_WHEN_FULL && s_full) || 
                  (DROP_OVERSIZE_FRAME && s_in_frame && s_full) ||
                  (DROP_BAD_FRAME && s_in_frame && s_full);
    wire s_mark = (MARK_WHEN_FULL && s_full) && !s_drop;
    
    // Status flags
    reg s_overflow, s_bad_frame, s_good_frame;
    reg m_overflow, m_bad_frame, m_good_frame;
    
    // Write enable
    wire s_write_en = s_axis_tvalid && s_axis_tready && !s_pause_active;
    
    // Read enable
    wire m_read_en = m_axis_tvalid && m_axis_tready && !m_pause_active;
    
    // Output valid
    assign m_axis_tvalid = !m_empty && !m_pause_active;
    
    // Input ready
    assign s_axis_tready = !s_full && !s_pause_active;
    
    // Pause ack
    assign s_pause_ack = s_paused;
    assign m_pause_ack = m_paused;
    
    // Status outputs
    assign s_status_depth = depth_s;
    assign s_status_depth_commit = depth_s;
    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;
    
    assign m_status_depth = depth_m;
    assign m_status_depth_commit = depth_m;
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;
    
    // Data packing
    wire [TOTAL_DATA_WIDTH-1:0] s_data;
    assign s_data = {
        (USER_ENABLE ? s_axis_tuser : {USER_WIDTH{1'b0}}),
        (DEST_ENABLE ? s_axis_tdest : {DEST_WIDTH{1'b0}}),
        (ID_ENABLE ? s_axis_tid : {ID_WIDTH{1'b0}}),
        (LAST_ENABLE ? s_axis_tlast : 1'b0),
        (KEEP_ENABLE ? s_axis_tkeep : {KEEP_WIDTH{1'b0}}),
        s_axis_tdata
    };
    
    // Data unpacking
    wire [TOTAL_DATA_WIDTH-1:0] m_data;
    assign m_axis_tdata = m_data[DATA_WIDTH-1:0];
    assign m_axis_tkeep = KEEP_ENABLE ? m_data[DATA_WIDTH +: KEEP_WIDTH] : {KEEP_WIDTH{1'b0}};
    assign m_axis_tlast = LAST_ENABLE ? m_data[DATA_WIDTH + KEEP_ENABLE*KEEP_WIDTH] : 1'b0;
    assign m_axis_tid = ID_ENABLE ? m_data[DATA_WIDTH + KEEP_ENABLE*KEEP_WIDTH + LAST_ENABLE +: ID_WIDTH] : {ID_WIDTH{1'b0}};
    assign m_axis_tdest = DEST_ENABLE ? m_data[DATA_WIDTH + KEEP_ENABLE*KEEP_WIDTH + LAST_ENABLE + ID_ENABLE*ID_WIDTH +: DEST_WIDTH] : {DEST_WIDTH{1'b0}};
    assign m_axis_tuser = USER_ENABLE ? m_data[DATA_WIDTH + KEEP_ENABLE*KEEP_WIDTH + LAST_ENABLE + ID_ENABLE*ID_WIDTH + DEST_ENABLE*DEST_WIDTH +: USER_WIDTH] : {USER_WIDTH{1'b0}};
    
    // Write pointer synchronization
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            w_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            w_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            w_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            w_ptr_sync1 <= w_ptr;
            w_ptr_sync2 <= w_ptr_sync1;
            w_ptr_sync3 <= w_ptr_sync2;
        end
    end
    
    // Read pointer synchronization
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            r_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            r_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            r_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            r_ptr_sync1 <= r_ptr;
            r_ptr_sync2 <= r_ptr_sync1;
            r_ptr_sync3 <= r_ptr_sync2;
        end
    end
    
    // Write logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            w_ptr <= {PTR_WIDTH{1'b0}};
            w_addr <= {ADDR_WIDTH{1'b0}};
            s_in_frame <= 1'b0;
            s_overflow <= 1'b0;
            s_bad_frame <= 1'b0;
            s_good_frame <= 1'b0;
            s_paused <= 1'b0;
        end else begin
            if (s_write_en) begin
                mem[w_addr] <= s_data;
                w_ptr <= w_ptr + 1'b1;
                w_addr <= w_addr + 1'b1;
                
                if (LAST_ENABLE && s_axis_tlast) begin
                    s_in_frame <= 1'b0;
                    s_good_frame <= 1'b1;
                end else begin
                    s_in_frame <= 1'b1;
                end
            end
            
            if (s_pause_req) begin
                s_paused <= 1'b1;
            end else if (s_paused && !s_pause_req) begin
                s_paused <= 1'b0;
            end
        end
    end
    
    // Read logic
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            r_ptr <= {PTR_WIDTH{1'b0}};
            r_addr <= {ADDR_WIDTH{1'b0}};
            m_in_frame <= 1'b0;
            m_overflow <= 1'b0;
            m_bad_frame <= 1'b0;
            m_good_frame <= 1'b0;
            m_paused <= 1'b0;
        end else begin
            if (m_read_en) begin
                m_data <= mem[r_addr];
                r_ptr <= r_ptr + 1'b1;
                r_addr <= r_addr + 1'b1;
                
                if (LAST_ENABLE && m_axis_tlast) begin
                    m_in_frame <= 1'b0;
                    m_good_frame <= 1'b1;
                end else begin
                    m_in_frame <= 1'b1;
                end
            end
            
            if (m_pause_req) begin
                m_paused <= 1'b1;
            end else if (m_paused && !m_pause_req) begin
                m_paused <= 1'b0;
            end
        end
    end

endmodule
