module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   s_pause_req,
    output wire                   s_pause_ack,
    input  wire                   m_pause_req,
    output wire                   m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] s_status_depth,
    output wire [$clog2(DEPTH):0] s_status_depth_commit,
    output wire                   s_status_overflow,
    output wire                   s_status_bad_frame,
    output wire                   s_status_good_frame,
    output wire [$clog2(DEPTH):0] m_status_depth,
    output wire [$clog2(DEPTH):0] m_status_depth_commit,
    output wire                   m_status_overflow,
    output wire                   m_status_bad_frame,
    output wire                   m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;

    // Total data width for storage
    localparam TOTAL_DATA_WIDTH = DATA_WIDTH + KEEP_WIDTH + 1 + ID_WIDTH + DEST_WIDTH + USER_WIDTH;

    // RAM storage
    reg [TOTAL_DATA_WIDTH-1:0] ram [0:DEPTH-1];

    // Write pointer (s_clk domain)
    reg [PTR_WIDTH-1:0] w_ptr;
    // Read pointer (m_clk domain)
    reg [PTR_WIDTH-1:0] r_ptr;

    // Synchronizers for read pointer to s_clk domain
    reg [PTR_WIDTH-1:0] r_ptr_sync1, r_ptr_sync2, r_ptr_sync3;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            r_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            r_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            r_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            r_ptr_sync1 <= r_ptr;
            r_ptr_sync2 <= r_ptr_sync1;
            r_ptr_sync3 <= r_ptr_sync2;
        end
    end

    // Synchronizers for write pointer to m_clk domain
    reg [PTR_WIDTH-1:0] w_ptr_sync1, w_ptr_sync2, w_ptr_sync3;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            w_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            w_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            w_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            w_ptr_sync1 <= w_ptr;
            w_ptr_sync2 <= w_ptr_sync1;
            w_ptr_sync3 <= w_ptr_sync2;
        end
    end

    // Depth calculations
    wire [PTR_WIDTH-1:0] s_depth = w_ptr - r_ptr_sync3;
    wire [PTR_WIDTH-1:0] m_depth = w_ptr_sync3 - r_ptr;

    // Full and empty conditions
    wire s_full = (s_depth == DEPTH);
    wire m_empty = (m_depth == 0);

    // Pause logic
    reg s_paused, m_paused;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) s_paused <= 1'b0;
        else if (PAUSE_ENABLE) s_paused <= s_pause_req;
        else s_paused <= 1'b0;
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) m_paused <= 1'b0;
        else if (PAUSE_ENABLE) m_paused <= m_pause_req;
        else m_paused <= 1'b0;
    end

    assign s_pause_ack = s_paused;
    assign m_pause_ack = m_paused;

    // Frame tracking in s_clk domain
    reg s_in_frame;
    reg s_frame_overflow;
    reg s_frame_bad;
    reg s_frame_good;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_in_frame <= 1'b0;
            s_frame_overflow <= 1'b0;
            s_frame_bad <= 1'b0;
            s_frame_good <= 1'b0;
        end else begin
            s_frame_overflow <= 1'b0;
            s_frame_bad <= 1'b0;
            s_frame_good <= 1'b0;

            if (s_axis_tvalid && s_axis_tready) begin
                if (s_in_frame) begin
                    if (s_axis_tlast) begin
                        s_in_frame <= 1'b0;
                        s_frame_good <= 1'b1;
                    end else if (DROP_OVERSIZE_FRAME && s_full) begin
                        s_frame_overflow <= 1'b1;
                        s_in_frame <= 1'b0;
                    end
                end else begin
                    if (s_axis_tlast) begin
                        s_frame_good <= 1'b1;
                    end else begin
                        s_in_frame <= 1'b1;
                    end
                end
            end
        end
    end

    // Write enable
    wire s_write_en = s_axis_tvalid && s_axis_tready && !s_paused;

    // Determine if we should drop or mark
    wire s_drop = (DROP_WHEN_FULL && s_full) || (DROP_OVERSIZE_FRAME && s_in_frame && s_full);
    wire s_mark = (MARK_WHEN_FULL && s_full) && !s_drop;

    // Actual write to RAM
    wire s_ram_write = s_write_en && !s_drop;

    // Write address
    wire [ADDR_WIDTH-1:0] w_addr = w_ptr[ADDR_WIDTH-1:0];

    // Write data
    wire [TOTAL_DATA_WIDTH-1:0] w_data;
    assign w_data[DATA_WIDTH-1:0] = s_axis_tdata;
    assign w_data[DATA_WIDTH +: KEEP_WIDTH] = s_axis_tkeep;
    assign w_data[DATA_WIDTH + KEEP_WIDTH] = s_axis_tlast;
    assign w_data[DATA_WIDTH + KEEP_WIDTH + 1 +: ID_WIDTH] = s_axis_tid;
    assign w_data[DATA_WIDTH + KEEP_WIDTH + 1 + ID_WIDTH +: DEST_WIDTH] = s_axis_tdest;
    assign w_data[DATA_WIDTH + KEEP_WIDTH + 1 + ID_WIDTH + DEST_WIDTH +: USER_WIDTH] = s_axis_tuser;

    // RAM write
    always @(posedge s_clk) begin
        if (s_ram_write) begin
            ram[w_addr] <= w_data;
        end
    end

    // Update write pointer
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) w_ptr <= {PTR_WIDTH{1'b0}};
        else if (s_ram_write) w_ptr <= w_ptr + 1'b1;
    end

    // tready logic
    wire s_tready_base = !s_full && !s_paused;
    assign s_axis_tready = s_tready_base;

    // Read side
    wire [ADDR_WIDTH-1:0] r_addr = r_ptr[ADDR_WIDTH-1:0];

    // RAM read
    reg [TOTAL_DATA_WIDTH-1:0] r_data;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) r_data <= {TOTAL_DATA_WIDTH{1'b0}};
        else if (!m_empty) r_data <= ram[r_addr];
    end

    // Output FIFO for pipeline
    reg [TOTAL_DATA_WIDTH-1:0] out_fifo [0:1];
    reg [1:0] out_w_ptr, out_r_ptr;
    reg out_w_valid, out_r_valid;

    // Read from RAM into output FIFO
    wire m_ram_read = !m_empty && !out_w_valid && !m_paused;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            out_w_valid <= 1'b0;
            out_r_valid <= 1'b0;
            out_w_ptr <= 2'b00;
            out_r_ptr <= 2'b00;
        end else begin
            if (m_ram_read) begin
                out_fifo[out_w_ptr[0]] <= r_data;
                out_w_ptr <= out_w_ptr + 1'b1;
                out_w_valid <= 1'b1;
            end
            if (m_axis_tvalid && m_axis_tready) begin
                out_r_ptr <= out_r_ptr + 1'b1;
                out_r_valid <= 1'b0;
            end
        end
    end

    // Output data
    wire [TOTAL_DATA_WIDTH-1:0] m_out_data = out_fifo[out_r_ptr[0]];

    assign m_axis_tdata = m_out_data[DATA_WIDTH-1:0];
    assign m_axis_tkeep = m_out_data[DATA_WIDTH +: KEEP_WIDTH];
    assign m_axis_tlast = m_out_data[DATA_WIDTH + KEEP_WIDTH];
    assign m_axis_tid = m_out_data[DATA_WIDTH + KEEP_WIDTH + 1 +: ID_WIDTH];
    assign m_axis_tdest = m_out_data[DATA_WIDTH + KEEP_WIDTH + 1 + ID_WIDTH +: DEST_WIDTH];
    assign m_axis_tuser = m_out_data[DATA_WIDTH + KEEP_WIDTH + 1 + ID_WIDTH + DEST_WIDTH +: USER_WIDTH];

    assign m_axis_tvalid = out_r_valid && !m_paused;

    // Update read pointer
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) r_ptr <= {PTR_WIDTH{1'b0}};
        else if (m_axis_tvalid && m_axis_tready) r_ptr <= r_ptr + 1'b1;
    end

    // Status outputs
    assign s_status_depth = s_depth;
    assign s_status_depth_commit = s_depth;
    assign s_status_overflow = s_frame_overflow;
    assign s_status_bad_frame = s_frame_bad;
    assign s_status_good_frame = s_frame_good;

    assign m_status_depth = m_depth;
    assign m_status_depth_commit = m_depth;
    assign m_status_overflow = 1'b0;
    assign m_status_bad_frame = 1'b0;
    assign m_status_good_frame = 1'b0;

endmodule
