```verilog
module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   s_pause_req,
    output wire                   s_pause_ack,
    input  wire                   m_pause_req,
    output wire                   m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] s_status_depth,
    output wire [$clog2(DEPTH):0] s_status_depth_commit,
    output wire                   s_status_overflow,
    output wire                   s_status_bad_frame,
    output wire                   s_status_good_frame,
    output wire [$clog2(DEPTH):0] m_status_depth,
    output wire [$clog2(DEPTH):0] m_status_depth_commit,
    output wire                   m_status_overflow,
    output wire                   m_status_bad_frame,
    output wire                   m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Total width of stored data
    localparam TOTAL_DATA_WIDTH = DATA_WIDTH + 
                                  (KEEP_ENABLE ? KEEP_WIDTH : 0) +
                                  (LAST_ENABLE ? 1 : 0) +
                                  (ID_ENABLE ? ID_WIDTH : 0) +
                                  (DEST_ENABLE ? DEST_WIDTH : 0) +
                                  (USER_ENABLE ? USER_WIDTH : 0);

    // Internal signals
    reg [TOTAL_DATA_WIDTH-1:0] mem [0:DEPTH-1];
    
    // Write pointer (s_clk domain)
    reg [PTR_WIDTH-1:0] w_ptr;
    reg [ADDR_WIDTH-1:0] w_addr;
    
    // Read pointer (m_clk domain)
    reg [PTR_WIDTH-1:1] r_ptr;
    reg [ADDR_WIDTH-1:0] r_addr;
    
    // CDC synchronizers for write pointer to m_clk domain
    reg [PTR_WIDTH-1:0] w_ptr_sync1, w_ptr_sync2, w_ptr_sync3;
    
    // CDC synchronizers for read pointer to s_clk domain
    reg [PTR_WIDTH-1:0] r_ptr_sync1, r_ptr_sync2, r_ptr_sync3;
    
    // CDC synchronizers for pause signals
    reg s_pause_req_sync1, s_pause_req_sync2, s_pause_req_sync3;
    reg m_pause_req_sync1, m_pause_req_sync2, m_pause_req_sync3;
    
    // Pause state
    reg s_paused, m_paused;
    
    // Status flags
    reg s_overflow, m_overflow;
    reg s_bad_frame, m_bad_frame;
    reg s_good_frame, m_good_frame;
    
    // Frame tracking
    reg s_in_frame, m_in_frame;
    reg [USER_WIDTH-1:0] s_frame_user, m_frame_user;
    
    // Depth calculation
    wire [PTR_WIDTH-1:0] w_ptr_m = w_ptr_sync3;
    wire [PTR_WIDTH-1:0] r_ptr_s = r_ptr_sync3;
    
    wire [PTR_WIDTH-1:0] depth_s = w_ptr - r_ptr_s;
    wire [PTR_WIDTH-1:0] depth_m = w_ptr_m - r_ptr;
    
    wire s_full = (depth_s == DEPTH);
    wire m_empty = (depth_m == 0);
    
    // Write enable
    wire s_write_en = s_axis_tvalid && s_axis_tready && !s_paused;
    
    // Read enable
    wire m_read_en = m_axis_tvalid && m_axis_tready && !m_paused;
    
    // Tready generation
    wire s_tready_base = !s_full || DROP_WHEN_FULL;
    assign s_axis_tready = s_tready_base && !s_paused;
    
    // Tvalid generation
    assign m_axis_tvalid = !m_empty && !m_paused;
    
    // Data output
    wire [TOTAL_DATA_WIDTH-1:0] r_data = mem[r_addr];
    
    assign m_axis_tdata = r_data[DATA_WIDTH-1:0];
    assign m_axis_tkeep = KEEP_ENABLE ? r_data[DATA_WIDTH +: KEEP_WIDTH] : {KEEP_WIDTH{1'b0}};
    assign m_axis_tlast = LAST_ENABLE ? r_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0)] : 1'b0;
    assign m_axis_tid = ID_ENABLE ? r_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) +: ID_WIDTH] : {ID_WIDTH{1'b0}};
    assign m_axis_tdest = DEST_ENABLE ? r_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0) +: DEST_WIDTH] : {DEST_WIDTH{1'b0}};
    assign m_axis_tuser = USER_ENABLE ? r_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0) + (DEST_ENABLE ? DEST_WIDTH : 0) +: USER_WIDTH] : {USER_WIDTH{1'b0}};
    
    // Write data packing
    wire [TOTAL_DATA_WIDTH-1:0] w_data;
    assign w_data[DATA_WIDTH-1:0] = s_axis_tdata;
    assign w_data[DATA_WIDTH +: KEEP_WIDTH] = KEEP_ENABLE ? s_axis_tkeep : {KEEP_WIDTH{1'b0}};
    assign w_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0)] = LAST_ENABLE ? s_axis_tlast : 1'b0;
    assign w_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) +: ID_WIDTH] = ID_ENABLE ? s_axis_tid : {ID_WIDTH{1'b0}};
    assign w_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0) +: DEST_WIDTH] = DEST_ENABLE ? s_axis_tdest : {DEST_WIDTH{1'b0}};
    assign w_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0) + (DEST_ENABLE ? DEST_WIDTH : 0) +: USER_WIDTH] = USER_ENABLE ? s_axis_tuser : {USER_WIDTH{1'b0}};
    
    // Write pointer update
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            w_ptr <= {PTR_WIDTH{1'b0}};
            w_addr <= {ADDR_WIDTH{1'b0}};
            s_overflow <= 1'b0;
            s_bad_frame <= 1'b0;
            s_good_frame <= 1'b0;
            s_in_frame <= 1'b0;
            s_frame_user <= {USER_WIDTH{1'b0}};
            s_paused <= 1'b0;
        end else begin
            if (s_write_en) begin
                mem[w_addr] <= w_data;
                w_ptr <= w_ptr + 1'b1;
                w_addr <= w_addr + 1'b1;
            end
            
            // Overflow detection
            if (s_axis_tvalid && s_full && !DROP_WHEN_FULL) begin
                s_overflow <= 1'b1;
            end
            
            // Frame tracking
            if (s_write_en) begin
                if (!s_in_frame) begin
                    s_in_frame <= 1'b1;
                    s_frame_user <= USER_ENABLE ? s_axis_tuser : {USER_WIDTH{1'b0}};
                end
                if (LAST_ENABLE && s_axis_tlast) begin
                    s_in_frame <= 1'b0;
                    // Check for bad frame
                    if (USER_ENABLE && (s_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                        s_bad_frame <= 1'b1;
                    end else begin
                        s_good_frame <= 1'b1;
                    end
                end
            end
            
            // Pause handling
            if (PAUSE_ENABLE) begin
                if (s_pause_req_sync3) begin
                    s_paused <= 1'b1;
                end else if (!s_pause_req_sync3) begin
                    s_paused <= 1'b0;
                end
            end
        end
    end
    
    // Read pointer update
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            r_ptr <= {PTR_WIDTH{1'b0}};
            r_addr <= {ADDR_WIDTH{1'b0}};
            m_overflow <= 1'b0;
            m_bad_frame <= 1'b0;
            m_good_frame <= 1'b0;
            m_in_frame <= 1'b0;
            m_frame_user <= {USER_WIDTH{1'b0}};
            m_paused <= 1'b0;
        end else begin
            if (m_read_en) begin
                r_ptr <= r_ptr + 1'b1;
                r_addr <= r_addr + 1'b1;
            end
            
            // Frame tracking on read side
            if (m_read_en) begin
                if (!m_in_frame) begin
                    m_in_frame <= 1'b1;
                    m_frame_user <= USER_ENABLE ? m_axis_tuser : {USER_WIDTH{1'b0}};
                end
                if (LAST_ENABLE && m_axis_tlast) begin
                    m_in_frame <= 1'b0;
                    if (USER_ENABLE && (m_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                        m_bad_frame <= 1'b1;
                    end else begin
                        m_good_frame <= 1'b1;
                    end
                end
            end
            
            // Pause handling
            if (PAUSE_ENABLE) begin
                if (m_pause_req_sync3) begin
                    m_paused <= 1'b1;
                end else if (!m_pause_req_sync3) begin
                    m_paused <= 1'b0;
                end
            end
        end
    end
    
    // CDC synchronizers for write pointer
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            w_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            w_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            w_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            w_ptr_sync1 <= w_ptr;
            w_ptr_sync2 <= w_ptr_sync1;
            w_ptr_sync3 <= w_ptr_sync2;
        end
    end
    
    // CDC synchronizers for read pointer
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            r_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            r_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            r_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            r_ptr_sync1 <= r_ptr;
            r_ptr_sync2 <= r_ptr_sync1;
            r_ptr_sync3 <= r_ptr_sync2;
        end
    end
    
    // CDC synchronizers for pause signals
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_pause_req_sync1 <= 1'b0;
            s_pause_req_sync2 <= 1'b0;
            s_pause_req_sync3 <= 1'b0;
        end else begin
            s_pause_req_sync1 <= s_pause_req;
            s_pause_req_sync2 <= s_pause_req_sync1;
            s_pause_req_sync3 <= s_pause_req_sync2;
        end
    end
    
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_pause_req_sync1 <= 1'b0;
            m_pause_req_sync2 <= 1'b0;
            m_pause_req_sync3 <= 1'b0;
        end else begin
            m_pause_req_sync1 <= m_pause_req;
            m_pause_req_sync2 <= m_pause_req_sync1;
            m_pause_req_sync3 <= m_pause_req_sync2;
        end
    end
    
    // Pause ack outputs
    assign s_pause_ack = PAUSE_ENABLE ? s_paused : 1'b0;
    assign m_pause_ack = PAUSE_ENABLE ? m_paused : 1'b0;
    
    // Status outputs
    assign s_status_depth = depth_s;
    assign s_status_depth_commit = depth_s;
    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;
    
    assign m_status_depth = depth_m;
    assign m_status_depth_commit = depth_m;
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;

endmodule
