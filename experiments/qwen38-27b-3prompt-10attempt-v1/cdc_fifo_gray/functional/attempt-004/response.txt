module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Source domain
    reg [PTR_W-1:0] src_wr_ptr;
    reg [PTR_W-1:0] src_rd_ptr_synced;
    reg [WIDTH-1:0] mem [0:DEPTH-1];

    wire src_wr_en = src_valid_i & src_ready_o;
    wire src_full = (src_wr_ptr[LOG_DEPTH] != src_rd_ptr_synced[LOG_DEPTH]) &&
                    (src_wr_ptr[LOG_DEPTH-1:0] == src_rd_ptr_synced[LOG_DEPTH-1:0]);
    assign src_ready_o = ~src_full;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_wr_ptr <= {PTR_W{1'b0}};
        else if (src_wr_en)
            src_wr_ptr <= src_wr_ptr + 1'b1;
    end

    always @(posedge src_clk_i) begin
        if (src_wr_en)
            mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
    end

    // Destination domain
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [PTR_W-1:0] dst_wr_ptr_synced;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;

    wire dst_rd_en = dst_valid_o & dst_ready_i;
    wire dst_empty = (dst_rd_ptr == dst_wr_ptr_synced);
    assign dst_valid_o = dst_valid_reg;
    assign dst_data_o = dst_data_reg;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_valid_reg <= 1'b0;
            dst_data_reg <= {WIDTH{1'b0}};
        end else begin
            if (dst_rd_en) begin
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_valid_reg <= ~dst_empty;
                if (~dst_empty)
                    dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
            end else if (dst_valid_reg && dst_empty) begin
                // This case shouldn't happen if logic is correct, but handle it
                dst_valid_reg <= 1'b0;
            end
        end
    end

    // Synchronizers
    // Source write pointer to destination domain
    reg [PTR_W-1:0] src_wr_ptr_sync [0:SYNC_STAGES-1];
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_sync[0] <= {PTR_W{1'b0}};
            src_wr_ptr_sync[1] <= {PTR_W{1'b0}};
            if (SYNC_STAGES > 2) begin
                integer i;
                for (i = 2; i < SYNC_STAGES; i = i + 1)
                    src_wr_ptr_sync[i] <= {PTR_W{1'b0}};
            end
        end else begin
            src_wr_ptr_sync[0] <= src_wr_ptr;
            src_wr_ptr_sync[1] <= src_wr_ptr_sync[0];
            if (SYNC_STAGES > 2) begin
                integer i;
                for (i = 2; i < SYNC_STAGES; i = i + 1)
                    src_wr_ptr_sync[i] <= src_wr_ptr_sync[i-1];
            end
        end
    end
    assign dst_wr_ptr_synced = src_wr_ptr_sync[SYNC_STAGES-1];

    // Destination read pointer to source domain
    reg [PTR_W-1:0] dst_rd_ptr_sync [0:SYNC_STAGES-1];
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_sync[0] <= {PTR_W{1'b0}};
            dst_rd_ptr_sync[1] <= {PTR_W{1'b0}};
            if (SYNC_STAGES > 2) begin
                integer i;
                for (i = 2; i < SYNC_STAGES; i = i + 1)
                    dst_rd_ptr_sync[i] <= {PTR_W{1'b0}};
            end
        end else begin
            dst_rd_ptr_sync[0] <= dst_rd_ptr;
            dst_rd_ptr_sync[1] <= dst_rd_ptr_sync[0];
            if (SYNC_STAGES > 2) begin
                integer i;
                for (i = 2; i < SYNC_STAGES; i = i + 1)
                    dst_rd_ptr_sync[i] <= dst_rd_ptr_sync[i-1];
            end
        end
    end
    assign src_rd_ptr_synced = dst_rd_ptr_sync[SYNC_STAGES-1];

endmodule
