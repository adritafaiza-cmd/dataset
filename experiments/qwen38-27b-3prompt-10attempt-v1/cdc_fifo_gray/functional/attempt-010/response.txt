module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Source domain
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_W-1:0] src_wr_ptr;
    reg [PTR_W-1:0] dst_rd_ptr_synced;

    // Destination domain
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [PTR_W-1:0] src_wr_ptr_synced;

    // Synchronizer registers for write pointer (src -> dst)
    reg [PTR_W-1:0] wr_ptr_sync [0:SYNC_STAGES-1];

    // Synchronizer registers for read pointer (dst -> src)
    reg [PTR_W-1:0] rd_ptr_sync [0:SYNC_STAGES-1];

    // Full/Empty flags
    wire src_full;
    wire dst_empty;

    // Write enable
    wire src_wr_en;

    // Read enable
    wire dst_rd_en;

    // Source domain logic
    assign src_wr_en = src_valid_i & src_ready_o;
    assign src_ready_o = ~src_full;

    // Destination domain logic
    assign dst_rd_en = dst_valid_o & dst_ready_i;
    assign dst_valid_o = ~dst_empty;

    // Full condition: write pointer has lapped read pointer by DEPTH
    assign src_full = (src_wr_ptr[PTR_W-1] != dst_rd_ptr_synced[PTR_W-1]) &&
                      (src_wr_ptr[LOG_DEPTH-1:0] == dst_rd_ptr_synced[LOG_DEPTH-1:0]);

    // Empty condition: read pointer equals write pointer
    assign dst_empty = (dst_rd_ptr == src_wr_ptr_synced);

    // Write to memory in source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
        end else begin
            if (src_wr_en) begin
                mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
                src_wr_ptr <= src_wr_ptr + 1'b1;
            end
        end
    end

    // Read from memory in destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_data_o <= {WIDTH{1'b0}};
        end else begin
            if (dst_rd_en) begin
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
            end
            if (!dst_empty) begin
                dst_data_o <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
            end
        end
    end

    // Synchronize write pointer from src to dst
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            wr_ptr_sync[0] <= {PTR_W{1'b0}};
            wr_ptr_sync[1] <= {PTR_W{1'b0}};
            if (SYNC_STAGES > 2) begin
                wr_ptr_sync[2] <= {PTR_W{1'b0}};
            end
            if (SYNC_STAGES > 3) begin
                wr_ptr_sync[3] <= {PTR_W{1'b0}};
            end
        end else begin
            wr_ptr_sync[0] <= src_wr_ptr;
            wr_ptr_sync[1] <= wr_ptr_sync[0];
            if (SYNC_STAGES > 2) begin
                wr_ptr_sync[2] <= wr_ptr_sync[1];
            end
            if (SYNC_STAGES > 3) begin
                wr_ptr_sync[3] <= wr_ptr_sync[2];
            end
        end
    end

    assign src_wr_ptr_synced = (SYNC_STAGES == 2) ? wr_ptr_sync[1] :
                               (SYNC_STAGES == 3) ? wr_ptr_sync[2] :
                               wr_ptr_sync[3];

    // Synchronize read pointer from dst to src
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            rd_ptr_sync[0] <= {PTR_W{1'b0}};
            rd_ptr_sync[1] <= {PTR_W{1'b0}};
            if (SYNC_STAGES > 2) begin
                rd_ptr_sync[2] <= {PTR_W{1'b0}};
            end
            if (SYNC_STAGES > 3) begin
                rd_ptr_sync[3] <= {PTR_W{1'b0}};
            end
        end else begin
            rd_ptr_sync[0] <= dst_rd_ptr;
            rd_ptr_sync[1] <= rd_ptr_sync[0];
            if (SYNC_STAGES > 2) begin
                rd_ptr_sync[2] <= rd_ptr_sync[1];
            end
            if (SYNC_STAGES > 3) begin
                rd_ptr_sync[3] <= rd_ptr_sync[2];
            end
        end
    end

    assign dst_rd_ptr_synced = (SYNC_STAGES == 2) ? rd_ptr_sync[1] :
                               (SYNC_STAGES == 3) ? rd_ptr_sync[2] :
                               rd_ptr_sync[3];

endmodule
