module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Source domain
    reg  [PTR_W-1:0] src_wr_ptr;
    reg  [PTR_W-1:0] src_rd_ptr_sync;
    reg  [WIDTH-1:0] mem [0:DEPTH-1];

    wire src_wr_en = src_valid_i & src_ready_o;
    wire src_full = (src_wr_ptr[PTR_W-1] != src_rd_ptr_sync[PTR_W-1]) &&
                    (src_wr_ptr[LOG_DEPTH-1:0] == src_rd_ptr_sync[LOG_DEPTH-1:0]);
    assign src_ready_o = ~src_full;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_wr_ptr <= {PTR_W{1'b0}};
        else if (src_wr_en)
            src_wr_ptr <= src_wr_ptr + 1'b1;
    end

    always @(posedge src_clk_i) begin
        if (src_wr_en)
            mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
    end

    // Destination domain
    reg  [PTR_W-1:0] dst_rd_ptr;
    reg  [PTR_W-1:0] dst_wr_ptr_sync;
    reg  [WIDTH-1:0] dst_data_reg;
    reg              dst_valid_reg;

    wire dst_rd_en = dst_valid_reg & dst_ready_i;
    wire dst_empty = (dst_rd_ptr == dst_wr_ptr_sync);
    assign dst_valid_o = dst_valid_reg;
    assign dst_data_o = dst_data_reg;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_rd_en) begin
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_valid_reg <= ~dst_empty;
                if (~dst_empty)
                    dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
            end else if (~dst_empty && ~dst_valid_reg) begin
                dst_valid_reg <= 1'b1;
                dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
            end
        end
    end

    // Synchronizers
    reg [SYNC_STAGES-1:0] src_wr_ptr_sync;
    reg [SYNC_STAGES-1:0] dst_rd_ptr_sync;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            src_wr_ptr_sync <= {SYNC_STAGES{1'b0}};
        else
            src_wr_ptr_sync <= {src_wr_ptr_sync[SYNC_STAGES-2:0], src_wr_ptr[PTR_W-1]};
    end

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            dst_rd_ptr_sync <= {SYNC_STAGES{1'b0}};
        else
            dst_rd_ptr_sync <= {dst_rd_ptr_sync[SYNC_STAGES-2:0], dst_rd_ptr[PTR_W-1]};
    end

    // Reconstruct full pointers from synchronized MSB and local LSB
    wire [PTR_W-1:0] src_rd_ptr_sync_full = {src_wr_ptr_sync[SYNC_STAGES-1], src_rd_ptr_sync[LOG_DEPTH-1:0]};
    wire [PTR_W-1:0] dst_wr_ptr_sync_full = {dst_rd_ptr_sync[SYNC_STAGES-1], dst_wr_ptr_sync[LOG_DEPTH-1:0]};

    // Assign the reconstructed pointers to the signals used in full/empty checks
    // We need to use the synchronized MSB with the local LSB
    // For src domain: src_rd_ptr_sync should be {synced MSB from dst, local LSB}
    // But we only synced the MSB. We need to reconstruct properly.
    
    // Actually, let's use a different approach: sync the entire pointer but only the MSB changes slowly.
    // Standard approach: sync the full pointer, but the MSB is the one that matters for full/empty.
    // Let's sync the full pointer with SYNC_STAGES.

    // Redo: sync full pointers
    reg [SYNC_STAGES-1:0][PTR_W-1:0] src_wr_ptr_sync_full;
    reg [SYNC_STAGES-1:0][PTR_W-1:0] dst_rd_ptr_sync_full;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            src_wr_ptr_sync_full <= {SYNC_STAGES{1'b0}};
        else
            src_wr_ptr_sync_full <= {src_wr_ptr_sync_full[SYNC_STAGES-2:0], src_wr_ptr};
    end

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            dst_rd_ptr_sync_full <= {SYNC_STAGES{1'b0}};
        else
            dst_rd_ptr_sync_full <= {dst_rd_ptr_sync_full[SYNC_STAGES-2:0], dst_rd_ptr};
    end

    // Use the last stage of the synchronizer
    wire [PTR_W-1:0] src_rd_ptr_sync_full = src_wr_ptr_sync_full[SYNC_STAGES-1];
    wire [PTR_W-1:0] dst_wr_ptr_sync_full = dst_rd_ptr_sync_full[SYNC_STAGES-1];

    // Override the earlier assignments
    // The earlier src_rd_ptr_sync and dst_wr_ptr_sync are not used anymore
    // Let's clean up and use only the full pointer sync

endmodule
