module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Source domain
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_W-1:0] src_wr_ptr;
    reg [PTR_W-1:0] dst_rd_ptr_src_sync;

    // Destination domain
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [PTR_W-1:0] src_wr_ptr_dst_sync;

    // Synchronizer registers
    reg [PTR_W-1:0] src_wr_ptr_sync [0:SYNC_STAGES-1];
    reg [PTR_W-1:0] dst_rd_ptr_sync [0:SYNC_STAGES-1];

    // Source domain logic
    wire src_wr_en = src_valid_i & src_ready_o;
    wire [PTR_W-1:0] src_wr_ptr_next = src_wr_ptr + 1'b1;

    // Full condition: write pointer catches up to read pointer
    wire src_full = (src_wr_ptr[PTR_W-1] != dst_rd_ptr_src_sync[PTR_W-1]) &&
                    (src_wr_ptr[PTR_W-2:0] == dst_rd_ptr_src_sync[PTR_W-2:0]);

    assign src_ready_o = ~src_full;

    // Write to memory
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
        end else if (src_wr_en) begin
            mem[src_wr_ptr[PTR_W-2:0]] <= src_data_i;
            src_wr_ptr <= src_wr_ptr_next;
        end
    end

    // Synchronize write pointer to destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_sync[0] <= {PTR_W{1'b0}};
            src_wr_ptr_sync[1] <= {PTR_W{1'b0}};
            if (SYNC_STAGES > 2) begin
                integer i;
                for (i = 2; i < SYNC_STAGES; i = i + 1)
                    src_wr_ptr_sync[i] <= {PTR_W{1'b0}};
            end
        end else begin
            src_wr_ptr_sync[0] <= src_wr_ptr;
            src_wr_ptr_sync[1] <= src_wr_ptr_sync[0];
            if (SYNC_STAGES > 2) begin
                integer i;
                for (i = 2; i < SYNC_STAGES; i = i + 1)
                    src_wr_ptr_sync[i] <= src_wr_ptr_sync[i-1];
            end
        end
    end

    assign src_wr_ptr_dst_sync = src_wr_ptr_sync[SYNC_STAGES-1];

    // Destination domain logic
    wire dst_rd_en = dst_valid_o & dst_ready_i;
    wire [PTR_W-1:0] dst_rd_ptr_next = dst_rd_ptr + 1'b1;

    // Empty condition: read pointer catches up to write pointer
    wire dst_empty = (dst_rd_ptr[PTR_W-1] != src_wr_ptr_dst_sync[PTR_W-1]) &&
                     (dst_rd_ptr[PTR_W-2:0] == src_wr_ptr_dst_sync[PTR_W-2:0]);

    assign dst_valid_o = ~dst_empty;
    assign dst_data_o = mem[dst_rd_ptr[PTR_W-2:0]];

    // Read from memory
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
        end else if (dst_rd_en) begin
            dst_rd_ptr <= dst_rd_ptr_next;
        end
    end

    // Synchronize read pointer to source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_sync[0] <= {PTR_W{1'b0}};
            dst_rd_ptr_sync[1] <= {PTR_W{1'b0}};
            if (SYNC_STAGES > 2) begin
                integer i;
                for (i = 2; i < SYNC_STAGES; i = i + 1)
                    dst_rd_ptr_sync[i] <= {PTR_W{1'b0}};
            end
        end else begin
            dst_rd_ptr_sync[0] <= dst_rd_ptr;
            dst_rd_ptr_sync[1] <= dst_rd_ptr_sync[0];
            if (SYNC_STAGES > 2) begin
                integer i;
                for (i = 2; i < SYNC_STAGES; i = i + 1)
                    dst_rd_ptr_sync[i] <= dst_rd_ptr_sync[i-1];
            end
        end
    end

    assign dst_rd_ptr_src_sync = dst_rd_ptr_sync[SYNC_STAGES-1];

endmodule
