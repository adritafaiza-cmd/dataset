module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;

    // Storage
    reg [WIDTH-1:0] mem [0:DEPTH-1];

    // Source side pointers
    reg [PTR_WIDTH-1:0] src_wr_ptr;
    reg [PTR_WIDTH-1:0] src_rd_ptr_synced;
    reg [PTR_WIDTH-1:0] src_rd_ptr_synced_next;

    // Destination side pointers
    reg [PTR_WIDTH-1:0] dst_rd_ptr;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_synced;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_synced_next;

    // Gray code conversion functions
    function [PTR_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_WIDTH-1:0] gray2bin;
        input [PTR_WIDTH-1:0] gray;
        reg [PTR_WIDTH-1:0] bin;
        integer i;
        begin
            bin[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1)
                bin[i] = bin[i+1] ^ gray[i];
            gray2bin = bin;
        end
    endfunction

    // Source side: write pointer and gray code
    wire [PTR_WIDTH-1:0] src_wr_ptr_gray = bin2gray(src_wr_ptr);

    // Destination side: read pointer and gray code
    wire [PTR_WIDTH-1:0] dst_rd_ptr_gray = bin2gray(dst_rd_ptr);

    // Synchronizers: source read pointer (from dst domain)
    reg [PTR_WIDTH-1:0] src_rd_ptr_sync [0:SYNC_STAGES-1];
    integer i;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            for (i = 0; i < SYNC_STAGES; i = i + 1)
                src_rd_ptr_sync[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            src_rd_ptr_sync[0] <= dst_rd_ptr_gray;
            for (i = 1; i < SYNC_STAGES; i = i + 1)
                src_rd_ptr_sync[i] <= src_rd_ptr_sync[i-1];
        end
    end

    assign src_rd_ptr_synced = gray2bin(src_rd_ptr_sync[SYNC_STAGES-1]);

    // Synchronizers: destination write pointer (from src domain)
    reg [PTR_WIDTH-1:0] dst_wr_ptr_sync [0:SYNC_STAGES-1];
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            for (i = 0; i < SYNC_STAGES; i = i + 1)
                dst_wr_ptr_sync[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            dst_wr_ptr_sync[0] <= src_wr_ptr_gray;
            for (i = 1; i < SYNC_STAGES; i = i + 1)
                dst_wr_ptr_sync[i] <= dst_wr_ptr_sync[i-1];
        end
    end

    assign dst_wr_ptr_synced = gray2bin(dst_wr_ptr_sync[SYNC_STAGES-1]);

    // Source side: full check
    wire src_full = (src_wr_ptr[PTR_WIDTH-1:1] == src_rd_ptr_synced[PTR_WIDTH-1:1]) &&
                    (src_wr_ptr[0] != src_rd_ptr_synced[0]);

    assign src_ready_o = !src_full;

    // Source side: write operation
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
        end else if (src_valid_i && src_ready_o) begin
            mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
            src_wr_ptr <= src_wr_ptr + 1'b1;
        end
    end

    // Destination side: empty check
    wire dst_empty = (dst_rd_ptr == dst_wr_ptr_synced);

    assign dst_valid_o = !dst_empty;

    // Destination side: read operation
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
        end else if (dst_valid_o && dst_ready_i) begin
            dst_rd_ptr <= dst_rd_ptr + 1'b1;
        end
    end

    // Output data
    assign dst_data_o = mem[dst_rd_ptr[LOG_DEPTH-1:0]];

endmodule
