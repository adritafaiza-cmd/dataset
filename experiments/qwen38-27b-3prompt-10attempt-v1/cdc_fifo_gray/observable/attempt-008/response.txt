module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;

    // Source side
    reg  [WIDTH-1:0] mem [0:DEPTH-1];
    reg  [PTR_WIDTH-1:0] src_wr_ptr;
    reg  [PTR_WIDTH-1:0] src_rd_ptr_gray_sync;
    reg  [PTR_WIDTH-1:0] src_rd_ptr_gray_sync_next;
    reg  [PTR_WIDTH-1:0] src_rd_ptr_gray_sync_reg [0:SYNC_STAGES-1];

    // Destination side
    reg  [PTR_WIDTH-1:0] dst_rd_ptr;
    reg  [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync;
    reg  [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync_next;
    reg  [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync_reg [0:SYNC_STAGES-1];

    // Gray code conversion functions
    function [PTR_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_WIDTH-1:0] gray2bin;
        input [PTR_WIDTH-1:0] gray;
        reg [PTR_WIDTH-1:0] bin;
        integer i;
        begin
            bin[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1)
                bin[i] = gray[i] ^ bin[i+1];
            gray2bin = bin;
        end
    endfunction

    // Source side: write pointer and memory write
    wire src_wr_ptr_gray = bin2gray(src_wr_ptr);
    wire src_full = (src_wr_ptr_gray == src_rd_ptr_gray_sync) &&
                    (src_wr_ptr[PTR_WIDTH-1] != src_rd_ptr_gray_sync[PTR_WIDTH-1]);
    wire src_can_write = src_valid_i && !src_full;

    assign src_ready_o = !src_full;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
        else if (src_can_write) begin
            mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
            src_wr_ptr <= src_wr_ptr + 1'b1;
        end
    end

    // Source side: synchronize read pointer from destination
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rd_ptr_gray_sync_reg[0] <= {PTR_WIDTH{1'b0}};
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                src_rd_ptr_gray_sync_reg[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            src_rd_ptr_gray_sync_reg[0] <= dst_rd_ptr_gray_sync_next;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                src_rd_ptr_gray_sync_reg[i] <= src_rd_ptr_gray_sync_reg[i-1];
        end
    end

    assign src_rd_ptr_gray_sync = src_rd_ptr_gray_sync_reg[SYNC_STAGES-1];

    // Destination side: read pointer and output
    wire dst_rd_ptr_gray = bin2gray(dst_rd_ptr);
    wire dst_empty = (dst_rd_ptr_gray == dst_wr_ptr_gray_sync);
    wire dst_can_read = dst_ready_i && !dst_empty;

    assign dst_valid_o = !dst_empty;
    assign dst_data_o = mem[dst_rd_ptr[LOG_DEPTH-1:0]];

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
        else if (dst_can_read)
            dst_rd_ptr <= dst_rd_ptr + 1'b1;
    end

    // Destination side: synchronize write pointer from source
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync_reg[0] <= {PTR_WIDTH{1'b0}};
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                dst_wr_ptr_gray_sync_reg[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync_reg[0] <= src_wr_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                dst_wr_ptr_gray_sync_reg[i] <= dst_wr_ptr_gray_sync_reg[i-1];
        end
    end

    assign dst_wr_ptr_gray_sync = dst_wr_ptr_gray_sync_reg[SYNC_STAGES-1];

    // Assign the next gray values for synchronization
    assign dst_rd_ptr_gray_sync_next = dst_rd_ptr_gray;
    assign src_wr_ptr_gray_sync_next = src_wr_ptr_gray;

endmodule
