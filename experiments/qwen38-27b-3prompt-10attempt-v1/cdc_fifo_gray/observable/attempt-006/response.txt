```verilog
module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Storage
    reg [WIDTH-1:0] mem [0:DEPTH-1];

    // Source side
    reg [PTR_W-1:0] src_wr_ptr;
    reg [PTR_W-1:0] src_rd_ptr_gray_sync;
    wire [PTR_W-1:0] src_rd_ptr_gray;
    wire [PTR_W-1:0] src_rd_ptr_bin;
    wire src_full;
    wire src_can_write;

    // Destination side
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [PTR_W-1:0] dst_wr_ptr_gray_sync;
    wire [PTR_W-1:0] dst_wr_ptr_gray;
    wire [PTR_W-1:0] dst_wr_ptr_bin;
    wire dst_empty;
    wire dst_can_read;

    // Gray code conversion functions
    function [PTR_W-1:0] bin2gray;
        input [PTR_W-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_W-1:0] gray2bin;
        input [PTR_W-1:0] gray;
        integer i;
        begin
            gray2bin[PTR_W-1] = gray[PTR_W-1];
            for (i = PTR_W-2; i >= 0; i = i - 1)
                gray2bin[i] = gray[i] ^ gray2bin[i+1];
        end
    endfunction

    // Source side: write pointer
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_wr_ptr <= {PTR_W{1'b0}};
        else if (src_valid_i && src_ready_o)
            src_wr_ptr <= src_wr_ptr + 1'b1;
    end

    // Source side: gray code of write pointer
    assign dst_wr_ptr_gray = bin2gray(src_wr_ptr);

    // Destination side: read pointer
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_rd_ptr <= {PTR_W{1'b0}};
        else if (dst_valid_o && dst_ready_i)
            dst_rd_ptr <= dst_rd_ptr + 1'b1;
    end

    // Destination side: gray code of read pointer
    assign src_rd_ptr_gray = bin2gray(dst_rd_ptr);

    // Synchronize read pointer gray from dst to src
    reg [SYNC_STAGES-1:0][PTR_W-1:0] src_rd_ptr_gray_sync_reg;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rd_ptr_gray_sync_reg[0] <= {PTR_W{1'b0}};
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                src_rd_ptr_gray_sync_reg[i] <= {PTR_W{1'b0}};
        end
        else begin
            src_rd_ptr_gray_sync_reg[0] <= src_rd_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                src_rd_ptr_gray_sync_reg[i] <= src_rd_ptr_gray_sync_reg[i-1];
        end
    end
    assign src_rd_ptr_gray_sync = src_rd_ptr_gray_sync_reg[SYNC_STAGES-1];
    assign src_rd_ptr_bin = gray2bin(src_rd_ptr_gray_sync);

    // Synchronize write pointer gray from src to dst
    reg [SYNC_STAGES-1:0][PTR_W-1:0] dst_wr_ptr_gray_sync_reg;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync_reg[0] <= {PTR_W{1'b0}};
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                dst_wr_ptr_gray_sync_reg[i] <= {PTR_W{1'b0}};
        end
        else begin
            dst_wr_ptr_gray_sync_reg[0] <= dst_wr_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                dst_wr_ptr_gray_sync_reg[i] <= dst_wr_ptr_gray_sync_reg[i-1];
        end
    end
    assign dst_wr_ptr_gray_sync = dst_wr_ptr_gray_sync_reg[SYNC_STAGES-1];
    assign dst_wr_ptr_bin = gray2bin(dst_wr_ptr_gray_sync);

    // Full/Empty detection
    // Full: write pointer has lapped read pointer by DEPTH
    // Compare upper bits (excluding LSB) and check if they differ by 1 in the MSB
    assign src_full = (src_wr_ptr[PTR_W-1] != src_rd_ptr_bin[PTR_W-1]) &&
                      (src_wr_ptr[PTR_W-2:0] == src_rd_ptr_bin[PTR_W-2:0]);

    assign dst_empty = (dst_rd_ptr == dst_wr_ptr_bin);

    // Flow control
    assign src_ready_o = !src_full;
    assign src_can_write = src_valid_i && src_ready_o;

    assign dst_valid_o = !dst_empty;
    assign dst_can_read = dst_valid_o && dst_ready_i;

    // Memory write
    always @(posedge src_clk_i) begin
        if (src_can_write)
            mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
    end

    // Memory read
    reg [WIDTH-1:0] dst_data_reg;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_data_reg <= {WIDTH{1'b0}};
        else if (dst_can_read)
            dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
    end

    assign dst_data_o = dst_data_reg;

endmodule
