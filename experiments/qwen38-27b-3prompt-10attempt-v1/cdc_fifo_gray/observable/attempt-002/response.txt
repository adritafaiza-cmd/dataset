module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;

    // Source side
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_WIDTH-1:0] src_wr_ptr;
    reg [PTR_WIDTH-1:0] src_rd_ptr_gray_sync;
    reg [PTR_WIDTH-1:0] src_rd_ptr_bin;

    // Destination side
    reg [PTR_WIDTH-1:0] dst_rd_ptr;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_bin;

    // Gray code conversion functions
    function [PTR_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_WIDTH-1:0] gray2bin;
        input [PTR_WIDTH-1:0] gray;
        reg [PTR_WIDTH-1:0] bin;
        integer i;
        begin
            bin[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1)
                bin[i] = gray[i] ^ bin[i+1];
            gray2bin = bin;
        end
    endfunction

    // Source side: write pointer and memory write
    wire src_wr_en = src_valid_i & src_ready_o;
    wire [PTR_WIDTH-1:0] src_wr_ptr_gray = bin2gray(src_wr_ptr);

    // Destination side: read pointer and memory read
    wire dst_rd_en = dst_valid_o & dst_ready_i;
    wire [PTR_WIDTH-1:0] dst_rd_ptr_gray = bin2gray(dst_rd_ptr);

    // Synchronize read pointer to source domain
    reg [PTR_WIDTH-1:0] src_rd_ptr_gray_sync_r [0:SYNC_STAGES-1];
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rd_ptr_gray_sync_r[0] <= {PTR_WIDTH{1'b0}};
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                src_rd_ptr_gray_sync_r[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            src_rd_ptr_gray_sync_r[0] <= dst_rd_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                src_rd_ptr_gray_sync_r[i] <= src_rd_ptr_gray_sync_r[i-1];
        end
    end

    assign src_rd_ptr_gray_sync = src_rd_ptr_gray_sync_r[SYNC_STAGES-1];
    assign src_rd_ptr_bin = gray2bin(src_rd_ptr_gray_sync);

    // Synchronize write pointer to destination domain
    reg [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync_r [0:SYNC_STAGES-1];
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync_r[0] <= {PTR_WIDTH{1'b0}};
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                dst_wr_ptr_gray_sync_r[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync_r[0] <= src_wr_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                dst_wr_ptr_gray_sync_r[i] <= dst_wr_ptr_gray_sync_r[i-1];
        end
    end

    assign dst_wr_ptr_gray_sync = dst_wr_ptr_gray_sync_r[SYNC_STAGES-1];
    assign dst_wr_ptr_bin = gray2bin(dst_wr_ptr_gray_sync);

    // Source side: ready logic
    // FIFO is full when the upper bit of write pointer differs from upper bit of read pointer
    // and the lower bits are equal
    wire src_full = (src_wr_ptr[PTR_WIDTH-1] != src_rd_ptr_bin[PTR_WIDTH-1]) &&
                    (src_wr_ptr[PTR_WIDTH-2:0] == src_rd_ptr_bin[PTR_WIDTH-2:0]);
    assign src_ready_o = ~src_full;

    // Destination side: valid logic
    // FIFO is empty when write pointer equals read pointer
    wire dst_empty = (dst_wr_ptr_bin == dst_rd_ptr);
    assign dst_valid_o = ~dst_empty;

    // Source side: write to memory
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
        end else if (src_wr_en) begin
            mem[src_wr_ptr[PTR_WIDTH-2:0]] <= src_data_i;
            src_wr_ptr <= src_wr_ptr + 1'b1;
        end
    end

    // Destination side: read from memory
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
            dst_data_o <= {WIDTH{1'b0}};
        end else if (dst_rd_en) begin
            dst_data_o <= mem[dst_rd_ptr[PTR_WIDTH-2:0]];
            dst_rd_ptr <= dst_rd_ptr + 1'b1;
        end
    end

endmodule
