```verilog
module axis_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    input  wire                   clk,
    input  wire                   rst,
    /*
     * AXI input
     */
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   pause_req,
    output wire                   pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] status_depth,
    output wire [$clog2(DEPTH):0] status_depth_commit,
    output wire                   status_overflow,
    output wire                   status_bad_frame,
    output wire                   status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam DEPTH_MASK = DEPTH - 1;

    // Internal FIFO storage
    reg [DATA_WIDTH-1:0] mem_data [0:DEPTH-1];
    reg [KEEP_WIDTH-1:0] mem_keep [0:DEPTH-1];
    reg                  mem_last [0:DEPTH-1];
    reg [ID_WIDTH-1:0]   mem_id   [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] mem_dest [0:DEPTH-1];
    reg [USER_WIDTH-1:0] mem_user [0:DEPTH-1];

    // Write pointer and read pointer
    reg [ADDR_WIDTH:0] wptr;
    reg [ADDR_WIDTH:0] rptr;

    // Frame tracking
    reg frame_active;
    reg [ADDR_WIDTH:0] frame_start_ptr;
    reg frame_oversize;
    reg frame_bad;

    // Status registers
    reg status_overflow_reg;
    reg status_bad_frame_reg;
    reg status_good_frame_reg;

    // Pause logic
    reg pause_active;

    // Write enable
    wire write_en;
    wire read_en;

    // Full and empty conditions
    wire fifo_full;
    wire fifo_empty;

    // Frame full condition (for frame-based FIFO)
    wire frame_full;

    // Calculate depth
    wire [$clog2(DEPTH):0] depth = wptr - rptr;

    // Determine if FIFO is full
    assign fifo_full = (wptr[ADDR_WIDTH] != rptr[ADDR_WIDTH]) && 
                       (wptr[ADDR_WIDTH-1:0] == rptr[ADDR_WIDTH-1:0]);

    // Determine if FIFO is empty
    assign fifo_empty = (wptr == rptr);

    // Frame full: if frame is active and we've filled the frame
    wire frame_at_capacity;
    assign frame_at_capacity = frame_active && (depth >= DEPTH[ADDR_WIDTH-1:0]);

    // Write enable logic
    wire can_write;
    assign can_write = s_axis_tvalid && !rst && 
                       (!fifo_full || DROP_WHEN_FULL || MARK_WHEN_FULL) &&
                       (!pause_active || !PAUSE_ENABLE) &&
                       (!frame_active || !frame_at_capacity || DROP_OVERSIZE_FRAME);

    assign write_en = can_write;

    // Read enable
    assign read_en = m_axis_tready && !fifo_empty && !rst;

    // Frame oversize detection
    wire frame_oversize_det;
    assign frame_oversize_det = frame_active && s_axis_tlast && 
                                (depth + 1 > DEPTH[ADDR_WIDTH-1:0]);

    // Frame bad detection based on user signal
    wire frame_bad_det;
    assign frame_bad_det = frame_active && s_axis_tlast && 
                           (USER_ENABLE && (s_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE);

    // Determine if we should drop the frame
    wire drop_frame;
    assign drop_frame = (DROP_OVERSIZE_FRAME && frame_oversize_det) ||
                        (DROP_BAD_FRAME && frame_bad_det);

    // Determine if we should mark the frame as bad
    wire mark_frame_bad;
    assign mark_frame_bad = (MARK_WHEN_FULL && fifo_full && s_axis_tvalid) ||
                            (DROP_BAD_FRAME == 0 && frame_bad_det);

    // Write data to memory
    always @(posedge clk) begin
        if (rst) begin
            wptr <= 0;
            rptr <= 0;
            frame_active <= 0;
            frame_start_ptr <= 0;
            frame_oversize <= 0;
            frame_bad <= 0;
            status_overflow_reg <= 0;
            status_bad_frame_reg <= 0;
            status_good_frame_reg <= 0;
            pause_active <= 0;
        end else begin
            // Handle pause
            if (PAUSE_ENABLE) begin
                if (pause_req) begin
                    pause_active <= 1;
                end else if (m_axis_tready && !fifo_empty) begin
                    pause_active <= 0;
                end
            end

            // Write to FIFO
            if (write_en) begin
                mem_data[wptr[ADDR_WIDTH-1:0]] <= s_axis_tdata;
                mem_keep[wptr[ADDR_WIDTH-1:0]] <= s_axis_tkeep;
                mem_last[wptr[ADDR_WIDTH-1:0]] <= s_axis_tlast;
                mem_id[wptr[ADDR_WIDTH-1:0]]   <= s_axis_tid;
                mem_dest[wptr[ADDR_WIDTH-1:0]] <= s_axis_tdest;
                mem_user[wptr[ADDR_WIDTH-1:0]] <= s_axis_tuser;

                // Update frame tracking
                if (!frame_active) begin
                    frame_active <= 1;
                    frame_start_ptr <= wptr;
                    frame_oversize <= 0;
                    frame_bad <= 0;
                end else begin
                    if (s_axis_tlast) begin
                        frame_active <= 0;
                    end
                end

                // Check for oversize
                if (frame_active && !s_axis_tlast) begin
                    if (depth + 1 >= DEPTH[ADDR_WIDTH-1:0]) begin
                        frame_oversize <= 1;
                    end
                end

                // Check for bad frame
                if (frame_active && s_axis_tlast) begin
                    if (USER_ENABLE && (s_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                        frame_bad <= 1;
                    end
                end

                wptr <= wptr + 1;
            end

            // Read from FIFO
            if (read_en) begin
                rptr <= rptr + 1;
            end

            // Status updates
            if (write_en && fifo_full && !DROP_WHEN_FULL && !MARK_WHEN_FULL) begin
                status_overflow_reg <= 1;
            end

            if (frame_active && s_axis_tlast && write_en) begin
                if (frame_bad || (DROP_BAD_FRAME && frame_bad_det)) begin
                    status_bad_frame_reg <= 1;
                end else begin
                    status_good_frame_reg <= 1;
                end
            end
        end
    end

    // Output data
    assign m_axis_tdata = mem_data[rptr[ADDR_WIDTH-1:0]];
    assign m_axis_tkeep = mem_keep[rptr[ADDR_WIDTH-1:0]];
    assign m_axis_tlast = mem_last[rptr[ADDR_WIDTH-1:0]];
    assign m_axis_tid   = mem_id[rptr[ADDR_WIDTH-1:0]];
    assign m_axis_tdest = mem_dest[rptr[ADDR_WIDTH-1:0]];
    assign m_axis_tuser = mem_user[rptr[ADDR_WIDTH-1:0]];

    // Output valid
    assign m_axis_tvalid = !fifo_empty && !rst;

    // Input ready
    assign s_axis_tready = !rst && 
                           (!fifo_full || DROP_WHEN_FULL || MARK_WHEN_FULL) &&
                           (!pause_active || !PAUSE_ENABLE) &&
                           (!frame_active || !frame_at_capacity || DROP_OVERSIZE_FRAME);

    // Pause acknowledge
    assign pause_ack = pause_active;

    // Status outputs
    assign status_depth = depth;
    assign status_depth_commit = depth;
    assign status_overflow = status_overflow_reg;
    assign status_bad_frame = status_bad_frame_reg;
    assign status_good_frame = status_good_frame_reg;

endmodule
