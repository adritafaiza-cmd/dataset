```verilog
module async_bidir_ramif_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "FALSE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir,
       output wire             o_ram_a_clk,
       output wire [DSIZE-1:0] o_ram_a_wdata,
       input wire [DSIZE-1:0]  i_ram_a_rdata,
       output wire [ASIZE-1:0] o_ram_a_addr,
       output wire             o_ram_a_rinc,
       output wire             o_ram_a_winc,
       output wire             o_ram_b_clk,
       output wire [DSIZE-1:0] o_ram_b_wdata,
       input wire [DSIZE-1:0]  i_ram_b_rdata,
       output wire [ASIZE-1:0] o_ram_b_addr,
       output wire             o_ram_b_rinc,
       output wire             o_ram_b_winc
       );

  localparam DEPTH = 1 << ASIZE;
  localparam PTR_W = ASIZE + 1;
  localparam AFULL_MARGIN = DEPTH/4;
  localparam AEMPTY_MARGIN = DEPTH/4;

  // Side A registers
  reg [PTR_W-1:0] a_wr_ptr;
  reg [PTR_W-1:0] a_rd_ptr;
  reg [PTR_W-1:0] a_wr_ptr_gray;
  reg [PTR_W-1:0] a_rd_ptr_gray;
  reg [PTR_W-1:0] a_wr_ptr_gray_sync;
  reg [PTR_W-1:0] a_rd_ptr_gray_sync;
  reg [PTR_W-1:0] a_wr_ptr_gray_sync2;
  reg [PTR_W-1:0] a_rd_ptr_gray_sync2;

  // Side B registers
  reg [PTR_W-1:0] b_wr_ptr;
  reg [PTR_W-1:0] b_rd_ptr;
  reg [PTR_W-1:0] b_wr_ptr_gray;
  reg [PTR_W-1:0] b_rd_ptr_gray;
  reg [PTR_W-1:0] b_wr_ptr_gray_sync;
  reg [PTR_W-1:0] b_rd_ptr_gray_sync;
  reg [PTR_W-1:0] b_wr_ptr_gray_sync2;
  reg [PTR_W-1:0] b_rd_ptr_gray_sync2;

  // Gray code conversion
  function [PTR_W-1:0] bin2gray;
    input [PTR_W-1:0] bin;
    begin
      bin2gray = bin ^ (bin >> 1);
    end
  endfunction

  // Side A logic
  wire a_wr_en = a_winc & ~a_full;
  wire a_rd_en = a_rinc & ~a_empty;

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= {PTR_W{1'b0}};
      a_rd_ptr <= {PTR_W{1'b0}};
      a_wr_ptr_gray <= {PTR_W{1'b0}};
      a_rd_ptr_gray <= {PTR_W{1'b0}};
      a_wr_ptr_gray_sync <= {PTR_W{1'b0}};
      a_rd_ptr_gray_sync <= {PTR_W{1'b0}};
      a_wr_ptr_gray_sync2 <= {PTR_W{1'b0}};
      a_rd_ptr_gray_sync2 <= {PTR_W{1'b0}};
    end else begin
      if (a_wr_en) begin
        a_wr_ptr <= a_wr_ptr + 1'b1;
        a_wr_ptr_gray <= bin2gray(a_wr_ptr + 1'b1);
      end
      if (a_rd_en) begin
        a_rd_ptr <= a_rd_ptr + 1'b1;
        a_rd_ptr_gray <= bin2gray(a_rd_ptr + 1'b1);
      end
      a_wr_ptr_gray_sync <= a_wr_ptr_gray;
      a_rd_ptr_gray_sync <= a_rd_ptr_gray;
      a_wr_ptr_gray_sync2 <= a_wr_ptr_gray_sync;
      a_rd_ptr_gray_sync2 <= a_rd_ptr_gray_sync;
    end
  end

  // Side B logic
  wire b_wr_en = b_winc & ~b_full;
  wire b_rd_en = b_rinc & ~b_empty;

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= {PTR_W{1'b0}};
      b_rd_ptr <= {PTR_W{1'b0}};
      b_wr_ptr_gray <= {PTR_W{1'b0}};
      b_rd_ptr_gray <= {PTR_W{1'b0}};
      b_wr_ptr_gray_sync <= {PTR_W{1'b0}};
      b_rd_ptr_gray_sync <= {PTR_W{1'b0}};
      b_wr_ptr_gray_sync2 <= {PTR_W{1'b0}};
      b_rd_ptr_gray_sync2 <= {PTR_W{1'b0}};
    end else begin
      if (b_wr_en) begin
        b_wr_ptr <= b_wr_ptr + 1'b1;
        b_wr_ptr_gray <= bin2gray(b_wr_ptr + 1'b1);
      end
      if (b_rd_en) begin
        b_rd_ptr <= b_rd_ptr + 1'b1;
        b_rd_ptr_gray <= bin2gray(b_rd_ptr + 1'b1);
      end
      b_wr_ptr_gray_sync <= b_wr_ptr_gray;
      b_rd_ptr_gray_sync <= b_rd_ptr_gray;
      b_wr_ptr_gray_sync2 <= b_wr_ptr_gray_sync;
      b_rd_ptr_gray_sync2 <= b_rd_ptr_gray_sync;
    end
  end

  // Full/Empty flags
  wire a_full = (a_wr_ptr_gray_sync2 == a_rd_ptr_gray);
  wire a_empty = (a_rd_ptr_gray_sync2 == a_wr_ptr_gray);
  wire b_full = (b_wr_ptr_gray_sync2 == b_rd_ptr_gray);
  wire b_empty = (b_rd_ptr_gray_sync2 == b_wr_ptr_gray);

  // Almost full/empty
  wire [PTR_W-1:0] a_wr_minus_rd = a_wr_ptr - a_rd_ptr;
  wire [PTR_W-1:0] b_wr_minus_rd = b_wr_ptr - b_rd_ptr;
  wire a_afull = (a_wr_minus_rd >= (DEPTH - AFULL_MARGIN));
  wire a_aempty = (a_wr_minus_rd <= AEMPTY_MARGIN);
  wire b_afull = (b_wr_minus_rd >= (DEPTH - AFULL_MARGIN));
  wire b_aempty = (b_wr_minus_rd <= AEMPTY_MARGIN);

  // RAM interface
  assign o_ram_a_clk = a_clk;
  assign o_ram_a_wdata = a_wdata;
  assign o_ram_a_addr = a_wr_ptr[ASIZE-1:0];
  assign o_ram_a_winc = a_wr_en;
  assign o_ram_a_rinc = a_rd_en;
  assign a_rdata = i_ram_a_rdata;

  assign o_ram_b_clk = b_clk;
  assign o_ram_b_wdata = b_wdata;
  assign o_ram_b_addr = b_wr_ptr[ASIZE-1:0];
  assign o_ram_b_winc = b_wr_en;
  assign o_ram_b_rinc = b_rd_en;
  assign b_rdata = i_ram_b_rdata;

  assign a_full = a_full;
  assign a_afull = a_afull;
  assign a_empty = a_empty;
  assign a_aempty = a_aempty;
  assign b_full = b_full;
  assign b_afull = b_afull;
  assign b_empty = b_empty;
  assign b_aempty = b_aempty;

endmodule
