  // wr_en_b = b_winc & b_dir  (B writes to FIFO)

  // RAM read: A side, when rd_en_a

  reg [AWIDTH-1:0] a2b_rd_ptr;

  reg [AWIDTH-1:0] a2b_rd_ptr_gray_sync1;
  reg [AWIDTH-1:0] a2b_rd_ptr_gray_sync2;

  reg [AWIDTH-1:0] b2a_rd_ptr;

  reg [AWIDTH-1:0] b2a_wr_ptr_gray_sync1;
  reg [AWIDTH-1:0] b2a_wr_ptr_gray_sync2;

  wire a2b_rd_en = b_rinc & ~b_dir;

  wire b2a_wr_en = b_winc & b_dir;
  wire b2a_rd_en = a_rinc & ~a_dir;

  wire [AWIDTH-1:0] a2b_rd_ptr_gray_sync2_diff;
  
  // Convert gray to binary for comparison
  function [AWIDTH-1:0] gray2bin;
    input [AWIDTH-1:0] gray;
    integer i;
    begin
      gray2bin[AWIDTH-1] = gray[AWIDTH-1];
      for (i = AWIDTH-2; i >= 0; i = i - 1)
        gray2bin[i] = gray2bin[i+1] ^ gray[i];
    end
  endfunction

    input [AWIDTH-1:0] bin;
    begin
      bin2gray = bin ^ (bin >> 1);
    end
  endfunction

  wire [AWIDTH-1:0] a2b_rd_ptr_bin_sync = gray2bin(a2b_rd_ptr_gray_sync2);

  wire a2b_full = (a2b_wr_ptr_bin_sync[AWIDTH-1:1] == a2b_rd_ptr_bin_sync[AWIDTH-1:1]) &

  wire a2b_empty = (a2b_wr_ptr_bin_sync == a2b_rd_ptr_bin_sync);

  wire [AWIDTH-1:0] b2a_rd_ptr_bin_sync = gray2bin(b2a_rd_ptr_gray_sync2);

  wire b2a_empty = (b2a_wr_ptr_bin_sync == b2a_rd_ptr_bin_sync);

  // A side: empty when B-to-A is empty (A can't read)

  wire [AWIDTH-1:0] rd_ptr_bin_sync_a = gray2bin(rd_ptr_gray_sync2);

  wire [AWIDTH-1:0] rd_ptr_bin_sync_b = gray2bin(rd_ptr_gray_sync2);
