```verilog
module async_bidir_ramif_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "FALSE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir,
       output wire             o_ram_a_clk,
       output wire [DSIZE-1:0] o_ram_a_wdata,
       input wire [DSIZE-1:0]  i_ram_a_rdata,
       output wire [ASIZE-1:0] o_ram_a_addr,
       output wire             o_ram_a_rinc,
       output wire             o_ram_a_winc,
       output wire             o_ram_b_clk,
       output wire [DSIZE-1:0] o_ram_b_wdata,
       input wire [DSIZE-1:0]  i_ram_b_rdata,
       output wire [ASIZE-1:0] o_ram_b_addr,
       output wire             o_ram_b_rinc,
       output wire             o_ram_b_winc
       );

  localparam AWIDTH = ASIZE + 1;
  localparam DEPTH  = 1 << ASIZE;
  localparam AFULL_THRESH  = DEPTH - 2;
  localparam AEMPTY_THRESH = 1;

  // A-side state
  reg [AWIDTH-1:0] a_wr_ptr;
  reg [AWIDTH-1:0] a_rd_ptr;
  reg [AWIDTH-1:0] a_wr_ptr_b;
  reg [AWIDTH-1:0] a_rd_ptr_b;

  // B-side state
  reg [AWIDTH-1:0] b_wr_ptr;
  reg [AWIDTH-1:0] b_rd_ptr;
  reg [AWIDTH-1:0] b_wr_ptr_a;
  reg [AWIDTH-1:0] b_rd_ptr_a;

  // A-side status
  wire a_full   = (a_wr_ptr == a_rd_ptr_b) && (a_wr_ptr[ASIZE] != a_rd_ptr_b[ASIZE]);
  wire a_empty  = (a_wr_ptr_b == a_rd_ptr);
  wire a_afull  = (a_wr_ptr - a_rd_ptr_b) >= AFULL_THRESH;
  wire a_aempty = (a_wr_ptr_b - a_rd_ptr) <= AEMPTY_THRESH;

  // B-side status
  wire b_full   = (b_wr_ptr == b_rd_ptr_a) && (b_wr_ptr[ASIZE] != b_rd_ptr_a[ASIZE]);
  wire b_empty  = (b_wr_ptr_a == b_rd_ptr);
  wire b_afull  = (b_wr_ptr - b_rd_ptr_a) >= AFULL_THRESH;
  wire b_aempty = (b_wr_ptr_a - b_rd_ptr) <= AEMPTY_THRESH;

  // A-side RAM control
  wire a_ram_wen = a_winc && !a_full;
  wire a_ram_ren = a_rinc && !a_empty;
  wire a_ram_we  = a_ram_wen;
  wire a_ram_re  = a_ram_ren;

  assign o_ram_a_clk   = a_clk;
  assign o_ram_a_wdata = a_wdata;
  assign o_ram_a_addr  = a_wr_ptr[ASIZE-1:0];
  assign o_ram_a_winc  = a_ram_we;
  assign o_ram_a_rinc  = a_ram_re;

  // B-side RAM control
  wire b_ram_wen = b_winc && !b_full;
  wire b_ram_ren = b_rinc && !b_empty;
  wire b_ram_we  = b_ram_wen;
  wire b_ram_re  = b_ram_ren;

  assign o_ram_b_clk   = b_clk;
  assign o_ram_b_wdata = b_wdata;
  assign o_ram_b_addr  = b_wr_ptr[ASIZE-1:0];
  assign o_ram_b_winc  = b_ram_we;
  assign o_ram_b_rinc  = b_ram_re;

  // A-side read data
  reg [DSIZE-1:0] a_rdata_reg;
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n)
      a_rdata_reg <= {DSIZE{1'b0}};
    else if (a_ram_re)
      a_rdata_reg <= i_ram_a_rdata;
  end

  // Fall-through logic for A
  wire a_fallthrough = (FALLTHROUGH == "TRUE") && a_empty && a_ram_re;
  assign a_rdata = a_fallthrough ? a_wdata : a_rdata_reg;

  // B-side read data
  reg [DSIZE-1:0] b_rdata_reg;
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n)
      b_rdata_reg <= {DSIZE{1'b0}};
    else if (b_ram_re)
      b_rdata_reg <= i_ram_b_rdata;
  end

  // Fall-through logic for B
  wire b_fallthrough = (FALLTHROUGH == "TRUE") && b_empty && b_ram_re;
  assign b_rdata = b_fallthrough ? b_wdata : b_rdata_reg;

  // A-side pointer updates
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= {AWIDTH{1'b0}};
      a_rd_ptr <= {AWIDTH{1'b0}};
    end else begin
      if (a_ram_we)
        a_wr_ptr <= a_wr_ptr + 1'b1;
      if (a_ram_re)
        a_rd_ptr <= a_rd_ptr + 1'b1;
    end
  end

  // B-side pointer updates
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= {AWIDTH{1'b0}};
      b_rd_ptr <= {AWIDTH{1'b0}};
    end else begin
      if (b_ram_we)
        b_wr_ptr <= b_wr_ptr + 1'b1;
      if (b_ram_re)
        b_rd_ptr <= b_rd_ptr + 1'b1;
    end
  end

  // A-side pointer snapshots for B (crossing to B domain)
  // a_wr_ptr crosses to B as a_wr_ptr_b
  // a_rd_ptr crosses to B as a_rd_ptr_b
  // These are used by B to determine A's full/empty status

  // B-side pointer snapshots for A (crossing to A domain)
  // b_wr_ptr crosses to A as b_wr_ptr_a
  // b_rd_ptr crosses to A as b_rd_ptr_a
  // These are used by A to determine B's full/empty status

  // Gray code conversion for CDC
  function [AWIDTH-1:0] bin2gray;
    input [AWIDTH-1:0] bin;
    begin
      bin2gray = bin ^ (bin >> 1);
    end
  endfunction

  function [AWIDTH-1:0] gray2bin;
    input [AWIDTH-1:0] gray;
    integer i;
    begin
      gray2bin[AWIDTH-1] = gray[AWIDTH-1];
      for (i = AWIDTH-2; i >= 0; i = i - 1)
        gray2bin[i] = gray2bin[i+1] ^ gray[i];
    end
  endfunction

  // A-side: send a_wr_ptr and a_rd_ptr to B domain
  reg [AWIDTH-1:0] a_wr_ptr_gray;
  reg [AWIDTH-1:0] a_rd_ptr_gray;
  reg [AWIDTH-1:0] a_wr_ptr_gray_b;
  reg [AWIDTH-1:0] a_rd_ptr_gray_b;

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr_gray <= {AWIDTH{1'b0}};
      a_rd_ptr_gray <= {AWIDTH{1'b0}};
    end else begin
      a_wr_ptr_gray <= bin2gray(a_wr_ptr);
      a_rd_ptr_gray <= bin2gray(a_rd_ptr);
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_wr_ptr_gray_b <= {AWIDTH{1'b0}};
      a_rd_ptr_gray_b <= {AWIDTH{1'b0}};
    end else begin
      a_wr_ptr_gray_b <= a_wr_ptr_gray;
      a_rd_ptr_gray_b <= a_rd_ptr_gray;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_wr_ptr_b <= {AWIDTH{1'b0}};
      a_rd_ptr_b <= {AWIDTH{1'b0}};
    end else begin
      a_wr_ptr_b <= gray2bin(a_wr_ptr_gray_b);
      a_rd_ptr_b <= gray2bin(a_rd_ptr_gray_b);
    end
  end

  // B-side: send b_wr_ptr and b_rd_ptr to A domain
  reg [AWIDTH-1:0] b_wr_ptr_gray;
  reg [AWIDTH-1:0] b_rd_ptr_gray;
  reg [AWIDTH-1:0] b_wr_ptr_gray_a;
  reg [AWIDTH-1:0] b_rd_ptr_gray_a;

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr_gray <= {AWIDTH{1'b0}};
      b_rd_ptr_gray <= {AWIDTH{1'b0}};
    end else begin
      b_wr_ptr_gray <= bin2gray(b_wr_ptr);
      b_rd_ptr_gray <= bin2gray(b_rd_ptr);
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_wr_ptr_gray_a <= {AWIDTH{1'b0}};
      b_rd_ptr_gray_a <= {AWIDTH{1'b0}};
    end else begin
      b_wr_ptr_gray_a <= b_wr_ptr_gray;
      b_rd_ptr_gray_a <= b_rd_ptr_gray;
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_wr_ptr_a <= {AWIDTH{1'b0}};
      b_rd_ptr_a <= {AWIDTH{1'b0}};
    end else begin
      b_wr_ptr_a <= gray2bin(b_wr_ptr_gray_a);
      b_rd_ptr_a <= gray2bin(b_rd_ptr_gray_a);
    end
  end

endmodule
