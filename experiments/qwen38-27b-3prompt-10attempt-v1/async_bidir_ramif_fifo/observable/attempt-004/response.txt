module async_bidir_ramif_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "FALSE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir,
       output wire             o_ram_a_clk,
       output wire [DSIZE-1:0] o_ram_a_wdata,
       input wire [DSIZE-1:0]  i_ram_a_rdata,
       output wire [ASIZE-1:0] o_ram_a_addr,
       output wire             o_ram_a_rinc,
       output wire             o_ram_a_winc,
       output wire             o_ram_b_clk,
       output wire [DSIZE-1:0] o_ram_b_wdata,
       input wire [DSIZE-1:0]  i_ram_b_rdata,
       output wire [ASIZE-1:0] o_ram_b_addr,
       output wire             o_ram_b_rinc,
       output wire             o_ram_b_winc
       );

  localparam AWIDTH = ASIZE + 1;
  localparam DEPTH  = 1 << ASIZE;

  // ---------------------------------------------------------------------------
  // Side A logic (clocked by a_clk)
  // ---------------------------------------------------------------------------

  reg [AWIDTH-1:0] a_wr_ptr;
  reg [AWIDTH-1:0] a_rd_ptr;

  // Synchronizers for B's pointers into A domain
  reg [AWIDTH-1:0] b_wr_ptr_sync1, b_wr_ptr_sync2;
  reg [AWIDTH-1:0] b_rd_ptr_sync1, b_rd_ptr_sync2;

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_wr_ptr_sync1 <= {AWIDTH{1'b0}};
      b_wr_ptr_sync2 <= {AWIDTH{1'b0}};
    end else begin
      b_wr_ptr_sync1 <= b_wr_ptr_sync2;
      b_wr_ptr_sync2 <= b_wr_ptr;
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_rd_ptr_sync1 <= {AWIDTH{1'b0}};
      b_rd_ptr_sync2 <= {AWIDTH{1'b0}};
    end else begin
      b_rd_ptr_sync1 <= b_rd_ptr_sync2;
      b_rd_ptr_sync2 <= b_rd_ptr;
    end
  end

  // A-side status
  wire a_full  = (a_wr_ptr == b_rd_ptr_sync2);
  wire a_empty = (a_rd_ptr == b_wr_ptr_sync2);

  wire a_afull  = (a_wr_ptr[ASIZE] == b_rd_ptr_sync2[ASIZE]) &&
                  (a_wr_ptr[ASIZE-1:0] == b_rd_ptr_sync2[ASIZE-1:0] + ASIZE'(DEPTH - 2));
  wire a_aempty = (a_rd_ptr[ASIZE] == b_wr_ptr_sync2[ASIZE]) &&
                  (a_rd_ptr[ASIZE-1:0] == b_wr_ptr_sync2[ASIZE-1:0] + ASIZE'(DEPTH - 2));

  // A-side RAM port control
  wire a_ram_wen = a_winc && !a_full;
  wire a_ram_ren = a_rinc && !a_empty;

  assign o_ram_a_clk   = a_clk;
  assign o_ram_a_wdata = a_wdata;
  assign o_ram_a_addr  = a_wr_ptr[ASIZE-1:0];
  assign o_ram_a_winc  = a_ram_wen;
  assign o_ram_a_rinc  = a_ram_ren;

  // A-side read data: fall-through or from RAM
  wire a_fallthrough = (FALLTHROUGH == "TRUE") && a_ram_wen && a_ram_ren;
  assign a_rdata = a_fallthrough ? a_wdata : i_ram_a_rdata;

  // A-side pointer updates
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= {AWIDTH{1'b0}};
      a_rd_ptr <= {AWIDTH{1'b0}};
    end else begin
      if (a_ram_wen) a_wr_ptr <= a_wr_ptr + 1'b1;
      if (a_ram_ren) a_rd_ptr <= a_rd_ptr + 1'b1;
    end
  end

  // ---------------------------------------------------------------------------
  // Side B logic (clocked by b_clk)
  // ---------------------------------------------------------------------------

  reg [AWIDTH-1:0] b_wr_ptr;
  reg [AWIDTH-1:0] b_rd_ptr;

  // Synchronizers for A's pointers into B domain
  reg [AWIDTH-1:0] a_wr_ptr_sync1, a_wr_ptr_sync2;
  reg [AWIDTH-1:0] a_rd_ptr_sync1, a_rd_ptr_sync2;

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_wr_ptr_sync1 <= {AWIDTH{1'b0}};
      a_wr_ptr_sync2 <= {AWIDTH{1'b0}};
    end else begin
      a_wr_ptr_sync1 <= a_wr_ptr_sync2;
      a_wr_ptr_sync2 <= a_wr_ptr;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_rd_ptr_sync1 <= {AWIDTH{1'b0}};
      a_rd_ptr_sync2 <= {AWIDTH{1'b0}};
    end else begin
      a_rd_ptr_sync1 <= a_rd_ptr_sync2;
      a_rd_ptr_sync2 <= a_rd_ptr;
    end
  end

  // B-side status
  wire b_full  = (b_wr_ptr == a_rd_ptr_sync2);
  wire b_empty = (b_rd_ptr == a_wr_ptr_sync2);

  wire b_afull  = (b_wr_ptr[ASIZE] == a_rd_ptr_sync2[ASIZE]) &&
                  (b_wr_ptr[ASIZE-1:0] == a_rd_ptr_sync2[ASIZE-1:0] + ASIZE'(DEPTH - 2));
  wire b_aempty = (b_rd_ptr[ASIZE] == a_wr_ptr_sync2[ASIZE]) &&
                  (b_rd_ptr[ASIZE-1:0] == a_wr_ptr_sync2[ASIZE-1:0] + ASIZE'(DEPTH - 2));

  // B-side RAM port control
  wire b_ram_wen = b_winc && !b_full;
  wire b_ram_ren = b_rinc && !b_empty;

  assign o_ram_b_clk   = b_clk;
  assign o_ram_b_wdata = b_wdata;
  assign o_ram_b_addr  = b_wr_ptr[ASIZE-1:0];
  assign o_ram_b_winc  = b_ram_wen;
  assign o_ram_b_rinc  = b_ram_ren;

  // B-side read data: fall-through or from RAM
  wire b_fallthrough = (FALLTHROUGH == "TRUE") && b_ram_wen && b_ram_ren;
  assign b_rdata = b_fallthrough ? b_wdata : i_ram_b_rdata;

  // B-side pointer updates
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= {AWIDTH{1'b0}};
      b_rd_ptr <= {AWIDTH{1'b0}};
    end else begin
      if (b_ram_wen) b_wr_ptr <= b_wr_ptr + 1'b1;
      if (b_ram_ren) b_rd_ptr <= b_rd_ptr + 1'b1;
    end
  end

  // ---------------------------------------------------------------------------
  // Status outputs
  // ---------------------------------------------------------------------------

  assign a_full   = a_full;
  assign a_afull  = a_afull;
  assign a_empty  = a_empty;
  assign a_aempty = a_aempty;

  assign b_full   = b_full;
  assign b_afull  = b_afull;
  assign b_empty  = b_empty;
  assign b_aempty = b_aempty;

endmodule
