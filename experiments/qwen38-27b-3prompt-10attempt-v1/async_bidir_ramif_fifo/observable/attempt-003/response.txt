```verilog
module async_bidir_ramif_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "FALSE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir,
       output wire             o_ram_a_clk,
       output wire [DSIZE-1:0] o_ram_a_wdata,
       input wire [DSIZE-1:0]  i_ram_a_rdata,
       output wire [ASIZE-1:0] o_ram_a_addr,
       output wire             o_ram_a_rinc,
       output wire             o_ram_a_winc,
       output wire             o_ram_b_clk,
       output wire [DSIZE-1:0] o_ram_b_wdata,
       input wire [DSIZE-1:0]  i_ram_b_rdata,
       output wire [ASIZE-1:0] o_ram_b_addr,
       output wire             o_ram_b_rinc,
       output wire             o_ram_b_winc
       );

  localparam AWIDTH = ASIZE + 1;
  localparam DEPTH  = 1 << ASIZE;
  localparam AFULL_OFFSET = 2;
  localparam AEMPTY_OFFSET = 1;

  // A-side state
  reg [AWIDTH-1:0] a_wr_ptr;
  reg [AWIDTH-1:0] a_rd_ptr;
  reg [AWIDTH-1:0] b_wr_ptr_a;
  reg [AWIDTH-1:0] b_rd_ptr_a;

  // B-side state
  reg [AWIDTH-1:0] b_wr_ptr;
  reg [AWIDTH-1:0] b_rd_ptr;
  reg [AWIDTH-1:0] a_wr_ptr_b;
  reg [AWIDTH-1:0] a_rd_ptr_b;

  // CDC synchronizers
  reg [AWIDTH-1:0] a_wr_ptr_b_sync1, a_wr_ptr_b_sync2;
  reg [AWIDTH-1:0] a_rd_ptr_b_sync1, a_rd_ptr_b_sync2;
  reg [AWIDTH-1:0] b_wr_ptr_a_sync1, b_wr_ptr_a_sync2;
  reg [AWIDTH-1:0] b_rd_ptr_a_sync1, b_rd_ptr_a_sync2;

  // Direction sync
  reg a_dir_b_sync1, a_dir_b_sync2;
  reg b_dir_a_sync1, b_dir_a_sync2;

  // A-side status
  wire a_full   = (a_wr_ptr == b_rd_ptr_a);
  wire a_empty  = (a_rd_ptr == b_wr_ptr_a);
  wire a_afull  = (a_wr_ptr[AWIDTH-1:1] == b_rd_ptr_a[AWIDTH-1:1]) &&
                  (a_wr_ptr[0] != b_rd_ptr_a[0]) &&
                  (a_wr_ptr[0] == 1'b1);
  wire a_aempty = (a_rd_ptr[AWIDTH-1:1] == b_wr_ptr_a[AWIDTH-1:1]) &&
                  (a_rd_ptr[0] != b_wr_ptr_a[0]) &&
                  (a_rd_ptr[0] == 1'b0);

  // B-side status
  wire b_full   = (b_wr_ptr == a_rd_ptr_b);
  wire b_empty  = (b_rd_ptr == a_wr_ptr_b);
  wire b_afull  = (b_wr_ptr[AWIDTH-1:1] == a_rd_ptr_b[AWIDTH-1:1]) &&
                  (b_wr_ptr[0] != a_rd_ptr_b[0]) &&
                  (b_wr_ptr[0] == 1'b1);
  wire b_aempty = (b_rd_ptr[AWIDTH-1:1] == a_wr_ptr_b[AWIDTH-1:1]) &&
                  (b_rd_ptr[0] != b_wr_ptr_b[0]) &&
                  (b_rd_ptr[0] == 1'b0);

  // A-side RAM control
  wire a_ram_wen = a_winc && !a_full;
  wire a_ram_ren = a_rinc && !a_empty;
  wire a_ram_fallthrough = (FALLTHROUGH == "TRUE") && a_winc && a_rinc && !a_full && !a_empty;

  // B-side RAM control
  wire b_ram_wen = b_winc && !b_full;
  wire b_ram_ren = b_rinc && !b_empty;
  wire b_ram_fallthrough = (FALLTHROUGH == "TRUE") && b_winc && b_rinc && !b_full && !b_empty;

  // A-side RAM outputs
  assign o_ram_a_clk   = a_clk;
  assign o_ram_a_wdata = a_wdata;
  assign o_ram_a_addr  = a_wr_ptr[ASIZE-1:0];
  assign o_ram_a_winc  = a_ram_wen;
  assign o_ram_a_rinc  = a_ram_ren;

  // B-side RAM outputs
  assign o_ram_b_clk   = b_clk;
  assign o_ram_b_wdata = b_wdata;
  assign o_ram_b_addr  = b_wr_ptr[ASIZE-1:0];
  assign o_ram_b_winc  = b_ram_wen;
  assign o_ram_b_rinc  = b_ram_ren;

  // A-side read data
  assign a_rdata = a_ram_fallthrough ? a_wdata : i_ram_a_rdata;

  // B-side read data
  assign b_rdata = b_ram_fallthrough ? b_wdata : i_ram_b_rdata;

  // A-side pointer updates
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= {AWIDTH{1'b0}};
      a_rd_ptr <= {AWIDTH{1'b0}};
      b_wr_ptr_a <= {AWIDTH{1'b0}};
      b_rd_ptr_a <= {AWIDTH{1'b0}};
    end else begin
      if (a_ram_wen)
        a_wr_ptr <= a_wr_ptr + 1'b1;
      if (a_ram_ren)
        a_rd_ptr <= a_rd_ptr + 1'b1;
      b_wr_ptr_a <= b_wr_ptr_a_sync2;
      b_rd_ptr_a <= b_rd_ptr_a_sync2;
    end
  end

  // B-side pointer updates
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= {AWIDTH{1'b0}};
      b_rd_ptr <= {AWIDTH{1'b0}};
      a_wr_ptr_b <= {AWIDTH{1'b0}};
      a_rd_ptr_b <= {AWIDTH{1'b0}};
    end else begin
      if (b_ram_wen)
        b_wr_ptr <= b_wr_ptr + 1'b1;
      if (b_ram_ren)
        b_rd_ptr <= b_rd_ptr + 1'b1;
      a_wr_ptr_b <= a_wr_ptr_b_sync2;
      a_rd_ptr_b <= a_rd_ptr_b_sync2;
    end
  end

  // CDC: A write pointer to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_wr_ptr_b_sync1 <= {AWIDTH{1'b0}};
      a_wr_ptr_b_sync2 <= {AWIDTH{1'b0}};
    end else begin
      a_wr_ptr_b_sync1 <= a_wr_ptr;
      a_wr_ptr_b_sync2 <= a_wr_ptr_b_sync1;
    end
  end

  // CDC: A read pointer to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_rd_ptr_b_sync1 <= {AWIDTH{1'b0}};
      a_rd_ptr_b_sync2 <= {AWIDTH{1'b0}};
    end else begin
      a_rd_ptr_b_sync1 <= a_rd_ptr;
      a_rd_ptr_b_sync2 <= a_rd_ptr_b_sync1;
    end
  end

  // CDC: B write pointer to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_wr_ptr_a_sync1 <= {AWIDTH{1'b0}};
      b_wr_ptr_a_sync2 <= {AWIDTH{1'b0}};
    end else begin
      b_wr_ptr_a_sync1 <= b_wr_ptr;
      b_wr_ptr_a_sync2 <= b_wr_ptr_a_sync1;
    end
  end

  // CDC: B read pointer to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_rd_ptr_a_sync1 <= {AWIDTH{1'b0}};
      b_rd_ptr_a_sync2 <= {AWIDTH{1'b0}};
    end else begin
      b_rd_ptr_a_sync1 <= b_rd_ptr;
      b_rd_ptr_a_sync2 <= b_rd_ptr_a_sync1;
    end
  end

  // CDC: A direction to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_dir_b_sync1 <= 1'b0;
      a_dir_b_sync2 <= 1'b0;
    end else begin
      a_dir_b_sync1 <= a_dir;
      a_dir_b_sync2 <= a_dir_b_sync1;
    end
  end

  // CDC: B direction to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_dir_a_sync1 <= 1'b0;
      b_dir_a_sync2 <= 1'b0;
    end else begin
      b_dir_a_sync1 <= b_dir;
      b_dir_a_sync2 <= b_dir_a_sync1;
    end
  end

endmodule
