module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

    logic [1:0] sync_rst;
    logic       init_cnt_en;
    logic [1:0] init_cnt;

    // Synchronize reset release
    always_ff @(posedge clk_i) begin
        sync_rst <= {sync_rst[0], rst_ni};
    end

    // Synchronized reset output
    assign rst_no = sync_rst[1];

    // Initialization counter: counts for a few cycles after reset release
    // init_no is low during initialization, high after
    always_ff @(posedge clk_i) begin
        if (!rst_no) begin
            init_cnt <= 2'd0;
        end else if (init_cnt < 2'd3) begin
            init_cnt <= init_cnt + 1'b1;
        end
    end

    assign init_no = (init_cnt >= 2'd3);

endmodule
