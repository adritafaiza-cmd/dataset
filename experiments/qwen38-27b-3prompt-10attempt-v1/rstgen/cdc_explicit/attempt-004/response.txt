module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

    // Synchronization chain for reset release
    logic [1:0] sync_rst;

    // Initialize to reset state (active-low = 0)
    // On power-up, we want rst_no to be 0 (reset asserted)
    // The sync chain should start at 0 and transition to 1 when rst_ni goes high

    // Synchronize the reset input
    always_ff @(posedge clk_i) begin
        sync_rst[0] <= rst_ni;
        sync_rst[1] <= sync_rst[0];
    end

    // In test mode, bypass synchronization
    assign rst_no = test_mode_i ? rst_ni : sync_rst[1];

    // init_no: short initialization qualifier after reset release
    // init_no should be low for a few cycles after rst_no releases, then go high
    // Use a small counter or shift register to generate a pulse

    logic [2:0] init_cnt;

    always_ff @(posedge clk_i) begin
        if (!rst_no) begin
            init_cnt <= 3'd0;
        end else if (init_cnt < 3'd3) begin
            init_cnt <= init_cnt + 3'd1;
        end
    end

    // init_no is high when initialization is complete
    assign init_no = (init_cnt == 3'd3);

endmodule
