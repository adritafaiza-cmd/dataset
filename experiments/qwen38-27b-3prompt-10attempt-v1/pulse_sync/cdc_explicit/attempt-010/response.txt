module pulse_sync #(
    parameter STAGES = 2
)(
    input  clk_a,
    input  rstn_a,
    input  clk_b,
    input  rstn_b,
    input  pulseA_i,
    output pulseB_o,
    output busy_o
);

    // Domain A: handshake request
    reg req_a;
    reg ack_a;
    reg ack_a_d1;
    reg ack_a_d2;

    // Domain B: handshake acknowledge
    reg ack_b;
    reg req_b_d1;
    reg req_b_d2;

    // Synchronize ack from B to A (2 stages)
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a) begin
            ack_a_d1 <= 1'b0;
            ack_a_d2 <= 1'b0;
        end else begin
            ack_a_d1 <= ack_b;
            ack_a_d2 <= ack_a_d1;
        end
    end

    // Synchronize req from A to B (2 stages)
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b) begin
            req_b_d1 <= 1'b0;
            req_b_d2 <= 1'b0;
        end else begin
            req_b_d1 <= req_a;
            req_b_d2 <= req_b_d1;
        end
    end

    // Domain A: request generation and clear
    // req_a is set when pulseA_i is high and not busy
    // req_a is cleared when ack_a_d2 (synchronized ack) is high
    wire busy_a = req_a & ~ack_a_d2;
    wire can_send = !busy_a;

    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a) begin
            req_a <= 1'b0;
        end else begin
            if (pulseA_i && can_send) begin
                req_a <= 1'b1;
            end else if (ack_a_d2) begin
                req_a <= 1'b0;
            end
        end
    end

    // Domain B: acknowledge generation and clear
    // ack_b is set when req_b_d2 (synchronized req) is high
    // ack_b is cleared when req_b_d2 is low (req_a has been cleared)
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b) begin
            ack_b <= 1'b0;
        end else begin
            if (req_b_d2) begin
                ack_b <= 1'b1;
            end else begin
                ack_b <= 1'b0;
            end
        end
    end

    // Generate pulseB_o: one-cycle pulse when req_b_d2 transitions from 0 to 1
    reg req_b_d2_prev;
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b) begin
            req_b_d2_prev <= 1'b0;
        end else begin
            req_b_d2_prev <= req_b_d2;
        end
    end

    assign pulseB_o = req_b_d2 & ~req_b_d2_prev;

    // busy_o is high while a pulse is in flight (req_a is high)
    assign busy_o = req_a;

endmodule
