module pulse_sync #(
    parameter STAGES = 2
)(
    input  clk_a,
    input  rstn_a,
    input  clk_b,
    input  rstn_b,
    input  pulseA_i,
    output pulseB_o,
    output busy_o
);

    // Domain A: level synchronizer
    reg [STAGES-1:0] sync_a;
    integer i;
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            sync_a <= {STAGES{1'b0}};
        else begin
            sync_a[0] <= pulseA_i;
            for (i = 1; i < STAGES; i = i + 1)
                sync_a[i] <= sync_a[i-1];
        end
    end

    // Domain B: level synchronizer
    reg [STAGES-1:0] sync_b;
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            sync_b <= {STAGES{1'b0}};
        else begin
            sync_b[0] <= sync_a[STAGES-1];
            for (i = 1; i < STAGES; i = i + 1)
                sync_b[i] <= sync_b[i-1];
        end
    end

    // Domain B: edge detect
    reg sync_b_d;
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            sync_b_d <= 1'b0;
        else
            sync_b_d <= sync_b[STAGES-1];
    end

    wire edge_b = sync_b[STAGES-1] & ~sync_b_d;

    // Domain B: pulse generation
    reg pulseB_r;
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            pulseB_r <= 1'b0;
        else
            pulseB_r <= edge_b;
    end

    assign pulseB_o = pulseB_r;

    // Domain A: busy generation
    reg busy_a;
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            busy_a <= 1'b0;
        else
            busy_a <= sync_b[STAGES-1];
    end

    assign busy_o = busy_a;

endmodule
