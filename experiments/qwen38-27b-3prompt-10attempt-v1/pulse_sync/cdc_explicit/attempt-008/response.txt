```verilog
module pulse_sync #(
    parameter STAGES = 2
)(
    input  clk_a,
    input  rstn_a,
    input  clk_b,
    input  rstn_b,
    input  pulseA_i,
    output pulseB_o,
    output busy_o
);

    // Domain A: handshake request
    reg req_a;
    reg ack_a;

    // Domain B: handshake acknowledge
    reg ack_b;
    reg req_b;

    // Synchronizers
    // req_a crosses from A to B
    reg [STAGES-1:0] req_sync;
    // ack_b crosses from B to A
    reg [STAGES-1:0] ack_sync;

    // Synchronize req_a into domain B
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            req_sync <= {STAGES{1'b0}};
        else
            req_sync <= {req_sync[STAGES-2:0], req_a};
    end

    // Synchronize ack_b into domain A
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            ack_sync <= {STAGES{1'b0}};
        else
            ack_sync <= {ack_sync[STAGES-2:0], ack_b};
    end

    // Domain A logic
    // req_a is set when pulseA_i is received and not busy
    // req_a is cleared when ack_b (synchronized) is high
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a) begin
            req_a <= 1'b0;
            ack_a <= 1'b0;
        end else begin
            if (ack_sync[STAGES-1])
                req_a <= 1'b0;
            else if (pulseA_i && !req_a)
                req_a <= 1'b1;
        end
    end

    // Domain B logic
    // ack_b is set when req_a (synchronized) is high
    // ack_b is cleared when req_a (synchronized) is low
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b) begin
            ack_b <= 1'b0;
            req_b <= 1'b0;
        end else begin
            if (req_sync[STAGES-1])
                ack_b <= 1'b1;
            else
                ack_b <= 1'b0;
        end
    end

    // Generate pulseB_o: rising edge of ack_b
    reg ack_b_d;
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            ack_b_d <= 1'b0;
        else
            ack_b_d <= ack_b;
    end

    assign pulseB_o = ack_b & ~ack_b_d;

    // busy_o is high when req_a is high (pulse in flight)
    assign busy_o = req_a;

endmodule
