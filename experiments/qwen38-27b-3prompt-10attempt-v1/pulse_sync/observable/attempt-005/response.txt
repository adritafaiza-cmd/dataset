module pulse_sync #(
    parameter STAGES = 2
)(
    input  clk_a,
    input  rstn_a,
    input  clk_b,
    input  rstn_b,
    input  pulseA_i,
    output pulseB_o,
    output busy_o
);

    // Synchronizer for pulseA_i into clk_b domain
    reg [STAGES-1:0] sync_a2b;
    integer i;
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            sync_a2b <= {STAGES{1'b0}};
        else begin
            sync_a2b[0] <= pulseA_i;
            for (i = 1; i < STAGES; i = i + 1)
                sync_a2b[i] <= sync_a2b[i-1];
        end
    end

    // Detect rising edge of synchronized signal
    reg sync_a2b_d;
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            sync_a2b_d <= 1'b0;
        else
            sync_a2b_d <= sync_a2b[STAGES-1];
    end

    wire edge_a2b = sync_a2b[STAGES-1] & ~sync_a2b_d;

    // Busy flag in clk_b domain
    reg busy_b;
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            busy_b <= 1'b0;
        else if (edge_a2b)
            busy_b <= 1'b1;
        else if (pulseB_o)
            busy_b <= 1'b0;
    end

    // Generate pulseB_o: one cycle pulse when edge detected and not busy
    assign pulseB_o = edge_a2b & ~busy_b;

    // Synchronize busy_b back to clk_a domain
    reg [STAGES-1:0] sync_b2a;
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            sync_b2a <= {STAGES{1'b0}};
        else begin
            sync_b2a[0] <= busy_b;
            for (i = 1; i < STAGES; i = i + 1)
                sync_b2a[i] <= sync_b2a[i-1];
        end
    end

    assign busy_o = sync_b2a[STAGES-1];

endmodule
