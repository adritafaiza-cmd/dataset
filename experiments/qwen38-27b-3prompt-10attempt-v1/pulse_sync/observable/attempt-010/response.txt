```verilog
module pulse_sync #(
    parameter STAGES = 2
)(
    input  clk_a,
    input  rstn_a,
    input  clk_b,
    input  rstn_b,
    input  pulseA_i,
    output pulseB_o,
    output busy_o
);

    // Request signal in clk_a domain (to be synchronized to clk_b)
    reg req_a;

    // Acknowledge signal in clk_b domain (to be synchronized to clk_a)
    reg ack_b;

    // Synchronizer for request (A -> B)
    reg [STAGES-1:0] req_sync;

    // Synchronizer for acknowledge (B -> A)
    reg [STAGES-1:0] ack_sync;

    // Internal signals
    wire req_b;   // Request in clk_b domain
    wire ack_a;   // Acknowledge in clk_a domain

    // Synchronize request from A to B
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            req_sync <= {STAGES{1'b0}};
        else
            req_sync <= {req_sync[STAGES-2:0], req_a};
    end

    // Synchronize acknowledge from B to A
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            ack_sync <= {STAGES{1'b0}};
        else
            ack_sync <= {ack_sync[STAGES-2:0], ack_b};
    end

    // Extract synchronized signals
    assign req_b = req_sync[STAGES-1];
    assign ack_a = ack_sync[STAGES-1];

    // clk_a domain logic
    // req_a is set when pulseA_i is received and not busy
    // req_a is cleared when ack_a is received
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            req_a <= 1'b0;
        else if (ack_a)
            req_a <= 1'b0;
        else if (pulseA_i && !req_a)
            req_a <= 1'b1;
    end

    // clk_b domain logic
    // ack_b is set when req_b is received and not busy
    // ack_b is cleared when req_b is no longer asserted (or after one cycle)
    // Actually, we need to generate a pulse on pulseB_o when req_b transitions from 0 to 1
    // And set ack_b to acknowledge, then clear it after one cycle

    reg req_b_d;
    reg ack_b_d;

    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b) begin
            req_b_d <= 1'b0;
            ack_b_d <= 1'b0;
        end
        else begin
            req_b_d <= req_b;
            if (req_b && !req_b_d) begin
                // Rising edge of req_b: generate pulse and acknowledge
                ack_b <= 1'b1;
            end
            else if (ack_b) begin
                // Clear ack_b after one cycle
                ack_b <= 1'b0;
            end
        end
    end

    // Generate pulseB_o on rising edge of req_b
    assign pulseB_o = req_b && !req_b_d;

    // busy_o is high when a pulse is in flight (req_a is set in clk_a domain)
    assign busy_o = req_a;

endmodule
