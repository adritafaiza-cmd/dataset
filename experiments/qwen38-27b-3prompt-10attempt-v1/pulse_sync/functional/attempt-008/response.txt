```verilog
module pulse_sync #(
    parameter STAGES = 2
)(
    input  clk_a,
    input  rstn_a,
    input  clk_b,
    input  rstn_b,
    input  pulseA_i,
    output pulseB_o,
    output busy_o
);

    // Domain A: level synchronizer
    reg [STAGES-1:0] sync_a;
    reg level_a;

    // Domain B: level synchronizer
    reg [STAGES-1:0] sync_b;
    reg level_b;
    reg level_b_prev;

    // Generate pulse in domain B
    assign pulseB_o = level_b & ~level_b_prev;

    // Busy signal: high while pulse is in flight (level_b is high)
    assign busy_o = level_b;

    // Domain A: synchronize pulseA_i into a level
    // When pulseA_i is high and not busy, set level_a
    // level_a is cleared when level_b is synchronized back to domain A
    reg level_a_sync;
    reg [STAGES-1:0] sync_b_to_a;

    // Synchronize level_b back to domain A
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            sync_b_to_a <= {STAGES{1'b0}};
        else
            sync_b_to_a <= {sync_b_to_a[STAGES-2:0], level_b};
    end

    assign level_a_sync = sync_b_to_a[STAGES-1];

    // Set level_a when pulseA_i is asserted and not busy
    // Clear level_a when level_b is synchronized back
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            level_a <= 1'b0;
        else if (pulseA_i && !level_a_sync)
            level_a <= 1'b1;
        else if (level_a_sync)
            level_a <= 1'b0;
    end

    // Synchronize level_a to domain B
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            sync_b <= {STAGES{1'b0}};
        else
            sync_b <= {sync_b[STAGES-2:0], level_a};
    end

    assign level_b = sync_b[STAGES-1];

    // Track previous level_b for pulse generation
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            level_b_prev <= 1'b0;
        else
            level_b_prev <= level_b;
    end

endmodule
