module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain: request signal
  reg src_req;
  
  // Destination domain: acknowledge signal
  reg dst_ack;
  
  // Synchronizers
  reg [1:0] req_sync;
  reg [1:0] ack_sync;
  
  // Destination domain: data and valid
  reg [WIDTH-1:0] dst_data_reg;
  reg dst_valid_reg;
  
  // Source domain: destination acknowledge synchronized
  wire dst_ack_src = ack_sync[1];
  
  // Destination domain: source request synchronized
  wire src_req_dst = req_sync[1];
  
  // Source domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_req <= 1'b0;
    end else begin
      if (src_valid_i && src_ready_o) begin
        src_req <= 1'b1;
      end else if (dst_ack_src) begin
        src_req <= 1'b0;
      end
    end
  end
  
  // Source ready: can accept new data when not currently holding a request
  // that hasn't been acknowledged yet
  assign src_ready_o = ~src_req;
  
  // Synchronize src_req to destination domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      req_sync <= 2'b00;
    end else begin
      req_sync <= {req_sync[0], src_req};
    end
  end
  
  // Synchronize dst_ack to source domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      ack_sync <= 2'b00;
    end else begin
      ack_sync <= {ack_sync[0], dst_ack};
    end
  end
  
  // Destination domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_ack <= 1'b0;
      dst_data_reg <= {WIDTH{1'b0}};
      dst_valid_reg <= 1'b0;
    end else begin
      if (dst_valid_reg) begin
        if (dst_ready_i) begin
          // Transfer complete, acknowledge
          dst_ack <= 1'b1;
          dst_valid_reg <= 1'b0;
        end
      end else begin
        if (src_req_dst) begin
          // New request received
          dst_data_reg <= src_data_i;
          dst_valid_reg <= 1'b1;
          dst_ack <= 1'b0;
        end
      end
    end
  end
  
  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

endmodule
