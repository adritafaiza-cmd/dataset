module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain state
  reg src_valid_reg;
  reg [WIDTH-1:0] src_data_reg;

  // Destination domain state
  reg dst_valid_reg;
  reg [WIDTH-1:0] dst_data_reg;

  // Handshake signals crossing domains
  // src_to_dst_valid: indicates source has data ready to send
  // dst_to_src_ready: indicates destination is ready to accept new data

  // We use a simple 2-phase handshake with data latched in source domain
  // and synchronized to destination domain.

  // Source domain:
  // - When src_valid_i && src_ready_o, latch data and assert src_to_dst_valid
  // - src_ready_o is high when we don't have outstanding data (src_valid_reg == 0)
  //   OR when destination has acknowledged (dst_to_src_ready synchronized)

  // Actually, let's use a simpler approach:
  // Source latches data when valid && ready.
  // src_ready_o = ~src_valid_reg (no outstanding item in source)
  // When src_valid_reg is set, we need to wait for destination to consume it.
  // Destination sees the data via synchronization.

  // Let's use a standard async FIFO-like approach with single entry:
  // Source domain:
  //   - Latch data on valid && ready
  //   - Assert a "data available" signal that crosses to destination
  // Destination domain:
  //   - Synchronize the "data available" signal
  //   - When synchronized valid is high and dst_ready_i is high, complete transfer
  //   - Deassert the "data available" signal (crosses back to source)
  // Source domain:
  //   - Synchronize the deassertion to clear src_valid_reg

  // Signals crossing from source to destination:
  //   src_data_cross: the data (multi-bit, needs to be stable)
  //   src_valid_cross: valid indicator

  // Signals crossing from destination to source:
  //   dst_ack_cross: acknowledgment that destination consumed the data

  // Source domain registers
  reg src_valid_reg;
  reg [WIDTH-1:0] src_data_reg;

  // Destination domain registers
  reg dst_valid_reg;
  reg [WIDTH-1:0] dst_data_reg;

  // Synchronization chains
  // src_valid_cross: 2-flop sync in destination domain
  reg [1:0] src_valid_sync;
  // dst_ack_cross: 2-flop sync in source domain
  reg [1:0] dst_ack_sync;

  // Source domain logic
  wire src_ready_o = ~src_valid_reg;

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_valid_reg <= 1'b0;
      src_data_reg  <= {WIDTH{1'b0}};
    end else begin
      if (src_valid_i && src_ready_o) begin
        src_valid_reg <= 1'b1;
        src_data_reg  <= src_data_i;
      end else if (dst_ack_sync[1]) begin
        src_valid_reg <= 1'b0;
      end
    end
  end

  // Destination domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid_reg <= 1'b0;
      dst_data_reg  <= {WIDTH{1'b0}};
    end else begin
      if (src_valid_sync[1] && !dst_valid_reg) begin
        dst_valid_reg <= 1'b1;
        dst_data_reg  <= src_data_reg;
      end else if (dst_valid_reg && dst_ready_i) begin
        dst_valid_reg <= 1'b0;
      end
    end
  end

  // Synchronization of src_valid to destination domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_valid_sync <= 2'b00;
    end else begin
      src_valid_sync <= {src_valid_sync[0], src_valid_reg};
    end
  end

  // Synchronization of dst_ack to source domain
  // dst_ack is asserted when destination has consumed the data
  wire dst_ack = dst_valid_reg && dst_ready_i;

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_ack_sync <= 2'b00;
    end else begin
      dst_ack_sync <= {dst_ack_sync[0], dst_ack};
    end
  end

  // Outputs
  assign src_ready_o = ~src_valid_reg;
  assign dst_data_o  = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

endmodule
