module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain state
  reg src_valid_reg;
  reg [WIDTH-1:0] src_data_reg;

  // Destination domain state
  reg dst_valid_reg;
  reg [WIDTH-1:0] dst_data_reg;

  // CDC synchronizers for valid signal
  // src_valid_reg -> dst domain
  reg [1:0] src_valid_sync;
  // dst_valid_reg -> src domain
  reg [1:0] dst_valid_sync;

  // Synchronize src_valid_reg to dst domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni)
      src_valid_sync <= 2'b00;
    else
      src_valid_sync <= {src_valid_sync[0], src_valid_reg};
  end

  // Synchronize dst_valid_reg to src domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni)
      dst_valid_sync <= 2'b00;
    else
      dst_valid_sync <= {dst_valid_sync[0], dst_valid_reg};
  end

  // Source domain logic
  // src_ready_o is high when we can accept a new item
  // We can accept if src_valid_reg is low (no pending item) OR if the destination has consumed it
  // Actually, we can only accept a new item if there is no pending item in the source register
  // The source register holds an item until the destination acknowledges it (dst_valid_reg goes low)
  // So src_ready_o = !src_valid_reg

  assign src_ready_o = !src_valid_reg;

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_valid_reg <= 1'b0;
      src_data_reg  <= {WIDTH{1'b0}};
    end else begin
      if (src_valid_i && src_ready_o) begin
        src_valid_reg <= 1'b1;
        src_data_reg  <= src_data_i;
      end else if (dst_valid_sync[1] == 1'b0) begin
        // Destination has consumed the item (dst_valid_reg is low)
        // Clear the source valid
        src_valid_reg <= 1'b0;
      end
    end
  end

  // Destination domain logic
  // dst_valid_o is high when there is a pending item
  assign dst_valid_o = dst_valid_reg;
  assign dst_data_o  = dst_data_reg;

  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid_reg <= 1'b0;
      dst_data_reg  <= {WIDTH{1'b0}};
    end else begin
      if (src_valid_sync[1] == 1'b1 && !dst_valid_reg) begin
        // New item available from source, and no pending item
        dst_valid_reg <= 1'b1;
        dst_data_reg  <= src_data_reg;
      end else if (dst_valid_reg && dst_ready_i) begin
        // Item is being consumed
        dst_valid_reg <= 1'b0;
      end
    end
  end

endmodule
