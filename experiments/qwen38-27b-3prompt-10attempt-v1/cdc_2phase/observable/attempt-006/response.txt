module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source side: capture data and generate request
  reg [WIDTH-1:0] src_data_reg;
  reg src_req;
  reg src_ack;

  // Destination side: capture data and generate ack
  reg [WIDTH-1:0] dst_data_reg;
  reg dst_ack;
  reg dst_req;

  // Synchronizers for request (src -> dst)
  reg req_sync1;
  reg req_sync2;

  // Synchronizers for ack (dst -> src)
  reg ack_sync1;
  reg ack_sync2;

  // Source clock domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_reg <= {WIDTH{1'b0}};
      src_req      <= 1'b0;
      src_ack      <= 1'b0;
    end else begin
      // If not currently holding a pending request, accept new data
      if (!src_req) begin
        if (src_valid_i) begin
          src_data_reg <= src_data_i;
          src_req      <= 1'b1;
        end
      end else begin
        // Waiting for ack
        if (src_ack) begin
          src_req <= 1'b0;
          src_ack <= 1'b0;
        end
      end
    end
  end

  // Source ready: ready when not holding a pending request
  assign src_ready_o = ~src_req;

  // Synchronize request to destination clock domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      req_sync1 <= 1'b0;
      req_sync2 <= 1'b0;
    end else begin
      req_sync1 <= src_req;
      req_sync2 <= req_sync1;
    end
  end

  // Destination clock domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg <= {WIDTH{1'b0}};
      dst_ack      <= 1'b0;
      dst_req      <= 1'b0;
    end else begin
      // If not currently holding a pending ack, accept new request
      if (!dst_ack) begin
        if (req_sync2) begin
          dst_data_reg <= src_data_reg;
          dst_ack      <= 1'b1;
        end
      end else begin
        // Waiting for dst to consume the data
        if (dst_valid_o && dst_ready_i) begin
          dst_ack <= 1'b0;
        end
      end
    end
  end

  // Destination valid: valid when holding data and not yet consumed
  assign dst_valid_o = dst_ack;
  assign dst_data_o  = dst_data_reg;

  // Synchronize ack to source clock domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      ack_sync1 <= 1'b0;
      ack_sync2 <= 1'b0;
    end else begin
      ack_sync1 <= dst_ack;
      ack_sync2 <= ack_sync1;
    end
  end

  // Update src_ack from synchronized ack
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ack <= 1'b0;
    end else begin
      src_ack <= ack_sync2;
    end
  end

endmodule
