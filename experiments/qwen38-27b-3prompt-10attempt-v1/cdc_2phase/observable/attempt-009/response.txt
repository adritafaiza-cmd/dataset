module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source side: capture data and generate request pulse
  reg [WIDTH-1:0] src_data_reg;
  reg             src_req;
  wire            src_ready;

  // Destination side: receive request, generate response pulse
  reg [WIDTH-1:0] dst_data_reg;
  reg             dst_ack;
  wire            dst_valid;

  // Synchronizers for request (src -> dst)
  reg [1:0] req_sync;
  wire      req_synced = req_sync[1];

  // Synchronizers for response (dst -> src)
  reg [1:0] ack_sync;
  wire      ack_synced = ack_sync[1];

  // Source clock domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_reg <= {WIDTH{1'b0}};
      src_req      <= 1'b0;
    end else begin
      if (src_valid_i && src_ready) begin
        src_data_reg <= src_data_i;
        src_req      <= 1'b1;
      end else if (ack_synced) begin
        src_req <= 1'b0;
      end
    end
  end

  assign src_ready = !src_req;

  // Synchronize request from src to dst
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      req_sync <= 2'b00;
    end else begin
      req_sync <= {req_sync[0], src_req};
    end
  end

  // Destination clock domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg <= {WIDTH{1'b0}};
      dst_ack      <= 1'b0;
    end else begin
      if (req_synced && !dst_ack) begin
        dst_data_reg <= src_data_reg;
        dst_ack      <= 1'b1;
      end else if (dst_ack && dst_ready_i) begin
        dst_ack <= 1'b0;
      end
    end
  end

  assign dst_valid = dst_ack;
  assign dst_data_o = dst_data_reg;

  // Synchronize response from dst to src
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      ack_sync <= 2'b00;
    end else begin
      ack_sync <= {ack_sync[0], dst_ack};
    end
  end

endmodule
