```verilog
module uart_top (
    wb_clk_i,
    wb_rst_i, wb_adr_i, wb_dat_i, wb_dat_o, wb_we_i, wb_stb_i, wb_cyc_i, wb_ack_o, wb_sel_i,
    int_o,
    stx_pad_o, srx_pad_i,
    rts_pad_o, cts_pad_i, dtr_pad_o, dsr_pad_i, ri_pad_i, dcd_pad_i
);

    // Wishbone signals
    input  wire wb_clk_i;
    input  wire wb_rst_i;
    input  wire [7:0] wb_adr_i;
    input  wire [7:0] wb_dat_i;
    output reg  [7:0] wb_dat_o;
    input  wire wb_we_i;
    input  wire wb_stb_i;
    input  wire wb_cyc_i;
    output reg  wb_ack_o;
    input  wire [3:0] wb_sel_i;

    // UART signals
    output reg  int_o;
    output reg  stx_pad_o;
    input  wire srx_pad_i;
    output reg  rts_pad_o;
    input  wire cts_pad_i;
    output reg  dtr_pad_o;
    input  wire dsr_pad_i;
    input  wire ri_pad_i;
    input  wire dcd_pad_i;

    // Internal signals
    reg [7:0] rbr_thr;       // Receive Buffer Register / Transmit Holding Register
    reg [7:0] ier;           // Interrupt Enable Register
    reg [7:0] iir;           // Interrupt Identification Register
    reg [7:0] lcr;           // Line Control Register
    reg [7:0] mcr;           // Modem Control Register
    reg [7:0] lsr;           // Line Status Register
    reg [7:0] msr;           // Modem Status Register
    reg [7:0] scr;           // Scratch Register
    reg [7:0] fcr;           // FIFO Control Register
    reg [7:0] lcr_dll;       // Divisor Latch Low
    reg [7:0] lcr_dlh;       // Divisor Latch High
    reg [7:0] iir_reg;       // Internal IIR register

    // FIFO signals
    reg [7:0] rx_fifo [0:15];
    reg [3:0] rx_fifo_wr_ptr;
    reg [3:0] rx_fifo_rd_ptr;
    reg [3:0] rx_fifo_count;
    reg [7:0] tx_fifo [0:15];
    reg [3:0] tx_fifo_wr_ptr;
    reg [3:0] tx_fifo_rd_ptr;
    reg [3:0] tx_fifo_count;

    // Baud rate generator
    reg [15:0] baud_divisor;
    reg [15:0] baud_counter;
    reg baud_tick;

    // Transmitter state machine
    reg tx_active;
    reg [3:0] tx_bit_count;
    reg [7:0] tx_shift_reg;
    reg tx_start_sent;

    // Receiver state machine
    reg rx_active;
    reg [3:0] rx_bit_count;
    reg [7:0] rx_shift_reg;
    reg rx_start_detected;
    reg [1:0] rx_sync;
    reg rx_sync_valid;

    // Modem control
    reg [3:0] modem_status;
    reg [3:0] modem_status_prev;

    // Interrupt generation
    reg int_pending;

    // Wishbone address decoding
    wire [2:0] wb_addr = wb_adr_i[2:0];
    wire wb_write = wb_we_i & wb_stb_i & wb_cyc_i;
    wire wb_read = ~wb_we_i & wb_stb_i & wb_cyc_i;

    // Baud rate calculation
    wire [15:0] baud_div = (lcr[7]) ? {lcr_dlh, lcr_dll} : 16'd1;

    // FIFO enable
    wire fifo_enabled = fcr[0];

    // DLAB bit
    wire dlab = lcr[7];

    // Interrupt enable bits
    wire ier_rbr = ier[0];
    wire ier_thr = ier[1];
    wire ier_lsr = ier[2];
    wire ier_ms  = ier[3];

    // Line status bits
    wire lsr_dr = (rx_fifo_count > 0);
    wire lsr_oe = 1'b0; // Overrun error (simplified)
    wire lsr_pe = 1'b0; // Parity error (simplified)
    wire lsr_fe = 1'b0; // Framing error (simplified)
    wire lsr_bi = 1'b0; // Break interrupt (simplified)
    wire lsr_thre = (tx_fifo_count == 0);
    wire lsr_temt = (tx_fifo_count == 0) & ~tx_active;
    wire lsr_err = lsr_oe | lsr_pe | lsr_fe | lsr_bi;

    // Modem status bits
    wire msr_dcd = dcd_pad_i;
    wire msr_ri  = ri_pad_i;
    wire msr_ds  = dsr_pad_i;
    wire msr_cts = cts_pad_i;

    // Modem status change detection
    wire msr_dcd_chg = msr_dcd ^ modem_status_prev[0];
    wire msr_ri_chg  = msr_ri  ^ modem_status_prev[1];
    wire msr_ds_chg  = msr_ds  ^ modem_status_prev[2];
    wire msr_cts_chg = msr_cts ^ modem_status_prev[3];
    wire msr_chg = msr_dcd_chg | msr_ri_chg | msr_ds_chg | msr_cts_chg;

    // Interrupt sources
    wire int_rbr = ier_rbr & lsr_dr;
    wire int_thr = ier_thr & lsr_thre;
    wire int_lsr = ier_lsr & lsr_err;
    wire int_ms  = ier_ms  & msr_chg;

    // IIR value calculation
    function [7:0] calc_iir;
        input [7:0] lsr_in;
        input [3:0] msr_in;
        input [3:0] msr_chg_in;
        begin
            if (lsr_in[5]) // THRE
                calc_iir = 8'h02;
            else if (lsr_in[0]) // DR
                calc_iir = 8'h06;
            else if (lsr_in[6] | lsr_in[4] | lsr_in[3] | lsr_in[2]) // Error
                calc_iir = 8'h04;
            else if (msr_chg_in[0] | msr_chg_in[1] | msr_chg_in[2] | msr_chg_in[3]) // Modem status change
                calc_iir = 8'h00;
            else
                calc_iir = 8'h01; // No interrupt pending
        end
    endfunction

    // Wishbone read data
    reg [7:0] wb_read_data;

    always @(*) begin
        case (wb_addr)
            3'd0: begin
                if (dlab)
                    wb_read_data = lcr_dll;
                else
                    wb_read_data = rbr_thr;
            end
            3'd1: begin
                if (dlab)
                    wb_read_data = lcr_dlh;
                else
                    wb_read_data = ier;
            end
            3'd2: wb_read_data = iir_reg;
            3'd3: wb_read_data = lcr;
            3'd4: wb_read_data = mcr;
            3'd5: wb_read_data = lsr;
            3'd6: wb_read_data = msr;
            3'd7: wb_read_data = scr;
            default: wb_read_data = 8'h00;
        endcase
    end

    // Wishbone interface
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            wb_ack_o <= 1'b0;
            wb_dat_o <= 8'h00;
        end else begin
            wb_ack_o <= 1'b0;
            if (wb_stb_i & wb_cyc_i) begin
                wb_ack_o <= 1'b1;
                if (wb_read)
                    wb_dat_o <= wb_read_data;
            end
        end
    end

    // Register writes
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            rbr_thr <= 8'h00;
            ier <= 8'h00;
            lcr <= 8'h03; // 8N1
            mcr <= 8'h00;
            scr <= 8'h00;
            fcr <= 8'h00;
            lcr_dll <= 8'h00;
            lcr_dlh <= 8'h00;
            rx_fifo_wr_ptr <= 4'd0;
            rx_fifo_rd_ptr <= 4'd0;
            rx_fifo_count <= 4'd0;
            tx_fifo_wr_ptr <= 4'd0;
            tx_fifo_rd_ptr <= 4'd0;
            tx_fifo_count <= 4'd0;
            tx_active <= 1'b0;
            tx_bit_count <= 4'd0;
            tx_shift_reg <= 8'h00;
            tx_start_sent <= 1'b0;
            rx_active <= 1'b0;
            rx_bit_count <= 4'd0;
            rx_shift_reg <= 8'h00;
            rx_start_detected <= 1'b0;
            rx_sync <= 2'b00;
            rx_sync_valid <= 1'b0;
            modem_status <= 4'b0000;
            modem_status_prev <= 4'b0000;
            int_pending <= 1'b0;
            baud_counter <= 16'd0;
            baud_tick <= 1'b0;
            stx_pad_o <= 1'b1;
            rts_pad_o <= 1'b0;
            dtr_pad_o <= 1'b0;
        end else begin
            // Baud rate generator
            baud_tick <= 1'b0;
            if (baud_div == 16'd0) begin
                baud_counter <= 16'd0;
                baud_tick <= 1'b1;
            end else begin
                if (baud_counter >= baud_div - 1) begin
                    baud_counter <= 16'd0;
                    baud_tick <= 1'b1;
                end else begin
                    baud_counter <= baud_counter + 1;
                end
            end

            // Transmitter
            if (tx_active) begin
                if (baud_tick) begin
                    if (tx_bit_count == 4'd0) begin
                        // Start bit
                        stx_pad_o <= 1'b0;
                        tx_bit_count <= 4'd1;
                        tx_start_sent <= 1'b1;
                    end else if (tx_bit_count <= 4'd8) begin
                        // Data bits
                        stx_pad_o <= tx_shift_reg[tx_bit_count - 1];
                        tx_bit_count <= tx_bit_count + 1;
                    end else if (tx_bit_count == 4'd9) begin
                        // Stop bit
                        stx_pad_o <= 1'b1;
                        tx_bit_count <= 4'd10;
                    end else begin
                        // Done
                        tx_active <= 1'b0;
                        tx_bit_count <= 4'd0;
                        tx_start_sent <= 1'b0;
                        stx_pad_o <= 1'b1;
                        // Load next from FIFO
                        if (tx_fifo_count > 0) begin
                            tx_shift_reg <= tx_fifo[tx_fifo_rd_ptr];
                            tx_fifo_rd_ptr <= tx_fifo_rd_ptr + 1;
                            tx_fifo_count <= tx_fifo_count - 1;
                            tx_active <= 1'b1;
                            tx_bit_count <= 4'd0;
                        end
                    end
                end
            end else begin
                stx_pad_o <= 1'b1;
                // Load from FIFO if available
                if (tx_fifo_count > 0) begin
                    tx_shift_reg <= tx_fifo[tx_fifo_rd_ptr];
                    tx_fifo_rd_ptr <= tx_fifo_rd_ptr + 1;
                    tx_fifo_count <= tx_fifo_count - 1;
                    tx_active <= 1'b1;
                    tx_bit_count <= 4'd0;
                    tx_start_sent <= 1'b0;
                end
            end

            // Receiver
            rx_sync <= {rx_sync[0], srx_pad_i};
            if (rx_sync == 2'b01) begin
                rx_sync_valid <= 1'b1;
            end else begin
                rx_sync_valid <= 1'b0;
            end

            if (rx_active) begin
                if (baud_tick) begin
                    if (rx_bit_count == 4'd0) begin
                        // Start bit
                        rx_shift_reg <= {srx_pad_i, rx_shift_reg[7:1]};
                        rx_bit_count <= 4'd1;
                    end else if (rx_bit_count <= 4'd8) begin
                        // Data bits
                        rx_shift_reg <= {srx_pad_i, rx_shift_reg[7:1]};
                        rx_bit_count <= rx_bit_count + 1;
                    end else if (rx_bit_count == 4'd9) begin
                        // Stop bit
                        rx_bit_count <= 4'd10;
                    end else begin
                        // Done
                        rx_active <= 1'b0;
                        rx_bit_count <= 4'd0;
                        // Store in FIFO
                        if (fifo_enabled) begin
                            rx_fifo[rx_fifo_wr_ptr] <= rx_shift_reg;
                            rx_fifo_wr_ptr <= rx_fifo_wr_ptr + 1;
                            rx_fifo_count <= rx_fifo_count + 1;
                        end else begin
                            rbr_thr <= rx_shift_reg;
                        end
                    end
                end
            end else begin
                // Detect start bit
                if (srx_pad_i == 1'b0) begin
                    rx_active <= 1'b1;
                    rx_bit_count <= 4'd0;
                    rx_shift_reg <= 8'h00;
                end
            end

            // Modem status
            modem_status <= {msr_cts, msr_ds, msr_ri, msr_dcd};
            modem_status_prev <= modem_status;

            // Modem control outputs
            rts_pad_o <= mcr[1];
            dtr_pad_o <= mcr[0];

            // Wishbone writes
            if (wb_write) begin
                case (wb_addr)
                    3'd0: begin
                        if (dlab)
                            lcr_dll <= wb_dat_i;
                        else
                            rbr_thr <= wb_dat_i;
                        // If writing to THR, load into TX FIFO
                        if (~dlab) begin
                            if (fifo_enabled) begin
                                tx_fifo[tx_fifo_wr_ptr] <= wb_dat_i;
                                tx_fifo_wr_ptr <= tx_fifo_wr_ptr + 1;
                                tx_fifo_count <= tx_fifo_count + 1;
                            end else begin
                                tx_shift_reg <= wb_dat_i;
                                tx_active <= 1'b1;
                                tx_bit_count <= 4'd0;
                                tx_start_sent <= 1'b0;
                            end
                        end
                    end
                    3'd1: begin
                        if (dlab)
                            lcr_dlh <= wb_dat_i;
                        else
                            ier <= wb_dat_i;
                    end
                    3'd2: begin
                        // IIR read clears interrupt
                        // No write action needed
                    end
                    3'd3: lcr <= wb_dat_i;
                    3'd4: mcr <= wb_dat_i;
                    3'd5: begin
                        // LSR read clears some bits
                        // No write action needed
                    end
                    3'd6: begin
                        // MSR read clears some bits
                        // No write action needed
                    end
                    3'd7: scr <= wb_dat_i;
                    default: ;
                endcase
            end

            // FIFO control
            if (wb_write & (wb_addr == 3'd2) & dlab) begin
                fcr <= wb_dat_i;
                if (wb_dat_i[2]) begin
                    // Clear RX FIFO
                    rx_fifo_wr_ptr <= 4'd0;
                    rx_fifo_rd_ptr <= 4'd0;
                    rx_fifo_count <= 4'd0;
                end
                if (wb_dat_i[1]) begin
                    // Clear TX FIFO
                    tx_fifo_wr_ptr <= 4'd0;
                    tx_fifo_rd_ptr <= 4'd0;
                    tx_fifo_count <= 4'd0;
                end
            end

            // Interrupt generation
            int_pending <= int_rbr | int_thr | int_lsr | int_ms;
            int_o <= int_pending & (ier_rbr | ier_thr | ier_lsr | ier_ms);

            // IIR register
            iir_reg <= calc_iir(lsr, msr, {msr_cts_chg, msr_ds_chg, msr_ri_chg, msr_dcd_chg});
        end
    end

endmodule
