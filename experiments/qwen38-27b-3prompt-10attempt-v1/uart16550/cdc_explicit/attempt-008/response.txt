// UART TX module
module uart_tx #(
    parameter BAUD_DIVISOR = 16
) (
    input wire clk,
    input wire rst,
    input wire [7:0] data,
    input wire enable,
    output reg done,
    output reg tx_pad,
    input wire [15:0] baud_divisor,
    input wire [1:0] data_bits,
    input wire parity_enable,
    input wire parity_type,
    input wire [1:0] stop_bits,
    input wire break_ctl
);

    localparam IDLE = 3'd0;
    localparam START = 3'd1;
    localparam DATA = 3'd2;
    localparam PARITY = 3'd3;
    localparam STOP = 3'd4;

    reg [2:0] state;
    reg [15:0] baud_counter;
    reg [7:0] shift_reg;
    reg [3:0] bit_count;
    reg parity_bit;
    reg [15:0] baud_div;

    assign baud_div = baud_divisor == 0 ? 16'd1 : baud_divisor;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= IDLE;
            baud_counter <= 16'd0;
            shift_reg <= 8'd0;
            bit_count <= 4'd0;
            parity_bit <= 1'b0;
            done <= 1'b0;
            tx_pad <= 1'b1;
        end else begin
            case (state)
                IDLE: begin
                    tx_pad <= 1'b1;
                    done <= 1'b0;
                    if (enable) begin
                        state <= START;
                        baud_counter <= 16'd0;
                        shift_reg <= data;
                        bit_count <= 4'd0;
                    end
                end
                START: begin
                    tx_pad <= 1'b0;
                    if (baud_counter == baud_div - 1) begin
                        baud_counter <= 16'd0;
                        state <= DATA;
                        bit_count <= 4'd0;
                    end else begin
                        baud_counter <= baud_counter + 1;
                    end
                end
                DATA: begin
                    if (baud_counter == baud_div - 1) begin
                        baud_counter <= 16'd0;
                        tx_pad <= shift_reg[0];
                        shift_reg <= {1'b0, shift_reg[7:1]};
                        bit_count <= bit_count + 1;
                        if (bit_count == data_bits - 1) begin
                            state <= parity_enable ? PARITY : STOP;
                        end
                    end else begin
                        baud_counter <= baud_counter + 1;
                    end
                end
                PARITY: begin
                    if (baud_counter == baud_div - 1) begin
                        baud_counter <= 16'd0;
                        tx_pad <= parity_bit;
                        state <= STOP;
                    end else begin
                        baud_counter <= baud_counter + 1;
                    end
                end
                STOP: begin
                    if (baud_counter == baud_div - 1) begin
                        baud_counter <= 16'd0;
                        tx_pad <= 1'b1;
                        done <= 1'b1;
                        state <= IDLE;
                    end else begin
                        baud_counter <= baud_counter + 1;
                    end
                end
                default: state <= IDLE;
            end
        end
    end

    always @(*) begin
        parity_bit = 1'b0;
        if (parity_enable) begin
            case (parity_type)
                1'b0: parity_bit = ^data; // Even
                1'b1: parity_bit = ~(^data); // Odd
            endcase
        end
    end

endmodule

// UART RX module
module uart_rx #(
    parameter BAUD_DIVISOR = 16
) (
    input wire clk,
    input wire rst,
    input wire rx_pad,
    output reg [7:0] data,
    output reg valid,
    input wire [15:0] baud_divisor,
    input wire [1:0] data_bits,
    input wire parity_enable,
    input wire parity_type,
    input wire [1:0] stop_bits,
    output reg fe,
    output reg pe,
    output reg oe,
    output reg bi
);

    localparam IDLE = 3'd0;
    localparam START = 3'd1;
    localparam DATA = 3'd2;
    localparam PARITY = 3'd3;
    localparam STOP = 3'd4;

    reg [2:0] state;
    reg [15:0] baud_counter;
    reg [7:0] shift_reg;
    reg [3:0] bit_count;
    reg rx_pad_sync1;
    reg rx_pad_sync2;
    reg rx_pad_sync3;
    reg [15:0] baud_div;

    assign baud_div = baud_divisor == 0 ? 16'd1 : baud_divisor;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rx_pad_sync1 <= 1'b1;
            rx_pad_sync2 <= 1'b1;
            rx_pad_sync3 <= 1'b1;
        end else begin
            rx_pad_sync1 <= rx_pad;
            rx_pad_sync2 <= rx_pad_sync1;
            rx_pad_sync3 <= rx_pad_sync2;
        end
    end

    wire rx_pad_stable = rx_pad_sync2;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= IDLE;
            baud_counter <= 16'd0;
            shift_reg <= 8'd0;
            bit_count <= 4'd0;
            data <= 8'd0;
            valid <= 1'b0;
            fe <= 1'b0;
            pe <= 1'b0;
            oe <= 1'b0;
            bi <= 1'b0;
        end else begin
            valid <= 1'b0;
            case (state)
                IDLE: begin
                    if (rx_pad_stable == 1'b0) begin
                        state <= START;
                        baud_counter <= 16'd0;
                    end
                end
                START: begin
                    if (baud_counter == baud_div/2 - 1) begin
                        baud_counter <= 16'd0;
                        if (rx_pad_stable == 1'b0) begin
                            state <= DATA;
                            bit_count <= 4'd0;
                            shift_reg <= 8'd0;
                        end else begin
                            state <= IDLE;
                        end
                    end else begin
                        baud_counter <= baud_counter + 1;
                    end
                end
                DATA: begin
                    if (baud_counter == baud_div - 1) begin
                        baud_counter <= 16'd0;
                        shift_reg <= {rx_pad_stable, shift_reg[7:1]};
                        bit_count <= bit_count + 1;
                        if (bit_count == data_bits - 1) begin
                            state <= parity_enable ? PARITY : STOP;
                        end
                    end else begin
                        baud_counter <= baud_counter + 1;
                    end
                end
                PARITY: begin
                    if (baud_counter == baud_div - 1) begin
                        baud_counter <= 16'd0;
                        state <= STOP;
                    end else begin
                        baud_counter <= baud_counter + 1;
                    end
                end
                STOP: begin
                    if (baud_counter == baud_div - 1) begin
                        baud_counter <= 16'd0;
                        data <= shift_reg;
                        valid <= 1'b1;
                        state <= IDLE;
                    end else begin
                        baud_counter <= baud_counter + 1;
                    end
                end
                default: state <= IDLE;
            end
        end
    end

endmodule

// FIFO 8x16
module fifo_8x16 (
    input wire clk,
    input wire rst,
    input wire wr_en,
    input wire rd_en,
    input wire reset,
    input wire [7:0] data_in,
    output wire [7:0] data_out,
    output wire empty,
    output wire full,
    output wire [3:0] count
);

    reg [7:0] mem [0:15];
    reg [4:0] wr_ptr;
    reg [4:0] rd_ptr;
    reg [3:0] cnt;

    assign data_out = mem[rd_ptr[3:0]];
    assign empty = (cnt == 4'd0);
    assign full = (cnt == 4'd16);
    assign count = cnt;

    always @(posedge clk or posedge rst) begin
        if (rst || reset) begin
            wr_ptr <= 5'd0;
            rd_ptr <= 5'd0;
            cnt <= 4'd0;
        end else begin
            if (wr_en && !full) begin
                mem[wr_ptr[3:0]] <= data_in;
                wr_ptr <= wr_ptr + 1;
            end
            if (rd_en && !empty) begin
                rd_ptr <= rd_ptr + 1;
            end
            if (wr_en && !full && rd_en && !empty) begin
                cnt <= cnt;
            end else if (wr_en && !full) begin
                cnt <= cnt + 1;
            end else if (rd_en && !empty) begin
                cnt <= cnt - 1;
            end
        end
    end

endmodule

// Register file
module reg_file (
    input wire clk,
    input wire rst,
    input wire we,
    input wire [2:0] addr,
    input wire [7:0] data_in,
    output reg [7:0] rbr_thr,
    output reg [7:0] iir_fcr,
    output reg [7:0] lcr,
    output reg [7:0] mcr,
    output reg [7:0] lsr,
    output reg [7:0] msr,
    output reg [7:0] scr,
    output reg [7:0] dll,
    output reg [7:0] dlh
);

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rbr_thr <= 8'd0;
            iir_fcr <= 8'd1;
            lcr <= 8'd0;
            mcr <= 8'd0;
            lsr <= 8'd0;
            msr <= 8'd0;
            scr <= 8'd0;
            dll <= 8'd0;
            dlh <= 8'd0;
        end else if (we) begin
            case (addr)
                3'd0: begin
                    if (lcr[5]) dll <= data_in;
                    else rbr_thr <= data_in;
                end
                3'd1: begin
                    if (lcr[5]) dlh <= data_in;
                    else iir_fcr <= data_in;
                end
                3'd2: lcr <= data_in;
                3'd3: mcr <= data_in;
                3'd6: scr <= data_in;
                default: ;
            endcase
        end
    end

endmodule
