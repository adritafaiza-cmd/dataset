module uart_top (
    input  wire wb_clk_i,
    input  wire wb_rst_i,
    input  wire [7:0] wb_adr_i,
    input  wire [7:0] wb_dat_i,
    output wire [7:0] wb_dat_o,
    input  wire wb_we_i,
    input  wire wb_stb_i,
    input  wire wb_cyc_i,
    output wire wb_ack_o,
    input  wire [3:0] wb_sel_i,
    output wire int_o,
    output wire stx_pad_o,

    output wire rts_pad_o,
    input  wire cts_pad_i,
    output wire dtr_pad_o,

);

    wire wb_valid = wb_cyc_i & wb_stb_i;
    wire wb_write = wb_valid & wb_we_i;
    wire wb_read  = wb_valid & ~wb_we_i;

    wire lsr_bi = reg_lsr[8];

    wire msr_ri_d = reg_msr[5];
    wire msr_ds_d = reg_msr[6];
    wire msr_cts_d = reg_msr[7];

        .reset(fifo_reset_rx),
        .push(rx_valid),

        .break_in(rx_break),
        .pop(fifo_rx_valid),
        .data_out(fifo_rx_data),
        .error_out(),
        .break_out(),
        .count(fifo_rx_count),
        .overflow(fifo_rx_overflow),
        .empty(rx_empty)

    tx_fifo u_tx_fifo (
        .clk(wb_clk_i),
        .rst(wb_rst_i),
        .reset(fifo_reset_tx),
        .push(fifo_tx_valid),
        .data_in(fifo_tx_data),
        .pop(tx_start),
        .data_out(tx_data),
        .count(fifo_tx_count),
        .empty(fifo_tx_empty)
    );

    // Register file
    reg_file u_reg_file (

        .rst(wb_rst_i),
        .wb_valid(wb_valid),
        .wb_write(wb_write),
        .wb_read(wb_read),
        .wb_adr(wb_adr_i),
        .wb_dat_i(wb_dat_i),
        .wb_dat_o(wb_dat_o),
        .wb_ack_o(wb_ack_o),
        .dlab(dlab),
        .reg_rbr(reg_rbr),

        .reg_mcr(reg_mcr),
        .reg_lsr(reg_lsr),
        .reg_msr(reg_msr),
        .reg_scr(reg_scr),
        .reg_dll(reg_dll),
        .reg_dlm(reg_dlm),
        .fifo_rx_count(fifo_rx_count),
        .fifo_tx_count(fifo_tx_count),
        .fifo_rx_overflow(fifo_rx_overflow),
        .fifo_tx_empty(fifo_tx_empty),
        .rx_valid(rx_valid),
        .rx_data(rx_data),
        .rx_error(rx_error),
        .rx_break(rx_break),
        .tx_done(tx_done),
        .tx_empty(tx_empty),
        .msr_dcd(msr_dcd),
        .msr_ri(msr_ri),
        .msr_ds(msr_ds),
        .msr_cts(msr_cts),
        .msr_dcd_d(msr_dcd_d),
        .msr_ri_d(msr_ri_d),
        .msr_ds_d(msr_ds_d),
        .msr_cts_d(msr_cts_d),
        .dtr(dtr),
        .rts(rts),
        .out2(out2),
        .out1(out1),
        .loopback(loopback),
        .parity_enable(parity_enable),
        .stop_bits(stop_bits),
        .word_length(word_length),
        .break_int(break_int),
        .send_break(send_break),
        .fifo_enabled(fifo_enabled),
        .fifo_reset_rx(fifo_reset_rx),
        .fifo_reset_tx(fifo_reset_tx),
        .fifo_rx_trig(fifo_rx_trig),
        .fifo_tx_trig(fifo_tx_trig),
        .ier_rlsi(ier_rlsi),
        .ier_thre(ier_thre),
        .ier_rx(ier_rx),
        .ier_tx(ier_tx),

        .lsr_bi(lsr_bi),
        .lsr_thre(lsr_thre),
        .lsr_temt(lsr_temt)

    assign rts_pad_o = rts;
    assign dtr_pad_o = dtr;

    // MSR edge detection
    reg msr_dcd_r, msr_ri_r, msr_ds_r, msr_cts_r;
    reg msr_dcd_d_r, msr_ri_d_r, msr_ds_d_r, msr_cts_d_r;

    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            msr_dcd_r <= 1'b0;
            msr_ri_r <= 1'b0;
            msr_ds_r <= 1'b0;
            msr_cts_r <= 1'b0;
            msr_dcd_d_r <= 1'b0;
            msr_ri_d_r <= 1'b0;
            msr_ds_d_r <= 1'b0;
            msr_cts_d_r <= 1'b0;
        end else begin
            msr_dcd_r <= dcd_pad_i;
            msr_ri_r <= ri_pad_i;
            msr_ds_r <= dsr_pad_i;
            msr_cts_r <= cts_pad_i;

            msr_ri_d_r <= (ri_pad_i != msr_ri_r);

    assign msr_ds = dsr_pad_i;
    assign msr_cts = cts_pad_i;

    assign msr_ds_d = msr_ds_d_r;
    assign msr_cts_d = msr_cts_d_r;

module uart_core #(
    parameter BAUD_DIVISOR = 16'd1
) (
    input  wire clk,
    input  wire rst,
    input  wire [15:0] baud_divisor,
    input  wire tx_start,
    input  wire [7:0] tx_data,
    output wire tx_done,
    output wire tx_empty,
    output wire rx_valid,
    output wire [7:0] rx_data,
    output wire rx_error,
    output wire rx_break,
    output wire rx_empty,
    output wire stx_pad_o,
    input  wire srx_pad_i,
    input  wire loopback
);

    // Baud rate generator
    reg [15:0] baud_counter;
    wire baud_tick;
    
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            baud_counter <= 16'd0;
        end else if (baud_counter == baud_divisor - 1) begin
            baud_counter <= 16'd0;
        end else begin
            baud_counter <= baud_counter + 1'b1;
        end
    end
    
    assign baud_tick = (baud_counter == baud_divisor - 1);

    // TX state machine
    localparam TX_IDLE = 3'd0;
    localparam TX_START = 3'd1;
    localparam TX_DATA = 3'd2;
    localparam TX_PARITY = 3'd3;
    localparam TX_STOP1 = 3'd4;
    localparam TX_STOP2 = 3'd5;

    reg [2:0] tx_state;
    reg [3:0] tx_bit_count;
    reg [7:0] tx_shift_reg;
    reg tx_parity_bit;
    reg tx_stop_bit;
    reg tx_done_r;
    reg tx_empty_r;

    wire [7:0] tx_data_eff = loopback ? tx_data : tx_data;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            tx_state <= TX_IDLE;
            tx_bit_count <= 4'd0;
            tx_shift_reg <= 8'd0;
            tx_parity_bit <= 1'b0;
            tx_stop_bit <= 1'b0;
            tx_done_r <= 1'b0;
            tx_empty_r <= 1'b1;
        end else begin
            tx_done_r <= 1'b0;
            
            case (tx_state)
                TX_IDLE: begin
                    tx_empty_r <= 1'b1;
                    if (tx_start) begin
                        tx_state <= TX_START;
                        tx_shift_reg <= tx_data_eff;
                        tx_empty_r <= 1'b0;
                    end
                end
                TX_START: begin
                    if (baud_tick) begin
                        tx_state <= TX_DATA;
                        tx_bit_count <= 4'd0;
                    end
                end
                TX_DATA: begin
                    if (baud_tick) begin
                        if (tx_bit_count == 4'd7) begin
                            tx_state <= TX_PARITY;
                            tx_parity_bit <= 1'b0; // Simplified parity
                        end else begin
                            tx_bit_count <= tx_bit_count + 1'b1;
                            tx_shift_reg <= {tx_shift_reg[6:0], 1'b0};
                        end
                    end
                end
                TX_PARITY: begin
                    if (baud_tick) begin
                        tx_state <= TX_STOP1;
                        tx_stop_bit <= 1'b1;
                    end
                end
                TX_STOP1: begin
                    if (baud_tick) begin
                        tx_state <= TX_STOP2;
                    end
                end
                TX_STOP2: begin
                    if (baud_tick) begin
                        tx_state <= TX_IDLE;
                        tx_done_r <= 1'b1;
                    end
                end
                default: tx_state <= TX_IDLE;
            endcase
        end
    end

    assign tx_done = tx_done_r;
    assign tx_empty = tx_empty_r;
    assign stx_pad_o = (tx_state == TX_IDLE) ? 1'b1 : 
                       (tx_state == TX_START) ? 1'b0 :
                       (tx_state == TX_DATA) ? tx_shift_reg[0] :
                       (tx_state == TX_PARITY) ? tx_parity_bit :
                       (tx_state == TX_STOP1) ? 1'b1 :
                       (tx_state == TX_STOP2) ? 1'b1 : 1'b1;

    // RX state machine
    localparam RX_IDLE = 3'd0;
    localparam RX_START = 3'd1;
    localparam RX_DATA = 3'd2;
    localparam RX_PARITY = 3'd3;
    localparam RX_STOP1 = 3'd4;
    localparam RX_STOP2 = 3'd5;

    reg [2:0] rx_state;
    reg [3:0] rx_bit_count;
    reg [7:0] rx_shift_reg;
    reg rx_valid_r;
    reg rx_error_r;
    reg rx_break_r;
    reg rx_empty_r;

    wire srx_eff = loopback ? stx_pad_o : srx_pad_i;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rx_state <= RX_IDLE;
            rx_bit_count <= 4'd0;
            rx_shift_reg <= 8'd0;
            rx_valid_r <= 1'b0;
            rx_error_r <= 1'b0;
            rx_break_r <= 1'b0;
            rx_empty_r <= 1'b1;
        end else begin
            rx_valid_r <= 1'b0;
            rx_error_r <= 1'b0;
            rx_break_r <= 1'b0;
            
            case (rx_state)
                RX_IDLE: begin
                    rx_empty_r <= 1'b1;
                    if (!srx_eff) begin
                        rx_state <= RX_START;
                        rx_empty_r <= 1'b0;
                    end
                end
                RX_START: begin
                    if (baud_tick) begin
                        if (!srx_eff) begin
                            rx_state <= RX_DATA;
                            rx_bit_count <= 4'd0;
                        end else begin
                            rx_state <= RX_IDLE;
                            rx_break_r <= 1'b1;
                        end
                    end
                end
                RX_DATA: begin
                    if (baud_tick) begin
                        if (rx_bit_count == 4'd7) begin
                            rx_state <= RX_PARITY;
                        end else begin
                            rx_bit_count <= rx_bit_count + 1'b1;
                            rx_shift_reg <= {srx_eff, rx_shift_reg[7:1]};
                        end
                    end
                end
                RX_PARITY: begin
                    if (baud_tick) begin
                        rx_state <= RX_STOP1;
                    end
                end
                RX_STOP1: begin
                    if (baud_tick) begin
                        rx_state <= RX_STOP2;
                    end
                end
                RX_STOP2: begin
                    if (baud_tick) begin
                        rx_state <= RX_IDLE;
                        rx_valid_r <= 1'b1;
                        rx_empty_r <= 1'b1;
                    end
                end
                default: rx_state <= RX_IDLE;
            endcase
        end
    end

    assign rx_valid = rx_valid_r;
    assign rx_data = rx_shift_reg;
    assign rx_error = rx_error_r;
    assign rx_break = rx_break_r;
    assign rx_empty = rx_empty_r;

endmodule

module rx_fifo (
    input  wire clk,
    input  wire rst,
    input  wire reset,
    input  wire push,
    input  wire [7:0] data_in,
    input  wire error_in,
    input  wire break_in,
    input  wire pop,
    output wire [7:0] data_out,
    output wire error_out,
    output wire break_out,
    output wire [3:0] count,
    output wire overflow,
    output wire empty
);

    reg [7:0] mem [0:15];
    reg [3:0] wr_ptr, rd_ptr;
    reg [3:0] count_r;
    reg overflow_r;
    reg empty_r;

    wire full = (count_r == 4'd16);
    wire empty = (count_r == 4'd0);

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_ptr <= 4'd0;
            rd_ptr <= 4'd0;
            count_r <= 4'd0;
            overflow_r <= 1'b0;
            empty_r <= 1'b1;
        end else if (reset) begin
            wr_ptr <= 4'd0;
            rd_ptr <= 4'd0;
            count_r <= 4'd0;
            overflow_r <= 1'b0;
            empty_r <= 1'b1;
        end else begin
            overflow_r <= 1'b0;
            
            if (push && !full) begin
                mem[wr_ptr] <= data_in;
                wr_ptr <= wr_ptr + 1'b1;
                count_r <= count_r + 1'b1;
                empty_r <= 1'b0;
            end else if (push && full) begin
                overflow_r <= 1'b1;
            end
            
            if (pop && !empty) begin
                rd_ptr <= rd_ptr + 1'b1;
                count_r <= count_r - 1'b1;
                empty_r <= (count_r == 4'd1);
            end
        end
    end

    assign data_out = mem[rd_ptr];
    assign error_out = 1'b0;
    assign break_out = 1'b0;
    assign count = count_r;
    assign overflow = overflow_r;
    assign empty = empty_r;

endmodule

module tx_fifo (
    input  wire clk,
    input  wire rst,
    input  wire reset,
    input  wire push,
    input  wire [7:0] data_in,
    input  wire pop,
    output wire [7:0] data_out,
    output wire [3:0] count,
    output wire empty
);

    reg [7:0] mem [0:15];
    reg [3:0] wr_ptr, rd_ptr;
    reg [3:0] count_r;
    reg empty_r;

    wire full = (count_r == 4'd16);
    wire empty = (count_r == 4'd0);

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_ptr <= 4'd0;
            rd_ptr <= 4'd0;
            count_r <= 4'd0;
            empty_r <= 1'b1;
        end else if (reset) begin
            wr_ptr <= 4'd0;
            rd_ptr <= 4'd0;
            count_r <= 4'd0;
            empty_r <= 1'b1;
        end else begin
            if (push && !full) begin
                mem[wr_ptr] <= data_in;
                wr_ptr <= wr_ptr + 1'b1;
                count_r <= count_r + 1'b1;
                empty_r <= 1'b0;
            end
            
            if (pop && !empty) begin
                rd_ptr <= rd_ptr + 1'b1;
                count_r <= count_r - 1'b1;
                empty_r <= (count_r == 4'd1);
            end
        end
    end

    assign data_out = mem[rd_ptr];
    assign count = count_r;
    assign empty = empty_r;

endmodule
