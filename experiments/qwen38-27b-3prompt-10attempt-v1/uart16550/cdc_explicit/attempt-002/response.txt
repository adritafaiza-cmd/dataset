```verilog
module uart_top (
    input  wire        wb_clk_i,
    input  wire        wb_rst_i,
    input  wire [7:0]  wb_adr_i,
    input  wire [7:0]  wb_dat_i,
    output wire [7:0]  wb_dat_o,
    input  wire        wb_we_i,
    input  wire        wb_stb_i,
    input  wire        wb_cyc_i,
    output wire        wb_ack_o,
    input  wire [3:0]  wb_sel_i,
    output wire        int_o,
    output wire        stx_pad_o,
    input  wire        srx_pad_i,
    output wire        rts_pad_o,
    input  wire        cts_pad_i,
    output wire        dtr_pad_o,
    input  wire        dsr_pad_i,
    input  wire        ri_pad_i,
    input  wire        dcd_pad_i
);

    // Wishbone signals
    wire        wb_valid = wb_cyc_i & wb_stb_i;
    wire        wb_write = wb_valid & wb_we_i;
    wire        wb_read  = wb_valid & ~wb_we_i;

    // Register addresses
    localparam ADDR_THR   = 8'h00;
    localparam ADDR_RBR   = 8'h00;
    localparam ADDR_DLL   = 8'h00;
    localparam ADDR_DLH   = 8'h01;
    localparam ADDR_IER   = 8'h01;
    localparam ADDR_DLM   = 8'h02;
    localparam ADDR_FCR   = 8'h02;
    localparam ADDR_LCR   = 8'h03;
    localparam ADDR_MCR   = 8'h04;
    localparam ADDR_LSR   = 8'h05;
    localparam ADDR_MSR   = 8'h06;
    localparam ADDR_SCR   = 8'h07;

    // Internal signals
    reg  [7:0]  ier_reg;
    reg  [7:0]  lcr_reg;
    reg  [7:0]  mcr_reg;
    reg  [7:0]  scr_reg;
    reg  [7:0]  fcr_reg;
    reg  [7:0]  dll_reg;
    reg  [7:0]  dlh_reg;
    reg  [7:0]  dlm_reg;

    // FIFO signals
    reg  [7:0]  tx_fifo [0:15];
    reg  [3:0]  tx_fifo_wr_ptr;
    reg  [3:0]  tx_fifo_rd_ptr;
    reg  [7:0]  rx_fifo [0:15];
    reg  [3:0]  rx_fifo_wr_ptr;
    reg  [3:0]  rx_fifo_rd_ptr;

    // UART state
    reg  [7:0]  tx_shift_reg;
    reg  [3:0]  tx_bit_cnt;
    reg         tx_active;
    reg  [7:0]  rx_shift_reg;
    reg  [3:0]  rx_bit_cnt;
    reg         rx_active;
    reg         rx_start_bit;
    reg  [15:0] baud_divider;
    reg  [15:0] baud_counter;
    reg         baud_tick;

    // Modem control
    reg         dtr_reg;
    reg         rts_reg;
    reg         loopback_reg;

    // Interrupts
    wire        int_thr_empty;
    wire        int_rx_data;
    wire        int_rx_timeout;
    wire        int_line_status;
    wire        int_modem_status;
    wire        int_enabled;
    wire        int_raw;

    // LSR bits
    wire        lsr_thr_empty;
    wire        lsr_tx_empty;
    wire        lsr_rx_data;
    wire        lsr_overrun;
    wire        lsr_parity_err;
    wire        lsr_framing_err;
    wire        lsr_break_err;
    wire        lsr_fifo_err;

    // MSR bits
    wire        msr_dcd;
    wire        msr_ri;
    wire        msr_dsr;
    wire        msr_cts;
    wire        msr_dcd_change;
    wire        msr_ri_change;
    wire        msr_dsr_change;
    wire        msr_cts_change;

    // FIFO error flags
    reg         overrun_flag;
    reg         parity_err_flag;
    reg         framing_err_flag;
    reg         break_err_flag;

    // Modem status change detection
    reg         dcd_prev;
    reg         ri_prev;
    reg         dsr_prev;
    reg         cts_prev;

    // Baud rate calculation
    wire [15:0] baud_div = {dlh_reg, dll_reg};
    wire [15:0] baud_div_eff = (baud_div == 16'd0) ? 16'd1 : baud_div;

    // DLAB bit
    wire        dlab = lcr_reg[7];

    // FIFO enable
    wire        fifo_enable = fcr_reg[0];

    // Parity settings
    wire        parity_enable = lcr_reg[0];
    wire [1:0]  parity_type   = lcr_reg[2:1];
    wire        stop_bits     = lcr_reg[4]; // 0=1 stop, 1=1.5/2 stop
    wire        word_length   = lcr_reg[3:2]; // 00=5, 01=6, 10=7, 11=8

    // Calculate number of data bits
    wire [3:0]  num_data_bits = (word_length == 2'b00) ? 4'd5 :
                                (word_length == 2'b01) ? 4'd6 :
                                (word_length == 2'b10) ? 4'd7 : 4'd8;

    // Parity bit calculation
    wire        parity_bit;
    assign parity_bit = (parity_type == 2'b00) ? 1'b0 : // Even
                        (parity_type == 2'b01) ? 1'b1 : // Odd
                        (parity_type == 2'b10) ? 1'b0 : // Mark
                        (parity_type == 2'b11) ? 1'b1;  // Space

    // TX FIFO full/empty
    wire [3:0]  tx_fifo_count = tx_fifo_wr_ptr - tx_fifo_rd_ptr;
    wire        tx_fifo_empty = (tx_fifo_wr_ptr == tx_fifo_rd_ptr);
    wire        tx_fifo_full  = (tx_fifo_count == 4'd16);

    // RX FIFO full/empty
    wire [3:0]  rx_fifo_count = rx_fifo_wr_ptr - rx_fifo_rd_ptr;
    wire        rx_fifo_empty = (rx_fifo_wr_ptr == rx_fifo_rd_ptr);
    wire        rx_fifo_full  = (rx_fifo_count == 4'd16);

    // LSR signals
    assign lsr_thr_empty = tx_fifo_empty;
    assign lsr_tx_empty  = tx_fifo_empty & ~tx_active;
    assign lsr_rx_data   = ~rx_fifo_empty;
    assign lsr_overrun   = overrun_flag;
    assign lsr_parity_err = parity_err_flag;
    assign lsr_framing_err = framing_err_flag;
    assign lsr_break_err   = break_err_flag;
    assign lsr_fifo_err    = overrun_flag | parity_err_flag | framing_err_flag | break_err_flag;

    // MSR signals
    assign msr_dcd = dcd_pad_i;
    assign msr_ri  = ri_pad_i;
    assign msr_dsr = dsr_pad_i;
    assign msr_cts = cts_pad_i;
    assign msr_dcd_change = dcd_pad_i ^ dcd_prev;
    assign msr_ri_change  = ri_pad_i ^ ri_prev;
    assign msr_dsr_change = dsr_pad_i ^ dsr_prev;
    assign msr_cts_change = cts_pad_i ^ cts_prev;

    // Interrupt sources
    assign int_thr_empty   = ier_reg[0] & lsr_thr_empty;
    assign int_rx_data     = ier_reg[1] & lsr_rx_data;
    assign int_rx_timeout  = ier_reg[2] & lsr_rx_data & rx_fifo_empty; // Simplified
    assign int_line_status = ier_reg[3] & (lsr_overrun | lsr_parity_err | lsr_framing_err | lsr_break_err);
    assign int_modem_status = ier_reg[4] & (msr_dcd_change | msr_ri_change | msr_dsr_change | msr_cts_change);

    assign int_enabled = ier_reg[5];
    assign int_raw = int_enabled & (int_thr_empty | int_rx_data | int_rx_timeout | int_line_status | int_modem_status);

    assign int_o = int_raw;

    // Modem control outputs
    assign dtr_pad_o = dtr_reg;
    assign rts_pad_o = rts_reg;

    // TX output
    assign stx_pad_o = tx_active ? tx_shift_reg[0] : 1'b1;

    // Baud rate generator
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            baud_counter <= 16'd0;
            baud_tick <= 1'b0;
        end else begin
            baud_tick <= 1'b0;
            if (baud_counter >= baud_div_eff - 16'd1) begin
                baud_counter <= 16'd0;
                baud_tick <= 1'b1;
            end else begin
                baud_counter <= baud_counter + 16'd1;
            end
        end
    end

    // TX logic
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            tx_shift_reg <= 8'd0;
            tx_bit_cnt <= 4'd0;
            tx_active <= 1'b0;
        end else if (baud_tick) begin
            if (tx_active) begin
                if (tx_bit_cnt == 4'd0) begin
                    // Start bit
                    tx_shift_reg <= 8'd0;
                    tx_bit_cnt <= 4'd1;
                end else if (tx_bit_cnt <= num_data_bits) begin
                    // Data bits
                    tx_shift_reg <= {1'b0, tx_shift_reg[7:1]};
                    tx_bit_cnt <= tx_bit_cnt + 4'd1;
                end else if (parity_enable && tx_bit_cnt == num_data_bits + 4'd1) begin
                    // Parity bit
                    tx_shift_reg <= {1'b0, tx_shift_reg[7:1]};
                    tx_bit_cnt <= tx_bit_cnt + 4'd1;
                end else if (tx_bit_cnt == (parity_enable ? num_data_bits + 4'd2 : num_data_bits + 4'd1)) begin
                    // Stop bit
                    tx_shift_reg <= {1'b1, tx_shift_reg[7:1]};
                    tx_bit_cnt <= tx_bit_cnt + 4'd1;
                end else begin
                    // End of transmission
                    tx_active <= 1'b0;
                    tx_bit_cnt <= 4'd0;
                end
            end else if (~tx_fifo_empty) begin
                // Start new transmission
                tx_active <= 1'b1;
                tx_bit_cnt <= 4'd0;
                tx_shift_reg <= {1'b1, tx_fifo[tx_fifo_rd_ptr], 6'b0};
                tx_fifo_rd_ptr <= tx_fifo_rd_ptr + 4'd1;
            end
        end
    end

    // RX logic
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            rx_shift_reg <= 8'd0;
            rx_bit_cnt <= 4'd0;
            rx_active <= 1'b0;
            rx_start_bit <= 1'b0;
            overrun_flag <= 1'b0;
            parity_err_flag <= 1'b0;
            framing_err_flag <= 1'b0;
            break_err_flag <= 1'b0;
        end else if (baud_tick) begin
            if (rx_active) begin
                if (rx_bit_cnt == 4'd0) begin
                    // Start bit
                    rx_shift_reg <= {1'b0, rx_shift_reg[7:1]};
                    rx_bit_cnt <= 4'd1;
                end else if (rx_bit_cnt <= num_data_bits) begin
                    // Data bits
                    rx_shift_reg <= {1'b0, rx_shift_reg[7:1]};
                    rx_bit_cnt <= rx_bit_cnt + 4'd1;
                end else if (parity_enable && rx_bit_cnt == num_data_bits + 4'd1) begin
                    // Parity bit
                    rx_shift_reg <= {1'b0, rx_shift_reg[7:1]};
                    rx_bit_cnt <= rx_bit_cnt + 4'd1;
                end else if (rx_bit_cnt == (parity_enable ? num_data_bits + 4'd2 : num_data_bits + 4'd1)) begin
                    // Stop bit
                    if (srx_pad_i) begin
                        // Valid stop bit
                        rx_active <= 1'b0;
                        rx_bit_cnt <= 4'd0;
                        if (~rx_fifo_full) begin
                            rx_fifo[rx_fifo_wr_ptr] <= rx_shift_reg;
                            rx_fifo_wr_ptr <= rx_fifo_wr_ptr + 4'd1;
                        end else begin
                            overrun_flag <= 1'b1;
                        end
                    end else begin
                        // Framing error
                        framing_err_flag <= 1'b1;
                        rx_active <= 1'b0;
                        rx_bit_cnt <= 4'd0;
                    end
                end else begin
                    // End of reception
                    rx_active <= 1'b0;
                    rx_bit_cnt <= 4'd0;
                end
            end else if (srx_pad_i == 1'b0) begin
                // Start bit detected
                rx_active <= 1'b1;
                rx_bit_cnt <= 4'd0;
                rx_shift_reg <= {1'b0, rx_shift_reg[7:1]};
            end
        end
    end

    // Wishbone register read
    reg [7:0] wb_dat_o_reg;
    always @(*) begin
        case (wb_adr_i)
            ADDR_RBR: begin
                if (dlab)
                    wb_dat_o_reg = dll_reg;
                else if (~rx_fifo_empty)
                    wb_dat_o_reg = rx_fifo[rx_fifo_rd_ptr];
                else
                    wb_dat_o_reg = 8'd0;
            end
            ADDR_DLH: begin
                if (dlab)
                    wb_dat_o_reg = dlh_reg;
                else
                    wb_dat_o_reg = ier_reg;
            end
            ADDR_DLM: begin
                if (dlab)
                    wb_dat_o_reg = dlm_reg;
                else
                    wb_dat_o_reg = fcr_reg;
            end
            ADDR_LCR: wb_dat_o_reg = lcr_reg;
            ADDR_MCR: wb_dat_o_reg = mcr_reg;
            ADDR_LSR: begin
                wb_dat_o_reg = {1'b0, lsr_fifo_err, lsr_break_err, lsr_framing_err, lsr_parity_err, lsr_overrun, lsr_rx_data, lsr_tx_empty, lsr_thr_empty};
            end
            ADDR_MSR: begin
                wb_dat_o_reg = {1'b0, msr_cts_change, msr_dsr_change, msr_ri_change, msr_dcd_change, msr_cts, msr_dsr, msr_ri, msr_dcd};
            end
            ADDR_SCR: wb_dat_o_reg = scr_reg;
            default:  wb_dat_o_reg = 8'd0;
        endcase
    end

    assign wb_dat_o = wb_dat_o_reg;

    // Wishbone register write
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            ier_reg <= 8'd0;
            lcr_reg <= 8'd0;
            mcr_reg <= 8'd0;
            scr_reg <= 8'd0;
            fcr_reg <= 8'd0;
            dll_reg <= 8'd0;
            dlh_reg <= 8'd0;
            dlm_reg <= 8'd0;
            dtr_reg <= 1'b0;
            rts_reg <= 1'b0;
            loopback_reg <= 1'b0;
            tx_fifo_wr_ptr <= 4'd0;
            tx_fifo_rd_ptr <= 4'd0;
            rx_fifo_wr_ptr <= 4'd0;
            rx_fifo_rd_ptr <= 4'd0;
            overrun_flag <= 1'b0;
            parity_err_flag <= 1'b0;
            framing_err_flag <= 1'b0;
            break_err_flag <= 1'b0;
            dcd_prev <= 1'b0;
            ri_prev <= 1'b0;
            dsr_prev <= 1'b0;
            cts_prev <= 1'b0;
        end else begin
            // Update modem status previous values
            dcd_prev <= dcd_pad_i;
            ri_prev <= ri_pad_i;
            dsr_prev <= dsr_pad_i;
            cts_prev <= cts_pad_i;

            if (wb_write) begin
                case (wb_adr_i)
                    ADDR_THR: begin
                        if (dlab) begin
                            dll_reg <= wb_dat_i;
                        end else begin
                            if (~tx_fifo_full) begin
                                tx_fifo[tx_fifo_wr_ptr] <= wb_dat_i;
                                tx_fifo_wr_ptr <= tx_fifo_wr_ptr + 4'd1;
                            end
                        end
                    end
                    ADDR_DLH: begin
                        if (dlab) begin
                            dlh_reg <= wb_dat_i;
                        end else begin
                            ier_reg <= wb_dat_i;
                        end
                    end
                    ADDR_DLM: begin
                        if (dlab) begin
                            dlm_reg <= wb_dat_i;
                        end else begin
                            fcr_reg <= wb_dat_i;
                            if (wb_dat_i[2]) begin
                                // FIFO reset
                                tx_fifo_wr_ptr <= 4'd0;
                                tx_fifo_rd_ptr <= 4'd0;
                                rx_fifo_wr_ptr <= 4'd0;
                                rx_fifo_rd_ptr <= 4'd0;
                                overrun_flag <= 1'b0;
                                parity_err_flag <= 1'b0;
                                framing_err_flag <= 1'b0;
                                break_err_flag <= 1'b0;
                            end
                        end
                    end
                    ADDR_LCR: begin
                        lcr_reg <= wb_dat_i;
                    end
                    ADDR_MCR: begin
                        mcr_reg <= wb_dat_i;
                        dtr_reg <= wb_dat_i[0];
                        rts_reg <= wb_dat_i[1];
                        loopback_reg <= wb_dat_i[4];
                    end
                    ADDR_SCR: begin
                        scr_reg <= wb_dat_i;
                    end
                    default: ;
                endcase
            end

            // Clear error flags on read
            if (wb_read && wb_adr_i == ADDR_LSR) begin
                overrun_flag <= 1'b0;
                parity_err_flag <= 1'b0;
                framing_err_flag <= 1'b0;
                break_err_flag <= 1'b0;
            end

            // Clear modem status change flags on read
            if (wb_read && wb_adr_i == ADDR_MSR) begin
                dcd_prev <= dcd_pad_i;
                ri_prev <= ri_pad_i;
                dsr_prev <= dsr_pad_i;
                cts_prev <= cts_pad_i;
            end
        end
    end

    // Wishbone ACK
    reg wb_ack_o_reg;
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            wb_ack_o_reg <= 1'b0;
        end else begin
            wb_ack_o_reg <= wb_valid;
        end
    end

    assign wb_ack_o = wb_ack_o_reg;

endmodule
