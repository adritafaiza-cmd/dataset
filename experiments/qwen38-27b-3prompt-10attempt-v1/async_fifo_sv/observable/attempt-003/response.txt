module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 1 << ADDR_WIDTH;
    localparam PTR_WIDTH = ADDR_WIDTH + 1;

    // Write domain
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_WIDTH-1:0]  wptr;
    reg [PTR_WIDTH-1:0]  rptr_wsync;

    // Read domain
    reg [PTR_WIDTH-1:0]  rptr;
    reg [PTR_WIDTH-1:0]  wptr_rsync;

    // Synchronizer for read pointer into write domain
    reg [PTR_WIDTH-1:0] rptr_sync [0:SYNC_STAGES-1];

    // Synchronizer for write pointer into read domain
    reg [PTR_WIDTH-1:0] wptr_sync [0:SYNC_STAGES-1];

    // Write clock domain logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {PTR_WIDTH{1'b0}};
        end else if (winc && !wfull) begin
            wptr <= wptr + 1'b1;
        end
    end

    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_sync[0] <= {PTR_WIDTH{1'b0}};
        end else begin
            rptr_sync[0] <= rptr;
        end
    end

    genvar i;
    generate
        for (i = 1; i < SYNC_STAGES; i = i + 1) begin : rptr_sync_gen
            always @(posedge wclk or negedge wrst_n) begin
                if (!wrst_n) begin
                    rptr_sync[i] <= {PTR_WIDTH{1'b0}};
                end else begin
                    rptr_sync[i] <= rptr_sync[i-1];
                end
            end
        end
    endgenerate

    assign rptr_wsync = rptr_sync[SYNC_STAGES-1];

    // Full condition: write pointer has lapped read pointer by DEPTH
    // In binary, full when upper bits match and lower bits differ by 1
    // Specifically: wptr[ADDR_WIDTH:1] == rptr_wsync[ADDR_WIDTH:1] && wptr[0] != rptr_wsync[0]
    // But since we use PTR_WIDTH = ADDR_WIDTH+1, the MSB is the wrap bit.
    // Full when wptr[ADDR_WIDTH:1] == rptr_wsync[ADDR_WIDTH:1] && wptr[0] != rptr_wsync[0]
    // Actually, standard approach: full when (wptr - rptr_wsync) == DEPTH
    // Which means wptr[ADDR_WIDTH:1] == rptr_wsync[ADDR_WIDTH:1] and wptr[0] != rptr_wsync[0]
    // Let's use: wptr[ADDR_WIDTH:1] == rptr_wsync[ADDR_WIDTH:1] && wptr[0] != rptr_wsync[0]
    // Wait, that's not quite right. Let me think again.
    // wptr and rptr are PTR_WIDTH bits. DEPTH = 2^ADDR_WIDTH.
    // Full when wptr - rptr_wsync == DEPTH (mod 2^PTR_WIDTH)
    // This means the upper ADDR_WIDTH bits are equal and the LSB differs.
    // Actually: wptr[ADDR_WIDTH:1] == rptr_wsync[ADDR_WIDTH:1] && wptr[0] != rptr_wsync[0]
    // Hmm, let me verify: if wptr = 0b10000 and rptr = 0b00000, DEPTH=16, ADDR_WIDTH=4, PTR_WIDTH=5
    // wptr[4:1] = 0b100, rptr[4:1] = 0b000. Not equal. So not full. But wptr - rptr = 16 = DEPTH. So it should be full.
    // My condition is wrong. Let me reconsider.
    // wptr - rptr = DEPTH means wptr = rptr + DEPTH (mod 2^PTR_WIDTH)
    // Since DEPTH = 2^ADDR_WIDTH, adding DEPTH flips bit ADDR_WIDTH and leaves lower bits unchanged.
    // So wptr[ADDR_WIDTH-1:0] == rptr[ADDR_WIDTH-1:0] and wptr[ADDR_WIDTH] != rptr[ADDR_WIDTH]
    // That is: wptr[ADDR_WIDTH-1:0] == rptr_wsync[ADDR_WIDTH-1:0] && wptr[ADDR_WIDTH] != rptr_wsync[ADDR_WIDTH]
    
    assign wfull = (wptr[ADDR_WIDTH-1:0] == rptr_wsync[ADDR_WIDTH-1:0]) && 
                   (wptr[ADDR_WIDTH] != rptr_wsync[ADDR_WIDTH]);

    // Write data to memory
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            // No action needed for memory
        end else if (winc && !wfull) begin
            mem[wptr[ADDR_WIDTH-1:0]] <= wdata;
        end
    end

    assign waddr = wptr;

    // Read clock domain logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {PTR_WIDTH{1'b0}};
        end else if (rinc && !rempty) begin
            rptr <= rptr + 1'b1;
        end
    end

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_sync[0] <= {PTR_WIDTH{1'b0}};
        end else begin
            wptr_sync[0] <= wptr;
        end
    end

    generate
        for (i = 1; i < SYNC_STAGES; i = i + 1) begin : wptr_sync_gen
            always @(posedge rclk or negedge rrst_n) begin
                if (!rrst_n) begin
                    wptr_sync[i] <= {PTR_WIDTH{1'b0}};
                end else begin
                    wptr_sync[i] <= wptr_sync[i-1];
                end
            end
        end
    endgenerate

    assign wptr_rsync = wptr_sync[SYNC_STAGES-1];

    // Empty condition: wptr_rsync == rptr
    assign rempty = (wptr_rsync == rptr);

    // Read data from memory
    assign rdata = mem[rptr[ADDR_WIDTH-1:0]];

    assign raddr = rptr;

endmodule
