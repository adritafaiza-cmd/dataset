module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 1 << ADDR_WIDTH;

    // Write domain
    reg [ADDR_WIDTH:0] wptr;
    reg [ADDR_WIDTH:0] wptr_synced_r; // read pointer synced to write domain

    // Read domain
    reg [ADDR_WIDTH:0] rptr;
    reg [ADDR_WIDTH:0] rptr_synced_w; // write pointer synced to read domain

    // FIFO memory
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];

    // Write enable
    wire wvalid = winc & ~wfull;

    // Read enable
    wire rvalid = rinc & ~rempty;

    // Write pointer increment
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n)
            wptr <= {(ADDR_WIDTH+1){1'b0}};
        else if (wvalid)
            wptr <= wptr + 1'b1;
    end

    // Read pointer increment
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n)
            rptr <= {(ADDR_WIDTH+1){1'b0}};
        else if (rvalid)
            rptr <= rptr + 1'b1;
    end

    // Write to memory
    always @(posedge wclk) begin
        if (wvalid)
            mem[wptr[ADDR_WIDTH-1:0]] <= wdata;
    end

    // Read from memory
    always @(posedge rclk) begin
        if (rvalid)
            rdata <= mem[rptr[ADDR_WIDTH-1:0]];
    end

    // Synchronize write pointer to read domain
    // We need to sync the write pointer (wptr) to the read domain for rempty calculation
    // and sync the read pointer (rptr) to the write domain for wfull calculation.

    // Sync wptr to read domain
    reg [ADDR_WIDTH:0] wptr_sync_r [0:SYNC_STAGES-1];
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_sync_r[0] <= {(ADDR_WIDTH+1){1'b0}};
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                wptr_sync_r[i] <= {(ADDR_WIDTH+1){1'b0}};
        end else begin
            wptr_sync_r[0] <= wptr;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                wptr_sync_r[i] <= wptr_sync_r[i-1];
        end
    end

    // Sync rptr to write domain
    reg [ADDR_WIDTH:0] rptr_sync_w [0:SYNC_STAGES-1];
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_sync_w[0] <= {(ADDR_WIDTH+1){1'b0}};
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                rptr_sync_w[i] <= {(ADDR_WIDTH+1){1'b0}};
        end else begin
            rptr_sync_w[0] <= rptr;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                rptr_sync_w[i] <= rptr_sync_w[i-1];
        end
    end

    // Full condition: write pointer has lapped read pointer by DEPTH
    // wfull = (wptr[ADDR_WIDTH] != rptr_synced_w[ADDR_WIDTH]) && (wptr[ADDR_WIDTH-1:0] == rptr_synced_w[ADDR_WIDTH-1:0])
    assign wfull = (wptr[ADDR_WIDTH] != rptr_sync_w[SYNC_STAGES-1][ADDR_WIDTH]) &&
                   (wptr[ADDR_WIDTH-1:0] == rptr_sync_w[SYNC_STAGES-1][ADDR_WIDTH-1:0]);

    // Empty condition: write pointer equals read pointer
    // rempty = (wptr_synced_r == rptr)
    assign rempty = (wptr_sync_r[SYNC_STAGES-1] == rptr);

    // Debug pointer outputs
    assign waddr = wptr;
    assign raddr = rptr;

endmodule
