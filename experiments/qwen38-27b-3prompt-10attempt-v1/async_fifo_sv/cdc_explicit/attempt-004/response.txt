module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 1 << ADDR_WIDTH;

    // Write domain
    reg [ADDR_WIDTH:0] wptr;
    reg [ADDR_WIDTH:0] wptr_sync_r; // read pointer synchronized to write domain
    reg [ADDR_WIDTH:0] wptr_sync_r_next;
    reg [ADDR_WIDTH:0] wptr_sync_r_1;
    reg [ADDR_WIDTH:0] wptr_sync_r_2;

    // Read domain
    reg [ADDR_WIDTH:0] rptr;
    reg [ADDR_WIDTH:0] rptr_sync_w; // write pointer synchronized to read domain
    reg [ADDR_WIDTH:0] rptr_sync_w_next;
    reg [ADDR_WIDTH:0] rptr_sync_w_1;
    reg [ADDR_WIDTH:0] rptr_sync_w_2;

    // Memory
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];

    // Write enable
    wire wvalid = winc & ~wfull;

    // Read enable
    wire rvalid = rinc & ~rempty;

    // Full condition: write pointer has lapped read pointer by DEPTH
    // wfull = (wptr[ADDR_WIDTH] != rptr_sync_r[ADDR_WIDTH]) && 
    //         (wptr[ADDR_WIDTH-1:0] == rptr_sync_r[ADDR_WIDTH-1:0])
    wire wfull_internal = (wptr[ADDR_WIDTH] != wptr_sync_r[ADDR_WIDTH]) && 
                          (wptr[ADDR_WIDTH-1:0] == wptr_sync_r[ADDR_WIDTH-1:0]);

    // Empty condition: read pointer equals write pointer
    wire rempty_internal = (rptr == rptr_sync_w);

    // Assign outputs
    assign wfull = wfull_internal;
    assign waddr = wptr;
    assign rdata = mem[rptr[ADDR_WIDTH-1:0]];
    assign rempty = rempty_internal;
    assign raddr = rptr;

    // Write pointer update
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n)
            wptr <= {(ADDR_WIDTH+1){1'b0}};
        else if (wvalid)
            wptr <= wptr + 1'b1;
    end

    // Read pointer update
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n)
            rptr <= {(ADDR_WIDTH+1){1'b0}};
        else if (rvalid)
            rptr <= rptr + 1'b1;
    end

    // Synchronize read pointer to write domain
    // We need to sync rptr from read domain to write domain
    // Use a 2-stage synchronizer for the pointer
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr_sync_r_1 <= {(ADDR_WIDTH+1){1'b0}};
            wptr_sync_r_2 <= {(ADDR_WIDTH+1){1'b0}};
        end else begin
            wptr_sync_r_1 <= rptr;
            wptr_sync_r_2 <= wptr_sync_r_1;
        end
    end

    assign wptr_sync_r = wptr_sync_r_2;

    // Synchronize write pointer to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr_sync_w_1 <= {(ADDR_WIDTH+1){1'b0}};
            rptr_sync_w_2 <= {(ADDR_WIDTH+1){1'b0}};
        end else begin
            rptr_sync_w_1 <= wptr;
            rptr_sync_w_2 <= rptr_sync_w_1;
        end
    end

    assign rptr_sync_w = rptr_sync_w_2;

    // Memory write
    always @(posedge wclk) begin
        if (wvalid)
            mem[wptr[ADDR_WIDTH-1:0]] <= wdata;
    end

endmodule
