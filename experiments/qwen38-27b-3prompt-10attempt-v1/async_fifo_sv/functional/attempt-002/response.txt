module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 1 << ADDR_WIDTH;

    // Memory
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];

    // Write domain
    reg [ADDR_WIDTH:0] wptr;
    reg [ADDR_WIDTH:0] rptr_synced;
    wire               wfull;

    // Read domain
    reg [ADDR_WIDTH:0] rptr;
    reg [ADDR_WIDTH:0] wptr_synced;
    wire               rempty;

    // Gray code pointers
    wire [ADDR_WIDTH:0] wptr_gray;
    wire [ADDR_WIDTH:0] rptr_gray;

    // Synchronizer for write pointer into read domain
    reg [ADDR_WIDTH:0] wptr_gray_sync [0:SYNC_STAGES-1];

    // Synchronizer for read pointer into write domain
    reg [ADDR_WIDTH:0] rptr_gray_sync [0:SYNC_STAGES-1];

    // Gray code conversion
    assign wptr_gray = wptr ^ (wptr >> 1);
    assign rptr_gray = rptr ^ (rptr >> 1);

    // Write domain logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {(ADDR_WIDTH+1){1'b0}};
        end else if (winc && !wfull) begin
            wptr <= wptr + 1'b1;
        end
    end

    // Write memory
    always @(posedge wclk) begin
        if (winc && !wfull) begin
            mem[wptr[ADDR_WIDTH-1:0]] <= wdata;
        end
    end

    // Full condition: write pointer gray code equals read pointer gray code with MSB different
    assign wfull = (wptr_gray == rptr_synced) && (wptr[ADDR_WIDTH] != rptr_synced[ADDR_WIDTH]);

    // Synchronize read pointer gray code into write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_sync[0] <= {(ADDR_WIDTH+1){1'b0}};
            rptr_gray_sync[1] <= {(ADDR_WIDTH+1){1'b0}};
            if (SYNC_STAGES > 2) begin
                for (int i = 2; i < SYNC_STAGES; i++) begin
                    rptr_gray_sync[i] <= {(ADDR_WIDTH+1){1'b0}};
                end
            end
        end else begin
            rptr_gray_sync[0] <= rptr_gray;
            rptr_gray_sync[1] <= rptr_gray_sync[0];
            if (SYNC_STAGES > 2) begin
                for (int i = 2; i < SYNC_STAGES; i++) begin
                    rptr_gray_sync[i] <= rptr_gray_sync[i-1];
                end
            end
        end
    end

    assign rptr_synced = rptr_gray_sync[SYNC_STAGES-1];

    // Read domain logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {(ADDR_WIDTH+1){1'b0}};
        end else if (rinc && !rempty) begin
            rptr <= rptr + 1'b1;
        end
    end

    // Read data
    assign rdata = mem[rptr[ADDR_WIDTH-1:0]];

    // Empty condition: read pointer gray code equals write pointer gray code
    assign rempty = (rptr_gray == wptr_synced);

    // Synchronize write pointer gray code into read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_sync[0] <= {(ADDR_WIDTH+1){1'b0}};
            wptr_gray_sync[1] <= {(ADDR_WIDTH+1){1'b0}};
            if (SYNC_STAGES > 2) begin
                for (int i = 2; i < SYNC_STAGES; i++) begin
                    wptr_gray_sync[i] <= {(ADDR_WIDTH+1){1'b0}};
                end
            end
        end else begin
            wptr_gray_sync[0] <= wptr_gray;
            wptr_gray_sync[1] <= wptr_gray_sync[0];
            if (SYNC_STAGES > 2) begin
                for (int i = 2; i < SYNC_STAGES; i++) begin
                    wptr_gray_sync[i] <= wptr_gray_sync[i-1];
                end
            end
        end
    end

    assign wptr_synced = wptr_gray_sync[SYNC_STAGES-1];

    // Debug outputs
    assign waddr = wptr;
    assign raddr = rptr;

endmodule
