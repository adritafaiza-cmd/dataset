module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Storage
    reg [WIDTH-1:0] mem [0:DEPTH-1];

    // Source domain pointers
    reg [PTR_W-1:0] src_wr_ptr;
    reg [PTR_W-1:0] src_rd_ptr_sync;

    // Destination domain pointers
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [PTR_W-1:0] dst_wr_ptr_sync;

    // Synchronizers
    reg [PTR_W-1:0] src_wr_ptr_sync1, src_wr_ptr_sync2;
    reg [PTR_W-1:0] dst_rd_ptr_sync1, dst_rd_ptr_sync2;

    // Destination output register
    reg [WIDTH-1:0] dst_data_reg;
    reg             dst_valid_reg;

    // Compute full/empty
    wire src_full = (src_wr_ptr[LOG_DEPTH-1:0] == src_rd_ptr_sync[LOG_DEPTH-1:0]) &&
                    (src_wr_ptr[LOG_DEPTH] != src_rd_ptr_sync[LOG_DEPTH]);

    wire dst_empty = (dst_rd_ptr[LOG_DEPTH-1:0] == dst_wr_ptr_sync[LOG_DEPTH-1:0]) &&
                     (dst_rd_ptr[LOG_DEPTH] == dst_wr_ptr_sync[LOG_DEPTH]);

    // Source ready
    assign src_ready_o = ~src_full;

    // Destination valid
    assign dst_valid_o = dst_valid_reg;
    assign dst_data_o  = dst_data_reg;

    // Source clock domain: write pointer and synchronizer for read pointer
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
        end else begin
            if (src_valid_i && src_ready_o) begin
                mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
                src_wr_ptr <= src_wr_ptr + 1'b1;
            end
        end
    end

    // Synchronize source write pointer to destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_sync1 <= {PTR_W{1'b0}};
            src_wr_ptr_sync2 <= {PTR_W{1'b0}};
        end else begin
            src_wr_ptr_sync1 <= src_wr_ptr;
            src_wr_ptr_sync2 <= src_wr_ptr_sync1;
        end
    end

    // Synchronize destination read pointer to source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_sync1 <= {PTR_W{1'b0}};
            dst_rd_ptr_sync2 <= {PTR_W{1'b0}};
        end else begin
            dst_rd_ptr_sync1 <= dst_rd_ptr;
            dst_rd_ptr_sync2 <= dst_rd_ptr_sync1;
        end
    end

    // Assign synchronized pointers
    assign src_rd_ptr_sync = dst_rd_ptr_sync2;
    assign dst_wr_ptr_sync = src_wr_ptr_sync2;

    // Destination clock domain: read pointer and output register
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr    <= {PTR_W{1'b0}};
            dst_data_reg  <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_valid_reg && dst_ready_i) begin
                // Consume current output
                dst_valid_reg <= 1'b0;
                dst_rd_ptr    <= dst_rd_ptr + 1'b1;
            end else if (!dst_valid_reg && !dst_empty) begin
                // Load next word from memory
                dst_data_reg  <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                dst_valid_reg <= 1'b1;
                dst_rd_ptr    <= dst_rd_ptr + 1'b1;
            end
        end
    end

endmodule
