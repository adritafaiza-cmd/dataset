module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;

    // Storage
    reg [WIDTH-1:0] mem [0:DEPTH-1];

    // Source domain pointers
    reg [PTR_WIDTH-1:0] src_wr_ptr;
    reg [PTR_WIDTH-1:0] src_rd_ptr_sync;

    // Destination domain pointers
    reg [PTR_WIDTH-1:0] dst_rd_ptr;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_sync;

    // Synchronizers
    reg [PTR_WIDTH-1:0] src_wr_ptr_sync1, src_wr_ptr_sync2;
    reg [PTR_WIDTH-1:0] dst_rd_ptr_sync1, dst_rd_ptr_sync2;

    // Destination output register
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;

    // Source ready logic
    wire src_full;
    wire src_can_write;

    // Destination ready logic
    wire dst_empty;
    wire dst_can_read;

    // Compute full/empty
    // Full: src_wr_ptr - src_rd_ptr_sync == DEPTH
    // Empty: dst_rd_ptr == dst_wr_ptr_sync
    assign src_full = (src_wr_ptr[PTR_WIDTH-1:1] == src_rd_ptr_sync[PTR_WIDTH-1:1]) &&
                      (src_wr_ptr[0] != src_rd_ptr_sync[0]);
    // Actually, for a 2-phase FIFO with pointer width LOG_DEPTH+1:
    // Full when the upper bits are equal and the LSBs differ (wr_ptr is one ahead in the wrap-around sense)
    // Let's use a simpler comparison:
    // Full: (src_wr_ptr - src_rd_ptr_sync) == DEPTH
    // Since DEPTH = 2^LOG_DEPTH, and pointers are LOG_DEPTH+1 bits:
    // Full when src_wr_ptr[LOG_DEPTH:1] == src_rd_ptr_sync[LOG_DEPTH:1] and src_wr_ptr[0] != src_rd_ptr_sync[0]
    // But actually, the standard way:
    // Full: src_wr_ptr[PTR_WIDTH-1:1] == src_rd_ptr_sync[PTR_WIDTH-1:1] && src_wr_ptr[0] != src_rd_ptr_sync[0]
    // This works because when full, the write pointer has wrapped around and is exactly DEPTH ahead.

    // Let me reconsider. With PTR_WIDTH = LOG_DEPTH + 1:
    // The difference (wr_ptr - rd_ptr) mod 2^(LOG_DEPTH+1) gives the number of entries.
    // Full when difference == DEPTH = 2^LOG_DEPTH
    // Empty when difference == 0

    // For full: wr_ptr[LOG_DEPTH:1] == rd_ptr[LOG_DEPTH:1] and wr_ptr[0] != rd_ptr[0]
    // For empty: wr_ptr == rd_ptr

    assign src_full = (src_wr_ptr[PTR_WIDTH-1:1] == src_rd_ptr_sync[PTR_WIDTH-1:1]) &&
                      (src_wr_ptr[0] != src_rd_ptr_sync[0]);

    assign dst_empty = (dst_rd_ptr == dst_wr_ptr_sync);

    assign src_can_write = src_valid_i && !src_full;
    assign dst_can_read = !dst_empty && dst_ready_i;

    // Source clock domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
        end else if (src_can_write) begin
            src_wr_ptr <= src_wr_ptr + 1'b1;
            mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
        end
    end

    assign src_ready_o = !src_full;

    // Synchronize src_wr_ptr to dst domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            src_wr_ptr_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            src_wr_ptr_sync1 <= src_wr_ptr;
            src_wr_ptr_sync2 <= src_wr_ptr_sync1;
        end
    end

    assign dst_wr_ptr_sync = src_wr_ptr_sync2;

    // Synchronize dst_rd_ptr to src domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            dst_rd_ptr_sync1 <= dst_rd_ptr;
            dst_rd_ptr_sync2 <= dst_rd_ptr_sync1;
        end
    end

    assign src_rd_ptr_sync = dst_rd_ptr_sync2;

    // Destination clock domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_can_read) begin
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                dst_valid_reg <= 1'b1;
            end else begin
                dst_valid_reg <= 1'b0;
            end
        end
    end

    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;

endmodule
