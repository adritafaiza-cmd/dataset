module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_W = LOG_DEPTH;
    localparam CNT_W = LOG_DEPTH + 1;

    // Source domain
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [ADDR_W-1:0] wr_ptr;
    reg [ADDR_W-1:0] rd_ptr_src;
    reg [ADDR_W-1:0] rd_ptr_src_sync;
    reg [ADDR_W-1:0] rd_ptr_src_sync2;

    // Destination domain
    reg [ADDR_W-1:0] rd_ptr;
    reg [ADDR_W-1:0] wr_ptr_dst;
    reg [ADDR_W-1:0] wr_ptr_dst_sync;
    reg [ADDR_W-1:0] wr_ptr_dst_sync2;
    reg [WIDTH-1:0] dst_reg;
    reg dst_valid_reg;

    // Synchronizers for read pointer (src -> dst)
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            rd_ptr_src_sync <= {ADDR_W{1'b0}};
            rd_ptr_src_sync2 <= {ADDR_W{1'b0}};
        end else begin
            rd_ptr_src_sync <= rd_ptr_src;
            rd_ptr_src_sync2 <= rd_ptr_src_sync;
        end
    end

    // Synchronizers for write pointer (dst -> src)
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            wr_ptr_dst_sync <= {ADDR_W{1'b0}};
            wr_ptr_dst_sync2 <= {ADDR_W{1'b0}};
        end else begin
            wr_ptr_dst_sync <= wr_ptr_dst;
            wr_ptr_dst_sync2 <= wr_ptr_dst_sync;
        end
    end

    // Source domain logic
    wire src_wr_en = src_valid_i & src_ready_o;
    wire [CNT_W-1:0] src_count = {1'b0, wr_ptr} - {1'b0, rd_ptr_src_sync2};
    wire src_full = (src_count == CNT_W'(DEPTH));
    assign src_ready_o = !src_full;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            wr_ptr <= {ADDR_W{1'b0}};
            rd_ptr_src <= {ADDR_W{1'b0}};
        end else begin
            if (src_wr_en)
                mem[wr_ptr] <= src_data_i;
            if (src_wr_en)
                wr_ptr <= wr_ptr + 1'b1;
            if (dst_valid_reg & dst_ready_i)
                rd_ptr_src <= rd_ptr_src + 1'b1;
        end
    end

    // Destination domain logic
    wire [CNT_W-1:0] dst_count = {1'b0, wr_ptr_dst_sync2} - {1'b0, rd_ptr};
    wire dst_empty = (dst_count == {CNT_W{1'b0}});
    wire dst_wr_en = !dst_empty;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            rd_ptr <= {ADDR_W{1'b0}};
            wr_ptr_dst <= {ADDR_W{1'b0}};
            dst_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_wr_en) begin
                dst_reg <= mem[rd_ptr];
                rd_ptr <= rd_ptr + 1'b1;
                dst_valid_reg <= 1'b1;
            end else begin
                dst_valid_reg <= 1'b0;
            end
            if (dst_valid_reg & dst_ready_i)
                wr_ptr_dst <= wr_ptr_dst + 1'b1;
        end
    end

    assign dst_data_o = dst_reg;
    assign dst_valid_o = dst_valid_reg;

endmodule
