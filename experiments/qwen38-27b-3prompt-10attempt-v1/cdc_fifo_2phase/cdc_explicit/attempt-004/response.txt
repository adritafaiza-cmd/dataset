module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_W = LOG_DEPTH;
    localparam CNT_W = LOG_DEPTH + 1;

    // Source domain
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [ADDR_W-1:0] wr_ptr;
    reg [ADDR_W-1:0] rd_ptr_src; // synchronized read pointer in src domain

    // Destination domain
    reg [ADDR_W-1:0] rd_ptr;
    reg [ADDR_W-1:0] wr_ptr_dst; // synchronized write pointer in dst domain
    reg [WIDTH-1:0] dst_reg;
    reg dst_valid_reg;

    // Synchronization registers
    reg [ADDR_W-1:0] wr_ptr_sync1, wr_ptr_sync2;
    reg [ADDR_W-1:0] rd_ptr_sync1, rd_ptr_sync2;

    // Source domain logic
    wire src_wr_en = src_valid_i & src_ready_o;
    wire src_full;

    // Compute src_ready_o: FIFO is not full
    // Full when wr_ptr == rd_ptr_src and the MSB differs (Gray code comparison)
    // Using binary pointers: full when (wr_ptr + 1) == rd_ptr_src and MSB differs
    // Actually, let's use Gray code for CDC safety.

    // Convert binary to Gray
    function [ADDR_W-1:0] bin2gray;
        input [ADDR_W-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Source domain: write pointer in Gray
    wire [ADDR_W-1:0] wr_ptr_gray = bin2gray({1'b0, wr_ptr});

    // Destination domain: read pointer in Gray
    wire [ADDR_W-1:0] rd_ptr_gray = bin2gray({1'b0, rd_ptr});

    // Synchronize write pointer to destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            wr_ptr_sync1 <= {ADDR_W{1'b0}};
            wr_ptr_sync2 <= {ADDR_W{1'b0}};
        end else begin
            wr_ptr_sync1 <= wr_ptr_gray;
            wr_ptr_sync2 <= wr_ptr_sync1;
        end
    end

    // Synchronize read pointer to source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            rd_ptr_sync1 <= {ADDR_W{1'b0}};
            rd_ptr_sync2 <= {ADDR_W{1'b0}};
        end else begin
            rd_ptr_sync1 <= rd_ptr_gray;
            rd_ptr_sync2 <= rd_ptr_sync1;
        end
    end

    // Source domain: determine if FIFO is full
    // Full when wr_ptr_gray == rd_ptr_sync2 and MSB differs
    // Actually, with Gray code, full when wr_ptr_gray == rd_ptr_sync2 ^ {1'b1, {ADDR_W-1{1'b0}}}
    // Let's use the standard approach:
    // Full when (wr_ptr + 1) == rd_ptr in binary, which in Gray means:
    // wr_ptr_gray == rd_ptr_sync2 and MSB of wr_ptr != MSB of rd_ptr_sync2
    // Simpler: convert synchronized Gray back to binary and compare

    wire [ADDR_W-1:0] rd_ptr_sync2_bin = gray2bin(rd_ptr_sync2);
    wire [ADDR_W-1:0] wr_ptr_gray_inv = wr_ptr_gray ^ {1'b1, {ADDR_W-1{1'b0}}};

    // Full condition: wr_ptr_gray == rd_ptr_sync2 and MSB differs
    // Or equivalently: wr_ptr_gray == (rd_ptr_sync2 + 1) in Gray
    // Standard: full when wr_ptr_gray == rd_ptr_sync2 and MSB(wr_ptr) != MSB(rd_ptr_sync2)
    // Let's just use: full when (wr_ptr + 1) == rd_ptr_sync2_bin
    wire src_full = (wr_ptr + 1'b1) == rd_ptr_sync2_bin;

    assign src_ready_o = !src_full;

    // Write to memory
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            wr_ptr <= {ADDR_W{1'b0}};
        end else if (src_wr_en) begin
            mem[wr_ptr] <= src_data_i;
            wr_ptr <= wr_ptr + 1'b1;
        end
    end

    // Destination domain logic
    wire dst_rd_en = dst_valid_o & dst_ready_i;
    wire dst_empty;

    // Empty when rd_ptr == wr_ptr_sync2_bin
    wire [ADDR_W-1:0] wr_ptr_sync2_bin = gray2bin(wr_ptr_sync2);
    wire dst_empty = (rd_ptr == wr_ptr_sync2_bin);

    // Destination valid: we have data to deliver
    // dst_valid_o is high when FIFO is not empty
    assign dst_valid_o = !dst_empty;

    // Read from memory and register output
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            rd_ptr <= {ADDR_W{1'b0}};
            dst_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_rd_en) begin
                rd_ptr <= rd_ptr + 1'b1;
            end
            if (!dst_empty) begin
                dst_reg <= mem[rd_ptr];
                dst_valid_reg <= 1'b1;
            end else begin
                dst_valid_reg <= 1'b0;
            end
        end
    end

    assign dst_data_o = dst_reg;

    // Gray to binary conversion
    function [ADDR_W-1:0] gray2bin;
        input [ADDR_W-1:0] gray;
        integer i;
        begin
            gray2bin[ADDR_W-1] = gray[ADDR_W-1];
            for (i = ADDR_W-2; i >= 0; i = i - 1) begin
                gray2bin[i] = gray2bin[i+1] ^ gray[i];
            end
        end
    endfunction

endmodule
