module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;

    // Source domain
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [ADDR_WIDTH-1:0] wr_ptr;
    reg [PTR_WIDTH-1:0] wr_ptr_gray;
    reg [PTR_WIDTH-1:0] rd_ptr_gray_src;

    // Destination domain
    reg [ADDR_WIDTH-1:0] rd_ptr;
    reg [PTR_WIDTH-1:0] rd_ptr_gray;
    reg [PTR_WIDTH-1:0] wr_ptr_gray_dst;
    reg [WIDTH-1:0] dst_reg;
    reg dst_valid_reg;

    // Gray code conversion functions
    function [PTR_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_WIDTH-1:0] gray2bin;
        input [PTR_WIDTH-1:0] gray;
        reg [PTR_WIDTH-1:0] bin;
        integer i;
        begin
            bin[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1)
                bin[i] = bin[i+1] ^ gray[i];
            gray2bin = bin;
        end
    endfunction

    // Source domain logic
    wire [PTR_WIDTH-1:0] wr_ptr_bin = gray2bin(wr_ptr_gray);
    wire [ADDR_WIDTH-1:0] wr_addr = wr_ptr_bin[ADDR_WIDTH-1:0];
    wire [PTR_WIDTH-1:0] rd_ptr_bin_src = gray2bin(rd_ptr_gray_src);
    wire [ADDR_WIDTH-1:0] rd_addr_src = rd_ptr_bin_src[ADDR_WIDTH-1:0];

    wire src_full = (wr_ptr_gray == rd_ptr_gray_src) && (wr_ptr_bin[PTR_WIDTH-1] != rd_ptr_bin_src[PTR_WIDTH-1]);
    wire src_can_write = src_valid_i && !src_full;

    assign src_ready_o = !src_full;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            wr_ptr_gray <= {PTR_WIDTH{1'b0}};
        end else if (src_can_write) begin
            mem[wr_addr] <= src_data_i;
            wr_ptr_gray <= bin2gray(wr_ptr_bin + 1'b1);
        end
    end

    // Synchronize rd_ptr_gray from dst to src
    reg [PTR_WIDTH-1:0] rd_ptr_gray_sync1;
    reg [PTR_WIDTH-1:0] rd_ptr_gray_sync2;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            rd_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            rd_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            rd_ptr_gray_sync1 <= rd_ptr_gray;
            rd_ptr_gray_sync2 <= rd_ptr_gray_sync1;
        end
    end

    assign rd_ptr_gray_src = rd_ptr_gray_sync2;

    // Destination domain logic
    wire [PTR_WIDTH-1:0] rd_ptr_bin = gray2bin(rd_ptr_gray);
    wire [ADDR_WIDTH-1:0] rd_addr = rd_ptr_bin[ADDR_WIDTH-1:0];
    wire [PTR_WIDTH-1:0] wr_ptr_bin_dst = gray2bin(wr_ptr_gray_dst);
    wire [ADDR_WIDTH-1:0] wr_addr_dst = wr_ptr_bin_dst[ADDR_WIDTH-1:0];

    wire dst_empty = (rd_ptr_gray == wr_ptr_gray_dst);
    wire dst_can_read = dst_valid_reg && dst_ready_i;

    assign dst_valid_o = dst_valid_reg;
    assign dst_data_o = dst_reg;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            rd_ptr_gray <= {PTR_WIDTH{1'b0}};
            dst_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_can_read) begin
                dst_valid_reg <= 1'b0;
                rd_ptr_gray <= bin2gray(rd_ptr_bin + 1'b1);
            end else if (!dst_empty && !dst_valid_reg) begin
                dst_reg <= mem[rd_addr];
                dst_valid_reg <= 1'b1;
            end
        end
    end

    // Synchronize wr_ptr_gray from src to dst
    reg [PTR_WIDTH-1:0] wr_ptr_gray_sync1;
    reg [PTR_WIDTH-1:0] wr_ptr_gray_sync2;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            wr_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            wr_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            wr_ptr_gray_sync1 <= wr_ptr_gray;
            wr_ptr_gray_sync2 <= wr_ptr_gray_sync1;
        end
    end

    assign wr_ptr_gray_dst = wr_ptr_gray_sync2;

endmodule
