module edge_propagator (
  input  logic clk_tx_i,
  input  logic rstn_tx_i,
  input  logic edge_i,
  input  logic clk_rx_i,
  input  logic rstn_rx_i,
  output logic edge_o
);

  // TX domain: detect edge and generate a pulse to toggle the level
  logic edge_d1, edge_d2;
  logic toggle_tx;
  logic toggle_tx_d1, toggle_tx_d2;

  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i) begin
      edge_d1 <= 1'b0;
      edge_d2 <= 1'b0;
      toggle_tx <= 1'b0;
    end else begin
      edge_d1 <= edge_i;
      edge_d2 <= edge_d1;
      if (edge_d1 && !edge_d2) begin
        toggle_tx <= ~toggle_tx;
      end
    end
  end

  // Synchronize toggle_tx to RX domain
  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      toggle_tx_d1 <= 1'b0;
      toggle_tx_d2 <= 1'b0;
    end else begin
      toggle_tx_d1 <= toggle_tx;
      toggle_tx_d2 <= toggle_tx_d1;
    end
  end

  // RX domain: detect edge on synchronized toggle and generate output pulse
  logic toggle_rx_d1, toggle_rx_d2;
  logic edge_rx;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      toggle_rx_d1 <= 1'b0;
      toggle_rx_d2 <= 1'b0;
      edge_o <= 1'b0;
    end else begin
      toggle_rx_d1 <= toggle_tx_d2;
      toggle_rx_d2 <= toggle_rx_d1;
      edge_o <= (toggle_rx_d1 != toggle_rx_d2);
    end
  end

endmodule
