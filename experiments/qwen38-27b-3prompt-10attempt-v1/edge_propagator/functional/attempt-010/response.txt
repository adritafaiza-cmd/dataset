module edge_propagator (
  input  logic clk_tx_i,
  input  logic rstn_tx_i,
  input  logic edge_i,
  input  logic clk_rx_i,
  input  logic rstn_rx_i,
  output logic edge_o
);

  // TX domain: detect edge and generate a toggle
  logic tx_edge_detected;
  logic tx_toggle;
  logic tx_toggle_reg;

  // Detect rising edge of edge_i in TX domain
  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i) begin
      tx_toggle_reg <= 1'b0;
    end else if (edge_i) begin
      tx_toggle_reg <= ~tx_toggle_reg;
    end
  end

  // Synchronize tx_toggle_reg to RX domain
  logic rx_sync1;
  logic rx_sync2;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      rx_sync1 <= 1'b0;
      rx_sync2 <= 1'b0;
    end else begin
      rx_sync1 <= tx_toggle_reg;
      rx_sync2 <= rx_sync1;
    end
  end

  // Detect edge on synchronized signal in RX domain
  logic rx_edge;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      edge_o <= 1'b0;
    end else begin
      edge_o <= (rx_sync2 != rx_sync1);
    end
  end

endmodule
