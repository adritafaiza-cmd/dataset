module edge_propagator (
  input  logic clk_tx_i,
  input  logic rstn_tx_i,
  input  logic edge_i,
  input  logic clk_rx_i,
  input  logic rstn_rx_i,
  output logic edge_o
);

  // TX domain: detect rising edge of edge_i and generate a pulse
  logic edge_i_d1, edge_i_d2;
  logic tx_pulse;

  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i) begin
      edge_i_d1 <= 1'b0;
      edge_i_d2 <= 1'b0;
    end else begin
      edge_i_d1 <= edge_i;
      edge_i_d2 <= edge_i_d1;
    end
  end

  assign tx_pulse = edge_i_d1 & ~edge_i_d2;

  // TX domain: pulse to level converter
  logic tx_level;

  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i)
      tx_level <= 1'b0;
    else if (tx_pulse)
      tx_level <= 1'b1;
  end

  // Synchronize tx_level into RX domain
  logic sync_d1, sync_d2, sync_d3;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      sync_d1 <= 1'b0;
      sync_d2 <= 1'b0;
      sync_d3 <= 1'b0;
    end else begin
      sync_d1 <= tx_level;
      sync_d2 <= sync_d1;
      sync_d3 <= sync_d2;
    end
  end

  // RX domain: detect rising edge of synchronized level and generate output pulse
  logic rx_level_d1;
  logic rx_pulse;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i)
      rx_level_d1 <= 1'b0;
    else
      rx_level_d1 <= sync_d3;
  end

  assign rx_pulse = sync_d3 & ~rx_level_d1;

  // RX domain: clear tx_level by sending a clear pulse back to TX domain
  // Use a handshake: when rx_pulse occurs, assert rx_clear which is synchronized back to TX
  logic rx_clear;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i)
      rx_clear <= 1'b0;
    else if (rx_pulse)
      rx_clear <= 1'b1;
  end

  // Synchronize rx_clear into TX domain
  logic clear_sync_d1, clear_sync_d2;

  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i) begin
      clear_sync_d1 <= 1'b0;
      clear_sync_d2 <= 1'b0;
    end else begin
      clear_sync_d1 <= rx_clear;
      clear_sync_d2 <= clear_sync_d1;
    end
  end

  // TX domain: clear tx_level when clear signal arrives
  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i)
      tx_level <= 1'b0;
    else if (clear_sync_d2)
      tx_level <= 1'b0;
    else if (tx_pulse)
      tx_level <= 1'b1;
  end

  // RX domain: deassert rx_clear after it has been acknowledged (simplified: deassert after one cycle)
  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i)
      rx_clear <= 1'b0;
    else if (rx_pulse)
      rx_clear <= 1'b1;
    else
      rx_clear <= 1'b0;
  end

  assign edge_o = rx_pulse;

endmodule
