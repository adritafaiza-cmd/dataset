module edge_propagator (
  input  logic clk_tx_i,
  input  logic rstn_tx_i,
  input  logic edge_i,
  input  logic clk_rx_i,
  input  logic rstn_rx_i,
  output logic edge_o
);

  // TX side: capture edge_i into a level signal
  logic tx_level;

  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i)
      tx_level <= 1'b0;
    else if (edge_i)
      tx_level <= 1'b1;
  end

  // Synchronize tx_level into RX domain
  logic rx_sync0, rx_sync1;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      rx_sync0 <= 1'b0;
      rx_sync1 <= 1'b0;
    end else begin
      rx_sync0 <= tx_level;
      rx_sync1 <= rx_sync0;
    end
  end

  // RX side: detect rising edge of rx_sync1 to produce a single-cycle pulse
  logic rx_prev;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i)
      rx_prev <= 1'b0;
    else
      rx_prev <= rx_sync1;
  end

  assign edge_o = rx_sync1 & ~rx_prev;

endmodule
