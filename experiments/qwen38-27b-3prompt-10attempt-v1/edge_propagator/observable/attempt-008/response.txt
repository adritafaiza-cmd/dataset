module edge_propagator (
  input  logic clk_tx_i,
  input  logic rstn_tx_i,
  input  logic edge_i,
  input  logic clk_rx_i,
  input  logic rstn_rx_i,
  output logic edge_o
);

  // TX domain: capture edge_i and generate a pulse to toggle the level
  logic tx_edge_d;
  logic tx_pulse;
  logic tx_level;

  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i) begin
      tx_edge_d <= 1'b0;
      tx_level  <= 1'b0;
    end else begin
      tx_edge_d <= edge_i;
      if (edge_i && !tx_edge_d)
        tx_level <= ~tx_level;
    end
  end

  assign tx_pulse = tx_level;

  // Synchronizer: 2-FF synchronizer for tx_level into RX domain
  logic rx_sync0;
  logic rx_sync1;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      rx_sync0 <= 1'b0;
      rx_sync1 <= 1'b0;
    end else begin
      rx_sync0 <= tx_pulse;
      rx_sync1 <= rx_sync0;
    end
  end

  // RX domain: detect rising edge of synchronized level to produce one-cycle pulse
  logic rx_level_d;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      rx_level_d <= 1'b0;
      edge_o     <= 1'b0;
    end else begin
      rx_level_d <= rx_sync1;
      edge_o     <= rx_sync1 && !rx_level_d;
    end
  end

endmodule
