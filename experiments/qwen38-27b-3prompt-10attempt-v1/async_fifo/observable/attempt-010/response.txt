```verilog
module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    localparam AWIDTH = ASIZE;
    localparam DEPTH  = 1 << ASIZE;
    localparam AWIDTH_PLUS_1 = ASIZE + 1;

    // Write domain
    reg  [AWIDTH-1:0] wptr;
    reg  [AWIDTH_PLUS_1-1:0] wptr_gray;
    reg  [AWIDTH_PLUS_1-1:0] wptr_gray_next;
    reg  [AWIDTH_PLUS_1-1:0] rptr_gray_w;
    reg  [AWIDTH_PLUS_1-1:0] rptr_gray_w1;
    reg  [AWIDTH_PLUS_1-1:0] rptr_gray_w2;

    // Read domain
    reg  [AWIDTH-1:0] rptr;
    reg  [AWIDTH_PLUS_1-1:0] rptr_gray;
    reg  [AWIDTH_PLUS_1-1:0] rptr_gray_next;
    reg  [AWIDTH_PLUS_1-1:0] wptr_gray_r;
    reg  [AWIDTH_PLUS_1-1:0] wptr_gray_r1;
    reg  [AWIDTH_PLUS_1-1:0] wptr_gray_r2;

    // Memory
    reg  [DSIZE-1:0] mem [0:DEPTH-1];

    // Write enable
    wire wfull_local;
    wire awfull_local;
    wire winc_eff;

    // Read enable
    wire rempty_local;
    wire arempty_local;
    wire rinc_eff;

    // Gray code conversion
    function [AWIDTH_PLUS_1-1:0] bin2gray;
        input [AWIDTH_PLUS_1-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Write pointer update
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {AWIDTH{1'b0}};
            wptr_gray <= {AWIDTH_PLUS_1{1'b0}};
        end else begin
            wptr <= wptr + (winc_eff ? 1'b1 : 1'b0);
            wptr_gray <= wptr_gray_next;
        end
    end

    assign wptr_gray_next = bin2gray(wptr + (winc_eff ? 1'b1 : 1'b0));

    // Read pointer update
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {AWIDTH{1'b0}};
            rptr_gray <= {AWIDTH_PLUS_1{1'b0}};
        end else begin
            rptr <= rptr + (rinc_eff ? 1'b1 : 1'b0);
            rptr_gray <= rptr_gray_next;
        end
    end

    assign rptr_gray_next = bin2gray(rptr + (rinc_eff ? 1'b1 : 1'b0));

    // Memory write
    always @(posedge wclk) begin
        if (winc_eff) begin
            mem[wptr] <= wdata;
        end
    end

    // Memory read
    reg [DSIZE-1:0] rdata_reg;
    always @(posedge rclk) begin
        if (rinc_eff) begin
            rdata_reg <= mem[rptr];
        end
    end

    // Fall-through logic
    wire fallthrough_en = (FALLTHROUGH == "TRUE");
    wire fallthrough_valid = fallthrough_en & rempty_local & winc_eff;
    assign rdata = fallthrough_valid ? wdata : rdata_reg;

    // Synchronize read pointer gray to write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_w <= {AWIDTH_PLUS_1{1'b0}};
            rptr_gray_w1 <= {AWIDTH_PLUS_1{1'b0}};
            rptr_gray_w2 <= {AWIDTH_PLUS_1{1'b0}};
        end else begin
            rptr_gray_w <= rptr_gray;
            rptr_gray_w1 <= rptr_gray_w;
            rptr_gray_w2 <= rptr_gray_w1;
        end
    end

    // Synchronize write pointer gray to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_r <= {AWIDTH_PLUS_1{1'b0}};
            wptr_gray_r1 <= {AWIDTH_PLUS_1{1'b0}};
            wptr_gray_r2 <= {AWIDTH_PLUS_1{1'b0}};
        end else begin
            wptr_gray_r <= wptr_gray;
            wptr_gray_r1 <= wptr_gray_r;
            wptr_gray_r2 <= wptr_gray_r1;
        end
    end

    // Full/empty detection in write domain
    // Full: write pointer gray MSB differs from read pointer gray MSB, and lower bits are complement
    wire [AWIDTH_PLUS_1-1:0] wptr_gray_w2;
    assign wptr_gray_w2 = wptr_gray;

    wire wfull_raw = (wptr_gray_w2[AWIDTH] != rptr_gray_w2[AWIDTH]) &&
                     (wptr_gray_w2[AWIDTH-1:0] == ~rptr_gray_w2[AWIDTH-1:0]);

    // Almost full: one less than full
    wire [AWIDTH_PLUS_1-1:0] wptr_gray_next_w2;
    assign wptr_gray_next_w2 = wptr_gray_next;

    wire awfull_raw = (wptr_gray_next_w2[AWIDTH] != rptr_gray_w2[AWIDTH]) &&
                      (wptr_gray_next_w2[AWIDTH-1:0] == ~rptr_gray_w2[AWIDTH-1:0]);

    assign wfull = wfull_raw;
    assign awfull = awfull_raw;

    // Empty detection in read domain
    wire rempty_raw = (wptr_gray_r2 == rptr_gray);

    // Almost empty: one more than empty
    wire [AWIDTH_PLUS_1-1:0] rptr_gray_next_r2;
    assign rptr_gray_next_r2 = rptr_gray_next;

    wire arempty_raw = (wptr_gray_r2 == rptr_gray_next_r2);

    assign rempty = rempty_raw;
    assign arempty = arempty_raw;

    // Effective write enable: only write if not full
    assign winc_eff = winc & ~wfull;

    // Effective read enable: only read if not empty
    assign rinc_eff = rinc & ~rempty;

endmodule
