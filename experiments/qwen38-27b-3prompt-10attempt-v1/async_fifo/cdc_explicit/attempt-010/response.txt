```verilog
module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    localparam DEPTH = 1 << ASIZE;
    localparam AWIDTH = ASIZE + 1;
    localparam FULL_THRESHOLD = DEPTH - 1;
    localparam EMPTY_THRESHOLD = 1;

    // Write domain
    reg [AWIDTH-1:0] wptr;
    reg [ASIZE-1:0]  wptr_gray;
    reg [ASIZE-1:0]  rptr_gray_w;
    reg [ASIZE-1:0]  rptr_gray_w_sync1;
    reg [ASIZE-1:0]  rptr_gray_w_sync2;

    // Read domain
    reg [AWIDTH-1:0] rptr;
    reg [ASIZE-1:0]  rptr_gray;
    reg [ASIZE-1:0]  wptr_gray_r;
    reg [ASIZE-1:0]  wptr_gray_r_sync1;
    reg [ASIZE-1:0]  wptr_gray_r_sync2;

    // Memory
    reg [DSIZE-1:0] mem [0:DEPTH-1];

    // Write enable
    wire wvalid = winc & ~wfull;

    // Read enable
    wire rvalid = rinc & ~rempty;

    // Gray code conversion
    function [ASIZE-1:0] bin2gray;
        input [ASIZE-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Write pointer update
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {AWIDTH{1'b0}};
        end else if (wvalid) begin
            wptr <= wptr + 1'b1;
        end
    end

    // Write pointer gray code
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr_gray <= {ASIZE{1'b0}};
        end else begin
            wptr_gray <= bin2gray(wptr[ASIZE:1]);
        end
    end

    // Read pointer gray code
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {AWIDTH{1'b0}};
        end else if (rvalid) begin
            rptr <= rptr + 1'b1;
        end
    end

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr_gray <= {ASIZE{1'b0}};
        end else begin
            rptr_gray <= bin2gray(rptr[ASIZE:1]);
        end
    end

    // Synchronize read pointer gray to write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_w_sync1 <= {ASIZE{1'b0}};
            rptr_gray_w_sync2 <= {ASIZE{1'b0}};
        end else begin
            rptr_gray_w_sync1 <= rptr_gray;
            rptr_gray_w_sync2 <= rptr_gray_w_sync1;
        end
    end

    // Synchronize write pointer gray to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_r_sync1 <= {ASIZE{1'b0}};
            wptr_gray_r_sync2 <= {ASIZE{1'b0}};
        end else begin
            wptr_gray_r_sync1 <= wptr_gray;
            wptr_gray_r_sync2 <= wptr_gray_r_sync1;
        end
    end

    // Full detection in write domain
    // Full when write pointer has lapped read pointer by DEPTH
    // In gray code, full when wptr_gray[ASIZE] == rptr_gray_sync[ASIZE] and wptr_gray[ASIZE-1:0] == ~rptr_gray_sync[ASIZE-1:0]
    wire [ASIZE:0] wptr_gray_ext = {wptr[ASIZE], wptr_gray};
    wire [ASIZE:0] rptr_gray_w_ext = {rptr_gray_w_sync2[ASIZE-1], rptr_gray_w_sync2};
    
    // Actually, let's use the standard approach:
    // Full when the MSB of write pointer equals MSB of read pointer, and the lower bits are inverted
    wire wfull_internal = (wptr[ASIZE] == rptr_gray_w_sync2[ASIZE-1]) && 
                          (wptr[ASIZE-1:0] == ~rptr_gray_w_sync2);
    
    // Wait, the standard gray code full detection:
    // wptr_gray has ASIZE+1 bits conceptually, but we only store ASIZE bits of gray
    // Let me reconsider. The write pointer is AWIDTH bits. The gray code of the upper ASIZE bits is stored.
    // Full condition: wptr[ASIZE] == rptr_gray_sync[ASIZE-1] && wptr[ASIZE-1:0] == ~rptr_gray_sync[ASIZE-1:0]
    
    assign wfull = wfull_internal;

    // Almost full: DEPTH - FULL_THRESHOLD words remaining
    // awfull when (wptr - rptr_sync) >= FULL_THRESHOLD
    // In terms of distance: awfull when the difference is >= FULL_THRESHOLD
    // We can compute this by checking if wptr[ASIZE:1] - rptr_gray_sync[ASIZE:1] >= FULL_THRESHOLD
    // But since we're in gray code, it's easier to convert back or use a different approach.
    
    // Let's use a simpler approach: awfull when the number of free slots <= 1
    // i.e., when wptr - rptr_sync >= DEPTH - 1
    // This is equivalent to wfull being about to happen, so awfull = wfull_internal or one step before
    
    // Actually, let's compute the distance properly.
    // The distance in write domain: wptr - rptr_sync (where rptr_sync is the synchronized read pointer)
    // We need to convert rptr_gray_sync back to binary, or use the property that
    // awfull when wptr[ASIZE:1] - rptr_gray_sync[ASIZE:1] >= FULL_THRESHOLD
    
    // For simplicity and correctness, let's use a counter-based approach or direct comparison.
    // Since ASIZE is small, we can afford a small combinational logic.
    
    // Convert rptr_gray_sync to binary for distance calculation
    function [ASIZE-1:0] gray2bin;
        input [ASIZE-1:0] gray;
        integer i;
        begin
            gray2bin[ASIZE-1] = gray[ASIZE-1];
            for (i = ASIZE-2; i >= 0; i = i - 1)
                gray2bin[i] = gray2bin[i+1] ^ gray[i];
        end
    endfunction

    wire [ASIZE-1:0] rptr_bin_w = gray2bin(rptr_gray_w_sync2);
    wire [ASIZE-1:0] wptr_low = wptr[ASIZE-1:0];
    
    // Distance = wptr - rptr_sync (mod DEPTH)
    // awfull when distance >= FULL_THRESHOLD
    wire [ASIZE:0] dist_w = {1'b0, wptr_low} - {1'b0, rptr_bin_w};
    assign awfull = (dist_w >= FULL_THRESHOLD[ASIZE:0]);

    // Empty detection in read domain
    wire rempty_internal = (rptr[ASIZE] == wptr_gray_r_sync2[ASIZE-1]) && 
                           (rptr[ASIZE-1:0] == wptr_gray_r_sync2);
    assign rempty = rempty_internal;

    // Almost empty: DEPTH - EMPTY_THRESHOLD words remaining
    // arempty when (wptr_sync - rptr) <= EMPTY_THRESHOLD
    // i.e., when distance <= 1
    wire [ASIZE-1:0] wptr_bin_r = gray2bin(wptr_gray_r_sync2);
    wire [ASIZE-1:0] rptr_low = rptr[ASIZE-1:0];
    wire [ASIZE:0] dist_r = {1'b0, wptr_bin_r} - {1'b0, rptr_low};
    assign arempty = (dist_r <= EMPTY_THRESHOLD[ASIZE:0]);

    // Memory write
    always @(posedge wclk) begin
        if (wvalid) begin
            mem[wptr[ASIZE-1:0]] <= wdata;
        end
    end

    // Read data
    wire [ASIZE-1:0] raddr = rptr[ASIZE-1:0];
    wire [DSIZE-1:0] mem_rdata = mem[raddr];

    // Fall-through logic
    reg [DSIZE-1:0] rdata_reg;
    
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rdata_reg <= {DSIZE{1'b0}};
        end else if (rvalid) begin
            rdata_reg <= mem_rdata;
        end
    end

    assign rdata = (FALLTHROUGH == "TRUE") ? mem_rdata : rdata_reg;

endmodule
