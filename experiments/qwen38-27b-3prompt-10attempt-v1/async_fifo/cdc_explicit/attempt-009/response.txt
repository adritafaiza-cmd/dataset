```verilog
module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    localparam DEPTH = 1 << ASIZE;
    localparam AWIDTH = ASIZE + 1;
    localparam FULL_THRESHOLD = DEPTH - 1;
    localparam EMPTY_THRESHOLD = 1;

    // Write domain
    reg [AWIDTH-1:0] wptr;
    reg [ASIZE-1:0]  wptr_gray;
    reg [ASIZE-1:0]  rptr_gray_w;
    reg [ASIZE-1:0]  rptr_gray_w_sync1;
    reg [ASIZE-1:0]  rptr_gray_w_sync2;

    // Read domain
    reg [AWIDTH-1:0] rptr;
    reg [ASIZE-1:0]  rptr_gray;
    reg [ASIZE-1:0]  wptr_gray_r;
    reg [ASIZE-1:0]  wptr_gray_r_sync1;
    reg [ASIZE-1:0]  wptr_gray_r_sync2;

    // Memory
    reg [DSIZE-1:0] mem [0:DEPTH-1];

    // Write enable
    wire wvalid = winc & ~wfull;
    wire rvalid = rinc & ~rempty;

    // Gray code conversion
    function [ASIZE-1:0] bin2gray;
        input [AWIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Write pointer update
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {AWIDTH{1'b0}};
        end else if (wvalid) begin
            wptr <= wptr + 1'b1;
        end
    end

    // Write pointer gray code
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr_gray <= {ASIZE{1'b0}};
        end else begin
            wptr_gray <= bin2gray(wptr)[ASIZE-1:0];
        end
    end

    // Read pointer gray code (read domain)
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {AWIDTH{1'b0}};
        end else if (rvalid) begin
            rptr <= rptr + 1'b1;
        end
    end

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr_gray <= {ASIZE{1'b0}};
        end else begin
            rptr_gray <= bin2gray(rptr)[ASIZE-1:1];
        end
    end

    // Synchronize read pointer gray to write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_w_sync1 <= {ASIZE{1'b0}};
            rptr_gray_w_sync2 <= {ASIZE{1'b0}};
        end else begin
            rptr_gray_w_sync1 <= rptr_gray;
            rptr_gray_w_sync2 <= rptr_gray_w_sync1;
        end
    end

    // Synchronize write pointer gray to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_r_sync1 <= {ASIZE{1'b0}};
            wptr_gray_r_sync2 <= {ASIZE{1'b0}};
        end else begin
            wptr_gray_r_sync1 <= wptr_gray;
            wptr_gray_r_sync2 <= wptr_gray_r_sync1;
        end
    end

    // Full detection in write domain
    // Full when write pointer has lapped read pointer by DEPTH
    // In gray code: wptr_gray == rptr_gray_w_sync2 with MSB of wptr != MSB of rptr
    wire [ASIZE-1:0] wptr_gray_full = bin2gray(wptr)[ASIZE-1:0];
    wire wfull_internal = (wptr_gray_full == rptr_gray_w_sync2) && 
                          (wptr[ASIZE] != rptr_gray_w_sync2[ASIZE-1]);
    assign wfull = wfull_internal;

    // Almost full: one less than full
    wire [AWIDTH-1:0] wptr_next = wptr + 1'b1;
    wire [ASIZE-1:0] wptr_gray_next = bin2gray(wptr_next)[ASIZE-1:0];
    wire awfull_internal = (wptr_gray_next == rptr_gray_w_sync2) && 
                           (wptr_next[ASIZE] != rptr_gray_w_sync2[ASIZE-1]);
    assign awfull = awfull_internal;

    // Empty detection in read domain
    // Empty when read pointer == write pointer
    wire [ASIZE-1:0] rptr_gray_empty = bin2gray(rptr)[ASIZE-1:0];
    wire rempty_internal = (rptr_gray_empty == wptr_gray_r_sync2) && 
                           (rptr[ASIZE] == wptr_gray_r_sync2[ASIZE-1]);
    assign rempty = rempty_internal;

    // Almost empty: one more than empty
    wire [AWIDTH-1:0] rptr_next = rptr + 1'b1;
    wire [ASIZE-1:0] rptr_gray_next = bin2gray(rptr_next)[ASIZE-1:0];
    wire arempty_internal = (rptr_gray_next == wptr_gray_r_sync2) && 
                            (rptr_next[ASIZE] == wptr_gray_r_sync2[ASIZE-1]);
    assign arempty = arempty_internal;

    // Memory write
    always @(posedge wclk) begin
        if (wvalid) begin
            mem[wptr[ASIZE-1:0]] <= wdata;
        end
    end

    // Read data output
    reg [DSIZE-1:0] rdata_reg;
    wire [ASIZE-1:0] raddr = rptr[ASIZE-1:0];

    if (FALLTHROUGH == "TRUE") begin
        assign rdata = mem[raddr];
    end else begin
        always @(posedge rclk or negedge rrst_n) begin
            if (!rrst_n) begin
                rdata_reg <= {DSIZE{1'b0}};
            end else if (rvalid) begin
                rdata_reg <= mem[raddr];
            end
        end
        assign rdata = rdata_reg;
    end

endmodule
