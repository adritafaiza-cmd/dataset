        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,

        input  wire             rrst_n,
        input  wire             rinc,

    localparam AWIDTH = ASIZE + 1;
    localparam AWIDTH_M1 = AWIDTH - 1;

        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

        begin
            bin[ASIZE-1] = gray[ASIZE-1];
            for (i = ASIZE-2; i >= 0; i = i - 1)
                bin[i] = gray[i] ^ bin[i+1];
            gray2bin = bin;
        end
    endfunction

            rptr_gray <= bin2gray(rptr_bin[ASIZE-1:0] + 1'b1);
        end
    end

    // Memory write
    always @(posedge wclk) begin
        if (winc_eff)
            mem[wptr_bin[ASIZE-1:0]] <= wdata;
    end

    // Memory read
    wire [ASIZE-1:0] raddr;
    assign raddr = rptr_bin[ASIZE-1:0];

    // Read data
    reg [DSIZE-1:0] rdata_reg;
    always @(posedge rclk) begin
        if (rinc_eff)
            rdata_reg <= mem[raddr];
    end

    assign rdata = (FALLTHROUGH == "TRUE") ? mem[raddr] : rdata_reg;

            wptr_gray_r_sync1 <= {ASIZE{1'b0}};
            wptr_gray_r_sync2 <= {ASIZE{1'b0}};

            wptr_gray_r_sync1 <= wptr_gray;
            wptr_gray_r_sync2 <= wptr_gray_r_sync1;

            rptr_gray_w_sync1 <= {ASIZE{1'b0}};
            rptr_gray_w_sync2 <= {ASIZE{1'b0}};
        end else begin
            rptr_gray_w_sync1 <= rptr_gray;
            rptr_gray_w_sync2 <= rptr_gray_w_sync1;
        end
    end

endmodule
