module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    localparam DEPTH = 1 << ASIZE;
    localparam AWFULL_THRESH = DEPTH - 2;
    localparam AEMPTY_THRESH = 2;

    // Write pointer (ASIZE+1 bits: MSB is extra bit for full detection)
    reg [ASIZE:0] wptr;
    // Read pointer (ASIZE+1 bits)
    reg [ASIZE:0] rptr;

    // Memory
    reg [DSIZE-1:0] mem [0:DEPTH-1];

    // Write enable
    wire wvalid = winc & ~wfull;

    // Read enable
    wire rvalid = rinc & ~rempty;

    // Write pointer increment
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n)
            wptr <= 0;
        else if (wvalid)
            wptr <= wptr + 1;
    end

    // Read pointer increment
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n)
            rptr <= 0;
        else if (rvalid)
            rptr <= rptr + 1;
    end

    // Memory write
    always @(posedge wclk) begin
        if (wvalid)
            mem[wptr[ASIZE-1:0]] <= wdata;
    end

    // Gray code conversion for write pointer
    wire [ASIZE:0] wptr_gray = wptr ^ {wptr[ASIZE:0], 1'b0}[ASIZE+1:1];
    // Actually, standard gray code: gray = bin ^ (bin >> 1)
    wire [ASIZE:0] wptr_gray_calc = wptr ^ (wptr >> 1);

    // Gray code conversion for read pointer
    wire [ASIZE:0] rptr_gray_calc = rptr ^ (rptr >> 1);

    // Synchronize write pointer gray to read domain
    reg [ASIZE:0] wptr_gray_r1, wptr_gray_r2;
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_r1 <= 0;
            wptr_gray_r2 <= 0;
        end else begin
            wptr_gray_r1 <= wptr_gray_calc;
            wptr_gray_r2 <= wptr_gray_r1;
        end
    end

    // Synchronize read pointer gray to write domain
    reg [ASIZE:0] rptr_gray_r1, rptr_gray_r2;
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_r1 <= 0;
            rptr_gray_r2 <= 0;
        end else begin
            rptr_gray_r1 <= rptr_gray_calc;
            rptr_gray_r2 <= rptr_gray_r1;
        end
    end

    // Convert synchronized gray codes back to binary
    // Gray to binary conversion
    function [ASIZE:0] gray2bin;
        input [ASIZE:0] gray;
        integer i;
        begin
            gray2bin[ASIZE] = gray[ASIZE];
            for (i = ASIZE-1; i >= 0; i = i-1)
                gray2bin[i] = gray2bin[i+1] ^ gray[i];
        end
    endfunction

    wire [ASIZE:0] wptr_gray_synced = gray2bin(wptr_gray_r2);
    wire [ASIZE:0] rptr_gray_synced = gray2bin(rptr_gray_r2);

    // Full condition: write pointer MSB differs from read pointer MSB,
    // and lower bits are equal
    wire wfull = (wptr[ASIZE] != rptr_gray_synced[ASIZE]) &&
                 (wptr[ASIZE-1:0] == rptr_gray_synced[ASIZE-1:0]);

    // Empty condition: read pointer MSB differs from write pointer MSB,
    // and lower bits are equal
    wire rempty = (rptr[ASIZE] != wptr_gray_synced[ASIZE]) &&
                  (rptr[ASIZE-1:0] == wptr_gray_synced[ASIZE-1:0]);

    // Almost full: number of entries >= AWFULL_THRESH
    // Count = wptr - rptr (mod 2^(ASIZE+1))
    // In the write domain, we have wptr and rptr_gray_synced
    // We need to compute the fill level in the write domain
    wire [ASIZE:0] fill_level_w = wptr - rptr_gray_synced;
    wire awfull = (fill_level_w >= AWFULL_THRESH);

    // Almost empty: number of entries <= AEMPTY_THRESH
    // In the read domain, we have rptr and wptr_gray_synced
    wire [ASIZE:0] fill_level_r = wptr_gray_synced - rptr;
    wire arempty = (fill_level_r <= AEMPTY_THRESH);

    // Read data
    // For fall-through: rdata is the current oldest unread word
    // For non-fall-through: rdata is registered
    reg [DSIZE-1:0] rdata_reg;
    wire [DSIZE-1:0] mem_rdata = mem[rptr[ASIZE-1:0]];

    assign rdata = (FALLTHROUGH == "TRUE") ? mem_rdata : rdata_reg;

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n)
            rdata_reg <= 0;
        else if (rvalid)
            rdata_reg <= mem[rptr[ASIZE-1:0]];
    end

endmodule
