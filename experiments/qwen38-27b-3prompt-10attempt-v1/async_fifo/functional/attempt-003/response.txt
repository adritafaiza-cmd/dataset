module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    localparam DEPTH = 1 << ASIZE;
    localparam AWFULL_THRESH = DEPTH - 2;
    localparam AEMPTY_THRESH = 1;

    // Memory
    reg [DSIZE-1:0] mem [0:DEPTH-1];

    // Write domain
    reg [ASIZE:0] wptr;
    reg [ASIZE:0] rptr_w;
    reg [ASIZE:0] rptr_w_sync1;
    reg [ASIZE:0] rptr_w_sync2;

    // Read domain
    reg [ASIZE:0] rptr;
    reg [ASIZE:0] wptr_r;
    reg [ASIZE:0] wptr_r_sync1;
    reg [ASIZE:0] wptr_r_sync2;

    // Write enable
    wire wvalid = winc & ~wfull;

    // Read enable
    wire rvalid = rinc & ~rempty;

    // Write pointer update
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n)
            wptr <= {(ASIZE+1){1'b0}};
        else if (wvalid)
            wptr <= wptr + 1'b1;
    end

    // Read pointer update
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n)
            rptr <= {(ASIZE+1){1'b0}};
        else if (rvalid)
            rptr <= rptr + 1'b1;
    end

    // Synchronize read pointer to write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_w_sync1 <= {(ASIZE+1){1'b0}};
            rptr_w_sync2 <= {(ASIZE+1){1'b0}};
        end
        else begin
            rptr_w_sync1 <= rptr;
            rptr_w_sync2 <= rptr_w_sync1;
        end
    end

    // Synchronize write pointer to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_r_sync1 <= {(ASIZE+1){1'b0}};
            wptr_r_sync2 <= {(ASIZE+1){1'b0}};
        end
        else begin
            wptr_r_sync1 <= wptr;
            wptr_r_sync2 <= wptr_r_sync1;
        end
    end

    // Write full: wptr - rptr_w_sync2 == DEPTH
    wire wfull = (wptr[ASIZE] != rptr_w_sync2[ASIZE]) &&
                 (wptr[ASIZE-1:0] == rptr_w_sync2[ASIZE-1:0]);

    // Read empty: rptr - wptr_r_sync2 == 0
    wire rempty = (rptr == wptr_r_sync2);

    // Almost full: wptr - rptr_w_sync2 >= DEPTH - 2
    wire [ASIZE:0] wdiff = wptr - rptr_w_sync2;
    wire awfull = (wdiff >= AWFULL_THRESH);

    // Almost empty: wptr_r_sync2 - rptr <= 1
    wire [ASIZE:0] rdiff = wptr_r_sync2 - rptr;
    wire arempty = (rdiff <= AEMPTY_THRESH);

    // Memory write
    always @(posedge wclk) begin
        if (wvalid)
            mem[wptr[ASIZE-1:0]] <= wdata;
    end

    // Read data
    reg [DSIZE-1:0] rdata_reg;

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n)
            rdata_reg <= {DSIZE{1'b0}};
        else if (rvalid)
            rdata_reg <= mem[rptr[ASIZE-1:0]];
    end

    // Fall-through: combinational read of current oldest word
    wire [DSIZE-1:0] rdata_comb = mem[rptr[ASIZE-1:0]];

    assign rdata = (FALLTHROUGH == "TRUE") ? rdata_comb : rdata_reg;

endmodule
