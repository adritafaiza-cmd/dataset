module areset_sync #(
    parameter STAGES = 2
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);

    localparam N = (STAGES < 1) ? 1 : STAGES;

    reg [N-1:0] sync_reg;

    always @(posedge clk or posedge async_rst_i) begin
        if (async_rst_i)
            sync_reg <= {N{1'b1}};
        else
            sync_reg <= {sync_reg[N-2:0], 1'b0};
    end

    assign sync_rst_o = sync_reg[N-1];

endmodule
