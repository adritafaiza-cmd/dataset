module areset_deassert_sync #(
    parameter CHAINS = 2,
    parameter RST_POL = 1'b1
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);

    // Normalize reset polarity: internal active-high reset
    wire rst_active = (RST_POL == 1'b1) ? async_rst_i : ~async_rst_i;

    // Synchronizer chain for deassertion (release)
    reg [CHAINS-1:0] sync_chain;

    // Initialize chain to reset state (active)
    integer i;
    initial begin
        for (i = 0; i < CHAINS; i = i + 1)
            sync_chain[i] = 1'b1;
    end

    always @(posedge clk or posedge rst_active) begin
        if (rst_active) begin
            sync_chain <= {CHAINS{1'b1}};
        end else begin
            sync_chain[0] <= 1'b0;
            for (i = 1; i < CHAINS; i = i + 1)
                sync_chain[i] <= sync_chain[i-1];
        end
    end

    // Output: active if either the async reset is asserted OR the sync chain is not fully released
    // When rst_active is high, output is active immediately.
    // When rst_active is low, output follows the last stage of the sync chain.
    // The sync chain starts all 1s (active) and shifts in 0s.
    // Output is active if rst_active OR sync_chain[CHAINS-1] is 1.
    assign sync_rst_o = rst_active | sync_chain[CHAINS-1];

endmodule
