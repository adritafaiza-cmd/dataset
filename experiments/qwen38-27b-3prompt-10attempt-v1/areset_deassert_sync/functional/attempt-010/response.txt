module areset_deassert_sync #(
    parameter CHAINS = 2,
    parameter RST_POL = 1'b1
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);

    localparam DEASSERT_POL = ~RST_POL;

    reg [CHAINS-1:0] sync_chain;

    always @(posedge clk or posedge async_rst_i) begin
        if (async_rst_i)
            sync_chain <= {CHAINS{1'b1}};
        else
            sync_chain <= {sync_chain[CHAINS-2:0], DEASSERT_POL};
    end

    assign sync_rst_o = RST_POL ? (async_rst_i | ~sync_chain[CHAINS-1])
                                : (async_rst_i & ~sync_chain[CHAINS-1]);

endmodule
