module axis_adapter #
(
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1
)
(
    input  wire                     clk,
    input  wire                     rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser
);

    localparam S_BYTES = (S_DATA_WIDTH + 7) / 8;
    localparam M_BYTES = (M_DATA_WIDTH + 7) / 8;
    localparam MAX_BYTES = (S_BYTES > M_BYTES) ? S_BYTES : M_BYTES;
    localparam MAX_BITS = MAX_BYTES * 8;

    // State machine
    localparam ST_IDLE = 2'd0;
    localparam ST_SPLIT = 2'd1;
    localparam ST_MERGE = 2'd2;

    reg [1:0] state;
    reg [MAX_BITS-1:0] buffer;
    reg [3:0] buf_bytes; // number of valid bytes in buffer (0 to MAX_BYTES)
    reg [3:0] s_bytes;   // number of valid bytes from current s beat
    reg [3:0] m_bytes;   // number of valid bytes to output in current m beat
    reg s_tlast_reg;
    reg [ID_WIDTH-1:0] s_tid_reg;
    reg [DEST_WIDTH-1:0] s_tdest_reg;
    reg [USER_WIDTH-1:0] s_tuser_reg;

    // Helper: count valid bytes from tkeep
    function [3:0] count_bytes;
        input [S_KEEP_WIDTH-1:0] keep;
        integer i;
        begin
            count_bytes = 0;
            for (i = 0; i < S_KEEP_WIDTH; i = i + 1)
                if (keep[i])
                    count_bytes = count_bytes + 1;
        end
    endfunction

    // Helper: extract bytes from s_axis_tdata based on tkeep
    // We'll build the buffer by shifting in valid bytes
    // For simplicity, we assume tkeep is contiguous from LSB (standard AXI-Stream)
    // But to be safe, we handle general case by packing valid bytes

    // Function to pack valid bytes from s_axis_tdata into a byte array
    // Returns packed bytes in lower bits
    function [MAX_BITS-1:0] pack_s_bytes;
        input [S_DATA_WIDTH-1:0] data;
        input [S_KEEP_WIDTH-1:0] keep;
        integer i, byte_idx;
        reg [7:0] tmp_byte;
        begin
            pack_s_bytes = 0;
            byte_idx = 0;
            for (i = 0; i < S_KEEP_WIDTH; i = i + 1) begin
                if (keep[i]) begin
                    tmp_byte = data[i*8 +: 8];
                    pack_s_bytes[byte_idx*8 +: 8] = tmp_byte;
                    byte_idx = byte_idx + 1;
                end
            end
        end
    endfunction

    // Function to unpack bytes from buffer to m_axis_tdata
    function [M_DATA_WIDTH-1:0] unpack_m_bytes;
        input [MAX_BITS-1:0] buf;
        input [3:0] num_bytes;
        integer i;
        begin
            unpack_m_bytes = 0;
            for (i = 0; i < M_KEEP_WIDTH; i = i + 1) begin
                if (i < num_bytes)
                    unpack_m_bytes[i*8 +: 8] = buf[i*8 +: 8];
                else
                    unpack_m_bytes[i*8 +: 8] = 8'h00;
            end
        end
    endfunction

    // Function to generate m_axis_tkeep
    function [M_KEEP_WIDTH-1:0] gen_m_keep;
        input [3:0] num_bytes;
        integer i;
        begin
            gen_m_keep = 0;
            for (i = 0; i < M_KEEP_WIDTH; i = i + 1) begin
                if (i < num_bytes)
                    gen_m_keep[i] = 1'b1;
                else
                    gen_m_keep[i] = 1'b0;
            end
        end
    endfunction

    // Determine if we can accept a new s beat
    wire s_can_accept = (state == ST_IDLE) || 
                        ((state == ST_SPLIT) && (buf_bytes + s_bytes <= MAX_BYTES));

    // Determine if we can output an m beat
    wire m_can_output = (state != ST_IDLE) && (buf_bytes > 0);

    // Calculate how many bytes to output in current m beat
    wire [3:0] bytes_to_output = (buf_bytes >= M_BYTES) ? M_BYTES[3:0] : buf_bytes;

    // Determine if this m beat is the last
    wire m_is_last = (state == ST_SPLIT) ? (s_tlast_reg && (buf_bytes <= M_BYTES)) :
                     (state == ST_MERGE) ? (s_tlast_reg && (buf_bytes <= M_BYTES)) :
                     1'b0;

    // Assign outputs
    assign s_axis_tready = s_can_accept;
    assign m_axis_tvalid = m_can_output;
    assign m_axis_tdata = unpack_m_bytes(buffer, bytes_to_output);
    assign m_axis_tkeep = gen_m_keep(bytes_to_output);
    assign m_axis_tlast = m_is_last;
    assign m_axis_tid = s_tid_reg;
    assign m_axis_tdest = s_tdest_reg;
    assign m_axis_tuser = s_tuser_reg;

    // State machine logic
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= ST_IDLE;
            buffer <= {MAX_BITS{1'b0}};
            buf_bytes <= 4'd0;
            s_bytes <= 4'd0;
            s_tlast_reg <= 1'b0;
            s_tid_reg <= {ID_WIDTH{1'b0}};
            s_tdest_reg <= {DEST_WIDTH{1'b0}};
            s_tuser_reg <= {USER_WIDTH{1'b0}};
        end else begin
            case (state)
                ST_IDLE: begin
                    if (s_axis_tvalid && s_axis_tready) begin
                        // Accept s beat
                        s_bytes <= count_bytes(s_axis_tkeep);
                        s_tlast_reg <= s_axis_tlast;
                        s_tid_reg <= s_axis_tid;
                        s_tdest_reg <= s_axis_tdest;
                        s_tuser_reg <= s_axis_tuser;
                        
                        // Pack bytes into buffer
                        buffer <= pack_s_bytes(s_axis_tdata, s_axis_tkeep);
                        buf_bytes <= count_bytes(s_axis_tkeep);
                        
                        if (count_bytes(s_axis_tkeep) >= M_BYTES) begin
                            // Need to split or output directly
                            if (count_bytes(s_axis_tkeep) == M_BYTES) begin
                                // Exact fit, output and go to idle
                                state <= ST_IDLE;
                                buf_bytes <= 4'd0;
                            end else begin
                                // Split
                                state <= ST_SPLIT;
                            end
                        end else begin
                            // Merge: need more data
                            state <= ST_MERGE;
                        end
                    end
                end
                
                ST_SPLIT: begin
                    if (m_axis_tvalid && m_axis_tready) begin
                        // Output M_BYTES from buffer
                        // Shift buffer left by M_BYTES
                        buffer <= buffer << (M_BYTES * 8);
                        buf_bytes <= buf_bytes - M_BYTES;
                        
                        if (buf_bytes == M_BYTES) begin
                            // Exactly one more m beat needed
                            state <= ST_IDLE;
                            buf_bytes <= 4'd0;
                        end else if (buf_bytes == 0) begin
                            // Done
                            state <= ST_IDLE;
                        end else begin
                            // Still have data, continue splitting
                            state <= ST_SPLIT;
                        end
                    end
                end
                
                ST_MERGE: begin
                    if (s_axis_tvalid && s_axis_tready) begin
                        // Accept new s beat and merge into buffer
                        // Shift existing buffer up by s_bytes
                        // Pack new bytes and append
                        buffer <= (buffer << (count_bytes(s_axis_tkeep) * 8)) | 
                                  pack_s_bytes(s_axis_tdata, s_axis_tkeep);
                        buf_bytes <= buf_bytes + count_bytes(s_axis_tkeep);
                        
                        // Update sideband if this is the last beat
                        if (s_axis_tlast) begin
                            s_tlast_reg <= 1'b1;
                            s_tid_reg <= s_axis_tid;
                            s_tdest_reg <= s_axis_tdest;
                            s_tuser_reg <= s_axis_tuser;
                        end
                        
                        if (buf_bytes >= M_BYTES) begin
                            state <= ST_SPLIT;
                        end else begin
                            state <= ST_MERGE;
                        end
                    end
                end
                
                default: begin
                    state <= ST_IDLE;
                end
            endcase
        end
    end

endmodule
