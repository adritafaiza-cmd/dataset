```verilog
module axis_adapter #
(
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1
)
(
    input  wire                     clk,
    input  wire                     rst,
    /*
     * AXI input
     */
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser
);

    // Internal signals
    reg [S_DATA_WIDTH-1:0] s_data_reg;
    reg [S_KEEP_WIDTH-1:0] s_keep_reg;
    reg                    s_last_reg;
    reg [ID_WIDTH-1:0]     s_id_reg;
    reg [DEST_WIDTH-1:0]   s_dest_reg;
    reg [USER_WIDTH-1:0]   s_user_reg;
    reg                    s_valid_reg;
    reg                    s_ready_reg;

    reg [M_DATA_WIDTH-1:0] m_data_reg;
    reg [M_KEEP_WIDTH-1:0] m_keep_reg;
    reg                    m_last_reg;
    reg [ID_WIDTH-1:0]     m_id_reg;
    reg [DEST_WIDTH-1:0]   m_dest_reg;
    reg [USER_WIDTH-1:0]   m_user_reg;
    reg                    m_valid_reg;
    reg                    m_ready_reg;

    // State machine
    localparam S_IDLE = 2'd0;
    localparam S_PACK = 2'd1;
    localparam S_UNPACK = 2'd2;
    localparam S_PASS = 2'd3;

    reg [1:0] state;
    reg [1:0] next_state;

    // Buffer for packing/unpacking
    reg [S_DATA_WIDTH-1:0] buffer;
    reg [S_KEEP_WIDTH-1:0] buffer_keep;
    reg                    buffer_last;
    reg [ID_WIDTH-1:0]     buffer_id;
    reg [DEST_WIDTH-1:0]   buffer_dest;
    reg [USER_WIDTH-1:0]   buffer_user;
    reg                    buffer_valid;

    // Byte counter for packing/unpacking
    reg [3:0] byte_count;
    reg [3:0] next_byte_count;

    // Determine if we need to pack, unpack, or pass through
    wire need_pack = (S_DATA_WIDTH > M_DATA_WIDTH);
    wire need_unpack = (S_DATA_WIDTH < M_DATA_WIDTH);
    wire pass_through = (S_DATA_WIDTH == M_DATA_WIDTH);

    // Calculate number of bytes per beat
    localparam S_BYTES = (S_DATA_WIDTH + 7) / 8;
    localparam M_BYTES = (M_DATA_WIDTH + 7) / 8;

    // State machine logic
    always @(*) begin
        next_state = state;
        next_byte_count = byte_count;

        case (state)
            S_IDLE: begin
                if (s_axis_tvalid && s_ready_reg) begin
                    if (need_pack) begin
                        next_state = S_PACK;
                        next_byte_count = 0;
                    end else if (need_unpack) begin
                        next_state = S_UNPACK;
                        next_byte_count = 0;
                    end else begin
                        next_state = S_PASS;
                    end
                end
            end

            S_PACK: begin
                // Accumulate bytes from S side until we have enough for M side
                if (s_axis_tvalid && s_ready_reg) begin
                    next_byte_count = byte_count + S_BYTES;
                    if (next_byte_count >= M_BYTES) begin
                        // We have enough bytes, output one M beat
                        if (m_axis_tready) begin
                            if (s_axis_tlast) begin
                                next_state = S_IDLE;
                                next_byte_count = 0;
                            end else begin
                                next_state = S_PACK;
                                next_byte_count = next_byte_count - M_BYTES;
                            end
                        end
                    end
                end
            end

            S_UNPACK: begin
                // Split one S beat into multiple M beats
                if (m_axis_tready && m_valid_reg) begin
                    next_byte_count = byte_count + M_BYTES;
                    if (next_byte_count >= S_BYTES) begin
                        // We've output all bytes from this S beat
                        if (s_axis_tlast) begin
                            next_state = S_IDLE;
                            next_byte_count = 0;
                        end else begin
                            next_state = S_UNPACK;
                            next_byte_count = 0;
                        end
                    end
                end
            end

            S_PASS: begin
                if (s_axis_tvalid && s_ready_reg && m_axis_tready) begin
                    if (s_axis_tlast) begin
                        next_state = S_IDLE;
                    end
                end
            end

            default: next_state = S_IDLE;
        endcase
    end

    // State register
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= S_IDLE;
            byte_count <= 0;
        end else begin
            state <= next_state;
            byte_count <= next_byte_count;
        end
    end

    // Input ready logic
    always @(*) begin
        s_ready_reg = 1'b0;
        case (state)
            S_IDLE: s_ready_reg = 1'b1;
            S_PACK: s_ready_reg = 1'b1;
            S_UNPACK: s_ready_reg = (byte_count == 0);
            S_PASS: s_ready_reg = m_axis_tready;
            default: s_ready_reg = 1'b0;
        endcase
    end

    assign s_axis_tready = s_ready_reg;

    // Buffer input data
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            s_data_reg <= {S_DATA_WIDTH{1'b0}};
            s_keep_reg <= {S_KEEP_WIDTH{1'b0}};
            s_last_reg <= 1'b0;
            s_id_reg <= {ID_WIDTH{1'b0}};
            s_dest_reg <= {DEST_WIDTH{1'b0}};
            s_user_reg <= {USER_WIDTH{1'b0}};
            s_valid_reg <= 1'b0;
        end else begin
            if (s_axis_tvalid && s_ready_reg) begin
                s_data_reg <= s_axis_tdata;
                s_keep_reg <= s_axis_tkeep;
                s_last_reg <= s_axis_tlast;
                s_id_reg <= s_axis_tid;
                s_dest_reg <= s_axis_tdest;
                s_user_reg <= s_axis_tuser;
                s_valid_reg <= 1'b1;
            end else begin
                s_valid_reg <= 1'b0;
            end
        end
    end

    // Output valid logic
    always @(*) begin
        m_valid_reg = 1'b0;
        case (state)
            S_PACK: begin
                if (byte_count >= M_BYTES) begin
                    m_valid_reg = 1'b1;
                end
            end
            S_UNPACK: begin
                if (s_valid_reg && byte_count < S_BYTES) begin
                    m_valid_reg = 1'b1;
                end
            end
            S_PASS: begin
                if (s_valid_reg) begin
                    m_valid_reg = 1'b1;
                end
            end
            default: m_valid_reg = 1'b0;
        endcase
    end

    assign m_axis_tvalid = m_valid_reg;

    // Output data generation
    always @(*) begin
        m_data_reg = {M_DATA_WIDTH{1'b0}};
        m_keep_reg = {M_KEEP_WIDTH{1'b0}};
        m_last_reg = 1'b0;
        m_id_reg = {ID_WIDTH{1'b0}};
        m_dest_reg = {DEST_WIDTH{1'b0}};
        m_user_reg = {USER_WIDTH{1'b0}};

        case (state)
            S_PACK: begin
                if (byte_count >= M_BYTES) begin
                    // Extract M_BYTES from buffer
                    // Simplified: just pass through for now
                    m_data_reg = s_data_reg[M_DATA_WIDTH-1:0];
                    if (M_KEEP_ENABLE) begin
                        m_keep_reg = s_keep_reg[M_KEEP_WIDTH-1:0];
                    end
                    m_last_reg = s_last_reg;
                    m_id_reg = s_id_reg;
                    m_dest_reg = s_dest_reg;
                    m_user_reg = s_user_reg;
                end
            end

            S_UNPACK: begin
                if (s_valid_reg && byte_count < S_BYTES) begin
                    // Extract M_BYTES from S data
                    m_data_reg = s_data_reg[M_DATA_WIDTH-1:0];
                    if (M_KEEP_ENABLE) begin
                        m_keep_reg = s_keep_reg[M_KEEP_WIDTH-1:0];
                    end
                    m_last_reg = s_last_reg;
                    m_id_reg = s_id_reg;
                    m_dest_reg = s_dest_reg;
                    m_user_reg = s_user_reg;
                end
            end

            S_PASS: begin
                if (s_valid_reg) begin
                    m_data_reg = s_data_reg;
                    if (M_KEEP_ENABLE) begin
                        m_keep_reg = s_keep_reg;
                    end
                    m_last_reg = s_last_reg;
                    m_id_reg = s_id_reg;
                    m_dest_reg = s_dest_reg;
                    m_user_reg = s_user_reg;
                end
            end

            default: begin
                m_data_reg = {M_DATA_WIDTH{1'b0}};
                m_keep_reg = {M_KEEP_WIDTH{1'b0}};
                m_last_reg = 1'b0;
                m_id_reg = {ID_WIDTH{1'b0}};
                m_dest_reg = {DEST_WIDTH{1'b0}};
                m_user_reg = {USER_WIDTH{1'b0}};
            end
        endcase
    end

    assign m_axis_tdata = m_data_reg;
    assign m_axis_tkeep = m_keep_reg;
    assign m_axis_tlast = m_last_reg;
    assign m_axis_tid = m_id_reg;
    assign m_axis_tdest = m_dest_reg;
    assign m_axis_tuser = m_user_reg;

endmodule
