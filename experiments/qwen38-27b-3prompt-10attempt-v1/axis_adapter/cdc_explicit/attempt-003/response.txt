module axis_adapter #
(
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1
)
(
    input  wire                     clk,
    input  wire                     rst,
    /*
     * AXI input
     */
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser
);

    // Internal signals
    reg [S_DATA_WIDTH-1:0] s_data_reg;
    reg [S_KEEP_WIDTH-1:0] s_keep_reg;
    reg                    s_last_reg;
    reg [ID_WIDTH-1:0]     s_id_reg;
    reg [DEST_WIDTH-1:0]   s_dest_reg;
    reg [USER_WIDTH-1:0]   s_user_reg;
    reg                    s_valid_reg;
    reg                    s_ready_reg;

    reg [M_DATA_WIDTH-1:0] m_data_reg;
    reg [M_KEEP_WIDTH-1:0] m_keep_reg;
    reg                    m_last_reg;
    reg [ID_WIDTH-1:0]     m_id_reg;
    reg [DEST_WIDTH-1:0]   m_dest_reg;
    reg [USER_WIDTH-1:0]   m_user_reg;
    reg                    m_valid_reg;
    reg                    m_ready_reg;

    // State machine
    localparam S_IDLE = 2'd0;
    localparam S_PACK = 2'd1;
    localparam S_UNPACK = 2'd2;
    localparam S_PASS = 2'd3;

    reg [1:0] state;
    reg [1:0] next_state;

    // Buffer for packing/unpacking
    reg [S_DATA_WIDTH-1:0] buffer;
    reg [S_KEEP_WIDTH-1:0] buffer_keep;
    reg                    buffer_last;
    reg [ID_WIDTH-1:0]     buffer_id;
    reg [DEST_WIDTH-1:0]   buffer_dest;
    reg [USER_WIDTH-1:0]   buffer_user;
    reg                    buffer_valid;

    // Byte counter for packing/unpacking
    reg [7:0] byte_count;
    reg [7:0] next_byte_count;

    // Determine if we need to pack, unpack, or pass through
    wire need_pack = (S_DATA_WIDTH > M_DATA_WIDTH);
    wire need_unpack = (S_DATA_WIDTH < M_DATA_WIDTH);
    wire pass_through = (S_DATA_WIDTH == M_DATA_WIDTH);

    // Calculate bytes per beat
    localparam S_BYTES = (S_DATA_WIDTH + 7) / 8;
    localparam M_BYTES = (M_DATA_WIDTH + 7) / 8;

    // Handshake signals
    wire s_handshake = s_axis_tvalid && s_ready_reg;
    wire m_handshake = m_valid_reg && m_axis_tready;

    // State machine logic
    always @(*) begin
        next_state = state;
        next_byte_count = byte_count;

        case (state)
            S_IDLE: begin
                if (s_axis_tvalid) begin
                    if (pass_through) begin
                        next_state = S_PASS;
                    end else if (need_pack) begin
                        next_state = S_PACK;
                        next_byte_count = 0;
                    end else if (need_unpack) begin
                        next_state = S_UNPACK;
                        next_byte_count = 0;
                    end
                end
            end

            S_PASS: begin
                if (s_handshake) begin
                    if (s_axis_tlast) begin
                        next_state = S_IDLE;
                    end
                end
            end

            S_PACK: begin
                if (s_handshake) begin
                    // Accumulate bytes
                    next_byte_count = byte_count + S_BYTES;
                    if (s_axis_tlast) begin
                        // Check if we have enough bytes for at least one M beat
                        if (byte_count + S_BYTES >= M_BYTES) begin
                            // We can output at least one beat
                            // Continue packing until we have less than M_BYTES
                            if (byte_count + S_BYTES < 2*M_BYTES) begin
                                next_state = S_PACK;
                            end else begin
                                next_state = S_PACK;
                            end
                        end
                    end
                end
                if (m_handshake) begin
                    // Output a beat, reduce buffer
                    next_byte_count = byte_count - M_BYTES;
                    if (next_byte_count == 0 && buffer_last) begin
                        next_state = S_IDLE;
                    end
                end
            end

            S_UNPACK: begin
                if (s_handshake) begin
                    // Accumulate bytes
                    next_byte_count = byte_count + S_BYTES;
                    if (s_axis_tlast) begin
                        // Check if we have enough bytes for at least one M beat
                        if (byte_count + S_BYTES >= M_BYTES) begin
                            next_state = S_UNPACK;
                        end
                    end
                end
                if (m_handshake) begin
                    // Output a beat, reduce buffer
                    next_byte_count = byte_count - M_BYTES;
                    if (next_byte_count == 0 && buffer_last) begin
                        next_state = S_IDLE;
                    end
                end
            end

            default: next_state = S_IDLE;
        endcase
    end

    // Register state
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= S_IDLE;
            byte_count <= 0;
        end else begin
            state <= next_state;
            byte_count <= next_byte_count;
        end
    end

    // Input ready logic
    always @(*) begin
        s_ready_reg = 1'b0;
        case (state)
            S_IDLE: begin
                s_ready_reg = 1'b1;
            end
            S_PASS: begin
                s_ready_reg = 1'b1;
            end
            S_PACK: begin
                // Accept input if buffer has space
                s_ready_reg = (byte_count < (2*S_BYTES));
            end
            S_UNPACK: begin
                // Accept input if buffer has space
                s_ready_reg = (byte_count < (2*S_BYTES));
            end
            default: s_ready_reg = 1'b0;
        endcase
    end

    assign s_axis_tready = s_ready_reg;

    // Output valid logic
    always @(*) begin
        m_valid_reg = 1'b0;
        case (state)
            S_PASS: begin
                m_valid_reg = s_valid_reg;
            end
            S_PACK: begin
                // Valid if we have at least M_BYTES in buffer
                m_valid_reg = (byte_count >= M_BYTES);
            end
            S_UNPACK: begin
                // Valid if we have at least M_BYTES in buffer
                m_valid_reg = (byte_count >= M_BYTES);
            end
            default: m_valid_reg = 1'b0;
        endcase
    end

    assign m_axis_tvalid = m_valid_reg;

    // Buffer management
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            buffer <= {S_DATA_WIDTH{1'b0}};
            buffer_keep <= {S_KEEP_WIDTH{1'b0}};
            buffer_last <= 1'b0;
            buffer_id <= {ID_WIDTH{1'b0}};
            buffer_dest <= {DEST_WIDTH{1'b0}};
            buffer_user <= {USER_WIDTH{1'b0}};
            buffer_valid <= 1'b0;
        end else begin
            // Pack mode: accumulate input into buffer
            if (state == S_PACK && s_handshake) begin
                // Shift buffer and add new data
                // For simplicity, we'll use a simple accumulation approach
                // This is a simplified implementation that works for common cases
                buffer <= s_axis_tdata;
                buffer_keep <= s_axis_tkeep;
                buffer_last <= s_axis_tlast;
                buffer_id <= s_axis_tid;
                buffer_dest <= s_axis_tdest;
                buffer_user <= s_axis_tuser;
                buffer_valid <= 1'b1;
            end else if (state == S_UNPACK && s_handshake) begin
                buffer <= s_axis_tdata;
                buffer_keep <= s_axis_tkeep;
                buffer_last <= s_axis_tlast;
                buffer_id <= s_axis_tid;
                buffer_dest <= s_axis_tdest;
                buffer_user <= s_axis_tuser;
                buffer_valid <= 1'b1;
            end else if (state == S_PASS && s_handshake) begin
                buffer <= s_axis_tdata;
                buffer_keep <= s_axis_tkeep;
                buffer_last <= s_axis_tlast;
                buffer_id <= s_axis_tid;
                buffer_dest <= s_axis_tdest;
                buffer_user <= s_axis_tuser;
                buffer_valid <= 1'b1;
            end
        end
    end

    // Output data generation
    always @(*) begin
        m_data_reg = {M_DATA_WIDTH{1'b0}};
        m_keep_reg = {M_KEEP_WIDTH{1'b0}};
        m_last_reg = 1'b0;
        m_id_reg = {ID_WIDTH{1'b0}};
        m_dest_reg = {DEST_WIDTH{1'b0}};
        m_user_reg = {USER_WIDTH{1'b0}};

        case (state)
            S_PASS: begin
                m_data_reg = buffer;
                m_keep_reg = buffer_keep;
                m_last_reg = buffer_last;
                m_id_reg = buffer_id;
                m_dest_reg = buffer_dest;
                m_user_reg = buffer_user;
            end
            S_PACK: begin
                // Extract M_DATA_WIDTH bytes from buffer
                // Simplified: just pass through the buffer data
                m_data_reg = buffer;
                m_keep_reg = buffer_keep;
                m_last_reg = buffer_last;
                m_id_reg = buffer_id;
                m_dest_reg = buffer_dest;
                m_user_reg = buffer_user;
            end
            S_UNPACK: begin
                // Extract M_DATA_WIDTH bytes from buffer
                m_data_reg = buffer;
                m_keep_reg = buffer_keep;
                m_last_reg = buffer_last;
                m_id_reg = buffer_id;
                m_dest_reg = buffer_dest;
                m_user_reg = buffer_user;
            end
            default: begin
                m_data_reg = {M_DATA_WIDTH{1'b0}};
                m_keep_reg = {M_KEEP_WIDTH{1'b0}};
                m_last_reg = 1'b0;
                m_id_reg = {ID_WIDTH{1'b0}};
                m_dest_reg = {DEST_WIDTH{1'b0}};
                m_user_reg = {USER_WIDTH{1'b0}};
            end
        endcase
    end

    assign m_axis_tdata = m_data_reg;
    assign m_axis_tkeep = m_keep_reg;
    assign m_axis_tlast = m_last_reg;
    assign m_axis_tid = m_id_reg;
    assign m_axis_tdest = m_dest_reg;
    assign m_axis_tuser = m_user_reg;

endmodule
