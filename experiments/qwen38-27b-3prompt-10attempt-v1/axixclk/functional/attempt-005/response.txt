			ar_fifo_full = ar_fifo_wr_ptr[FIFO_ADDR_WIDTH-1] < ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1];
			ar_fifo_empty = 1'b0;
		end
	end

	// AR channel outputs
	assign s_arready = !ar_fifo_full;
	assign s_arvalid = S_AXI_ARVALID;
	assign s_arid = S_AXI_ARID;
	assign s_araddr = S_AXI_ARADDR;
	assign s_arlen = S_AXI_ARLEN;

	assign s_arcache = S_AXI_ARCACHE;
	assign s_arprot = S_AXI_ARPROT;
	assign s_arqos = S_AXI_ARQOS;

	assign m_arid = ar_fifo_id[ar_fifo_rd_ptr];
	assign m_araddr = ar_fifo_addr[ar_fifo_rd_ptr];
	assign m_arlen = ar_fifo_len[ar_fifo_rd_ptr];
	assign m_arsize = ar_fifo_size[ar_fifo_rd_ptr];
	assign m_arburst = ar_fifo_burst[ar_fifo_rd_ptr];

	assign m_arqos = ar_fifo_qos[ar_fifo_rd_ptr];
	assign m_arvalid = !ar_fifo_empty;
	assign m_arready = M_AXI_ARREADY;

	// R Channel: M to S
	// FIFO for R channel
	reg [C_S_AXI_ID_WIDTH-1 : 0] r_fifo_id [0 : FIFO_DEPTH-1];
	reg [C_S_AXI_DATA_WIDTH-1 : 0] r_fifo_data [0 : FIFO_DEPTH-1];
	reg [1 : 0] r_fifo_resp [0 : FIFO_DEPTH-1];
	reg r_fifo_last [0 : FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH-1 : 0] r_fifo_wr_ptr, r_fifo_rd_ptr;
	reg r_fifo_full, r_fifo_empty;

	// R FIFO write logic (M clock)
	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			r_fifo_wr_ptr <= {FIFO_ADDR_WIDTH{1'b0}};
			r_fifo_full <= 1'b0;
		end else begin
			if (m_rvalid && m_rready) begin
				r_fifo_id[r_fifo_wr_ptr] <= M_AXI_RID;
				r_fifo_data[r_fifo_wr_ptr] <= M_AXI_RDATA;
				r_fifo_resp[r_fifo_wr_ptr] <= M_AXI_RRESP;
				r_fifo_last[r_fifo_wr_ptr] <= M_AXI_RLAST;
				r_fifo_wr_ptr <= r_fifo_wr_ptr + 1'b1;
			end
		end
	end

	// R FIFO read logic (S clock)
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			r_fifo_rd_ptr <= {FIFO_ADDR_WIDTH{1'b0}};
			r_fifo_empty <= 1'b1;
		end else begin
			if (s_rvalid && s_rready) begin
				r_fifo_rd_ptr <= r_fifo_rd_ptr + 1'b1;
			end
		end
	end

	// R FIFO status
	always @(*) begin
		if (r_fifo_wr_ptr == r_fifo_rd_ptr) begin
			r_fifo_full = 1'b0;
			r_fifo_empty = 1'b1;
		end else if (r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1] != r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1]) begin
			r_fifo_full = r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1] > r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1];
			r_fifo_empty = 1'b0;
		end else begin
			r_fifo_full = r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1] < r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1];
			r_fifo_empty = 1'b0;
		end
	end

	// R channel outputs
	assign m_rready = !r_fifo_full;
	assign m_rvalid = M_AXI_RVALID;
	assign m_rid = M_AXI_RID;
	assign m_rdata = M_AXI_RDATA;
	assign m_rresp = M_AXI_RRESP;
	assign m_rlast = M_AXI_RLAST;

	assign s_rid = r_fifo_id[r_fifo_rd_ptr];
	assign s_rdata = r_fifo_data[r_fifo_rd_ptr];
	assign s_rresp = r_fifo_resp[r_fifo_rd_ptr];
	assign s_rlast = r_fifo_last[r_fifo_rd_ptr];
	assign s_rvalid = !r_fifo_empty;
	assign s_rready = S_AXI_RREADY;

	// Final port assignments
	assign S_AXI_AWREADY = s_awready;
	assign S_AXI_WREADY = s_wready;
	assign S_AXI_BID = s_bid;

	assign M_AXI_AWADDR = m_awaddr;
