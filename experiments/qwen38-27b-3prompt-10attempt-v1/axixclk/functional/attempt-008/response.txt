	localparam integer FIFO_DEPTH = 1 << LGFIFO;
	localparam integer FIFO_ADDR_WIDTH = LGFIFO;

	localparam integer AR_FIFO_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4;
	localparam integer W_FIFO_WIDTH = C_S_AXI_DATA_WIDTH + (C_S_AXI_DATA_WIDTH/8) + 1;
	localparam integer B_FIFO_WIDTH = C_S_AXI_ID_WIDTH + 2;
	localparam integer R_FIFO_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_DATA_WIDTH + 2 + 1;

	// Internal signals
	wire [AW_FIFO_WIDTH-1:0] aw_fifo_din;
	wire aw_fifo_wr_en;
	wire aw_fifo_rd_en;
	wire aw_fifo_full;
	wire aw_fifo_empty;
	wire [AW_FIFO_WIDTH-1:0] aw_fifo_dout;

	wire [AR_FIFO_WIDTH-1:0] ar_fifo_din;

	wire b_fifo_rd_en;

	wire b_fifo_empty;
	wire [B_FIFO_WIDTH-1:0] b_fifo_dout;

	wire [R_FIFO_WIDTH-1:0] r_fifo_din;
	wire r_fifo_wr_en;
	wire r_fifo_rd_en;
	wire r_fifo_full;
	wire r_fifo_empty;
	wire [R_FIFO_WIDTH-1:0] r_fifo_dout;

	// AW channel packing
	assign aw_fifo_din = {S_AXI_AWID, S_AXI_AWADDR, S_AXI_AWLEN, S_AXI_AWSIZE, S_AXI_AWBURST, S_AXI_AWLOCK, S_AXI_AWCACHE, S_AXI_AWPROT, S_AXI_AWQOS};
	assign aw_fifo_wr_en = S_AXI_AWVALID && S_AXI_AWREADY;
	assign S_AXI_AWREADY = !aw_fifo_full;

	// AR channel packing
	assign ar_fifo_din = {S_AXI_ARID, S_AXI_ARADDR, S_AXI_ARLEN, S_AXI_ARSIZE, S_AXI_ARBURST, S_AXI_ARLOCK, S_AXI_ARCACHE, S_AXI_ARPROT, S_AXI_ARQOS};
	assign ar_fifo_wr_en = S_AXI_ARVALID && S_AXI_ARREADY;
	assign S_AXI_ARREADY = !ar_fifo_full;

	// W channel packing
	assign w_fifo_din = {S_AXI_WDATA, S_AXI_WSTRB, S_AXI_WLAST};
	assign w_fifo_wr_en = S_AXI_WVALID && S_AXI_WREADY;
	assign S_AXI_WREADY = !w_fifo_full;

	// B channel unpacking
	assign S_AXI_BID = b_fifo_dout[C_S_AXI_ID_WIDTH-1:0];
	assign S_AXI_BRESP = b_fifo_dout[C_S_AXI_ID_WIDTH+1:C_S_AXI_ID_WIDTH];
	assign S_AXI_BVALID = !b_fifo_empty;
	assign b_fifo_rd_en = S_AXI_BVALID && S_AXI_BREADY;

	// R channel unpacking
	assign S_AXI_RID = r_fifo_dout[C_S_AXI_ID_WIDTH-1:0];
	assign S_AXI_RDATA = r_fifo_dout[C_S_AXI_ID_WIDTH+C_S_AXI_DATA_WIDTH-1:C_S_AXI_ID_WIDTH];
	assign S_AXI_RRESP = r_fifo_dout[C_S_AXI_ID_WIDTH+C_S_AXI_DATA_WIDTH+1:C_S_AXI_ID_WIDTH+C_S_AXI_DATA_WIDTH];
	assign S_AXI_RLAST = r_fifo_dout[C_S_AXI_ID_WIDTH+C_S_AXI_DATA_WIDTH+2];
	assign S_AXI_RVALID = !r_fifo_empty;
	assign r_fifo_rd_en = S_AXI_RVALID && S_AXI_RREADY;

	// AW FIFO instantiation
	axixclk_fifo #(
		.DATA_WIDTH(AW_FIFO_WIDTH),
		.ADDR_WIDTH(FIFO_ADDR_WIDTH),
		.DEPTH(FIFO_DEPTH)
	) u_aw_fifo (
		.clk(S_AXI_ACLK),
		.rst_n(S_AXI_ARESETN),
		.din(aw_fifo_din),
		.wr_en(aw_fifo_wr_en),
		.rd_en(aw_fifo_rd_en),
		.full(aw_fifo_full),
		.empty(aw_fifo_empty),
		.dout(aw_fifo_dout)
	);

	// AR FIFO instantiation
	axixclk_fifo #(
		.DATA_WIDTH(AR_FIFO_WIDTH),
		.ADDR_WIDTH(FIFO_ADDR_WIDTH),
		.DEPTH(FIFO_DEPTH)
	) u_ar_fifo (
		.clk(S_AXI_ACLK),
		.rst_n(S_AXI_ARESETN),
		.din(ar_fifo_din),
		.wr_en(ar_fifo_wr_en),
		.rd_en(ar_fifo_rd_en),
		.full(ar_fifo_full),
		.empty(ar_fifo_empty),
		.dout(ar_fifo_dout)
	);

	// W FIFO instantiation
	axixclk_fifo #(
		.DATA_WIDTH(W_FIFO_WIDTH),
		.ADDR_WIDTH(FIFO_ADDR_WIDTH),
		.DEPTH(FIFO_DEPTH)
	) u_w_fifo (
		.clk(S_AXI_ACLK),
		.rst_n(S_AXI_ARESETN),
		.din(w_fifo_din),
		.wr_en(w_fifo_wr_en),
		.rd_en(w_fifo_rd_en),
		.full(w_fifo_full),
		.empty(w_fifo_empty),
		.dout(w_fifo_dout)
	);

	// B FIFO instantiation
	axixclk_fifo #(
		.DATA_WIDTH(B_FIFO_WIDTH),
		.ADDR_WIDTH(FIFO_ADDR_WIDTH),
		.DEPTH(FIFO_DEPTH)
	) u_b_fifo (
		.clk(M_AXI_ACLK),
		.rst_n(M_AXI_ARESETN),
		.din(b_fifo_din),
		.wr_en(b_fifo_wr_en),
		.rd_en(b_fifo_rd_en),
		.full(b_fifo_full),
		.empty(b_fifo_empty),
		.dout(b_fifo_dout)
	);

		.DATA_WIDTH(R_FIFO_WIDTH),
		.ADDR_WIDTH(FIFO_ADDR_WIDTH),
		.DEPTH(FIFO_DEPTH)
	) u_r_fifo (
		.clk(M_AXI_ACLK),
		.rst_n(M_AXI_ARESETN),
		.din(r_fifo_din),
		.wr_en(r_fifo_wr_en),
		.rd_en(r_fifo_rd_en),
		.full(r_fifo_full),
		.empty(r_fifo_empty),
		.dout(r_fifo_dout)
	);

	// AW channel crossing to M clock domain
	wire [AW_FIFO_WIDTH-1:0] aw_m_data;
	wire aw_m_valid;
	wire aw_m_ready;

	assign aw_fifo_rd_en = aw_m_ready;

	axixclk_async_fifo #(
		.DATA_WIDTH(AW_FIFO_WIDTH),
		.ADDR_WIDTH(FIFO_ADDR_WIDTH),
		.DEPTH(FIFO_DEPTH),
		.FFS(XCLOCK_FFS)
	) u_aw_async_fifo (
		.wr_clk(S_AXI_ACLK),
		.wr_rst_n(S_AXI_ARESETN),
		.wr_din(aw_fifo_dout),
		.wr_en(aw_fifo_rd_en),
		.wr_full(aw_fifo_full),
		.wr_empty(aw_fifo_empty),
		.rd_clk(M_AXI_ACLK),
		.rd_rst_n(M_AXI_ARESETN),
		.rd_dout(aw_m_data),
		.rd_en(aw_m_ready),
		.rd_full(aw_m_full),
		.rd_empty(aw_m_empty)
	);

	wire aw_m_full;
	wire aw_m_empty;

	// AW master side
	reg [AW_FIFO_WIDTH-1:0] aw_m_reg;
	reg aw_m_valid_reg;

	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			aw_m_valid_reg <= 1'b0;
			aw_m_reg <= {AW_FIFO_WIDTH{1'b0}};
		end else begin
			if (!aw_m_empty) begin
				aw_m_reg <= aw_m_data;
				aw_m_valid_reg <= 1'b1;
			end else if (aw_m_valid_reg && M_AXI_AWREADY) begin
				aw_m_valid_reg <= 1'b0;
			end
		end
	end

	assign aw_m_valid = aw_m_valid_reg;
	assign aw_m_ready = !aw_m_full && (!aw_m_valid_reg || M_AXI_AWREADY);

	assign M_AXI_AWID = aw_m_reg[C_S_AXI_ID_WIDTH-1:0];
	assign M_AXI_AWADDR = aw_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH-1:C_S_AXI_ID_WIDTH];
	assign M_AXI_AWLEN = aw_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+7:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH];
	assign M_AXI_AWSIZE = aw_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+10:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+8];
	assign M_AXI_AWBURST = aw_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+12:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+11];
	assign M_AXI_AWLOCK = aw_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+13];
	assign M_AXI_AWCACHE = aw_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+17:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+14];
	assign M_AXI_AWPROT = aw_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+20:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+18];
	assign M_AXI_AWQOS = aw_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+24:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+21];
	assign M_AXI_AWVALID = aw_m_valid;

	// AR channel crossing to M clock domain
	wire [AR_FIFO_WIDTH-1:0] ar_m_data;
	wire ar_m_valid;
	wire ar_m_ready;

	assign ar_fifo_rd_en = ar_m_ready;

		.DATA_WIDTH(AR_FIFO_WIDTH),
		.ADDR_WIDTH(FIFO_ADDR_WIDTH),
		.DEPTH(FIFO_DEPTH),
		.FFS(XCLOCK_FFS)
	) u_ar_async_fifo (

		.wr_rst_n(S_AXI_ARESETN),
		.wr_din(ar_fifo_dout),
		.wr_en(ar_fifo_rd_en),
		.wr_full(ar_fifo_full),
		.wr_empty(ar_fifo_empty),
		.rd_clk(M_AXI_ACLK),
		.rd_rst_n(M_AXI_ARESETN),
		.rd_dout(ar_m_data),
		.rd_en(ar_m_ready),
		.rd_full(ar_m_full),
		.rd_empty(ar_m_empty)
	);

	wire ar_m_full;
	wire ar_m_empty;

	// AR master side
	reg [AR_FIFO_WIDTH-1:0] ar_m_reg;
	reg ar_m_valid_reg;

	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			ar_m_valid_reg <= 1'b0;
			ar_m_reg <= {AR_FIFO_WIDTH{1'b0}};
		end else begin
			if (!ar_m_empty) begin
				ar_m_reg <= ar_m_data;
				ar_m_valid_reg <= 1'b1;
			end else if (ar_m_valid_reg && M_AXI_ARREADY) begin
				ar_m_valid_reg <= 1'b0;
			end
		end
	end

	assign ar_m_valid = ar_m_valid_reg;
	assign ar_m_ready = !ar_m_full && (!ar_m_valid_reg || M_AXI_ARREADY);

	assign M_AXI_ARID = ar_m_reg[C_S_AXI_ID_WIDTH-1:0];
	assign M_AXI_ARADDR = ar_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH-1:C_S_AXI_ID_WIDTH];
	assign M_AXI_ARLEN = ar_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+7:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH];
	assign M_AXI_ARSIZE = ar_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+10:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+8];
	assign M_AXI_ARBURST = ar_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+12:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+11];
	assign M_AXI_ARLOCK = ar_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+13];
	assign M_AXI_ARCACHE = ar_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+17:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+14];
	assign M_AXI_ARPROT = ar_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+20:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+18];
	assign M_AXI_ARQOS = ar_m_reg[C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+24:C_S_AXI_ID_WIDTH+C_S_AXI_ADDR_WIDTH+21];
	assign M_AXI_ARVALID = ar_m_valid;

	// W channel crossing to M clock domain
	wire [W_FIFO_WIDTH-1:0] w_m_data;
	wire w_m_valid;
	wire w_m_ready;

	assign w_fifo_rd_en = w_m_ready;

	axixclk_async_fifo #(
		.DATA_WIDTH(W_FIFO_WIDTH),
		.ADDR_WIDTH(FIFO_ADDR_WIDTH),
		.DEPTH(FIFO_DEPTH),
		.FFS(XCLOCK_FFS)
	) u_w_async_fifo (
		.wr_clk(S_AXI_ACLK),
		.wr_rst_n(S_AXI_ARESETN),
		.wr_din(w_fifo_dout),
		.wr_en(w_fifo_rd_en),
		.wr_full(w_fifo_full),
		.wr_empty(w_fifo_empty),
		.rd_clk(M_AXI_ACLK),
		.rd_rst_n(M_AXI_ARESETN),
		.rd_dout(w_m_data),
		.rd_en(w_m_ready),
		.rd_full(w_m_full),
		.rd_empty(w_m_empty)
	);

	wire w_m_full;
	wire w_m_empty;

	// W master side with extra register
	reg [W_FIFO_WIDTH-1:0] w_m_reg1;
	reg [W_FIFO_WIDTH-1:0] w_m_reg2;
	reg w_m_valid_reg1;
	reg w_m_valid_reg2;

	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			w_m_valid_reg1 <= 1'b0;
			w_m_valid_reg2 <= 1'b0;
			w_m_reg1 <= {W_FIFO_WIDTH{1'b0}};
			w_m_reg2 <= {W_FIFO_WIDTH{1'b0}};
		end else begin
			// First stage
			if (!w_m_empty) begin
				w_m_reg1 <= w_m_data;
				w_m_valid_reg1 <= 1'b1;
			end else if (w_m_valid_reg1 && w_m_valid_reg2) begin
				w_m_valid_reg1 <= 1'b0;
			end

			// Second stage
			if (w_m_valid_reg1) begin
				w_m_reg2 <= w_m_reg1;
				w_m_valid_reg2 <= 1'b1;
			end else if (w_m_valid_reg2 && M_AXI_WREADY) begin
				w_m_valid_reg2 <= 1'b0;
			end
		end
	end

	assign w_m_ready = !w_m_full && (!w_m_valid_reg1 || w_m_valid_reg2);

	assign M_AXI_WSTRB = w_m_reg2[C_S_AXI_DATA_WIDTH+(C_S_AXI_DATA_WIDTH/8)-1:C_S_AXI_DATA_WIDTH];
	assign M_AXI_WLAST = w_m_reg2[C_S_AXI_DATA_WIDTH+(C_S_AXI_DATA_WIDTH/8)];
	assign M_AXI_WVALID = w_m_valid;

	// B channel crossing to S clock domain
	wire [B_FIFO_WIDTH-1:0] b_s_data;
	wire b_s_valid;
	wire b_s_ready;

	assign b_fifo_wr_en = M_AXI_BVALID && M_AXI_BREADY;
	assign M_AXI_BREADY = !b_fifo_full;

	assign b_fifo_din = {M_AXI_BID, M_AXI_BRESP};

	axixclk_async_fifo #(
		.DATA_WIDTH(B_FIFO_WIDTH),
		.ADDR_WIDTH(FIFO_ADDR_WIDTH),
		.DEPTH(FIFO_DEPTH),
		.FFS(XCLOCK_FFS)
	) u_b_async_fifo (
		.wr_clk(M_AXI_ACLK),
		.wr_rst_n(M_AXI_ARESETN),
		.wr_din(b_fifo_din),
		.wr_en(b_fifo_wr_en),
		.wr_full(b_fifo_full),
		.wr_empty(b_fifo_empty),
		.rd_clk(S_AXI_ACLK),
		.rd_rst_n(S_AXI_ARESETN),
		.rd_dout(b_s_data),
		.rd_en(b_s_ready),
		.rd_full(b_s_full),
		.rd_empty(b_s_empty)
	);

	wire b_s_full;
	wire b_s_empty;

	// B slave side
	reg [B_FIFO_WIDTH-1:0] b_s_reg;
	reg b_s_valid_reg;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			b_s_valid_reg <= 1'b0;
			b_s_reg <= {B_FIFO_WIDTH{1'b0}};
		end else begin
			if (!b_s_empty) begin
				b_s_reg <= b_s_data;
				b_s_valid_reg <= 1'b1;
			end else if (b_s_valid_reg && S_AXI_BREADY) begin
				b_s_valid_reg <= 1'b0;
			end
		end
	end

	assign b_s_valid = b_s_valid_reg;
	assign b_s_ready = !b_s_full && (!b_s_valid_reg || S_AXI_BREADY);

	assign S_AXI_BID = b_s_reg[C_S_AXI_ID_WIDTH-1:0];
	assign S_AXI_BRESP = b_s_reg[C_S_AXI_ID_WIDTH+1:C_S_AXI_ID_WIDTH];
	assign S_AXI_BVALID = b_s_valid;

	// R channel crossing to S clock domain
	wire [R_FIFO_WIDTH-1:0] r_s_data;
	wire r_s_valid;
	wire r_s_ready;

	assign r_fifo_wr_en = M_AXI_RVALID && M_AXI_RREADY;
	assign M_AXI_RREADY = !r_fifo_full;

	assign r_fifo_din = {M_AXI_RID, M_AXI_RDATA, M_AXI_RRESP, M_AXI_RLAST};

	axixclk_async_fifo #(
		.DATA_WIDTH(R_FIFO_WIDTH),
		.ADDR_WIDTH(FIFO_ADDR_WIDTH),
		.DEPTH(FIFO_DEPTH),
		.FFS(XCLOCK_FFS)
	) u_r_async_fifo (
		.wr_clk(M_AXI_ACLK),
		.wr_rst_n(M_AXI_ARESETN),
		.wr_din(r_fifo_din),
		.wr_en(r_fifo_wr_en),
		.wr_full(r_fifo_full),
		.wr_empty(r_fifo_empty),
		.rd_clk(S_AXI_ACLK),
		.rd_rst_n(S_AXI_ARESETN),
		.rd_dout(r_s_data),
		.rd_en(r_s_ready),
		.rd_full(r_s_full),
		.rd_empty(r_s_empty)
	);

	wire r_s_full;
	wire r_s_empty;

	// R slave side
	reg [R_FIFO_WIDTH-1:0] r_s_reg;
	reg r_s_valid_reg;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			r_s_valid_reg <= 1'b0;
			r_s_reg <= {R_FIFO_WIDTH{1'b0}};
		end else begin
			if (!r_s_empty) begin
				r_s_reg <= r_s_data;
				r_s_valid_reg <= 1'b1;
			end else if (r_s_valid_reg && S_AXI_RREADY) begin
				r_s_valid_reg <= 1'b0;
			end
		end
	end

	assign r_s_valid = r_s_valid_reg;
	assign r_s_ready = !r_s_full && (!r_s_valid_reg || S_AXI_RREADY);

	assign S_AXI_RID = r_s_reg[C_S_AXI_ID_WIDTH-1:0];
	assign S_AXI_RDATA = r_s_reg[C_S_AXI_ID_WIDTH+C_S_AXI_DATA_WIDTH-1:C_S_AXI_ID_WIDTH];
	assign S_AXI_RRESP = r_s_reg[C_S_AXI_ID_WIDTH+C_S_AXI_DATA_WIDTH+1:C_S_AXI_ID_WIDTH+C_S_AXI_DATA_WIDTH];
	assign S_AXI_RLAST = r_s_reg[C_S_AXI_ID_WIDTH+C_S_AXI_DATA_WIDTH+2];
	assign S_AXI_RVALID = r_s_valid;

	// Reset synchronization
	reg [1:0] m_reset_sync;

	always @(posedge M_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			m_reset_sync <= 2'b00;
		end else begin
			m_reset_sync <= {m_reset_sync[0], 1'b1};
		end
	end

	assign M_AXI_ARESETN = m_reset_sync[1];

endmodule

// Synchronous FIFO module
module axixclk_fifo #(
	parameter DATA_WIDTH = 32,
	parameter ADDR_WIDTH = 4,
	parameter DEPTH = 16
) (
	input wire clk,
	input wire rst_n,
	input wire [DATA_WIDTH-1:0] din,
	input wire wr_en,
	input wire rd_en,
	output wire full,
	output wire empty,
	output wire [DATA_WIDTH-1:0] dout
);

	reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];
	reg [ADDR_WIDTH:0] wr_ptr;
	reg [ADDR_WIDTH:0] rd_ptr;

	wire [ADDR_WIDTH-1:0] wr_addr = wr_ptr[ADDR_WIDTH-1:0];
	wire [ADDR_WIDTH-1:0] rd_addr = rd_ptr[ADDR_WIDTH-1:0];

	assign full = (wr_ptr[ADDR_WIDTH] != rd_ptr[ADDR_WIDTH]) && (wr_ptr[ADDR_WIDTH-1:0] == rd_ptr[ADDR_WIDTH-1:0]);
	assign empty = (wr_ptr == rd_ptr);

	always @(posedge clk) begin
		if (wr_en && !full) begin
			mem[wr_addr] <= din;
			wr_ptr <= wr_ptr + 1'b1;
		end
	end

	always @(posedge clk) begin
		if (rd_en && !empty) begin
			rd_ptr <= rd_ptr + 1'b1;
		end
	end

	assign dout = mem[rd_addr];

endmodule

// Asynchronous FIFO module
module axixclk_async_fifo #(
	parameter DATA_WIDTH = 32,
	parameter ADDR_WIDTH = 4,
	parameter DEPTH = 16,
	parameter FFS = 2
) (
	input wire wr_clk,
	input wire wr_rst_n,
	input wire [DATA_WIDTH-1:0] wr_din,
	input wire wr_en,
	output wire wr_full,
	output wire wr_empty,
	input wire rd_clk,
	input wire rd_rst_n,
	output wire [DATA_WIDTH-1:0] rd_dout,
	input wire rd_en,
	output wire rd_full,
	output wire rd_empty
);

	// Write pointer
	reg [ADDR_WIDTH:0] wr_ptr;
	reg [ADDR_WIDTH:0] wr_ptr_rd_sync [0:FFS-1];

	// Read pointer
	reg [ADDR_WIDTH:0] rd_ptr;
	reg [ADDR_WIDTH:0] rd_ptr_wr_sync [0:FFS-1];

	// Memory
	reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];

	// Write address
	wire [ADDR_WIDTH-1:0] wr_addr = wr_ptr[ADDR_WIDTH-1:0];

	// Read address
	wire [ADDR_WIDTH-1:0] rd_addr = rd_ptr[ADDR_WIDTH-1:0];

	// Synchronize write pointer to read clock domain
	always @(posedge rd_clk or negedge rd_rst_n) begin
		if (!rd_rst_n) begin
			wr_ptr_rd_sync[0] <= {(ADDR_WIDTH+1){1'b0}};
		end else begin
			wr_ptr_rd_sync[0] <= wr_ptr;
		end
	end

	generate
		genvar i;
		for (i = 1; i < FFS; i = i + 1) begin : gen_wr_sync
			always @(posedge rd_clk or negedge rd_rst_n) begin
				if (!rd_rst_n) begin
					wr_ptr_rd_sync[i] <= {(ADDR_WIDTH+1){1'b0}};
				end else begin
					wr_ptr_rd_sync[i] <= wr_ptr_rd_sync[i-1];
				end
			end
		end
	endgenerate

	// Synchronize read pointer to write clock domain
	always @(posedge wr_clk or negedge wr_rst_n) begin
		if (!wr_rst_n) begin
			rd_ptr_wr_sync[0] <= {(ADDR_WIDTH+1){1'b0}};
		end else begin
			rd_ptr_wr_sync[0] <= rd_ptr;
		end
	end

	generate
		genvar j;
		for (j = 1; j < FFS; j = j + 1) begin : gen_rd_sync
			always @(posedge wr_clk or negedge wr_rst_n) begin
				if (!wr_rst_n) begin
					rd_ptr_wr_sync[j] <= {(ADDR_WIDTH+1){1'b0}};
				end else begin
					rd_ptr_wr_sync[j] <= rd_ptr_wr_sync[j-1];
				end
			end
		end
	endgenerate

	// Full condition
	assign wr_full = (wr_ptr[ADDR_WIDTH] != rd_ptr_wr_sync[FFS-1][ADDR_WIDTH]) && 
	                  (wr_ptr[ADDR_WIDTH-1:0] == rd_ptr_wr_sync[FFS-1][ADDR_WIDTH-1:0]);

	// Empty condition
	assign rd_empty = (rd_ptr == wr_ptr_rd_sync[FFS-1]);

	// Write to memory
	always @(posedge wr_clk) begin
		if (wr_en && !wr_full) begin
			mem[wr_addr] <= wr_din;
			wr_ptr <= wr_ptr + 1'b1;
		end
	end

	// Read from memory
	always @(posedge rd_clk) begin
		if (rd_en && !rd_empty) begin
			rd_ptr <= rd_ptr + 1'b1;
		end
	end

	assign rd_dout = mem[rd_addr];
	assign wr_empty = rd_empty;
	assign rd_full = wr_full;

endmodule
