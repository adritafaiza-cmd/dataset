	localparam integer FIFO_ADDR_WIDTH = LGFIFO;
	localparam integer FFS_STAGES = XCLOCK_FFS;

	// Internal signals
	wire [C_S_AXI_ID_WIDTH-1 : 0] aw_id_m;
	wire [C_S_AXI_ADDR_WIDTH-1 : 0] aw_addr_m;
	wire [7 : 0] aw_len_m;
	wire [2 : 0] aw_size_m;
	wire [1 : 0] aw_burst_m;
	wire aw_lock_m;
	wire [3 : 0] aw_cache_m;
	wire [2 : 0] aw_prot_m;
	wire [3 : 0] aw_qos_m;

	wire aw_ready_s;

	wire [(C_S_AXI_DATA_WIDTH/8)-1 : 0] w_strb_m;
	wire w_last_m;
	wire w_valid_m;
	wire w_ready_s;

	wire [2 : 0] ar_size_m;

		end

				if (aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1 : 0] == aw_fifo_rd_ptr[FIFO_ADDR_WIDTH-1 : 0]) begin
					aw_fifo_full <= 1'b1;
				end
			end else begin
				aw_fifo_full <= 1'b0;
			end

				aw_fifo_addr[aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1 : 0]] <= S_AXI_AWADDR;

				aw_fifo_qos[aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1 : 0]] <= S_AXI_AWQOS;
				aw_fifo_wr_ptr <= aw_fifo_wr_ptr + 1'b1;
			end
		end
	end

	assign M_AXI_AWPROT = aw_prot_m;

	assign M_AXI_AWVALID = aw_valid_m;

	// FIFO for W channel

	assign M_AXI_ARADDR = ar_addr_m;

	// R Channel: Master to Slave
	// FIFO for R channel
	reg [C_S_AXI_ID_WIDTH-1 : 0] r_fifo_id [0 : FIFO_DEPTH-1];
	reg [C_S_AXI_DATA_WIDTH-1 : 0] r_fifo_data [0 : FIFO_DEPTH-1];
	reg [1 : 0] r_fifo_resp [0 : FIFO_DEPTH-1];
	reg r_fifo_last [0 : FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH : 0] r_fifo_wr_ptr;
	reg [FIFO_ADDR_WIDTH : 0] r_fifo_rd_ptr;
	reg r_fifo_empty;
	reg r_fifo_full;

	// R FIFO write logic (M_AXI_ACLK domain)
	always @(posedge M_AXI_ACLK) begin
		if (!M_AXI_ARESETN) begin
			r_fifo_wr_ptr <= 0;
			r_fifo_rd_ptr <= 0;
			r_fifo_empty <= 1'b1;
			r_fifo_full <= 1'b0;
		end else begin
			if (r_fifo_wr_ptr[FIFO_ADDR_WIDTH] != r_fifo_rd_ptr[FIFO_ADDR_WIDTH]) begin
				if (r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1 : 0] == r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1 : 0]) begin
					r_fifo_full <= 1'b1;
				end
			end else begin
				r_fifo_full <= 1'b0;
			end

			if (r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1 : 0] != r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1 : 0]) begin
				r_fifo_empty <= 1'b0;
			end else begin
				r_fifo_empty <= 1'b1;
			end

			if (M_AXI_RVALID && !r_fifo_full) begin
				r_fifo_id[r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1 : 0]] <= M_AXI_RID;
				r_fifo_data[r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1 : 0]] <= M_AXI_RDATA;
				r_fifo_resp[r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1 : 0]] <= M_AXI_RRESP;
				r_fifo_last[r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1 : 0]] <= M_AXI_RLAST;
				r_fifo_wr_ptr <= r_fifo_wr_ptr + 1'b1;
			end
		end
	end

	assign M_AXI_RREADY = !r_fifo_full;

	// R FIFO read logic (S_AXI_ACLK domain)
	always @(posedge S_AXI_ACLK) begin
		if (!S_AXI_ARESETN) begin
			r_fifo_rd_ptr <= 0;
		end else begin
			if (r_valid_s && S_AXI_RREADY) begin
				r_fifo_rd_ptr <= r_fifo_rd_ptr + 1'b1;
			end
		end
	end

	// R FIFO read data
	assign r_id_s = r_fifo_id[r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1 : 0]];
	assign r_data_s = r_fifo_data[r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1 : 0]];
	assign r_resp_s = r_fifo_resp[r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1 : 0]];
	assign r_last_s = r_fifo_last[r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1 : 0]];

	// R valid signal: FIFO not empty
	assign r_valid_s = !r_fifo_empty;

	// R output to slave
	assign S_AXI_RID = r_id_s;
	assign S_AXI_RDATA = r_data_s;
	assign S_AXI_RRESP = r_resp_s;
	assign S_AXI_RLAST = r_last_s;
	assign S_AXI_RVALID = r_valid_s;
