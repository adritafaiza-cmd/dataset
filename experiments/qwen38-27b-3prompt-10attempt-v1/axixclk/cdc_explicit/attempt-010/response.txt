module axixclk #(
		parameter integer C_S_AXI_ID_WIDTH	= 2,
		parameter integer C_S_AXI_DATA_WIDTH	= 32,
		parameter integer C_S_AXI_ADDR_WIDTH	= 6,
		parameter [0:0]	OPT_WRITE_ONLY = 1'b0,

		input	wire [7 : 0]			S_AXI_AWLEN,
		input	wire [2 : 0]			S_AXI_AWSIZE,
		input	wire [1 : 0]			S_AXI_AWBURST,
		input	wire				S_AXI_AWLOCK,
		input	wire [3 : 0]			S_AXI_AWCACHE,

		input	wire				S_AXI_WLAST,
		input	wire				S_AXI_WVALID,
		output	wire				S_AXI_WREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_BID,
		output	wire [1 : 0]			S_AXI_BRESP,
		output	wire				S_AXI_BVALID,
		input	wire				S_AXI_BREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_ARID,
		input	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	S_AXI_ARADDR,
		input	wire [7 : 0]			S_AXI_ARLEN,
		input	wire [2 : 0]			S_AXI_ARSIZE,
		input	wire [1 : 0]			S_AXI_ARBURST,
		input	wire				S_AXI_ARLOCK,
		input	wire [3 : 0]			S_AXI_ARCACHE,
		input	wire [2 : 0]			S_AXI_ARPROT,
		input	wire [3 : 0]			S_AXI_ARQOS,
		input	wire				S_AXI_ARVALID,
		output	wire				S_AXI_ARREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_RID,
		output	wire [C_S_AXI_DATA_WIDTH-1 : 0]	S_AXI_RDATA,
		output	wire [1 : 0]			S_AXI_RRESP,
		output	wire				S_AXI_RLAST,
		output	wire				S_AXI_RVALID,
		input	wire				S_AXI_RREADY,
		input	wire				M_AXI_ACLK,
		output	wire				M_AXI_ARESETN,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_AWID,
		output	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	M_AXI_AWADDR,
		output	wire [7 : 0]			M_AXI_AWLEN,
		output	wire [2 : 0]			M_AXI_AWSIZE,
		output	wire [1 : 0]			M_AXI_AWBURST,
		output	wire				M_AXI_AWLOCK,
		output	wire [3 : 0]			M_AXI_AWCACHE,

		output	wire				M_AXI_AWVALID,
		input	wire				M_AXI_AWREADY,
		output	wire [C_S_AXI_DATA_WIDTH-1 : 0]	M_AXI_WDATA,
		output	wire [(C_S_AXI_DATA_WIDTH/8)-1 : 0] M_AXI_WSTRB,
		output	wire				M_AXI_WLAST,
		output	wire				M_AXI_WVALID,

		output	wire				M_AXI_BREADY,

		output	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	M_AXI_ARADDR,
		output	wire [7 : 0]			M_AXI_ARLEN,
		output	wire [2 : 0]			M_AXI_ARSIZE,
		output	wire [1 : 0]			M_AXI_ARBURST,
		output	wire				M_AXI_ARLOCK,
		output	wire [3 : 0]			M_AXI_ARCACHE,
		output	wire [2 : 0]			M_AXI_ARPROT,
		output	wire [3 : 0]			M_AXI_ARQOS,
		output	wire				M_AXI_ARVALID,
		input	wire				M_AXI_ARREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_RID,
		input	wire [C_S_AXI_DATA_WIDTH-1 : 0]	M_AXI_RDATA,
		input	wire [1 : 0]			M_AXI_RRESP,
		input	wire				M_AXI_RLAST,
		input	wire				M_AXI_RVALID,
		output	wire				M_AXI_RREADY
	);

	// Local parameters
	localparam integer FIFO_DEPTH = 1 << LGFIFO;
	localparam integer FIFO_ADDR_WIDTH = LGFIFO;
	localparam integer AW_FIFO_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4;
	localparam integer AR_FIFO_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4;
	localparam integer W_FIFO_WIDTH = C_S_AXI_DATA_WIDTH + (C_S_AXI_DATA_WIDTH/8) + 1;
	localparam integer B_FIFO_WIDTH = C_S_AXI_ID_WIDTH + 2;
	localparam integer R_FIFO_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_DATA_WIDTH + 2 + 1;

	// Reset synchronization for M_AXI domain
	reg [1:0] m_reset_sync;
	reg m_reset_sync_next;
	wire m_reset_active;

	always @(posedge M_AXI_ACLK) begin
		m_reset_sync <= {m_reset_sync[0], S_AXI_ARESETN};
	end

	assign m_reset_active = ~m_reset_sync[1];
	assign M_AXI_ARESETN = m_reset_sync[1];

	// AW channel: S to M
	reg [AW_FIFO_WIDTH-1:0] aw_fifo_mem [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] aw_wr_ptr;
	reg [FIFO_ADDR_WIDTH:0] aw_rd_ptr;
	wire aw_wr_en;
	wire aw_rd_en;
	wire aw_full;
	wire aw_empty;
	wire [AW_FIFO_WIDTH-1:0] aw_wr_data;
	wire [AW_FIFO_WIDTH-1:0] aw_rd_data;

	assign aw_wr_data = {S_AXI_AWID, S_AXI_AWADDR, S_AXI_AWLEN, S_AXI_AWSIZE, S_AXI_AWBURST, S_AXI_AWLOCK, S_AXI_AWCACHE, S_AXI_AWPROT, S_AXI_AWQOS};
	assign aw_full = (aw_wr_ptr[FIFO_ADDR_WIDTH] != aw_rd_ptr[FIFO_ADDR_WIDTH]) && (aw_wr_ptr[FIFO_ADDR_WIDTH-1:0] == aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
	assign aw_empty = (aw_wr_ptr == aw_rd_ptr);
	assign aw_rd_data = aw_fifo_mem[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];

	assign S_AXI_AWREADY = ~aw_full;
	assign aw_wr_en = S_AXI_AWVALID & S_AXI_AWREADY;

	always @(posedge S_AXI_ACLK) begin
		if (aw_wr_en) begin
			aw_fifo_mem[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= aw_wr_data;
			aw_wr_ptr <= aw_wr_ptr + 1'b1;
		end
	end

	// AW channel: M side read
	reg [AW_FIFO_WIDTH-1:0] m_aw_id;
	reg [C_S_AXI_ADDR_WIDTH-1:0] m_aw_addr;
	reg [7:0] m_aw_len;
	reg [2:0] m_aw_size;
	reg [1:0] m_aw_burst;
	reg m_aw_lock;
	reg [3:0] m_aw_cache;
	reg [2:0] m_aw_prot;
	reg [3:0] m_aw_qos;
	reg m_aw_valid;

	assign M_AXI_AWID = m_aw_id;
	assign M_AXI_AWADDR = m_aw_addr;
	assign M_AXI_AWLEN = m_aw_len;
	assign M_AXI_AWSIZE = m_aw_size;
	assign M_AXI_AWBURST = m_aw_burst;
	assign M_AXI_AWLOCK = m_aw_lock;
	assign M_AXI_AWCACHE = m_aw_cache;
	assign M_AXI_AWPROT = m_aw_prot;
	assign M_AXI_AWQOS = m_aw_qos;
	assign M_AXI_AWVALID = m_aw_valid;

	assign aw_rd_en = m_aw_valid & M_AXI_AWREADY;

	always @(posedge M_AXI_ACLK) begin
		if (m_reset_active) begin
			m_aw_valid <= 1'b0;
		end else begin
			if (aw_rd_en) begin
				m_aw_valid <= 1'b0;
			end else if (~aw_empty) begin
				m_aw_id <= aw_rd_data[AW_FIFO_WIDTH-1 -: C_S_AXI_ID_WIDTH];
				m_aw_addr <= aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-1 -: C_S_AXI_ADDR_WIDTH];
				m_aw_len <= aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-1 -: 8];
				m_aw_size <= aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-9 -: 3];
				m_aw_burst <= aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-12 -: 2];
				m_aw_lock <= aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-14];
				m_aw_cache <= aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-15 -: 4];
				m_aw_prot <= aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-19 -: 3];
				m_aw_qos <= aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-22 -: 4];
				m_aw_valid <= 1'b1;
				aw_rd_ptr <= aw_rd_ptr + 1'b1;
			end
		end
	end

	// W channel: S to M
	reg [W_FIFO_WIDTH-1:0] w_fifo_mem [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] w_wr_ptr;
	reg [FIFO_ADDR_WIDTH:0] w_rd_ptr;
	wire w_wr_en;
	wire w_rd_en;
	wire w_full;
	wire w_empty;
	wire [W_FIFO_WIDTH-1:0] w_wr_data;
	wire [W_FIFO_WIDTH-1:0] w_rd_data;

	assign w_wr_data = {S_AXI_WDATA, S_AXI_WSTRB, S_AXI_WLAST};
	assign w_full = (w_wr_ptr[FIFO_ADDR_WIDTH] != w_rd_ptr[FIFO_ADDR_WIDTH]) && (w_wr_ptr[FIFO_ADDR_WIDTH-1:0] == w_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
	assign w_empty = (w_wr_ptr == w_rd_ptr);
	assign w_rd_data = w_fifo_mem[w_rd_ptr[FIFO_ADDR_WIDTH-1:0]];

	assign S_AXI_WREADY = ~w_full;
	assign w_wr_en = S_AXI_WVALID & S_AXI_WREADY;

	always @(posedge S_AXI_ACLK) begin
		if (w_wr_en) begin
			w_fifo_mem[w_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= w_wr_data;
			w_wr_ptr <= w_wr_ptr + 1'b1;
		end
	end

	// W channel: M side read
	reg [C_S_AXI_DATA_WIDTH-1:0] m_w_data;
	reg [(C_S_AXI_DATA_WIDTH/8)-1:0] m_w_strb;
	reg m_w_last;
	reg m_w_valid;

	assign M_AXI_WDATA = m_w_data;
	assign M_AXI_WSTRB = m_w_strb;
	assign M_AXI_WLAST = m_w_last;
	assign M_AXI_WVALID = m_w_valid;

	assign w_rd_en = m_w_valid & M_AXI_WREADY;

	always @(posedge M_AXI_ACLK) begin
		if (m_reset_active) begin
			m_w_valid <= 1'b0;
		end else begin
			if (w_rd_en) begin
				m_w_valid <= 1'b0;
			end else if (~w_empty) begin
				m_w_data <= w_rd_data[W_FIFO_WIDTH-1 -: C_S_AXI_DATA_WIDTH];
				m_w_strb <= w_rd_data[C_S_AXI_DATA_WIDTH-1 -: (C_S_AXI_DATA_WIDTH/8)];
				m_w_last <= w_rd_data[0];
				m_w_valid <= 1'b1;
				w_rd_ptr <= w_rd_ptr + 1'b1;
			end
		end
	end

	// B channel: M to S
	reg [B_FIFO_WIDTH-1:0] b_fifo_mem [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] b_wr_ptr;
	reg [FIFO_ADDR_WIDTH:0] b_rd_ptr;
	wire b_wr_en;
	wire b_rd_en;
	wire b_full;
	wire b_empty;
	wire [B_FIFO_WIDTH-1:0] b_wr_data;
	wire [B_FIFO_WIDTH-1:0] b_rd_data;

	assign b_wr_data = {M_AXI_BID, M_AXI_BRESP};
	assign b_full = (b_wr_ptr[FIFO_ADDR_WIDTH] != b_rd_ptr[FIFO_ADDR_WIDTH]) && (b_wr_ptr[FIFO_ADDR_WIDTH-1:0] == b_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
	assign b_empty = (b_wr_ptr == b_rd_ptr);
	assign b_rd_data = b_fifo_mem[b_rd_ptr[FIFO_ADDR_WIDTH-1:0]];

	assign M_AXI_BREADY = ~b_full;
	assign b_wr_en = M_AXI_BVALID & M_AXI_BREADY;

	always @(posedge M_AXI_ACLK) begin
		if (b_wr_en) begin
			b_fifo_mem[b_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= b_wr_data;
			b_wr_ptr <= b_wr_ptr + 1'b1;
		end
	end

	// B channel: S side read
	reg [C_S_AXI_ID_WIDTH-1:0] s_b_id;
	reg [1:0] s_b_resp;
	reg s_b_valid;

	assign S_AXI_BID = s_b_id;
	assign S_AXI_BRESP = s_b_resp;
	assign S_AXI_BVALID = s_b_valid;

	assign b_rd_en = s_b_valid & S_AXI_BREADY;

	always @(posedge S_AXI_ACLK) begin
		if (!S_AXI_ARESETN) begin
			s_b_valid <= 1'b0;
		end else begin
			if (b_rd_en) begin
				s_b_valid <= 1'b0;
			end else if (~b_empty) begin
				s_b_id <= b_rd_data[B_FIFO_WIDTH-1 -: C_S_AXI_ID_WIDTH];
				s_b_resp <= b_rd_data[C_S_AXI_ID_WIDTH-1 -: 2];
				s_b_valid <= 1'b1;
				b_rd_ptr <= b_rd_ptr + 1'b1;
			end
		end
	end

	// AR channel: S to M
	reg [AR_FIFO_WIDTH-1:0] ar_fifo_mem [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] ar_wr_ptr;
	reg [FIFO_ADDR_WIDTH:0] ar_rd_ptr;
	wire ar_wr_en;
	wire ar_rd_en;
	wire ar_full;
	wire ar_empty;
	wire [AR_FIFO_WIDTH-1:0] ar_wr_data;
	wire [AR_FIFO_WIDTH-1:0] ar_rd_data;

	assign ar_wr_data = {S_AXI_ARID, S_AXI_ARADDR, S_AXI_ARLEN, S_AXI_ARSIZE, S_AXI_ARBURST, S_AXI_ARLOCK, S_AXI_ARCACHE, S_AXI_ARPROT, S_AXI_ARQOS};
	assign ar_full = (ar_wr_ptr[FIFO_ADDR_WIDTH] != ar_rd_ptr[FIFO_ADDR_WIDTH]) && (ar_wr_ptr[FIFO_ADDR_WIDTH-1:0] == ar_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
	assign ar_empty = (ar_wr_ptr == ar_rd_ptr);
	assign ar_rd_data = ar_fifo_mem[ar_rd_ptr[FIFO_ADDR_WIDTH-1:0]];

	assign ar_wr_en = S_AXI_ARVALID & S_AXI_ARREADY;

			ar_fifo_mem[ar_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= ar_wr_data;

	reg [C_S_AXI_ID_WIDTH-1:0] m_ar_id;
	reg [C_S_AXI_ADDR_WIDTH-1:0] m_ar_addr;
	reg [7:0] m_ar_len;
	reg [2:0] m_ar_size;
	reg [1:0] m_ar_burst;
	reg m_ar_lock;
	reg [3:0] m_ar_cache;
	reg [2:0] m_ar_prot;
	reg [3:0] m_ar_qos;

	assign M_AXI_ARADDR = m_ar_addr;
	assign M_AXI_ARLEN = m_ar_len;
	assign M_AXI_ARSIZE = m_ar_size;
	assign M_AXI_ARBURST = m_ar_burst;
	assign M_AXI_ARLOCK = m_ar_lock;
	assign M_AXI_ARCACHE = m_ar_cache;
	assign M_AXI_ARPROT = m_ar_prot;
	assign M_AXI_ARQOS = m_ar_qos;
	assign M_AXI_ARVALID = m_ar_valid;

			m_ar_valid <= 1'b0;

				m_ar_valid <= 1'b0;
			end else if (~ar_empty) begin

				m_ar_size <= ar_rd_data[AR_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-9 -: 3];
				m_ar_burst <= ar_rd_data[AR_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-12 -: 2];
				m_ar_lock <= ar_rd_data[AR_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-14];
				m_ar_cache <= ar_rd_data[AR_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-15 -: 4];
				m_ar_prot <= ar_rd_data[AR_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-19 -: 3];
				m_ar_qos <= ar_rd_data[AR_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-22 -: 4];
				m_ar_valid <= 1'b1;
				ar_rd_ptr <= ar_rd_ptr + 1'b1;
			end
		end
	end

	// R channel: M to S
	reg [R_FIFO_WIDTH-1:0] r_fifo_mem [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] r_wr_ptr;
	reg [FIFO_ADDR_WIDTH:0] r_rd_ptr;
	wire r_wr_en;
	wire r_rd_en;
	wire r_full;
	wire r_empty;
	wire [R_FIFO_WIDTH-1:0] r_wr_data;
	wire [R_FIFO_WIDTH-1:0] r_rd_data;

	assign r_full = (r_wr_ptr[FIFO_ADDR_WIDTH] != r_rd_ptr[FIFO_ADDR_WIDTH]) && (r_wr_ptr[FIFO_ADDR_WIDTH-1:0] == r_rd_ptr[FIFO_ADDR_WIDTH-1:0]);

	assign r_wr_en = M_AXI_RVALID & M_AXI_RREADY;

	always @(posedge M_AXI_ACLK) begin
		if (r_wr_en) begin
			r_fifo_mem[r_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= r_wr_data;
			r_wr_ptr <= r_wr_ptr + 1'b1;
		end
	end

	// R channel: S side read
	reg [C_S_AXI_ID_WIDTH-1:0] s_r_id;
	reg [C_S_AXI_DATA_WIDTH-1:0] s_r_data;
	reg [1:0] s_r_resp;
	reg s_r_last;
	reg s_r_valid;

	assign S_AXI_RID = s_r_id;
	assign S_AXI_RDATA = s_r_data;
	assign S_AXI_RRESP = s_r_resp;
	assign S_AXI_RLAST = s_r_last;
	assign S_AXI_RVALID = s_r_valid;

	assign r_rd_en = s_r_valid & S_AXI_RREADY;

	always @(posedge S_AXI_ACLK) begin

			s_r_valid <= 1'b0;
		end else begin
			if (r_rd_en) begin
				s_r_valid <= 1'b0;
			end else if (~r_empty) begin
				s_r_id <= r_rd_data[R_FIFO_WIDTH-1 -: C_S_AXI_ID_WIDTH];
				s_r_data <= r_rd_data[C_S_AXI_ID_WIDTH-1 -: C_S_AXI_DATA_WIDTH];
				s_r_resp <= r_rd_data[C_S_AXI_ID_WIDTH+C_S_AXI_DATA_WIDTH-1 -: 2];
				s_r_last <= r_rd_data[0];
				s_r_valid <= 1'b1;
				r_rd_ptr <= r_rd_ptr + 1'b1;
			end
		end
	end
