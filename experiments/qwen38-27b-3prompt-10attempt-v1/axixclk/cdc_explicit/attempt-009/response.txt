module axixclk #(

		input	wire [7 : 0]			S_AXI_AWLEN,
		input	wire [2 : 0]			S_AXI_AWSIZE,
		input	wire [1 : 0]			S_AXI_AWBURST,
		input	wire				S_AXI_AWLOCK,
		input	wire [3 : 0]			S_AXI_AWCACHE,

		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_BID,
		output	wire [1 : 0]			S_AXI_BRESP,
		output	wire				S_AXI_BVALID,
		input	wire				S_AXI_BREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_ARID,

		input	wire [2 : 0]			S_AXI_ARSIZE,
		input	wire [1 : 0]			S_AXI_ARBURST,
		input	wire				S_AXI_ARLOCK,
		input	wire [3 : 0]			S_AXI_ARCACHE,
		input	wire [2 : 0]			S_AXI_ARPROT,
		input	wire [3 : 0]			S_AXI_ARQOS,
		input	wire				S_AXI_ARVALID,
		output	wire				S_AXI_ARREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_RID,
		output	wire [C_S_AXI_DATA_WIDTH-1 : 0]	S_AXI_RDATA,
		output	wire [1 : 0]			S_AXI_RRESP,
		output	wire				S_AXI_RLAST,
		output	wire				S_AXI_RVALID,
		input	wire				S_AXI_RREADY,
		input	wire				M_AXI_ACLK,
		output	wire				M_AXI_ARESETN,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_AWID,
		output	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	M_AXI_AWADDR,
		output	wire [7 : 0]			M_AXI_AWLEN,

		output	wire [2 : 0]			M_AXI_ARPROT,
		output	wire [3 : 0]			M_AXI_ARQOS,

		input	wire				M_AXI_ARREADY,

	localparam integer AR_FIFO_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4;
	localparam integer W_FIFO_WIDTH = C_S_AXI_DATA_WIDTH + (C_S_AXI_DATA_WIDTH/8) + 1;
	localparam integer B_FIFO_WIDTH = C_S_AXI_ID_WIDTH + 2;
	localparam integer R_FIFO_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_DATA_WIDTH + 2 + 1;

		if (!S_AXI_ARESETN) begin
			m_reset_sync <= 2'b00;
		end else begin
			m_reset_sync <= {m_reset_sync[0], 1'b1};
		end
	end
	assign M_AXI_ARESETN = m_reset_sync[1];

	// AW channel: S to M
	reg [AW_FIFO_WIDTH-1:0] aw_fifo_mem [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] aw_wr_ptr, aw_rd_ptr;
	reg aw_wr_en, aw_rd_en;
	wire aw_full, aw_empty;

	                 (aw_wr_ptr[FIFO_ADDR_WIDTH-1:0] == aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
	assign aw_empty = (aw_wr_ptr == aw_rd_ptr);

	// AW write side (S clock)
	reg [AW_FIFO_WIDTH-1:0] aw_wr_data;
	assign aw_wr_data = {S_AXI_AWID, S_AXI_AWADDR, S_AXI_AWLEN, S_AXI_AWSIZE, S_AXI_AWBURST, 
	                     S_AXI_AWLOCK, S_AXI_AWCACHE, S_AXI_AWPROT, S_AXI_AWQOS};
	assign S_AXI_AWREADY = !aw_full;
	assign aw_wr_en = S_AXI_AWVALID && S_AXI_AWREADY;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			aw_wr_ptr <= {FIFO_ADDR_WIDTH+1{1'b0}};
		end else if (aw_wr_en) begin
			aw_fifo_mem[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= aw_wr_data;
			aw_wr_ptr <= aw_wr_ptr + 1'b1;
		end
	end

	reg [AW_FIFO_WIDTH-1:0] aw_rd_data;

	assign M_AXI_AWADDR = aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-1 : AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH];
	assign M_AXI_AWLEN = aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-1 : AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8];
	assign M_AXI_AWSIZE = aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-1 : AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3];
	assign M_AXI_AWBURST = aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-1 : AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2];
	assign M_AXI_AWLOCK = aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1];
	assign M_AXI_AWCACHE = aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-1 : AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4];
	assign M_AXI_AWPROT = aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4-1 : AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4-3];
	assign M_AXI_AWQOS = aw_rd_data[AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4-3-1 : AW_FIFO_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4-3-4];
	assign M_AXI_AWVALID = !aw_empty;
	assign aw_rd_en = M_AXI_AWVALID && M_AXI_AWREADY;

	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			aw_rd_ptr <= {FIFO_ADDR_WIDTH+1{1'b0}};
		end else if (aw_rd_en) begin
			aw_rd_data <= aw_fifo_mem[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			aw_rd_ptr <= aw_rd_ptr + 1'b1;
		end
	end

	// W channel: S to M
	reg [W_FIFO_WIDTH-1:0] w_fifo_mem [0:FIFO_DEPTH-1];

		if (!M_AXI_ARESETN) begin
			w_rd_ptr <= {FIFO_ADDR_WIDTH+1{1'b0}};
			w_valid_reg <= 1'b0;
		end else begin
			if (w_rd_en) begin
				w_rd_ptr <= w_rd_ptr + 1'b1;
				w_valid_reg <= !w_empty;
			end else if (!w_valid_reg && !w_empty) begin
				w_rd_data <= w_fifo_mem[w_rd_ptr[FIFO_ADDR_WIDTH-1:0]];

	assign S_AXI_BVALID = !b_empty;
	assign b_rd_en = S_AXI_BVALID && S_AXI_BREADY;

		if (!S_AXI_ARESETN) begin
			b_rd_ptr <= {FIFO_ADDR_WIDTH+1{1'b0}};
		end else if (b_rd_en) begin
			b_rd_data <= b_fifo_mem[b_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			b_rd_ptr <= b_rd_ptr + 1'b1;
		end

	reg [FIFO_ADDR_WIDTH:0] ar_wr_ptr, ar_rd_ptr;

	assign ar_empty = (ar_wr_ptr == ar_rd_ptr);

	reg [AR_FIFO_WIDTH-1:0] ar_wr_data;
	assign ar_wr_data = {S_AXI_ARID, S_AXI_ARADDR, S_AXI_ARLEN, S_AXI_ARSIZE, S_AXI_ARBURST, 
	                     S_AXI_ARLOCK, S_AXI_ARCACHE, S_AXI_ARPROT, S_AXI_ARQOS};
	assign S_AXI_ARREADY = !ar_full;
	assign ar_wr_en = S_AXI_ARVALID && S_AXI_ARREADY;
