	// Local parameters
	localparam integer AW_FIFO_DEPTH = 1 << LGFIFO;
	localparam integer AW_FIFO_PTR_WIDTH = LGFIFO;
	localparam integer W_FIFO_DEPTH = 1 << LGFIFO;
	localparam integer W_FIFO_PTR_WIDTH = LGFIFO;
	localparam integer B_FIFO_DEPTH = 1 << LGFIFO;
	localparam integer B_FIFO_PTR_WIDTH = LGFIFO;
	localparam integer AR_FIFO_DEPTH = 1 << LGFIFO;
	localparam integer AR_FIFO_PTR_WIDTH = LGFIFO;
	localparam integer R_FIFO_DEPTH = 1 << LGFIFO;
	localparam integer R_FIFO_PTR_WIDTH = LGFIFO;

	localparam integer AW_DATA_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4;
	localparam integer W_DATA_WIDTH = C_S_AXI_DATA_WIDTH + (C_S_AXI_DATA_WIDTH/8) + 1;
	localparam integer B_DATA_WIDTH = C_S_AXI_ID_WIDTH + 2;
	localparam integer AR_DATA_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4;
	localparam integer R_DATA_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_DATA_WIDTH + 2 + 1;

	// Reset synchronization
	reg [1:0] m_reset_sync;
	reg m_reset_sync_next;
	always @(posedge M_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			m_reset_sync <= 2'b00;
		end else begin
			m_reset_sync <= {m_reset_sync[0], 1'b1};
		end
	end
	assign M_AXI_ARESETN = m_reset_sync[1];

	// AW channel: S to M
	reg [AW_FIFO_PTR_WIDTH-1:0] aw_wr_ptr;
	reg [AW_FIFO_PTR_WIDTH-1:0] aw_rd_ptr;
	reg [AW_DATA_WIDTH-1:0] aw_fifo_mem [0:AW_FIFO_DEPTH-1];
	reg aw_wr_en;
	reg aw_rd_en;
	wire aw_full;
	wire aw_empty;

	assign aw_full = (aw_wr_ptr == aw_rd_ptr) && (aw_wr_ptr[AW_FIFO_PTR_WIDTH-1] != aw_rd_ptr[AW_FIFO_PTR_WIDTH-1]);
	assign aw_empty = (aw_wr_ptr == aw_rd_ptr);

	// AW write side (S clock)
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			aw_wr_ptr <= {AW_FIFO_PTR_WIDTH{1'b0}};
			aw_wr_en <= 1'b0;
		end else begin
			aw_wr_en <= S_AXI_AWVALID && !aw_full;
			if (S_AXI_AWVALID && !aw_full) begin
				aw_fifo_mem[aw_wr_ptr[AW_FIFO_PTR_WIDTH-2:0]] <= {
					S_AXI_AWID,
					S_AXI_AWADDR,
					S_AXI_AWLEN,
					S_AXI_AWSIZE,
					S_AXI_AWBURST,
					S_AXI_AWLOCK,
					S_AXI_AWCACHE,
					S_AXI_AWPROT,
					S_AXI_AWQOS
				};
				aw_wr_ptr <= aw_wr_ptr + 1'b1;
			end
		end
	end

	assign S_AXI_AWREADY = !aw_full;

	// AW read side (M clock)
	reg [AW_DATA_WIDTH-1:0] aw_rd_data;
	reg [AW_FIFO_PTR_WIDTH-1:0] aw_rd_ptr_m;
	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			aw_rd_ptr_m <= {AW_FIFO_PTR_WIDTH{1'b0}};
			aw_rd_en <= 1'b0;
			aw_rd_data <= {AW_DATA_WIDTH{1'b0}};
		end else begin
			aw_rd_en <= !aw_empty && M_AXI_AWREADY;
			if (!aw_empty && M_AXI_AWREADY) begin
				aw_rd_data <= aw_fifo_mem[aw_rd_ptr_m[AW_FIFO_PTR_WIDTH-2:0]];
				aw_rd_ptr_m <= aw_rd_ptr_m + 1'b1;
			end
		end
	end

	// AW output
	assign M_AXI_AWID = aw_rd_data[AW_DATA_WIDTH-1 : AW_DATA_WIDTH-C_S_AXI_ID_WIDTH];
	assign M_AXI_AWADDR = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-1 : AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH];
	assign M_AXI_AWLEN = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-1 : AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8];
	assign M_AXI_AWSIZE = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-1 : AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3];
	assign M_AXI_AWBURST = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-1 : AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2];
	assign M_AXI_AWLOCK = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1];
	assign M_AXI_AWCACHE = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-1 : AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4];
	assign M_AXI_AWPROT = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4-1 : AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4-3];
	assign M_AXI_AWQOS = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4-3-1 : 0];
	assign M_AXI_AWVALID = !aw_empty;

	// W channel: S to M
	reg [W_FIFO_PTR_WIDTH-1:0] w_wr_ptr;
	reg [W_FIFO_PTR_WIDTH-1:0] w_rd_ptr;
	reg [W_DATA_WIDTH-1:0] w_fifo_mem [0:W_FIFO_DEPTH-1];
	reg w_wr_en;
	reg w_rd_en;
	wire w_full;
	wire w_empty;

	assign w_full = (w_wr_ptr == w_rd_ptr) && (w_wr_ptr[W_FIFO_PTR_WIDTH-1] != w_rd_ptr[W_FIFO_PTR_WIDTH-1]);
	assign w_empty = (w_wr_ptr == w_rd_ptr);

	// W write side (S clock)
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			w_wr_ptr <= {W_FIFO_PTR_WIDTH{1'b0}};
			w_wr_en <= 1'b0;
		end else begin
			w_wr_en <= S_AXI_WVALID && !w_full;
			if (S_AXI_WVALID && !w_full) begin
				w_fifo_mem[w_wr_ptr[W_FIFO_PTR_WIDTH-2:0]] <= {
					S_AXI_WDATA,
					S_AXI_WSTRB,
					S_AXI_WLAST
				};
				w_wr_ptr <= w_wr_ptr + 1'b1;
			end
		end
	end

	assign S_AXI_WREADY = !w_full;

	// W read side (M clock)
	reg [W_DATA_WIDTH-1:0] w_rd_data;
	reg [W_FIFO_PTR_WIDTH-1:0] w_rd_ptr_m;
	reg w_valid_reg;
	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			w_rd_ptr_m <= {W_FIFO_PTR_WIDTH{1'b0}};
			w_rd_en <= 1'b0;
			w_rd_data <= {W_DATA_WIDTH{1'b0}};
			w_valid_reg <= 1'b0;
		end else begin
			w_rd_en <= !w_empty && M_AXI_WREADY;
			if (!w_empty && M_AXI_WREADY) begin
				w_rd_data <= w_fifo_mem[w_rd_ptr_m[W_FIFO_PTR_WIDTH-2:0]];
				w_rd_ptr_m <= w_rd_ptr_m + 1'b1;
			end
			// Hold one extra M_AXI clock before M_AXI_WVALID
			w_valid_reg <= !w_empty;
		end
	end

	assign M_AXI_WDATA = w_rd_data[W_DATA_WIDTH-1 : W_DATA_WIDTH-C_S_AXI_DATA_WIDTH];
	assign M_AXI_WSTRB = w_rd_data[W_DATA_WIDTH-C_S_AXI_DATA_WIDTH-1 : W_DATA_WIDTH-C_S_AXI_DATA_WIDTH-(C_S_AXI_DATA_WIDTH/8)];
	assign M_AXI_WLAST = w_rd_data[0];
	assign M_AXI_WVALID = w_valid_reg;

	// B channel: M to S
	reg [B_FIFO_PTR_WIDTH-1:0] b_wr_ptr;
	reg [B_FIFO_PTR_WIDTH-1:0] b_rd_ptr;
	reg [B_DATA_WIDTH-1:0] b_fifo_mem [0:B_FIFO_DEPTH-1];
	reg b_wr_en;
	reg b_rd_en;
	wire b_full;
	wire b_empty;

	assign b_full = (b_wr_ptr == b_rd_ptr) && (b_wr_ptr[B_FIFO_PTR_WIDTH-1] != b_rd_ptr[B_FIFO_PTR_WIDTH-1]);
	assign b_empty = (b_wr_ptr == b_rd_ptr);

	// B write side (M clock)
	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			b_wr_ptr <= {B_FIFO_PTR_WIDTH{1'b0}};
			b_wr_en <= 1'b0;
		end else begin
			b_wr_en <= M_AXI_BVALID && !b_full;
			if (M_AXI_BVALID && !b_full) begin
				b_fifo_mem[b_wr_ptr[B_FIFO_PTR_WIDTH-2:0]] <= {
					M_AXI_BID,
					M_AXI_BRESP
				};
				b_wr_ptr <= b_wr_ptr + 1'b1;
			end
		end
	end

	assign M_AXI_BREADY = !b_full;

	// B read side (S clock)
	reg [B_DATA_WIDTH-1:0] b_rd_data;
	reg [B_FIFO_PTR_WIDTH-1:0] b_rd_ptr_s;
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			b_rd_ptr_s <= {B_FIFO_PTR_WIDTH{1'b0}};
			b_rd_en <= 1'b0;
			b_rd_data <= {B_DATA_WIDTH{1'b0}};
		end else begin
			b_rd_en <= !b_empty && S_AXI_BREADY;
			if (!b_empty && S_AXI_BREADY) begin
				b_rd_data <= b_fifo_mem[b_rd_ptr_s[B_FIFO_PTR_WIDTH-2:0]];
				b_rd_ptr_s <= b_rd_ptr_s + 1'b1;
			end
		end
	end

	assign S_AXI_BID = b_rd_data[B_DATA_WIDTH-1 : B_DATA_WIDTH-C_S_AXI_ID_WIDTH];
	assign S_AXI_BRESP = b_rd_data[C_S_AXI_ID_WIDTH-1 : 0];
	assign S_AXI_BVALID = !b_empty;

	// AR channel: S to M
	reg [AR_FIFO_PTR_WIDTH-1:0] ar_wr_ptr;
	reg [AR_FIFO_PTR_WIDTH-1:0] ar_rd_ptr;
	reg [AR_DATA_WIDTH-1:0] ar_fifo_mem [0:AR_FIFO_DEPTH-1];
	reg ar_wr_en;
	reg ar_rd_en;
	wire ar_full;
	wire ar_empty;

	assign ar_full = (ar_wr_ptr == ar_rd_ptr) && (ar_wr_ptr[AR_FIFO_PTR_WIDTH-1] != ar_rd_ptr[AR_FIFO_PTR_WIDTH-1]);
	assign ar_empty = (ar_wr_ptr == ar_rd_ptr);

	// AR write side (S clock)
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			ar_wr_ptr <= {AR_FIFO_PTR_WIDTH{1'b0}};
			ar_wr_en <= 1'b0;
		end else begin
			ar_wr_en <= S_AXI_ARVALID && !ar_full;
			if (S_AXI_ARVALID && !ar_full) begin
				ar_fifo_mem[ar_wr_ptr[AR_FIFO_PTR_WIDTH-2:0]] <= {
					S_AXI_ARID,
					S_AXI_ARADDR,
					S_AXI_ARLEN,
					S_AXI_ARSIZE,
					S_AXI_ARBURST,
					S_AXI_ARLOCK,
					S_AXI_ARCACHE,
					S_AXI_ARPROT,
					S_AXI_ARQOS
				};
				ar_wr_ptr <= ar_wr_ptr + 1'b1;
			end
		end
	end

	assign S_AXI_ARREADY = !ar_full;

	// AR read side (M clock)
	reg [AR_DATA_WIDTH-1:0] ar_rd_data;
	reg [AR_FIFO_PTR_WIDTH-1:0] ar_rd_ptr_m;
	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			ar_rd_ptr_m <= {AR_FIFO_PTR_WIDTH{1'b0}};
			ar_rd_en <= 1'b0;
			ar_rd_data <= {AR_DATA_WIDTH{1'b0}};
		end else begin
			ar_rd_en <= !ar_empty && M_AXI_ARREADY;
			if (!ar_empty && M_AXI_ARREADY) begin
				ar_rd_data <= ar_fifo_mem[ar_rd_ptr_m[AR_FIFO_PTR_WIDTH-2:0]];
				ar_rd_ptr_m <= ar_rd_ptr_m + 1'b1;
			end
		end
	end

	assign M_AXI_ARADDR = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-1 : AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH];
	assign M_AXI_ARLEN = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-1 : AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8];
	assign M_AXI_ARSIZE = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-1 : AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3];
	assign M_AXI_ARBURST = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-1 : AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2];
	assign M_AXI_ARLOCK = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1];
	assign M_AXI_ARCACHE = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-1 : AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4];
	assign M_AXI_ARPROT = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4-1 : AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4-3];
	assign M_AXI_ARQOS = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-4-3-1 : 0];

	reg [R_FIFO_PTR_WIDTH-1:0] r_wr_ptr;
	reg [R_FIFO_PTR_WIDTH-1:0] r_rd_ptr;

	reg r_wr_en;

	wire r_empty;

	assign r_empty = (r_wr_ptr == r_rd_ptr);

	// R write side (M clock)
	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			r_wr_ptr <= {R_FIFO_PTR_WIDTH{1'b0}};
			r_wr_en <= 1'b0;
		end else begin
			r_wr_en <= M_AXI_RVALID && !r_full;
			if (M_AXI_RVALID && !r_full) begin
				r_fifo_mem[r_wr_ptr[R_FIFO_PTR_WIDTH-2:0]] <= {
					M_AXI_RID,
					M_AXI_RDATA,
					M_AXI_RRESP,
					M_AXI_RLAST
				};
				r_wr_ptr <= r_wr_ptr + 1'b1;
			end
		end
	end

	reg [R_FIFO_PTR_WIDTH-1:0] r_rd_ptr_s;
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			r_rd_ptr_s <= {R_FIFO_PTR_WIDTH{1'b0}};
			r_rd_en <= 1'b0;
