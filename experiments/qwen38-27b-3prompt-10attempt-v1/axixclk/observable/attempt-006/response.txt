module axixclk #(

		parameter integer C_S_AXI_DATA_WIDTH	= 32,

		input	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	S_AXI_AWADDR,
		input	wire [7 : 0]			S_AXI_AWLEN,
		input	wire [2 : 0]			S_AXI_AWSIZE,
		input	wire [1 : 0]			S_AXI_AWBURST,
		input	wire				S_AXI_AWLOCK,

		input	wire [C_S_AXI_DATA_WIDTH-1 : 0]	S_AXI_WDATA,
		input	wire [(C_S_AXI_DATA_WIDTH/8)-1 : 0] S_AXI_WSTRB,
		input	wire				S_AXI_WLAST,
		input	wire				S_AXI_WVALID,
		output	wire				S_AXI_WREADY,
		output	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_BID,
		output	wire [1 : 0]			S_AXI_BRESP,
		output	wire				S_AXI_BVALID,
		input	wire				S_AXI_BREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	S_AXI_ARID,
		input	wire [C_S_AXI_ADDR_WIDTH-1 : 0]	S_AXI_ARADDR,
		input	wire [7 : 0]			S_AXI_ARLEN,
		input	wire [2 : 0]			S_AXI_ARSIZE,

		input	wire				S_AXI_ARVALID,
		output	wire				S_AXI_ARREADY,

	localparam integer AW_FIFO_DEPTH = 2**LGFIFO;
	localparam integer AW_FIFO_ADDR_WIDTH = LGFIFO;
	localparam integer W_FIFO_DEPTH = 2**LGFIFO;
	localparam integer W_FIFO_ADDR_WIDTH = LGFIFO;
	localparam integer B_FIFO_DEPTH = 2**LGFIFO;
	localparam integer B_FIFO_ADDR_WIDTH = LGFIFO;
	localparam integer AR_FIFO_DEPTH = 2**LGFIFO;
	localparam integer AR_FIFO_ADDR_WIDTH = LGFIFO;
	localparam integer R_FIFO_DEPTH = 2**LGFIFO;
	localparam integer R_FIFO_ADDR_WIDTH = LGFIFO;

	localparam integer AW_DATA_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4;
	localparam integer W_DATA_WIDTH = C_S_AXI_DATA_WIDTH + (C_S_AXI_DATA_WIDTH/8) + 1;
	localparam integer B_DATA_WIDTH = C_S_AXI_ID_WIDTH + 2;
	localparam integer AR_DATA_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_ADDR_WIDTH + 8 + 3 + 2 + 1 + 4 + 3 + 4;
	localparam integer R_DATA_WIDTH = C_S_AXI_ID_WIDTH + C_S_AXI_DATA_WIDTH + 2 + 1;

	// Reset synchronization
	reg [1:0] m_reset_sync;
	reg m_reset_n;
	always @(posedge M_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			m_reset_sync <= 2'b00;
			m_reset_n <= 1'b0;
		end else begin
			m_reset_sync <= {m_reset_sync[0], 1'b1};
			m_reset_n <= m_reset_sync[1];
		end
	end
	assign M_AXI_ARESETN = m_reset_n;

	reg [AW_DATA_WIDTH-1:0] aw_fifo_mem [0:AW_FIFO_DEPTH-1];
	reg [AW_FIFO_ADDR_WIDTH:0] aw_wr_ptr, aw_rd_ptr;
	wire aw_full = (aw_wr_ptr[AW_FIFO_ADDR_WIDTH:1] == aw_rd_ptr[AW_FIFO_ADDR_WIDTH:1]) && (aw_wr_ptr[0] != aw_rd_ptr[0]);
	wire aw_empty = (aw_wr_ptr == aw_rd_ptr);

	// AW write side (S clock)
	reg [AW_DATA_WIDTH-1:0] aw_wr_data;
	assign aw_wr_data = {S_AXI_AWID, S_AXI_AWADDR, S_AXI_AWLEN, S_AXI_AWSIZE, S_AXI_AWBURST, S_AXI_AWLOCK, S_AXI_AWCACHE, S_AXI_AWPROT, S_AXI_AWQOS};
	assign S_AXI_AWREADY = !aw_full;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			aw_wr_ptr <= 0;
		end else if (S_AXI_AWVALID && S_AXI_AWREADY) begin
			aw_fifo_mem[aw_wr_ptr[AW_FIFO_ADDR_WIDTH-1:0]] <= aw_wr_data;
			aw_wr_ptr <= aw_wr_ptr + 1'b1;
		end
	end

	// AW read side (M clock)
	reg [AW_DATA_WIDTH-1:0] aw_rd_data;
	assign M_AXI_AWID = aw_rd_data[AW_DATA_WIDTH-1 -: C_S_AXI_ID_WIDTH];
	assign M_AXI_AWADDR = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-1 -: C_S_AXI_ADDR_WIDTH];
	assign M_AXI_AWLEN = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-1 -: 8];
	assign M_AXI_AWSIZE = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-1 -: 3];
	assign M_AXI_AWBURST = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-1 -: 2];
	assign M_AXI_AWLOCK = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1];
	assign M_AXI_AWCACHE = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-1 -: 4];
	assign M_AXI_AWPROT = aw_rd_data[AW_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-1-4-1 -: 3];
	assign M_AXI_AWQOS = aw_rd_data[0:3];
	assign M_AXI_AWVALID = !aw_empty;

	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			aw_rd_ptr <= 0;
			aw_rd_data <= 0;
		end else if (M_AXI_AWVALID && M_AXI_AWREADY) begin
			aw_rd_ptr <= aw_rd_ptr + 1'b1;
		end
		if (!aw_empty) begin
			aw_rd_data <= aw_fifo_mem[aw_rd_ptr[AW_FIFO_ADDR_WIDTH-1:0]];
		end
	end

	// W Channel: S to M
	reg [W_DATA_WIDTH-1:0] w_fifo_mem [0:W_FIFO_DEPTH-1];
	reg [W_FIFO_ADDR_WIDTH:0] w_wr_ptr, w_rd_ptr;
	wire w_full = (w_wr_ptr[W_FIFO_ADDR_WIDTH:1] == w_rd_ptr[W_FIFO_ADDR_WIDTH:1]) && (w_wr_ptr[0] != w_rd_ptr[0]);
	wire w_empty = (w_wr_ptr == w_rd_ptr);

	// W write side (S clock)
	reg [W_DATA_WIDTH-1:0] w_wr_data;
	assign w_wr_data = {S_AXI_WDATA, S_AXI_WSTRB, S_AXI_WLAST};
	assign S_AXI_WREADY = !w_full;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			w_wr_ptr <= 0;
		end else if (S_AXI_WVALID && S_AXI_WREADY) begin
			w_fifo_mem[w_wr_ptr[W_FIFO_ADDR_WIDTH-1:0]] <= w_wr_data;
			w_wr_ptr <= w_wr_ptr + 1'b1;
		end
	end

	// W read side (M clock) - with extra register for M_AXI_WVALID
	reg [W_DATA_WIDTH-1:0] w_rd_data_reg1, w_rd_data_reg2;
	reg w_valid_reg1, w_valid_reg2;
	
	assign M_AXI_WDATA = w_rd_data_reg2[W_DATA_WIDTH-1 -: C_S_AXI_DATA_WIDTH];
	assign M_AXI_WSTRB = w_rd_data_reg2[C_S_AXI_DATA_WIDTH-1 -: (C_S_AXI_DATA_WIDTH/8)];
	assign M_AXI_WLAST = w_rd_data_reg2[0];
	assign M_AXI_WVALID = w_valid_reg2;

	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			w_rd_ptr <= 0;
			w_rd_data_reg1 <= 0;
			w_rd_data_reg2 <= 0;
			w_valid_reg1 <= 1'b0;
			w_valid_reg2 <= 1'b0;
		end else begin
			// First stage: read from FIFO
			if (!w_empty) begin
				w_rd_data_reg1 <= w_fifo_mem[w_rd_ptr[W_FIFO_ADDR_WIDTH-1:0]];
				w_valid_reg1 <= 1'b1;
			end else begin
				w_valid_reg1 <= 1'b0;
			end
			
			// Second stage: hold for one extra clock
			w_rd_data_reg2 <= w_rd_data_reg1;
			w_valid_reg2 <= w_valid_reg1;
			
			// Advance read pointer when data is consumed
			if (M_AXI_WVALID && M_AXI_WREADY) begin
				w_rd_ptr <= w_rd_ptr + 1'b1;
			end
		end
	end

	// B Channel: M to S
	reg [B_DATA_WIDTH-1:0] b_fifo_mem [0:B_FIFO_DEPTH-1];
	reg [B_FIFO_ADDR_WIDTH:0] b_wr_ptr, b_rd_ptr;
	wire b_full = (b_wr_ptr[B_FIFO_ADDR_WIDTH:1] == b_rd_ptr[B_FIFO_ADDR_WIDTH:1]) && (b_wr_ptr[0] != b_rd_ptr[0]);
	wire b_empty = (b_wr_ptr == b_rd_ptr);

	// B write side (M clock)
	reg [B_DATA_WIDTH-1:0] b_wr_data;
	assign b_wr_data = {M_AXI_BID, M_AXI_BRESP};
	assign M_AXI_BREADY = !b_full;

	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			b_wr_ptr <= 0;
		end else if (M_AXI_BVALID && M_AXI_BREADY) begin
			b_fifo_mem[b_wr_ptr[B_FIFO_ADDR_WIDTH-1:0]] <= b_wr_data;
			b_wr_ptr <= b_wr_ptr + 1'b1;
		end
	end

	// B read side (S clock)
	reg [B_DATA_WIDTH-1:0] b_rd_data;
	assign S_AXI_BID = b_rd_data[B_DATA_WIDTH-1 -: C_S_AXI_ID_WIDTH];
	assign S_AXI_BRESP = b_rd_data[1:0];
	assign S_AXI_BVALID = !b_empty;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			b_rd_ptr <= 0;
			b_rd_data <= 0;
		end else if (S_AXI_BVALID && S_AXI_BREADY) begin
			b_rd_ptr <= b_rd_ptr + 1'b1;
		end
		if (!b_empty) begin
			b_rd_data <= b_fifo_mem[b_rd_ptr[B_FIFO_ADDR_WIDTH-1:0]];
		end
	end

	// AR Channel: S to M
	reg [AR_DATA_WIDTH-1:0] ar_fifo_mem [0:AR_FIFO_DEPTH-1];
	reg [AR_FIFO_ADDR_WIDTH:0] ar_wr_ptr, ar_rd_ptr;
	wire ar_full = (ar_wr_ptr[AR_FIFO_ADDR_WIDTH:1] == ar_rd_ptr[AR_FIFO_ADDR_WIDTH:1]) && (ar_wr_ptr[0] != ar_rd_ptr[0]);
	wire ar_empty = (ar_wr_ptr == ar_rd_ptr);

	// AR write side (S clock)
	reg [AR_DATA_WIDTH-1:0] ar_wr_data;
	assign ar_wr_data = {S_AXI_ARID, S_AXI_ARADDR, S_AXI_ARLEN, S_AXI_ARSIZE, S_AXI_ARBURST, S_AXI_ARLOCK, S_AXI_ARCACHE, S_AXI_ARPROT, S_AXI_ARQOS};
	assign S_AXI_ARREADY = !ar_full;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			ar_wr_ptr <= 0;
		end else if (S_AXI_ARVALID && S_AXI_ARREADY) begin
			ar_fifo_mem[ar_wr_ptr[AR_FIFO_ADDR_WIDTH-1:0]] <= ar_wr_data;
			ar_wr_ptr <= ar_wr_ptr + 1'b1;
		end
	end

	// AR read side (M clock)
	reg [AR_DATA_WIDTH-1:0] ar_rd_data;
	assign M_AXI_ARID = ar_rd_data[AR_DATA_WIDTH-1 -: C_S_AXI_ID_WIDTH];
	assign M_AXI_ARADDR = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-1 -: C_S_AXI_ADDR_WIDTH];
	assign M_AXI_ARLEN = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-1 -: 8];
	assign M_AXI_ARSIZE = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-1 -: 3];
	assign M_AXI_ARBURST = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-1 -: 2];
	assign M_AXI_ARLOCK = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1];
	assign M_AXI_ARCACHE = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-1 -: 4];
	assign M_AXI_ARPROT = ar_rd_data[AR_DATA_WIDTH-C_S_AXI_ID_WIDTH-C_S_AXI_ADDR_WIDTH-8-3-2-1-1-4-1 -: 3];

	reg [R_FIFO_ADDR_WIDTH:0] r_wr_ptr, r_rd_ptr;
