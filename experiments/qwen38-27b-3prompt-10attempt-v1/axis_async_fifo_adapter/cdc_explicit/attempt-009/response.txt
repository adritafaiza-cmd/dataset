module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Internal signals
    wire s_wr_en;
    wire m_rd_en;
    wire s_full;
    wire m_empty;
    wire s_overflow;
    wire m_overflow;
    wire s_bad_frame;
    wire m_bad_frame;
    wire s_good_frame;
    wire m_good_frame;
    wire [S_DATA_WIDTH-1:0] s_wr_data;
    wire [S_KEEP_WIDTH-1:0] s_wr_keep;
    wire s_wr_last;
    wire [ID_WIDTH-1:0] s_wr_id;
    wire [DEST_WIDTH-1:0] s_wr_dest;
    wire [USER_WIDTH-1:0] s_wr_user;
    wire [M_DATA_WIDTH-1:0] m_rd_data;
    wire [M_KEEP_WIDTH-1:0] m_rd_keep;
    wire m_rd_last;
    wire [ID_WIDTH-1:0] m_rd_id;
    wire [DEST_WIDTH-1:0] m_rd_dest;
    wire [USER_WIDTH-1:0] m_rd_user;
    wire [PTR_WIDTH-1:0] s_wr_ptr;
    wire [PTR_WIDTH-1:0] m_rd_ptr;
    wire [PTR_WIDTH-1:0] s_wr_ptr_m;
    wire [PTR_WIDTH-1:0] m_rd_ptr_s;
    wire [ADDR_WIDTH:0] s_depth;
    wire [ADDR_WIDTH:0] m_depth;

    // Pause signals
    wire s_pause_active;
    wire m_pause_active;

    // Reset synchronization
    reg [1:0] s_rst_sync;
    reg [1:0] m_rst_sync;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_rst_sync <= 2'b00;
        else
            s_rst_sync <= {s_rst_sync[0], 1'b0};
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_rst_sync <= 2'b00;
        else
            m_rst_sync <= {m_rst_sync[0], 1'b0};
    end

    wire s_rst_synced = s_rst_sync[1];
    wire m_rst_synced = m_rst_sync[1];

    // Pause logic
    assign s_pause_active = PAUSE_ENABLE ? s_pause_req : 1'b0;
    assign m_pause_active = PAUSE_ENABLE ? m_pause_req : 1'b0;
    assign s_pause_ack = PAUSE_ENABLE ? s_pause_req : 1'b0;
    assign m_pause_ack = PAUSE_ENABLE ? m_pause_req : 1'b0;

    // Write enable
    assign s_wr_en = s_axis_tvalid && s_axis_tready && !s_pause_active;

    // Read enable
    assign m_rd_en = m_axis_tvalid && m_axis_tready && !m_pause_active;

    // FIFO control
    assign s_full = (s_wr_ptr == m_rd_ptr_s);
    assign m_empty = (m_rd_ptr == s_wr_ptr_m);

    // Overflow detection
    assign s_overflow = s_axis_tvalid && !s_axis_tready && !s_pause_active;
    assign m_overflow = m_axis_tvalid && !m_axis_tready && !m_pause_active;

    // Frame detection
    assign s_bad_frame = 1'b0;
    assign m_bad_frame = 1'b0;
    assign s_good_frame = s_axis_tvalid && s_axis_tready && s_axis_tlast && !s_pause_active;
    assign m_good_frame = m_axis_tvalid && m_axis_tready && m_axis_tlast && !m_pause_active;

    // Data path
    assign s_wr_data = s_axis_tdata;
    assign s_wr_keep = S_KEEP_ENABLE ? s_axis_tkeep : {S_KEEP_WIDTH{1'b1}};
    assign s_wr_last = s_axis_tlast;
    assign s_wr_id = ID_ENABLE ? s_axis_tid : {ID_WIDTH{1'b0}};
    assign s_wr_dest = DEST_ENABLE ? s_axis_tdest : {DEST_WIDTH{1'b0}};
    assign s_wr_user = USER_ENABLE ? s_axis_tuser : {USER_WIDTH{1'b0}};

    // Width adapter: S_DATA_WIDTH to M_DATA_WIDTH
    // For simplicity, assume S_DATA_WIDTH == M_DATA_WIDTH or handle basic cases
    // In a real implementation, this would be more complex
    assign m_rd_data = M_DATA_WIDTH == S_DATA_WIDTH ? s_wr_data : 
                       (M_DATA_WIDTH > S_DATA_WIDTH ? {M_DATA_WIDTH-S_DATA_WIDTH{1'b0}, s_wr_data} : 
                       s_wr_data[M_DATA_WIDTH-1:0]);
    assign m_rd_keep = M_KEEP_ENABLE ? (M_KEEP_WIDTH == S_KEEP_WIDTH ? s_wr_keep : 
                       (M_KEEP_WIDTH > S_KEEP_WIDTH ? {M_KEEP_WIDTH-S_KEEP_WIDTH{1'b1}, s_wr_keep} : 
                       s_wr_keep[M_KEEP_WIDTH-1:0])) : {M_KEEP_WIDTH{1'b1}};
    assign m_rd_last = s_wr_last;
    assign m_rd_id = ID_ENABLE ? s_wr_id : {ID_WIDTH{1'b0}};
    assign m_rd_dest = DEST_ENABLE ? s_wr_dest : {DEST_WIDTH{1'b0}};
    assign m_rd_user = USER_ENABLE ? s_wr_user : {USER_WIDTH{1'b0}};

    // FIFO implementation
    reg [S_DATA_WIDTH-1:0] mem_data [0:DEPTH-1];
    reg [S_KEEP_WIDTH-1:0] mem_keep [0:DEPTH-1];
    reg mem_last [0:DEPTH-1];
    reg [ID_WIDTH-1:0] mem_id [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] mem_dest [0:DEPTH-1];
    reg [USER_WIDTH-1:0] mem_user [0:DEPTH-1];

    // Write pointer
    reg [PTR_WIDTH-1:0] s_wr_ptr_reg;
    // Read pointer
    reg [PTR_WIDTH-1:0] m_rd_ptr_reg;

    // Pointer synchronization
    reg [PTR_WIDTH-1:0] s_wr_ptr_m_reg;
    reg [PTR_WIDTH-1:0] m_rd_ptr_s_reg;

    always @(posedge s_clk or posedge s_rst_synced) begin
        if (s_rst_synced)
            s_wr_ptr_reg <= {PTR_WIDTH{1'b0}};
        else if (s_wr_en)
            s_wr_ptr_reg <= s_wr_ptr_reg + 1'b1;
    end

    always @(posedge m_clk or posedge m_rst_synced) begin
        if (m_rst_synced)
            m_rd_ptr_reg <= {PTR_WIDTH{1'b0}};
        else if (m_rd_en)
            m_rd_ptr_reg <= m_rd_ptr_reg + 1'b1;
    end

    // Synchronize write pointer to m_clk domain
    always @(posedge m_clk or posedge m_rst_synced) begin
        if (m_rst_synced)
            s_wr_ptr_m_reg <= {PTR_WIDTH{1'b0}};
        else
            s_wr_ptr_m_reg <= s_wr_ptr_reg;
    end

    // Synchronize read pointer to s_clk domain
    always @(posedge s_clk or posedge s_rst_synced) begin
        if (s_rst_synced)
            m_rd_ptr_s_reg <= {PTR_WIDTH{1'b0}};
        else
            m_rd_ptr_s_reg <= m_rd_ptr_reg;
    end

    assign s_wr_ptr = s_wr_ptr_reg;
    assign m_rd_ptr = m_rd_ptr_reg;
    assign s_wr_ptr_m = s_wr_ptr_m_reg;
    assign m_rd_ptr_s = m_rd_ptr_s_reg;

    // Memory write
    always @(posedge s_clk or posedge s_rst_synced) begin
        if (s_rst_synced) begin
            // No action needed for memory
        end else if (s_wr_en) begin
            mem_data[s_wr_ptr_reg[ADDR_WIDTH-1:0]] <= s_wr_data;
            mem_keep[s_wr_ptr_reg[ADDR_WIDTH-1:0]] <= s_wr_keep;
            mem_last[s_wr_ptr_reg[ADDR_WIDTH-1:0]] <= s_wr_last;
            mem_id[s_wr_ptr_reg[ADDR_WIDTH-1:0]] <= s_wr_id;
            mem_dest[s_wr_ptr_reg[ADDR_WIDTH-1:0]] <= s_wr_dest;
            mem_user[s_wr_ptr_reg[ADDR_WIDTH-1:0]] <= s_wr_user;
        end
    end

    // Memory read
    wire [ADDR_WIDTH-1:0] m_rd_addr = m_rd_ptr_reg[ADDR_WIDTH-1:0];
    wire [S_DATA_WIDTH-1:0] mem_rd_data = mem_data[m_rd_addr];
    wire [S_KEEP_WIDTH-1:0] mem_rd_keep = mem_keep[m_rd_addr];
    wire mem_rd_last = mem_last[m_rd_addr];
    wire [ID_WIDTH-1:0] mem_rd_id = mem_id[m_rd_addr];
    wire [DEST_WIDTH-1:0] mem_rd_dest = mem_dest[m_rd_addr];
    wire [USER_WIDTH-1:0] mem_rd_user = mem_user[m_rd_addr];

    // Output FIFO (if enabled)
    reg [M_DATA_WIDTH-1:0] out_fifo_data [0:DEPTH-1];
    reg [M_KEEP_WIDTH-1:0] out_fifo_keep [0:DEPTH-1];
    reg out_fifo_last [0:DEPTH-1];
    reg [ID_WIDTH-1:0] out_fifo_id [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] out_fifo_dest [0:DEPTH-1];
    reg [USER_WIDTH-1:0] out_fifo_user [0:DEPTH-1];
    reg [PTR_WIDTH-1:0] out_fifo_wr_ptr;
    reg [PTR_WIDTH-1:0] out_fifo_rd_ptr;

    wire out_fifo_wr_en = m_rd_en;
    wire out_fifo_rd_en = m_axis_tvalid && m_axis_tready && !m_pause_active;
    wire out_fifo_full = (out_fifo_wr_ptr == out_fifo_rd_ptr);
    wire out_fifo_empty = (out_fifo_wr_ptr == out_fifo_rd_ptr);

    always @(posedge m_clk or posedge m_rst_synced) begin
        if (m_rst_synced) begin
            out_fifo_wr_ptr <= {PTR_WIDTH{1'b0}};
            out_fifo_rd_ptr <= {PTR_WIDTH{1'b0}};
        end else begin
            if (out_fifo_wr_en)
                out_fifo_wr_ptr <= out_fifo_wr_ptr + 1'b1;
            if (out_fifo_rd_en)
                out_fifo_rd_ptr <= out_fifo_rd_ptr + 1'b1;
        end
    end

    always @(posedge m_clk or posedge m_rst_synced) begin
        if (m_rst_synced) begin
            // No action
        end else if (out_fifo_wr_en) begin
            out_fifo_data[out_fifo_wr_ptr[ADDR_WIDTH-1:0]] <= mem_rd_data;
            out_fifo_keep[out_fifo_wr_ptr[ADDR_WIDTH-1:0]] <= mem_rd_keep;
            out_fifo_last[out_fifo_wr_ptr[ADDR_WIDTH-1:0]] <= mem_rd_last;
            out_fifo_id[out_fifo_wr_ptr[ADDR_WIDTH-1:0]] <= mem_rd_id;
            out_fifo_dest[out_fifo_wr_ptr[ADDR_WIDTH-1:0]] <= mem_rd_dest;
            out_fifo_user[out_fifo_wr_ptr[ADDR_WIDTH-1:0]] <= mem_rd_user;
        end
    end

    wire [ADDR_WIDTH-1:0] out_fifo_rd_addr = out_fifo_rd_ptr[ADDR_WIDTH-1:0];
    wire [M_DATA_WIDTH-1:0] out_fifo_rd_data = out_fifo_data[out_fifo_rd_addr];
    wire [M_KEEP_WIDTH-1:0] out_fifo_rd_keep = out_fifo_keep[out_fifo_rd_addr];
    wire out_fifo_rd_last = out_fifo_last[out_fifo_rd_addr];
    wire [ID_WIDTH-1:0] out_fifo_rd_id = out_fifo_id[out_fifo_rd_addr];
    wire [DEST_WIDTH-1:0] out_fifo_rd_dest = out_fifo_dest[out_fifo_rd_addr];
    wire [USER_WIDTH-1:0] out_fifo_rd_user = out_fifo_user[out_fifo_rd_addr];

    // Output assignment
    assign m_axis_tdata = OUTPUT_FIFO_ENABLE ? out_fifo_rd_data : m_rd_data;
    assign m_axis_tkeep = OUTPUT_FIFO_ENABLE ? out_fifo_rd_keep : m_rd_keep;
    assign m_axis_tvalid = OUTPUT_FIFO_ENABLE ? !out_fifo_empty : !m_empty;
    assign m_axis_tlast = OUTPUT_FIFO_ENABLE ? out_fifo_rd_last : m_rd_last;
    assign m_axis_tid = OUTPUT_FIFO_ENABLE ? out_fifo_rd_id : m_rd_id;
    assign m_axis_tdest = OUTPUT_FIFO_ENABLE ? out_fifo_rd_dest : m_rd_dest;
    assign m_axis_tuser = OUTPUT_FIFO_ENABLE ? out_fifo_rd_user : m_rd_user;

    // Input ready
    assign s_axis_tready = !s_full && !s_pause_active;

    // Depth calculation
    wire [PTR_WIDTH-1:0] s_depth_raw = s_wr_ptr_reg - m_rd_ptr_s_reg;
    wire [PTR_WIDTH-1:0] m_depth_raw = s_wr_ptr_m_reg - m_rd_ptr_reg;

    assign s_status_depth = s_depth_raw[ADDR_WIDTH:0];
    assign s_status_depth_commit = s_depth_raw[ADDR_WIDTH:0];
    assign m_status_depth = m_depth_raw[ADDR_WIDTH:0];
    assign m_status_depth_commit = m_depth_raw[ADDR_WIDTH:0];

    // Status
    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;

endmodule
