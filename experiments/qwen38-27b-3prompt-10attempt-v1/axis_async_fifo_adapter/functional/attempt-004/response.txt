module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Internal signals
    wire s_wr_en;
    wire m_rd_en;
    wire s_full;
    wire m_empty;
    wire [S_DATA_WIDTH-1:0] s_wr_data;
    wire [S_KEEP_WIDTH-1:0] s_wr_keep;
    wire s_wr_last;
    wire [ID_WIDTH-1:0] s_wr_id;
    wire [DEST_WIDTH-1:0] s_wr_dest;
    wire [USER_WIDTH-1:0] s_wr_user;

    wire [M_DATA_WIDTH-1:0] m_rd_data;
    wire [M_KEEP_WIDTH-1:0] m_rd_keep;
    wire m_rd_last;
    wire [ID_WIDTH-1:0] m_rd_id;
    wire [DEST_WIDTH-1:0] m_rd_dest;
    wire [USER_WIDTH-1:0] m_rd_user;

    // Pause logic
    reg s_pause_active;
    reg m_pause_active;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_pause_active <= 1'b0;
        else if (PAUSE_ENABLE)
            s_pause_active <= s_pause_req;
        else
            s_pause_active <= 1'b0;
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_pause_active <= 1'b0;
        else if (PAUSE_ENABLE)
            m_pause_active <= m_pause_req;
        else
            m_pause_active <= 1'b0;
    end

    assign s_pause_ack = s_pause_active;
    assign m_pause_ack = m_pause_active;

    // Write enable logic
    wire s_wr_allowed = s_axis_tvalid && s_axis_tready && !s_pause_active;
    assign s_wr_en = s_wr_allowed;

    // Read enable logic
    wire m_rd_allowed = m_axis_tvalid && m_axis_tready && !m_pause_active;
    assign m_rd_en = m_rd_allowed;

    // Data path
    assign s_wr_data = s_axis_tdata;
    assign s_wr_keep = s_axis_tkeep;
    assign s_wr_last = s_axis_tlast;
    assign s_wr_id = s_axis_tid;
    assign s_wr_dest = s_axis_tdest;
    assign s_wr_user = s_axis_tuser;

    // Instantiate the core async FIFO with width adapter
    axis_async_fifo_core #(
        .DEPTH(DEPTH),
        .S_DATA_WIDTH(S_DATA_WIDTH),
        .S_KEEP_ENABLE(S_KEEP_ENABLE),
        .S_KEEP_WIDTH(S_KEEP_WIDTH),
        .M_DATA_WIDTH(M_DATA_WIDTH),
        .M_KEEP_ENABLE(M_KEEP_ENABLE),
        .M_KEEP_WIDTH(M_KEEP_WIDTH),
        .ID_ENABLE(ID_ENABLE),
        .ID_WIDTH(ID_WIDTH),
        .DEST_ENABLE(DEST_ENABLE),
        .DEST_WIDTH(DEST_WIDTH),
        .USER_ENABLE(USER_ENABLE),
        .USER_WIDTH(USER_WIDTH),
        .RAM_PIPELINE(RAM_PIPELINE),
        .OUTPUT_FIFO_ENABLE(OUTPUT_FIFO_ENABLE),
        .FRAME_FIFO(FRAME_FIFO),
        .USER_BAD_FRAME_VALUE(USER_BAD_FRAME_VALUE),
        .USER_BAD_FRAME_MASK(USER_BAD_FRAME_MASK),
        .DROP_OVERSIZE_FRAME(DROP_OVERSIZE_FRAME),
        .DROP_BAD_FRAME(DROP_BAD_FRAME),
        .DROP_WHEN_FULL(DROP_WHEN_FULL),
        .MARK_WHEN_FULL(MARK_WHEN_FULL),
        .PAUSE_ENABLE(PAUSE_ENABLE),
        .FRAME_PAUSE(FRAME_PAUSE)
    ) u_fifo (
        .s_clk(s_clk),
        .s_rst(s_rst),
        .s_wr_en(s_wr_en),
        .s_wr_data(s_wr_data),
        .s_wr_keep(s_wr_keep),
        .s_wr_last(s_wr_last),
        .s_wr_id(s_wr_id),
        .s_wr_dest(s_wr_dest),
        .s_wr_user(s_wr_user),
        .s_full(s_full),
        .s_status_depth(s_status_depth),
        .s_status_depth_commit(s_status_depth_commit),
        .s_status_overflow(s_status_overflow),
        .s_status_bad_frame(s_status_bad_frame),
        .s_status_good_frame(s_status_good_frame),

        .m_clk(m_clk),
        .m_rst(m_rst),
        .m_rd_en(m_rd_en),
        .m_rd_data(m_rd_data),
        .m_rd_keep(m_rd_keep),
        .m_rd_last(m_rd_last),
        .m_rd_id(m_rd_id),
        .m_rd_dest(m_rd_dest),
        .m_rd_user(m_rd_user),
        .m_empty(m_empty),
        .m_status_depth(m_status_depth),
        .m_status_depth_commit(m_status_depth_commit),
        .m_status_overflow(m_status_overflow),
        .m_status_bad_frame(m_status_bad_frame),
        .m_status_good_frame(m_status_good_frame)
    );

    // Output assignment
    assign m_axis_tdata = m_rd_data;
    assign m_axis_tkeep = m_rd_keep;
    assign m_axis_tlast = m_rd_last;
    assign m_axis_tid = m_rd_id;
    assign m_axis_tdest = m_rd_dest;
    assign m_axis_tuser = m_rd_user;

    // Valid and Ready logic
    assign m_axis_tvalid = !m_empty;
    assign s_axis_tready = !s_full;

endmodule

module axis_async_fifo_core #(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
) (
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire                     s_wr_en,
    input  wire [S_DATA_WIDTH-1:0]  s_wr_data,
    input  wire [S_KEEP_WIDTH-1:0]  s_wr_keep,
    input  wire                     s_wr_last,
    input  wire [ID_WIDTH-1:0]      s_wr_id,
    input  wire [DEST_WIDTH-1:0]    s_wr_dest,
    input  wire [USER_WIDTH-1:0]    s_wr_user,
    output wire                     s_full,
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,

    input  wire                     m_clk,
    input  wire                     m_rst,
    input  wire                     m_rd_en,
    output wire [M_DATA_WIDTH-1:0]  m_rd_data,
    output wire [M_KEEP_WIDTH-1:0]  m_rd_keep,
    output wire                     m_rd_last,
    output wire [ID_WIDTH-1:0]      m_rd_id,
    output wire [DEST_WIDTH-1:0]    m_rd_dest,
    output wire [USER_WIDTH-1:0]    m_rd_user,
    output wire                     m_empty,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Write pointer (s_clk domain)
    reg [PTR_WIDTH-1:0] s_wr_ptr;
    reg [ADDR_WIDTH-1:0] s_wr_addr;

    // Read pointer (m_clk domain)
    reg [PTR_WIDTH-1:0] m_rd_ptr;
    reg [ADDR_WIDTH-1:0] m_rd_addr;

    // Synchronized pointers
    reg [PTR_WIDTH-1:0] s_wr_ptr_m_sync;
    reg [PTR_WIDTH-1:0] m_rd_ptr_s_sync;

    // RAM storage
    reg [S_DATA_WIDTH-1:0] ram_data [0:DEPTH-1];
    reg [S_KEEP_WIDTH-1:0] ram_keep [0:DEPTH-1];
    reg ram_last [0:DEPTH-1];
    reg [ID_WIDTH-1:0] ram_id [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] ram_dest [0:DEPTH-1];
    reg [USER_WIDTH-1:0] ram_user [0:DEPTH-1];

    // Status flags
    reg s_overflow;
    reg s_bad_frame;
    reg s_good_frame;
    reg m_overflow;
    reg m_bad_frame;
    reg m_good_frame;

    // Depth calculation
    wire [PTR_WIDTH-1:0] s_depth_raw = s_wr_ptr - m_rd_ptr_s_sync;
    wire [PTR_WIDTH-1:0] m_depth_raw = m_rd_ptr - s_wr_ptr_m_sync;

    assign s_status_depth = s_depth_raw;
    assign s_status_depth_commit = s_depth_raw;
    assign m_status_depth = m_depth_raw;
    assign m_status_depth_commit = m_depth_raw;

    assign s_full = (s_depth_raw == DEPTH);
    assign m_empty = (m_depth_raw == 0);

    // Write logic (s_clk domain)
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= {PTR_WIDTH{1'b0}};
            s_wr_addr <= {ADDR_WIDTH{1'b0}};
            s_overflow <= 1'b0;
            s_bad_frame <= 1'b0;
            s_good_frame <= 1'b0;
        end else begin
            if (s_wr_en) begin
                ram_data[s_wr_addr] <= s_wr_data;
                ram_keep[s_wr_addr] <= s_wr_keep;
                ram_last[s_wr_addr] <= s_wr_last;
                ram_id[s_wr_addr] <= s_wr_id;
                ram_dest[s_wr_addr] <= s_wr_dest;
                ram_user[s_wr_addr] <= s_wr_user;

                s_wr_addr <= s_wr_addr + 1'b1;
                s_wr_ptr <= s_wr_ptr + 1'b1;

                if (s_wr_last) begin
                    s_good_frame <= 1'b1;
                end
            end
        end
    end

    // Read logic (m_clk domain)
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr <= {PTR_WIDTH{1'b0}};
            m_rd_addr <= {ADDR_WIDTH{1'b0}};
            m_overflow <= 1'b0;
            m_bad_frame <= 1'b0;
            m_good_frame <= 1'b0;
        end else begin
            if (m_rd_en) begin
                m_rd_addr <= m_rd_addr + 1'b1;
                m_rd_ptr <= m_rd_ptr + 1'b1;

                if (ram_last[m_rd_addr]) begin
                    m_good_frame <= 1'b1;
                end
            end
        end
    end

    // Output data (combinational read from RAM)
    assign m_rd_data = ram_data[m_rd_addr];
    assign m_rd_keep = ram_keep[m_rd_addr];
    assign m_rd_last = ram_last[m_rd_addr];
    assign m_rd_id = ram_id[m_rd_addr];
    assign m_rd_dest = ram_dest[m_rd_addr];
    assign m_rd_user = ram_user[m_rd_addr];

    // Pointer synchronization
    // s_wr_ptr to m_clk domain
    reg [PTR_WIDTH-1:0] s_wr_ptr_m_sync_r1;
    reg [PTR_WIDTH-1:0] s_wr_ptr_m_sync_r2;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_wr_ptr_m_sync_r1 <= {PTR_WIDTH{1'b0}};
            s_wr_ptr_m_sync_r2 <= {PTR_WIDTH{1'b0}};
        end else begin
            s_wr_ptr_m_sync_r1 <= s_wr_ptr;
            s_wr_ptr_m_sync_r2 <= s_wr_ptr_m_sync_r1;
        end
    end
    assign s_wr_ptr_m_sync = s_wr_ptr_m_sync_r2;

    // m_rd_ptr to s_clk domain
    reg [PTR_WIDTH-1:0] m_rd_ptr_s_sync_r1;
    reg [PTR_WIDTH-1:0] m_rd_ptr_s_sync_r2;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_rd_ptr_s_sync_r1 <= {PTR_WIDTH{1'b0}};
            m_rd_ptr_s_sync_r2 <= {PTR_WIDTH{1'b0}};
        end else begin
            m_rd_ptr_s_sync_r1 <= m_rd_ptr;
            m_rd_ptr_s_sync_r2 <= m_rd_ptr_s_sync_r1;
        end
    end
    assign m_rd_ptr_s_sync = m_rd_ptr_s_sync_r2;

    // Status outputs
    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;

endmodule
