```verilog
module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam S_BYTES = S_DATA_WIDTH / 8;
    localparam M_BYTES = M_DATA_WIDTH / 8;
    localparam S_KEEP_W = S_KEEP_ENABLE ? S_KEEP_WIDTH : 1;
    localparam M_KEEP_W = M_KEEP_ENABLE ? M_KEEP_WIDTH : 1;
    localparam ID_W = ID_ENABLE ? ID_WIDTH : 1;
    localparam DEST_W = DEST_ENABLE ? DEST_WIDTH : 1;
    localparam USER_W = USER_ENABLE ? USER_WIDTH : 1;

    // Internal signals
    wire [S_DATA_WIDTH-1:0] s_tdata;
    wire [S_KEEP_W-1:0] s_tkeep;
    wire s_tvalid;
    wire s_tready;
    wire s_tlast;
    wire [ID_W-1:0] s_tid;
    wire [DEST_W-1:0] s_tdest;
    wire [USER_W-1:0] s_tuser;

    assign s_tdata = s_axis_tdata;
    assign s_tkeep = S_KEEP_ENABLE ? s_axis_tkeep : {S_KEEP_W{1'b1}};
    assign s_tvalid = s_axis_tvalid;
    assign s_tlast = s_axis_tlast;
    assign s_tid = ID_ENABLE ? s_axis_tid : {ID_W{1'b0}};
    assign s_tdest = DEST_ENABLE ? s_axis_tdest : {DEST_W{1'b0}};
    assign s_tuser = USER_ENABLE ? s_axis_tuser : {USER_W{1'b0}};

    // Write side (s_clk domain)
    reg [PTR_WIDTH-1:0] w_ptr;
    reg [ADDR_WIDTH-1:0] w_addr;
    reg [S_DATA_WIDTH-1:0] w_data;
    reg [S_KEEP_W-1:0] w_keep;
    reg w_last;
    reg [ID_W-1:0] w_id;
    reg [DEST_W-1:0] w_dest;
    reg [USER_W-1:0] w_user;
    reg w_valid;
    reg w_overflow;
    reg w_bad_frame;
    reg w_good_frame;

    // Read side (m_clk domain)
    reg [PTR_WIDTH-1:0] r_ptr;
    reg [ADDR_WIDTH-1:0] r_addr;
    reg [S_DATA_WIDTH-1:0] r_data;
    reg [S_KEEP_W-1:0] r_keep;
    reg r_last;
    reg [ID_W-1:0] r_id;
    reg [DEST_W-1:0] r_dest;
    reg [USER_W-1:0] r_user;
    reg r_valid;
    reg r_overflow;
    reg r_bad_frame;
    reg r_good_frame;

    // Synchronizers
    reg [PTR_WIDTH-1:0] w_ptr_sync1, w_ptr_sync2, w_ptr_sync3;
    reg [PTR_WIDTH-1:0] r_ptr_sync1, r_ptr_sync2, r_ptr_sync3;

    // Write pointer synchronization to m_clk
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            w_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            w_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            w_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            w_ptr_sync1 <= w_ptr;
            w_ptr_sync2 <= w_ptr_sync1;
            w_ptr_sync3 <= w_ptr_sync2;
        end
    end

    // Read pointer synchronization to s_clk
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            r_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            r_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            r_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            r_ptr_sync1 <= r_ptr;
            r_ptr_sync2 <= r_ptr_sync1;
            r_ptr_sync3 <= r_ptr_sync2;
        end
    end

    // Depth calculation
    wire [PTR_WIDTH-1:0] w_ptr_m = w_ptr_sync3;
    wire [PTR_WIDTH-1:0] r_ptr_s = r_ptr_sync3;
    wire [PTR_WIDTH-1:0] depth_s = w_ptr - r_ptr_s;
    wire [PTR_WIDTH-1:0] depth_m = w_ptr_m - r_ptr;

    // Full/Empty detection
    wire s_full = (depth_s == DEPTH);
    wire m_empty = (depth_m == 0);

    // Pause logic
    wire s_pause = PAUSE_ENABLE ? s_pause_req : 1'b0;
    wire m_pause = PAUSE_ENABLE ? m_pause_req : 1'b0;

    // Write ready
    wire s_write_en = s_tvalid && s_tready && !s_pause;
    assign s_tready = !s_full && !s_pause;
    assign s_axis_tready = s_tready;

    // Read ready
    wire m_read_en = m_axis_tready && m_axis_tvalid && !m_pause;
    assign m_axis_tvalid = r_valid;

    // Write logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            w_ptr <= {PTR_WIDTH{1'b0}};
            w_addr <= {ADDR_WIDTH{1'b0}};
            w_data <= {S_DATA_WIDTH{1'b0}};
            w_keep <= {S_KEEP_W{1'b0}};
            w_last <= 1'b0;
            w_id <= {ID_W{1'b0}};
            w_dest <= {DEST_W{1'b0}};
            w_user <= {USER_W{1'b0}};
            w_valid <= 1'b0;
            w_overflow <= 1'b0;
            w_bad_frame <= 1'b0;
            w_good_frame <= 1'b0;
        end else begin
            if (s_write_en) begin
                w_data <= s_tdata;
                w_keep <= s_tkeep;
                w_last <= s_tlast;
                w_id <= s_tid;
                w_dest <= s_tdest;
                w_user <= s_tuser;
                w_valid <= 1'b1;
                w_addr <= w_addr + 1'b1;
                w_ptr <= w_ptr + 1'b1;
                
                if (s_tlast) begin
                    w_good_frame <= 1'b1;
                    w_bad_frame <= 1'b0;
                end
                
                if (s_full && !DROP_WHEN_FULL) begin
                    w_overflow <= 1'b1;
                end
            end
        end
    end

    // Read logic
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            r_ptr <= {PTR_WIDTH{1'b0}};
            r_addr <= {ADDR_WIDTH{1'b0}};
            r_data <= {S_DATA_WIDTH{1'b0}};
            r_keep <= {S_KEEP_W{1'b0}};
            r_last <= 1'b0;
            r_id <= {ID_W{1'b0}};
            r_dest <= {DEST_W{1'b0}};
            r_user <= {USER_W{1'b0}};
            r_valid <= 1'b0;
            r_overflow <= 1'b0;
            r_bad_frame <= 1'b0;
            r_good_frame <= 1'b0;
        end else begin
            if (m_read_en) begin
                r_addr <= r_addr + 1'b1;
                r_ptr <= r_ptr + 1'b1;
                r_valid <= 1'b0;
                
                if (r_last) begin
                    r_good_frame <= 1'b1;
                    r_bad_frame <= 1'b0;
                end
            end else if (!m_empty && !r_valid) begin
                r_valid <= 1'b1;
            end
        end
    end

    // RAM storage
    reg [S_DATA_WIDTH-1:0] ram_data [0:DEPTH-1];
    reg [S_KEEP_W-1:0] ram_keep [0:DEPTH-1];
    reg ram_last [0:DEPTH-1];
    reg [ID_W-1:0] ram_id [0:DEPTH-1];
    reg [DEST_W-1:0] ram_dest [0:DEPTH-1];
    reg [USER_W-1:0] ram_user [0:DEPTH-1];

    // Write to RAM
    always @(posedge s_clk) begin
        if (s_write_en) begin
            ram_data[w_addr] <= s_tdata;
            ram_keep[w_addr] <= s_tkeep;
            ram_last[w_addr] <= s_tlast;
            ram_id[w_addr] <= s_tid;
            ram_dest[w_addr] <= s_tdest;
            ram_user[w_addr] <= s_tuser;
        end
    end

    // Read from RAM
    wire [ADDR_WIDTH-1:0] r_addr_ram = r_addr;
    wire [S_DATA_WIDTH-1:0] ram_rdata = ram_data[r_addr_ram];
    wire [S_KEEP_W-1:0] ram_rkeep = ram_keep[r_addr_ram];
    wire ram_rlast = ram_last[r_addr_ram];
    wire [ID_W-1:0] ram_rid = ram_id[r_addr_ram];
    wire [DEST_W-1:0] ram_rdest = ram_dest[r_addr_ram];
    wire [USER_W-1:0] ram_ruser = ram_user[r_addr_ram];

    // Update read registers
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            r_data <= {S_DATA_WIDTH{1'b0}};
            r_keep <= {S_KEEP_W{1'b0}};
            r_last <= 1'b0;
            r_id <= {ID_W{1'b0}};
            r_dest <= {DEST_W{1'b0}};
            r_user <= {USER_W{1'b0}};
        end else if (!m_empty && !r_valid) begin
            r_data <= ram_rdata;
            r_keep <= ram_rkeep;
            r_last <= ram_rlast;
            r_id <= ram_rid;
            r_dest <= ram_rdest;
            r_user <= ram_ruser;
        end
    end

    // Width adapter: S_DATA_WIDTH to M_DATA_WIDTH
    // For simplicity, assume S_DATA_WIDTH == M_DATA_WIDTH or direct mapping
    // In a real implementation, this would handle byte alignment and keep signals
    wire [M_DATA_WIDTH-1:0] m_tdata;
    wire [M_KEEP_W-1:0] m_tkeep;
    wire m_tlast;
    wire [ID_W-1:0] m_tid;
    wire [DEST_W-1:0] m_tdest;
    wire [USER_W-1:0] m_tuser;

    // Direct mapping for same width
    assign m_tdata = (M_DATA_WIDTH == S_DATA_WIDTH) ? r_data[M_DATA_WIDTH-1:0] : {M_DATA_WIDTH{1'b0}};
    assign m_tkeep = M_KEEP_ENABLE ? (M_KEEP_W == S_KEEP_W ? r_keep : {M_KEEP_W{1'b1}}) : {M_KEEP_W{1'b1}};
    assign m_tlast = r_last;
    assign m_tid = r_id;
    assign m_tdest = r_dest;
    assign m_tuser = r_user;

    // Output assignments
    assign m_axis_tdata = m_tdata;
    assign m_axis_tkeep = m_tkeep;
    assign m_axis_tlast = m_tlast;
    assign m_axis_tid = ID_ENABLE ? m_tid : {ID_WIDTH{1'b0}};
    assign m_axis_tdest = DEST_ENABLE ? m_tdest : {DEST_WIDTH{1'b0}};
    assign m_axis_tuser = USER_ENABLE ? m_tuser : {USER_WIDTH{1'b0}};

    // Pause acknowledgments
    assign s_pause_ack = PAUSE_ENABLE ? s_pause_req : 1'b0;
    assign m_pause_ack = PAUSE_ENABLE ? m_pause_req : 1'b0;

    // Status outputs
    assign s_status_depth = depth_s;
    assign s_status_depth_commit = depth_s;
    assign s_status_overflow = w_overflow;
    assign s_status_bad_frame = w_bad_frame;
    assign s_status_good_frame = w_good_frame;

    assign m_status_depth = depth_m;
    assign m_status_depth_commit = depth_m;
    assign m_status_overflow = r_overflow;
    assign m_status_bad_frame = r_bad_frame;
    assign m_status_good_frame = r_good_frame;

endmodule
