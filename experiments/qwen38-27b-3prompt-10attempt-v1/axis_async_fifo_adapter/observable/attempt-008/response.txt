```verilog
module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam S_BYTES = S_DATA_WIDTH / 8;
    localparam M_BYTES = M_DATA_WIDTH / 8;
    localparam MAX_BYTES = (S_BYTES > M_BYTES) ? S_BYTES : M_BYTES;
    localparam ACC_WIDTH = $clog2(MAX_BYTES) + 1;

    // Internal signals
    wire [S_DATA_WIDTH-1:0] s_tdata;
    wire [S_KEEP_WIDTH-1:0] s_tkeep;
    wire s_tvalid;
    wire s_tready;
    wire s_tlast;
    wire [ID_WIDTH-1:0] s_tid;
    wire [DEST_WIDTH-1:0] s_tdest;
    wire [USER_WIDTH-1:0] s_tuser;

    wire [M_DATA_WIDTH-1:0] m_tdata;
    wire [M_KEEP_WIDTH-1:0] m_tkeep;
    wire m_tvalid;
    wire m_tready;
    wire m_tlast;
    wire [ID_WIDTH-1:0] m_tid;
    wire [DEST_WIDTH-1:0] m_tdest;
    wire [USER_WIDTH-1:0] m_tuser;

    assign s_tdata = s_axis_tdata;
    assign s_tkeep = S_KEEP_ENABLE ? s_axis_tkeep : {S_KEEP_WIDTH{1'b1}};
    assign s_tvalid = s_axis_tvalid;
    assign s_tlast = s_axis_tlast;
    assign s_tid = ID_ENABLE ? s_axis_tid : {ID_WIDTH{1'b0}};
    assign s_tdest = DEST_ENABLE ? s_axis_tdest : {DEST_WIDTH{1'b0}};
    assign s_tuser = USER_ENABLE ? s_axis_tuser : {USER_WIDTH{1'b0}};

    assign m_tdata = m_axis_tdata;
    assign m_tkeep = M_KEEP_ENABLE ? m_axis_tkeep : {M_KEEP_WIDTH{1'b1}};
    assign m_tvalid = m_axis_tvalid;
    assign m_tready = m_axis_tready;
    assign m_tlast = m_axis_tlast;
    assign m_tid = m_axis_tid;
    assign m_tdest = m_axis_tdest;
    assign m_tuser = m_axis_tuser;

    // FIFO control signals
    wire s_wr_en;
    wire m_rd_en;
    wire s_full;
    wire m_empty;
    wire [ADDR_WIDTH-1:0] s_wr_addr;
    wire [ADDR_WIDTH-1:0] m_rd_addr;

    // Width adapter state
    reg [ACC_WIDTH-1:0] s_acc_bytes;
    reg [ACC_WIDTH-1:0] m_acc_bytes;
    reg [S_DATA_WIDTH-1:0] s_acc_data;
    reg [M_DATA_WIDTH-1:0] m_acc_data;
    reg s_acc_valid;
    reg m_acc_valid;
    reg s_tlast_reg;
    reg m_tlast_reg;
    reg [ID_WIDTH-1:0] s_tid_reg;
    reg [DEST_WIDTH-1:0] s_tdest_reg;
    reg [USER_WIDTH-1:0] s_tuser_reg;
    reg m_tid_reg;
    reg m_tdest_reg;
    reg m_tuser_reg;

    // Pause logic
    reg s_pause;
    reg m_pause;
    wire s_pause_active;
    wire m_pause_active;

    assign s_pause_active = PAUSE_ENABLE ? (s_pause_req | s_pause) : 1'b0;
    assign m_pause_active = PAUSE_ENABLE ? (m_pause_req | m_pause) : 1'b0;

    // Status signals
    reg s_overflow;
    reg m_overflow;
    reg s_bad_frame;
    reg m_bad_frame;
    reg s_good_frame;
    reg m_good_frame;

    // FIFO depth tracking
    reg [PTR_WIDTH-1:0] s_wr_ptr;
    reg [PTR_WIDTH-1:0] m_rd_ptr;
    wire [PTR_WIDTH-1:0] s_wr_ptr_sync;
    wire [PTR_WIDTH-1:0] m_rd_ptr_sync;

    // Clock domain crossing for write pointer
    reg [PTR_WIDTH-1:0] s_wr_ptr_reg;
    reg [PTR_WIDTH-1:0] s_wr_ptr_sync1;
    reg [PTR_WIDTH-1:0] s_wr_ptr_sync2;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_wr_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            s_wr_ptr_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            s_wr_ptr_sync1 <= s_wr_ptr_reg;
            s_wr_ptr_sync2 <= s_wr_ptr_sync1;
        end
    end

    assign s_wr_ptr_sync = s_wr_ptr_sync2;

    // Clock domain crossing for read pointer
    reg [PTR_WIDTH-1:0] m_rd_ptr_reg;
    reg [PTR_WIDTH-1:0] m_rd_ptr_sync1;
    reg [PTR_WIDTH-1:0] m_rd_ptr_sync2;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_rd_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            m_rd_ptr_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            m_rd_ptr_sync1 <= m_rd_ptr_reg;
            m_rd_ptr_sync2 <= m_rd_ptr_sync1;
        end
    end

    assign m_rd_ptr_sync = m_rd_ptr_sync2;

    // Depth calculation
    wire [PTR_WIDTH-1:0] s_depth = s_wr_ptr - m_rd_ptr_sync;
    wire [PTR_WIDTH-1:0] m_depth = s_wr_ptr_sync - m_rd_ptr;

    assign s_status_depth = s_depth;
    assign s_status_depth_commit = s_depth;
    assign m_status_depth = m_depth;
    assign m_status_depth_commit = m_depth;

    // Full and empty conditions
    assign s_full = (s_depth >= DEPTH);
    assign m_empty = (m_depth == 0);

    // Write enable
    wire s_wr_en_raw = s_tvalid & s_tready & ~s_full & ~s_pause_active;
    assign s_wr_en = s_wr_en_raw;

    // Read enable
    wire m_rd_en_raw = m_tvalid & m_tready & ~m_empty & ~m_pause_active;
    assign m_rd_en = m_rd_en_raw;

    // Pointer updates
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= {PTR_WIDTH{1'b0}};
            s_wr_ptr_reg <= {PTR_WIDTH{1'b0}};
        end else if (s_wr_en) begin
            s_wr_ptr <= s_wr_ptr + 1'b1;
            s_wr_ptr_reg <= s_wr_ptr + 1'b1;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr <= {PTR_WIDTH{1'b0}};
            m_rd_ptr_reg <= {PTR_WIDTH{1'b0}};
        end else if (m_rd_en) begin
            m_rd_ptr <= m_rd_ptr + 1'b1;
            m_rd_ptr_reg <= m_rd_ptr + 1'b1;
        end
    end

    // Overflow detection
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_overflow <= 1'b0;
        end else if (s_tvalid & s_full & ~s_pause_active) begin
            s_overflow <= 1'b1;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_overflow <= 1'b0;
        end else if (m_tvalid & m_empty & ~m_pause_active) begin
            m_overflow <= 1'b0;
        end
    end

    // Bad frame detection
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_bad_frame <= 1'b0;
            s_good_frame <= 1'b0;
        end else if (s_wr_en) begin
            if (s_tlast) begin
                if (USER_ENABLE && (s_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                    s_bad_frame <= 1'b1;
                    s_good_frame <= 1'b0;
                end else begin
                    s_bad_frame <= 1'b0;
                    s_good_frame <= 1'b1;
                end
            end
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_bad_frame <= 1'b0;
            m_good_frame <= 1'b0;
        end else if (m_rd_en) begin
            if (m_tlast) begin
                if (USER_ENABLE && (m_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                    m_bad_frame <= 1'b1;
                    m_good_frame <= 1'b0;
                end else begin
                    m_bad_frame <= 1'b0;
                    m_good_frame <= 1'b1;
                end
            end
        end
    end

    // Pause acknowledgment
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_pause <= 1'b0;
        end else if (s_pause_req) begin
            s_pause <= 1'b1;
        end else if (s_depth == 0) begin
            s_pause <= 1'b0;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_pause <= 1'b0;
        end else if (m_pause_req) begin
            m_pause <= 1'b1;
        end else if (m_depth == 0) begin
            m_pause <= 1'b0;
        end
    end

    assign s_pause_ack = s_pause;
    assign m_pause_ack = m_pause;

    // Width adapter - Source side
    wire s_bytes_valid = S_KEEP_ENABLE ? (s_tkeep != 0) : 1'b1;
    wire [ACC_WIDTH-1:0] s_bytes_in = S_KEEP_ENABLE ? s_tkeep : S_BYTES[ACC_WIDTH-1:0];

    wire s_frame_end = s_tlast;
    wire s_drop = (DROP_OVERSIZE_FRAME && s_frame_end && (s_acc_bytes + s_bytes_in > MAX_BYTES[ACC_WIDTH-1:0])) ||
                  (DROP_BAD_FRAME && USER_ENABLE && (s_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE);

    wire s_accept = s_tvalid & s_tready & ~s_full & ~s_pause_active & ~s_drop;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_acc_bytes <= {ACC_WIDTH{1'b0}};
            s_acc_data <= {S_DATA_WIDTH{1'b0}};
            s_acc_valid <= 1'b0;
            s_tlast_reg <= 1'b0;
            s_tid_reg <= {ID_WIDTH{1'b0}};
            s_tdest_reg <= {DEST_WIDTH{1'b0}};
            s_tuser_reg <= {USER_WIDTH{1'b0}};
        end else if (s_accept) begin
            if (s_acc_valid) begin
                // Combine with existing accumulator
                if (s_acc_bytes + s_bytes_in <= M_BYTES[ACC_WIDTH-1:0]) begin
                    // Can fit in one output beat
                    s_acc_data <= {s_acc_data, s_tdata};
                    s_acc_bytes <= s_acc_bytes + s_bytes_in;
                    s_tlast_reg <= s_tlast_reg | s_tlast;
                    s_tid_reg <= s_tid;
                    s_tdest_reg <= s_tdest;
                    s_tuser_reg <= s_tuser;
                end else begin
                    // Need to output and start new
                    s_acc_data <= s_tdata;
                    s_acc_bytes <= s_bytes_in;
                    s_tlast_reg <= s_tlast;
                    s_tid_reg <= s_tid;
                    s_tdest_reg <= s_tdest;
                    s_tuser_reg <= s_tuser;
                end
            end else begin
                s_acc_data <= s_tdata;
                s_acc_bytes <= s_bytes_in;
                s_acc_valid <= 1'b1;
                s_tlast_reg <= s_tlast;
                s_tid_reg <= s_tid;
                s_tdest_reg <= s_tdest;
                s_tuser_reg <= s_tuser;
            end
        end
    end

    // Width adapter - Destination side
    wire m_bytes_out = M_KEEP_ENABLE ? m_tkeep : M_BYTES[ACC_WIDTH-1:0];

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_acc_bytes <= {ACC_WIDTH{1'b0}};
            m_acc_data <= {M_DATA_WIDTH{1'b0}};
            m_acc_valid <= 1'b0;
            m_tlast_reg <= 1'b0;
            m_tid_reg <= {ID_WIDTH{1'b0}};
            m_tdest_reg <= {DEST_WIDTH{1'b0}};
            m_tuser_reg <= {USER_WIDTH{1'b0}};
        end else if (m_rd_en) begin
            if (m_acc_valid) begin
                m_acc_data <= m_acc_data >> (m_bytes_out * 8);
                m_acc_bytes <= m_acc_bytes - m_bytes_out;
                if (m_acc_bytes == 0) begin
                    m_acc_valid <= 1'b0;
                    m_tlast_reg <= 1'b0;
                end
            end
        end
    end

    // FIFO data storage
    reg [S_DATA_WIDTH-1:0] mem [0:DEPTH-1];
    reg [S_KEEP_WIDTH-1:0] mem_keep [0:DEPTH-1];
    reg mem_tlast [0:DEPTH-1];
    reg [ID_WIDTH-1:0] mem_tid [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] mem_tdest [0:DEPTH-1];
    reg [USER_WIDTH-1:0] mem_tuser [0:DEPTH-1];

    // Write to memory
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            // No action
        end else if (s_wr_en) begin
            mem[s_wr_addr] <= s_tdata;
            mem_keep[s_wr_addr] <= s_tkeep;
            mem_tlast[s_wr_addr] <= s_tlast;
            mem_tid[s_wr_addr] <= s_tid;
            mem_tdest[s_wr_addr] <= s_tdest;
            mem_tuser[s_wr_addr] <= s_tuser;
        end
    end

    // Read from memory
    wire [S_DATA_WIDTH-1:0] rd_data;
    wire [S_KEEP_WIDTH-1:0] rd_keep;
    wire rd_tlast;
    wire [ID_WIDTH-1:0] rd_tid;
    wire [DEST_WIDTH-1:0] rd_tdest;
    wire [USER_WIDTH-1:0] rd_tuser;

    assign rd_data = mem[m_rd_addr];
    assign rd_keep = mem_keep[m_rd_addr];
    assign rd_tlast = mem_tlast[m_rd_addr];
    assign rd_tid = mem_tid[m_rd_addr];
    assign rd_tdest = mem_tdest[m_rd_addr];
    assign rd_tuser = mem_tuser[m_rd_addr];

    // Output ready logic
    assign s_axis_tready = ~s_full & ~s_pause_active & ~s_drop;
    assign m_axis_tvalid = ~m_empty & ~m_pause_active;
    assign m_axis_tdata = rd_data;
    assign m_axis_tkeep = rd_keep;
    assign m_axis_tlast = rd_tlast;
    assign m_axis_tid = rd_tid;
    assign m_axis_tdest = rd_tdest;
    assign m_axis_tuser = rd_tuser;

    // Status outputs
    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;

endmodule
