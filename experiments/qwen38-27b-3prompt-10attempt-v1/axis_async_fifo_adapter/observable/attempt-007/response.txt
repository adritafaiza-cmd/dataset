module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam S_BYTES = S_DATA_WIDTH / 8;
    localparam M_BYTES = M_DATA_WIDTH / 8;
    localparam MAX_BYTES = (S_BYTES > M_BYTES) ? S_BYTES : M_BYTES;
    localparam ACC_WIDTH = $clog2(MAX_BYTES) + 1;
    localparam ACC_BITS = ACC_WIDTH * 8;

    // Internal signals
    wire [S_DATA_WIDTH-1:0] s_tdata_reg;
    wire [S_KEEP_WIDTH-1:0] s_tkeep_reg;
    wire s_tlast_reg;
    wire [ID_WIDTH-1:0] s_tid_reg;
    wire [DEST_WIDTH-1:0] s_tdest_reg;
    wire [USER_WIDTH-1:0] s_tuser_reg;
    wire s_tvalid_reg;
    wire s_tready_int;

    wire [M_DATA_WIDTH-1:0] m_tdata_reg;
    wire [M_KEEP_WIDTH-1:0] m_tkeep_reg;
    wire m_tlast_reg;
    wire [ID_WIDTH-1:0] m_tid_reg;
    wire [DEST_WIDTH-1:0] m_tdest_reg;
    wire [USER_WIDTH-1:0] m_tuser_reg;
    wire m_tvalid_reg;
    wire m_tready_int;

    // FIFO pointers
    reg [PTR_WIDTH-1:0] s_wr_ptr;
    reg [PTR_WIDTH-1:0] m_rd_ptr;
    reg [PTR_WIDTH-1:0] s_wr_ptr_sync_m;
    reg [PTR_WIDTH-1:0] m_rd_ptr_sync_s;

    // RAM
    reg [S_DATA_WIDTH-1:0] ram_data [0:DEPTH-1];
    reg [S_KEEP_WIDTH-1:0] ram_keep [0:DEPTH-1];
    reg ram_last [0:DEPTH-1];
    reg [ID_WIDTH-1:0] ram_tid [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] ram_tdest [0:DEPTH-1];
    reg [USER_WIDTH-1:0] ram_tuser [0:DEPTH-1];

    // Accumulator state
    reg [ACC_BITS-1:0] acc_data;
    reg [ACC_WIDTH-1:0] acc_bytes;
    reg acc_last;
    reg [ID_WIDTH-1:0] acc_tid;
    reg [DEST_WIDTH-1:0] acc_tdest;
    reg [USER_WIDTH-1:0] acc_tuser;
    reg acc_valid;

    // Status
    reg s_overflow;
    reg s_bad_frame;
    reg s_good_frame;
    reg m_overflow;
    reg m_bad_frame;
    reg m_good_frame;

    // Pause
    reg s_pause_ack_reg;
    reg m_pause_ack_reg;

    // Synchronous reset for s_clk domain
    reg s_rst_sync;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_rst_sync <= 1'b1;
        else
            s_rst_sync <= 1'b0;
    end

    // Synchronous reset for m_clk domain
    reg m_rst_sync;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_rst_sync <= 1'b1;
        else
            m_rst_sync <= 1'b0;
    end

    // Clock domain crossing: s_wr_ptr to m_clk
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            s_wr_ptr_sync_m <= {PTR_WIDTH{1'b0}};
        else
            s_wr_ptr_sync_m <= s_wr_ptr;
    end

    // Clock domain crossing: m_rd_ptr to s_clk
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            m_rd_ptr_sync_s <= {PTR_WIDTH{1'b0}};
        else
            m_rd_ptr_sync_s <= m_rd_ptr;
    end

    // Depth calculation
    wire [PTR_WIDTH-1:0] s_depth = s_wr_ptr - m_rd_ptr_sync_s;
    wire [PTR_WIDTH-1:0] m_depth = s_wr_ptr_sync_m - m_rd_ptr;

    assign s_status_depth = s_depth;
    assign s_status_depth_commit = s_depth;
    assign m_status_depth = m_depth;
    assign m_status_depth_commit = m_depth;

    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;

    assign s_pause_ack = s_pause_ack_reg;
    assign m_pause_ack = m_pause_ack_reg;

    // Pause logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_pause_ack_reg <= 1'b0;
        else if (PAUSE_ENABLE)
            s_pause_ack_reg <= s_pause_req;
        else
            s_pause_ack_reg <= 1'b0;
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_pause_ack_reg <= 1'b0;
        else if (PAUSE_ENABLE)
            m_pause_ack_reg <= m_pause_req;
        else
            m_pause_ack_reg <= 1'b0;
    end

    // Write enable
    wire s_write_en = s_tvalid_reg && s_tready_int && !s_rst_sync;
    wire s_full = (s_depth == DEPTH);
    wire s_pause_active = PAUSE_ENABLE && s_pause_req;

    // Determine if we should accept data
    wire s_accept = s_tvalid_reg && !s_full && !s_pause_active;
    assign s_tready_int = !s_full && !s_pause_active;
    assign s_axis_tready = s_tready_int;

    // Write to RAM
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= {PTR_WIDTH{1'b0}};
            s_overflow <= 1'b0;
            s_bad_frame <= 1'b0;
            s_good_frame <= 1'b0;
        end else begin
            if (s_tvalid_reg && s_tready_int) begin
                ram_data[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_tdata_reg;
                ram_keep[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_tkeep_reg;
                ram_last[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_tlast_reg;
                ram_tid[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_tid_reg;
                ram_tdest[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_tdest_reg;
                ram_tuser[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_tuser_reg;
                s_wr_ptr <= s_wr_ptr + 1'b1;
            end
            if (s_tvalid_reg && s_full)
                s_overflow <= 1'b1;
            else
                s_overflow <= 1'b0;
        end
    end

    // Read from RAM in m_clk domain
    wire m_read_en = m_tvalid_reg && m_tready_int && !m_rst_sync;
    wire m_empty = (m_depth == 0);
    wire m_pause_active = PAUSE_ENABLE && m_pause_req;

    // Read data
    wire [S_DATA_WIDTH-1:0] rd_data = ram_data[m_rd_ptr[ADDR_WIDTH-1:0]];
    wire [S_KEEP_WIDTH-1:0] rd_keep = ram_keep[m_rd_ptr[ADDR_WIDTH-1:0]];
    wire rd_last = ram_last[m_rd_ptr[ADDR_WIDTH-1:0]];
    wire [ID_WIDTH-1:0] rd_tid = ram_tid[m_rd_ptr[ADDR_WIDTH-1:0]];
    wire [DEST_WIDTH-1:0] rd_tdest = ram_tdest[m_rd_ptr[ADDR_WIDTH-1:0]];
    wire [USER_WIDTH-1:0] rd_tuser = ram_tuser[m_rd_ptr[ADDR_WIDTH-1:0]];

    // Accumulator logic
    // Calculate how many bytes are valid in current beat
    reg [ACC_WIDTH-1:0] beat_bytes;
    integer i;
    always @* begin
        beat_bytes = 0;
        for (i = 0; i < S_KEEP_WIDTH; i = i + 1)
            beat_bytes = beat_bytes + rd_keep[i];
    end

    // Determine if we can output a full M_DATA_WIDTH beat
    wire can_output = acc_valid && (acc_bytes >= M_BYTES);
    wire output_last = can_output && acc_last && (acc_bytes == M_BYTES);

    // Generate output data
    reg [M_DATA_WIDTH-1:0] out_data;
    reg [M_KEEP_WIDTH-1:0] out_keep;
    reg out_last;
    reg [ID_WIDTH-1:0] out_tid;
    reg [DEST_WIDTH-1:0] out_tdest;
    reg [USER_WIDTH-1:0] out_tuser;

    always @* begin
        out_data = {M_DATA_WIDTH{1'b0}};
        out_keep = {M_KEEP_WIDTH{1'b0}};
        out_last = 1'b0;
        out_tid = {ID_WIDTH{1'b0}};
        out_tdest = {DEST_WIDTH{1'b0}};
        out_tuser = {USER_WIDTH{1'b0}};

        if (can_output) begin
            // Extract M_BYTES from acc_data
            // acc_data is stored LSB first
            for (i = 0; i < M_BYTES; i = i + 1)
                out_data[i*8 +: 8] = acc_data[i*8 +: 8];
            
            // Generate keep
            for (i = 0; i < M_KEEP_WIDTH; i = i + 1)
                out_keep[i] = 1'b1;
            
            out_last = output_last;
            out_tid = acc_tid;
            out_tdest = acc_tdest;
            out_tuser = acc_tuser;
        end
    end

    assign m_tdata_reg = out_data;
    assign m_tkeep_reg = out_keep;
    assign m_tlast_reg = out_last;
    assign m_tid_reg = out_tid;
    assign m_tdest_reg = out_tdest;
    assign m_tuser_reg = out_tuser;
    assign m_tvalid_reg = can_output;
    assign m_tready_int = !m_pause_active;
    assign m_axis_tdata = m_tdata_reg;
    assign m_axis_tkeep = m_tkeep_reg;
    assign m_axis_tvalid = m_tvalid_reg;
    assign m_axis_tlast = m_tlast_reg;
    assign m_axis_tid = m_tid_reg;
    assign m_axis_tdest = m_tdest_reg;
    assign m_axis_tuser = m_tuser_reg;

    // Accumulator update
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            acc_data <= {ACC_BITS{1'b0}};
            acc_bytes <= {ACC_WIDTH{1'b0}};
            acc_last <= 1'b0;
            acc_tid <= {ID_WIDTH{1'b0}};
            acc_tdest <= {DEST_WIDTH{1'b0}};
            acc_tuser <= {USER_WIDTH{1'b0}};
            acc_valid <= 1'b0;
            m_rd_ptr <= {PTR_WIDTH{1'b0}};
            m_overflow <= 1'b0;
            m_bad_frame <= 1'b0;
            m_good_frame <= 1'b0;
        end else begin
            // Read from RAM
            if (m_read_en) begin
                m_rd_ptr <= m_rd_ptr + 1'b1;
            end

            // Update accumulator
            if (m_read_en) begin
                // Add new data to accumulator
                // Shift acc_data left by beat_bytes*8 bits and add new data
                // For simplicity, we'll do byte-wise addition
                // This is a simplified implementation
                if (acc_bytes + beat_bytes <= MAX_BYTES) begin
                    // Add bytes
                    for (i = 0; i < S_BYTES; i = i + 1) begin
                        if (rd_keep[i]) begin
                            acc_data[(acc_bytes + i)*8 +: 8] <= rd_data[i*8 +: 8];
                        end
                    end
                    acc_bytes <= acc_bytes + beat_bytes;
                    if (rd_last)
                        acc_last <= 1'b1;
                    acc_tid <= rd_tid;
                    acc_tdest <= rd_tdest;
                    acc_tuser <= rd_tuser;
                    acc_valid <= 1'b1;
                end
            end

            // Output data
            if (m_read_en && can_output) begin
                // Remove M_BYTES from accumulator
                // Shift acc_data right by M_BYTES*8 bits
                for (i = M_BYTES; i < MAX_BYTES; i = i + 1)
                    acc_data[(i - M_BYTES)*8 +: 8] <= acc_data[i*8 +: 8];
                acc_bytes <= acc_bytes - M_BYTES;
                if (output_last) begin
                    acc_last <= 1'b0;
                    acc_valid <= 1'b0;
                    m_good_frame <= 1'b1;
                end
            end
        end
    end

    // Input register
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_tdata_reg <= {S_DATA_WIDTH{1'b0}};
            s_tkeep_reg <= {S_KEEP_WIDTH{1'b0}};
            s_tlast_reg <= 1'b0;
            s_tid_reg <= {ID_WIDTH{1'b0}};
            s_tdest_reg <= {DEST_WIDTH{1'b0}};
            s_tuser_reg <= {USER_WIDTH{1'b0}};
            s_tvalid_reg <= 1'b0;
        end else begin
            s_tdata_reg <= s_axis_tdata;
            s_tkeep_reg <= s_axis_tkeep;
            s_tlast_reg <= s_axis_tlast;
            s_tid_reg <= s_axis_tid;
            s_tdest_reg <= s_axis_tdest;
            s_tuser_reg <= s_axis_tuser;
            s_tvalid_reg <= s_axis_tvalid;
        end
    end

endmodule
