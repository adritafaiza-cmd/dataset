```verilog
module wbxclk #(
	parameter AW=32,
	parameter DW=32,
	parameter LGFIFO = 5
) (
	input	wire			i_wb_clk, i_reset,
	input	wire			i_wb_cyc, i_wb_stb, i_wb_we,
	input	wire	[(AW-1):0]	i_wb_addr,
	input	wire	[(DW-1):0]	i_wb_data,
	input	wire	[(DW/8-1):0]	i_wb_sel,
	output	wire			o_wb_stall,
	output	reg			o_wb_ack,
	output	reg	[(DW-1):0]	o_wb_data,
	output	reg			o_wb_err,
	input	wire			i_xclk_clk,
	output	reg			o_xclk_cyc,
	output	reg			o_xclk_stb,
	output	reg			o_xclk_we,
	output	reg	[(AW-1):0]	o_xclk_addr,
	output	reg	[(DW-1):0]	o_xclk_data,
	output	reg	[(DW/8-1):0]	o_xclk_sel,
	input	wire			i_xclk_stall,
	input	wire			i_xclk_ack,
	input	wire	[(DW-1):0]	i_xclk_data,
	input	wire			i_xclk_err
);

	localparam FIFO_DEPTH = 1 << LGFIFO;
	localparam FIFO_ADDR_WIDTH = LGFIFO;
	localparam DATA_WIDTH = AW + DW + (DW/8) + 1; // addr + data + sel + we

	// Request FIFO (wb -> xclk)
	reg [DATA_WIDTH-1:0] req_fifo_mem [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] req_wr_ptr, req_rd_ptr;
	reg [FIFO_ADDR_WIDTH:0] req_wr_ptr_sync1, req_wr_ptr_sync2;
	reg [FIFO_ADDR_WIDTH:0] req_rd_ptr_sync1, req_rd_ptr_sync2;

	// Response FIFO (xclk -> wb)
	reg [DW+1:0] rsp_fifo_mem [0:FIFO_DEPTH-1]; // data + err
	reg [FIFO_ADDR_WIDTH:0] rsp_wr_ptr, rsp_rd_ptr;
	reg [FIFO_ADDR_WIDTH:0] rsp_wr_ptr_sync1, rsp_wr_ptr_sync2;
	reg [FIFO_ADDR_WIDTH:0] rsp_rd_ptr_sync1, rsp_rd_ptr_sync2;

	// Request FIFO write logic (wb domain)
	wire req_wr_en = i_wb_cyc & i_wb_stb & ~o_wb_stall;
	wire [DATA_WIDTH-1:0] req_wr_data = {i_wb_we, i_wb_addr, i_wb_data, i_wb_sel};

	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset)
			req_wr_ptr <= {(FIFO_ADDR_WIDTH+1){1'b0}};
		else if (req_wr_en)
			req_wr_ptr <= req_wr_ptr + 1'b1;
	end

	// Request FIFO read logic (xclk domain)
	wire [FIFO_ADDR_WIDTH:0] req_wr_ptr_sync = req_wr_ptr_sync2;
	wire [FIFO_ADDR_WIDTH:0] req_rd_ptr_sync = req_rd_ptr_sync2;
	wire req_empty = (req_wr_ptr_sync == req_rd_ptr_sync);
	wire req_full = (req_wr_ptr_sync == req_rd_ptr_sync + FIFO_DEPTH);
	wire req_rd_en = ~req_empty & ~i_xclk_stall;

	always @(posedge i_xclk_clk) begin
		if (req_rd_en)
			req_rd_ptr <= req_rd_ptr + 1'b1;
	end

	always @(posedge i_xclk_clk) begin
		req_wr_ptr_sync1 <= req_wr_ptr;
		req_wr_ptr_sync2 <= req_wr_ptr_sync1;
	end

	always @(posedge i_wb_clk) begin
		req_rd_ptr_sync1 <= req_rd_ptr;
		req_rd_ptr_sync2 <= req_rd_ptr_sync1;
	end

	// Request FIFO read data
	wire [DATA_WIDTH-1:0] req_rd_data = req_fifo_mem[req_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
	wire [AW-1:0] req_addr = req_rd_data[DW+(DW/8)+1 +: AW];
	wire [DW-1:0] req_data = req_rd_data[(DW/8)+1 +: DW];
	wire [DW/8-1:0] req_sel = req_rd_data[0 +: DW/8];
	wire req_we = req_rd_data[DW+(DW/8)+1];

	// Output to xclk
	always @(posedge i_xclk_clk) begin
		if (req_rd_en) begin
			o_xclk_cyc <= 1'b1;
			o_xclk_stb <= 1'b1;
			o_xclk_we <= req_we;
			o_xclk_addr <= req_addr;
			o_xclk_data <= req_data;
			o_xclk_sel <= req_sel;
		end else begin
			o_xclk_cyc <= 1'b0;
			o_xclk_stb <= 1'b0;
		end
	end

	// Response FIFO write logic (xclk domain)
	wire rsp_wr_en = i_xclk_ack;
	wire [DW+1:0] rsp_wr_data = {i_xclk_err, i_xclk_data};

	always @(posedge i_xclk_clk) begin
		if (rsp_wr_en)
			rsp_wr_ptr <= rsp_wr_ptr + 1'b1;
	end

	// Response FIFO read logic (wb domain)
	wire [FIFO_ADDR_WIDTH:0] rsp_wr_ptr_sync = rsp_wr_ptr_sync2;
	wire [FIFO_ADDR_WIDTH:0] rsp_rd_ptr_sync = rsp_rd_ptr_sync2;
	wire rsp_empty = (rsp_wr_ptr_sync == rsp_rd_ptr_sync);
	wire rsp_rd_en = ~rsp_empty;

	always @(posedge i_wb_clk) begin
		if (rsp_rd_en)
			rsp_rd_ptr <= rsp_rd_ptr + 1'b1;
	end

	always @(posedge i_wb_clk) begin
		rsp_wr_ptr_sync1 <= rsp_wr_ptr;
		rsp_wr_ptr_sync2 <= rsp_wr_ptr_sync1;
	end

	always @(posedge i_xclk_clk) begin
		rsp_rd_ptr_sync1 <= rsp_rd_ptr;
		rsp_rd_ptr_sync2 <= rsp_rd_ptr_sync1;
	end

	// Response FIFO read data
	wire [DW+1:0] rsp_rd_data = rsp_fifo_mem[rsp_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
	wire [DW-1:0] rsp_data = rsp_rd_data[0 +: DW];
	wire rsp_err = rsp_rd_data[DW];

	// Output to wb
	always @(posedge i_wb_clk) begin
		if (rsp_rd_en) begin
			o_wb_ack <= 1'b1;
			o_wb_data <= rsp_data;
			o_wb_err <= rsp_err;
		end else begin
			o_wb_ack <= 1'b0;
		end
	end

	// Stall logic: stall when request FIFO is full
	assign o_wb_stall = req_full;

	// FIFO memory write (wb domain for req, xclk domain for rsp)
	always @(posedge i_wb_clk) begin
		if (req_wr_en)
			req_fifo_mem[req_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= req_wr_data;
	end

	always @(posedge i_xclk_clk) begin
		if (rsp_wr_en)
			rsp_fifo_mem[rsp_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= rsp_wr_data;
	end

endmodule
