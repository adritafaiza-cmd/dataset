			DW=32,
			LGFIFO = 5
	) (

	input	wire			i_wb_cyc, i_wb_stb, i_wb_we,

	input	wire	[(DW-1):0]	i_wb_data,

	reg [FIFO_ADDR_W:0] req_wr_ptr, req_rd_ptr;
	reg [FIFO_ADDR_W:0] req_count;

	// Store: data, err
	localparam RSP_W = DW + 1;

	reg [RSP_W-1:0] rsp_fifo_mem [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_W:0] rsp_wr_ptr, rsp_rd_ptr;
	reg [FIFO_ADDR_W:0] rsp_count;

	wire req_empty = (req_count == 0);

	// Response FIFO full/empty
	wire rsp_full = (rsp_count == FIFO_DEPTH);
	wire rsp_empty = (rsp_count == 0);

	assign o_wb_stall = (req_count != 0);

		end else if (!req_empty_xclk && !i_xclk_stall) begin

	reg [FIFO_ADDR_W:0] rsp_wr_ptr_g, rsp_rd_ptr_g;
