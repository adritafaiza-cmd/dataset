module wbxclk #(
	parameter	AW=32,
			DW=32,
			LGFIFO = 5
) (
	input	wire			i_wb_clk, i_reset,
	input	wire			i_wb_cyc, i_wb_stb, i_wb_we,
	input	wire	[(AW-1):0]	i_wb_addr,
	input	wire	[(DW-1):0]	i_wb_data,
	input	wire	[(DW/8-1):0]	i_wb_sel,
	output	wire			o_wb_stall,
	output	reg			o_wb_ack,
	output	reg	[(DW/8-1):0]	o_wb_data,
	output	reg			o_wb_err,
	input	wire			i_xclk_clk,
	output	reg			o_xclk_cyc,
	output	reg			o_xclk_stb,
	output	reg			o_xclk_we,
	output	reg	[(AW-1):0]	o_xclk_addr,
	output	reg	[(DW-1):0]	o_xclk_data,
	output	reg	[(DW/8-1):0]	o_xclk_sel,
	input	wire			i_xclk_stall,
	input	wire			i_xclk_ack,
	input	wire	[(DW-1):0]	i_xclk_data,
	input	wire	[(DW-8-1):0]	i_xclk_err
);

	// FIFO depth
	localparam FIFO_DEPTH = 1 << LGFIFO;
	localparam FIFO_WIDTH = AW + DW + (DW/8) + 1; // addr + data + sel + we

	// Request FIFO: WB side writes, XCLK side reads
	reg [FIFO_WIDTH-1:0] req_fifo [0:FIFO_DEPTH-1];
	reg [LGFIFO:0] req_wr_ptr, req_rd_ptr;
	reg [LGFIFO:0] req_count;

	// Response FIFO: XCLK side writes, WB side reads
	reg [DW+1:0] rsp_fifo [0:FIFO_DEPTH-1]; // data + err
	reg [LGFIFO:0] rsp_wr_ptr, rsp_rd_ptr;
	reg [LGFIFO:0] rsp_count;

	// WB side: request valid
	wire req_valid = i_wb_cyc && i_wb_stb;
	wire req_full = (req_count == FIFO_DEPTH);
	wire req_can_write = req_valid && !req_full;

	// XCLK side: request read
	wire req_empty = (req_count == 0);
	wire req_can_read = !req_empty && !i_xclk_stall;

	// WB side: response read
	wire rsp_empty = (rsp_count == 0);
	wire rsp_can_read = !rsp_empty;

	// XCLK side: response write
	wire rsp_full = (rsp_count == FIFO_DEPTH);
	wire rsp_can_write = i_xclk_ack && !rsp_full;

	// o_wb_stall is the outstanding-request count
	assign o_wb_stall = (req_count != 0);

	// WB clock domain: request FIFO write
	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset) begin
			req_wr_ptr <= 0;
			req_count <= 0;
		end else begin
			if (req_can_write) begin
				req_fifo[req_wr_ptr[LGFIFO-1:0]] <= {i_wb_addr, i_wb_data, i_wb_sel, i_wb_we};
				req_wr_ptr <= req_wr_ptr + 1;
			end
			if (req_can_read) begin
				req_count <= req_count - 1;
			end
			if (req_can_write && req_can_read) begin
				// both happen, count stays same
			end else if (req_can_write) begin
				req_count <= req_count + 1;
			end else if (req_can_read) begin
				req_count <= req_count - 1;
			end
		end
	end

	// WB clock domain: response FIFO read
	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset) begin
			rsp_rd_ptr <= 0;
			rsp_count <= 0;
			o_wb_ack <= 0;
			o_wb_data <= 0;
			o_wb_err <= 0;
		end else begin
			if (rsp_can_read) begin
				o_wb_ack <= 1;
				o_wb_data <= rsp_fifo[rsp_rd_ptr[LGFIFO-1:0]][DW:1];
				o_wb_err <= rsp_fifo[rsp_rd_ptr[LGFIFO-1:0]][0];
				rsp_rd_ptr <= rsp_rd_ptr + 1;
				rsp_count <= rsp_count - 1;
			end else begin
				o_wb_ack <= 0;
			end
		end
	end

	// XCLK clock domain: request FIFO read
	always @(posedge i_xclk_clk or posedge i_reset) begin
		if (i_reset) begin
			req_rd_ptr <= 0;
			req_count <= 0;
			o_xclk_cyc <= 0;
			o_xclk_stb <= 0;
			o_xclk_we <= 0;
			o_xclk_addr <= 0;
			o_xclk_data <= 0;
			o_xclk_sel <= 0;
		end else begin
			if (req_can_read) begin
				o_xclk_cyc <= 1;
				o_xclk_stb <= 1;
				o_xclk_we <= req_fifo[req_rd_ptr[LGFIFO-1:0]][0];
				o_xclk_sel <= req_fifo[req_rd_ptr[LGFIFO-1:0]][DW+1:DW+2];
				o_xclk_data <= req_fifo[req_rd_ptr[LGFIFO-1:0]][DW+DW+1:DW+2];
				o_xclk_addr <= req_fifo[req_rd_ptr[LGFIFO-1:0]][FIFO_WIDTH-1:DW+DW+2];
				req_rd_ptr <= req_rd_ptr + 1;
				req_count <= req_count - 1;
			end else begin
				o_xclk_cyc <= 0;
				o_xclk_stb <= 0;
			end
		end
	end

	// XCLK clock domain: response FIFO write
	always @(posedge i_xclk_clk or posedge i_reset) begin
		if (i_reset) begin
			rsp_wr_ptr <= 0;
			rsp_count <= 0;
		end else begin
			if (rsp_can_write) begin
				rsp_fifo[rsp_wr_ptr[LGFIFO-1:0]] <= {i_xclk_data, i_xclk_err};
				rsp_wr_ptr <= rsp_wr_ptr + 1;
				rsp_count <= rsp_count + 1;
			end
		end
	end

endmodule
