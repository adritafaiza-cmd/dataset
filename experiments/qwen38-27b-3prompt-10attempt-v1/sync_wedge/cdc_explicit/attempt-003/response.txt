module sync_wedge #(
    parameter STAGES = 2
)(
    input  clk_i,
    input  rst_ni,
    input  en_i,
    input  serial_i,
    output r_edge_o,
    output f_edge_o,
    output serial_o
);

    // Ensure at least 2 stages for safe CDC
    localparam SYNC_STAGES = (STAGES < 2) ? 2 : STAGES;

    // Synchronizer chain
    reg [SYNC_STAGES-1:0] sync_reg;

    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            sync_reg <= {SYNC_STAGES{1'b0}};
        else
            sync_reg <= {sync_reg[SYNC_STAGES-2:0], serial_i};
    end

    // Synchronized output
    assign serial_o = sync_reg[SYNC_STAGES-1];

    // Previous synchronized value for edge detection
    reg serial_d;

    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            serial_d <= 1'b0;
        else
            serial_d <= serial_o;
    end

    // Edge detection
    wire rising  = serial_o & ~serial_d;
    wire falling = ~serial_o & serial_d;

    // Gated edge outputs
    assign r_edge_o = rising  & en_i;
    assign f_edge_o = falling & en_i;

endmodule
