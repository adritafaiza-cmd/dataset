module sync_wedge #(
    parameter STAGES = 2
)(
    input  clk_i,
    input  rst_ni,
    input  en_i,
    input  serial_i,
    output r_edge_o,
    output f_edge_o,
    output serial_o
);

    // Synchronizer chain
    reg [STAGES-1:0] sync_reg;
    reg              sync_prev;

    // Edge detection registers
    reg              r_edge_reg;
    reg              f_edge_reg;

    // Synchronize the asynchronous input
    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            sync_reg <= {STAGES{1'b0}};
        end else begin
            sync_reg[0] <= serial_i;
            if (STAGES > 1) begin
                sync_reg[1] <= sync_reg[0];
            end
            if (STAGES > 2) begin
                sync_reg[2] <= sync_reg[1];
            end
            if (STAGES > 3) begin
                sync_reg[3] <= sync_reg[2];
            end
        end
    end

    // The synchronized output is the last stage of the synchronizer
    wire sync_bit = sync_reg[STAGES-1];

    // Track the previous value of the synchronized bit for edge detection
    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            sync_prev <= 1'b0;
        end else begin
            sync_prev <= sync_bit;
        end
    end

    // Edge detection
    wire rising_edge  = sync_bit & ~sync_prev;
    wire falling_edge = ~sync_bit & sync_prev;

    // Edge outputs gated by enable
    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            r_edge_reg <= 1'b0;
            f_edge_reg <= 1'b0;
        end else begin
            r_edge_reg <= en_i & rising_edge;
            f_edge_reg <= en_i & falling_edge;
        end
    end

    assign serial_o = sync_bit;
    assign r_edge_o = r_edge_reg;
    assign f_edge_o = f_edge_reg;

endmodule
