module sync_wedge #(
    parameter STAGES = 2
)(
    input  clk_i,
    input  rst_ni,
    input  en_i,
    input  serial_i,
    output r_edge_o,
    output f_edge_o,
    output serial_o
);

    localparam SYNC_STAGES = (STAGES < 2) ? 2 : STAGES;

    reg [SYNC_STAGES-1:0] sync_reg;
    reg                   prev_sync;

    wire sync_bit = sync_reg[SYNC_STAGES-1];

    assign serial_o = sync_bit;

    wire rising  = sync_bit & ~prev_sync;
    wire falling = ~sync_bit & prev_sync;

    assign r_edge_o = en_i & rising;
    assign f_edge_o = en_i & falling;

    integer i;
    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            sync_reg <= {SYNC_STAGES{1'b0}};
            prev_sync <= 1'b0;
        end else begin
            sync_reg[0] <= serial_i;
            for (i = 1; i < SYNC_STAGES; i = i + 1)
                sync_reg[i] <= sync_reg[i-1];
            prev_sync <= sync_bit;
        end
    end

endmodule
