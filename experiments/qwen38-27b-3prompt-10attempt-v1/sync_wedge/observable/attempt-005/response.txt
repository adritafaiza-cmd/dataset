```verilog
module sync_wedge #(
    parameter STAGES = 2
)(
    input  clk_i,
    input  rst_ni,
    input  en_i,
    input  serial_i,
    output r_edge_o,
    output f_edge_o,
    output serial_o
);

    reg [STAGES-1:0] sync_reg;
    reg              serial_d;
    reg              serial_dd;

    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            sync_reg <= {STAGES{1'b0}};
            serial_d <= 1'b0;
            serial_dd <= 1'b0;
        end else begin
            sync_reg <= {sync_reg[STAGES-2:0], serial_i};
            serial_d <= sync_reg[STAGES-1];
            serial_dd <= serial_d;
        end
    end

    assign serial_o = sync_reg[STAGES-1];
    assign r_edge_o = en_i & serial_d & ~serial_dd;
    assign f_edge_o = en_i & ~serial_d & serial_dd;

endmodule
