module axis_register #
(
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter REG_TYPE = 2
)
(
    input  wire                   clk,
    input  wire                   rst,
    /*
     * AXI Stream input
     */
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI Stream output
     */
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser
);

    // Internal signals
    wire [DATA_WIDTH-1:0]  reg_tdata;
    wire [KEEP_WIDTH-1:0]  reg_tkeep;
    wire                   reg_tvalid;
    wire                   reg_tready;
    wire                   reg_tlast;
    wire [ID_WIDTH-1:0]    reg_tid;
    wire [DEST_WIDTH-1:0]  reg_tdest;
    wire [USER_WIDTH-1:0]  reg_tuser;

    // Wire mode (REG_TYPE = 0)
    generate
        if (REG_TYPE == 0) begin : gen_wire
            assign s_axis_tready = m_axis_tready;
            assign m_axis_tdata  = s_axis_tdata;
            assign m_axis_tkeep  = s_axis_tkeep;
            assign m_axis_tvalid = s_axis_tvalid;
            assign m_axis_tlast  = s_axis_tlast;
            assign m_axis_tid    = s_axis_tid;
            assign m_axis_tdest  = s_axis_tdest;
            assign m_axis_tuser  = s_axis_tuser;
        end
        else if (REG_TYPE == 1) begin : gen_simple_buffer
            // Simple buffer: one register stage
            reg_simple_buffer #(
                .DATA_WIDTH(DATA_WIDTH),
                .KEEP_ENABLE(KEEP_ENABLE),
                .KEEP_WIDTH(KEEP_WIDTH),
                .LAST_ENABLE(LAST_ENABLE),
                .ID_ENABLE(ID_ENABLE),
                .ID_WIDTH(ID_WIDTH),
                .DEST_ENABLE(DEST_ENABLE),
                .DEST_WIDTH(DEST_WIDTH),
                .USER_ENABLE(USER_ENABLE),
                .USER_WIDTH(USER_WIDTH)
            ) u_simple_buffer (
                .clk(clk),
                .rst(rst),
                .s_axis_tdata(s_axis_tdata),
                .s_axis_tkeep(s_axis_tkeep),
                .s_axis_tvalid(s_axis_tvalid),
                .s_axis_tready(s_axis_tready),
                .s_axis_tlast(s_axis_tlast),
                .s_axis_tid(s_axis_tid),
                .s_axis_tdest(s_axis_tdest),
                .s_axis_tuser(s_axis_tuser),
                .m_axis_tdata(m_axis_tdata),
                .m_axis_tkeep(m_axis_tkeep),
                .m_axis_tvalid(m_axis_tvalid),
                .m_axis_tready(m_axis_tready),
                .m_axis_tlast(m_axis_tlast),
                .m_axis_tid(m_axis_tid),
                .m_axis_tdest(m_axis_tdest),
                .m_axis_tuser(m_axis_tuser)
            );
        end
        else if (REG_TYPE == 2) begin : gen_skid_buffer
            // Skid buffer: two register stages with skid
            reg_skid_buffer #(
                .DATA_WIDTH(DATA_WIDTH),
                .KEEP_ENABLE(KEEP_ENABLE),
                .KEEP_WIDTH(KEEP_WIDTH),
                .LAST_ENABLE(LAST_ENABLE),
                .ID_ENABLE(ID_ENABLE),
                .ID_WIDTH(ID_WIDTH),
                .DEST_ENABLE(DEST_ENABLE),
                .DEST_WIDTH(DEST_WIDTH),
                .USER_ENABLE(USER_ENABLE),
                .USER_WIDTH(USER_WIDTH)
            ) u_skid_buffer (
                .clk(clk),
                .rst(rst),
                .s_axis_tdata(s_axis_tdata),
                .s_axis_tkeep(s_axis_tkeep),
                .s_axis_tvalid(s_axis_tvalid),
                .s_axis_tready(s_axis_tready),
                .s_axis_tlast(s_axis_tlast),
                .s_axis_tid(s_axis_tid),
                .s_axis_tdest(s_axis_tdest),
                .s_axis_tuser(s_axis_tuser),
                .m_axis_tdata(m_axis_tdata),
                .m_axis_tkeep(m_axis_tkeep),
                .m_axis_tvalid(m_axis_tvalid),
                .m_axis_tready(m_axis_tready),
                .m_axis_tlast(m_axis_tlast),
                .m_axis_tid(m_axis_tid),
                .m_axis_tdest(m_axis_tdest),
                .m_axis_tuser(m_axis_tuser)
            );
        end
    endgenerate

endmodule

module reg_simple_buffer #
(
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1
)
(
    input  wire                   clk,
    input  wire                   rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser
);

    reg [DATA_WIDTH-1:0]  tdata_reg;
    reg [KEEP_WIDTH-1:0]  tkeep_reg;
    reg                   tvalid_reg;
    reg                   tlast_reg;
    reg [ID_WIDTH-1:0]    tid_reg;
    reg [DEST_WIDTH-1:0]  tdest_reg;
    reg [USER_WIDTH-1:0]  tuser_reg;

    wire s_axis_tready = m_axis_tready;
    wire m_axis_tvalid = tvalid_reg;

    assign m_axis_tdata  = tdata_reg;
    assign m_axis_tkeep  = tkeep_reg;
    assign m_axis_tlast  = tlast_reg;
    assign m_axis_tid    = tid_reg;
    assign m_axis_tdest  = tdest_reg;
    assign m_axis_tuser  = tuser_reg;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            tvalid_reg <= 1'b0;
        end else begin
            if (s_axis_tvalid && m_axis_tready) begin
                tvalid_reg <= 1'b1;
                tdata_reg  <= s_axis_tdata;
                tkeep_reg  <= s_axis_tkeep;
                tlast_reg  <= s_axis_tlast;
                tid_reg    <= s_axis_tid;
                tdest_reg  <= s_axis_tdest;
                tuser_reg  <= s_axis_tuser;
            end else if (m_axis_tready) begin
                tvalid_reg <= 1'b0;
            end
        end
    end

endmodule

module reg_skid_buffer #
(
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1
)
(
    input  wire                   clk,
    input  wire                   rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser
);

    // Main register
    reg [DATA_WIDTH-1:0]  main_tdata;
    reg [KEEP_WIDTH-1:0]  main_tkeep;
    reg                   main_tvalid;
    reg                   main_tlast;
    reg [ID_WIDTH-1:0]    main_tid;
    reg [DEST_WIDTH-1:0]  main_tdest;
    reg [USER_WIDTH-1:0]  main_tuser;

    // Skid register
    reg [DATA_WIDTH-1:0]  skid_tdata;
    reg [KEEP_WIDTH-1:0]  skid_tkeep;
    reg                   skid_tvalid;
    reg                   skid_tlast;
    reg [ID_WIDTH-1:0]    skid_tid;
    reg [DEST_WIDTH-1:0]  skid_tdest;
    reg [USER_WIDTH-1:0]  skid_tuser;

    wire main_tready = m_axis_tready;
    wire m_axis_tvalid = main_tvalid;

    assign m_axis_tdata  = main_tdata;
    assign m_axis_tkeep  = main_tkeep;
    assign m_axis_tlast  = main_tlast;
    assign m_axis_tid    = main_tid;
    assign m_axis_tdest  = main_tdest;
    assign m_axis_tuser  = main_tuser;

    // s_axis_tready: accept input if main is empty, or if main can be drained to skid
    wire main_empty = ~main_tvalid;
    wire skid_empty = ~skid_tvalid;
    wire main_drain = main_tvalid && m_axis_tready;
    wire skid_drain = skid_tvalid && m_axis_tready;

    // s_axis_tready is high when:
    // - main is empty, OR
    // - main is valid and m_axis_tready is high (main can drain, and if skid is empty, input can go to main; if skid is full, input can go to skid only if main drains)
    // Actually, for a skid buffer:
    // s_axis_tready = main_empty || (main_tvalid && m_axis_tready)
    // But we also need to consider skid. The standard skid buffer logic:
    // s_axis_tready = ~main_tvalid || (main_tvalid && m_axis_tready)
    // This is equivalent to: s_axis_tready = ~main_tvalid || m_axis_tready
    // Wait, that's not quite right. Let me think again.
    
    // A skid buffer has two storage locations: main and skid.
    // s_axis_tready should be high when we can accept a new beat.
    // We can accept a new beat if:
    // 1. Main is empty (new beat goes to main)
    // 2. Main is full but m_axis_tready is high (main drains, new beat goes to main, and if skid was full, skid drains to main... no wait)
    
    // Let me use the standard approach:
    // s_axis_tready = ~main_tvalid || (main_tvalid && m_axis_tready)
    // This means: if main is empty, accept. If main is full, accept only if m_axis_tready is high (so main can drain).
    // But what about skid? If skid is full and main is full, we can't accept.
    // Actually, the condition should be:
    // s_axis_tready = ~main_tvalid || (main_tvalid && m_axis_tready)
    // This works because if main is full and m_axis_tready is low, we can't accept.
    // If main is full and m_axis_tready is high, we can accept (main will drain, and the new beat goes to main).
    // But what if skid is also full? Then when main drains, skid moves to main, and the new beat goes to skid.
    // So the condition is correct.
    
    wire s_axis_tready = ~main_tvalid || (main_tvalid && m_axis_tready);

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            main_tvalid <= 1'b0;
            skid_tvalid <= 1'b0;
        end else begin
            // Determine what happens this cycle
            wire s_axis_handshake = s_axis_tvalid && s_axis_tready;
            wire m_axis_handshake = m_axis_tvalid && m_axis_tready;

            // Update main register
            if (s_axis_handshake && main_empty) begin
                // New beat goes to main
                main_tvalid <= 1'b1;
                main_tdata  <= s_axis_tdata;
                main_tkeep  <= s_axis_tkeep;
                main_tlast  <= s_axis_tlast;
                main_tid    <= s_axis_tid;
                main_tdest  <= s_axis_tdest;
                main_tuser  <= s_axis_tuser;
            end else if (s_axis_handshake && main_tvalid && m_axis_tready) begin
                // Main drains, new beat goes to main
                main_tvalid <= 1'b1;
                main_tdata  <= s_axis_tdata;
                main_tkeep  <= s_axis_tkeep;
                main_tlast  <= s_axis_tlast;
                main_tid    <= s_axis_tid;
                main_tdest  <= s_axis_tdest;
                main_tuser  <= s_axis_tuser;
            end else if (m_axis_handshake) begin
                // Main drains, no new beat
                if (skid_tvalid) begin
                    // Skid moves to main
                    main_tvalid <= 1'b1;
                    main_tdata  <= skid_tdata;
                    main_tkeep  <= skid_tkeep;
                    main_tlast  <= skid_tlast;
                    main_tid    <= skid_tid;
                    main_tdest  <= skid_tdest;
                    main_tuser  <= skid_tuser;
                end else begin
                    main_tvalid <= 1'b0;
                end
            end
            // else: no change to main

            // Update skid register
            if (s_axis_handshake && main_tvalid && m_axis_tready && skid_tvalid) begin
                // Main drains, skid moves to main, new beat goes to skid
                skid_tvalid <= 1'b1;
                skid_tdata  <= s_axis_tdata;
                skid_tkeep  <= s_axis_tkeep;
                skid_tlast  <= s_axis_tlast;
                skid_tid    <= s_axis_tid;
                skid_tdest  <= s_axis_tdest;
                skid_tuser  <= s_axis_tuser;
            end else if (s_axis_handshake && main_tvalid && m_axis_tready && ~skid_tvalid) begin
                // Main drains, skid is empty, new beat goes to skid? No, new beat goes to main.
                // Wait, if main drains and skid is empty, the new beat goes to main. Skid stays empty.
                // So no change to skid.
            end else if (m_axis_handshake && skid_tvalid) begin
                // Main drains, skid moves to main, skid becomes empty
                skid_tvalid <= 1'b0;
            end
            // else: no change to skid
        end
    end

endmodule
