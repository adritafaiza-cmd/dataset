module apb_regs #(
  parameter int unsigned        NoApbRegs    = 32'd0,
  parameter int unsigned        ApbAddrWidth = 32'd0,
  parameter int unsigned        AddrOffset   = 32'd4,
  parameter int unsigned        ApbDataWidth = 32'd0,
  parameter int unsigned        RegDataWidth = 32'd0,
  parameter bit [NoApbRegs-1:0] ReadOnly     = 32'h0,
  parameter type                req_t        = logic,
  parameter type                resp_t       = logic,
  parameter type apb_addr_t                  = logic[ApbAddrWidth-1:0],
  parameter type reg_data_t                  = logic[RegDataWidth-1:0]
) (
  input  logic                      pclk_i,
  input  logic                      preset_ni,
  input  req_t                      req_i,
  output resp_t                     resp_o,
  input  apb_addr_t                 base_addr_i,
  input  reg_data_t [NoApbRegs-1:0] reg_init_i,
  output reg_data_t [NoApbRegs-1:0] reg_q_o
);

  // Local parameters
  localparam int unsigned ADDR_SHIFT = $clog2(AddrOffset);
  localparam int unsigned REG_INDEX_WIDTH = $clog2(NoApbRegs);

  // APB signal extraction (assuming standard APB3/4 structure)
  // We need to extract: psel, penable, pwrite, paddr, pwdata, pready, preslverr
  // Since req_t and resp_t are parameterized types, we assume they have standard APB fields.
  // For a generic implementation, we'll use direct field access assuming the types
  // are structured with standard APB signal names.

  // Extract APB signals from req_t
  wire psel;
  wire penable;
  wire pwrite;
  wire [ApbAddrWidth-1:0] paddr;
  wire [ApbDataWidth-1:0] pwdata;

  assign psel    = req_i.psel;
  assign penable = req_i.penable;
  assign pwrite  = req_i.pwrite;
  assign paddr   = req_i.paddr;
  assign pwdata  = req_i.pwdata;

  // Response signals
  wire pready;
  wire preslverr;

  assign resp_o.pready    = pready;
  assign resp_o.preslverr = preslverr;

  // Register file
  reg_data_t [NoApbRegs-1:0] reg_q;

  // Address decoding
  // Calculate register index from address
  wire [ApbAddrWidth-1:0] addr_offset = paddr - base_addr_i;
  wire [REG_INDEX_WIDTH-1:0] reg_index = (AddrOffset == 1) ? 
    addr_offset[REG_INDEX_WIDTH-1:0] : 
    addr_offset[ADDR_SHIFT + REG_INDEX_WIDTH-1:ADDR_SHIFT];

  wire addr_valid = (reg_index < NoApbRegs);

  // Write enable
  wire write_en = psel & penable & pwrite & addr_valid;

  // Read data
  reg_data_t read_data;
  assign read_data = reg_q[reg_index];

  // Register update logic
  always @(posedge pclk_i or negedge preset_ni) begin
    if (!preset_ni) begin
      for (int i = 0; i < NoApbRegs; i++) begin
        reg_q[i] <= reg_init_i[i];
      end
    end else begin
      if (write_en) begin
        if (!ReadOnly[reg_index]) begin
          reg_q[reg_index] <= pwdata[RegDataWidth-1:0];
        end
      end
    end
  end

  // Drive output
  always @(*) begin
    for (int i = 0; i < NoApbRegs; i++) begin
      reg_q_o[i] = reg_q[i];
    end
  end

  // APB ready and error
  assign pready    = 1'b1;
  assign preslverr = 1'b0;

endmodule
