module apb_regs #(
  parameter int unsigned        NoApbRegs    = 32'd0,
  parameter int unsigned        ApbAddrWidth = 32'd0,
  parameter int unsigned        AddrOffset   = 32'd4,
  parameter int unsigned        ApbDataWidth = 32'd0,
  parameter int unsigned        RegDataWidth = 32'd0,
  parameter bit [NoApbRegs-1:0] ReadOnly     = 32'h0,
  parameter type                req_t        = logic,
  parameter type                resp_t       = logic,
  parameter type apb_addr_t                  = logic[ApbAddrWidth-1:0],
  parameter type reg_data_t                  = logic[RegDataWidth-1:0]
) (
  input  logic                      pclk_i,
  input  logic                      preset_ni,
  input  req_t                      req_i,
  output resp_t                     resp_o,
  input  apb_addr_t                 base_addr_i,
  input  reg_data_t [NoApbRegs-1:0] reg_init_i,
  output reg_data_t [NoApbRegs-1:0] reg_q_o
);

  wire        pwrite;

  // Let's assume req_t is a struct with fields: psel, penable, pwrite, paddr, pwdata, pstrb

  // Actually, since req_t is a parameter type, we need to cast or use a different method
  // Let's assume the testbench provides req_t as a struct with these fields
  // We'll use a typedef approach or direct field access

  // Let's define the APB signals from req_t
  // Assuming req_t is a struct with: psel, penable, pwrite, paddr, pwdata, pstrb
  assign psel    = req_t.psel;
  assign penable = req_t.penable;
  assign pwrite  = req_t.pwrite;
  assign paddr   = req_t.paddr;
  assign pwdata  = req_t.pwdata;
  assign pstrb   = req_t.pstrb;

  // APB response signals
  wire [ApbDataWidth-1:0] prdata;
  wire        pready;
  wire        pslverr;

  // Register file
  reg_data_t [NoApbRegs-1:0] reg_q;

  // Address decoding
  // Calculate register index from address
  // addr = base_addr + index * AddrOffset
  // index = (addr - base_addr) / AddrOffset

  wire [ApbAddrWidth-1:0] addr_offset;
  wire [ApbAddrWidth-1:0] addr_diff;
  wire [ApbAddrWidth-1:0] reg_index;
  wire                    addr_valid;

  assign addr_offset = paddr - base_addr_i;
  assign addr_diff   = addr_offset;
  assign reg_index   = addr_diff / AddrOffset;
  assign addr_valid  = (reg_index < NoApbRegs) && (addr_offset % AddrOffset == 0);

  // Read data
  wire [RegDataWidth-1:0] read_data;
  assign read_data = addr_valid ? reg_q[reg_index] : '0;

  // Convert read data to APB data width
  wire [ApbDataWidth-1:0] prdata_wire;
  assign prdata_wire = (ApbDataWidth >= RegDataWidth) ? 
                       {{(ApbDataWidth-RegDataWidth){1'b0}}, read_data} : 
                       read_data[ApbDataWidth-1:0];

  assign prdata = prdata_wire;

  // Write enable
  wire        write_en;
  wire [ApbDataWidth-1:0] write_data;
  wire [RegDataWidth-1:0] write_data_reg;

  assign write_en = psel & penable & pwrite & addr_valid;
  assign write_data = pwdata;
  assign write_data_reg = (RegDataWidth >= ApbDataWidth) ? 
                          {{(RegDataWidth-ApbDataWidth){1'b0}}, write_data} : 
                          write_data[RegDataWidth-1:0];

  // Apply read-only mask
  wire [RegDataWidth-1:0] ro_mask;
  assign ro_mask = ReadOnly[reg_index] ? '1 : '0;

  // Register update
  always @(posedge pclk_i or negedge preset_ni) begin
    if (!preset_ni) begin
      for (int i = 0; i < NoApbRegs; i++) begin
        reg_q[i] <= reg_init_i[i];
      end
    end else begin
      if (write_en) begin
        // Apply read-only mask: keep original bits where RO is set
        reg_q[reg_index] <= (reg_q[reg_index] & ro_mask) | (write_data_reg & ~ro_mask);
      end
    end
  end

  // Drive output
  assign reg_q_o = reg_q;

  // APB response
  assign pready  = 1'b1;
  assign pslverr = 1'b0;

  // Build response
  // Assuming resp_t is a struct with: pready, pslverr, prdata
  assign resp_o = resp_t'{
    pready:  pready,
    pslverr: pslverr,
    prdata:  prdata
  };

endmodule
