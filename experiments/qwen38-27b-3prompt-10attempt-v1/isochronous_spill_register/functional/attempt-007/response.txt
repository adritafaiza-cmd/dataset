module isochronous_spill_register #(
    parameter WIDTH = 8
)(
    input              src_clk_i,
    input              src_rst_ni,
    input              src_valid_i,
    output             src_ready_o,
    input  [WIDTH-1:0] src_data_i,
    input              dst_clk_i,
    input              dst_rst_ni,
    output             dst_valid_o,
    input              dst_ready_i,
    output [WIDTH-1:0] dst_data_o
);

    // Stage 0 (first register)
    reg [WIDTH-1:0] stage0_data;
    reg             stage0_valid;

    // Stage 1 (second register)
    reg [WIDTH-1:0] stage1_data;
    reg             stage1_valid;

    // Source domain: stage0
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            stage0_valid <= 1'b0;
            stage0_data  <= {WIDTH{1'b0}};
        end else begin
            if (src_valid_i && src_ready_o) begin
                stage0_valid <= 1'b1;
                stage0_data  <= src_data_i;
            end else if (stage0_valid && dst_ready_i) begin
                stage0_valid <= 1'b0;
            end
        end
    end

    // Destination domain: stage1
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            stage1_valid <= 1'b0;
            stage1_data  <= {WIDTH{1'b0}};
        end else begin
            if (dst_ready_i) begin
                stage1_valid <= stage0_valid;
                stage1_data  <= stage0_data;
            end
        end
    end

    // Outputs
    assign src_ready_o = !stage0_valid || dst_ready_i;
    assign dst_valid_o = stage1_valid;
    assign dst_data_o  = stage1_data;

endmodule
