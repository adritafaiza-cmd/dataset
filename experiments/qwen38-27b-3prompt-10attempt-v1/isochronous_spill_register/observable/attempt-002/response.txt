module isochronous_spill_register #(
    parameter WIDTH = 8
)(
    input              src_clk_i,
    input              src_rst_ni,
    input              src_valid_i,
    output             src_ready_o,
    input  [WIDTH-1:0] src_data_i,
    input              dst_clk_i,
    input              dst_rst_ni,
    output             dst_valid_o,
    input              dst_ready_i,
    output [WIDTH-1:0] dst_data_o
);

    // Source clock domain
    reg [WIDTH-1:0] src_reg0;
    reg             src_reg0_valid;
    reg [WIDTH-1:0] src_reg1;
    reg             src_reg1_valid;

    // Destination clock domain
    reg [WIDTH-1:0] dst_reg0;
    reg             dst_reg0_valid;
    reg [WIDTH-1:0] dst_reg1;
    reg             dst_reg1_valid;

    // Synchronization of source valid signals to destination domain
    reg [1:0] src_reg0_valid_sync;
    reg [1:0] src_reg1_valid_sync;

    // Synchronization of destination valid signals to source domain
    reg [1:0] dst_reg0_valid_sync;
    reg [1:0] dst_reg1_valid_sync;

    // Synchronization of destination ready to source domain
    reg [1:0] dst_ready_sync;

    // Synchronization of source ready to destination domain
    reg [1:0] src_ready_sync;

    // Source domain logic
    wire src_can_accept = !src_reg0_valid || !src_reg1_valid;
    assign src_ready_o = src_can_accept;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_reg0_valid <= 1'b0;
            src_reg1_valid <= 1'b0;
            src_reg0       <= {WIDTH{1'b0}};
            src_reg1       <= {WIDTH{1'b0}};
        end else begin
            if (src_valid_i && src_ready_o) begin
                if (!src_reg0_valid) begin
                    src_reg0       <= src_data_i;
                    src_reg0_valid <= 1'b1;
                end else if (!src_reg1_valid) begin
                    src_reg1       <= src_data_i;
                    src_reg1_valid <= 1'b1;
                end
            end
        end
    end

    // Destination domain logic
    wire dst_can_accept = !dst_reg0_valid || !dst_reg1_valid;
    assign dst_valid_o = dst_reg0_valid || dst_reg1_valid;
    assign dst_data_o  = dst_reg0_valid ? dst_reg0 : dst_reg1;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_reg0_valid <= 1'b0;
            dst_reg1_valid <= 1'b0;
            dst_reg0       <= {WIDTH{1'b0}};
            dst_reg1       <= {WIDTH{1'b0}};
        end else begin
            // Accept from source if destination has space
            if (dst_can_accept) begin
                if (!dst_reg0_valid && src_reg0_valid_sync[1]) begin
                    dst_reg0       <= src_reg0;
                    dst_reg0_valid <= 1'b1;
                end else if (!dst_reg1_valid && src_reg1_valid_sync[1]) begin
                    dst_reg1       <= src_reg1;
                    dst_reg1_valid <= 1'b1;
                end
            end

            // Pop to destination if ready
            if (dst_valid_o && dst_ready_i) begin
                if (dst_reg0_valid) begin
                    dst_reg0_valid <= 1'b0;
                    if (dst_reg1_valid) begin
                        dst_reg0       <= dst_reg1;
                        dst_reg1_valid <= 1'b0;
                    end
                end else if (dst_reg1_valid) begin
                    dst_reg1_valid <= 1'b0;
                end
            end
        end
    end

    // Synchronizers: source valid to destination
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_reg0_valid_sync <= 2'b00;
            src_reg1_valid_sync <= 2'b00;
        end else begin
            src_reg0_valid_sync <= {src_reg0_valid_sync[0], src_reg0_valid};
            src_reg1_valid_sync <= {src_reg1_valid_sync[0], src_reg1_valid};
        end
    end

    // Synchronizers: destination valid to source
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_reg0_valid_sync <= 2'b00;
            dst_reg1_valid_sync <= 2'b00;
        end else begin
            dst_reg0_valid_sync <= {dst_reg0_valid_sync[0], dst_reg0_valid};
            dst_reg1_valid_sync <= {dst_reg1_valid_sync[0], dst_reg1_valid};
        end
    end

    // Synchronizers: destination ready to source
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_ready_sync <= 2'b00;
        end else begin
            dst_ready_sync <= {dst_ready_sync[0], dst_ready_i};
        end
    end

    // Synchronizers: source ready to destination
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_ready_sync <= 2'b00;
        end else begin
            src_ready_sync <= {src_ready_sync[0], src_ready_o};
        end
    end

endmodule
