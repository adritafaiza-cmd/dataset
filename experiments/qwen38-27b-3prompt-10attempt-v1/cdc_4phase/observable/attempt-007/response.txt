module cdc_4phase #(
  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Internal signals for the four-phase handshake
  // req: source -> destination (request)
  // ack: destination -> source (acknowledge)
  //
  // Four-phase protocol:
  //   Idle: req=0, ack=0
  //   Transfer: req=1, ack=0 (data valid on dst side)
  //   Ack: req=1, ack=1 (dst has accepted, src can deassert req)
  //   Return: req=0, ack=1 (dst deasserts ack, back to idle)
  //
  // We use a 2-bit encoded state for each side to track the phase.
  // Actually, let's use a simpler approach with two 1-bit signals
  // that are synchronized across the clock domain.

  // Source side state
  // src_req: the request signal generated by source
  // src_ack_sync: synchronized version of dst_ack
  //
  // Destination side state
  // dst_req_sync: synchronized version of src_req
  // dst_ack: the acknowledge signal generated by destination

  // Let's use a 2-bit state machine on each side.
  // State encoding:
  //   00: Idle (req=0, ack=0)
  //   01: Request sent (req=1, ack=0)
  //   10: Ack received (req=1, ack=1)
  //   11: Return (req=0, ack=1)

  // Source side
  reg [1:0] src_state;
  reg [WIDTH-1:0] src_data_reg;
  wire src_req;
  wire src_ack_sync;

  // Destination side
  reg [1:0] dst_state;
  reg [WIDTH-1:0] dst_data_reg;
  wire dst_req_sync;
  wire dst_ack;

  // The req signal is driven by source, synchronized to dst
  // The ack signal is driven by dst, synchronized to src

  // Source drives req based on its state
  assign src_req = (src_state == 2'b01) || (src_state == 2'b10);

  // Destination drives ack based on its state
  assign dst_ack = (dst_state == 2'b10) || (dst_state == 2'b11);

  // Synchronize src_req to dst clock domain
  reg [1:0] req_sync_reg;
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni)
      req_sync_reg <= 2'b00;
    else
      req_sync_reg <= {req_sync_reg[0], src_req};
  end
  assign dst_req_sync = req_sync_reg[1];

  // Synchronize dst_ack to src clock domain
  reg [1:0] ack_sync_reg;
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni)
      ack_sync_reg <= 2'b00;
    else
      ack_sync_reg <= {ack_sync_reg[0], dst_ack};
  end
  assign src_ack_sync = ack_sync_reg[1];

  // Source state machine
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_state <= 2'b00;
      src_data_reg <= {WIDTH{1'b0}};
    end else begin
      case (src_state)
        2'b00: begin
          // Idle: wait for valid
          if (src_valid_i) begin
            src_data_reg <= src_data_i;
            src_state <= 2'b01;
          end
        end
        2'b01: begin
          // Request sent: wait for ack
          if (src_ack_sync) begin
            src_state <= 2'b10;
          end
        end
        2'b10: begin
          // Ack received: deassert req
          src_state <= 2'b11;
        end
        2'b11: begin
          // Return: wait for ack to deassert
          if (!src_ack_sync) begin
            src_state <= 2'b00;
          end
        end
        default: src_state <= 2'b00;
      endcase
    end
  end

  // Destination state machine
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_state <= 2'b00;
      dst_data_reg <= {WIDTH{1'b0}};
    end else begin

          // Idle: wait for req
          if (dst_req_sync) begin
            dst_data_reg <= src_data_i; // This is wrong - src_data_i is in src domain
            dst_state <= 2'b01;
          end
        end
        2'b01: begin
          // Req received: assert ack
          dst_state <= 2'b10;
        end
        2'b10: begin
          // Ack asserted: wait for req to deassert
          if (!dst_req_sync) begin
            dst_state <= 2'b11;
          end
        end
        2'b11: begin
          // Return: deassert ack
          dst_state <= 2'b00;
        end
        default: dst_state <= 2'b00;
      endcase
    end
  end

  // Wait, the above approach has issues. The data needs to be transferred
  // across the clock domain as well. Let me reconsider.

  // Actually, in a four-phase handshake, the data is stable while req is high.
  // So we can synchronize the data along with req. But data is WIDTH bits,

  // Let me redo this more carefully.

  // The key insight: in a four-phase handshake, the data is valid and stable
  // while req is high. So we can use a multi-bit synchronizer for the data,
  // but we need to ensure the data is stable before we sample it.

  // A better approach: use the four-phase handshake to control when data
  // is sampled. The data should be synchronized using a pipeline that is
  // only updated when the handshake indicates data is valid.

  // Let me use a different approach:
  // 1. Source asserts req when it has valid data
  // 2. Data is held stable while req is high
  // 3. Destination synchronizes req and data
  // 4. When dst sees req go high, it samples the synchronized data
  // 5. Destination asserts ack
  // 6. Source sees ack, deasserts req
  // 7. Destination sees req deassert, deasserts ack

  // Since data is stable while req is high, we can synchronize the data
  // bits using a 2-stage synchronizer, but we need to ensure we only
  // sample when req is stable high.

  // Actually, the standard approach for four-phase CDC with data is:
  // - Use the handshake to ensure data is stable
  // - Synchronize the data bits along with the req signal
  // - The data is only valid when req is high and stable

  // Source side
  reg [1:0] src_state;
  reg [WIDTH-1:0] src_data_hold;
  wire src_req;

  // Destination side
  reg [1:0] dst_state;
  reg [WIDTH-1:0] dst_data_hold;

  reg [1:0] req_sync;
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni)
      req_sync <= 2'b00;
    else

  end

  reg [1:0] ack_sync;
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni)
      ack_sync <= 2'b00;
    else
      ack_sync <= {ack_sync[0], dst_ack};
  end
  wire src_ack = ack_sync[1];

  // Synchronize data from src to dst
  // Data is stable while req is high, so we can synchronize it
  reg [WIDTH-1:0] data_sync1;
  reg [WIDTH-1:0] data_sync2;
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      data_sync1 <= {WIDTH{1'b0}};
      data_sync2 <= {WIDTH{1'b0}};
    end else begin
      data_sync1 <= src_data_hold;
      data_sync2 <= data_sync1;
    end
  end

  // Source state machine
  assign src_req = (src_state == 2'b01) || (src_state == 2'b10);

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_state <= 2'b00;
      src_data_hold <= {WIDTH{1'b0}};
    end else begin
      case (src_state)
        2'b00: begin
          if (src_valid_i) begin
            src_data_hold <= src_data_i;
            src_state <= 2'b01;
          end
        end
        2'b01: begin
          if (src_ack)
            src_state <= 2'b10;
        end
        2'b10: begin
          src_state <= 2'b11;
        end
        2'b11: begin
          if (!src_ack)
            src_state <= 2'b00;
        end
        default: src_state <= 2'b00;
      endcase
    end
  end

  // Destination state machine
  assign dst_ack = (dst_state == 2'b10) || (dst_state == 2'b11);

  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_state <= 2'b00;
      dst_data_hold <= {WIDTH{1'b0}};
    end else begin
      case (dst_state)
        2'b00: begin
          if (dst_req) begin
            dst_data_hold <= data_sync2;
            dst_state <= 2'b01;
          end
        end
        2'b01: begin
          dst_state <= 2'b10;
        end
        2'b10: begin
          if (!dst_req)
            dst_state <= 2'b11;
        end
        2'b11: begin
          dst_state <= 2'b00;
        end
        default: dst_state <= 2'b00;
      endcase
    end
  end

  // Outputs
  assign src_ready_o = (src_state == 2'b00);
  assign dst_valid_o = (dst_state == 2'b01) || (dst_state == 2'b10);
  assign dst_data_o = dst_data_hold;

endmodule
