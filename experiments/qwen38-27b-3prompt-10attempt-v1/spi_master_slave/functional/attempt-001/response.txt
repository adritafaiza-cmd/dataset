module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

    // State machine states
    localparam S_IDLE    = 2'd0;
    localparam S_WAIT_WR = 2'd1;
    localparam S_SHIFT   = 2'd2;
    localparam S_DONE    = 2'd3;

    reg [1:0] state;
    reg [1:0] state_next;

    // Data registers
    reg [N-1:0] tx_data;
    reg [N-1:0] rx_data;
    reg [N-1:0] tx_shift;
    reg [N-1:0] rx_shift;

    // Bit counter
    reg [$clog2(N)-1:0] bit_cnt;

    // SCLK divider counter
    reg [$clog2(SPI_2X_CLK_DIV)-1:0] clk_div_cnt;
    reg sclk_phase;

    // Handshake signals
    reg wr_ack_int;
    reg do_valid_int;

    // Synchronous reset for pclk domain
    reg [1:0] rst_sync_p;
    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i)
            rst_sync_p <= 2'b11;
        else
            rst_sync_p <= {rst_sync_p[0], 1'b0};
    end
    wire rst_p = rst_sync_p[1];

    // Synchronous reset for sclk domain
    reg [1:0] rst_sync_s;
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i)
            rst_sync_s <= 2'b11;
        else
            rst_sync_s <= {rst_sync_s[0], 1'b0};
    end
    wire rst_s = rst_sync_s[1];

    // PCLK domain: handle write handshake and data loading
    always @(posedge pclk_i or posedge rst_p) begin
        if (rst_p) begin
            wr_ack_o   <= 1'b0;
            di_req_o   <= 1'b0;
            do_valid_o <= 1'b0;
            do_o       <= {N{1'b0}};
            tx_data    <= {N{1'b0}};
            state      <= S_IDLE;
        end else begin
            case (state)
                S_IDLE: begin
                    di_req_o <= 1'b0;
                    wr_ack_o <= 1'b0;
                    do_valid_o <= 1'b0;
                    if (wren_i) begin
                        tx_data <= di_i;
                        state   <= S_WAIT_WR;
                    end
                end
                S_WAIT_WR: begin
                    di_req_o <= 1'b0;
                    wr_ack_o <= 1'b1;
                    do_valid_o <= 1'b0;
                    state <= S_SHIFT;
                end
                S_SHIFT: begin
                    di_req_o <= 1'b0;
                    wr_ack_o <= 1'b0;
                    do_valid_o <= 1'b0;
                    // Wait for shift to complete (signaled from sclk domain)
                    if (do_valid_int) begin
                        do_o       <= rx_data;
                        do_valid_o <= 1'b1;
                        state      <= S_DONE;
                    end
                end
                S_DONE: begin
                    di_req_o <= 1'b1;
                    wr_ack_o <= 1'b0;
                    do_valid_o <= 1'b0;
                    if (wren_i) begin
                        tx_data <= di_i;
                        state   <= S_WAIT_WR;
                    end else begin
                        state <= S_IDLE;
                    end
                end
                default: state <= S_IDLE;
            endcase
        end
    end

    // SCLK domain: SPI shifting logic
    always @(posedge sclk_i or posedge rst_s) begin
        if (rst_s) begin
            spi_ssel_o  <= 1'b0;
            spi_sck_o   <= 1'b0;
            spi_mosi_o  <= 1'b0;
            bit_cnt     <= {($clog2(N)){1'b0}};
            clk_div_cnt <= {($clog2(SPI_2X_CLK_DIV)){1'b0}};
            sclk_phase  <= 1'b0;
            tx_shift    <= {N{1'b0}};
            rx_shift    <= {N{1'b0}};
            rx_data     <= {N{1'b0}};
            do_valid_int <= 1'b0;
        end else begin
            // Default: clear handshake
            do_valid_int <= 1'b0;

            case (state)
                S_IDLE, S_WAIT_WR: begin
                    spi_ssel_o <= 1'b0;
                    spi_sck_o  <= 1'b0;
                    spi_mosi_o <= 1'b0;
                    clk_div_cnt <= {($clog2(SPI_2X_CLK_DIV)){1'b0}};
                    sclk_phase  <= 1'b0;
                    bit_cnt     <= {($clog2(N)){1'b0}};
                end
                S_SHIFT: begin
                    // Load data at start of shift
                    if (bit_cnt == {($clog2(N)){1'b0}} && clk_div_cnt == {($clog2(SPI_2X_CLK_DIV)){1'b0}} && sclk_phase == 1'b0) begin
                        tx_shift <= tx_data;
                        rx_shift <= {N{1'b0}};
                        spi_ssel_o <= 1'b1;
                    end

                    // Clock divider
                    if (clk_div_cnt == SPI_2X_CLK_DIV - 1) begin
                        clk_div_cnt <= {($clog2(SPI_2X_CLK_DIV)){1'b0}};
                        sclk_phase  <= ~sclk_phase;

                        if (sclk_phase == 1'b0) begin
                            // Rising edge of SCK: shift out MSB, capture MISO
                            spi_sck_o <= 1'b1;
                            spi_mosi_o <= tx_shift[N-1];
                            rx_shift <= {spi_miso_i, rx_shift[N-1:1]};
                            tx_shift <= {tx_shift[N-2:0], 1'b0};

                            if (bit_cnt == N - 1) begin
                                // Last bit
                                rx_data <= {spi_miso_i, rx_shift[N-1:1]};
                                do_valid_int <= 1'b1;
                                spi_ssel_o <= 1'b0;
                                spi_sck_o  <= 1'b0;
                                bit_cnt    <= {($clog2(N)){1'b0}};
                            end else begin
                                bit_cnt <= bit_cnt + 1'b1;
                            end
                        end else begin
                            // Falling edge of SCK
                            spi_sck_o <= 1'b0;
                        end
                    end else begin
                        clk_div_cnt <= clk_div_cnt + 1'b1;
                    end
                end
                S_DONE: begin
                    spi_ssel_o <= 1'b0;
                    spi_sck_o  <= 1'b0;
                    spi_mosi_o <= 1'b0;
                end
                default: begin
                    spi_ssel_o <= 1'b0;
                    spi_sck_o  <= 1'b0;
                    spi_mosi_o <= 1'b0;
                end
            endcase
        end
    end

endmodule
