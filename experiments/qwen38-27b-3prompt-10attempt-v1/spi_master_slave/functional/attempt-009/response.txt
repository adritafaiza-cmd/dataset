```verilog
module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

    // State machine states
    localparam S_IDLE    = 2'd0;
    localparam S_WAIT_WR = 2'd1;
    localparam S_SHIFT   = 2'd2;
    localparam S_DONE    = 2'd3;

    reg [1:0] state;
    reg [N-1:0] tx_shift_reg;
    reg [N-1:0] rx_shift_reg;
    reg [N-1:0] bit_count;
    reg [N-1:0] tx_word;
    reg         tx_valid;

    // SCLK divider counter
    reg [3:0] clk_div_cnt;
    reg       clk_div_tick;

    // SCK phase control
    reg       sck_phase;

    // Asynchronous reset for sclk domain
    reg       rst_sclk;
    reg       rst_sclk_d;

    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            rst_sclk <= 1'b1;
            rst_sclk_d <= 1'b1;
        end else begin
            rst_sclk <= 1'b0;
            rst_sclk_d <= 1'b0;
        end
    end

    // Generate tick for SCK generation
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            clk_div_cnt <= 4'd0;
            clk_div_tick <= 1'b0;
        end else begin
            if (clk_div_cnt >= SPI_2X_CLK_DIV - 1) begin
                clk_div_cnt <= 4'd0;
                clk_div_tick <= 1'b1;
            end else begin
                clk_div_cnt <= clk_div_cnt + 4'd1;
                clk_div_tick <= 1'b0;
            end
        end
    end

    // SPI state machine in sclk domain
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            state <= S_IDLE;
            spi_ssel_o <= 1'b0;
            spi_sck_o <= 1'b0;
            tx_shift_reg <= {N{1'b0}};
            rx_shift_reg <= {N{1'b0}};
            bit_count <= {N{1'b0}};
            sck_phase <= 1'b0;
        end else begin
            case (state)
                S_IDLE: begin
                    spi_ssel_o <= 1'b0;
                    spi_sck_o <= 1'b0;
                    if (tx_valid) begin
                        state <= S_SHIFT;
                        spi_ssel_o <= 1'b1;
                        tx_shift_reg <= tx_word;
                        rx_shift_reg <= {N{1'b0}};
                        bit_count <= {N{1'b0}};
                        sck_phase <= 1'b0;
                    end
                end

                S_SHIFT: begin
                    if (clk_div_tick) begin
                        if (sck_phase == 1'b0) begin
                            // Rising edge of SCK
                            spi_sck_o <= 1'b1;
                            sck_phase <= 1'b1;
                            // Capture MISO on rising edge (SPI mode 0)
                            rx_shift_reg <= {spi_miso_i, rx_shift_reg[N-1:1]};
                        end else begin
                            // Falling edge of SCK
                            spi_sck_o <= 1'b0;
                            sck_phase <= 1'b0;
                            // Shift out next bit on falling edge
                            if (bit_count == N - 1) begin
                                state <= S_DONE;
                                spi_ssel_o <= 1'b0;
                            end else begin
                                bit_count <= bit_count + 1'b1;
                                tx_shift_reg <= {1'b0, tx_shift_reg[N-1:1]};
                            end
                        end
                    end
                end

                S_DONE: begin
                    spi_ssel_o <= 1'b0;
                    spi_sck_o <= 1'b0;
                    state <= S_IDLE;
                end

                default: begin
                    state <= S_IDLE;
                    spi_ssel_o <= 1'b0;
                    spi_sck_o <= 1'b0;
                end
            endcase
        end
    end

    // MOSI output
    assign spi_mosi_o = tx_shift_reg[N-1];

    // PCLK domain: handshake and data transfer
    reg [N-1:0] rx_word_pclk;
    reg         rx_valid_pclk;
    reg         rx_valid_pclk_d;

    // Synchronize rx_valid from sclk domain to pclk domain
    reg         rx_valid_sync1;
    reg         rx_valid_sync2;

    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            rx_valid_sync1 <= 1'b0;
            rx_valid_sync2 <= 1'b0;
        end else begin
            rx_valid_sync1 <= rx_valid;
            rx_valid_sync2 <= rx_valid_sync1;
        end
    end

    // Synchronize rx_shift_reg from sclk domain to pclk domain
    reg [N-1:0] rx_shift_reg_sync1;
    reg [N-1:0] rx_shift_reg_sync2;

    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            rx_shift_reg_sync1 <= {N{1'b0}};
            rx_shift_reg_sync2 <= {N{1'b0}};
        end else begin
            rx_shift_reg_sync1 <= rx_shift_reg;
            rx_shift_reg_sync2 <= rx_shift_reg_sync1;
        end
    end

    // Generate rx_valid in sclk domain when shift is complete
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            rx_valid <= 1'b0;
        end else begin
            if (state == S_DONE) begin
                rx_valid <= 1'b1;
            end else begin
                rx_valid <= 1'b0;
            end
        end
    end

    // PCLK domain state machine for parallel interface
    reg [1:0] p_state;
    localparam P_IDLE    = 2'd0;
    localparam P_WAIT_WR = 2'd1;
    localparam P_WAIT_RX = 2'd2;

    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            p_state <= P_IDLE;
            di_req_o <= 1'b0;
            wr_ack_o <= 1'b0;
            do_valid_o <= 1'b0;
            do_o <= {N{1'b0}};
            tx_word <= {N{1'b0}};
            tx_valid <= 1'b0;
        end else begin
            case (p_state)
                P_IDLE: begin
                    di_req_o <= 1'b0;
                    wr_ack_o <= 1'b0;
                    do_valid_o <= 1'b0;
                    if (wren_i) begin
                        p_state <= P_WAIT_WR;
                    end
                end

                P_WAIT_WR: begin
                    if (wren_i) begin
                        tx_word <= di_i;
                        tx_valid <= 1'b1;
                        wr_ack_o <= 1'b1;
                        p_state <= P_WAIT_RX;
                    end else begin
                        p_state <= P_IDLE;
                    end
                end

                P_WAIT_RX: begin
                    wr_ack_o <= 1'b0;
                    if (rx_valid_sync2) begin
                        do_o <= rx_shift_reg_sync2;
                        do_valid_o <= 1'b1;
                        p_state <= P_IDLE;
                    end
                end

                default: begin
                    p_state <= P_IDLE;
                    di_req_o <= 1'b0;
                    wr_ack_o <= 1'b0;
                    do_valid_o <= 1'b0;
                end
            endcase
        end
    end

    // Clear tx_valid after it has been consumed by sclk domain
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            tx_valid <= 1'b0;
        end else begin
            if (state == S_SHIFT && bit_count == 0 && sck_phase == 1'b0) begin
                tx_valid <= 1'b0;
            end
        end
    end

endmodule
