module axi_dma #
(
    parameter AXI_DATA_WIDTH = 32,
    parameter AXI_ADDR_WIDTH = 16,
    parameter AXI_STRB_WIDTH = (AXI_DATA_WIDTH/8),
    parameter AXI_ID_WIDTH = 8,
    parameter AXI_MAX_BURST_LEN = 16,
    parameter AXIS_DATA_WIDTH = AXI_DATA_WIDTH,
    parameter AXIS_KEEP_ENABLE = (AXIS_DATA_WIDTH>8),
    parameter AXIS_KEEP_WIDTH = (AXIS_DATA_WIDTH/8),
    parameter AXIS_LAST_ENABLE = 1,
    parameter AXIS_ID_ENABLE = 0,
    parameter AXIS_ID_WIDTH = 8,
    parameter AXIS_DEST_ENABLE = 0,
    parameter AXIS_DEST_WIDTH = 8,
    parameter AXIS_USER_ENABLE = 1,
    parameter AXIS_USER_WIDTH = 1,
    parameter LEN_WIDTH = 20,
    parameter TAG_WIDTH = 8,
    parameter ENABLE_SG = 0,
    parameter ENABLE_UNALIGNED = 0
)
(
    input  wire                       clk,
    input  wire                       rst,
    /*
     * AXI read descriptor input
     */
    input  wire [AXI_ADDR_WIDTH-1:0]  s_axis_read_desc_addr,
    input  wire [LEN_WIDTH-1:0]       s_axis_read_desc_len,
    input  wire [TAG_WIDTH-1:0]       s_axis_read_desc_tag,
    input  wire [AXIS_ID_WIDTH-1:0]   s_axis_read_desc_id,
    input  wire [AXIS_DEST_WIDTH-1:0] s_axis_read_desc_dest,
    input  wire [AXIS_USER_WIDTH-1:0] s_axis_read_desc_user,
    input  wire                       s_axis_read_desc_valid,
    output wire                       s_axis_read_desc_ready,
    /*
     * AXI read descriptor status output
     */
    output wire [TAG_WIDTH-1:0]       m_axis_read_desc_status_tag,
    output wire [3:0]                 m_axis_read_desc_status_error,
    output wire                       m_axis_read_desc_status_valid,
    /*
     * AXI stream read data output
     */
    output wire [AXIS_DATA_WIDTH-1:0] m_axis_read_data_tdata,
    output wire [AXIS_KEEP_WIDTH-1:0] m_axis_read_data_tkeep,
    output wire                       m_axis_read_data_tvalid,
    input  wire                       m_axis_read_data_tready,
    output wire                       m_axis_read_data_tlast,
    output wire [AXIS_ID_WIDTH-1:0]   m_axis_read_data_tid,
    output wire [AXIS_DEST_WIDTH-1:0] m_axis_read_data_tdest,
    output wire [AXIS_USER_WIDTH-1:0] m_axis_read_data_tuser,
    /*
     * AXI write descriptor input
     */
    input  wire [AXI_ADDR_WIDTH-1:0]  s_axis_write_desc_addr,
    input  wire [LEN_WIDTH-1:0]       s_axis_write_desc_len,
    input  wire [TAG_WIDTH-1:0]       s_axis_write_desc_tag,
    input  wire                       s_axis_write_desc_valid,
    output wire                       s_axis_write_desc_ready,
    /*
     * AXI write descriptor status output
     */
    output wire [LEN_WIDTH-1:0]       m_axis_write_desc_status_len,
    output wire [TAG_WIDTH-1:0]       m_axis_write_desc_status_tag,
    output wire [AXIS_ID_WIDTH-1:0]   m_axis_write_desc_status_id,
    output wire [AXIS_DEST_WIDTH-1:0] m_axis_write_desc_status_dest,
    output wire [AXIS_USER_WIDTH-1:0] m_axis_write_desc_status_user,
    output wire [3:0]                 m_axis_write_desc_status_error,
    output wire                       m_axis_write_desc_status_valid,
    /*
     * AXI stream write data input
     */
    input  wire [AXIS_DATA_WIDTH-1:0] s_axis_write_data_tdata,
    input  wire [AXIS_KEEP_WIDTH-1:0] s_axis_write_data_tkeep,
    input  wire                       s_axis_write_data_tvalid,
    output wire                       s_axis_write_data_tready,
    input  wire                       s_axis_write_data_tlast,
    input  wire [AXIS_ID_WIDTH-1:0]   s_axis_write_data_tid,
    input  wire [AXIS_DEST_WIDTH-1:0] s_axis_write_data_tdest,
    input  wire [AXIS_USER_WIDTH-1:0] s_axis_write_data_tuser,
    /*
     * AXI master interface
     */
    output wire [AXI_ID_WIDTH-1:0]    m_axi_awid,
    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_awaddr,
    output wire [7:0]                 m_axi_awlen,
    output wire [2:0]                 m_axi_awsize,
    output wire [1:0]                 m_axi_awburst,
    output wire                       m_axi_awlock,
    output wire [3:0]                 m_axi_awcache,
    output wire [2:0]                 m_axi_awprot,
    output wire                       m_axi_awvalid,
    input  wire                       m_axi_awready,
    output wire [AXI_DATA_WIDTH-1:0]  m_axi_wdata,
    output wire [AXI_STRB_WIDTH-1:0]  m_axi_wstrb,
    output wire                       m_axi_wlast,
    output wire                       m_axi_wvalid,
    input  wire                       m_axi_wready,
    input  wire [AXI_ID_WIDTH-1:0]    m_axi_bid,
    input  wire [1:0]                 m_axi_bresp,
    input  wire                       m_axi_bvalid,
    output wire                       m_axi_bready,
    output wire [AXI_ID_WIDTH-1:0]    m_axi_arid,
    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_araddr,
    output wire [7:0]                 m_axi_arlen,
    output wire [2:0]                 m_axi_arsize,
    output wire [1:0]                 m_axi_arburst,
    output wire                       m_axi_arlock,
    output wire [3:0]                 m_axi_arcache,
    output wire [2:0]                 m_axi_arprot,
    output wire                       m_axi_arvalid,
    input  wire                       m_axi_arready,
    input  wire [AXI_ID_WIDTH-1:0]    m_axi_rid,
    input  wire [AXI_DATA_WIDTH-1:0]  m_axi_rdata,
    input  wire [1:0]                 m_axi_rresp,
    input  wire                       m_axi_rlast,
    input  wire                       m_axi_rvalid,
    output wire                       m_axi_rready,
    /*
     * Configuration
     */
    input  wire                       read_enable,
    input  wire                       write_enable,
    input  wire                       write_abort
);

    // Local parameters
    localparam AWLEN_WIDTH = 8;
    localparam BURST_LEN_WIDTH = 8;
    localparam DATA_BYTES = AXI_DATA_WIDTH / 8;
    localparam AXI_SIZE = $clog2(DATA_BYTES);
    
    // Read channel state
    reg [AXI_ADDR_WIDTH-1:0] rd_addr;
    reg [LEN_WIDTH-1:0] rd_len;
    reg [TAG_WIDTH-1:0] rd_tag;
    reg [AXIS_ID_WIDTH-1:0] rd_id;
    reg [AXIS_DEST_WIDTH-1:0] rd_dest;
    reg [AXIS_USER_WIDTH-1:0] rd_user;
    reg rd_active;
    reg [LEN_WIDTH-1:0] rd_remaining;
    reg [7:0] rd_burst_remaining;
    reg rd_burst_active;
    reg [AXI_ADDR_WIDTH-1:0] rd_burst_addr;
    reg [7:0] rd_burst_len;
    reg rd_status_valid;
    reg [TAG_WIDTH-1:0] rd_status_tag;
    reg [3:0] rd_status_error;
    
    // Write channel state
    reg [AXI_ADDR_WIDTH-1:0] wr_addr;
    reg [LEN_WIDTH-1:0] wr_len;
    reg [TAG_WIDTH-1:0] wr_tag;
    reg wr_active;
    reg [LEN_WIDTH-1:0] wr_remaining;
    reg [7:0] wr_burst_remaining;
    reg wr_burst_active;
    reg [AXI_ADDR_WIDTH-1:0] wr_burst_addr;
    reg [7:0] wr_burst_len;
    reg wr_status_valid;
    reg [LEN_WIDTH-1:0] wr_status_len;
    reg [TAG_WIDTH-1:0] wr_status_tag;
    reg [AXIS_ID_WIDTH-1:0] wr_status_id;
    reg [AXIS_DEST_WIDTH-1:0] wr_status_dest;
    reg [AXIS_USER_WIDTH-1:0] wr_status_user;
    reg [3:0] wr_status_error;
    
    // FIFO for read data
    reg [AXIS_DATA_WIDTH-1:0] rd_fifo_data [0:15];
    reg [AXIS_KEEP_WIDTH-1:0] rd_fifo_keep [0:15];
    reg [AXIS_ID_WIDTH-1:0] rd_fifo_id [0:15];
    reg [AXIS_DEST_WIDTH-1:0] rd_fifo_dest [0:15];
    reg [AXIS_USER_WIDTH-1:0] rd_fifo_user [0:15];
    reg rd_fifo_last [0:15];
    reg [4:0] rd_fifo_wr_ptr;
    reg [4:0] rd_fifo_rd_ptr;
    reg rd_fifo_empty;
    reg rd_fifo_full;
    
    // FIFO for write data
    reg [AXIS_DATA_WIDTH-1:0] wr_fifo_data [0:15];
    reg [AXIS_KEEP_WIDTH-1:0] wr_fifo_keep [0:15];
    reg [AXIS_ID_WIDTH-1:0] wr_fifo_id [0:15];
    reg [AXIS_DEST_WIDTH-1:0] wr_fifo_dest [0:15];
    reg [AXIS_USER_WIDTH-1:0] wr_fifo_user [0:15];
    reg wr_fifo_last [0:15];
    reg [4:0] wr_fifo_wr_ptr;
    reg [4:0] wr_fifo_rd_ptr;
    reg wr_fifo_empty;
    reg wr_fifo_full;
    
    // AXI write address channel
    reg aw_valid;
    reg [AXI_ID_WIDTH-1:0] aw_id;
    reg [AXI_ADDR_WIDTH-1:0] aw_addr;
    reg [7:0] aw_len;
    reg [2:0] aw_size;
    reg [1:0] aw_burst;
    reg aw_lock;
    reg [3:0] aw_cache;
    reg [2:0] aw_prot;
    
    // AXI write data channel
    reg w_valid;
    reg [AXI_DATA_WIDTH-1:0] w_data;
    reg [AXI_STRB_WIDTH-1:0] w_strb;
    reg w_last;
    
    // AXI write response channel
    reg b_ready;
    
    // AXI read address channel
    reg ar_valid;
    reg [AXI_ID_WIDTH-1:0] ar_id;
    reg [AXI_ADDR_WIDTH-1:0] ar_addr;
    reg [7:0] ar_len;
    reg [2:0] ar_size;
    reg [1:0] ar_burst;
    reg ar_lock;
    reg [3:0] ar_cache;
    reg [2:0] ar_prot;
    
    // AXI read data channel
    reg r_ready;
    
    // Helper functions
    function [7:0] calc_burst_len;
        input [LEN_WIDTH-1:0] remaining;
        begin
            if (remaining == 0)
                calc_burst_len = 0;
            else if (remaining > AXI_MAX_BURST_LEN)
                calc_burst_len = AXI_MAX_BURST_LEN - 1;
            else
                calc_burst_len = remaining - 1;
        end
    endfunction
    
    // Read descriptor ready
    assign s_axis_read_desc_ready = read_enable && !rd_active && !rd_burst_active;
    
    // Write descriptor ready
    assign s_axis_write_desc_ready = write_enable && !wr_active && !wr_burst_active && !write_abort;
    
    // Read status output
    assign m_axis_read_desc_status_tag = rd_status_tag;
    assign m_axis_read_desc_status_error = rd_status_error;
    assign m_axis_read_desc_status_valid = rd_status_valid;
    
    // Write status output
    assign m_axis_write_desc_status_len = wr_status_len;
    assign m_axis_write_desc_status_tag = wr_status_tag;
    assign m_axis_write_desc_status_id = wr_status_id;
    assign m_axis_write_desc_status_dest = wr_status_dest;
    assign m_axis_write_desc_status_user = wr_status_user;
    assign m_axis_write_desc_status_error = wr_status_error;
    assign m_axis_write_desc_status_valid = wr_status_valid;
    
    // Read data output
    assign m_axis_read_data_tdata = rd_fifo_data[rd_fifo_rd_ptr];
    assign m_axis_read_data_tkeep = rd_fifo_keep[rd_fifo_rd_ptr];
    assign m_axis_read_data_tvalid = !rd_fifo_empty;
    assign m_axis_read_data_tlast = rd_fifo_last[rd_fifo_rd_ptr];
    assign m_axis_read_data_tid = rd_fifo_id[rd_fifo_rd_ptr];
    assign m_axis_read_data_tdest = rd_fifo_dest[rd_fifo_rd_ptr];
    assign m_axis_read_data_tuser = rd_fifo_user[rd_fifo_rd_ptr];
    
    // Write data ready
    assign s_axis_write_data_tready = !wr_fifo_full && wr_active && !write_abort;
    
    // AXI AW channel
    assign m_axi_awid = aw_id;
    assign m_axi_awaddr = aw_addr;
    assign m_axi_awlen = aw_len;
    assign m_axi_awsize = aw_size;
    assign m_axi_awburst = aw_burst;
    assign m_axi_awlock = aw_lock;
    assign m_axi_awcache = aw_cache;
    assign m_axi_awprot = aw_prot;
    assign m_axi_awvalid = aw_valid;
    
    // AXI W channel
    assign m_axi_wdata = w_data;
    assign m_axi_wstrb = w_strb;
    assign m_axi_wlast = w_last;
    assign m_axi_wvalid = w_valid;
    
    // AXI B channel
    assign m_axi_bready = b_ready;
    
    // AXI AR channel
    assign m_axi_arid = ar_id;
    assign m_axi_araddr = ar_addr;
    assign m_axi_arlen = ar_len;
    assign m_axi_arsize = ar_size;
    assign m_axi_arburst = ar_burst;
    assign m_axi_arlock = ar_lock;
    assign m_axi_arcache = ar_cache;
    assign m_axi_arprot = ar_prot;
    assign m_axi_arvalid = ar_valid;
    
    // AXI R channel
    assign m_axi_rready = r_ready;
    
    // FIFO empty/full logic
    always @(*) begin
        rd_fifo_empty = (rd_fifo_wr_ptr == rd_fifo_rd_ptr);
        rd_fifo_full = (rd_fifo_wr_ptr[4] != rd_fifo_rd_ptr[4]) && 
                       (rd_fifo_wr_ptr[3:0] == rd_fifo_rd_ptr[3:0]);
        
        wr_fifo_empty = (wr_fifo_wr_ptr == wr_fifo_rd_ptr);
        wr_fifo_full = (wr_fifo_wr_ptr[4] != wr_fifo_rd_ptr[4]) && 
                       (wr_fifo_wr_ptr[3:0] == wr_fifo_rd_ptr[3:0]);
    end
    
    // Read descriptor acceptance
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rd_active <= 1'b0;
            rd_burst_active <= 1'b0;
            rd_addr <= {AXI_ADDR_WIDTH{1'b0}};
            rd_len <= {LEN_WIDTH{1'b0}};
            rd_tag <= {TAG_WIDTH{1'b0}};
            rd_id <= {AXIS_ID_WIDTH{1'b0}};
            rd_dest <= {AXIS_DEST_WIDTH{1'b0}};
            rd_user <= {AXIS_USER_WIDTH{1'b0}};
            rd_remaining <= {LEN_WIDTH{1'b0}};
            rd_burst_remaining <= 8'd0;
            rd_burst_addr <= {AXI_ADDR_WIDTH{1'b0}};
            rd_burst_len <= 8'd0;
            rd_status_valid <= 1'b0;
            rd_status_tag <= {TAG_WIDTH{1'b0}};
            rd_status_error <= 4'd0;
            rd_fifo_wr_ptr <= 5'd0;
            rd_fifo_rd_ptr <= 5'd0;
            ar_valid <= 1'b0;
            ar_id <= {AXI_ID_WIDTH{1'b0}};
            ar_addr <= {AXI_ADDR_WIDTH{1'b0}};
            ar_len <= 8'd0;
            ar_size <= 3'd0;
            ar_burst <= 2'd0;
            ar_lock <= 1'b0;
            ar_cache <= 4'd0;
            ar_prot <= 3'd0;
            r_ready <= 1'b0;
        end else begin
            // Accept read descriptor
            if (s_axis_read_desc_valid && s_axis_read_desc_ready) begin
                rd_active <= 1'b1;
                rd_addr <= s_axis_read_desc_addr;
                rd_len <= s_axis_read_desc_len;
                rd_tag <= s_axis_read_desc_tag;
                rd_id <= s_axis_read_desc_id;
                rd_dest <= s_axis_read_desc_dest;
                rd_user <= s_axis_read_desc_user;
                rd_remaining <= s_axis_read_desc_len;
                rd_burst_active <= 1'b0;
                rd_status_valid <= 1'b0;
            end
            
            // Start read burst
            if (rd_active && !rd_burst_active && rd_remaining > 0 && m_axi_arready) begin
                rd_burst_active <= 1'b1;
                rd_burst_addr <= rd_addr + (rd_len - rd_remaining) * DATA_BYTES;
                rd_burst_len <= calc_burst_len(rd_remaining);
                rd_burst_remaining <= rd_burst_len + 1;
                ar_valid <= 1'b1;
                ar_id <= rd_tag[AXI_ID_WIDTH-1:0];
                ar_addr <= rd_burst_addr;
                ar_len <= rd_burst_len;
                ar_size <= AXI_SIZE[2:0];
                ar_burst <= 2'b01; // INCR
                ar_lock <= 1'b0;
                ar_cache <= 4'b0011;
                ar_prot <= 3'b000;
                r_ready <= 1'b1;
            end
            
            // Complete read burst
            if (rd_burst_active && m_axi_rvalid && m_axi_rlast && m_axi_rready) begin
                rd_burst_active <= 1'b0;
                rd_remaining <= rd_remaining - (rd_burst_len + 1);
                ar_valid <= 1'b0;
                r_ready <= 1'b0;
                
                if (rd_remaining == 0) begin
                    rd_active <= 1'b0;
                    rd_status_valid <= 1'b1;
                    rd_status_tag <= rd_tag;
                    rd_status_error <= 4'd0;
                end
            end
            
            // Clear status when consumed
            if (rd_status_valid) begin
                rd_status_valid <= 1'b0;
            end
        end
    end
    
    // Read data FIFO write
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rd_fifo_wr_ptr <= 5'd0;
        end else if (rd_burst_active && m_axi_rvalid && m_axi_rready) begin
            rd_fifo_data[rd_fifo_wr_ptr] <= m_axi_rdata;
            rd_fifo_keep[rd_fifo_wr_ptr] <= {AXIS_KEEP_WIDTH{1'b1}};
            rd_fifo_id[rd_fifo_wr_ptr] <= rd_id;
            rd_fifo_dest[rd_fifo_wr_ptr] <= rd_dest;
            rd_fifo_user[rd_fifo_wr_ptr] <= rd_user;
            rd_fifo_last[rd_fifo_wr_ptr] <= m_axi_rlast;
            rd_fifo_wr_ptr <= rd_fifo_wr_ptr + 1'b1;
        end
    end
    
    // Read data FIFO read
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rd_fifo_rd_ptr <= 5'd0;
        end else if (!rd_fifo_empty && m_axis_read_data_tready) begin
            rd_fifo_rd_ptr <= rd_fifo_rd_ptr + 1'b1;
        end
    end
    
    // Write descriptor acceptance
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_active <= 1'b0;
            wr_burst_active <= 1'b0;
            wr_addr <= {AXI_ADDR_WIDTH{1'b0}};
            wr_len <= {LEN_WIDTH{1'b0}};
            wr_tag <= {TAG_WIDTH{1'b0}};
            wr_remaining <= {LEN_WIDTH{1'b0}};
            wr_burst_remaining <= 8'd0;
            wr_burst_addr <= {AXI_ADDR_WIDTH{1'b0}};
            wr_burst_len <= 8'd0;
            wr_status_valid <= 1'b0;
            wr_status_len <= {LEN_WIDTH{1'b0}};
            wr_status_tag <= {TAG_WIDTH{1'b0}};
            wr_status_id <= {AXIS_ID_WIDTH{1'b0}};
            wr_status_dest <= {AXIS_DEST_WIDTH{1'b0}};
            wr_status_user <= {AXIS_USER_WIDTH{1'b0}};
            wr_status_error <= 4'd0;
            wr_fifo_wr_ptr <= 5'd0;
            wr_fifo_rd_ptr <= 5'd0;
            aw_valid <= 1'b0;
            aw_id <= {AXI_ID_WIDTH{1'b0}};
            aw_addr <= {AXI_ADDR_WIDTH{1'b0}};
            aw_len <= 8'd0;
            aw_size <= 3'd0;
            aw_burst <= 2'd0;
            aw_lock <= 1'b0;
            aw_cache <= 4'd0;
            aw_prot <= 3'd0;
            w_valid <= 1'b0;
            w_data <= {AXI_DATA_WIDTH{1'b0}};
            w_strb <= {AXI_STRB_WIDTH{1'b0}};
            w_last <= 1'b0;
            b_ready <= 1'b0;
        end else begin
            // Accept write descriptor
            if (s_axis_write_desc_valid && s_axis_write_desc_ready) begin
                wr_active <= 1'b1;
                wr_addr <= s_axis_write_desc_addr;
                wr_len <= s_axis_write_desc_len;
                wr_tag <= s_axis_write_desc_tag;
                wr_remaining <= s_axis_write_desc_len;
                wr_burst_active <= 1'b0;
                wr_status_valid <= 1'b0;
            end
            
            // Start write burst
            if (wr_active && !wr_burst_active && wr_remaining > 0 && !wr_fifo_empty && m_axi_awready) begin
                wr_burst_active <= 1'b1;
                wr_burst_addr <= wr_addr + (wr_len - wr_remaining) * DATA_BYTES;
                wr_burst_len <= calc_burst_len(wr_remaining);
                wr_burst_remaining <= wr_burst_len + 1;
                aw_valid <= 1'b1;
                aw_id <= wr_tag[AXI_ID_WIDTH-1:0];
                aw_addr <= wr_burst_addr;
                aw_len <= wr_burst_len;
                aw_size <= AXI_SIZE[2:0];
                aw_burst <= 2'b01; // INCR
                aw_lock <= 1'b0;
                aw_cache <= 4'b0011;
                aw_prot <= 3'b000;
                w_valid <= 1'b1;
                b_ready <= 1'b1;
            end
            
            // Complete write burst
            if (wr_burst_active && m_axi_bvalid && m_axi_bready) begin
                wr_burst_active <= 1'b0;
                wr_remaining <= wr_remaining - (wr_burst_len + 1);
                aw_valid <= 1'b0;
                w_valid <= 1'b0;
                b_ready <= 1'b0;
                
                if (wr_remaining == 0) begin
                    wr_active <= 1'b0;
                    wr_status_valid <= 1'b1;
                    wr_status_len <= wr_len;
                    wr_status_tag <= wr_tag;
                    wr_status_id <= {AXIS_ID_WIDTH{1'b0}};
                    wr_status_dest <= {AXIS_DEST_WIDTH{1'b0}};
                    wr_status_user <= {AXIS_USER_WIDTH{1'b0}};
                    wr_status_error <= 4'd0;
                end
            end
            
            // Clear status when consumed
            if (wr_status_valid) begin
                wr_status_valid <= 1'b0;
            end
        end
    end
    
    // Write data FIFO write
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_fifo_wr_ptr <= 5'd0;
        end else if (s_axis_write_data_tvalid && s_axis_write_data_tready) begin
            wr_fifo_data[wr_fifo_wr_ptr] <= s_axis_write_data_tdata;
            wr_fifo_keep[wr_fifo_wr_ptr] <= s_axis_write_data_tkeep;
            wr_fifo_id[wr_fifo_wr_ptr] <= s_axis_write_data_tid;
            wr_fifo_dest[wr_fifo_wr_ptr] <= s_axis_write_data_tdest;
            wr_fifo_user[wr_fifo_wr_ptr] <= s_axis_write_data_tuser;
            wr_fifo_last[wr_fifo_wr_ptr] <= s_axis_write_data_tlast;
            wr_fifo_wr_ptr <= wr_fifo_wr_ptr + 1'b1;
        end
    end
    
    // Write data FIFO read and AXI W channel
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_fifo_rd_ptr <= 5'd0;
            w_data <= {AXI_DATA_WIDTH{1'b0}};
            w_strb <= {AXI_STRB_WIDTH{1'b0}};
            w_last <= 1'b0;
