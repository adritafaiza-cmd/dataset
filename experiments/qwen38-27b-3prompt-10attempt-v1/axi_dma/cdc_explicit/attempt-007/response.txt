    output wire [LEN_WIDTH-1:0]       m_axis_write_desc_status_len,

    output wire [AXIS_USER_WIDTH-1:0] m_axis_write_desc_status_user,
    output wire [3:0]                 m_axis_write_desc_status_error,
    output wire                       m_axis_write_desc_status_valid,
    /*
     * AXI stream write data input
     */
    input  wire [AXIS_DATA_WIDTH-1:0] s_axis_write_data_tdata,
    input  wire [AXIS_KEEP_WIDTH-1:0] s_axis_write_data_tkeep,

    output wire                       s_axis_write_data_tready,
    input  wire                       s_axis_write_data_tlast,

    input  wire [AXIS_USER_WIDTH-1:0] s_axis_write_data_tuser,
    /*
     * AXI master interface
     */
    output wire [AXI_ID_WIDTH-1:0]    m_axi_awid,
    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_awaddr,
    output wire [7:0]                 m_axi_awlen,
    output wire [2:0]                 m_axi_awsize,
    output wire [1:0]                 m_axi_awburst,
    output wire                       m_axi_awlock,
    output wire [3:0]                 m_axi_awcache,
    output wire [2:0]                 m_axi_awprot,
    output wire                       m_axi_awvalid,
    input  wire                       m_axi_awready,
    output wire [AXI_DATA_WIDTH-1:0]  m_axi_wdata,
    output wire [AXI_STRB_WIDTH-1:0]  m_axi_wstrb,
    output wire                       m_axi_wlast,
    output wire                       m_axi_wvalid,
    input  wire                       m_axi_wready,
    input  wire [AXI_ID_WIDTH-1:0]    m_axi_bid,
    input  wire [1:0]                 m_axi_bresp,
    input  wire                       m_axi_bvalid,
    output wire                       m_axi_bready,
    output wire [AXI_ID_WIDTH-1:0]    m_axi_arid,
    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_araddr,
    output wire [7:0]                 m_axi_arlen,
    output wire [2:0]                 m_axi_arsize,
    output wire [1:0]                 m_axi_arburst,
    output wire                       m_axi_arlock,
    output wire [3:0]                 m_axi_arcache,
    output wire [2:0]                 m_axi_arprot,
    output wire                       m_axi_arvalid,
    input  wire                       m_axi_arready,
    input  wire [AXI_ID_WIDTH-1:0]    m_axi_rid,
    input  wire [AXI_DATA_WIDTH-1:0]  m_axi_rdata,
    input  wire [1:0]                 m_axi_rresp,
    input  wire                       m_axi_rlast,
    input  wire                       m_axi_rvalid,
    output wire                       m_axi_rready,
    /*
     * Configuration
     */
    input  wire                       read_enable,
    input  wire                       write_enable,
    input  wire                       write_abort
);

    // Local parameters
    localparam DATA_BYTES = AXI_DATA_WIDTH / 8;
    localparam AWLEN_MAX  = AXI_MAX_BURST_LEN - 1;
    localparam AWLEN_WIDTH = 8;
    localparam AWLEN_MASK  = (1 << AWLEN_WIDTH) - 1;

    // Read channel state
    reg [AXI_ADDR_WIDTH-1:0] rd_addr;
    reg [LEN_WIDTH-1:0]      rd_len;
    reg [TAG_WIDTH-1:0]      rd_tag;
    reg [AXIS_ID_WIDTH-1:0]  rd_id;
    reg [AXIS_DEST_WIDTH-1:0] rd_dest;
    reg [AXIS_USER_WIDTH-1:0] rd_user;
    reg                      rd_active;
    reg [LEN_WIDTH-1:0]      rd_remaining;
    reg [AWLEN_WIDTH-1:0]    rd_burst_len;
    reg [AXI_ADDR_WIDTH-1:0] rd_burst_addr;
    reg                      rd_burst_pending;
    reg [3:0]                rd_error;

    // Write channel state
    reg [AXI_ADDR_WIDTH-1:0] wr_addr;
    reg [LEN_WIDTH-1:0]      wr_len;
    reg [TAG_WIDTH-1:0]      wr_tag;
    reg                      wr_active;
    reg [LEN_WIDTH-1:0]      wr_remaining;
    reg [AWLEN_WIDTH-1:0]    wr_burst_len;
    reg [AXI_ADDR_WIDTH-1:0] wr_burst_addr;
    reg                      wr_burst_pending;
    reg [3:0]                wr_error;
    reg [AXIS_ID_WIDTH-1:0]  wr_id;
    reg [AXIS_DEST_WIDTH-1:0] wr_dest;
    reg [AXIS_USER_WIDTH-1:0] wr_user;

    // Read data FIFO (simple register-based buffer)
    reg [AXIS_DATA_WIDTH-1:0] rd_data_buf [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_KEEP_WIDTH-1:0] rd_keep_buf [0:AXI_MAX_BURST_LEN-1];
    reg                       rd_last_buf [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_ID_WIDTH-1:0]   rd_id_buf   [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_DEST_WIDTH-1:0] rd_dest_buf [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_USER_WIDTH-1:0] rd_user_buf [0:AXI_MAX_BURST_LEN-1];
    reg [AWLEN_WIDTH:0]       rd_wr_ptr;
    reg [AWLEN_WIDTH:0]       rd_rd_ptr;
    reg                       rd_fifo_empty;
    reg                       rd_fifo_full;

    // Write data FIFO
    reg [AXI_DATA_WIDTH-1:0]  wr_data_buf [0:AXI_MAX_BURST_LEN-1];
    reg [AXI_STRB_WIDTH-1:0]  wr_strb_buf [0:AXI_MAX_BURST_LEN-1];
    reg                       wr_last_buf [0:AXI_MAX_BURST_LEN-1];
    reg [AWLEN_WIDTH:0]       wr_wr_ptr;
    reg [AWLEN_WIDTH:0]       wr_rd_ptr;
    reg                       wr_fifo_empty;
    reg                       wr_fifo_full;

    // AXI read address channel
    reg                      ar_valid;
    reg [AXI_ID_WIDTH-1:0]   ar_id;
    reg [AXI_ADDR_WIDTH-1:0] ar_addr;
    reg [AWLEN_WIDTH-1:0]    ar_len;
    reg [2:0]                ar_size;
    reg [1:0]                ar_burst;
    reg                      ar_lock;
    reg [3:0]                ar_cache;
    reg [2:0]                ar_prot;

    // AXI read data channel
    reg                      r_ready;

    // AXI write address channel
    reg                      aw_valid;
    reg [AXI_ID_WIDTH-1:0]   aw_id;
    reg [AXI_ADDR_WIDTH-1:0] aw_addr;
    reg [AWLEN_WIDTH-1:0]    aw_len;
    reg [2:0]                aw_size;
    reg [1:0]                aw_burst;
    reg                      aw_lock;
    reg [3:0]                aw_cache;
    reg [2:0]                aw_prot;

    // AXI write data channel
    reg                      w_valid;
    reg [AXI_DATA_WIDTH-1:0] w_data;
    reg [AXI_STRB_WIDTH-1:0] w_strb;
    reg                      w_last;

    // AXI write response channel
    reg                      b_ready;

    // Descriptor status
    reg [TAG_WIDTH-1:0]      rd_status_tag;
    reg [3:0]                rd_status_error;
    reg                      rd_status_valid;
    reg [LEN_WIDTH-1:0]      wr_status_len;
    reg [TAG_WIDTH-1:0]      wr_status_tag;
    reg [AXIS_ID_WIDTH-1:0]  wr_status_id;
    reg [AXIS_DEST_WIDTH-1:0] wr_status_dest;
    reg [AXIS_USER_WIDTH-1:0] wr_status_user;
    reg [3:0]                wr_status_error;
    reg                      wr_status_valid;

    // Helper signals
    wire rd_desc_accept = s_axis_read_desc_valid && s_axis_read_desc_ready;
    wire wr_desc_accept = s_axis_write_desc_valid && s_axis_write_desc_ready;

    // Read descriptor ready
    assign s_axis_read_desc_ready = read_enable && !rd_active && !rd_burst_pending;

    // Write descriptor ready
    assign s_axis_write_desc_ready = write_enable && !wr_active && !wr_burst_pending && !write_abort;

    // Read data output
    assign m_axis_read_data_tdata = rd_data_buf[rd_rd_ptr[AWLEN_WIDTH-1:0]];
    assign m_axis_read_data_tkeep = rd_keep_buf[rd_rd_ptr[AWLEN_WIDTH-1:0]];
    assign m_axis_read_data_tvalid = !rd_fifo_empty;
    assign m_axis_read_data_tlast = rd_last_buf[rd_rd_ptr[AWLEN_WIDTH-1:0]];
    assign m_axis_read_data_tid = rd_id_buf[rd_rd_ptr[AWLEN_WIDTH-1:0]];
    assign m_axis_read_data_tdest = rd_dest_buf[rd_rd_ptr[AWLEN_WIDTH-1:0]];
    assign m_axis_read_data_tuser = rd_user_buf[rd_rd_ptr[AWLEN_WIDTH-1:0]];

    // Write data input ready
    assign s_axis_write_data_tready = write_enable && wr_active && !wr_fifo_full && !write_abort;

    // AXI read address
    assign m_axi_arid = ar_id;
    assign m_axi_araddr = ar_addr;
    assign m_axi_arlen = ar_len;
    assign m_axi_arsize = ar_size;
    assign m_axi_arburst = ar_burst;
    assign m_axi_arlock = ar_lock;
    assign m_axi_arcache = ar_cache;
    assign m_axi_arprot = ar_prot;
    assign m_axi_arvalid = ar_valid;

    // AXI read data
    assign m_axi_rready = r_ready;

    // AXI write address
    assign m_axi_awid = aw_id;
    assign m_axi_awaddr = aw_addr;
    assign m_axi_awlen = aw_len;
    assign m_axi_awsize = aw_size;
    assign m_axi_awburst = aw_burst;
    assign m_axi_awlock = aw_lock;
    assign m_axi_awcache = aw_cache;
    assign m_axi_awprot = aw_prot;
    assign m_axi_awvalid = aw_valid;

    // AXI write data
    assign m_axi_wdata = w_data;
    assign m_axi_wstrb = w_strb;
    assign m_axi_wlast = w_last;
    assign m_axi_wvalid = w_valid;

    // AXI write response
    assign m_axi_bready = b_ready;

    // Descriptor status
    assign m_axis_read_desc_status_tag = rd_status_tag;
    assign m_axis_read_desc_status_error = rd_status_error;
    assign m_axis_read_desc_status_valid = rd_status_valid;

    assign m_axis_write_desc_status_len = wr_status_len;
    assign m_axis_write_desc_status_tag = wr_status_tag;
    assign m_axis_write_desc_status_id = wr_status_id;
    assign m_axis_write_desc_status_dest = wr_status_dest;
    assign m_axis_write_desc_status_user = wr_status_user;
    assign m_axis_write_desc_status_error = wr_status_error;
    assign m_axis_write_desc_status_valid = wr_status_valid;

    // FIFO empty/full
    assign rd_fifo_empty = (rd_wr_ptr == rd_rd_ptr);
    assign rd_fifo_full = (rd_wr_ptr[AWLEN_WIDTH] != rd_rd_ptr[AWLEN_WIDTH]) && 
                          (rd_wr_ptr[AWLEN_WIDTH-1:0] == rd_rd_ptr[AWLEN_WIDTH-1:0]);

    assign wr_fifo_empty = (wr_wr_ptr == wr_rd_ptr);
    assign wr_fifo_full = (wr_wr_ptr[AWLEN_WIDTH] != wr_rd_ptr[AWLEN_WIDTH]) && 
                          (wr_wr_ptr[AWLEN_WIDTH-1:0] == wr_rd_ptr[AWLEN_WIDTH-1:0]);

    // Read data FIFO write pointer increment
    wire rd_fifo_wr_en = m_axi_rvalid && m_axi_rready;
    wire rd_fifo_rd_en = m_axis_read_data_tvalid && m_axis_read_data_tready;

    // Write data FIFO write pointer increment
    wire wr_fifo_wr_en = s_axis_write_data_tvalid && s_axis_write_data_tready;
    wire wr_fifo_rd_en = m_axi_wvalid && m_axi_wready;

    // AXI read address handshake
    wire ar_handshake = ar_valid && m_axi_arready;

    // AXI write address handshake
    wire aw_handshake = aw_valid && m_axi_awready;

    // AXI write response handshake
    wire b_handshake = m_axi_bvalid && m_axi_bready;

    // Calculate burst length
    function [AWLEN_WIDTH-1:0] calc_burst_len;
        input [LEN_WIDTH-1:0] remaining;
        begin
            if (remaining > AWLEN_MAX)
                calc_burst_len = AWLEN_MAX;
            else
                calc_burst_len = remaining[AWLEN_WIDTH-1:0];
        end
    endfunction

    // Read channel logic
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rd_addr <= {AXI_ADDR_WIDTH{1'b0}};
            rd_len <= {LEN_WIDTH{1'b0}};
            rd_tag <= {TAG_WIDTH{1'b0}};
            rd_id <= {AXIS_ID_WIDTH{1'b0}};
            rd_dest <= {AXIS_DEST_WIDTH{1'b0}};
            rd_user <= {AXIS_USER_WIDTH{1'b0}};
            rd_active <= 1'b0;
            rd_remaining <= {LEN_WIDTH{1'b0}};
            rd_burst_len <= {AWLEN_WIDTH{1'b0}};
            rd_burst_addr <= {AXI_ADDR_WIDTH{1'b0}};
            rd_burst_pending <= 1'b0;
            rd_error <= 4'b0000;
            rd_wr_ptr <= {(AWLEN_WIDTH+1){1'b0}};
            rd_rd_ptr <= {(AWLEN_WIDTH+1){1'b0}};
            ar_valid <= 1'b0;
            ar_id <= {AXI_ID_WIDTH{1'b0}};
            ar_addr <= {AXI_ADDR_WIDTH{1'b0}};
            ar_len <= {AWLEN_WIDTH{1'b0}};
            ar_size <= 3'b000;
            ar_burst <= 2'b01;
            ar_lock <= 1'b0;
            ar_cache <= 4'b0000;
            ar_prot <= 3'b000;
            r_ready <= 1'b0;
            rd_status_tag <= {TAG_WIDTH{1'b0}};
            rd_status_error <= 4'b0000;
            rd_status_valid <= 1'b0;
        end else begin
            // Default
            rd_status_valid <= 1'b0;

            // Accept read descriptor
            if (rd_desc_accept) begin
                rd_addr <= s_axis_read_desc_addr;
                rd_len <= s_axis_read_desc_len;
                rd_tag <= s_axis_read_desc_tag;
                rd_id <= s_axis_read_desc_id;
                rd_dest <= s_axis_read_desc_dest;
                rd_user <= s_axis_read_desc_user;
                rd_active <= 1'b1;
                rd_remaining <= s_axis_read_desc_len;
                rd_error <= 4'b0000;
                rd_burst_pending <= 1'b0;
            end

            // Issue read address
            if (rd_active && !rd_burst_pending && !ar_valid) begin
                if (rd_remaining > 0) begin
                    rd_burst_len <= calc_burst_len(rd_remaining);
                    rd_burst_addr <= rd_addr;
                    ar_valid <= 1'b1;
                    ar_id <= rd_tag[AXI_ID_WIDTH-1:0];
                    ar_addr <= rd_addr;
                    ar_len <= calc_burst_len(rd_remaining) - 1;
                    ar_size <= $clog2(DATA_BYTES) - 1;
                    ar_burst <= 2'b01; // INCR
                    ar_lock <= 1'b0;
                    ar_cache <= 4'b0000;
                    ar_prot <= 3'b000;
                    rd_burst_pending <= 1'b1;
                end else begin
                    // Done
                    rd_active <= 1'b0;
                    rd_status_tag <= rd_tag;
                    rd_status_error <= rd_error;
                    rd_status_valid <= 1'b1;
                end
            end

            // AXI read address handshake
            if (ar_handshake) begin
                ar_valid <= 1'b0;
            end

            // AXI read data
            if (rd_burst_pending) begin
                r_ready <= 1'b1;
            end

            if (rd_fifo_wr_en) begin
                rd_data_buf[rd_wr_ptr[AWLEN_WIDTH-1:0]] <= m_axi_rdata;
                rd_keep_buf[rd_wr_ptr[AWLEN_WIDTH-1:0]] <= {AXIS_KEEP_WIDTH{1'b1}};
                rd_last_buf[rd_wr_ptr[AWLEN_WIDTH-1:0]] <= m_axi_rlast;
                rd_id_buf[rd_wr_ptr[AWLEN_WIDTH-1:0]] <= rd_id;
                rd_dest_buf[rd_wr_ptr[AWLEN_WIDTH-1:0]] <= rd_dest;
                rd_user_buf[rd_wr_ptr[AWLEN_WIDTH-1:0]] <= rd_user;
                rd_wr_ptr <= rd_wr_ptr + 1;
            end

            if (rd_fifo_rd_en) begin
                rd_rd_ptr <= rd_rd_ptr + 1;
            end

            // Check for burst completion
            if (rd_fifo_wr_en && m_axi_rlast) begin
                rd_burst_pending <= 1'b0;
                rd_addr <= rd_addr + (rd_burst_len * DATA_BYTES);
                rd_remaining <= rd_remaining - rd_burst_len;
                r_ready <= 1'b0;
            end

            // Check for read completion
            if (rd_active && !rd_burst_pending && rd_remaining == 0 && rd_fifo_empty) begin
                rd_active <= 1'b0;
                rd_status_tag <= rd_tag;
                rd_status_error <= rd_error;
                rd_status_valid <= 1'b1;
            end
        end
    end

    // Write channel logic
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_addr <= {AXI_ADDR_WIDTH{1'b0}};
            wr_len <= {LEN_WIDTH{1'b0}};
            wr_tag <= {TAG_WIDTH{1'b0}};
            wr_active <= 1'b0;
            wr_remaining <= {LEN_WIDTH{1'b0}};
            wr_burst_len <= {AWLEN_WIDTH{1'b0}};
            wr_burst_addr <= {AXI_ADDR_WIDTH{1'b0}};
            wr_burst_pending <= 1'b0;
            wr_error <= 4'b0000;
            wr_id <= {AXIS_ID_WIDTH{1'b0}};
            wr_dest <= {AXIS_DEST_WIDTH{1'b0}};
            wr_user <= {AXIS_USER_WIDTH{1'b0}};
            wr_wr_ptr <= {(AWLEN_WIDTH+1){1'b0}};
            wr_rd_ptr <= {(AWLEN_WIDTH+1){1'b0}};
            aw_valid <= 1'b0;
            aw_id <= {AXI_ID_WIDTH{1'b0}};
            aw_addr <= {AXI_ADDR_WIDTH{1'b0}};
            aw_len <= {AWLEN_WIDTH{1'b0}};
            aw_size <= 3'b000;
            aw_burst <= 2'b01;
            aw_lock <= 1'b0;
            aw_cache <= 4'b0000;
            aw_prot <= 3'b000;
            w_valid <= 1'b0;
            w_data <= {AXI_DATA_WIDTH{1'b0}};
            w_strb <= {AXI_STRB_WIDTH{1'b0}};
            w_last <= 1'b0;
            b_ready <= 1'b0;
            wr_status_len <= {LEN_WIDTH{1'b0}};
            wr_status_tag <= {TAG_WIDTH{1'b0}};
            wr_status_id <= {AXIS_ID_WIDTH{1'b0}};
            wr_status_dest <= {AXIS_DEST_WIDTH{1'b0}};
            wr_status_user <= {AXIS_USER_WIDTH{1'b0}};
            wr_status_error <= 4'b0000;
            wr_status_valid <= 1'b0;
        end else begin
            // Default
            wr_status_valid <= 1'b0;

            // Accept write descriptor
            if (wr_desc_accept) begin
                wr_addr <= s_axis_write_desc_addr;
                wr_len <= s_axis_write_desc_len;
                wr_tag <= s_axis_write_desc_tag;
                wr_active <= 1'b1;
                wr_remaining <= s_axis_write_desc_len;
                wr_error <= 4'b0000;
                wr_burst_pending <= 1'b0;
                wr_id <= s_axis_write_data_tid;
                wr_dest <= s_axis_write_data_tdest;
                wr_user <= s_axis_write_data_tuser;
            end

            // Issue write address
            if (wr_active && !wr_burst_pending && !aw_valid) begin
                if (wr_remaining > 0) begin
                    wr_burst_len <= calc_burst_len(wr_remaining);
                    wr_burst_addr <= wr_addr;
                    aw_valid <= 1'b1;
                    aw_id <= wr_tag[AXI_ID_WIDTH-1:0];
                    aw_addr <= wr_addr;
                    aw_len <= calc_burst_len(wr_remaining) - 1;
                    aw_size <= $clog2(DATA_BYTES) - 1;
                    aw_burst <= 2'b01; // INCR
                    aw_lock <= 1'b0;
                    aw_cache <= 4'b0000;
                    aw_prot <= 3'b000;
                    wr_burst_pending <= 1'b1;
                end else begin
                    // Done
                    wr_active <= 1'b0;
                    wr_status_len <= wr_len;
                    wr_status_tag <= wr_tag;
                    wr_status_id <= wr_id;
                    wr_status_dest <= wr_dest;
                    wr_status_user <= wr_user;
                    wr_status_error <= wr_error;
                    wr_status_valid <= 1'b1;
                end
            end

            // AXI write address handshake
            if (aw_handshake) begin
                aw_valid <= 1'b0;
            end

            // Write data FIFO
            if (wr_fifo_wr_en) begin
                wr_data_buf[wr_wr_ptr[AWLEN_WIDTH-1:0]] <= s_axis_write_data_tdata;
                wr_strb_buf[wr_wr_ptr[AWLEN_WIDTH-1:0]] <= s_axis_write_data_tkeep;
                wr_last_buf[wr_wr_ptr[AWLEN_WIDTH-1:0]] <= s_axis_write_data_tlast;
                wr_wr_ptr <= wr_wr_ptr + 1;
            end

            // AXI write data
            if (wr_burst_pending && !wr_fifo_empty) begin
                w_valid <= 1'b1;
                w_data <= wr_data_buf[wr_rd_ptr[AWLEN_WIDTH-1:0]];
                w_strb <= wr_strb_buf[wr_rd_ptr[AWLEN_WIDTH-1:0]];
                w_last <= wr_last_buf[wr_rd_ptr[AWLEN_WIDTH-1:0]];
            end

            if (wr_fifo_rd_en) begin
                wr_rd_ptr <= wr_rd_ptr + 1;
                if (wr_last_buf[wr_rd_ptr[AWLEN_WIDTH-1:0]]) begin
                    w_valid <= 1'b0;
                end
            end

            // Check for burst completion
            if (wr_fifo_rd_en && wr_last_buf[wr_rd_ptr[AWLEN_WIDTH-1:0]]) begin
                wr_burst_pending <= 1'b0;
                wr_addr <= wr_addr + (wr_burst_len * DATA_BYTES);
                wr_remaining <= wr_remaining - wr_burst_len;
            end

            // AXI write response
            if (wr_burst_pending) begin
                b_ready <= 1'b1;
            end

            if (b_handshake) begin
                if (m_axi_bresp != 2'b00) begin
                    wr_error <= {2'b00, m_axi_bresp};
                end
                b_ready <= 1'b0;
            end

            // Check for write completion
            if (wr_active && !wr_burst_pending && wr_remaining == 0 && wr_fifo_empty) begin
                wr_active <= 1'b0;
                wr_status_len <= wr_len;
                wr_status_tag <= wr_tag;
                wr_status_id <= wr_id;
                wr_status_dest <= wr_dest;
                wr_status_user <= wr_user;
                wr_status_error <= wr_error;
                wr_status_valid <= 1'b1;
            end

            // Write abort
            if (write_abort) begin
                wr_active <= 1'b0;
                wr_burst_pending <= 1'b0;
                aw_valid <= 1'b0;
                w_valid <= 1'b0;
                b_ready <= 1'b0;
                wr_status_len <= wr_len;
                wr_status_tag <= wr_tag;
                wr_status_id <= wr_id;
                wr_status_dest <= wr_dest;
                wr_status_user <= wr_user;
                wr_status_error <= 4'b1000; // Abort error
                wr_status_valid <= 1'b1;
            end
        end
    end

endmodule
