```verilog
module axi_dma #
(
    parameter AXI_DATA_WIDTH = 32,
    parameter AXI_ADDR_WIDTH = 16,
    parameter AXI_STRB_WIDTH = (AXI_DATA_WIDTH/8),
    parameter AXI_ID_WIDTH = 8,
    parameter AXI_MAX_BURST_LEN = 16,
    parameter AXIS_DATA_WIDTH = AXI_DATA_WIDTH,
    parameter AXIS_KEEP_ENABLE = (AXIS_DATA_WIDTH>8),
    parameter AXIS_KEEP_WIDTH = (AXIS_DATA_WIDTH/8),
    parameter AXIS_LAST_ENABLE = 1,
    parameter AXIS_ID_ENABLE = 0,
    parameter AXIS_ID_WIDTH = 8,
    parameter AXIS_DEST_ENABLE = 0,
    parameter AXIS_DEST_WIDTH = 8,
    parameter AXIS_USER_ENABLE = 1,
    parameter AXIS_USER_WIDTH = 1,
    parameter LEN_WIDTH = 20,
    parameter TAG_WIDTH = 8,
    parameter ENABLE_SG = 0,
    parameter ENABLE_UNALIGNED = 0
)
(
    input  wire                       clk,
    input  wire                       rst,
    /*
     * AXI read descriptor input
     */
    input  wire [AXI_ADDR_WIDTH-1:0]  s_axis_read_desc_addr,
    input  wire [LEN_WIDTH-1:0]       s_axis_read_desc_len,
    input  wire [TAG_WIDTH-1:0]       s_axis_read_desc_tag,
    input  wire [AXIS_ID_WIDTH-1:0]   s_axis_read_desc_id,
    input  wire [AXIS_DEST_WIDTH-1:0] s_axis_read_desc_dest,
    input  wire [AXIS_USER_WIDTH-1:0] s_axis_read_desc_user,
    input  wire                       s_axis_read_desc_valid,
    output wire                       s_axis_read_desc_ready,
    /*
     * AXI read descriptor status output
     */
    output wire [TAG_WIDTH-1:0]       m_axis_read_desc_status_tag,
    output wire [3:0]                 m_axis_read_desc_status_error,
    output wire                       m_axis_read_desc_status_valid,
    /*
     * AXI stream read data output
     */
    output wire [AXIS_DATA_WIDTH-1:0] m_axis_read_data_tdata,
    output wire [AXIS_KEEP_WIDTH-1:0] m_axis_read_data_tkeep,
    output wire                       m_axis_read_data_tvalid,
    input  wire                       m_axis_read_data_tready,
    output wire                       m_axis_read_data_tlast,
    output wire [AXIS_ID_WIDTH-1:0]   m_axis_read_data_tid,
    output wire [AXIS_DEST_WIDTH-1:0] m_axis_read_data_tdest,
    output wire [AXIS_USER_WIDTH-1:0] m_axis_read_data_tuser,
    /*
     * AXI write descriptor input
     */
    input  wire [AXI_ADDR_WIDTH-1:0]  s_axis_write_desc_addr,
    input  wire [LEN_WIDTH-1:0]       s_axis_write_desc_len,
    input  wire [TAG_WIDTH-1:0]       s_axis_write_desc_tag,
    input  wire                       s_axis_write_desc_valid,
    output wire                       s_axis_write_desc_ready,
    /*
     * AXI write descriptor status output
     */
    output wire [LEN_WIDTH-1:0]       m_axis_write_desc_status_len,
    output wire [TAG_WIDTH-1:0]       m_axis_write_desc_status_tag,
    output wire [AXIS_ID_WIDTH-1:0]   m_axis_write_desc_status_id,
    output wire [AXIS_DEST_WIDTH-1:0] m_axis_write_desc_status_dest,
    output wire [AXIS_USER_WIDTH-1:0] m_axis_write_desc_status_user,
    output wire [3:0]                 m_axis_write_desc_status_error,
    output wire                       m_axis_write_desc_status_valid,
    /*
     * AXI stream write data input
     */
    input  wire [AXIS_DATA_WIDTH-1:0] s_axis_write_data_tdata,
    input  wire [AXIS_KEEP_WIDTH-1:0] s_axis_write_data_tkeep,
    input  wire                       s_axis_write_data_tvalid,
    output wire                       s_axis_write_data_tready,
    input  wire                       s_axis_write_data_tlast,
    input  wire [AXIS_ID_WIDTH-1:0]   s_axis_write_data_tid,
    input  wire [AXIS_DEST_WIDTH-1:0] s_axis_write_data_tdest,
    input  wire [AXIS_USER_WIDTH-1:0] s_axis_write_data_tuser,
    /*
     * AXI master interface
     */
    output wire [AXI_ID_WIDTH-1:0]    m_axi_awid,
    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_awaddr,
    output wire [7:0]                 m_axi_awlen,
    output wire [2:0]                 m_axi_awsize,
    output wire [1:0]                 m_axi_awburst,
    output wire                       m_axi_awlock,
    output wire [3:0]                 m_axi_awcache,
    output wire [2:0]                 m_axi_awprot,
    output wire                       m_axi_awvalid,
    input  wire                       m_axi_awready,
    output wire [AXI_DATA_WIDTH-1:0]  m_axi_wdata,
    output wire [AXI_STRB_WIDTH-1:0]  m_axi_wstrb,
    output wire                       m_axi_wlast,
    output wire                       m_axi_wvalid,
    input  wire                       m_axi_wready,
    input  wire [AXI_ID_WIDTH-1:0]    m_axi_bid,
    input  wire [1:0]                 m_axi_bresp,
    input  wire                       m_axi_bvalid,
    output wire                       m_axi_bready,
    output wire [AXI_ID_WIDTH-1:0]    m_axi_arid,
    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_araddr,
    output wire [7:0]                 m_axi_arlen,
    output wire [2:0]                 m_axi_arsize,
    output wire [1:0]                 m_axi_arburst,
    output wire                       m_axi_arlock,
    output wire [3:0]                 m_axi_arcache,
    output wire [2:0]                 m_axi_arprot,
    output wire                       m_axi_arvalid,
    input  wire                       m_axi_arready,
    input  wire [AXI_ID_WIDTH-1:0]    m_axi_rid,
    input  wire [AXI_DATA_WIDTH-1:0]  m_axi_rdata,
    input  wire [1:0]                 m_axi_rresp,
    input  wire                       m_axi_rlast,
    input  wire                       m_axi_rvalid,
    output wire                       m_axi_rready,
    /*
     * Configuration
     */
    input  wire                       read_enable,
    input  wire                       write_enable,
    input  wire                       write_abort
);

    // Local parameters
    localparam AXI_DATA_BYTES = AXI_DATA_WIDTH / 8;
    localparam AXI_DATA_LOG2 = $clog2(AXI_DATA_WIDTH);
    localparam AXI_MAX_BURST_LEN_LOG2 = $clog2(AXI_MAX_BURST_LEN);
    localparam BURST_LEN_WIDTH = (AXI_MAX_BURST_LEN_LOG2 > 8) ? AXI_MAX_BURST_LEN_LOG2 : 8;
    localparam BURST_LEN_MASK = (1 << BURST_LEN_WIDTH) - 1;
    localparam BURST_LEN_MINUS_1 = (1 << BURST_LEN_WIDTH) - 2;
    
    // Read channel state
    reg [AXI_ADDR_WIDTH-1:0] rd_addr_reg;
    reg [LEN_WIDTH-1:0] rd_len_reg;
    reg [TAG_WIDTH-1:0] rd_tag_reg;
    reg [AXIS_ID_WIDTH-1:0] rd_id_reg;
    reg [AXIS_DEST_WIDTH-1:0] rd_dest_reg;
    reg [AXIS_USER_WIDTH-1:0] rd_user_reg;
    reg [LEN_WIDTH-1:0] rd_remaining_reg;
    reg [BURST_LEN_WIDTH-1:0] rd_burst_remaining_reg;
    reg rd_active;
    reg [3:0] rd_error_reg;
    
    // Write channel state
    reg [AXI_ADDR_WIDTH-1:0] wr_addr_reg;
    reg [LEN_WIDTH-1:0] wr_len_reg;
    reg [TAG_WIDTH-1:0] wr_tag_reg;
    reg [LEN_WIDTH-1:0] wr_remaining_reg;
    reg [BURST_LEN_WIDTH-1:0] wr_burst_remaining_reg;
    reg wr_active;
    reg [3:0] wr_error_reg;
    reg wr_aw_sent;
    reg wr_b_received;
    
    // Read data FIFO (simple register-based)
    reg [AXIS_DATA_WIDTH-1:0] rd_data_fifo [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_KEEP_WIDTH-1:0] rd_keep_fifo [0:AXI_MAX_BURST_LEN-1];
    reg rd_data_valid;
    reg rd_data_last;
    reg [AXIS_ID_WIDTH-1:0] rd_data_id;
    reg [AXIS_DEST_WIDTH-1:0] rd_data_dest;
    reg [AXIS_USER_WIDTH-1:0] rd_data_user;
    
    // Write data FIFO
    reg [AXIS_DATA_WIDTH-1:0] wr_data_fifo [0:AXI_MAX_BURST_LEN-1];
    reg [AXIS_KEEP_WIDTH-1:0] wr_keep_fifo [0:AXI_MAX_BURST_LEN-1];
    reg wr_data_valid;
    reg wr_data_last;
    reg [AXIS_ID_WIDTH-1:0] wr_data_id;
    reg [AXIS_DEST_WIDTH-1:0] wr_data_dest;
    reg [AXIS_USER_WIDTH-1:0] wr_data_user;
    
    // Status output registers
    reg [TAG_WIDTH-1:0] rd_status_tag_reg;
    reg [3:0] rd_status_error_reg;
    reg rd_status_valid_reg;
    
    reg [LEN_WIDTH-1:0] wr_status_len_reg;
    reg [TAG_WIDTH-1:0] wr_status_tag_reg;
    reg [AXIS_ID_WIDTH-1:0] wr_status_id_reg;
    reg [AXIS_DEST_WIDTH-1:0] wr_status_dest_reg;
    reg [AXIS_USER_WIDTH-1:0] wr_status_user_reg;
    reg [3:0] wr_status_error_reg;
    reg wr_status_valid_reg;
    
    // Helper functions
    function [BURST_LEN_WIDTH-1:0] calc_burst_len;
        input [LEN_WIDTH-1:0] remaining;
        begin
            if (remaining > AXI_MAX_BURST_LEN)
                calc_burst_len = AXI_MAX_BURST_LEN;
            else
                calc_burst_len = remaining;
        end
    endfunction
    
    // Read descriptor acceptance
    wire rd_desc_accept = s_axis_read_desc_valid && s_axis_read_desc_ready && read_enable && !rd_active;
    assign s_axis_read_desc_ready = read_enable && !rd_active;
    
    // Write descriptor acceptance
    wire wr_desc_accept = s_axis_write_desc_valid && s_axis_write_desc_ready && write_enable && !wr_active;
    assign s_axis_write_desc_ready = write_enable && !wr_active;
    
    // Read data output
    assign m_axis_read_data_tdata = rd_data_valid ? rd_data_fifo[0] : {AXIS_DATA_WIDTH{1'b0}};
    assign m_axis_read_data_tkeep = rd_data_valid ? rd_keep_fifo[0] : {AXIS_KEEP_WIDTH{1'b0}};
    assign m_axis_read_data_tvalid = rd_data_valid;
    assign m_axis_read_data_tlast = rd_data_valid && rd_data_last;
    assign m_axis_read_data_tid = rd_data_valid ? rd_data_id : {AXIS_ID_WIDTH{1'b0}};
    assign m_axis_read_data_tdest = rd_data_valid ? rd_data_dest : {AXIS_DEST_WIDTH{1'b0}};
    assign m_axis_read_data_tuser = rd_data_valid ? rd_data_user : {AXIS_USER_WIDTH{1'b0}};
    
    // Write data input
    assign s_axis_write_data_tready = wr_active && wr_data_valid && m_axi_wready;
    
    // AXI Read Address Channel
    assign m_axi_arid = {AXI_ID_WIDTH{1'b0}};
    assign m_axi_araddr = rd_addr_reg;
    assign m_axi_arlen = rd_burst_remaining_reg - 1;
    assign m_axi_arsize = AXI_DATA_LOG2[2:0];
    assign m_axi_arburst = 2'b01; // INCR
    assign m_axi_arlock = 1'b0;
    assign m_axi_arcache = 4'b0011;
    assign m_axi_arprot = 3'b000;
    assign m_axi_arvalid = rd_active && (rd_burst_remaining_reg > 0) && !rd_data_valid;
    
    // AXI Read Data Channel
    assign m_axi_rready = rd_active && !rd_data_valid;
    
    // AXI Write Address Channel
    assign m_axi_awid = {AXI_ID_WIDTH{1'b0}};
    assign m_axi_awaddr = wr_addr_reg;
    assign m_axi_awlen = wr_burst_remaining_reg - 1;
    assign m_axi_awsize = AXI_DATA_LOG2[2:0];
    assign m_axi_awburst = 2'b01; // INCR
    assign m_axi_awlock = 1'b0;
    assign m_axi_awcache = 4'b0011;
    assign m_axi_awprot = 3'b000;
    assign m_axi_awvalid = wr_active && !wr_aw_sent && (wr_burst_remaining_reg > 0);
    
    // AXI Write Data Channel
    assign m_axi_wdata = wr_data_valid ? wr_data_fifo[0] : {AXI_DATA_WIDTH{1'b0}};
    assign m_axi_wstrb = wr_data_valid ? wr_keep_fifo[0] : {AXI_STRB_WIDTH{1'b0}};
    assign m_axi_wlast = wr_data_valid && wr_data_last;
    assign m_axi_wvalid = wr_active && wr_data_valid;
    
    // AXI Write Response Channel
    assign m_axi_bready = wr_active && wr_aw_sent && !wr_b_received;
    
    // Read status output
    assign m_axis_read_desc_status_tag = rd_status_tag_reg;
    assign m_axis_read_desc_status_error = rd_status_error_reg;
    assign m_axis_read_desc_status_valid = rd_status_valid_reg;
    
    // Write status output
    assign m_axis_write_desc_status_len = wr_status_len_reg;
    assign m_axis_write_desc_status_tag = wr_status_tag_reg;
    assign m_axis_write_desc_status_id = wr_status_id_reg;
    assign m_axis_write_desc_status_dest = wr_status_dest_reg;
    assign m_axis_write_desc_status_user = wr_status_user_reg;
    assign m_axis_write_desc_status_error = wr_status_error_reg;
    assign m_axis_write_desc_status_valid = wr_status_valid_reg;
    
    // Read channel logic
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rd_addr_reg <= {AXI_ADDR_WIDTH{1'b0}};
            rd_len_reg <= {LEN_WIDTH{1'b0}};
            rd_tag_reg <= {TAG_WIDTH{1'b0}};
            rd_id_reg <= {AXIS_ID_WIDTH{1'b0}};
            rd_dest_reg <= {AXIS_DEST_WIDTH{1'b0}};
            rd_user_reg <= {AXIS_USER_WIDTH{1'b0}};
            rd_remaining_reg <= {LEN_WIDTH{1'b0}};
            rd_burst_remaining_reg <= {BURST_LEN_WIDTH{1'b0}};
            rd_active <= 1'b0;
            rd_error_reg <= 4'b0000;
            rd_data_valid <= 1'b0;
            rd_data_last <= 1'b0;
            rd_data_id <= {AXIS_ID_WIDTH{1'b0}};
            rd_data_dest <= {AXIS_DEST_WIDTH{1'b0}};
            rd_data_user <= {AXIS_USER_WIDTH{1'b0}};
            rd_status_tag_reg <= {TAG_WIDTH{1'b0}};
            rd_status_error_reg <= 4'b0000;
            rd_status_valid_reg <= 1'b0;
        end else begin
            // Accept new read descriptor
            if (rd_desc_accept) begin
                rd_addr_reg <= s_axis_read_desc_addr;
                rd_len_reg <= s_axis_read_desc_len;
                rd_tag_reg <= s_axis_read_desc_tag;
                rd_id_reg <= s_axis_read_desc_id;
                rd_dest_reg <= s_axis_read_desc_dest;
                rd_user_reg <= s_axis_read_desc_user;
                rd_remaining_reg <= s_axis_read_desc_len;
                rd_burst_remaining_reg <= calc_burst_len(s_axis_read_desc_len);
                rd_active <= 1'b1;
                rd_error_reg <= 4'b0000;
                rd_data_valid <= 1'b0;
                rd_data_last <= 1'b0;
            end
            
            // Read data from AXI
            if (m_axi_rvalid && m_axi_rready && rd_active) begin
                rd_data_fifo[0] <= m_axi_rdata;
                rd_keep_fifo[0] <= {AXIS_KEEP_WIDTH{1'b1}};
                rd_data_valid <= 1'b1;
                rd_data_last <= m_axi_rlast;
                rd_data_id <= rd_id_reg;
                rd_data_dest <= rd_dest_reg;
                rd_data_user <= rd_user_reg;
                
                if (m_axi_rresp != 2'b00) begin
                    rd_error_reg <= 4'b0001;
                end
                
                rd_burst_remaining_reg <= rd_burst_remaining_reg - 1;
                rd_remaining_reg <= rd_remaining_reg - 1;
                
                if (m_axi_rlast) begin
                    if (rd_remaining_reg == 1) begin
                        rd_active <= 1'b0;
                        rd_status_tag_reg <= rd_tag_reg;
                        rd_status_error_reg <= (m_axi_rresp != 2'b00) ? 4'b0001 : 4'b0000;
                        rd_status_valid_reg <= 1'b1;
                    end else begin
                        rd_addr_reg <= rd_addr_reg + AXI_DATA_BYTES;
                        rd_burst_remaining_reg <= calc_burst_len(rd_remaining_reg - 1);
                    end
                end
            end
            
            // Output read data
            if (rd_data_valid && m_axis_read_data_tready) begin
                rd_data_valid <= 1'b0;
            end
            
            // Clear status valid
            if (rd_status_valid_reg) begin
                rd_status_valid_reg <= 1'b0;
            end
        end
    end
    
    // Write channel logic
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_addr_reg <= {AXI_ADDR_WIDTH{1'b0}};
            wr_len_reg <= {LEN_WIDTH{1'b0}};
            wr_tag_reg <= {TAG_WIDTH{1'b0}};
            wr_remaining_reg <= {LEN_WIDTH{1'b0}};
            wr_burst_remaining_reg <= {BURST_LEN_WIDTH{1'b0}};
            wr_active <= 1'b0;
            wr_error_reg <= 4'b0000;
            wr_aw_sent <= 1'b0;
            wr_b_received <= 1'b0;
            wr_data_valid <= 1'b0;
            wr_data_last <= 1'b0;
            wr_data_id <= {AXIS_ID_WIDTH{1'b0}};
            wr_data_dest <= {AXIS_DEST_WIDTH{1'b0}};
            wr_data_user <= {AXIS_USER_WIDTH{1'b0}};
            wr_status_len_reg <= {LEN_WIDTH{1'b0}};
            wr_status_tag_reg <= {TAG_WIDTH{1'b0}};
            wr_status_id_reg <= {AXIS_ID_WIDTH{1'b0}};
            wr_status_dest_reg <= {AXIS_DEST_WIDTH{1'b0}};
            wr_status_user_reg <= {AXIS_USER_WIDTH{1'b0}};
            wr_status_error_reg <= 4'b0000;
            wr_status_valid_reg <= 1'b0;
        end else begin
            // Accept new write descriptor
            if (wr_desc_accept) begin
                wr_addr_reg <= s_axis_write_desc_addr;
                wr_len_reg <= s_axis_write_desc_len;
                wr_tag_reg <= s_axis_write_desc_tag;
                wr_remaining_reg <= s_axis_write_desc_len;
                wr_burst_remaining_reg <= calc_burst_len(s_axis_write_desc_len);
                wr_active <= 1'b1;
                wr_error_reg <= 4'b0000;
                wr_aw_sent <= 1'b0;
                wr_b_received <= 1'b0;
                wr_data_valid <= 1'b0;
                wr_data_last <= 1'b0;
            end
            
            // Send AW
            if (wr_active && !wr_aw_sent && m_axi_awready) begin
                wr_aw_sent <= 1'b1;
            end
            
            // Accept write data
            if (wr_active && wr_aw_sent && s_axis_write_data_tvalid && s_axis_write_data_tready) begin
                wr_data_fifo[0] <= s_axis_write_data_tdata;
                wr_keep_fifo[0] <= s_axis_write_data_tkeep;
                wr_data_valid <= 1'b1;
                wr_data_last <= s_axis_write_data_tlast;
                wr_data_id <= s_axis_write_data_tid;
                wr_data_dest <= s_axis_write_data_tdest;
                wr_data_user <= s_axis_write_data_tuser;
                
                wr_burst_remaining_reg <= wr_burst_remaining_reg - 1;
                wr_remaining_reg <= wr_remaining_reg - 1;
            end
            
            // Send W data
            if (wr_active && wr_data_valid && m_axi_wready) begin
                wr_data_valid <= 1'b0;
                
                if (wr_data_last) begin
                    if (wr_remaining_reg == 1) begin
                        wr_active <= 1'b0;
                        wr_status_len_reg <= wr_len_reg;
                        wr_status_tag_reg <= wr_tag_reg;
                        wr_status_id_reg <= wr_data_id;
                        wr_status_dest_reg <= wr_data_dest;
                        wr_status_user_reg <= wr_data_user;
                        wr_status_error_reg <= wr_error_reg;
                        wr_status_valid_reg <= 1'b1;
                    end else begin
                        wr_addr_reg <= wr_addr_reg + AXI_DATA_BYTES;
                        wr_burst_remaining_reg <= calc_burst_len(wr_remaining_reg - 1);
                        wr_aw_sent <= 1'b0;
                    end
                end
            end
            
            // Receive B response
            if (wr_active && wr_aw_sent && m_axi_bvalid && m_axi_bready) begin
                wr_b_received <= 1'b1;
                if (m_axi_bresp != 2'b00) begin
                    wr_error_reg <= 4'b0010;
                end
            end
            
            // Handle write abort
            if (write_abort && wr_active) begin
                wr_active <= 1'b0;
                wr_status_len_reg <= wr_len_reg;
                wr_status_tag_reg <= wr_tag_reg;
                wr_status_id_reg <= wr_data_id;
                wr_status_dest_reg <= wr_data_dest;
                wr_status_user_reg <= wr_data_user;
                wr_status_error_reg <= 4'b0100;
                wr_status_valid_reg <= 1'b1;
            end
            
            // Clear status valid
            if (wr_status_valid_reg) begin
                wr_status_valid_reg <= 1'b0;
            end
        end
    end

endmodule
