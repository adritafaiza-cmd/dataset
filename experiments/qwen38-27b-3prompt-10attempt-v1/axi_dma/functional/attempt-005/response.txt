module axi_dma #
(
    parameter AXI_DATA_WIDTH = 32,
    parameter AXI_ADDR_WIDTH = 16,
    parameter AXI_STRB_WIDTH = (AXI_DATA_WIDTH/8),
    parameter AXI_ID_WIDTH = 8,
    parameter AXI_MAX_BURST_LEN = 16,
    parameter AXIS_DATA_WIDTH = AXI_DATA_WIDTH,
    parameter AXIS_KEEP_ENABLE = (AXIS_DATA_WIDTH>8),
    parameter AXIS_KEEP_WIDTH = (AXIS_DATA_WIDTH/8),
    parameter AXIS_LAST_ENABLE = 1,
    parameter AXIS_ID_ENABLE = 0,
    parameter AXIS_ID_WIDTH = 8,
    parameter AXIS_DEST_ENABLE = 0,
    parameter AXIS_DEST_WIDTH = 8,
    parameter AXIS_USER_ENABLE = 1,
    parameter AXIS_USER_WIDTH = 1,
    parameter LEN_WIDTH = 20,
    parameter TAG_WIDTH = 8,
    parameter ENABLE_SG = 0,
    parameter ENABLE_UNALIGNED = 0
)
(
    input  wire                       clk,
    input  wire                       rst,
    /*
     * AXI read descriptor input
     */
    input  wire [AXI_ADDR_WIDTH-1:0]  s_axis_read_desc_addr,
    input  wire [LEN_WIDTH-1:0]       s_axis_read_desc_len,
    input  wire [TAG_WIDTH-1:0]       s_axis_read_desc_tag,
    input  wire [AXIS_ID_WIDTH-1:0]   s_axis_read_desc_id,
    input  wire [AXIS_DEST_WIDTH-1:0] s_axis_read_desc_dest,
    input  wire [AXIS_USER_WIDTH-1:0] s_axis_read_desc_user,
    input  wire                       s_axis_read_desc_valid,
    output wire                       s_axis_read_desc_ready,
    /*
     * AXI read descriptor status output
     */
    output wire [TAG_WIDTH-1:0]       m_axis_read_desc_status_tag,
    output wire [3:0]                 m_axis_read_desc_status_error,
    output wire                       m_axis_read_desc_status_valid,
    /*
     * AXI stream read data output
     */
    output wire [AXIS_DATA_WIDTH-1:0] m_axis_read_data_tdata,
    output wire [AXIS_KEEP_WIDTH-1:0] m_axis_read_data_tkeep,
    output wire                       m_axis_read_data_tvalid,
    input  wire                       m_axis_read_data_tready,
    output wire                       m_axis_read_data_tlast,
    output wire [AXIS_ID_WIDTH-1:0]   m_axis_read_data_tid,
    output wire [AXIS_DEST_WIDTH-1:0] m_axis_read_data_tdest,
    output wire [AXIS_USER_WIDTH-1:0] m_axis_read_data_tuser,
    /*
     * AXI write descriptor input
     */
    input  wire [AXI_ADDR_WIDTH-1:0]  s_axis_write_desc_addr,
    input  wire [LEN_WIDTH-1:0]       s_axis_write_desc_len,
    input  wire [TAG_WIDTH-1:0]       s_axis_write_desc_tag,
    input  wire                       s_axis_write_desc_valid,
    output wire                       s_axis_write_desc_ready,
    /*
     * AXI write descriptor status output
     */
    output wire [LEN_WIDTH-1:0]       m_axis_write_desc_status_len,
    output wire [TAG_WIDTH-1:0]       m_axis_write_desc_status_tag,
    output wire [AXIS_ID_WIDTH-1:0]   m_axis_write_desc_status_id,
    output wire [AXIS_DEST_WIDTH-1:0] m_axis_write_desc_status_dest,
    output wire [AXIS_USER_WIDTH-1:0] m_axis_write_desc_status_user,
    output wire [3:0]                 m_axis_write_desc_status_error,
    output wire                       m_axis_write_desc_status_valid,
    /*
     * AXI stream write data input
     */
    input  wire [AXIS_DATA_WIDTH-1:0] s_axis_write_data_tdata,
    input  wire [AXIS_KEEP_WIDTH-1:0] s_axis_write_data_tkeep,
    input  wire                       s_axis_write_data_tvalid,
    output wire                       s_axis_write_data_tready,
    input  wire                       s_axis_write_data_tlast,
    input  wire [AXIS_ID_WIDTH-1:0]   s_axis_write_data_tid,
    input  wire [AXIS_DEST_WIDTH-1:0] s_axis_write_data_tdest,
    input  wire [AXIS_USER_WIDTH-1:0] s_axis_write_data_tuser,
    /*
     * AXI master interface
     */
    output wire [AXI_ID_WIDTH-1:0]    m_axi_awid,
    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_awaddr,
    output wire [7:0]                 m_axi_awlen,
    output wire [2:0]                 m_axi_awsize,
    output wire [1:0]                 m_axi_awburst,
    output wire                       m_axi_awlock,
    output wire [3:0]                 m_axi_awcache,
    output wire [2:0]                 m_axi_awprot,
    output wire                       m_axi_awvalid,
    input  wire                       m_axi_awready,
    output wire [AXI_DATA_WIDTH-1:0]  m_axi_wdata,
    output wire [AXI_STRB_WIDTH-1:0]  m_axi_wstrb,
    output wire                       m_axi_wlast,
    output wire                       m_axi_wvalid,
    input  wire                       m_axi_wready,
    input  wire [AXI_ID_WIDTH-1:0]    m_axi_bid,
    input  wire [1:0]                 m_axi_bresp,
    input  wire                       m_axi_bvalid,
    output wire                       m_axi_bready,
    output wire [AXI_ID_WIDTH-1:0]    m_axi_arid,
    output wire [AXI_ADDR_WIDTH-1:0]  m_axi_araddr,
    output wire [7:0]                 m_axi_arlen,
    output wire [2:0]                 m_axi_arsize,
    output wire [1:0]                 m_axi_arburst,
    output wire                       m_axi_arlock,
    output wire [3:0]                 m_axi_arcache,
    output wire [2:0]                 m_axi_arprot,
    output wire                       m_axi_arvalid,
    input  wire                       m_axi_arready,
    input  wire [AXI_ID_WIDTH-1:0]    m_axi_rid,
    input  wire [AXI_DATA_WIDTH-1:0]  m_axi_rdata,
    input  wire [1:0]                 m_axi_rresp,
    input  wire                       m_axi_rlast,
    input  wire                       m_axi_rvalid,
    output wire                       m_axi_rready,
    /*
     * Configuration
     */
    input  wire                       read_enable,
    input  wire                       write_enable,
    input  wire                       write_abort
);

    // Local parameters
    localparam AXI_DATA_BYTES = AXI_DATA_WIDTH / 8;
    localparam AXI_DATA_LOG2 = $clog2(AXI_DATA_WIDTH);
    localparam AXI_MAX_BURST_LEN_LOG2 = $clog2(AXI_MAX_BURST_LEN);
    
    // State machine states
    localparam ST_IDLE = 2'd0;
    localparam ST_DESC = 2'd1;
    localparam ST_AXI = 2'd2;
    localparam ST_DONE = 2'd3;

    // Read channel state
    reg [1:0] rd_state;
    reg [AXI_ADDR_WIDTH-1:0] rd_addr;
    reg [LEN_WIDTH-1:0] rd_len;
    reg [LEN_WIDTH-1:0] rd_remaining;
    reg [TAG_WIDTH-1:0] rd_tag;
    reg [AXIS_ID_WIDTH-1:0] rd_id;
    reg [AXIS_DEST_WIDTH-1:0] rd_dest;
    reg [AXIS_USER_WIDTH-1:0] rd_user;
    reg [7:0] rd_burst_len;
    reg [AXI_ADDR_WIDTH-1:0] rd_axi_addr;
    reg rd_axi_valid;
    reg rd_axi_last_burst;
    reg [LEN_WIDTH-1:0] rd_burst_remaining;
    reg rd_data_valid;
    reg [AXIS_DATA_WIDTH-1:0] rd_data;
    reg [AXIS_KEEP_WIDTH-1:0] rd_keep;
    reg rd_data_last;
    reg rd_status_valid;
    reg [3:0] rd_status_error;

    // Write channel state
    reg [1:0] wr_state;
    reg [AXI_ADDR_WIDTH-1:0] wr_addr;
    reg [LEN_WIDTH-1:0] wr_len;
    reg [LEN_WIDTH-1:0] wr_remaining;
    reg [TAG_WIDTH-1:0] wr_tag;
    reg [7:0] wr_burst_len;
    reg [AXI_ADDR_WIDTH-1:0] wr_axi_addr;
    reg wr_axi_valid;
    reg wr_axi_last_burst;
    reg [LEN_WIDTH-1:0] wr_burst_remaining;
    reg wr_data_valid;
    reg [AXIS_DATA_WIDTH-1:0] wr_data;
    reg [AXIS_KEEP_WIDTH-1:0] wr_keep;
    reg wr_data_last;
    reg wr_status_valid;
    reg [3:0] wr_status_error;
    reg [LEN_WIDTH-1:0] wr_status_len;
    reg [AXIS_ID_WIDTH-1:0] wr_status_id;
    reg [AXIS_DEST_WIDTH-1:0] wr_status_dest;
    reg [AXIS_USER_WIDTH-1:0] wr_status_user;

    // AXI write data buffer
    reg [AXI_DATA_WIDTH-1:0] axi_wdata_reg;
    reg [AXI_STRB_WIDTH-1:0] axi_wstrb_reg;
    reg axi_wlast_reg;
    reg axi_wvalid_reg;

    // AXI read data buffer
    reg [AXI_DATA_WIDTH-1:0] axi_rdata_reg;
    reg axi_rlast_reg;
    reg axi_rvalid_reg;

    // Helper functions
    function [7:0] calc_burst_len;
        input [LEN_WIDTH-1:0] remaining;
        begin
            if (remaining >= AXI_MAX_BURST_LEN)
                calc_burst_len = AXI_MAX_BURST_LEN - 1;
            else
                calc_burst_len = remaining[7:0] - 1;
        end
    endfunction

    // Read descriptor ready
    assign s_axis_read_desc_ready = (rd_state == ST_IDLE) && read_enable;

    // Write descriptor ready
    assign s_axis_write_desc_ready = (wr_state == ST_IDLE) && write_enable;

    // Read data output
    assign m_axis_read_data_tdata = rd_data;
    assign m_axis_read_data_tkeep = rd_keep;
    assign m_axis_read_data_tvalid = rd_data_valid;
    assign m_axis_read_data_tlast = rd_data_last;
    assign m_axis_read_data_tid = rd_id;
    assign m_axis_read_data_tdest = rd_dest;
    assign m_axis_read_data_tuser = rd_user;

    // Read status output
    assign m_axis_read_desc_status_tag = rd_tag;
    assign m_axis_read_desc_status_error = rd_status_error;
    assign m_axis_read_desc_status_valid = rd_status_valid;

    // Write data ready
    assign s_axis_write_data_tready = (wr_state == ST_AXI) && wr_data_valid && !axi_wvalid_reg;

    // Write status output
    assign m_axis_write_desc_status_len = wr_status_len;
    assign m_axis_write_desc_status_tag = wr_tag;
    assign m_axis_write_desc_status_id = wr_status_id;
    assign m_axis_write_desc_status_dest = wr_status_dest;
    assign m_axis_write_desc_status_user = wr_status_user;
    assign m_axis_write_desc_status_error = wr_status_error;
    assign m_axis_write_desc_status_valid = wr_status_valid;

    // AXI Write Address Channel
    assign m_axi_awid = {AXI_ID_WIDTH{1'b0}};
    assign m_axi_awaddr = wr_axi_addr;
    assign m_axi_awlen = wr_burst_len;
    assign m_axi_awsize = AXI_DATA_LOG2[2:0];
    assign m_axi_awburst = 2'b01; // INCR
    assign m_axi_awlock = 1'b0;
    assign m_axi_awcache = 4'b0011;
    assign m_axi_awprot = 3'b000;
    assign m_axi_awvalid = wr_axi_valid;

    // AXI Write Data Channel
    assign m_axi_wdata = axi_wdata_reg;
    assign m_axi_wstrb = axi_wstrb_reg;
    assign m_axi_wlast = axi_wlast_reg;
    assign m_axi_wvalid = axi_wvalid_reg;

    // AXI Write Response Channel
    assign m_axi_bready = (wr_state == ST_AXI) && wr_axi_last_burst && (wr_burst_remaining == 1);

    // AXI Read Address Channel
    assign m_axi_arid = {AXI_ID_WIDTH{1'b0}};
    assign m_axi_araddr = rd_axi_addr;
    assign m_axi_arlen = rd_burst_len;
    assign m_axi_arsize = AXI_DATA_LOG2[2:0];
    assign m_axi_arburst = 2'b01; // INCR
    assign m_axi_arlock = 1'b0;
    assign m_axi_arcache = 4'b0011;
    assign m_axi_arprot = 3'b000;
    assign m_axi_arvalid = rd_axi_valid;

    // AXI Read Data Channel
    assign m_axi_rready = (rd_state == ST_AXI) && rd_data_valid;

    // Read state machine
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rd_state <= ST_IDLE;
            rd_addr <= {AXI_ADDR_WIDTH{1'b0}};
            rd_len <= {LEN_WIDTH{1'b0}};
            rd_remaining <= {LEN_WIDTH{1'b0}};
            rd_tag <= {TAG_WIDTH{1'b0}};
            rd_id <= {AXIS_ID_WIDTH{1'b0}};
            rd_dest <= {AXIS_DEST_WIDTH{1'b0}};
            rd_user <= {AXIS_USER_WIDTH{1'b0}};
            rd_burst_len <= 8'd0;
            rd_axi_addr <= {AXI_ADDR_WIDTH{1'b0}};
            rd_axi_valid <= 1'b0;
            rd_axi_last_burst <= 1'b0;
            rd_burst_remaining <= {LEN_WIDTH{1'b0}};
            rd_data_valid <= 1'b0;
            rd_data <= {AXIS_DATA_WIDTH{1'b0}};
            rd_keep <= {AXIS_KEEP_WIDTH{1'b0}};
            rd_data_last <= 1'b0;
            rd_status_valid <= 1'b0;
            rd_status_error <= 4'd0;
        end else begin
            case (rd_state)
                ST_IDLE: begin
                    rd_axi_valid <= 1'b0;
                    rd_data_valid <= 1'b0;
                    rd_status_valid <= 1'b0;
                    if (s_axis_read_desc_valid && read_enable) begin
                        rd_state <= ST_DESC;
                        rd_addr <= s_axis_read_desc_addr;
                        rd_len <= s_axis_read_desc_len;
                        rd_remaining <= s_axis_read_desc_len;
                        rd_tag <= s_axis_read_desc_tag;
                        rd_id <= s_axis_read_desc_id;
                        rd_dest <= s_axis_read_desc_dest;
                        rd_user <= s_axis_read_desc_user;
                    end
                end
                ST_DESC: begin
                    rd_state <= ST_AXI;
                    rd_axi_addr <= rd_addr;
                    rd_burst_len <= calc_burst_len(rd_remaining);
                    rd_burst_remaining <= rd_burst_len + 1;
                    rd_axi_valid <= 1'b1;
                    rd_axi_last_burst <= (rd_remaining <= (rd_burst_len + 1));
                end
                ST_AXI: begin
                    // Handle AXI read address handshake
                    if (rd_axi_valid && m_axi_arready) begin
                        rd_axi_valid <= 1'b0;
                    end

                    // Handle AXI read data
                    if (m_axi_rvalid && m_axi_rready) begin
                        rd_data <= m_axi_rdata;
                        rd_keep <= {AXIS_KEEP_WIDTH{1'b1}};
                        rd_data_last <= m_axi_rlast;
                        rd_data_valid <= 1'b1;
                        rd_burst_remaining <= rd_burst_remaining - 1;
                        
                        if (m_axi_rlast) begin
                            if (rd_axi_last_burst) begin
                                rd_state <= ST_DONE;
                                rd_status_valid <= 1'b1;
                                rd_status_error <= 4'd0;
                            end else begin
                                // Next burst
                                rd_axi_addr <= rd_axi_addr + (rd_burst_len + 1) * AXI_DATA_BYTES;
                                rd_burst_len <= calc_burst_len(rd_remaining - (rd_burst_len + 1));
                                rd_burst_remaining <= rd_burst_len + 1;
                                rd_axi_valid <= 1'b1;
                                rd_axi_last_burst <= ((rd_remaining - (rd_burst_len + 1)) <= (rd_burst_len + 1));
                            end
                        end
                    end

                    // Handle stream data output
                    if (rd_data_valid && m_axis_read_data_tready) begin
                        rd_data_valid <= 1'b0;
                    end
                end
                ST_DONE: begin
                    if (rd_status_valid) begin
                        rd_state <= ST_IDLE;
                        rd_status_valid <= 1'b0;
                    end
                end
                default: rd_state <= ST_IDLE;
            endcase
        end
    end

    // Write state machine
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_state <= ST_IDLE;
            wr_addr <= {AXI_ADDR_WIDTH{1'b0}};
            wr_len <= {LEN_WIDTH{1'b0}};
            wr_remaining <= {LEN_WIDTH{1'b0}};
            wr_tag <= {TAG_WIDTH{1'b0}};
            wr_burst_len <= 8'd0;
            wr_axi_addr <= {AXI_ADDR_WIDTH{1'b0}};
            wr_axi_valid <= 1'b0;
            wr_axi_last_burst <= 1'b0;
            wr_burst_remaining <= {LEN_WIDTH{1'b0}};
            wr_data_valid <= 1'b0;
            wr_data <= {AXIS_DATA_WIDTH{1'b0}};
            wr_keep <= {AXIS_KEEP_WIDTH{1'b0}};
            wr_data_last <= 1'b0;
            wr_status_valid <= 1'b0;
            wr_status_error <= 4'd0;
            wr_status_len <= {LEN_WIDTH{1'b0}};
            wr_status_id <= {AXIS_ID_WIDTH{1'b0}};
            wr_status_dest <= {AXIS_DEST_WIDTH{1'b0}};
            wr_status_user <= {AXIS_USER_WIDTH{1'b0}};
            axi_wdata_reg <= {AXI_DATA_WIDTH{1'b0}};
            axi_wstrb_reg <= {AXI_STRB_WIDTH{1'b0}};
            axi_wlast_reg <= 1'b0;
            axi_wvalid_reg <= 1'b0;
        end else begin
            case (wr_state)
                ST_IDLE: begin
                    wr_axi_valid <= 1'b0;
                    wr_data_valid <= 1'b0;
                    wr_status_valid <= 1'b0;
                    axi_wvalid_reg <= 1'b0;
                    if (s_axis_write_desc_valid && write_enable) begin
                        wr_state <= ST_DESC;
                        wr_addr <= s_axis_write_desc_addr;
                        wr_len <= s_axis_write_desc_len;
                        wr_remaining <= s_axis_write_desc_len;
                        wr_tag <= s_axis_write_desc_tag;
                        wr_status_len <= s_axis_write_desc_len;
                        wr_status_id <= s_axis_write_data_tid;
                        wr_status_dest <= s_axis_write_data_tdest;
                        wr_status_user <= s_axis_write_data_tuser;
                    end
                end
                ST_DESC: begin
                    wr_state <= ST_AXI;
                    wr_axi_addr <= wr_addr;
                    wr_burst_len <= calc_burst_len(wr_remaining);
                    wr_burst_remaining <= wr_burst_len + 1;
                    wr_axi_valid <= 1'b1;
                    wr_axi_last_burst <= (wr_remaining <= (wr_burst_len + 1));
                end
                ST_AXI: begin
                    // Handle AXI write address handshake
                    if (wr_axi_valid && m_axi_awready) begin
                        wr_axi_valid <= 1'b0;
                    end

                    // Handle stream data input
                    if (s_axis_write_data_tvalid && s_axis_write_data_tready) begin
                        wr_data <= s_axis_write_data_tdata;
                        wr_keep <= s_axis_write_data_tkeep;
                        wr_data_last <= s_axis_write_data_tlast;
                        wr_data_valid <= 1'b1;
                    end

                    // Handle AXI write data
                    if (wr_data_valid && !axi_wvalid_reg) begin
                        axi_wdata_reg <= wr_data;
                        axi_wstrb_reg <= wr_keep;
                        axi_wlast_reg <= (wr_burst_remaining == 1);
                        axi_wvalid_reg <= 1'b1;
                        wr_data_valid <= 1'b0;
                    end

                    if (axi_wvalid_reg && m_axi_wready) begin
                        axi_wvalid_reg <= 1'b0;
                        wr_burst_remaining <= wr_burst_remaining - 1;
                        
                        if (wr_burst_remaining == 1) begin
                            if (wr_axi_last_burst) begin
                                wr_state <= ST_DONE;
                                wr_status_valid <= 1'b1;
                                wr_status_error <= 4'd0;
                            end else begin
                                // Next burst
                                wr_axi_addr <= wr_axi_addr + (wr_burst_len + 1) * AXI_DATA_BYTES;
                                wr_burst_len <= calc_burst_len(wr_remaining - (wr_burst_len + 1));
                                wr_burst_remaining <= wr_burst_len + 1;
                                wr_axi_valid <= 1'b1;
                                wr_axi_last_burst <= ((wr_remaining - (wr_burst_len + 1)) <= (wr_burst_len + 1));
                            end
                        end
                    end

                    // Handle write abort
                    if (write_abort) begin
                        wr_state <= ST_DONE;
                        wr_status_valid <= 1'b1;
                        wr_status_error <= 4'd1;
                        axi_wvalid_reg <= 1'b0;
                    end
                end
                ST_DONE: begin
                    if (wr_status_valid) begin
                        wr_state <= ST_IDLE;
                        wr_status_valid <= 1'b0;
                    end
                end
                default: wr_state <= ST_IDLE;
            endcase
        end
    end

endmodule
