```verilog
module afifo #(
		parameter	LGFIFO = 3,
		parameter	WIDTH  = 16,
		parameter	NFF    = 2,
		parameter [0:0]	WRITE_ON_POSEDGE = 1'b1,
		parameter [0:0]	OPT_REGISTER_READS = 1'b1
	) (
		input	wire			i_wclk, i_wr_reset_n, i_wr,
		input	wire	[WIDTH-1:0]	i_wr_data,
		output	reg			o_wr_full,
		input	wire			i_rclk, i_rd_reset_n, i_rd,
		output	reg	[WIDTH-1:0]	o_rd_data,
		output	reg			o_rd_empty
	);

	localparam DEPTH = 1 << LGFIFO;
	localparam ADDR_W = LGFIFO;
	localparam CNT_W = LGFIFO + 1;

	// Write domain
	reg [WIDTH-1:0] mem [0:DEPTH-1];
	reg [ADDR_W-1:0] wr_ptr;
	reg [ADDR_W-1:0] wr_ptr_gray;
	reg [ADDR_W-1:0] rd_ptr_gray_w;
	reg [CNT_W-1:0] wr_count;
	reg [CNT_W-1:0] rd_count_w;

	// Read domain
	reg [ADDR_W-1:0] rd_ptr;
	reg [ADDR_W-1:0] rd_ptr_gray;
	reg [ADDR_W-1:0] wr_ptr_gray_r;
	reg [CNT_W-1:0] rd_count;
	reg [CNT_W-1:0] wr_count_r;

	// Synchronizers
	reg [ADDR_W-1:0] wr_ptr_gray_sync1, wr_ptr_gray_sync2;
	reg [ADDR_W-1:0] rd_ptr_gray_sync1, rd_ptr_gray_sync2;

	// Gray code conversion functions
	function [ADDR_W-1:0] bin2gray;
		input [ADDR_W-1:0] bin;
		begin
			bin2gray = bin ^ (bin >> 1);
		end
	endfunction

	function [ADDR_W-1:0] gray2bin;
		input [ADDR_W-1:0] gray;
		reg [ADDR_W-1:0] temp;
		begin
			temp[ADDR_W-1] = gray[ADDR_W-1];
			for (i = ADDR_W-2; i >= 0; i = i - 1)
				temp[i] = gray[i] ^ temp[i+1];
			gray2bin = temp;
		end
	endfunction

	// Write clock domain logic
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			wr_ptr <= {ADDR_W{1'b0}};
			wr_ptr_gray <= {ADDR_W{1'b0}};
			rd_ptr_gray_w <= {ADDR_W{1'b0}};
			wr_count <= {CNT_W{1'b0}};
			rd_count_w <= {CNT_W{1'b0}};
		end else begin
			if (i_wr && !o_wr_full) begin
				mem[wr_ptr] <= i_wr_data;
				wr_ptr <= wr_ptr + 1'b1;
				wr_count <= wr_count + 1'b1;
			end
			if (rd_ptr_gray_w != rd_ptr_gray_sync2) begin
				// Read pointer changed in read domain
				rd_count_w <= gray2bin(rd_ptr_gray_sync2);
			end
			rd_ptr_gray_w <= rd_ptr_gray_sync2;
		end
	end

	// Update gray code pointer in write domain
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n)
			wr_ptr_gray <= {ADDR_W{1'b0}};
		else
			wr_ptr_gray <= bin2gray(wr_ptr);
	end

	// Synchronize read pointer gray to write domain
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			rd_ptr_gray_sync1 <= {ADDR_W{1'b0}};
			rd_ptr_gray_sync2 <= {ADDR_W{1'b0}};
		end else begin
			rd_ptr_gray_sync1 <= rd_ptr_gray;
			rd_ptr_gray_sync2 <= rd_ptr_gray_sync1;
		end
	end

	// Full flag in write domain
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n)
			o_wr_full <= 1'b0;
		else
			o_wr_full <= (wr_count == DEPTH);
	end

	// Read clock domain logic
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			rd_ptr <= {ADDR_W{1'b0}};
			rd_ptr_gray <= {ADDR_W{1'b0}};
			wr_ptr_gray_r <= {ADDR_W{1'b0}};
			rd_count <= {CNT_W{1'b0}};
			wr_count_r <= {CNT_W{1'b0}};
		end else begin
			if (i_rd && !o_rd_empty) begin
				rd_ptr <= rd_ptr + 1'b1;
				rd_count <= rd_count + 1'b1;
			end
			if (wr_ptr_gray_r != wr_ptr_gray_sync2) begin
				// Write pointer changed in write domain
				wr_count_r <= gray2bin(wr_ptr_gray_sync2);
			end
			wr_ptr_gray_r <= wr_ptr_gray_sync2;
		end
	end

	// Update gray code pointer in read domain
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n)
			rd_ptr_gray <= {ADDR_W{1'b0}};
		else
			rd_ptr_gray <= bin2gray(rd_ptr);
	end

	// Synchronize write pointer gray to read domain
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			wr_ptr_gray_sync1 <= {ADDR_W{1'b0}};
			wr_ptr_gray_sync2 <= {ADDR_W{1'b0}};
		end else begin
			wr_ptr_gray_sync1 <= wr_ptr_gray;
			wr_ptr_gray_sync2 <= wr_ptr_gray_sync1;
		end
	end

	// Empty flag in read domain
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n)
			o_rd_empty <= 1'b1;
		else
			o_rd_empty <= (rd_count == wr_count_r);
	end

	// Read data output
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n)
			o_rd_data <= {WIDTH{1'b0}};
		else if (OPT_REGISTER_READS)
			o_rd_data <= mem[rd_ptr];
		else
			o_rd_data <= mem[rd_ptr];
	end

endmodule
