module afifo #(
		parameter	LGFIFO = 3,
		parameter	WIDTH  = 16,
		parameter	NFF    = 2,
		parameter [0:0]	WRITE_ON_POSEDGE = 1'b1,
		parameter [0:0]	OPT_REGISTER_READS = 1'b1
	) (
		input	wire			i_wclk, i_wr_reset_n, i_wr,
		input	wire	[WIDTH-1:0]	i_wr_data,
		output	reg			o_wr_full,
		input	wire			i_rclk, i_rd_reset_n, i_rd,
		output	reg	[WIDTH-1:0]	o_rd_data,
		output	reg			o_rd_empty
	);

	localparam DEPTH = 1 << LGFIFO;
	localparam ADDR_W = LGFIFO;
	localparam CNT_W = LGFIFO + 1;
	localparam GRAY_W = LGFIFO + 1;

	// Write domain
	reg [ADDR_W-1:0] w_addr;
	reg [ADDR_W-1:0] w_addr_gray;
	reg [CNT_W-1:0]  w_count;
	reg [WIDTH-1:0]  mem [0:DEPTH-1];

	// Read domain
	reg [ADDR_W-1:0] r_addr;
	reg [ADDR_W-1:0] r_addr_gray;
	reg [CNT_W-1:0]  r_count;
	reg [WIDTH-1:0]  r_data_reg;

	// Gray code conversion functions
	function [GRAY_W-1:0] bin2gray;
		input [CNT_W-1:0] bin;
		begin
			bin2gray = bin ^ (bin >> 1);
		end
	endfunction

	function [CNT_W-1:0] gray2bin;
		input [GRAY_W-1:0] gray;
		reg [CNT_W-1:0]   bin;
		begin
			bin[GRAY_W-1] = gray[GRAY_W-1];
			for (i = GRAY_W-2; i >= 0; i = i - 1)
				bin[i] = bin[i+1] ^ gray[i];
		end
	endfunction

	// Write clock domain logic
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			w_addr      <= {ADDR_W{1'b0}};
			w_addr_gray <= {GRAY_W{1'b0}};
			w_count     <= {CNT_W{1'b0}};
			o_wr_full   <= 1'b0;
		end else begin
			if (i_wr && !o_wr_full) begin
				mem[w_addr] <= i_wr_data;
				w_addr <= w_addr + 1'b1;
				w_count <= w_count + 1'b1;
			end
			w_addr_gray <= bin2gray(w_addr);
			o_wr_full   <= (w_count == DEPTH);
		end
	end

	// Read clock domain logic
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			r_addr      <= {ADDR_W{1'b0}};
			r_addr_gray <= {GRAY_W{1'b0}};
			r_count     <= {CNT_W{1'b0}};
			r_data_reg  <= {WIDTH{1'b0}};
			o_rd_empty  <= 1'b1;
			o_rd_data   <= {WIDTH{1'b0}};
		end else begin
			if (i_rd && !o_rd_empty) begin
				r_addr <= r_addr + 1'b1;
				r_count <= r_count + 1'b1;
				if (OPT_REGISTER_READS)
					r_data_reg <= mem[r_addr];
			end
			r_addr_gray <= bin2gray(r_addr);
			o_rd_empty  <= (r_count == 0);
			if (OPT_REGISTER_READS)
				o_rd_data <= r_data_reg;
			else
				o_rd_data <= mem[r_addr];
		end
	end

	// Synchronizers for write pointer gray code into read domain
	reg [GRAY_W-1:0] w_gray_sync [0:NFF-1];
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			w_gray_sync[0] <= {GRAY_W{1'b0}};
			for (i = 1; i < NFF; i = i + 1)
				w_gray_sync[i] <= {GRAY_W{1'b0}};
		end else begin
			w_gray_sync[0] <= w_addr_gray;
			for (i = 1; i < NFF; i = i + 1)
				w_gray_sync[i] <= w_gray_sync[i-1];
		end
	end

	// Synchronizers for read pointer gray code into write domain
	reg [GRAY_W-1:0] r_gray_sync [0:NFF-1];
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			r_gray_sync[0] <= {GRAY_W{1'b0}};
			for (i = 1; i < NFF; i = i + 1)
				r_gray_sync[i] <= {GRAY_W{1'b0}};
		end else begin
			r_gray_sync[0] <= r_addr_gray;
			for (i = 1; i < NFF; i = i + 1)
				r_gray_sync[i] <= r_gray_sync[i-1];
		end
	end

	// Read domain: compute write count from synchronized gray code
	wire [CNT_W-1:0] w_count_sync = gray2bin(w_gray_sync[NFF-1]);

	// Write domain: compute read count from synchronized gray code
	wire [CNT_W-1:0] r_count_sync = gray2bin(r_gray_sync[NFF-1]);

	// Update counts in respective domains
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n)
			r_count <= {CNT_W{1'b0}};
		else
			r_count <= w_count_sync;
	end

	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n)
			w_count <= {CNT_W{1'b0}};
		else
			w_count <= r_count_sync;
	end

endmodule
