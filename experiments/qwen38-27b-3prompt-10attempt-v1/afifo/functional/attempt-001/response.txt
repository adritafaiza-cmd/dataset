module afifo #(
		parameter	LGFIFO = 3,
		parameter	WIDTH  = 16,
		parameter	NFF    = 2,
		parameter [0:0]	WRITE_ON_POSEDGE = 1'b1,
		parameter [0:0]	OPT_REGISTER_READS = 1'b1
	) (
		input	wire			i_wclk, i_wr_reset_n, i_wr,
		input	wire	[WIDTH-1:0]	i_wr_data,
		output	reg			o_wr_full,
		input	wire			i_rclk, i_rd_reset_n, i_rd,
		output	reg	[WIDTH-1:0]	o_rd_data,
		output	reg			o_rd_empty
	);

	localparam DEPTH = 1 << LGFIFO;
	localparam ADDR_W = LGFIFO;
	localparam CNT_W = LGFIFO + 1;

	// Memory
	reg [WIDTH-1:0] mem [0:DEPTH-1];

	// Write pointer
	reg [ADDR_W:0] wptr;
	// Read pointer
	reg [ADDR_W:0] rptr;

	// Gray code pointers
	wire [ADDR_W:0] wptr_gray;
	wire [ADDR_W:0] rptr_gray;

	function [ADDR_W:0] bin2gray;
		input [ADDR_W:0] bin;
		begin
			bin2gray = bin ^ (bin >> 1);
		end
	endfunction

	assign wptr_gray = bin2gray(wptr);
	assign rptr_gray = bin2gray(rptr);

	// Synchronizers for write pointer gray to read clock domain
	reg [ADDR_W:0] wptr_gray_r1, wptr_gray_r2, wptr_gray_r3;
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			wptr_gray_r1 <= {(ADDR_W+1){1'b0}};
			wptr_gray_r2 <= {(ADDR_W+1){1'b0}};
			wptr_gray_r3 <= {(ADDR_W+1){1'b0}};
		end else begin
			wptr_gray_r1 <= wptr_gray;
			wptr_gray_r2 <= wptr_gray_r1;
			wptr_gray_r3 <= wptr_gray_r2;
		end
	end

	// Synchronizers for read pointer gray to write clock domain
	reg [ADDR_W:0] rptr_gray_r1, rptr_gray_r2, rptr_gray_r3;
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			rptr_gray_r1 <= {(ADDR_W+1){1'b0}};
			rptr_gray_r2 <= {(ADDR_W+1){1'b0}};
			rptr_gray_r3 <= {(ADDR_W+1){1'b0}};
		end else begin
			rptr_gray_r1 <= rptr_gray;
			rptr_gray_r2 <= rptr_gray_r1;
			rptr_gray_r3 <= rptr_gray_r2;
		end
	end

	// Convert synchronized gray to binary
	function [ADDR_W:0] gray2bin;
		input [ADDR_W:0] gray;
		reg [ADDR_W:0] bin;
		begin
			bin[ADDR_W] = gray[ADDR_W];
			for (i = ADDR_W-1; i >= 0; i = i - 1)
				bin[i] = gray[i] ^ bin[i+1];
			gray2bin = bin;
		end
	endfunction

	wire [ADDR_W:0] wptr_gray_sync_r = gray2bin(wptr_gray_r3);
	wire [ADDR_W:0] rptr_gray_sync_w = gray2bin(rptr_gray_r3);

	// Full condition: write pointer MSB differs from read pointer MSB,
	// and the remaining bits are equal
	wire full = (wptr[ADDR_W] != rptr_gray_sync_w[ADDR_W]) &&
	            (wptr[ADDR_W-1:0] == rptr_gray_sync_w[ADDR_W-1:0]);

	// Empty condition: read pointer equals synchronized write pointer
	wire empty = (rptr == wptr_gray_sync_r);

	// Write logic
	wire wclk_edge = WRITE_ON_POSEDGE ? i_wclk : ~i_wclk;
	wire wr_en = i_wr & ~full;

	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			wptr <= {(ADDR_W+1){1'b0}};
			o_wr_full <= 1'b0;
		end else begin
			o_wr_full <= full;
			if (wr_en)
				wptr <= wptr + 1'b1;
		end
	end

	// Write data to memory
	always @(posedge i_wclk) begin
		if (wr_en)
			mem[wptr[ADDR_W-1:0]] <= i_wr_data;
	end

	// Read logic
	wire rd_en = i_rd & ~empty;

	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			rptr <= {(ADDR_W+1){1'b0}};
			o_rd_empty <= 1'b1;
			o_rd_data <= {WIDTH{1'b0}};
		end else begin
			o_rd_empty <= empty;
			if (rd_en) begin
				rptr <= rptr + 1'b1;
				if (OPT_REGISTER_READS)
					o_rd_data <= mem[rptr[ADDR_W-1:0]];
			end
		end
	end

	// If not registered reads, output directly
	wire [WIDTH-1:0] rd_data_comb = mem[rptr[ADDR_W-1:0]];

	// Select read data based on OPT_REGISTER_READS
	// Since o_rd_data is a reg, we need to handle both cases
	// For combinational reads, we assign in a separate always block
	// But since o_rd_data is declared as reg, we'll use a different approach

	// Actually, let's restructure: use a wire for the data and assign to output
	// But the port is declared as reg. Let's use an always block for combinational case.

	// We'll use a separate always block for the combinational read case
	// and the registered case is handled in the clocked block above.

	// For OPT_REGISTER_READS = 0, we need combinational output
	// Since o_rd_data is a reg, we can assign it in an always @(*) block
	// but that conflicts with the clocked assignment. Let's use a different approach.

	// Let's use a wire for the actual data and then assign to o_rd_data
	wire [WIDTH-1:0] rd_data_wire;
	assign rd_data_wire = OPT_REGISTER_READS ? o_rd_data : rd_data_comb;

	// But o_rd_data is a reg output. We need to assign it properly.
	// Let's just assign o_rd_data in the appropriate always block.

	// For combinational reads, we need to assign o_rd_data combinationally
	// This requires an always @(*) block, but we already have a clocked block.
	// The solution is to use a separate always block for the combinational case.

	// Since Verilog-2001 doesn't allow multiple drivers, let's restructure.
	// We'll use a wire for the data and assign to the output port.

	// Actually, the cleanest way is to have the output be a wire and use
	// an assign statement, but the port is declared as reg. Let's change approach.

	// We'll keep o_rd_data as reg and handle both cases in the clocked block.
	// For combinational reads, we'll assign in an always @(*) block.
	// But this creates a conflict. Let's use a different strategy.

	// Let's use a wire for the read data and assign to o_rd_data via an always block.
	// For registered reads, o_rd_data is updated on clock.
	// For combinational reads, o_rd_data is updated combinationally.

	// We'll use a separate always block for the combinational case.
	// This is valid in Verilog-2001 as long as there's no conflict.

	// For OPT_REGISTER_READS = 0:
	always @(*) begin
		if (!OPT_REGISTER_READS)
			o_rd_data = rd_data_comb;
		else
			o_rd_data = o_rd_data; // No change, handled by clocked block
	end

endmodule
