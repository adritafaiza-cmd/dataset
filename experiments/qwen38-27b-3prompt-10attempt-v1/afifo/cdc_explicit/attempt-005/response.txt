		parameter [0:0]	OPT_REGISTER_READS = 1'b1

		output	reg			o_rd_empty
	);

	localparam PTR_W = LGFIFO + 1;

	reg [PTR_W-1:0] rd_ptr_w_sync13;
	reg [PTR_W-1:0] rd_ptr_w_sync14;
	reg [PTR_W-1:0] rd_ptr_w_sync15;
	reg [PTR_W-1:0] rd_ptr_w_sync16;

	reg [PTR_W-1:0] wr_ptr_r_sync1;
	reg [PTR_W-1:0] wr_ptr_r_sync2;
	reg [PTR_W-1:0] wr_ptr_r_sync3;
	reg [PTR_W-1:0] wr_ptr_r_sync4;

	reg [PTR_W-1:0] wr_ptr_r_sync6;
	reg [PTR_W-1:0] wr_ptr_r_sync7;
	reg [PTR_W-1:0] wr_ptr_r_sync8;

	reg [PTR_W-1:0] wr_ptr_r_sync10;
	reg [PTR_W-1:0] wr_ptr_r_sync11;
	reg [PTR_W-1:0] wr_ptr_r_sync12;
	reg [PTR_W-1:0] wr_ptr_r_sync13;
	reg [PTR_W-1:0] wr_ptr_r_sync14;
	reg [PTR_W-1:0] wr_ptr_r_sync15;
	reg [PTR_W-1:0] wr_ptr_r_sync16;

	wire wclk_negedge = WRITE_ON_POSEDGE ? ~i_wclk : i_wclk;

	wire rclk_edge = i_rclk;
	wire rclk_negedge = ~i_rclk;

	// Write domain logic
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			wr_ptr <= {PTR_W{1'b0}};

			rd_ptr_w_sync2 <= {PTR_W{1'b0}};
			rd_ptr_w_sync3 <= {PTR_W{1'b0}};
			rd_ptr_w_sync4 <= {PTR_W{1'b0}};
			rd_ptr_w_sync5 <= {PTR_W{1'b0}};

			rd_ptr_w_sync4 <= rd_ptr_w_sync5;

			end

			wr_ptr_r_sync2 <= {PTR_W{1'b0}};
			wr_ptr_r_sync3 <= {PTR_W{1'b0}};
			wr_ptr_r_sync4 <= {PTR_W{1'b0}};
			wr_ptr_r_sync5 <= {PTR_W{1'b0}};
			wr_ptr_r_sync6 <= {PTR_W{1'b0}};
			wr_ptr_r_sync7 <= {PTR_W{1'b0}};
			wr_ptr_r_sync8 <= {PTR_W{1'b0}};
			wr_ptr_r_sync9 <= {PTR_W{1'b0}};

			wr_ptr_r_sync11 <= {PTR_W{1'b0}};
			wr_ptr_r_sync12 <= {PTR_W{1'b0}};
			wr_ptr_r_sync13 <= {PTR_W{1'b0}};
			wr_ptr_r_sync14 <= {PTR_W{1'b0}};
			wr_ptr_r_sync15 <= {PTR_W{1'b0}};
			wr_ptr_r_sync16 <= {PTR_W{1'b0}};
			o_rd_data <= {WIDTH{1'b0}};
		end else begin
			// Synchronize write pointer from write domain
			wr_ptr_r_sync1 <= wr_ptr_r_sync2;
			wr_ptr_r_sync2 <= wr_ptr_r_sync3;
			wr_ptr_r_sync3 <= wr_ptr_r_sync4;
			wr_ptr_r_sync4 <= wr_ptr_r_sync5;
			wr_ptr_r_sync5 <= wr_ptr_r_sync6;
			wr_ptr_r_sync6 <= wr_ptr_r_sync7;
			wr_ptr_r_sync7 <= wr_ptr_r_sync8;
			wr_ptr_r_sync8 <= wr_ptr_r_sync9;
			wr_ptr_r_sync9 <= wr_ptr_r_sync10;
			wr_ptr_r_sync10 <= wr_ptr_r_sync11;
			wr_ptr_r_sync11 <= wr_ptr_r_sync12;
			wr_ptr_r_sync12 <= wr_ptr_r_sync13;
			wr_ptr_r_sync13 <= wr_ptr_r_sync14;
			wr_ptr_r_sync14 <= wr_ptr_r_sync15;
			wr_ptr_r_sync15 <= wr_ptr_r_sync16;
			wr_ptr_r_sync16 <= wr_ptr_r;

			// Read operation
			if (i_rd && !o_rd_empty) begin
				if (OPT_REGISTER_READS)
					o_rd_data <= mem[rd_ptr[LGFIFO-1:0]];
				rd_ptr <= rd_ptr + 1'b1;
			end
		end
	end

	// Empty flag in read domain
	always @(*) begin
		if (!i_rd_reset_n)
			o_rd_empty = 1'b1;
		else
			o_rd_empty = (rd_ptr == wr_ptr_r_sync16);
