```verilog
module apb_cdc #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_pclk_i,
    input                      src_preset_ni,
    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,
    input  [ADDR_WIDTH-1:0]    src_paddr_i,
    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH;

    // Source domain signals
    reg  [ADDR_WIDTH-1:0] src_addr_reg;
    reg  [DATA_WIDTH-1:0] src_wdata_reg;
    reg  [DATA_WIDTH/8-1:0] src_strb_reg;
    reg  [2:0] src_pprot_reg;
    reg  src_write_reg;
    reg  src_valid_reg;
    reg  src_valid_sync;
    reg  [1:0] src_valid_sync_r;

    // Destination domain signals
    reg  [ADDR_WIDTH-1:0] dst_addr_reg;
    reg  [DATA_WIDTH-1:0] dst_wdata_reg;
    reg  [DATA_WIDTH/8-1:0] dst_strb_reg;
    reg  [2:0] dst_pprot_reg;
    reg  dst_write_reg;
    reg  dst_valid_reg;
    reg  dst_valid_sync;
    reg  [1:0] dst_valid_sync_r;

    // Response path: dst -> src
    reg  [DATA_WIDTH-1:0] dst_resp_data_reg;
    reg  dst_resp_err_reg;
    reg  dst_resp_valid_reg;
    reg  dst_resp_valid_sync;
    reg  [1:0] dst_resp_valid_sync_r;

    // Source domain response registers
    reg  [DATA_WIDTH-1:0] src_resp_data_reg;
    reg  src_resp_err_reg;
    reg  src_resp_valid_reg;
    reg  src_resp_valid_sync;
    reg  [1:0] src_resp_valid_sync_r;

    // Source state machine
    reg  src_idle;
    reg  src_wait;

    // Destination state machine
    reg  dst_idle;
    reg  dst_setup;
    reg  dst_access;

    // 2-flop synchronizers for src_valid -> dst
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            src_valid_sync_r <= 2'b00;
        end else begin
            src_valid_sync_r <= {src_valid_sync_r[0], src_valid_reg};
        end
    end
    assign src_valid_sync = src_valid_sync_r[1];

    // 2-flop synchronizers for dst_valid -> src
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            dst_valid_sync_r <= 2'b00;
        end else begin
            dst_valid_sync_r <= {dst_valid_sync_r[0], dst_valid_reg};
        end
    end
    assign dst_valid_sync = dst_valid_sync_r[1];

    // 2-flop synchronizers for dst_resp_valid -> src
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            dst_resp_valid_sync_r <= 2'b00;
        end else begin
            dst_resp_valid_sync_r <= {dst_resp_valid_sync_r[0], dst_resp_valid_reg};
        end
    end
    assign dst_resp_valid_sync = dst_resp_valid_sync_r[1];

    // 2-flop synchronizers for src_resp_valid -> dst (not used, but for completeness if needed)
    // Actually, we don't need src_resp_valid to cross to dst.

    // Source domain logic
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_idle <= 1'b1;
            src_wait <= 1'b0;
            src_valid_reg <= 1'b0;
            src_addr_reg <= {ADDR_WIDTH{1'b0}};
            src_wdata_reg <= {DATA_WIDTH{1'b0}};
            src_strb_reg <= {(DATA_WIDTH/8){1'b0}};
            src_pprot_reg <= 3'b000;
            src_write_reg <= 1'b0;
            src_resp_data_reg <= {DATA_WIDTH{1'b0}};
            src_resp_err_reg <= 1'b0;
            src_resp_valid_reg <= 1'b0;
        end else begin
            if (src_idle) begin
                if (src_psel_i) begin
                    src_idle <= 1'b0;
                    src_wait <= 1'b1;
                    src_valid_reg <= 1'b1;
                    src_addr_reg <= src_paddr_i;
                    src_wdata_reg <= src_pwdata_i;
                    src_strb_reg <= src_pstrb_i;
                    src_pprot_reg <= src_pprot_i;
                    src_write_reg <= src_pwrite_i;
                end
            end else if (src_wait) begin
                if (dst_resp_valid_sync) begin
                    src_wait <= 1'b0;
                    src_idle <= 1'b1;
                    src_valid_reg <= 1'b0;
                    src_resp_data_reg <= dst_resp_data_reg;
                    src_resp_err_reg <= dst_resp_err_reg;
                    src_resp_valid_reg <= 1'b1;
                end
            end
        end
    end

    // Source outputs
    assign src_pready_o = src_idle | (src_wait & dst_resp_valid_sync);
    assign src_prdata_o = src_resp_data_reg;
    assign src_pslverr_o = src_resp_err_reg;

    // Destination domain logic
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            dst_idle <= 1'b1;
            dst_setup <= 1'b0;
            dst_access <= 1'b0;
            dst_valid_reg <= 1'b0;
            dst_addr_reg <= {ADDR_WIDTH{1'b0}};
            dst_wdata_reg <= {DATA_WIDTH{1'b0}};
            dst_strb_reg <= {(DATA_WIDTH/8){1'b0}};
            dst_pprot_reg <= 3'b000;
            dst_write_reg <= 1'b0;
            dst_resp_data_reg <= {DATA_WIDTH{1'b0}};
            dst_resp_err_reg <= 1'b0;
            dst_resp_valid_reg <= 1'b0;
        end else begin
            if (dst_idle) begin
                if (src_valid_sync) begin
                    dst_idle <= 1'b0;
                    dst_setup <= 1'b1;
                    dst_valid_reg <= 1'b1;
                    dst_addr_reg <= src_addr_reg;
                    dst_wdata_reg <= src_wdata_reg;
                    dst_strb_reg <= src_strb_reg;
                    dst_pprot_reg <= src_pprot_reg;
                    dst_write_reg <= src_write_reg;
                end
            end else if (dst_setup) begin
                dst_setup <= 1'b0;
                dst_access <= 1'b1;
            end else if (dst_access) begin
                if (dst_pready_i) begin
                    dst_access <= 1'b0;
                    dst_idle <= 1'b1;
                    dst_valid_reg <= 1'b0;
                    dst_resp_data_reg <= dst_prdata_i;
                    dst_resp_err_reg <= dst_pslverr_i;
                    dst_resp_valid_reg <= 1'b1;
                end
            end
        end
    end

    // Destination outputs
    assign dst_psel_o = dst_setup | dst_access;
    assign dst_penable_o = dst_access;
    assign dst_pwrite_o = dst_write_reg;
    assign dst_paddr_o = dst_addr_reg;
    assign dst_pwdata_o = dst_wdata_reg;
    assign dst_pstrb_o = dst_strb_reg;
    assign dst_pprot_o = dst_pprot_reg;

endmodule
