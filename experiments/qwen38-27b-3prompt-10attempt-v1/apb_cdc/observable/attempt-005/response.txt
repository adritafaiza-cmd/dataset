    input  [DATA_WIDTH-1:0]    src_pwdata_i,

    output                     dst_psel_o,

    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,

    reg [PTR_W-1:0] req_wr_ptr, req_rd_ptr;
    reg [PTR_W:0] req_count;

    reg [DATA_WIDTH-1:0] rsp_data_fifo [0:DEPTH-1];
    reg rsp_err_fifo [0:DEPTH-1];
    reg [PTR_W-1:0] rsp_wr_ptr, rsp_rd_ptr;
    reg [PTR_W:0] rsp_count;

    reg src_pready_r;
    reg [DATA_WIDTH-1:0] src_prdata_r;
    reg src_pslverr_r;

    // Destination side state

    reg dst_penable_r;
    reg dst_pwrite_r;
    reg [ADDR_WIDTH-1:0] dst_paddr_r;
    reg [DATA_WIDTH-1:0] dst_pwdata_r;
    reg [DATA_WIDTH/8-1:0] dst_pstrb_r;
    reg [2:0] dst_pprot_r;

    reg [2:0] req_count_sync;
    reg [2:0] rsp_count_sync;

    wire req_wr_en = src_psel_i & src_penable_i & src_pready_i;
    wire req_rd_en = (req_count > 0) & (rsp_count < DEPTH);

    // Response FIFO read (src clock)
    wire rsp_rd_en = src_busy & (rsp_count > 0);

    // Response FIFO write (dst clock)
    wire rsp_wr_en = dst_psel_r & dst_penable_r & dst_pready_i;

    // and destination is not currently busy
    wire dst_idle = !dst_psel_r;
    wire dst_start = dst_idle & (req_count > 0);

    // Pready is high when:
    // 1. Not busy and no pending request in FIFO (can accept new)
    // 2. Busy and response is available (can complete current)
    wire src_pready_w = !src_busy ? (req_count == 0) : (rsp_count > 0);

            end else if (rsp_rd_en) begin
                src_busy <= 1'b0;
            end

            if (rsp_rd_en) begin
                rsp_rd_ptr <= rsp_rd_ptr + 1'b1;
            end

            // Update response count
            if (rsp_rd_en && rsp_wr_en) begin
                rsp_count <= rsp_count;
            end else if (rsp_rd_en) begin
                rsp_count <= rsp_count - 1'b1;
            end else if (rsp_wr_en) begin
                rsp_count <= rsp_count + 1'b1;
            end
        end
    end

    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            req_rd_ptr <= {PTR_W{1'b0}};
            rsp_wr_ptr <= {PTR_W{1'b0}};
            dst_psel_r <= 1'b0;
            dst_penable_r <= 1'b0;
            dst_pwrite_r <= 1'b0;
            dst_paddr_r <= {ADDR_WIDTH{1'b0}};
            dst_pwdata_r <= {DATA_WIDTH{1'b0}};
            dst_pstrb_r <= {DATA_WIDTH/8{1'b0}};
            dst_pprot_r <= 3'b0;
        end else begin
            // Read from request FIFO
            if (req_rd_en) begin
                req_rd_ptr <= req_rd_ptr + 1'b1;
            end

            // Start new transaction
            if (dst_start) begin
                dst_psel_r <= 1'b1;
                dst_penable_r <= 1'b0;
                dst_pwrite_r <= req_write_fifo[req_rd_ptr];
                dst_paddr_r <= req_addr_fifo[req_rd_ptr];
                dst_pwdata_r <= req_fifo[req_rd_ptr];
                dst_pstrb_r <= req_strb_fifo[req_rd_ptr];
                dst_pprot_r <= req_prot_fifo[req_rd_ptr];
            end else if (dst_psel_r && !dst_penable_r) begin
                dst_penable_r <= 1'b1;
            end else if (dst_psel_r && dst_penable_r && dst_pready_i) begin
                dst_psel_r <= 1'b0;
                dst_penable_r <= 1'b0;
            end

            // Write to response FIFO
            if (rsp_wr_en) begin
                rsp_data_fifo[rsp_wr_ptr] <= dst_prdata_i;
                rsp_err_fifo[rsp_wr_ptr] <= dst_pslverr_i;
                rsp_wr_ptr <= rsp_wr_ptr + 1'b1;
            end
        end
    end

        if (!dst_preset_ni) begin
            req_count_sync <= 3'b0;
        end else begin
            req_count_sync <= req_count;
        end
    end

        if (!src_preset_ni) begin
            rsp_count_sync <= 3'b0;
        end else begin
            rsp_count_sync <= rsp_count;
        end
