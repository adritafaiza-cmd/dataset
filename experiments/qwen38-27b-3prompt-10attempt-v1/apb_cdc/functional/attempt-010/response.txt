module apb_cdc #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_pclk_i,
    input                      src_preset_ni,
    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,
    input  [ADDR_WIDTH-1:0]    src_paddr_i,
    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    // Source side state
    reg src_state;
    localparam SRC_IDLE = 1'b0;
    localparam SRC_WAIT = 1'b1;

    // Destination side state
    reg dst_state;
    localparam DST_IDLE = 1'b0;
    localparam DST_SETUP = 1'b1;
    localparam DST_ACCESS = 1'b10;

    // FIFO control signals
    reg src_wr_en;
    reg dst_rd_en;
    reg fifo_empty;
    reg fifo_full;

    // FIFO data
    reg [ADDR_WIDTH-1:0] fifo_addr;
    reg [DATA_WIDTH-1:0] fifo_wdata;
    reg [DATA_WIDTH/8-1:0] fifo_strb;
    reg [2:0] fifo_pprot;
    reg fifo_write;

    // Synchronizers
    reg [1:0] dst_pready_sync;
    reg [1:0] dst_pslverr_sync;
    reg [1:0] dst_prdata_sync; // We'll use a different approach for data

    // Actually, let's use a simpler approach with a single outstanding transaction
    // and synchronizers for the control signals.

    // Source side: capture request when in IDLE and PSEL is high
    wire src_req_valid = src_psel_i && !src_penable_i && !src_state;

    // Destination side: generate APB signals
    assign dst_psel_o = (dst_state == DST_SETUP) || (dst_state == DST_ACCESS);
    assign dst_penable_o = (dst_state == DST_ACCESS);
    assign dst_pwrite_o = fifo_write;
    assign dst_paddr_o = fifo_addr;
    assign dst_pwdata_o = fifo_wdata;
    assign dst_pstrb_o = fifo_strb;
    assign dst_pprot_o = fifo_pprot;

    // Source side outputs
    assign src_pready_o = (src_state == SRC_IDLE) && !src_psel_i;
    assign src_prdata_o = dst_prdata_i;
    assign src_pslverr_o = dst_pslverr_i;

    // Synchronizers for destination ready and error
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni)
            dst_pready_sync <= 2'b00;
        else
            dst_pready_sync <= {dst_pready_sync[0], dst_pready_i};
    end

    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni)
            dst_pslverr_sync <= 2'b00;
        else
            dst_pslverr_sync <= {dst_pslverr_sync[0], dst_pslverr_i};
    end

    // Synchronizer for read data - we need to capture it when the transfer completes
    reg [DATA_WIDTH-1:0] dst_prdata_reg;
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni)
            dst_prdata_reg <= {DATA_WIDTH{1'b0}};
        else if (dst_state == DST_ACCESS && dst_pready_i)
            dst_prdata_reg <= dst_prdata_i;
    end

    // Synchronizer for slave error
    reg dst_pslverr_reg;
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni)
            dst_pslverr_reg <= 1'b0;
        else if (dst_state == DST_ACCESS && dst_pready_i)
            dst_pslverr_reg <= dst_pslverr_i;
    end

    // Synchronizers to source domain
    reg [1:0] dst_done_sync;
    reg [1:0] dst_pslverr_src_sync;
    reg [DATA_WIDTH-1:0] dst_prdata_src_sync;

    // We need to synchronize the completion signal
    reg dst_done;
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni)
            dst_done <= 1'b0;
        else
            dst_done <= (dst_state == DST_ACCESS) && dst_pready_i;
    end

    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni)
            dst_done_sync <= 2'b00;
        else
            dst_done_sync <= {dst_done_sync[0], dst_done};
    end

    // Synchronize read data and error to source domain
    // We'll use a simple approach: when dst_done is synchronized, capture the data
    reg [DATA_WIDTH-1:0] src_prdata_reg;
    reg src_pslverr_reg;

    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_prdata_reg <= {DATA_WIDTH{1'b0}};
            src_pslverr_reg <= 1'b0;
        end else if (dst_done_sync[1] && !dst_done_sync[0]) begin
            // This is a rising edge of the synchronized done signal
            // But we need the data to be stable. Let's use a different approach.
        end
    end

    // Let's use a simpler approach: since we have only one outstanding transaction,
    // we can synchronize the data and error when the transfer completes.
    
    // Actually, let's redesign with a cleaner approach using a FIFO for the request
    // and synchronizers for the response.

    // Reset the module
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_state <= SRC_IDLE;
            src_wr_en <= 1'b0;
        end else begin
            case (src_state)
                SRC_IDLE: begin
                    src_wr_en <= 1'b0;
                    if (src_psel_i && !src_penable_i) begin
                        src_wr_en <= 1'b1;
                        src_state <= SRC_WAIT;
                    end
                end
                SRC_WAIT: begin
                    src_wr_en <= 1'b0;
                    if (dst_done_sync[1] && !dst_done_sync[0]) begin
                        src_state <= SRC_IDLE;
                    end
                end
                default: src_state <= SRC_IDLE;
            endcase
        end
    end

    // Capture request data
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            fifo_addr <= {ADDR_WIDTH{1'b0}};
            fifo_wdata <= {DATA_WIDTH{1'b0}};
            fifo_strb <= {(DATA_WIDTH/8){1'b0}};
            fifo_pprot <= 3'b000;
            fifo_write <= 1'b0;
        end else if (src_wr_en) begin
            fifo_addr <= src_paddr_i;
            fifo_wdata <= src_pwdata_i;
            fifo_strb <= src_pstrb_i;
            fifo_pprot <= src_pprot_i;
            fifo_write <= src_pwrite_i;
        end
    end

    // Destination side state machine
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            dst_state <= DST_IDLE;
            dst_rd_en <= 1'b0;
        end else begin
            dst_rd_en <= 1'b0;
            case (dst_state)
                DST_IDLE: begin
                    if (fifo_empty == 1'b0) begin
                        dst_state <= DST_SETUP;
                        dst_rd_en <= 1'b1;
                    end
                end
                DST_SETUP: begin
                    dst_state <= DST_ACCESS;
                end
                DST_ACCESS: begin
                    if (dst_pready_i) begin
                        dst_state <= DST_IDLE;
                    end
                end
                default: dst_state <= DST_IDLE;
            endcase
        end
    end

    // Simple FIFO implementation (depth = 2^LOG_DEPTH)
    // For simplicity, we'll use a single-entry FIFO since LOG_DEPTH=1 means 2 entries
    // but we only need one outstanding transaction.
    
    // Let's use a simple register-based approach since we only have one outstanding transaction
    reg fifo_valid;
    
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni)
            fifo_valid <= 1'b0;
        else if (src_wr_en)
            fifo_valid <= 1'b1;
        else if (dst_rd_en_sync[1] && !dst_rd_en_sync[0])
            fifo_valid <= 1'b0;
    end

    // Synchronize dst_rd_en to source domain
    reg [1:0] dst_rd_en_sync;
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni)
            dst_rd_en_sync <= 2'b00;
        else
            dst_rd_en_sync <= {dst_rd_en_sync[0], dst_rd_en};
    end

    // Synchronize fifo_valid to destination domain
    reg [1:0] fifo_valid_sync;
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni)
            fifo_valid_sync <= 2'b00;
        else
            fifo_valid_sync <= {fifo_valid_sync[0], fifo_valid};
    end

    assign fifo_empty = !fifo_valid_sync[1];
    assign fifo_full = fifo_valid;

    // Synchronize read data and error to source domain
    // We'll use a handshake approach
    reg [DATA_WIDTH-1:0] src_prdata_reg;
    reg src_pslverr_reg;
    reg src_data_valid;

    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_prdata_reg <= {DATA_WIDTH{1'b0}};
            src_pslverr_reg <= 1'b0;
            src_data_valid <= 1'b0;
        end else if (dst_done_sync[1] && !dst_done_sync[0]) begin
            src_prdata_reg <= dst_prdata_reg;
            src_pslverr_reg <= dst_pslverr_reg;
            src_data_valid <= 1'b1;
        end else if (src_state == SRC_IDLE && src_psel_i && !src_penable_i) begin
            src_data_valid <= 1'b0;
        end
    end

    assign src_prdata_o = src_prdata_reg;
    assign src_pslverr_o = src_pslverr_reg;

endmodule
