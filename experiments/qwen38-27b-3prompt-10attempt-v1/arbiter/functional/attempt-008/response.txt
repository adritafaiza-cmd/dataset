```verilog
module arbiter #
(
    parameter PORTS = 4,
    parameter ARB_TYPE_ROUND_ROBIN = 0,
    parameter ARB_BLOCK = 0,
    parameter ARB_BLOCK_ACK = 1,
    parameter ARB_LSB_HIGH_PRIORITY = 0
)
(
    input  wire                     clk,
    input  wire                     rst,
    input  wire [PORTS-1:0]         request,
    input  wire [PORTS-1:0]         acknowledge,
    output wire [PORTS-1:0]         grant,
    output wire                     grant_valid,
    output wire [$clog2(PORTS)-1:0] grant_encoded
);

    localparam NUM_BITS = $clog2(PORTS);

    reg [PORTS-1:0] grant_reg;
    reg [NUM_BITS-1:0] grant_encoded_reg;
    reg grant_valid_reg;

    // For round-robin, track the last granted port
    reg [NUM_BITS-1:0] last_grant;

    // Internal signals for arbitration
    wire [PORTS-1:0] arb_request;
    wire [PORTS-1:0] arb_grant;
    wire arb_valid;
    wire [NUM_BITS-1:0] arb_encoded;

    // Function to find the first set bit (LSB) or last set bit (MSB) depending on priority
    // We'll implement priority encoder logic

    // For fixed priority:
    // If ARB_LSB_HIGH_PRIORITY, LSB has highest priority
    // Otherwise, MSB has highest priority

    // For round-robin:
    // Rotate the request vector so that the port after last_grant has highest priority

    // Let's compute the grant based on the mode

    // First, determine the effective request for arbitration
    // In blocking mode, we hold the grant until release condition

    // Let's create a combinational block to compute the next grant

    // Priority encoder for fixed priority
    // LSB high priority: find lowest index with request
    // MSB high priority: find highest index with request

    // For round-robin, we rotate the request vector

    // Let's define helper functions

    // Function to get the index of the first (LSB) set bit
    function [NUM_BITS-1:0] find_lsb;
        input [PORTS-1:0] vec;
        integer i;
        begin
            find_lsb = 0;
            for (i = 0; i < PORTS; i = i + 1) begin
                if (vec[i]) begin
                    find_lsb = i[NUM_BITS-1:0];
                    disable find_lsb;
                end
            end
        end
    endfunction

    // Function to get the index of the last (MSB) set bit
    function [NUM_BITS-1:0] find_msb;
        input [PORTS-1:0] vec;
        integer i;
        begin
            find_msb = 0;
            for (i = PORTS-1; i >= 0; i = i - 1) begin
                if (vec[i]) begin
                    find_msb = i[NUM_BITS-1:0];
                    disable find_msb;
                end
            end
        end
    endfunction

    // For round-robin, we need to rotate the request vector
    // The port after last_grant should have highest priority
    // We can do this by rotating the request vector left by (last_grant + 1) positions
    // Then finding the LSB (or MSB depending on ARB_LSB_HIGH_PRIORITY) of the rotated vector
    // Then rotating the index back

    // Let's compute the rotated request
    wire [PORTS-1:0] rotated_request;
    wire [NUM_BITS-1:0] rotation_amount;

    assign rotation_amount = (last_grant + 1) % PORTS;

    // Rotate left by rotation_amount
    // rotated_request[i] = request[(i - rotation_amount + PORTS) % PORTS]
    genvar gi;
    generate
        for (gi = 0; gi < PORTS; gi = gi + 1) begin : gen_rot
            assign rotated_request[gi] = request[(gi - rotation_amount + PORTS) % PORTS];
        end
    endgenerate

    // Now compute the grant based on the mode
    // We'll use a combinational block

    wire [PORTS-1:0] next_grant;
    wire next_valid;
    wire [NUM_BITS-1:0] next_encoded;

    // Determine which request vector to use and which priority scheme
    // For round-robin: use rotated_request, and the priority is based on ARB_LSB_HIGH_PRIORITY
    //   If ARB_LSB_HIGH_PRIORITY, LSB of rotated_request wins -> corresponds to the port right after last_grant
    //   If not ARB_LSB_HIGH_PRIORITY, MSB of rotated_request wins -> corresponds to the port farthest from last_grant
    // For fixed priority: use request, and priority is based on ARB_LSB_HIGH_PRIORITY

    // Let's compute the grant index in the rotated space for round-robin
    wire [NUM_BITS-1:0] rotated_grant_idx;
    wire rotated_valid;

    // For round-robin with ARB_LSB_HIGH_PRIORITY: find LSB of rotated_request
    // For round-robin without ARB_LSB_HIGH_PRIORITY: find MSB of rotated_request
    // For fixed priority with ARB_LSB_HIGH_PRIORITY: find LSB of request
    // For fixed priority without ARB_LSB_HIGH_PRIORITY: find MSB of request

    // We can use conditional logic

    // Let's compute both possibilities and select

    wire [NUM_BITS-1:0] lsb_idx_rotated;
    wire [NUM_BITS-1:0] msb_idx_rotated;
    wire [NUM_BITS-1:0] lsb_idx_fixed;
    wire [NUM_BITS-1:0] msb_idx_fixed;

    assign lsb_idx_rotated = find_lsb(rotated_request);
    assign msb_idx_rotated = find_msb(rotated_request);
    assign lsb_idx_fixed = find_lsb(request);
    assign msb_idx_fixed = find_msb(request);

    // For round-robin, the actual grant index is (rotated_grant_idx + rotation_amount) % PORTS
    // Because we rotated left by rotation_amount, so index i in rotated corresponds to (i + rotation_amount) % PORTS in original

    wire [NUM_BITS-1:0] rr_grant_idx;
    wire rr_valid;

    // Select the rotated grant index based on ARB_LSB_HIGH_PRIORITY
    wire [NUM_BITS-1:0] sel_rotated_idx;
    assign sel_rotated_idx = ARB_LSB_HIGH_PRIORITY ? lsb_idx_rotated : msb_idx_rotated;

    // Convert from rotated index to original index
    // original_idx = (sel_rotated_idx + rotation_amount) % PORTS
    wire [NUM_BITS:0] rr_sum;
    assign rr_sum = sel_rotated_idx + rotation_amount;
    assign rr_grant_idx = rr_sum[NUM_BITS-1:0]; // Since PORTS is power of 2, modulo is just taking lower bits
    assign rr_valid = |rotated_request;

    // For fixed priority
    wire [NUM_BITS-1:0] fp_grant_idx;
    wire fp_valid;

    assign fp_grant_idx = ARB_LSB_HIGH_PRIORITY ? lsb_idx_fixed : msb_idx_fixed;
    assign fp_valid = |request;

    // Select between round-robin and fixed priority
    wire [NUM_BITS-1:0] comb_grant_idx;
    wire comb_valid;

    assign comb_grant_idx = ARB_TYPE_ROUND_ROBIN ? rr_grant_idx : fp_grant_idx;
    assign comb_valid = ARB_TYPE_ROUND_ROBIN ? rr_valid : fp_valid;

    // Now handle blocking
    // If ARB_BLOCK is set, hold the grant until:
    //   If ARB_BLOCK_ACK is nonzero: until acknowledge for the granted port is asserted
    //   If ARB_BLOCK_ACK is zero: until the request for the granted port drops

    // We need to track the current grant state

    // Let's use a state machine approach
    // State: IDLE or BLOCKED
    // In IDLE: grant is combinational from request
    // In BLOCKED: grant is held

    // Actually, let's think about this more carefully.
    // The grant output should reflect the current grant.
    // In non-blocking mode, grant is purely combinational.
    // In blocking mode, once a grant is given, it's held until the release condition.

    // Let's use a register to hold the grant in blocking mode

    // For simplicity, let's always use a register for the grant, and update it based on the mode

    // In non-blocking mode: grant_reg <= comb_grant (combinational)
    // In blocking mode: 
    //   If not currently blocked, and comb_valid, then grant_reg <= comb_grant, enter blocked state
    //   If currently blocked, check release condition
    //     If ARB_BLOCK_ACK: release when acknowledge[grant_reg] is asserted
    //     If not ARB_BLOCK_ACK: release when request[grant_reg] is deasserted
    //   When released, grant_reg <= 0, exit blocked state

    // Let's implement this

    reg blocked;

    // One-hot grant from index
    function [PORTS-1:0] idx_to_onehot;
        input [NUM_BITS-1:0] idx;
        integer i;
        begin
            idx_to_onehot = 0;
            for (i = 0; i < PORTS; i = i + 1) begin
                if (i == idx) begin
                    idx_to_onehot[i] = 1;
                    disable idx_to_onehot;
                end
            end
        end
    endfunction

    wire [PORTS-1:0] comb_grant_onehot;
    assign comb_grant_onehot = comb_valid ? idx_to_onehot(comb_grant_idx) : 0;

    // Release condition for blocking
    wire release;
    wire ack_release;
    wire req_release;

    assign ack_release = acknowledge[grant_encoded_reg];
    assign req_release = ~request[grant_encoded_reg];

    assign release = ARB_BLOCK_ACK ? ack_release : req_release;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            grant_reg <= 0;
            grant_encoded_reg <= 0;
            grant_valid_reg <= 0;
            blocked <= 0;
            last_grant <= 0;
        end else begin
            if (ARB_BLOCK) begin
                if (!blocked) begin
                    if (comb_valid) begin
                        grant_reg <= comb_grant_onehot;
                        grant_encoded_reg <= comb_grant_idx;
                        grant_valid_reg <= 1;
                        blocked <= 1;
                        last_grant <= comb_grant_idx;
                    end else begin
                        grant_reg <= 0;
                        grant_encoded_reg <= 0;
                        grant_valid_reg <= 0;
                    end
                end else begin
                    // Currently blocked
                    if (release) begin
                        grant_reg <= 0;
                        grant_encoded_reg <= 0;
                        grant_valid_reg <= 0;
                        blocked <= 0;
                    end else begin
                        // Hold the grant
                        // grant_reg, grant_encoded_reg, grant_valid_reg remain unchanged
                    end
                end
            end else begin
                // Non-blocking mode
                if (comb_valid) begin
                    grant_reg <= comb_grant_onehot;
                    grant_encoded_reg <= comb_grant_idx;
                    grant_valid_reg <= 1;
                    last_grant <= comb_grant_idx;
                end else begin
                    grant_reg <= 0;
                    grant_encoded_reg <= 0;
                    grant_valid_reg <= 0;
                end
            end
        end
    end

    assign grant = grant_reg;
    assign grant_valid = grant_valid_reg;
    assign grant_encoded = grant_encoded_reg;

endmodule
