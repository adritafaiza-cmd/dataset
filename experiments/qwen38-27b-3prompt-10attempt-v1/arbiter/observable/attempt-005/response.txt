module arbiter #
(
    parameter PORTS = 4,
    parameter ARB_TYPE_ROUND_ROBIN = 0,
    parameter ARB_BLOCK = 0,
    parameter ARB_BLOCK_ACK = 1,
    parameter ARB_LSB_HIGH_PRIORITY = 0
)
(
    input  wire                     clk,
    input  wire                     rst,
    input  wire [PORTS-1:0]         request,
    input  wire [PORTS-1:0]         acknowledge,
    output wire [PORTS-1:0]         grant,
    output wire                     grant_valid,
    output wire [$clog2(PORTS)-1:0] grant_encoded
);

    localparam NUM_PORTS = PORTS;
    localparam ENC_WIDTH = $clog2(PORTS);

    // State registers
    reg [NUM_PORTS-1:0] grant_reg;
    reg                 grant_valid_reg;
    reg [ENC_WIDTH-1:0] grant_encoded_reg;
    reg [NUM_PORTS-1:0] last_grant_mask; // For round-robin: mask of ports with higher priority than last granted

    // Combinational logic for next state
    wire [NUM_PORTS-1:0] next_grant;
    wire                 next_grant_valid;
    wire [ENC_WIDTH-1:0] next_grant_encoded;

    // Helper: find first set bit (LSB first)
    function [NUM_PORTS-1:0] find_first_set;
        input [NUM_PORTS-1:0] mask;
        integer i;
        begin
            find_first_set = {NUM_PORTS{1'b0}};
            for (i = 0; i < NUM_PORTS; i = i + 1) begin
                if (mask[i]) begin
                    find_first_set[i] = 1'b1;
                    break;
                end
            end
        end
    endfunction

    // Helper: find last set bit (MSB first)
    function [NUM_PORTS-1:0] find_last_set;
        input [NUM_PORTS-1:0] mask;
        integer i;
        begin
            find_last_set = {NUM_PORTS{1'b0}};
            for (i = NUM_PORTS-1; i >= 0; i = i - 1) begin
                if (mask[i]) begin
                    find_last_set[i] = 1'b1;
                    break;
                end
            end
        end
    endfunction

    // Helper: encode one-hot to binary
    function [ENC_WIDTH-1:0] onehot_to_binary;
        input [NUM_PORTS-1:0] onehot;
        integer i;
        begin
            onehot_to_binary = {ENC_WIDTH{1'b0}};
            for (i = 0; i < NUM_PORTS; i = i + 1) begin
                if (onehot[i]) begin
                    onehot_to_binary = i[ENC_WIDTH-1:0];
                    break;
                end
            end
        end
    endfunction

    // Determine the grant based on arbiter type
    wire [NUM_PORTS-1:0] arbiter_grant;
    wire                 arbiter_grant_valid;

    generate
        if (ARB_TYPE_ROUND_ROBIN) begin
            // Round-robin: rotate request based on last grant
            // Create a rotated request where ports after last_grant have higher priority
            wire [NUM_PORTS-1:0] rotated_request;
            wire [NUM_PORTS-1:0] rotated_request_high;
            wire [NUM_PORTS-1:0] rotated_request_low;
            wire [NUM_PORTS-1:0] rotated_request_combined;

            // If last_grant_mask is all zeros (no previous grant), treat as LSB high priority
            // last_grant_mask has bits set for ports with index > last_granted_port
            // We want to prioritize ports in order: last_granted+1, ..., PORTS-1, 0, ..., last_granted
            
            // Simpler approach: create a priority vector
            // For round-robin, we can use a rotating priority scheme
            // Let's use a different approach: find the first request after the last grant, wrapping around
            
            wire [NUM_PORTS-1:0] request_after_last;
            wire [NUM_PORTS-1:0] request_before_last;
            wire                 has_request_after;
            wire                 has_request_before;

            // request_after_last: bits from last_granted+1 to PORTS-1
            // request_before_last: bits from 0 to last_granted
            // We need to know the last granted port index
            
            // Alternative: use a shift register approach
            // Let's use a simpler method: create a priority mask
            
            // For round-robin, we can compute the grant as follows:
            // 1. If there's a request at a port higher than last_granted, grant the lowest such port
            // 2. Otherwise, if there's a request at a port lower than or equal to last_granted, grant the lowest such port
            
            wire [NUM_PORTS-1:0] higher_priority_request;
            wire [NUM_PORTS-1:0] lower_priority_request;
            wire                 higher_valid;
            wire                 lower_valid;

            // higher_priority_request: request bits where index > last_granted_index
            // lower_priority_request: request bits where index <= last_granted_index
            
            // We need to extract the last granted index from last_grant_mask
            // last_grant_mask has bits set for all ports with index > last_granted_index
            // So the last granted index is the position of the lowest set bit in ~last_grant_mask (if any)
            // Actually, let's store the last granted index directly
            
            // Let me restructure: store last_granted_index instead
            // But we already have last_grant_mask. Let's compute from it.
            
            // If last_grant_mask is 0, it means no previous grant or last grant was at port 0
            // Let's use a different state: last_granted_port
            
            // I'll restructure the state to use last_granted_port
        end
        else begin
            // Fixed priority
            if (ARB_LSB_HIGH_PRIORITY) begin
                assign arbiter_grant = find_first_set(request);
            end
            else begin
                assign arbiter_grant = find_last_set(request);
            end
            assign arbiter_grant_valid = |request;
        end
    endgenerate

    // Let me redo this with a cleaner approach using last_granted_port
    reg [ENC_WIDTH-1:0] last_granted_port;

    generate
        if (ARB_TYPE_ROUND_ROBIN) begin
            wire [NUM_PORTS-1:0] request_high;
            wire [NUM_PORTS-1:0] request_low;
            wire                 high_valid;
            wire                 low_valid;
            wire [NUM_PORTS-1:0] grant_high;
            wire [NUM_PORTS-1:0] grant_low;

            // Ports with index > last_granted_port
            genvar i;
            for (i = 0; i < NUM_PORTS; i = i + 1) begin : gen_high
                assign request_high[i] = (i > last_granted_port) ? request[i] : 1'b0;
            end

            // Ports with index <= last_granted_port
            for (i = 0; i < NUM_PORTS; i = i + 1) begin : gen_low
                assign request_low[i] = (i <= last_granted_port) ? request[i] : 1'b0;
            end

            assign high_valid = |request_high;
            assign low_valid  = |request_low;

            assign grant_high = find_first_set(request_high);
            assign grant_low  = find_first_set(request_low);

            // Prefer higher priority (ports after last grant)
            assign arbiter_grant = high_valid ? grant_high : grant_low;
            assign arbiter_grant_valid = high_valid | low_valid;
        end
        else begin
            if (ARB_LSB_HIGH_PRIORITY) begin
                assign arbiter_grant = find_first_set(request);
            end
            else begin
                assign arbiter_grant = find_last_set(request);
            end
            assign arbiter_grant_valid = |request;
        end
    endgenerate

    // Blocking logic
    wire block_active;
    wire release_event;

    generate
        if (ARB_BLOCK) begin
            if (ARB_BLOCK_ACK) begin
                // Block until acknowledge
                assign block_active = grant_valid_reg & ~(|acknowledge);
                assign release_event = grant_valid_reg & (|acknowledge);
            end
            else begin
                // No blocking (or always release)
                assign block_active = 1'b0;
                assign release_event = 1'b1;
            end
        end
        else begin
            assign block_active = 1'b0;
            assign release_event = 1'b1;
        end
    endgenerate

    // Next state logic
    wire [NUM_PORTS-1:0] next_grant_comb;
    wire                 next_grant_valid_comb;
    wire [ENC_WIDTH-1:0] next_grant_encoded_comb;

    assign next_grant_comb = block_active ? grant_reg : arbiter_grant;
    assign next_grant_valid_comb = block_active ? grant_valid_reg : arbiter_grant_valid;
    assign next_grant_encoded_comb = block_active ? grant_encoded_reg : onehot_to_binary(arbiter_grant);

    // Sequential logic
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            grant_reg <= {NUM_PORTS{1'b0}};
            grant_valid_reg <= 1'b0;
            grant_encoded_reg <= {ENC_WIDTH{1'b0}};
            last_granted_port <= {ENC_WIDTH{1'b0}};
        end
        else begin
            grant_reg <= next_grant_comb;
            grant_valid_reg <= next_grant_valid_comb;
            grant_encoded_reg <= next_grant_encoded_comb;
            
            // Update last_granted_port for round-robin
            if (ARB_TYPE_ROUND_ROBIN && next_grant_valid_comb && !block_active) begin
                last_granted_port <= next_grant_encoded_comb;
            end
        end
    end

    // Outputs
    assign grant = grant_reg;
    assign grant_valid = grant_valid_reg;
    assign grant_encoded = grant_encoded_reg;

endmodule
