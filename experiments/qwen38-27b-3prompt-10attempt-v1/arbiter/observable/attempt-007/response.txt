```verilog
module arbiter #
(
    parameter PORTS = 4,
    parameter ARB_TYPE_ROUND_ROBIN = 0,
    parameter ARB_BLOCK = 0,
    parameter ARB_BLOCK_ACK = 1,
    parameter ARB_LSB_HIGH_PRIORITY = 0
)
(
    input  wire                     clk,
    input  wire                     rst,
    input  wire [PORTS-1:0]         request,
    input  wire [PORTS-1:0]         acknowledge,
    output wire [PORTS-1:0]         grant,
    output wire                     grant_valid,
    output wire [$clog2(PORTS)-1:0] grant_encoded
);

    localparam GRANT_WIDTH = $clog2(PORTS);

    reg [PORTS-1:0] grant_reg;
    reg             grant_valid_reg;
    reg [GRANT_WIDTH-1:0] grant_encoded_reg;

    // For round-robin, track the last granted index
    reg [GRANT_WIDTH-1:0] last_grant_idx;

    // For blocking, track if a grant is currently held
    reg block_hold;

    // Function to find the lowest set bit index (LSB priority)
    function [GRANT_WIDTH-1:0] find_lsb;
        input [PORTS-1:0] mask;
        integer i;
        begin
            find_lsb = 0;
            for (i = 0; i < PORTS; i = i + 1) begin
                if (mask[i]) begin
                    find_lsb = i[GRANT_WIDTH-1:0];
                    disable find_lsb;
                end
            end
        end
    endfunction

    // Function to find the highest set bit index (MSB priority)
    function [GRANT_WIDTH-1:0] find_msb;
        input [PORTS-1:0] mask;
        integer i;
        begin
            find_msb = 0;
            for (i = PORTS-1; i >= 0; i = i - 1) begin
                if (mask[i]) begin
                    find_msb = i[GRANT_WIDTH-1:0];
                    disable find_msb;
                end
            end
        end
    endfunction

    // Round-robin: find the next requester after last_grant_idx
    function [GRANT_WIDTH-1:0] find_round_robin;
        input [PORTS-1:0] mask;
        input [GRANT_WIDTH-1:0] last_idx;
        integer i;
        integer start;
        begin
            find_round_robin = 0;
            start = last_idx + 1;
            if (start >= PORTS) start = 0;
            for (i = 0; i < PORTS; i = i + 1) begin
                if (start + i >= PORTS) begin
                    if (mask[(start + i) % PORTS]) begin
                        find_round_robin = ((start + i) % PORTS)[GRANT_WIDTH-1:0];
                        disable find_round_robin;
                    end
                end else begin
                    if (mask[start + i]) begin
                        find_round_robin = (start + i)[GRANT_WIDTH-1:0];
                        disable find_round_robin;
                    end
                end
            end
        end
    endfunction

    wire [PORTS-1:0] next_grant;
    wire             next_grant_valid;
    wire [GRANT_WIDTH-1:0] next_grant_encoded;

    // Determine the next grant based on arbiter type
    wire [PORTS-1:0] active_request;
    assign active_request = request;

    // Fixed priority
    wire [GRANT_WIDTH-1:0] fixed_priority_idx;
    wire                   fixed_priority_valid;
    assign fixed_priority_valid = |active_request;
    assign fixed_priority_idx = ARB_LSB_HIGH_PRIORITY ? find_lsb(active_request) : find_msb(active_request);

    // Round-robin
    wire [GRANT_WIDTH-1:0] round_robin_idx;
    wire                   round_robin_valid;
    assign round_robin_valid = |active_request;
    assign round_robin_idx = find_round_robin(active_request, last_grant_idx);

    // Select based on ARB_TYPE_ROUND_ROBIN
    assign next_grant_valid = ARB_TYPE_ROUND_ROBIN ? round_robin_valid : fixed_priority_valid;
    assign next_grant_encoded = ARB_TYPE_ROUND_ROBIN ? round_robin_idx : fixed_priority_idx;

    // Generate one-hot grant from encoded
    reg [PORTS-1:0] next_grant_onehot;
    integer j;
    always @* begin
        next_grant_onehot = {PORTS{1'b0}};
        for (j = 0; j < PORTS; j = j + 1) begin
            if (j == next_grant_encoded) begin
                next_grant_onehot[j] = 1'b1;
            end
        end
    end

    assign next_grant = next_grant_onehot;

    // Blocking logic
    wire release_event;
    assign release_event = (ARB_BLOCK_ACK == 1) ? acknowledge[next_grant_encoded] : 1'b1;

    // If blocking is enabled and we're holding a grant, keep it until release
    wire hold_grant;
    assign hold_grant = ARB_BLOCK && block_hold && !release_event;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            grant_reg <= {PORTS{1'b0}};
            grant_valid_reg <= 1'b0;
            grant_encoded_reg <= {GRANT_WIDTH{1'b0}};
            last_grant_idx <= {GRANT_WIDTH{1'b0}};
            block_hold <= 1'b0;
        end else begin
            if (hold_grant) begin
                // Hold the current grant
                grant_reg <= grant_reg;
                grant_valid_reg <= grant_valid_reg;
                grant_encoded_reg <= grant_encoded_reg;
                block_hold <= 1'b1;
            end else begin
                // Update grant
                grant_reg <= next_grant;
                grant_valid_reg <= next_grant_valid;
                grant_encoded_reg <= next_grant_encoded;
                
                // Update last grant index for round-robin
                if (next_grant_valid) begin
                    last_grant_idx <= next_grant_encoded;
                end
                
                // Update block hold
                if (ARB_BLOCK && next_grant_valid) begin
                    block_hold <= 1'b1;
                end else begin
                    block_hold <= 1'b0;
                end
            end
        end
    end

    assign grant = grant_reg;
    assign grant_valid = grant_valid_reg;
    assign grant_encoded = grant_encoded_reg;

endmodule
