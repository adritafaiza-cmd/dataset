```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;

    // Source domain signals
    reg [PTR_WIDTH-1:0] src_wr_ptr;
    reg [PTR_WIDTH-1:0] src_wr_ptr_gray;
    reg [PTR_WIDTH-1:0] src_rd_ptr_gray_sync;
    reg [PTR_WIDTH-1:0] src_rd_ptr_gray_sync1;
    reg [PTR_WIDTH-1:0] src_rd_ptr_gray_sync2;
    reg [WIDTH-1:0] src_mem [0:DEPTH-1];
    reg src_clear_pending;
    reg src_clear_in_progress;

    // Destination domain signals
    reg [PTR_WIDTH-1:0] dst_rd_ptr;
    reg [PTR_WIDTH-1:0] dst_rd_ptr_gray;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync1;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync2;
    reg dst_clear_pending;
    reg dst_clear_in_progress;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;

    // Helper functions
    function [PTR_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_WIDTH-1:0] gray2bin;
        input [PTR_WIDTH-1:0] gray;
        reg [PTR_WIDTH-1:0] bin;
        integer i;
        begin
            bin[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1)
                bin[i] = gray[i] ^ bin[i+1];
            gray2bin = bin;
        end
    endfunction

    // Source domain: clear handling
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_pending <= 1'b0;
            src_clear_in_progress <= 1'b0;
        end else begin
            if (src_clear_i && !src_clear_in_progress) begin
                src_clear_in_progress <= 1'b1;
                src_clear_pending <= 1'b1;
            end else if (src_clear_in_progress) begin
                // Wait until destination side has also cleared
                // We need to know when dst has cleared. Use a simple handshake:
                // src_clear_in_progress stays high until dst_clear_pending goes low
                // But we can't directly see dst_clear_pending in src domain.
                // Instead, we'll use a different approach: both sides clear independently,
                // and pending stays high until the local side is done.
                // For src side: done when we've reset pointers and memory is considered empty.
                // Since we're just resetting pointers, it's done in one cycle after clear request.
                src_clear_in_progress <= 1'b0;
                src_clear_pending <= 1'b0;
            end
        end
    end

    // Source domain: write pointer and memory
    wire src_full;
    wire src_empty;
    wire src_can_write;

    assign src_full = (src_wr_ptr_gray == src_rd_ptr_gray_sync) && 
                      (src_wr_ptr[PTR_WIDTH-1] != src_rd_ptr_gray_sync[PTR_WIDTH-1]);
    assign src_empty = (src_wr_ptr_gray == src_rd_ptr_gray_sync);
    assign src_can_write = !src_full && !src_clear_in_progress && !src_clear_pending;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
            src_wr_ptr_gray <= {PTR_WIDTH{1'b0}};
        end else if (src_clear_i || src_clear_in_progress) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
            src_wr_ptr_gray <= {PTR_WIDTH{1'b0}};
        end else if (src_valid_i && src_can_write) begin
            src_wr_ptr <= src_wr_ptr + 1'b1;
            src_wr_ptr_gray <= bin2gray(src_wr_ptr + 1'b1);
            src_mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
        end
    end

    // Source domain: synchronize destination read pointer
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rd_ptr_gray_sync <= {PTR_WIDTH{1'b0}};
            src_rd_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            src_rd_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            src_rd_ptr_gray_sync <= dst_rd_ptr_gray;
            src_rd_ptr_gray_sync1 <= src_rd_ptr_gray_sync;
            src_rd_ptr_gray_sync2 <= src_rd_ptr_gray_sync1;
        end
    end

    assign src_ready_o = src_can_write;
    assign src_clear_pending_o = src_clear_pending;

    // Destination domain: clear handling
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_pending <= 1'b0;
            dst_clear_in_progress <= 1'b0;
        end else begin
            if (dst_clear_i && !dst_clear_in_progress) begin
                dst_clear_in_progress <= 1'b1;
                dst_clear_pending <= 1'b1;
            end else if (dst_clear_in_progress) begin
                dst_clear_in_progress <= 1'b0;
                dst_clear_pending <= 1'b0;
            end
        end
    end

    // Destination domain: read pointer
    wire dst_full;
    wire dst_empty;
    wire dst_can_read;

    assign dst_full = (dst_rd_ptr_gray == dst_wr_ptr_gray_sync) && 
                      (dst_rd_ptr[PTR_WIDTH-1] != dst_wr_ptr_gray_sync[PTR_WIDTH-1]);
    assign dst_empty = (dst_rd_ptr_gray == dst_wr_ptr_gray_sync);
    assign dst_can_read = !dst_empty && !dst_clear_in_progress && !dst_clear_pending;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_gray <= {PTR_WIDTH{1'b0}};
        end else if (dst_clear_i || dst_clear_in_progress) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_gray <= {PTR_WIDTH{1'b0}};
        end else if (dst_valid_reg && dst_ready_i) begin
            dst_rd_ptr <= dst_rd_ptr + 1'b1;
            dst_rd_ptr_gray <= bin2gray(dst_rd_ptr + 1'b1);
        end
    end

    // Destination domain: synchronize source write pointer
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync <= {PTR_WIDTH{1'b0}};
            dst_wr_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            dst_wr_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync <= src_wr_ptr_gray;
            dst_wr_ptr_gray_sync1 <= dst_wr_ptr_gray_sync;
            dst_wr_ptr_gray_sync2 <= dst_wr_ptr_gray_sync1;
        end
    end

    // Destination domain: output data and valid
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else if (dst_clear_i || dst_clear_in_progress) begin
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else if (dst_can_read) begin
            dst_data_reg <= src_mem[dst_rd_ptr[LOG_DEPTH-1:0]];
            dst_valid_reg <= 1'b1;
        end else if (dst_valid_reg && dst_ready_i) begin
            dst_valid_reg <= 1'b0;
        end
    end

    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;
    assign dst_clear_pending_o = dst_clear_pending;

endmodule
