```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;
    localparam GRAY_WIDTH = LOG_DEPTH + 1;

    // Source domain signals
    reg  [PTR_WIDTH-1:0] src_wr_ptr;
    reg  [GRAY_WIDTH-1:0] src_wr_ptr_gray;
    reg  [GRAY_WIDTH-1:0] src_rd_ptr_gray_synced;
    reg  [WIDTH-1:0] mem [0:DEPTH-1];
    reg  src_clear_pending;
    reg  src_clear_active;

    // Destination domain signals
    reg  [PTR_WIDTH-1:0] dst_rd_ptr;
    reg  [GRAY_WIDTH-1:0] dst_rd_ptr_gray;
    reg  [GRAY_WIDTH-1:0] dst_wr_ptr_gray_synced;
    reg  [WIDTH-1:0] dst_data_reg;
    reg  dst_valid_reg;
    reg  dst_clear_pending;
    reg  dst_clear_active;

    // Synchronizer registers for source read pointer gray
    reg  [GRAY_WIDTH-1:0] src_rd_ptr_gray_sync [0:SYNC_STAGES-1];

    // Synchronizer registers for destination write pointer gray
    reg  [GRAY_WIDTH-1:0] dst_wr_ptr_gray_sync [0:SYNC_STAGES-1];

    // Clear request synchronizers
    reg  [1:0] src_clear_sync;
    reg  [1:0] dst_clear_sync;

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
            src_wr_ptr_gray <= {GRAY_WIDTH{1'b0}};
            src_clear_pending <= 1'b0;
            src_clear_active <= 1'b0;
            src_rd_ptr_gray_sync[0] <= {GRAY_WIDTH{1'b0}};
            src_rd_ptr_gray_sync[1] <= {GRAY_WIDTH{1'b0}};
            if (SYNC_STAGES > 2) begin
                src_rd_ptr_gray_sync[2] <= {GRAY_WIDTH{1'b0}};
            end
            if (SYNC_STAGES > 3) begin
                src_rd_ptr_gray_sync[3] <= {GRAY_WIDTH{1'b0}};
            end
            if (SYNC_STAGES > 4) begin
                src_rd_ptr_gray_sync[4] <= {GRAY_WIDTH{1'b0}};
            end
            if (SYNC_STAGES > 5) begin
                src_rd_ptr_gray_sync[5] <= {GRAY_WIDTH{1'b0}};
            end
            if (SYNC_STAGES > 6) begin
                src_rd_ptr_gray_sync[6] <= {GRAY_WIDTH{1'b0}};
            end
            if (SYNC_STAGES > 7) begin
                src_rd_ptr_gray_sync[7] <= {GRAY_WIDTH{1'b0}};
            end
            src_clear_sync <= 2'b00;
        end else begin
            // Synchronize destination clear request
            src_clear_sync <= {src_clear_sync[0], dst_clear_sync[1]};

            // Synchronize read pointer gray
            src_rd_ptr_gray_sync[0] <= dst_rd_ptr_gray;
            src_rd_ptr_gray_sync[1] <= src_rd_ptr_gray_sync[0];
            if (SYNC_STAGES > 2) begin
                src_rd_ptr_gray_sync[2] <= src_rd_ptr_gray_sync[1];
            end
            if (SYNC_STAGES > 3) begin
                src_rd_ptr_gray_sync[3] <= src_rd_ptr_gray_sync[2];
            end
            if (SYNC_STAGES > 4) begin
                src_rd_ptr_gray_sync[4] <= src_rd_ptr_gray_sync[3];
            end
            if (SYNC_STAGES > 5) begin
                src_rd_ptr_gray_sync[5] <= src_rd_ptr_gray_sync[4];
            end
            if (SYNC_STAGES > 6) begin
                src_rd_ptr_gray_sync[6] <= src_rd_ptr_gray_sync[5];
            end
            if (SYNC_STAGES > 7) begin
                src_rd_ptr_gray_sync[7] <= src_rd_ptr_gray_sync[6];
            end

            // Clear handling
            if (src_clear_i || src_clear_sync[1]) begin
                src_clear_active <= 1'b1;
                src_clear_pending <= 1'b1;
            end else if (src_clear_active) begin
                // Wait for destination to also be in clear state
                if (src_clear_sync[1]) begin
                    // Both sides are clearing, reset pointers
                    src_wr_ptr <= {PTR_WIDTH{1'b0}};
                    src_wr_ptr_gray <= {GRAY_WIDTH{1'b0}};
                    src_clear_active <= 1'b0;
                    src_clear_pending <= 1'b0;
                end
            end

            // Write pointer update
            if (!src_clear_active && src_valid_i && src_ready_o) begin
                mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
                src_wr_ptr <= src_wr_ptr + 1'b1;
                src_wr_ptr_gray <= src_wr_ptr + 1'b1;
            end
        end
    end

    // Convert binary to gray
    function [GRAY_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Source ready
    wire [GRAY_WIDTH-1:0] src_rd_ptr_gray_synced_wire;
    assign src_rd_ptr_gray_synced_wire = (SYNC_STAGES == 1) ? src_rd_ptr_gray_sync[0] :
                                          (SYNC_STAGES == 2) ? src_rd_ptr_gray_sync[1] :
                                          (SYNC_STAGES == 3) ? src_rd_ptr_gray_sync[2] :
                                          (SYNC_STAGES == 4) ? src_rd_ptr_gray_sync[3] :
                                          (SYNC_STAGES == 5) ? src_rd_ptr_gray_sync[4] :
                                          (SYNC_STAGES == 6) ? src_rd_ptr_gray_sync[5] :
                                          (SYNC_STAGES == 7) ? src_rd_ptr_gray_sync[6] :
                                                              src_rd_ptr_gray_sync[7];

    wire [PTR_WIDTH-1:0] src_rd_ptr_synced_bin;
    assign src_rd_ptr_synced_bin = gray2bin(src_rd_ptr_gray_synced_wire);

    function [PTR_WIDTH-1:0] gray2bin;
        input [GRAY_WIDTH-1:0] gray;
        reg [PTR_WIDTH-1:0] bin;
        integer i;
        begin
            bin[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1) begin
                bin[i] = bin[i+1] ^ gray[i];
            end
            gray2bin = bin;
        end
    endfunction

    assign src_ready_o = !src_clear_active && (src_wr_ptr != src_rd_ptr_synced_bin);
    assign src_clear_pending_o = src_clear_pending;

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_gray <= {GRAY_WIDTH{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
            dst_clear_pending <= 1'b0;
            dst_clear_active <= 1'b0;
            dst_wr_ptr_gray_sync[0] <= {GRAY_WIDTH{1'b0}};
            dst_wr_ptr_gray_sync[1] <= {GRAY_WIDTH{1'b0}};
            if (SYNC_STAGES > 2) begin
                dst_wr_ptr_gray_sync[2] <= {GRAY_WIDTH{1'b0}};
            end
            if (SYNC_STAGES > 3) begin
                dst_wr_ptr_gray_sync[3] <= {GRAY_WIDTH{1'b0}};
            end
            if (SYNC_STAGES > 4) begin
                dst_wr_ptr_gray_sync[4] <= {GRAY_WIDTH{1'b0}};
            end
            if (SYNC_STAGES > 5) begin
                dst_wr_ptr_gray_sync[5] <= {GRAY_WIDTH{1'b0}};
            end
            if (SYNC_STAGES > 6) begin
                dst_wr_ptr_gray_sync[6] <= {GRAY_WIDTH{1'b0}};
            end
            if (SYNC_STAGES > 7) begin
                dst_wr_ptr_gray_sync[7] <= {GRAY_WIDTH{1'b0}};
            end
            dst_clear_sync <= 2'b00;
        end else begin
            // Synchronize source clear request
            dst_clear_sync <= {dst_clear_sync[0], src_clear_sync[1]};

            // Synchronize write pointer gray
            dst_wr_ptr_gray_sync[0] <= src_wr_ptr_gray;
            dst_wr_ptr_gray_sync[1] <= dst_wr_ptr_gray_sync[0];
            if (SYNC_STAGES > 2) begin
                dst_wr_ptr_gray_sync[2] <= dst_wr_ptr_gray_sync[1];
            end
            if (SYNC_STAGES > 3) begin
                dst_wr_ptr_gray_sync[3] <= dst_wr_ptr_gray_sync[2];
            end
            if (SYNC_STAGES > 4) begin
                dst_wr_ptr_gray_sync[4] <= dst_wr_ptr_gray_sync[3];
            end
            if (SYNC_STAGES > 5) begin
                dst_wr_ptr_gray_sync[5] <= dst_wr_ptr_gray_sync[4];
            end
            if (SYNC_STAGES > 6) begin
                dst_wr_ptr_gray_sync[6] <= dst_wr_ptr_gray_sync[5];
            end
            if (SYNC_STAGES > 7) begin
                dst_wr_ptr_gray_sync[7] <= dst_wr_ptr_gray_sync[6];
            end

            // Clear handling
            if (dst_clear_i || dst_clear_sync[1]) begin
                dst_clear_active <= 1'b1;
                dst_clear_pending <= 1'b1;
            end else if (dst_clear_active) begin
                // Wait for source to also be in clear state
                if (dst_clear_sync[1]) begin
                    // Both sides are clearing, reset pointers
                    dst_rd_ptr <= {PTR_WIDTH{1'b0}};
                    dst_rd_ptr_gray <= {GRAY_WIDTH{1'b0}};
                    dst_valid_reg <= 1'b0;
                    dst_clear_active <= 1'b0;
                    dst_clear_pending <= 1'b0;
                end
            end

            // Read pointer update
            if (!dst_clear_active && dst_valid_reg && dst_ready_i) begin
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_rd_ptr_gray <= dst_rd_ptr + 1'b1;
            end

            // Data output
            if (!dst_clear_active) begin
                if (dst_valid_reg && dst_ready_i) begin
                    // Check if there's more data
                    if (dst_rd_ptr + 1'b1 != dst_wr_ptr_synced_bin) begin
                        dst_data_reg <= mem[(dst_rd_ptr + 1'b1)[LOG_DEPTH-1:0]];
                        dst_valid_reg <= 1'b1;
                    end else begin
                        dst_valid_reg <= 1'b0;
                    end
                end else if (!dst_valid_reg) begin
                    // Check if new data available
                    if (dst_rd_ptr != dst_wr_ptr_synced_bin) begin
                        dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                        dst_valid_reg <= 1'b1;
                    end
                end
            end else begin
                dst_valid_reg <= 1'b0;
            end
        end
    end

    wire [GRAY_WIDTH-1:0] dst_wr_ptr_gray_synced_wire;
    assign dst_wr_ptr_gray_synced_wire = (SYNC_STAGES == 1) ? dst_wr_ptr_gray_sync[0] :
                                          (SYNC_STAGES == 2) ? dst_wr_ptr_gray_sync[1] :
                                          (SYNC_STAGES == 3) ? dst_wr_ptr_gray_sync[2] :
                                          (SYNC_STAGES == 4) ? dst_wr_ptr_gray_sync[3] :
                                          (SYNC_STAGES == 5) ? dst_wr_ptr_gray_sync[4] :
                                          (SYNC_STAGES == 6) ? dst_wr_ptr_gray_sync[5] :
                                          (SYNC_STAGES == 7) ? dst_wr_ptr_gray_sync[6] :
                                                              dst_wr_ptr_gray_sync[7];

    wire [PTR_WIDTH-1:0] dst_wr_ptr_synced_bin;
    assign dst_wr_ptr_synced_bin = gray2bin(dst_wr_ptr_gray_synced_wire);

    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;
    assign dst_clear_pending_o = dst_clear_pending;

endmodule
