module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;
    localparam GRAY_W = LOG_DEPTH + 1;

    // Source domain
    reg [PTR_W-1:0] src_wr_ptr;
    reg [GRAY_W-1:0] src_wr_gray;
    reg [GRAY_W-1:0] src_rd_gray_sync [SYNC_STAGES-1:0];
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg src_clear_pending;
    reg src_clear_active;

    // Destination domain
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [GRAY_W-1:0] dst_rd_gray;
    reg [GRAY_W-1:0] dst_wr_gray_sync [SYNC_STAGES-1:0];
    reg dst_clear_pending;
    reg dst_clear_active;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;

    // Source domain logic
    wire [GRAY_W-1:0] src_rd_gray_synced = src_rd_gray_sync[SYNC_STAGES-1];
    wire [PTR_W-1:0] src_rd_ptr_bin = gray_to_bin(src_rd_gray_synced);
    wire [PTR_W-1:0] src_wr_ptr_next = src_wr_ptr + 1'b1;
    wire [GRAY_W-1:0] src_wr_gray_next = bin_to_gray(src_wr_ptr_next);

    wire src_full = (src_wr_ptr[PTR_W-1] == src_rd_ptr_bin[PTR_W-1]) &&
                    (src_wr_ptr[PTR_W-2:0] == src_rd_ptr_bin[PTR_W-2:0]);
    wire src_empty = (src_wr_ptr == src_rd_ptr_bin);

    wire src_can_write = src_valid_i && !src_full && !src_clear_active && !src_clear_pending;
    wire src_ready = !src_full && !src_clear_active && !src_clear_pending;

    // Destination domain logic
    wire [GRAY_W-1:0] dst_wr_gray_synced = dst_wr_gray_sync[SYNC_STAGES-1];
    wire [PTR_W-1:0] dst_wr_ptr_bin = gray_to_bin(dst_wr_gray_synced);
    wire [PTR_W-1:0] dst_rd_ptr_next = dst_rd_ptr + 1'b1;
    wire [GRAY_W-1:0] dst_rd_gray_next = bin_to_gray(dst_rd_ptr_next);

    wire dst_full = (dst_rd_ptr[PTR_W-1] == dst_wr_ptr_bin[PTR_W-1]) &&
                    (dst_rd_ptr[PTR_W-2:0] == dst_wr_ptr_bin[PTR_W-2:0]);
    wire dst_empty = (dst_rd_ptr == dst_wr_ptr_bin);

    wire dst_can_read = dst_valid_reg && dst_ready_i && !dst_clear_active && !dst_clear_pending;
    wire dst_valid = dst_valid_reg && !dst_clear_active && !dst_clear_pending;

    // Source clear logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_pending <= 1'b0;
            src_clear_active <= 1'b0;
        end else begin
            if (src_clear_i) begin
                src_clear_pending <= 1'b1;
                src_clear_active <= 1'b1;
            end else if (src_clear_pending) begin
                // Wait until destination side has also cleared
                // We need to know when dst has cleared. Use a handshake.
                // For simplicity, clear completes when both sides are cleared.
                // We'll use a simple approach: clear pending stays until dst_clear_pending is low
                // But we can't directly see dst_clear_pending in src domain.
                // Instead, we'll use a different approach: clear is done when both pointers are reset.
                // Let's use a simpler model: clear pending stays high for a fixed number of cycles
                // or until we detect that the other side has cleared.
                
                // Actually, let's use a different approach:
                // When clear is requested, we set clear_active and clear_pending.
                // We reset the local pointer and gray code.
                // We need to wait for the other side to also clear.
                // We can use a simple counter or wait for the synchronized pointer to match.
                
                // For now, let's just keep clear_pending high until we see that the other side
                // has also cleared. We'll use a flag that gets cleared when both sides are done.
                
                // Simplified: clear pending stays high for 2 cycles after clear is requested,
                // then we check if the other side has cleared by looking at the synchronized pointer.
                
                // Actually, the simplest correct approach:
                // 1. On clear request, set clear_active and clear_pending.
                // 2. Reset local pointer and gray code.
                // 3. Keep clear_pending high until the synchronized remote pointer matches the local reset pointer.
                
                // Let's implement this properly.
                src_clear_pending <= 1'b1;
                src_clear_active <= 1'b1;
            end else begin
                src_clear_pending <= 1'b0;
                src_clear_active <= 1'b0;
            end
        end
    end

    // Source pointer and memory write
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
            src_wr_gray <= {GRAY_W{1'b0}};
        end else if (src_clear_active) begin
            src_wr_ptr <= {PTR_W{1'b0}};
            src_wr_gray <= {GRAY_W{1'b0}};
        end else if (src_can_write) begin
            mem[src_wr_ptr[PTR_W-2:0]] <= src_data_i;
            src_wr_ptr <= src_wr_ptr_next;
            src_wr_gray <= src_wr_gray_next;
        end
    end

    // Source clear pending completion
    // Clear pending should stay high until the destination side has also cleared.
    // We can detect this by checking if the synchronized destination read pointer
    // matches the reset value (all zeros).
    wire src_clear_done = (src_rd_ptr_bin == {PTR_W{1'b0}});
    
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_pending <= 1'b0;
        end else if (src_clear_pending && src_clear_done) begin
            src_clear_pending <= 1'b0;
        end
    end

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_pending <= 1'b0;
            dst_clear_active <= 1'b0;
        end else begin
            if (dst_clear_i) begin
                dst_clear_pending <= 1'b1;
                dst_clear_active <= 1'b1;
            end else if (dst_clear_pending) begin
                dst_clear_pending <= 1'b1;
                dst_clear_active <= 1'b1;
            end else begin
                dst_clear_pending <= 1'b0;
                dst_clear_active <= 1'b0;
            end
        end
    end

    // Destination pointer and read
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_rd_gray <= {GRAY_W{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else if (dst_clear_active) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_rd_gray <= {GRAY_W{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else if (dst_can_read) begin
            dst_rd_ptr <= dst_rd_ptr_next;
            dst_rd_gray <= dst_rd_gray_next;
            dst_valid_reg <= 1'b0;
        end else if (!dst_empty && !dst_valid_reg) begin
            dst_data_reg <= mem[dst_rd_ptr[PTR_W-2:0]];
            dst_valid_reg <= 1'b1;
        end
    end

    // Destination clear pending completion
    wire dst_clear_done = (dst_wr_ptr_bin == {PTR_W{1'b0}});
    
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_pending <= 1'b0;
        end else if (dst_clear_pending && dst_clear_done) begin
            dst_clear_pending <= 1'b0;
        end
    end

    // Synchronizers
    integer i;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            for (i = 0; i < SYNC_STAGES; i = i + 1)
                src_rd_gray_sync[i] <= {GRAY_W{1'b0}};
        end else begin
            src_rd_gray_sync[0] <= dst_rd_gray;
            for (i = 1; i < SYNC_STAGES; i = i + 1)
                src_rd_gray_sync[i] <= src_rd_gray_sync[i-1];
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            for (i = 0; i < SYNC_STAGES; i = i + 1)
                dst_wr_gray_sync[i] <= {GRAY_W{1'b0}};
        end else begin
            dst_wr_gray_sync[0] <= src_wr_gray;
            for (i = 1; i < SYNC_STAGES; i = i + 1)
                dst_wr_gray_sync[i] <= dst_wr_gray_sync[i-1];
        end
    end

    // Outputs
    assign src_ready_o = src_ready;
    assign src_clear_pending_o = src_clear_pending;
    assign dst_valid_o = dst_valid;
    assign dst_data_o = dst_data_reg;
    assign dst_clear_pending_o = dst_clear_pending;

    // Gray code conversion functions
    function [GRAY_W-1:0] bin_to_gray;
        input [PTR_W-1:0] bin;
        begin
            bin_to_gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_W-1:0] gray_to_bin;
        input [GRAY_W-1:0] gray;
        reg [PTR_W-1:0] bin;
        begin
            bin[PTR_W-1] = gray[PTR_W-1];
            for (i = PTR_W-2; i >= 0; i = i - 1)
                bin[i] = gray[i] ^ bin[i+1];
            gray_to_bin = bin;
        end
    endfunction

endmodule
