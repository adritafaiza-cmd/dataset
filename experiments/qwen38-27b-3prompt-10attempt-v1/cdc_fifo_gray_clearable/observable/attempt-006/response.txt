```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;
    localparam GRAY_WIDTH = PTR_WIDTH;

    // Source domain
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_WIDTH-1:0] src_wr_ptr;
    reg [GRAY_WIDTH-1:0] src_wr_ptr_gray;
    reg [GRAY_WIDTH-1:0] src_rd_ptr_gray_sync [0:SYNC_STAGES-1];
    reg src_clear_pending;

    // Destination domain
    reg [PTR_WIDTH-1:0] dst_rd_ptr;
    reg [GRAY_WIDTH-1:0] dst_rd_ptr_gray;
    reg [GRAY_WIDTH-1:0] dst_wr_ptr_gray_sync [0:SYNC_STAGES-1];
    reg dst_clear_pending;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;

    // Convert binary to Gray
    function [GRAY_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = (bin >> 1) ^ bin;
        end
    endfunction

    // Convert Gray to binary
    function [PTR_WIDTH-1:0] gray2bin;
        input [GRAY_WIDTH-1:0] gray;
        integer i;
        reg [PTR_WIDTH-1:0] bin;
        begin
            bin[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1)
                bin[i] = gray[i] ^ bin[i+1];
            gray2bin = bin;
        end
    endfunction

    // Source domain logic
    wire [GRAY_WIDTH-1:0] src_rd_ptr_gray_synced = src_rd_ptr_gray_sync[SYNC_STAGES-1];
    wire [PTR_WIDTH-1:0] src_rd_ptr_bin = gray2bin(src_rd_ptr_gray_synced);
    wire [PTR_WIDTH-1:0] src_wr_ptr_gray_next = bin2gray(src_wr_ptr + 1);
    wire src_full = (src_wr_ptr_gray == src_rd_ptr_gray_synced) &&
                    (src_wr_ptr[PTR_WIDTH-1] != src_rd_ptr_bin[PTR_WIDTH-1]);
    wire src_can_write = src_valid_i && !src_full && !src_clear_pending;
    wire src_ready = !src_full && !src_clear_pending;

    // Destination domain logic
    wire [GRAY_WIDTH-1:0] dst_wr_ptr_gray_synced = dst_wr_ptr_gray_sync[SYNC_STAGES-1];
    wire [PTR_WIDTH-1:0] dst_wr_ptr_bin = gray2bin(dst_wr_ptr_gray_synced);
    wire [PTR_WIDTH-1:0] dst_rd_ptr_gray_next = bin2gray(dst_rd_ptr + 1);
    wire dst_empty = (dst_rd_ptr_gray == dst_wr_ptr_gray_synced) &&
                     (dst_rd_ptr[PTR_WIDTH-1] != dst_wr_ptr_bin[PTR_WIDTH-1]);
    wire dst_can_read = !dst_empty && !dst_clear_pending;
    wire dst_pop = dst_can_read && dst_ready_i;

    // Source clock domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
            src_wr_ptr_gray <= {GRAY_WIDTH{1'b0}};
            src_clear_pending <= 1'b0;
        end else begin
            if (src_clear_i) begin
                src_clear_pending <= 1'b1;
            end else if (src_clear_pending) begin
                // Wait for destination to acknowledge clear
                if (dst_clear_pending_synced) begin
                    src_clear_pending <= 1'b0;
                    src_wr_ptr <= {PTR_WIDTH{1'b0}};
                    src_wr_ptr_gray <= {GRAY_WIDTH{1'b0}};
                end
            end else if (src_can_write) begin
                mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
                src_wr_ptr <= src_wr_ptr + 1'b1;
                src_wr_ptr_gray <= src_wr_ptr_gray_next;
            end
        end
    end

    // Synchronize source write pointer Gray to destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync[0] <= {GRAY_WIDTH{1'b0}};
            dst_wr_ptr_gray_sync[1] <= {GRAY_WIDTH{1'b0}};
            dst_wr_ptr_gray_sync[2] <= {GRAY_WIDTH{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync[0] <= src_wr_ptr_gray;
            dst_wr_ptr_gray_sync[1] <= dst_wr_ptr_gray_sync[0];
            dst_wr_ptr_gray_sync[2] <= dst_wr_ptr_gray_sync[1];
        end
    end

    // Synchronize destination clear pending to source domain
    reg [SYNC_STAGES-1:0] dst_clear_pending_sync;
    wire dst_clear_pending_synced = dst_clear_pending_sync[SYNC_STAGES-1];

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_clear_pending_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            dst_clear_pending_sync[0] <= dst_clear_pending;
            dst_clear_pending_sync[1] <= dst_clear_pending_sync[0];
            dst_clear_pending_sync[2] <= dst_clear_pending_sync[1];
        end
    end

    // Destination clock domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_gray <= {GRAY_WIDTH{1'b0}};
            dst_clear_pending <= 1'b0;
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_clear_i) begin
                dst_clear_pending <= 1'b1;
                dst_valid_reg <= 1'b0;
            end else if (dst_clear_pending) begin
                // Wait for source to acknowledge clear
                if (src_clear_pending_synced) begin
                    dst_clear_pending <= 1'b0;
                    dst_rd_ptr <= {PTR_WIDTH{1'b0}};
                    dst_rd_ptr_gray <= {GRAY_WIDTH{1'b0}};
                    dst_valid_reg <= 1'b0;
                end
            end else begin
                if (dst_pop) begin
                    dst_rd_ptr <= dst_rd_ptr + 1'b1;
                    dst_rd_ptr_gray <= dst_rd_ptr_gray_next;
                    dst_valid_reg <= 1'b0;
                end else if (!dst_empty) begin
                    dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                    dst_valid_reg <= 1'b1;
                end
            end
        end
    end

    // Synchronize destination read pointer Gray to source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rd_ptr_gray_sync[0] <= {GRAY_WIDTH{1'b0}};
            src_rd_ptr_gray_sync[1] <= {GRAY_WIDTH{1'b0}};
            src_rd_ptr_gray_sync[2] <= {GRAY_WIDTH{1'b0}};
        end else begin
            src_rd_ptr_gray_sync[0] <= dst_rd_ptr_gray;
            src_rd_ptr_gray_sync[1] <= src_rd_ptr_gray_sync[0];
            src_rd_ptr_gray_sync[2] <= src_rd_ptr_gray_sync[1];
        end
    end

    // Synchronize source clear pending to destination domain
    reg [SYNC_STAGES-1:0] src_clear_pending_sync;
    wire src_clear_pending_synced = src_clear_pending_sync[SYNC_STAGES-1];

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_clear_pending_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            src_clear_pending_sync[0] <= src_clear_pending;
            src_clear_pending_sync[1] <= src_clear_pending_sync[0];
            src_clear_pending_sync[2] <= src_clear_pending_sync[1];
        end
    end

    // Outputs
    assign src_ready_o = src_ready;
    assign src_clear_pending_o = src_clear_pending;
    assign dst_clear_pending_o = dst_clear_pending;
    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;

endmodule
