```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Source domain
    reg [PTR_W-1:0] src_wr_ptr;
    reg [PTR_W-1:0] src_wr_ptr_gray;
    reg [WIDTH-1:0] mem [0:DEPTH-1];

    // Destination domain
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [PTR_W-1:0] dst_rd_ptr_gray;
    reg [WIDTH-1:0] dst_data_reg;
    reg             dst_valid_reg;

    // Clear pending signals
    reg src_clear_pending;
    reg dst_clear_pending;

    // Synchronizers
    reg [PTR_W-1:0] src_wr_ptr_gray_dst_sync [0:SYNC_STAGES-1];
    reg [PTR_W-1:0] dst_rd_ptr_gray_src_sync [0:SYNC_STAGES-1];

    // Clear synchronizers
    reg src_clear_dst_sync [0:SYNC_STAGES-1];
    reg dst_clear_src_sync [0:SYNC_STAGES-1];

    // Source domain logic
    wire [PTR_W-1:0] dst_rd_ptr_gray_src = dst_rd_ptr_gray_src_sync[SYNC_STAGES-1];
    wire [PTR_W-1:0] dst_rd_ptr_src = gray_to_bin(dst_rd_ptr_gray_src);
    wire src_full = (src_wr_ptr_gray == dst_rd_ptr_gray_src) && (src_wr_ptr[PTR_W-1] != dst_rd_ptr_src[PTR_W-1]);
    wire src_empty = (src_wr_ptr_gray == dst_rd_ptr_gray_src);

    wire src_can_write = src_valid_i && !src_full && !src_clear_pending;
    wire src_ready = !src_full && !src_clear_pending;

    // Destination domain logic
    wire [PTR_W-1:0] src_wr_ptr_gray_dst = src_wr_ptr_gray_dst_sync[SYNC_STAGES-1];
    wire [PTR_W-1:0] src_wr_ptr_dst = gray_to_bin(src_wr_ptr_gray_dst);
    wire dst_full = (dst_rd_ptr_gray == src_wr_ptr_gray_dst) && (dst_rd_ptr[PTR_W-1] != src_wr_ptr_dst[PTR_W-1]);
    wire dst_empty = (dst_rd_ptr_gray == src_wr_ptr_gray_dst);

    wire dst_can_read = dst_valid_reg && dst_ready_i && !dst_clear_pending;

    // Clear pending logic
    // src_clear_pending is set when src_clear_i is asserted, and cleared when dst_clear_src_sync indicates dst has processed the clear
    // dst_clear_pending is set when dst_clear_i is asserted, and cleared when src_clear_dst_sync indicates src has processed the clear

    // Source clock domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
            src_wr_ptr_gray <= {PTR_W{1'b0}};
            src_clear_pending <= 1'b0;
        end else begin
            // Handle clear
            if (src_clear_i) begin
                src_clear_pending <= 1'b1;
            end else if (dst_clear_src_sync[SYNC_STAGES-1]) begin
                src_clear_pending <= 1'b0;
            end

            // Write data
            if (src_can_write) begin
                mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
                src_wr_ptr <= src_wr_ptr + 1'b1;
                src_wr_ptr_gray <= bin_to_gray(src_wr_ptr + 1'b1);
            end
        end
    end

    // Destination clock domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_rd_ptr_gray <= {PTR_W{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
            dst_clear_pending <= 1'b0;
        end else begin
            // Handle clear
            if (dst_clear_i) begin
                dst_clear_pending <= 1'b1;
            end else if (src_clear_dst_sync[SYNC_STAGES-1]) begin
                dst_clear_pending <= 1'b0;
            end

            // Read data
            if (dst_can_read) begin
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_rd_ptr_gray <= bin_to_gray(dst_rd_ptr + 1'b1);
            end

            // Load new data if FIFO not empty and not clearing
            if (!dst_empty && !dst_clear_pending) begin
                dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                dst_valid_reg <= 1'b1;
            end else if (dst_clear_pending) begin
                dst_valid_reg <= 1'b0;
            end
        end
    end

    // Synchronizers for write pointer gray to destination domain
    integer i;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            for (i = 0; i < SYNC_STAGES; i = i + 1)
                src_wr_ptr_gray_dst_sync[i] <= {PTR_W{1'b0}};
        end else begin
            src_wr_ptr_gray_dst_sync[0] <= src_wr_ptr_gray;
            for (i = 1; i < SYNC_STAGES; i = i + 1)
                src_wr_ptr_gray_dst_sync[i] <= src_wr_ptr_gray_dst_sync[i-1];
        end
    end

    // Synchronizers for read pointer gray to source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            for (i = 0; i < SYNC_STAGES; i = i + 1)
                dst_rd_ptr_gray_src_sync[i] <= {PTR_W{1'b0}};
        end else begin
            dst_rd_ptr_gray_src_sync[0] <= dst_rd_ptr_gray;
            for (i = 1; i < SYNC_STAGES; i = i + 1)
                dst_rd_ptr_gray_src_sync[i] <= dst_rd_ptr_gray_src_sync[i-1];
        end
    end

    // Synchronizers for clear signals
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            for (i = 0; i < SYNC_STAGES; i = i + 1)
                src_clear_dst_sync[i] <= 1'b0;
        end else begin
            src_clear_dst_sync[0] <= src_clear_i;
            for (i = 1; i < SYNC_STAGES; i = i + 1)
                src_clear_dst_sync[i] <= src_clear_dst_sync[i-1];
        end
    end

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            for (i = 0; i < SYNC_STAGES; i = i + 1)
                dst_clear_src_sync[i] <= 1'b0;
        end else begin
            dst_clear_src_sync[0] <= dst_clear_i;
            for (i = 1; i < SYNC_STAGES; i = i + 1)
                dst_clear_src_sync[i] <= dst_clear_src_sync[i-1];
        end
    end

    // Outputs
    assign src_ready_o = src_ready;
    assign src_clear_pending_o = src_clear_pending;
    assign dst_clear_pending_o = dst_clear_pending;
    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;

    function [PTR_W-1:0] bin_to_gray;
        input [PTR_W-1:0] bin;
        begin
            bin_to_gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_W-1:0] gray_to_bin;
        input [PTR_W-1:0] gray;
        reg [PTR_W-1:0] b;
        integer j;
        begin
            b[PTR_W-1] = gray[PTR_W-1];
            for (j = PTR_W-2; j >= 0; j = j - 1)
                b[j] = gray[j] ^ b[j+1];
            gray_to_bin = b;
        end
    endfunction

endmodule
