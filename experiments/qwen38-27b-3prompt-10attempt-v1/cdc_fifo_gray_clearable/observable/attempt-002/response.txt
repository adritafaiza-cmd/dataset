```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;

    // Source domain
    reg [PTR_WIDTH-1:0] src_wr_ptr;
    reg [PTR_WIDTH-1:0] src_wr_ptr_gray;
    reg [PTR_WIDTH-1:0] src_rd_ptr_gray_sync;
    reg [PTR_WIDTH-1:0] src_rd_ptr_gray_sync_next;
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg src_clear_pending;

    // Destination domain
    reg [PTR_WIDTH-1:0] dst_rd_ptr;
    reg [PTR_WIDTH-1:0] dst_rd_ptr_gray;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync_next;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;
    reg dst_clear_pending;

    // Binary to Gray conversion
    function [PTR_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Gray to Binary conversion
    function [PTR_WIDTH-1:0] gray2bin;
        input [PTR_WIDTH-1:0] gray;
        reg [PTR_WIDTH-1:0] bin;
        integer i;
        begin
            bin[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1)
                bin[i] = gray[i] ^ bin[i+1];
            gray2bin = bin;
        end
    endfunction

    // Source domain logic
    wire [PTR_WIDTH-1:0] src_rd_ptr_bin = gray2bin(src_rd_ptr_gray_sync);
    wire src_full = (src_wr_ptr_gray == src_rd_ptr_gray_sync) && 
                    (src_wr_ptr[PTR_WIDTH-1] != src_rd_ptr_bin[PTR_WIDTH-1]);
    wire src_empty = (src_wr_ptr_gray == src_rd_ptr_gray_sync) && 
                     (src_wr_ptr[PTR_WIDTH-1] == src_rd_ptr_bin[PTR_WIDTH-1]);

    assign src_ready_o = ~src_full && ~src_clear_pending;

    // Clear pending logic in source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_clear_pending <= 1'b0;
        else if (src_clear_i)
            src_clear_pending <= 1'b1;
        else if (src_clear_pending && src_rd_ptr_gray_sync == src_wr_ptr_gray)
            src_clear_pending <= 1'b0;
    end

    assign src_clear_pending_o = src_clear_pending;

    // Write pointer update
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
            src_wr_ptr_gray <= {PTR_WIDTH{1'b0}};
        end
        else if (src_clear_i) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
            src_wr_ptr_gray <= {PTR_WIDTH{1'b0}};
        end
        else if (src_valid_i && src_ready_o) begin
            src_wr_ptr <= src_wr_ptr + 1'b1;
            src_wr_ptr_gray <= bin2gray(src_wr_ptr + 1'b1);
        end
    end

    // Memory write
    always @(posedge src_clk_i) begin
        if (src_valid_i && src_ready_o)
            mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
    end

    // Synchronize read pointer gray from destination domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_rd_ptr_gray_sync <= {PTR_WIDTH{1'b0}};
        else
            src_rd_ptr_gray_sync <= src_rd_ptr_gray_sync_next;
    end

    // Synchronizer chain for read pointer gray
    reg [SYNC_STAGES-1:0][PTR_WIDTH-1:0] src_rd_ptr_gray_sync_chain;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            for (integer i = 0; i < SYNC_STAGES; i = i + 1)
                src_rd_ptr_gray_sync_chain[i] <= {PTR_WIDTH{1'b0}};
        end
        else begin
            src_rd_ptr_gray_sync_chain[0] <= dst_rd_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                src_rd_ptr_gray_sync_chain[i] <= src_rd_ptr_gray_sync_chain[i-1];
        end
    end
    assign src_rd_ptr_gray_sync_next = src_rd_ptr_gray_sync_chain[SYNC_STAGES-1];

    // Destination domain logic
    wire [PTR_WIDTH-1:0] dst_wr_ptr_bin = gray2bin(dst_wr_ptr_gray_sync);
    wire dst_full = (dst_rd_ptr_gray == dst_wr_ptr_gray_sync) && 
                    (dst_rd_ptr[PTR_WIDTH-1] != dst_wr_ptr_bin[PTR_WIDTH-1]);
    wire dst_empty = (dst_rd_ptr_gray == dst_wr_ptr_gray_sync) && 
                     (dst_rd_ptr[PTR_WIDTH-1] == dst_wr_ptr_bin[PTR_WIDTH-1]);

    // Clear pending logic in destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_clear_pending <= 1'b0;
        else if (dst_clear_i)
            dst_clear_pending <= 1'b1;
        else if (dst_clear_pending && dst_wr_ptr_gray_sync == dst_rd_ptr_gray)
            dst_clear_pending <= 1'b0;
    end

    assign dst_clear_pending_o = dst_clear_pending;

    // Read pointer update
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_gray <= {PTR_WIDTH{1'b0}};
        end
        else if (dst_clear_i) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_gray <= {PTR_WIDTH{1'b0}};
        end
        else if (dst_valid_reg && dst_ready_i) begin
            dst_rd_ptr <= dst_rd_ptr + 1'b1;
            dst_rd_ptr_gray <= bin2gray(dst_rd_ptr + 1'b1);
        end
    end

    // Synchronize write pointer gray from source domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_wr_ptr_gray_sync <= {PTR_WIDTH{1'b0}};
        else
            dst_wr_ptr_gray_sync <= dst_wr_ptr_gray_sync_next;
    end

    // Synchronizer chain for write pointer gray
    reg [SYNC_STAGES-1:0][PTR_WIDTH-1:0] dst_wr_ptr_gray_sync_chain;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            for (integer i = 0; i < SYNC_STAGES; i = i + 1)
                dst_wr_ptr_gray_sync_chain[i] <= {PTR_WIDTH{1'b0}};
        end
        else begin
            dst_wr_ptr_gray_sync_chain[0] <= src_wr_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                dst_wr_ptr_gray_sync_chain[i] <= dst_wr_ptr_gray_sync_chain[i-1];
        end
    end
    assign dst_wr_ptr_gray_sync_next = dst_wr_ptr_gray_sync_chain[SYNC_STAGES-1];

    // Output data and valid
    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;

    // Data read and valid register update
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end
        else if (dst_clear_i) begin
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end
        else if (dst_valid_reg && dst_ready_i) begin
            // Current beat is being consumed; check if there's another word
            if (!dst_empty) begin
                dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                dst_valid_reg <= 1'b1;
            end
            else begin
                dst_valid_reg <= 1'b0;
            end
        end
        else if (!dst_valid_reg && !dst_empty) begin
            dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
            dst_valid_reg <= 1'b1;
        end
    end

endmodule
