```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Source domain signals
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_W-1:0] src_wr_ptr;
    reg [PTR_W-1:0] src_rd_ptr_gray;
    reg [PTR_W-1:0] src_rd_ptr_gray_sync;
    reg [PTR_W-1:0] src_rd_ptr_gray_sync2;
    reg [PTR_W-1:0] src_rd_ptr_gray_sync3;
    reg src_clear_pending;
    reg src_clear_sync;
    reg src_clear_sync2;
    reg src_clear_sync3;
    reg src_clear_done;

    // Destination domain signals
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [PTR_W-1:0] dst_wr_ptr_gray;
    reg [PTR_W-1:0] dst_wr_ptr_gray_sync;
    reg [PTR_W-1:0] dst_wr_ptr_gray_sync2;
    reg [PTR_W-1:0] dst_wr_ptr_gray_sync3;
    reg dst_clear_pending;
    reg dst_clear_sync;
    reg dst_clear_sync2;
    reg dst_clear_sync3;
    reg dst_clear_done;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;

    // Clear synchronization chains
    // Source clear to destination
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_sync  <= 1'b0;
            dst_clear_sync2 <= 1'b0;
            dst_clear_sync3 <= 1'b0;
        end else begin
            dst_clear_sync  <= src_clear_i;
            dst_clear_sync2 <= dst_clear_sync;
            dst_clear_sync3 <= dst_clear_sync2;
        end
    end

    // Destination clear to source
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_sync  <= 1'b0;
            src_clear_sync2 <= 1'b0;
            src_clear_sync3 <= 1'b0;
        end else begin
            src_clear_sync  <= dst_clear_i;
            src_clear_sync2 <= src_clear_sync;
            src_clear_sync3 <= src_clear_sync2;
        end
    end

    // Clear pending logic
    // Source side: pending when local clear or remote clear (synced) is active
    wire src_clear_active = src_clear_i | src_clear_sync3;
    wire dst_clear_active = dst_clear_i | dst_clear_sync3;

    // Source clear pending: high when clear is active and not yet done
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_pending <= 1'b0;
            src_clear_done    <= 1'b0;
        end else begin
            if (src_clear_active) begin
                src_clear_pending <= 1'b1;
                src_clear_done    <= 1'b0;
            end else if (src_clear_pending && !src_clear_active) begin
                // Clear has been deasserted, wait for it to propagate
                src_clear_pending <= 1'b0;
                src_clear_done    <= 1'b1;
            end
        end
    end

    // Destination clear pending
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_pending <= 1'b0;
            dst_clear_done    <= 1'b0;
        end else begin
            if (dst_clear_active) begin
                dst_clear_pending <= 1'b1;
                dst_clear_done    <= 1'b0;
            end else if (dst_clear_pending && !dst_clear_active) begin
                dst_clear_pending <= 1'b0;
                dst_clear_done    <= 1'b1;
            end
        end
    end

    assign src_clear_pending_o = src_clear_pending;
    assign dst_clear_pending_o = dst_clear_pending;

    // Gray code conversion
    function [PTR_W-1:0] bin2gray;
        input [PTR_W-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Source write pointer and memory
    wire src_wr_ptr_gray = bin2gray(src_wr_ptr);

    // Synchronize source write pointer gray to destination
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync  <= {PTR_W{1'b0}};
            dst_wr_ptr_gray_sync2 <= {PTR_W{1'b0}};
            dst_wr_ptr_gray_sync3 <= {PTR_W{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync  <= src_wr_ptr_gray;
            dst_wr_ptr_gray_sync2 <= dst_wr_ptr_gray_sync;
            dst_wr_ptr_gray_sync3 <= dst_wr_ptr_gray_sync2;
        end
    end

    // Synchronize destination read pointer gray to source
    wire dst_rd_ptr_gray = bin2gray(dst_rd_ptr);

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rd_ptr_gray_sync  <= {PTR_W{1'b0}};
            src_rd_ptr_gray_sync2 <= {PTR_W{1'b0}};
            src_rd_ptr_gray_sync3 <= {PTR_W{1'b0}};
        end else begin
            src_rd_ptr_gray_sync  <= dst_rd_ptr_gray;
            src_rd_ptr_gray_sync2 <= src_rd_ptr_gray_sync;
            src_rd_ptr_gray_sync3 <= src_rd_ptr_gray_sync2;
        end
    end

    // Convert synchronized gray to binary for comparison
    function [PTR_W-1:0] gray2bin;
        input [PTR_W-1:0] gray;
        integer i;
        begin
            gray2bin[PTR_W-1] = gray[PTR_W-1];
            for (i = PTR_W-2; i >= 0; i = i - 1)
                gray2bin[i] = gray2bin[i+1] ^ gray[i];
        end
    endfunction

    wire [PTR_W-1:0] src_rd_ptr_bin = gray2bin(src_rd_ptr_gray_sync3);
    wire [PTR_W-1:0] dst_wr_ptr_bin = gray2bin(dst_wr_ptr_gray_sync3);

    // FIFO full/empty
    wire src_fifo_full  = (src_wr_ptr[PTR_W-1:1] == src_rd_ptr_bin[PTR_W-1:1]) &&
                          (src_wr_ptr[0] != src_rd_ptr_bin[0]);
    wire dst_fifo_empty = (dst_rd_ptr == dst_wr_ptr_bin);

    // Source ready: not full and not in clear
    assign src_ready_o = !src_fifo_full && !src_clear_active;

    // Source write
    wire src_write = src_valid_i && src_ready_o;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
        end else if (src_clear_active) begin
            src_wr_ptr <= {PTR_W{1'b0}};
        end else if (src_write) begin
            mem[src_wr_ptr[PTR_W-2:0]] <= src_data_i;
            src_wr_ptr <= src_wr_ptr + 1'b1;
        end
    end

    // Destination read
    wire dst_read = dst_valid_reg && dst_ready_i;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr    <= {PTR_W{1'b0}};
            dst_data_reg  <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else if (dst_clear_active) begin
            dst_rd_ptr    <= {PTR_W{1'b0}};
            dst_data_reg  <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (!dst_fifo_empty) begin
                dst_data_reg  <= mem[dst_rd_ptr[PTR_W-2:0]];
                dst_valid_reg <= 1'b1;
                dst_rd_ptr    <= dst_rd_ptr + 1'b1;
            end else begin
                dst_valid_reg <= 1'b0;
            end
        end
    end

    assign dst_data_o  = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;

endmodule
