```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;
    localparam GRAY_W = LOG_DEPTH + 1;

    // Source domain
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_W-1:0] src_wr_ptr;
    reg [GRAY_W-1:0] src_wr_gray;
    reg [GRAY_W-1:0] src_rd_gray_sync [0:SYNC_STAGES-1];
    reg [GRAY_W-1:0] src_rd_gray;
    reg src_clear_pending;
    reg src_clear_active;

    // Destination domain
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [GRAY_W-1:0] dst_rd_gray;
    reg [GRAY_W-1:0] dst_wr_gray_sync [0:SYNC_STAGES-1];
    reg [GRAY_W-1:0] dst_wr_gray;
    reg dst_clear_pending;
    reg dst_clear_active;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;

    // Convert binary to Gray
    function [GRAY_W-1:0] bin2gray;
        input [PTR_W-1:0] bin;
        begin
            bin2gray = (bin >> 1) ^ bin;
        end
    endfunction

    // Convert Gray to binary
    function [PTR_W-1:0] gray2bin;
        input [GRAY_W-1:0] gray;
        integer i;
        begin
            gray2bin[GRAY_W-1] = gray[GRAY_W-1];
            for (i = GRAY_W-2; i >= 0; i = i - 1)
                gray2bin[i] = gray2bin[i+1] ^ gray[i];
        end
    endfunction

    // Source domain logic
    wire src_full;
    wire src_empty;
    wire src_can_write;
    wire src_can_read;

    assign src_full = (src_wr_gray == src_rd_gray) && (src_wr_ptr[PTR_W-1] != src_rd_ptr[PTR_W-1]);
    assign src_empty = (src_wr_gray == src_rd_gray) && (src_wr_ptr[PTR_W-1] == src_rd_ptr[PTR_W-1]);
    assign src_can_write = !src_full && !src_clear_active;
    assign src_can_read = !src_empty && !src_clear_active;

    assign src_ready_o = src_can_write;
    assign src_clear_pending_o = src_clear_pending;

    // Source write
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
            src_wr_gray <= {GRAY_W{1'b0}};
            src_clear_pending <= 1'b0;
            src_clear_active <= 1'b0;
        end else begin
            if (src_clear_i) begin
                src_clear_pending <= 1'b1;
                src_clear_active <= 1'b1;
            end else if (src_clear_pending) begin
                // Wait for destination to acknowledge clear
                if (dst_clear_pending_sync) begin
                    src_clear_pending <= 1'b0;
                    src_clear_active <= 1'b0;
                    src_wr_ptr <= {PTR_W{1'b0}};
                    src_wr_gray <= {GRAY_W{1'b0}};
                end
            end else if (src_valid_i && src_can_write) begin
                mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
                src_wr_ptr <= src_wr_ptr + 1'b1;
                src_wr_gray <= bin2gray(src_wr_ptr + 1'b1);
            end
        end
    end

    // Source read pointer sync (from destination)
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rd_gray_sync[0] <= {GRAY_W{1'b0}};
            src_rd_gray_sync[1] <= {GRAY_W{1'b0}};
            src_rd_gray_sync[2] <= {GRAY_W{1'b0}};
        end else begin
            src_rd_gray_sync[0] <= dst_rd_gray;
            src_rd_gray_sync[1] <= src_rd_gray_sync[0];
            src_rd_gray_sync[2] <= src_rd_gray_sync[1];
        end
    end
    assign src_rd_gray = src_rd_gray_sync[SYNC_STAGES-1];

    // Destination domain logic
    wire dst_full;
    wire dst_empty;
    wire dst_can_read;
    wire dst_can_write;

    assign dst_full = (dst_rd_gray == dst_wr_gray) && (dst_rd_ptr[PTR_W-1] != dst_wr_ptr[PTR_W-1]);
    assign dst_empty = (dst_rd_gray == dst_wr_gray) && (dst_rd_ptr[PTR_W-1] == dst_wr_ptr[PTR_W-1]);
    assign dst_can_read = !dst_empty && !dst_clear_active;
    assign dst_can_write = !dst_full && !dst_clear_active;

    assign dst_valid_o = dst_valid_reg;
    assign dst_data_o = dst_data_reg;
    assign dst_clear_pending_o = dst_clear_pending;

    // Destination read
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_rd_gray <= {GRAY_W{1'b0}};
            dst_clear_pending <= 1'b0;
            dst_clear_active <= 1'b0;
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_clear_i) begin
                dst_clear_pending <= 1'b1;
                dst_clear_active <= 1'b1;
                dst_valid_reg <= 1'b0;
            end else if (dst_clear_pending) begin
                // Wait for source to acknowledge clear
                if (src_clear_pending_sync) begin
                    dst_clear_pending <= 1'b0;
                    dst_clear_active <= 1'b0;
                    dst_rd_ptr <= {PTR_W{1'b0}};
                    dst_rd_gray <= {GRAY_W{1'b0}};
                    dst_valid_reg <= 1'b0;
                end
            end else if (dst_valid_reg && dst_ready_i) begin
                dst_valid_reg <= 1'b0;
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_rd_gray <= bin2gray(dst_rd_ptr + 1'b1);
            end else if (!dst_valid_reg && !dst_empty) begin
                dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                dst_valid_reg <= 1'b1;
            end
        end
    end

    // Destination write pointer sync (from source)
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_gray_sync[0] <= {GRAY_W{1'b0}};
            dst_wr_gray_sync[1] <= {GRAY_W{1'b0}};
            dst_wr_gray_sync[2] <= {GRAY_W{1'b0}};
        end else begin
            dst_wr_gray_sync[0] <= src_wr_gray;
            dst_wr_gray_sync[1] <= dst_wr_gray_sync[0];
            dst_wr_gray_sync[2] <= dst_wr_gray_sync[1];
        end
    end
    assign dst_wr_gray = dst_wr_gray_sync[SYNC_STAGES-1];

    // Clear pending sync signals
    reg src_clear_pending_sync [0:SYNC_STAGES-1];
    reg dst_clear_pending_sync [0:SYNC_STAGES-1];

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_pending_sync[0] <= 1'b0;
            src_clear_pending_sync[1] <= 1'b0;
            src_clear_pending_sync[2] <= 1'b0;
        end else begin
            src_clear_pending_sync[0] <= dst_clear_pending;
            src_clear_pending_sync[1] <= src_clear_pending_sync[0];
            src_clear_pending_sync[2] <= src_clear_pending_sync[1];
        end
    end
    wire src_clear_pending_synced = src_clear_pending_sync[SYNC_STAGES-1];

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_pending_sync[0] <= 1'b0;
            dst_clear_pending_sync[1] <= 1'b0;
            dst_clear_pending_sync[2] <= 1'b0;
        end else begin
            dst_clear_pending_sync[0] <= src_clear_pending;
            dst_clear_pending_sync[1] <= dst_clear_pending_sync[0];
            dst_clear_pending_sync[2] <= dst_clear_pending_sync[1];
        end
    end
    wire dst_clear_pending_synced = dst_clear_pending_sync[SYNC_STAGES-1];

    // Fix: use synced versions in the always blocks
    // Re-declare the always blocks with correct sync signals

endmodule
