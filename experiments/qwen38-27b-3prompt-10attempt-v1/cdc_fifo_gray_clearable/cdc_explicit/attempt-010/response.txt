```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;
    localparam GRAY_WIDTH = LOG_DEPTH + 1;

    // Source domain signals
    reg [PTR_WIDTH-1:0] src_wr_ptr;
    reg [GRAY_WIDTH-1:0] src_wr_ptr_gray;
    reg [GRAY_WIDTH-1:0] src_rd_ptr_gray_sync;
    reg [GRAY_WIDTH-1:0] src_rd_ptr_gray_sync_next;
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg src_clear_pending;
    reg src_clear_pending_next;
    reg src_ready;
    reg src_ready_next;
    reg src_valid;
    reg src_valid_next;
    reg src_data_reg [WIDTH-1:0];
    reg src_data_reg_next [WIDTH-1:0];

    // Destination domain signals
    reg [PTR_WIDTH-1:0] dst_rd_ptr;
    reg [GRAY_WIDTH-1:0] dst_rd_ptr_gray;
    reg [GRAY_WIDTH-1:0] dst_wr_ptr_gray_sync;
    reg [GRAY_WIDTH-1:0] dst_wr_ptr_gray_sync_next;
    reg dst_clear_pending;
    reg dst_clear_pending_next;
    reg dst_valid;
    reg dst_valid_next;
    reg [WIDTH-1:0] dst_data_reg;
    reg [WIDTH-1:0] dst_data_reg_next;

    // Helper functions
    function [GRAY_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_WIDTH-1:0] gray2bin;
        input [GRAY_WIDTH-1:0] gray;
        reg [PTR_WIDTH-1:0] bin;
        integer i;
        begin
            bin[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1)
                bin[i] = bin[i+1] ^ gray[i];
            gray2bin = bin;
        end
    endfunction

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
            src_wr_ptr_gray <= {GRAY_WIDTH{1'b0}};
            src_rd_ptr_gray_sync <= {GRAY_WIDTH{1'b0}};
            src_rd_ptr_gray_sync_next <= {GRAY_WIDTH{1'b0}};
            src_clear_pending <= 1'b0;
            src_clear_pending_next <= 1'b0;
            src_ready <= 1'b1;
            src_ready_next <= 1'b1;
            src_valid <= 1'b0;
            src_valid_next <= 1'b0;
            src_data_reg <= {WIDTH{1'b0}};
            src_data_reg_next <= {WIDTH{1'b0}};
        end else begin
            src_rd_ptr_gray_sync <= src_rd_ptr_gray_sync_next;
            src_clear_pending <= src_clear_pending_next;
            src_ready <= src_ready_next;
            src_valid <= src_valid_next;
            src_data_reg <= src_data_reg_next;

            if (src_clear_i) begin
                src_clear_pending_next <= 1'b1;
                src_ready_next <= 1'b0;
                src_valid_next <= 1'b0;
                src_data_reg_next <= {WIDTH{1'b0}};
                src_wr_ptr <= {PTR_WIDTH{1'b0}};
                src_wr_ptr_gray <= {GRAY_WIDTH{1'b0}};
            end else if (src_clear_pending) begin
                src_clear_pending_next <= 1'b0;
                src_ready_next <= 1'b1;
                src_valid_next <= 1'b0;
                src_data_reg_next <= {WIDTH{1'b0}};
            end else begin
                src_clear_pending_next <= 1'b0;
                if (src_valid_i && src_ready) begin
                    src_ready_next <= 1'b0;
                    src_valid_next <= 1'b1;
                    src_data_reg_next <= src_data_i;
                    mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
                    src_wr_ptr <= src_wr_ptr + 1'b1;
                    src_wr_ptr_gray <= bin2gray(src_wr_ptr + 1'b1);
                end else begin
                    src_ready_next <= 1'b1;
                    src_valid_next <= 1'b0;
                    src_data_reg_next <= {WIDTH{1'b0}};
                end
            end
        end
    end

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_gray <= {GRAY_WIDTH{1'b0}};
            dst_wr_ptr_gray_sync <= {GRAY_WIDTH{1'b0}};
            dst_wr_ptr_gray_sync_next <= {GRAY_WIDTH{1'b0}};
            dst_clear_pending <= 1'b0;
            dst_clear_pending_next <= 1'b0;
            dst_valid <= 1'b0;
            dst_valid_next <= 1'b0;
            dst_data_reg <= {WIDTH{1'b0}};
            dst_data_reg_next <= {WIDTH{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync <= dst_wr_ptr_gray_sync_next;
            dst_clear_pending <= dst_clear_pending_next;
            dst_valid <= dst_valid_next;
            dst_data_reg <= dst_data_reg_next;

            if (dst_clear_i) begin
                dst_clear_pending_next <= 1'b1;
                dst_valid_next <= 1'b0;
                dst_data_reg_next <= {WIDTH{1'b0}};
                dst_rd_ptr <= {PTR_WIDTH{1'b0}};
                dst_rd_ptr_gray <= {GRAY_WIDTH{1'b0}};
            end else if (dst_clear_pending) begin
                dst_clear_pending_next <= 1'b0;
                dst_valid_next <= 1'b0;
                dst_data_reg_next <= {WIDTH{1'b0}};
            end else begin
                dst_clear_pending_next <= 1'b0;
                if (dst_valid && dst_ready_i) begin
                    dst_valid_next <= 1'b0;
                    dst_data_reg_next <= {WIDTH{1'b0}};
                    dst_rd_ptr <= dst_rd_ptr + 1'b1;
                    dst_rd_ptr_gray <= bin2gray(dst_rd_ptr + 1'b1);
                end else begin
                    dst_valid_next <= 1'b0;
                    dst_data_reg_next <= {WIDTH{1'b0}};
                end
            end
        end
    end

    // Synchronizers
    // Source side: synchronize dst_rd_ptr_gray to src domain
    reg [GRAY_WIDTH-1:0] src_sync_stage [0:SYNC_STAGES-1];
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            for (integer i = 0; i < SYNC_STAGES; i = i + 1)
                src_sync_stage[i] <= {GRAY_WIDTH{1'b0}};
        end else begin
            src_sync_stage[0] <= dst_rd_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                src_sync_stage[i] <= src_sync_stage[i-1];
        end
    end

    // Destination side: synchronize src_wr_ptr_gray to dst domain
    reg [GRAY_WIDTH-1:0] dst_sync_stage [0:SYNC_STAGES-1];
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            for (integer i = 0; i < SYNC_STAGES; i = i + 1)
                dst_sync_stage[i] <= {GRAY_WIDTH{1'b0}};
        end else begin
            dst_sync_stage[0] <= src_wr_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                dst_sync_stage[i] <= dst_sync_stage[i-1];
        end
    end

    // Compute next values for synchronizer outputs
    always @(*) begin
        src_rd_ptr_gray_sync_next = src_sync_stage[SYNC_STAGES-1];
        dst_wr_ptr_gray_sync_next = dst_sync_stage[SYNC_STAGES-1];
    end

    // Compute FIFO full/empty
    wire [PTR_WIDTH-1:0] src_wr_ptr_bin = src_wr_ptr;
    wire [PTR_WIDTH-1:0] src_rd_ptr_bin = gray2bin(src_rd_ptr_gray_sync);
    wire [PTR_WIDTH-1:0] dst_rd_ptr_bin = dst_rd_ptr;
    wire [PTR_WIDTH-1:0] dst_wr_ptr_bin = gray2bin(dst_wr_ptr_gray_sync);

    wire src_fifo_full = (src_wr_ptr_bin == src_rd_ptr_bin) ? 1'b0 : 
                         (src_wr_ptr_bin[LOG_DEPTH] != src_rd_ptr_bin[LOG_DEPTH]) && 
                         (src_wr_ptr_bin[LOG_DEPTH-1:0] == src_rd_ptr_bin[LOG_DEPTH-1:0]);
    wire dst_fifo_empty = (dst_rd_ptr_bin == dst_wr_ptr_bin);

    // Update ready and valid based on FIFO status
    always @(*) begin
        if (src_clear_i || src_clear_pending) begin
            src_ready = 1'b0;
        end else begin
            src_ready = ~src_fifo_full;
        end

        if (dst_clear_i || dst_clear_pending) begin
            dst_valid = 1'b0;
        end else begin
            dst_valid = ~dst_fifo_empty;
        end
    end

    // Read data from memory when destination is ready and valid
    wire dst_read_enable = dst_valid && dst_ready_i && !dst_clear_i && !dst_clear_pending;
    wire [WIDTH-1:0] mem_read_data = mem[dst_rd_ptr[LOG_DEPTH-1:0]];

    always @(*) begin
        if (dst_read_enable) begin
            dst_data_reg_next = mem_read_data;
        end else begin
            dst_data_reg_next = dst_data_reg;
        end
    end

    // Assign outputs
    assign src_clear_pending_o = src_clear_pending;
    assign src_ready_o = src_ready;
    assign dst_clear_pending_o = dst_clear_pending;
    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid;

endmodule
