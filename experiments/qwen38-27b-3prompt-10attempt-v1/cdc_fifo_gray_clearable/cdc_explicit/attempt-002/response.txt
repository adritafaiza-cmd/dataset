```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Source domain signals
    reg  [PTR_W-1:0] src_wr_ptr;
    reg  [PTR_W-1:0] src_rd_ptr_gray_sync;
    reg  [WIDTH-1:0] mem [0:DEPTH-1];
    reg              src_clear_pending;
    reg              src_clear_active;

    // Destination domain signals
    reg  [PTR_W-1:0] dst_rd_ptr;
    reg  [PTR_W-1:0] dst_wr_ptr_gray_sync;
    reg  [WIDTH-1:0] dst_data_reg;
    reg              dst_valid_reg;
    reg              dst_clear_pending;
    reg              dst_clear_active;

    // Gray code conversion functions
    function [PTR_W-1:0] bin2gray;
        input [PTR_W-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_W-1:0] gray2bin;
        input [PTR_W-1:0] gray;
        reg [PTR_W-1:0] bin;
        integer i;
        begin
            bin[PTR_W-1] = gray[PTR_W-1];
            for (i = PTR_W-2; i >= 0; i = i - 1)
                bin[i] = bin[i+1] ^ gray[i];
            gray2bin = bin;
        end
    endfunction

    // Source domain: write pointer and memory
    wire [PTR_W-1:0] src_wr_ptr_gray = bin2gray(src_wr_ptr);
    wire [PTR_W-1:0] src_rd_ptr_bin = gray2bin(src_rd_ptr_gray_sync);

    wire src_full = (src_wr_ptr_gray == src_rd_ptr_gray_sync) && (src_wr_ptr[PTR_W-1] != src_rd_ptr_bin[PTR_W-1]);
    wire src_empty = (src_wr_ptr == src_rd_ptr_bin);

    // Source clear logic
    wire src_clear_req = src_clear_i | src_clear_active;
    wire src_clear_done = src_clear_active && src_empty;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
            src_clear_pending <= 1'b0;
            src_clear_active <= 1'b0;
        end else begin
            if (src_clear_i) begin
                src_clear_active <= 1'b1;
                src_clear_pending <= 1'b1;
            end else if (src_clear_active) begin
                if (src_empty) begin
                    src_clear_active <= 1'b0;
                    src_clear_pending <= 1'b0;
                end
            end
        end
    end

    // Source write
    wire src_write_en = src_valid_i & src_ready_o & ~src_clear_active;
    wire src_ready = ~src_full & ~src_clear_active;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
        end else if (src_write_en) begin
            mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
            src_wr_ptr <= src_wr_ptr + 1'b1;
        end
    end

    // Destination domain: read pointer and output
    wire [PTR_W-1:0] dst_rd_ptr_gray = bin2gray(dst_rd_ptr);
    wire [PTR_W-1:0] dst_wr_ptr_bin = gray2bin(dst_wr_ptr_gray_sync);

    wire dst_full = (dst_rd_ptr_gray == dst_wr_ptr_gray_sync) && (dst_rd_ptr[PTR_W-1] != dst_wr_ptr_bin[PTR_W-1]);
    wire dst_empty = (dst_rd_ptr == dst_wr_ptr_bin);

    // Destination clear logic
    wire dst_clear_req = dst_clear_i | dst_clear_active;
    wire dst_clear_done = dst_clear_active && dst_empty;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_clear_pending <= 1'b0;
            dst_clear_active <= 1'b0;
            dst_valid_reg <= 1'b0;
            dst_data_reg <= {WIDTH{1'b0}};
        end else begin
            if (dst_clear_i) begin
                dst_clear_active <= 1'b1;
                dst_clear_pending <= 1'b1;
                dst_valid_reg <= 1'b0;
            end else if (dst_clear_active) begin
                if (dst_empty && !dst_valid_reg) begin
                    dst_clear_active <= 1'b0;
                    dst_clear_pending <= 1'b0;
                end
            end
        end
    end

    // Destination read
    wire dst_read_en = dst_valid_reg & dst_ready_i & ~dst_clear_active;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_valid_reg <= 1'b0;
            dst_data_reg <= {WIDTH{1'b0}};
        end else begin
            if (dst_clear_active) begin
                dst_valid_reg <= 1'b0;
            end else if (dst_read_en) begin
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_valid_reg <= 1'b0;
            end else if (!dst_valid_reg && !dst_empty) begin
                dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                dst_valid_reg <= 1'b1;
            end
        end
    end

    // CDC synchronizers
    // Source read pointer gray -> destination domain
    reg [SYNC_STAGES-1:0][PTR_W-1:0] src_rd_ptr_gray_sync_pipe;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_rd_ptr_gray_sync_pipe <= {SYNC_STAGES{1'b0}};
        end else begin
            src_rd_ptr_gray_sync_pipe[0] <= dst_rd_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                src_rd_ptr_gray_sync_pipe[i] <= src_rd_ptr_gray_sync_pipe[i-1];
        end
    end
    assign src_rd_ptr_gray_sync = src_rd_ptr_gray_sync_pipe[SYNC_STAGES-1];

    // Destination write pointer gray -> source domain
    reg [SYNC_STAGES-1:0][PTR_W-1:0] dst_wr_ptr_gray_sync_pipe;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_wr_ptr_gray_sync_pipe <= {SYNC_STAGES{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync_pipe[0] <= src_wr_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                dst_wr_ptr_gray_sync_pipe[i] <= dst_wr_ptr_gray_sync_pipe[i-1];
        end
    end
    assign dst_wr_ptr_gray_sync = dst_wr_ptr_gray_sync_pipe[SYNC_STAGES-1];

    // Outputs
    assign src_ready_o = src_ready;
    assign src_clear_pending_o = src_clear_pending;
    assign dst_valid_o = dst_valid_reg;
    assign dst_data_o = dst_data_reg;
    assign dst_clear_pending_o = dst_clear_pending;

endmodule
