module isochronous_4phase_handshake (
    input  src_clk_i,
    input  src_rst_ni,
    input  src_valid_i,
    output src_ready_o,
    input  dst_clk_i,
    input  dst_rst_ni,
    output dst_valid_o,
    input  dst_ready_i
);

    // Source domain registers
    reg src_valid_r;
    reg src_ack_r;

    // Destination domain registers
    reg dst_valid_r;
    reg dst_ack_r;

    // Synchronizer: src_valid_r (src domain) to dst domain
    reg [1:0] src_valid_sync;

    // Synchronizer: dst_ack_r (dst domain) to src domain
    reg [1:0] dst_ack_sync;

    // Synchronizer: dst_valid_r (dst domain) to src domain - not needed for 4-phase
    // Synchronizer: src_ack_r (src domain) to dst domain - not needed for 4-phase

    // Source domain: drive valid, receive ack
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_valid_r <= 1'b0;
            src_ack_r   <= 1'b0;
        end else begin
            // Four-phase handshake source side
            // src_valid_r: valid signal driven by source
            // src_ack_r: ack signal received from destination (synchronized)
            
            // Phase 0: valid=0, ack=0 -> idle
            // Phase 1: valid=1, ack=0 -> request
            // Phase 2: valid=1, ack=1 -> acknowledged
            // Phase 3: valid=0, ack=1 -> deasserting valid
            // Phase 4: valid=0, ack=0 -> back to idle
            
            if (src_valid_r == 1'b0 && src_ack_r == 1'b0) begin
                // Idle state
                if (src_valid_i)
                    src_valid_r <= 1'b1;
                else
                    src_valid_r <= 1'b0;
            end else if (src_valid_r == 1'b1 && src_ack_r == 1'b0) begin
                // Request state, wait for ack
                src_valid_r <= 1'b1;
            end else if (src_valid_r == 1'b1 && src_ack_r == 1'b1) begin
                // Acknowledged state, deassert valid
                src_valid_r <= 1'b0;
            end else if (src_valid_r == 1'b0 && src_ack_r == 1'b1) begin
                // Deasserting valid, wait for ack to deassert
                src_valid_r <= 1'b0;
            end
        end
    end

    // Destination domain: receive valid, drive ack
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_valid_r <= 1'b0;
            dst_ack_r   <= 1'b0;

            // dst_valid_r: valid signal received from source (synchronized)
            // dst_ack_r: ack signal driven by destination

                dst_ack_r <= 1'b1;
            end else if (dst_valid_r == 1'b0 && dst_ack_r == 1'b1) begin

                dst_ack_r <= 1'b0;
