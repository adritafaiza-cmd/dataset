module isochronous_4phase_handshake (
    input  src_clk_i,
    input  src_rst_ni,
    input  src_valid_i,
    output src_ready_o,
    input  dst_clk_i,
    input  dst_rst_ni,
    output dst_valid_o,
    input  dst_ready_i
);

    // Synchronize src_valid_i into dst domain
    reg [2:0] src_valid_sync;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            src_valid_sync <= 3'b000;
        else
            src_valid_sync <= {src_valid_sync[1:0], src_valid_i};
    end
    wire src_valid_dst = src_valid_sync[2];

    // Synchronize dst_ready_i into src domain
    reg [2:0] dst_ready_sync;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            dst_ready_sync <= 3'b000;
        else
            dst_ready_sync <= {dst_ready_sync[1:0], dst_ready_i};
    end
    wire dst_ready_src = dst_ready_sync[2];

    // Synchronize dst_valid_o into src domain
    reg [2:0] dst_valid_sync;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            dst_valid_sync <= 3'b000;
        else
            dst_valid_sync <= {dst_valid_sync[1:0], dst_valid_o};
    end
    wire dst_valid_src = dst_valid_sync[2];

    // Synchronize src_ready_o into dst domain
    reg [2:0] src_ready_sync;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            src_ready_sync <= 3'b000;
        else
            src_ready_sync <= {src_ready_sync[1:0], src_ready_o};
    end
    wire src_ready_dst = src_ready_sync[2];

    // Four-phase handshake state machine in src domain
    // States: IDLE, WAIT_DST_READY, WAIT_SRC_VALID, WAIT_DST_VALID
    localparam IDLE          = 2'b00;
    localparam WAIT_DST_RDY  = 2'b01;
    localparam WAIT_SRC_VAL  = 2'b10;
    localparam WAIT_DST_VAL  = 2'b11;

    reg [1:0] src_state;
    reg       src_ready_reg;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_state     <= IDLE;
            src_ready_reg <= 1'b0;
        end else begin
            case (src_state)
                IDLE: begin
                    if (src_valid_i) begin
                        src_state     <= WAIT_DST_RDY;
                        src_ready_reg <= 1'b1;
                    end else begin
                        src_state     <= IDLE;
                        src_ready_reg <= 1'b0;
                    end
                end
                WAIT_DST_RDY: begin
                    if (dst_ready_src) begin
                        src_state     <= WAIT_SRC_VAL;
                        src_ready_reg <= 1'b0;
                    end else begin
                        src_state     <= WAIT_DST_RDY;
                        src_ready_reg <= 1'b1;
                    end
                end
                WAIT_SRC_VAL: begin
                    if (!src_valid_i) begin
                        src_state     <= WAIT_DST_VAL;
                        src_ready_reg <= 1'b0;
                    end else begin
                        src_state     <= WAIT_SRC_VAL;
                        src_ready_reg <= 1'b0;
                    end
                end
                WAIT_DST_VAL: begin
                    if (!dst_valid_src) begin
                        src_state     <= IDLE;
                        src_ready_reg <= 1'b0;
                    end else begin
                        src_state     <= WAIT_DST_VAL;
                        src_ready_reg <= 1'b0;
                    end
                end
                default: begin
                    src_state     <= IDLE;
                    src_ready_reg <= 1'b0;
                end
            endcase
        end
    end

    assign src_ready_o = src_ready_reg;

    // Four-phase handshake state machine in dst domain
    localparam DST_IDLE          = 2'b00;
    localparam DST_WAIT_SRC_RDY  = 2'b01;
    localparam DST_WAIT_DST_RDY  = 2'b10;
    localparam DST_WAIT_SRC_VAL  = 2'b11;

    reg [1:0] dst_state;
    reg       dst_valid_reg;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_state     <= DST_IDLE;
            dst_valid_reg <= 1'b0;
        end else begin
            case (dst_state)
                DST_IDLE: begin
                    if (src_valid_dst) begin
                        dst_state     <= DST_WAIT_SRC_RDY;
                        dst_valid_reg <= 1'b1;
                    end else begin
                        dst_state     <= DST_IDLE;
                        dst_valid_reg <= 1'b0;
                    end
                end
                DST_WAIT_SRC_RDY: begin
                    if (src_ready_dst) begin
                        dst_state     <= DST_WAIT_DST_RDY;
                        dst_valid_reg <= 1'b1;
                    end else begin
                        dst_state     <= DST_WAIT_SRC_RDY;
                        dst_valid_reg <= 1'b1;
                    end
                end
                DST_WAIT_DST_RDY: begin
                    if (dst_ready_i) begin
                        dst_state     <= DST_WAIT_SRC_VAL;
                        dst_valid_reg <= 1'b0;
                    end else begin
                        dst_state     <= DST_WAIT_DST_RDY;
                        dst_valid_reg <= 1'b1;
                    end
                end
                DST_WAIT_SRC_VAL: begin
                    if (!src_valid_dst) begin
                        dst_state     <= DST_IDLE;
                        dst_valid_reg <= 1'b0;
                    end else begin
                        dst_state     <= DST_WAIT_SRC_VAL;
                        dst_valid_reg <= 1'b0;
                    end
                end
                default: begin
                    dst_state     <= DST_IDLE;
                    dst_valid_reg <= 1'b0;
                end
            endcase
        end
    end

    assign dst_valid_o = dst_valid_reg;

endmodule
