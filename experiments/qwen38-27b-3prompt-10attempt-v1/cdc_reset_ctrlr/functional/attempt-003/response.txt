module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o,
    input  b_isolate_ack_i
);

    // State encoding
    localparam S_IDLE     = 2'd0;
    localparam S_ISOLATE  = 2'd1;
    localparam S_CLEAR    = 2'd2;
    localparam S_RELEASE  = 2'd3;

    // Local state
    reg [1:0] a_state;
    reg [1:0] b_state;

    // Synchronization registers
    reg [SYNC_STAGES-1:0] a_state_sync;
    reg [SYNC_STAGES-1:0] b_state_sync;

    // Synchronized state values
    wire [1:0] a_state_synced = a_state_sync[SYNC_STAGES-1:0];
    wire [1:0] b_state_synced = b_state_sync[SYNC_STAGES-1:0];

    // Synchronize a_state into b domain
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni)
            a_state_sync <= {SYNC_STAGES{1'b0}};
        else
            a_state_sync <= {a_state_sync[SYNC_STAGES-2:0], a_state};
    end

    // Synchronize b_state into a domain
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni)
            b_state_sync <= {SYNC_STAGES{1'b0}};
        else
            b_state_sync <= {b_state_sync[SYNC_STAGES-2:0], b_state};
    end

    // A domain state machine
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni)
            a_state <= S_IDLE;
        else begin
            case (a_state)
                S_IDLE: begin
                    if (a_clear_i || b_state_synced != S_IDLE)
                        a_state <= S_ISOLATE;
                end
                S_ISOLATE: begin
                    if (a_isolate_ack_i && b_state_synced == S_ISOLATE)
                        a_state <= S_CLEAR;
                end
                S_CLEAR: begin
                    if (a_clear_ack_i && b_state_synced == S_CLEAR)
                        a_state <= S_RELEASE;
                end
                S_RELEASE: begin
                    if (b_state_synced == S_IDLE)
                        a_state <= S_IDLE;
                end
                default: a_state <= S_IDLE;
            endcase
        end
    end

    // B domain state machine
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni)
            b_state <= S_IDLE;
        else begin
            case (b_state)
                S_IDLE: begin
                    if (b_clear_i || a_state_synced != S_IDLE)
                        b_state <= S_ISOLATE;
                end
                S_ISOLATE: begin
                    if (b_isolate_ack_i && a_state_synced == S_ISOLATE)
                        b_state <= S_CLEAR;
                end
                S_CLEAR: begin
                    if (b_clear_ack_i && a_state_synced == S_CLEAR)
                        b_state <= S_RELEASE;
                end
                S_RELEASE: begin
                    if (a_state_synced == S_IDLE)
                        b_state <= S_IDLE;
                end
                default: b_state <= S_IDLE;
            endcase
        end
    end

    // Outputs
    assign a_isolate_o = (a_state == S_ISOLATE) || (a_state == S_CLEAR) || (a_state == S_RELEASE);
    assign a_clear_o   = (a_state == S_CLEAR) || (a_state == S_RELEASE);
    assign b_isolate_o = (b_state == S_ISOLATE) || (b_state == S_CLEAR) || (b_state == S_RELEASE);
    assign b_clear_o   = (b_state == S_CLEAR) || (b_state == S_RELEASE);

endmodule
