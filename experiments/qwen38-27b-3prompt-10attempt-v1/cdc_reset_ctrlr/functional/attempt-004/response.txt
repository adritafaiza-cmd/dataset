module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o,
    input  b_isolate_ack_i
);

    // State encoding
    localparam S_IDLE    = 2'd0;
    localparam S_ISOLATE = 2'd1;
    localparam S_CLEAR   = 2'd2;

    // A domain state
    reg [1:0] a_state;
    // B domain state
    reg [1:0] b_state;

    // Synchronization signals: A -> B
    // a_isolate_req_sync: synchronized version of a_isolate_o
    // a_clear_req_sync: synchronized version of a_clear_o
    reg [SYNC_STAGES-1:0] a_isolate_req_sync_pipe;
    reg [SYNC_STAGES-1:0] a_clear_req_sync_pipe;
    wire a_isolate_req_sync = a_isolate_req_sync_pipe[SYNC_STAGES-1];
    wire a_clear_req_sync   = a_clear_req_sync_pipe[SYNC_STAGES-1];

    // Synchronization signals: B -> A
    // b_isolate_req_sync: synchronized version of b_isolate_o
    // b_clear_req_sync: synchronized version of b_clear_o
    reg [SYNC_STAGES-1:0] b_isolate_req_sync_pipe;
    reg [SYNC_STAGES-1:0] b_clear_req_sync_pipe;
    wire b_isolate_req_sync = b_isolate_req_sync_pipe[SYNC_STAGES-1];
    wire b_clear_req_sync   = b_clear_req_sync_pipe[SYNC_STAGES-1];

    // A domain: synchronize B's outputs to A domain
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            b_isolate_req_sync_pipe <= {SYNC_STAGES{1'b0}};
            b_clear_req_sync_pipe   <= {SYNC_STAGES{1'b0}};
        end else begin
            b_isolate_req_sync_pipe <= {b_isolate_req_sync_pipe[SYNC_STAGES-2:0], b_isolate_o};
            b_clear_req_sync_pipe   <= {b_clear_req_sync_pipe[SYNC_STAGES-2:0], b_clear_o};
        end
    end

    // B domain: synchronize A's outputs to B domain
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            a_isolate_req_sync_pipe <= {SYNC_STAGES{1'b0}};
            a_clear_req_sync_pipe   <= {SYNC_STAGES{1'b0}};
        end else begin
            a_isolate_req_sync_pipe <= {a_isolate_req_sync_pipe[SYNC_STAGES-2:0], a_isolate_o};
            a_clear_req_sync_pipe   <= {a_clear_req_sync_pipe[SYNC_STAGES-2:0], a_clear_o};
        end
    end

    // A domain state machine
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_state <= S_IDLE;
        end else begin
            case (a_state)
                S_IDLE: begin
                    if (a_clear_i || b_clear_req_sync) begin
                        a_state <= S_ISOLATE;
                    end
                end
                S_ISOLATE: begin
                    // Wait for both isolate acks (local and peer)
                    if (a_isolate_ack_i && b_isolate_req_sync) begin
                        a_state <= S_CLEAR;
                    end
                end
                S_CLEAR: begin
                    // Wait for both clear acks (local and peer)
                    if (a_clear_ack_i && b_clear_req_sync) begin
                        a_state <= S_IDLE;
                    end
                end
                default: a_state <= S_IDLE;
            endcase
        end
    end

    // B domain state machine
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_state <= S_IDLE;
        end else begin
            case (b_state)
                S_IDLE: begin
                    if (b_clear_i || a_clear_req_sync) begin
                        b_state <= S_ISOLATE;
                    end
                end
                S_ISOLATE: begin
                    // Wait for both isolate acks (local and peer)
                    if (b_isolate_ack_i && a_isolate_req_sync) begin
                        b_state <= S_CLEAR;
                    end
                end
                S_CLEAR: begin
                    // Wait for both clear acks (local and peer)
                    if (b_clear_ack_i && a_clear_req_sync) begin
                        b_state <= S_IDLE;
                    end
                end
                default: b_state <= S_IDLE;
            endcase
        end
    end

    // A domain outputs
    assign a_isolate_o = (a_state == S_ISOLATE) || (a_state == S_CLEAR);
    assign a_clear_o   = (a_state == S_CLEAR);

    // B domain outputs
    assign b_isolate_o = (b_state == S_ISOLATE) || (b_state == S_CLEAR);
    assign b_clear_o   = (b_state == S_CLEAR);

endmodule
