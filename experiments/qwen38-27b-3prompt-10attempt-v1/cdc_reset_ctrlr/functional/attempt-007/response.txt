module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o,
    input  b_isolate_ack_i
);

    // State encoding
    localparam S_IDLE     = 2'd0;
    localparam S_ISOLATE  = 2'd1;
    localparam S_CLEAR    = 2'd2;
    localparam S_RELEASE  = 2'd3;

    // Local state registers
    reg [1:0] a_state;
    reg [1:0] b_state;

    // Synchronized peer state
    reg [1:0] a_peer_state_sync;
    reg [1:0] b_peer_state_sync;

    // Synchronization chains for peer state
    // A side synchronizes B's state
    reg [1:0] b_state_sync_1;
    reg [1:0] b_state_sync_2;
    reg [1:0] b_state_sync_3;

    // B side synchronizes A's state
    reg [1:0] a_state_sync_1;
    reg [1:0] a_state_sync_2;
    reg [1:0] a_state_sync_3;

    // Synchronize B state into A domain
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            b_state_sync_1 <= 2'd0;
            b_state_sync_2 <= 2'd0;
            b_state_sync_3 <= 2'd0;
        end else begin
            b_state_sync_1 <= b_state;
            b_state_sync_2 <= b_state_sync_1;
            b_state_sync_3 <= b_state_sync_2;
        end
    end

    // Synchronize A state into B domain
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            a_state_sync_1 <= 2'd0;
            a_state_sync_2 <= 2'd0;
            a_state_sync_3 <= 2'd0;
        end else begin
            a_state_sync_1 <= a_state;
            a_state_sync_2 <= a_state_sync_1;
            a_state_sync_3 <= a_state_sync_2;
        end
    end

    // A domain state machine
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_state <= S_IDLE;
        end else begin
            case (a_state)
                S_IDLE: begin
                    if (a_clear_i) begin
                        a_state <= S_ISOLATE;
                    end
                end
                S_ISOLATE: begin
                    // Wait for both isolate acks: local ack and peer in ISOLATE or beyond
                    if (a_isolate_ack_i && (b_state_sync_3 == S_ISOLATE || b_state_sync_3 == S_CLEAR || b_state_sync_3 == S_RELEASE)) begin
                        a_state <= S_CLEAR;
                    end
                end
                S_CLEAR: begin
                    // Wait for both clear acks: local ack and peer in CLEAR or beyond
                    if (a_clear_ack_i && (b_state_sync_3 == S_CLEAR || b_state_sync_3 == S_RELEASE)) begin
                        a_state <= S_RELEASE;
                    end
                end
                S_RELEASE: begin
                    // Wait for peer to return to IDLE
                    if (b_state_sync_3 == S_IDLE) begin
                        a_state <= S_IDLE;
                    end
                end
                default: a_state <= S_IDLE;
            endcase
        end
    end

    // B domain state machine
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_state <= S_IDLE;
        end else begin
            case (b_state)
                S_IDLE: begin
                    if (b_clear_i) begin
                        b_state <= S_ISOLATE;
                    end
                end
                S_ISOLATE: begin
                    // Wait for both isolate acks: local ack and peer in ISOLATE or beyond
                    if (b_isolate_ack_i && (a_state_sync_3 == S_ISOLATE || a_state_sync_3 == S_CLEAR || a_state_sync_3 == S_RELEASE)) begin
                        b_state <= S_CLEAR;
                    end
                end
                S_CLEAR: begin
                    // Wait for both clear acks: local ack and peer in CLEAR or beyond
                    if (b_clear_ack_i && (a_state_sync_3 == S_CLEAR || a_state_sync_3 == S_RELEASE)) begin
                        b_state <= S_RELEASE;
                    end
                end
                S_RELEASE: begin
                    // Wait for peer to return to IDLE
                    if (a_state_sync_3 == S_IDLE) begin
                        b_state <= S_IDLE;
                    end
                end
                default: b_state <= S_IDLE;
            endcase
        end
    end

    // Output assignments
    assign a_isolate_o = (a_state == S_ISOLATE) || (a_state == S_CLEAR);
    assign a_clear_o   = (a_state == S_CLEAR);
    assign b_isolate_o = (b_state == S_ISOLATE) || (b_state == S_CLEAR);
    assign b_clear_o   = (b_state == S_CLEAR);

endmodule
