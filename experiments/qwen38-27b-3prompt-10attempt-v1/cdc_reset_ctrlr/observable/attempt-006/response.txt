module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o,
    input  b_isolate_ack_i
);

    // State encoding
    localparam S_IDLE    = 3'd0;
    localparam S_ISOLATE = 3'd1;
    localparam S_CLEAR   = 3'd2;
    localparam S_DEISO   = 3'd3;

    // A-side state
    reg [2:0] a_state;
    reg       a_isolate_o_r;
    reg       a_clear_o_r;

    // B-side state
    reg [2:0] b_state;
    reg       b_isolate_o_r;
    reg       b_clear_o_r;

    // A-side: B's state synchronized into A domain
    reg [2:0] b_state_sync;
    reg [2:0] b_state_sync_r1;
    reg [2:0] b_state_sync_r2;

    // B-side: A's state synchronized into B domain
    reg [2:0] a_state_sync;
    reg [2:0] a_state_sync_r1;
    reg [2:0] a_state_sync_r2;

    // A-side: B's isolate_ack synchronized into A domain
    reg       b_isolate_ack_sync;
    reg       b_isolate_ack_sync_r1;
    reg       b_isolate_ack_sync_r2;

    // B-side: A's isolate_ack synchronized into B domain
    reg       a_isolate_ack_sync;
    reg       a_isolate_ack_sync_r1;
    reg       a_isolate_ack_sync_r2;

    // A-side: B's clear_ack synchronized into A domain
    reg       b_clear_ack_sync;
    reg       b_clear_ack_sync_r1;
    reg       b_clear_ack_sync_r2;

    // B-side: A's clear_ack synchronized into B domain
    reg       a_clear_ack_sync;
    reg       a_clear_ack_sync_r1;
    reg       a_clear_ack_sync_r2;

    // Synchronize B state into A domain
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            b_state_sync_r1 <= 3'd0;
            b_state_sync_r2 <= 3'd0;
            b_state_sync    <= 3'd0;
        end else begin
            b_state_sync_r1 <= b_state;
            b_state_sync_r2 <= b_state_sync_r1;
            b_state_sync    <= b_state_sync_r2;
        end
    end

    // Synchronize A state into B domain
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            a_state_sync_r1 <= 3'd0;
            a_state_sync_r2 <= 3'd0;
            a_state_sync    <= 3'd0;
        end else begin
            a_state_sync_r1 <= a_state;
            a_state_sync_r2 <= a_state_sync_r1;
            a_state_sync    <= a_state_sync_r2;
        end
    end

    // Synchronize B isolate_ack into A domain
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            b_isolate_ack_sync_r1 <= 1'b0;
            b_isolate_ack_sync_r2 <= 1'b0;
            b_isolate_ack_sync    <= 1'b0;
        end else begin
            b_isolate_ack_sync_r1 <= b_isolate_ack_i;
            b_isolate_ack_sync_r2 <= b_isolate_ack_sync_r1;
            b_isolate_ack_sync    <= b_isolate_ack_sync_r2;
        end
    end

    // Synchronize A isolate_ack into B domain
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            a_isolate_ack_sync_r1 <= 1'b0;
            a_isolate_ack_sync_r2 <= 1'b0;
            a_isolate_ack_sync    <= 1'b0;
        end else begin
            a_isolate_ack_sync_r1 <= a_isolate_ack_i;
            a_isolate_ack_sync_r2 <= a_isolate_ack_sync_r1;
            a_isolate_ack_sync    <= a_isolate_ack_sync_r2;
        end
    end

    // Synchronize B clear_ack into A domain
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            b_clear_ack_sync_r1 <= 1'b0;
            b_clear_ack_sync_r2 <= 1'b0;
            b_clear_ack_sync    <= 1'b0;
        end else begin
            b_clear_ack_sync_r1 <= b_clear_ack_i;
            b_clear_ack_sync_r2 <= b_clear_ack_sync_r1;
            b_clear_ack_sync    <= b_clear_ack_sync_r2;
        end
    end

    // Synchronize A clear_ack into B domain
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            a_clear_ack_sync_r1 <= 1'b0;
            a_clear_ack_sync_r2 <= 1'b0;
            a_clear_ack_sync    <= 1'b0;
        end else begin
            a_clear_ack_sync_r1 <= a_clear_ack_i;
            a_clear_ack_sync_r2 <= a_clear_ack_sync_r1;
            a_clear_ack_sync    <= a_clear_ack_sync_r2;
        end
    end

    // A-side state machine
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_state       <= S_IDLE;
            a_isolate_o_r <= 1'b0;
            a_clear_o_r   <= 1'b0;
        end else begin
            case (a_state)
                S_IDLE: begin
                    a_isolate_o_r <= 1'b0;
                    a_clear_o_r   <= 1'b0;
                    if (a_clear_i || b_clear_i) begin
                        a_state <= S_ISOLATE;
                    end
                end
                S_ISOLATE: begin
                    a_isolate_o_r <= 1'b1;
                    a_clear_o_r   <= 1'b0;
                    // Wait for both sides to acknowledge isolate
                    if (a_isolate_ack_i && b_isolate_ack_sync) begin
                        a_state <= S_CLEAR;
                    end
                end
                S_CLEAR: begin
                    a_isolate_o_r <= 1'b1;
                    a_clear_o_r   <= 1'b1;
                    // Wait for both sides to acknowledge clear
                    if (a_clear_ack_i && b_clear_ack_sync) begin
                        a_state <= S_DEISO;
                    end
                end
                S_DEISO: begin
                    a_isolate_o_r <= 1'b0;
                    a_clear_o_r   <= 1'b0;
                    // Wait for both sides to acknowledge de-isolate (back to idle)
                    if (a_isolate_ack_i && b_isolate_ack_sync) begin
                        a_state <= S_IDLE;
                    end
                end
                default: begin
                    a_state       <= S_IDLE;
                    a_isolate_o_r <= 1'b0;
                    a_clear_o_r   <= 1'b0;
                end
            endcase
        end
    end

    // B-side state machine
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_state       <= S_IDLE;
            b_isolate_o_r <= 1'b0;
            b_clear_o_r   <= 1'b0;
        end else begin
            case (b_state)
                S_IDLE: begin
                    b_isolate_o_r <= 1'b0;
                    b_clear_o_r   <= 1'b0;
                    if (a_clear_i || b_clear_i) begin
                        b_state <= S_ISOLATE;
                    end
                end
                S_ISOLATE: begin
                    b_isolate_o_r <= 1'b1;
                    b_clear_o_r   <= 1'b0;
                    // Wait for both sides to acknowledge isolate
                    if (b_isolate_ack_i && a_isolate_ack_sync) begin
                        b_state <= S_CLEAR;
                    end
                end
                S_CLEAR: begin
                    b_isolate_o_r <= 1'b1;
                    b_clear_o_r   <= 1'b1;
                    // Wait for both sides to acknowledge clear
                    if (b_clear_ack_i && a_clear_ack_sync) begin
                        b_state <= S_DEISO;
                    end
                end
                S_DEISO: begin
                    b_isolate_o_r <= 1'b0;
                    b_clear_o_r   <= 1'b0;
                    // Wait for both sides to acknowledge de-isolate (back to idle)
                    if (b_isolate_ack_i && a_isolate_ack_sync) begin
                        b_state <= S_IDLE;
                    end
                end
                default: begin
                    b_state       <= S_IDLE;
                    b_isolate_o_r <= 1'b0;
                    b_clear_o_r   <= 1'b0;
                end
            endcase
        end
    end

    assign a_isolate_o = a_isolate_o_r;
    assign a_clear_o   = a_clear_o_r;
    assign b_isolate_o = b_isolate_o_r;
    assign b_clear_o   = b_clear_o_r;

endmodule
