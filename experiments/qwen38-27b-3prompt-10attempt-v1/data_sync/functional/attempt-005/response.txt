module data_sync #(
    parameter STAGES = 2,
    parameter DWIDTH = 8
)(
    input                  clk,
    input                  rstn,
    input  [DWIDTH-1:0]    din,
    input                  dready_i,
    output reg [DWIDTH-1:0] dout,
    output reg             dready_o
);

    // Choose a safe crossing depth: at least 2 stages for metastability
    localparam SYNC_STAGES = (STAGES < 2) ? 2 : STAGES;

    // Synchronizer flip-flops
    reg [SYNC_STAGES-1:0] sync_ff;
    reg [DWIDTH-1:0]      data_ff;

    // Pipeline registers for data and ready
    reg [DWIDTH-1:0]      data_pipe [0:SYNC_STAGES-1];
    reg                   ready_pipe [0:SYNC_STAGES-1];

    integer i;

    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            sync_ff    <= {SYNC_STAGES{1'b0}};
            data_ff    <= {DWIDTH{1'b0}};
            dout       <= {DWIDTH{1'b0}};
            dready_o   <= 1'b0;
            for (i = 0; i < SYNC_STAGES; i = i + 1) begin
                data_pipe[i]  <= {DWIDTH{1'b0}};
                ready_pipe[i] <= 1'b0;
            end
        end else begin
            // Synchronize the ready signal
            sync_ff[0] <= dready_i;
            for (i = 1; i < SYNC_STAGES; i = i + 1) begin
                sync_ff[i] <= sync_ff[i-1];
            end

            // Pipeline data and ready through the stages
            // Stage 0: capture data when dready_i is high
            if (dready_i) begin
                data_pipe[0]  <= din;
                ready_pipe[0] <= 1'b1;
            end else begin
                data_pipe[0]  <= data_pipe[0];
                ready_pipe[0] <= 1'b0;
            end

            // Subsequent stages
            for (i = 1; i < SYNC_STAGES; i = i + 1) begin
                data_pipe[i]  <= data_pipe[i-1];
                ready_pipe[i] <= ready_pipe[i-1];
            end

            // Output from the last stage
            dout     <= data_pipe[SYNC_STAGES-1];
            dready_o <= ready_pipe[SYNC_STAGES-1];
        end
    end

endmodule
