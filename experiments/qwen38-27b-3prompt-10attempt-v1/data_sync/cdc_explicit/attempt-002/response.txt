module data_sync #(
    parameter STAGES = 2,
    parameter DWIDTH = 8
)(
    input                  clk,
    input                  rstn,
    input  [DWIDTH-1:0]    din,
    input                  dready_i,
    output reg [DWIDTH-1:0] dout,
    output reg             dready_o
);

    // Use a safe minimum of 2 stages for synchronization
    localparam SYNC_STAGES = (STAGES < 2) ? 2 : STAGES;

    // Synchronization registers for the ready signal
    reg [SYNC_STAGES-1:0] dready_sync;

    // Synchronize dready_i into the local clock domain
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dready_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            dready_sync[0] <= dready_i;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                dready_sync[i] <= dready_sync[i-1];
            end
        end
    end

    // The synchronized ready signal
    wire dready_synced = dready_sync[SYNC_STAGES-1];

    // Data register to hold the coherent multi-bit data
    reg [DWIDTH-1:0] data_reg;

    // Capture data when synchronized ready is high
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            data_reg <= {DWIDTH{1'b0}};
        end else if (dready_synced) begin
            data_reg <= din;
        end
    end

    // Output assignments
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dout <= {DWIDTH{1'b0}};
            dready_o <= 1'b0;
        end else begin
            dout <= data_reg;
            dready_o <= dready_synced;
        end
    end

endmodule
