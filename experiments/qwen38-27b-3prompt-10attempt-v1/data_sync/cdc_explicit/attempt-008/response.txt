module data_sync #(
    parameter STAGES = 2,
    parameter DWIDTH = 8
)(
    input                  clk,
    input                  rstn,
    input  [DWIDTH-1:0]    din,
    input                  dready_i,
    output reg [DWIDTH-1:0] dout,
    output reg             dready_o
);

    // Ensure STAGES is at least 2 for safe CDC
    localparam SAFE_STAGES = (STAGES < 2) ? 2 : STAGES;

    // Synchronizer registers for the handshake signal
    reg [SAFE_STAGES-1:0] dready_sync;

    // Synchronize dready_i through SAFE_STAGES flip-flops
    integer i;
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dready_sync <= {SAFE_STAGES{1'b0}};
        end else begin
            dready_sync[0] <= dready_i;
            for (i = 1; i < SAFE_STAGES; i = i + 1) begin
                dready_sync[i] <= dready_sync[i-1];
            end
        end
    end

    // The synchronized ready signal
    wire dready_synced = dready_sync[SAFE_STAGES-1];

    // Data register and output ready pulse
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dout     <= {DWIDTH{1'b0}};
            dready_o <= 1'b0;
        end else begin
            dready_o <= 1'b0; // Default: no pulse
            if (dready_synced) begin
                dout     <= din;
                dready_o <= 1'b1;
            end
        end
    end

endmodule
