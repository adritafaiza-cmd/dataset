module data_sync #(
    parameter STAGES = 2,
    parameter DWIDTH = 8
)(
    input                  clk,
    input                  rstn,
    input  [DWIDTH-1:0]    din,
    input                  dready_i,
    output reg [DWIDTH-1:0] dout,
    output reg             dready_o
);

    // Ensure minimum 2 stages for safe multi-bit synchronization
    localparam SYNC_STAGES = (STAGES < 2) ? 2 : STAGES;

    // Synchronizer chain for the handshake signal
    reg [SYNC_STAGES-1:0] sync_chain;

    // Latched data and valid flag
    reg [DWIDTH-1:0] data_latch;
    reg              valid_latch;

    // Internal ready signal (synchronized version of dready_i)
    wire sync_ready = sync_chain[SYNC_STAGES-1];

    // Detect rising edge of synchronized ready
    reg sync_ready_d;
    wire sync_ready_rise = sync_ready & ~sync_ready_d;

    // Synchronize the valid flag and data using a handshake approach
    // Since we have a single clock, we can use a simple synchronizer for the control signal
    // and latch the data when the control signal transitions.

    // Synchronize dready_i
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            sync_chain <= {SYNC_STAGES{1'b0}};
        end else begin
            sync_chain[0] <= dready_i;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                sync_chain[i] <= sync_chain[i-1];
            end
        end
    end

    // Track previous synchronized ready
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            sync_ready_d <= 1'b0;
        end else begin
            sync_ready_d <= sync_ready;
        end
    end

    // Latch data and set valid when synchronized ready rises
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            data_latch  <= {DWIDTH{1'b0}};
            valid_latch <= 1'b0;
            dout        <= {DWIDTH{1'b0}};
            dready_o    <= 1'b0;
        end else begin
            if (sync_ready_rise) begin
                data_latch  <= din;
                valid_latch <= 1'b1;
            end
            if (valid_latch) begin
                dout     <= data_latch;
                dready_o <= 1'b1;
                valid_latch <= 1'b0;
            end else begin
                dready_o <= 1'b0;
            end
        end
    end

endmodule
