module data_sync #(
    parameter STAGES = 2,
    parameter DWIDTH = 8
)(
    input                  clk,
    input                  rstn,
    input  [DWIDTH-1:0]    din,
    input                  dready_i,
    output reg [DWIDTH-1:0] dout,
    output reg             dready_o
);

    localparam SYNC_STAGES = (STAGES < 2) ? 2 : STAGES;

    reg [SYNC_STAGES-1:0] dready_sync;
    reg [SYNC_STAGES-1:0] din_sync [0:DWIDTH-1];

    integer i;

    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dready_sync <= {SYNC_STAGES{1'b0}};
            for (i = 0; i < DWIDTH; i = i + 1)
                din_sync[i] <= {SYNC_STAGES{1'b0}};
        end else begin
            dready_sync <= {dready_sync[SYNC_STAGES-2:0], dready_i};
            for (i = 0; i < DWIDTH; i = i + 1)
                din_sync[i] <= {din_sync[i][SYNC_STAGES-2:0], din[i]};
        end
    end

    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dout <= {DWIDTH{1'b0}};
            dready_o <= 1'b0;
        end else begin
            dout <= din_sync[SYNC_STAGES-1];
            dready_o <= dready_sync[SYNC_STAGES-1];
        end
    end

endmodule
