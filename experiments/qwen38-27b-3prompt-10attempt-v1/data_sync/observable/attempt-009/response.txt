module data_sync #(
    parameter STAGES = 2,
    parameter DWIDTH = 8
)(
    input                  clk,
    input                  rstn,
    input  [DWIDTH-1:0]    din,
    input                  dready_i,
    output reg [DWIDTH-1:0] dout,
    output reg             dready_o
);

    // Synchronize din
    reg [DWIDTH-1:0] din_sync [0:STAGES-1];

    // Synchronize dready_i
    reg dready_i_sync [0:STAGES-1];

    // Track previous synchronized dready_i to detect rising edge
    reg dready_i_prev;

    // Internal ready flag
    reg internal_ready;

    // Synchronize din through STAGES flip-flops
    integer i;
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            for (i = 0; i < STAGES; i = i + 1) begin
                din_sync[i] <= {DWIDTH{1'b0}};
            end
        end else begin
            din_sync[0] <= din;
            for (i = 1; i < STAGES; i = i + 1) begin
                din_sync[i] <= din_sync[i-1];
            end
        end
    end

    // Synchronize dready_i through STAGES flip-flops
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            for (i = 0; i < STAGES; i = i + 1) begin
                dready_i_sync[i] <= 1'b0;
            end
        end else begin
            dready_i_sync[0] <= dready_i;
            for (i = 1; i < STAGES; i = i + 1) begin
                dready_i_sync[i] <= dready_i_sync[i-1];
            end
        end
    end

    // Detect rising edge of synchronized dready_i
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dready_i_prev <= 1'b0;
        end else begin
            dready_i_prev <= dready_i_sync[STAGES-1];
        end
    end

    wire dready_i_rise = dready_i_sync[STAGES-1] & ~dready_i_prev;

    // Internal ready: set when dready_i rises, cleared when we output
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            internal_ready <= 1'b0;
        end else begin
            if (dready_i_rise) begin
                internal_ready <= 1'b1;
            end else if (internal_ready) begin
                internal_ready <= 1'b0;
            end
        end
    end

    // Output logic
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dout     <= {DWIDTH{1'b0}};
            dready_o <= 1'b0;
        end else begin
            if (internal_ready) begin
                dout     <= din_sync[STAGES-1];
                dready_o <= 1'b1;
            end else begin
                dready_o <= 1'b0;
            end
        end
    end

endmodule
