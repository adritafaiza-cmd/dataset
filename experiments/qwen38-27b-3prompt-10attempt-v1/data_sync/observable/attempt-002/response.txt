module data_sync #(
    parameter STAGES = 2,
    parameter DWIDTH = 8
)(
    input                  clk,
    input                  rstn,
    input  [DWIDTH-1:0]    din,
    input                  dready_i,
    output reg [DWIDTH-1:0] dout,
    output reg             dready_o
);

    localparam SYNC_STAGES = (STAGES > 1) ? STAGES : 2;

    reg [SYNC_STAGES-1:0] dready_sync;
    reg [SYNC_STAGES-1:0] dready_sync_next;
    reg [DWIDTH-1:0]      din_sync;
    reg [DWIDTH-1:0]      din_sync_next;

    // Synchronize dready_i through STAGES flip-flops
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dready_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            dready_sync <= dready_sync_next;
        end
    end

    always @* begin
        dready_sync_next[0] = dready_i;
        for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
            dready_sync_next[i] = dready_sync[i-1];
        end
    end

    // Synchronize din through STAGES flip-flops
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            din_sync <= {DWIDTH{1'b0}};
        end else begin
            din_sync <= din_sync_next;
        end
    end

    always @* begin
        din_sync_next = din;
    end

    // Output: use the last stage of synchronization
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dout     <= {DWIDTH{1'b0}};
            dready_o <= 1'b0;
        end else begin
            dout     <= din_sync[SYNC_STAGES-1];
            dready_o <= dready_sync[SYNC_STAGES-1];
        end
    end

endmodule
