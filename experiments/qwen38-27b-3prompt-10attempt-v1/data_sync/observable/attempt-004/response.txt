module data_sync #(
    parameter STAGES = 2,
    parameter DWIDTH = 8
)(
    input                  clk,
    input                  rstn,
    input  [DWIDTH-1:0]    din,
    input                  dready_i,
    output reg [DWIDTH-1:0] dout,
    output reg             dready_o
);

    // Synchronize din
    reg [DWIDTH-1:0] din_sync [0:STAGES];
    // Synchronize dready_i
    reg dready_i_sync [0:STAGES];

    // Pipeline registers for data
    reg [DWIDTH-1:0] data_pipe [0:STAGES];
    // Pipeline registers for ready flag
    reg ready_pipe [0:STAGES];

    // Synchronization of din
    integer i;
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            for (i = 0; i < STAGES; i = i + 1) begin
                din_sync[i] <= {DWIDTH{1'b0}};
            end
        end else begin
            din_sync[0] <= din;
            for (i = 1; i < STAGES; i = i + 1) begin
                din_sync[i] <= din_sync[i-1];
            end
        end
    end

    // Synchronization of dready_i
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            for (i = 0; i < STAGES; i = i + 1) begin
                dready_i_sync[i] <= 1'b0;
            end
        end else begin
            dready_i_sync[0] <= dready_i;
            for (i = 1; i < STAGES; i = i + 1) begin
                dready_i_sync[i] <= dready_i_sync[i-1];
            end
        end
    end

    // Pipeline data and ready through STAGES stages
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            for (i = 0; i < STAGES; i = i + 1) begin
                data_pipe[i] <= {DWIDTH{1'b0}};
                ready_pipe[i] <= 1'b0;
            end
            dout <= {DWIDTH{1'b0}};
            dready_o <= 1'b0;
        end else begin
            // Stage 0: capture synchronized values
            data_pipe[0] <= din_sync[STAGES-1];
            ready_pipe[0] <= dready_i_sync[STAGES-1];

            // Propagate through pipeline
            for (i = 1; i < STAGES; i = i + 1) begin
                data_pipe[i] <= data_pipe[i-1];
                ready_pipe[i] <= ready_pipe[i-1];
            end

            // Output from last stage
            dout <= data_pipe[STAGES-1];
            dready_o <= ready_pipe[STAGES-1];
        end
    end

endmodule
