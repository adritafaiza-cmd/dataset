module apbslave #(
		parameter	C_APB_ADDR_WIDTH = 12,
		parameter	C_APB_DATA_WIDTH = 32,
		localparam	AW = C_APB_ADDR_WIDTH,
		localparam	DW = C_APB_DATA_WIDTH,
		localparam	APBLSB = $clog2(C_APB_DATA_WIDTH)-3
	) (
		input	wire			PCLK, PRESETn,
		input	wire			PSEL,
		input	wire			PENABLE,
		output	reg			PREADY,
		input	wire	[AW-1:0]	PADDR,
		input	wire			PWRITE,
		input	wire	[DW-1:0]	PWDATA,
		input	wire	[DW/8-1:0]	PWSTRB,
		input	wire	[2:0]		PPROT,
		output	reg	[DW-1:0]	PRDATA,
		output	wire			PSLVERR
	);

	// Internal register file
	reg [DW-1:0] mem [0:2**AW-1];

	// State machine
	localparam S_IDLE = 1'b0;
	localparam S_SETUP = 1'b1;
	localparam S_ACCESS = 1'b1;

	reg state;

	// PSLVERR is always low for in-range accesses
	assign PSLVERR = 1'b0;

	// PREADY logic
	always @(posedge PCLK or negedge PRESETn) begin
		if (!PRESETn) begin
			PREADY <= 1'b0;
			state <= S_IDLE;
		end else begin
			case (state)
				S_IDLE: begin
					PREADY <= 1'b0;
					if (PSEL) begin
						state <= S_SETUP;
					end
				end
				S_SETUP: begin
					PREADY <= 1'b0;
					if (PENABLE) begin
						state <= S_ACCESS;
					end
				end
				S_ACCESS: begin
					PREADY <= 1'b1;
					state <= S_IDLE;
				end
				default: begin
					PREADY <= 1'b0;
					state <= S_IDLE;
				end
			endcase
		end
	end

	// Read data output
	always @(posedge PCLK or negedge PRESETn) begin
		if (!PRESETn) begin
			PRDATA <= {DW{1'b0}};
		end else begin
			if (state == S_SETUP && !PWRITE) begin
				PRDATA <= mem[PADDR[AW-1:APBLSB]];
			end
		end
	end

	// Write data input
	always @(posedge PCLK or negedge PRESETn) begin
		if (!PRESETn) begin
			// No action needed for reset on memory
		end else begin
			if (state == S_ACCESS && PWRITE) begin
				// Byte-wise write using PWSTRB
				for (genvar i = 0; i < DW/8; i = i+1) begin : write_loop
					if (PWSTRB[i]) begin
						mem[PADDR[AW-1:APBLSB]][i*8 +: 8] <= PWDATA[i*8 +: 8];
					end
				end
			end
		end
	end

endmodule
