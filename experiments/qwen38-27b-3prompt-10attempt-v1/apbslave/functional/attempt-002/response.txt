module apbslave #(
		parameter	C_APB_ADDR_WIDTH = 12,
		parameter	C_APB_DATA_WIDTH = 32,
		localparam	AW = C_APB_ADDR_WIDTH,
		localparam	DW = C_APB_DATA_WIDTH,
		localparam	APBLSB = $clog2(C_APB_DATA_WIDTH)-3
	) (
		input	wire			PCLK, PRESETn,
		input	wire			PSEL,
		input	wire			PENABLE,
		output	reg			PREADY,
		input	wire	[AW-1:0]	PADDR,
		input	wire			PWRITE,
		input	wire	[DW-1:0]	PWDATA,
		input	wire	[DW/8-1:0]	PWSTRB,
		input	wire	[2:0]		PPROT,
		output	reg	[DW-1:0]	PRDATA,
		output	wire			PSLVERR
	);

	// Internal state
	reg [DW-1:0] mem [0:2**AW-1];
	reg [DW-1:0] rdata_reg;
	reg           rvalid;

	// PSLVERR is always low for in-range accesses
	assign PSLVERR = 1'b0;

	// PREADY logic: ready during setup phase (PSEL=1, PENABLE=0)
	// and during access phase (PSEL=1, PENABLE=1)
	// Standard APB: PREADY can be asserted in setup or access phase
	// For a simple slave, we assert PREADY when PSEL is high
	// Actually, PREADY should be asserted to complete the transfer.
	// In setup phase, if PREADY is high, the transfer completes in one cycle.
	// If PREADY is low, it waits until access phase.
	// For simplicity, we assert PREADY whenever PSEL is high.
	// But we need to be careful: PREADY should be stable during the access phase.
	// Let's use a simple approach: PREADY is high when PSEL is high.
	// This means the transfer completes in the setup phase if PREADY is high.
	
	// Actually, the standard behavior is:
	// - During setup phase (PSEL=1, PENABLE=0): PREADY can be high or low.
	//   If PREADY is high, the transfer completes in this cycle.
	//   If PREADY is low, the master asserts PENABLE in the next cycle.
	// - During access phase (PSEL=1, PENABLE=1): PREADY must be high to complete.
	
	// For a simple slave that always completes in one cycle, we can assert PREADY
	// whenever PSEL is high. This means the transfer completes in the setup phase.
	
	// Let's implement a simple memory-based slave.
	// We'll use a register file indexed by the address.
	
	// Address decoding: use the full address as index
	// For simplicity, we'll use the address directly as the memory index.
	// The address width AW determines the number of memory locations.
	
	// Write logic: on PSEL & PENABLE & PWRITE, write to memory
	// Read logic: on PSEL & PENABLE & ~PWRITE, read from memory
	
	// We need to handle the timing correctly.
	// In APB, the data is valid during the access phase.
	// For writes, PWDATA is valid during the access phase.
	// For reads, PRDATA must be valid during the access phase.
	
	// Let's use a simple approach:
	// - During setup phase, we can prepare the read data.
	// - During access phase, we present the read data.
	
	// Actually, for a simple slave, we can just read the memory in the setup phase
	// and present it in the access phase. Or we can read it in the access phase.
	
	// Let's use a two-phase approach:
	// - In setup phase (PSEL=1, PENABLE=0): if it's a read, read the memory and store in rdata_reg.
	// - In access phase (PSEL=1, PENABLE=1): present rdata_reg on PRDATA.
	
	// But if PREADY is high in the setup phase, the transfer completes in the setup phase,
	// and there is no access phase. So we need to handle both cases.
	
	// For simplicity, let's always complete the transfer in the access phase.
	// This means PREADY is low in the setup phase and high in the access phase.
	
	// Let's implement it this way:
	// - PREADY is low in the setup phase (PSEL=1, PENABLE=0).
	// - PREADY is high in the access phase (PSEL=1, PENABLE=1).
	
	// This ensures that the transfer always takes two cycles.
	
	// Let's implement the logic:
	
	always @(posedge PCLK or negedge PRESETn) begin
		if (!PRESETn) begin
			PREADY <= 1'b0;
			PRDATA <= {DW{1'b0}};
			rdata_reg <= {DW{1'b0}};
			rvalid <= 1'b0;
		end else begin
			// Default: PREADY is low
			PREADY <= 1'b0;
			
			if (PSEL) begin
				if (!PENABLE) begin
					// Setup phase
					// Prepare read data if it's a read
					if (!PWRITE) begin
						rdata_reg <= mem[PADDR[AW-1:APBLSB]];
						rvalid <= 1'b1;
					end
				end else begin
					// Access phase
					PREADY <= 1'b1;
					
					if (PWRITE) begin
						// Write to memory
						// Use PWSTRB to select bytes
						// For simplicity, we'll write the full word if any strobe is set
						// A more complete implementation would use PWSTRB to mask the write
						// For now, let's just write the full word
						mem[PADDR[AW-1:APBLSB]] <= PWDATA;
					end else begin
						// Read: present the data
						PRDATA <= rdata_reg;
					end
				end
			end
		end
	end

endmodule
