module apbslave #(
		parameter	C_APB_ADDR_WIDTH = 12,
		parameter	C_APB_DATA_WIDTH = 32,
		localparam	AW = C_APB_ADDR_WIDTH,
		localparam	DW = C_APB_DATA_WIDTH,
		localparam	APBLSB = $clog2(C_APB_DATA_WIDTH)-3
	) (
		input	wire			PCLK, PRESETn,
		input	wire			PSEL,
		input	wire			PENABLE,
		output	reg			PREADY,
		input	wire	[AW-1:0]	PADDR,
		input	wire			PWRITE,
		input	wire	[DW-1:0]	PWDATA,
		input	wire	[DW/8-1:0]	PWSTRB,
		input	wire	[2:0]		PPROT,
		output	reg	[DW-1:0]	PRDATA,
		output	wire			PSLVERR
	);

	// Internal register file
	reg [DW-1:0] mem [0:2**AW-1];

	// APB state machine
	localparam S_IDLE = 1'b0;
	localparam S_SETUP = 1'b1;
	localparam S_ACCESS = 1'b1; // Simplified: use PENABLE to distinguish

	// We'll use a simple approach:
	// PREADY is high when not in access phase
	// During setup phase (PSEL=1, PENABLE=0), PREADY can be high or low
	// During access phase (PSEL=1, PENABLE=1), PREADY indicates completion

	// For a simple slave, we can complete in one cycle:
	// PREADY is always high when PSEL is high, meaning the slave is ready
	// This is the simplest valid APB slave behavior

	// Let's implement a proper APB slave with a register file
	// The address is used to index into the register file
	// We'll use the upper bits of the address as the register index

	// Calculate the register index from the address
	wire [AW-APBLSB-1:0] reg_index = PADDR[AW-1:APBLSB];

	// Write enable: during access phase, write operation
	wire write_en = PSEL & PENABLE & PWRITE;

	// Read data: from the register file
	assign PRDATA = mem[reg_index];

	// PREADY: high when the slave is ready to complete the transfer
	// For a simple slave, we can assert PREADY in the access phase
	// This means the transfer completes in one cycle
	assign PREADY = PSEL;

	// PSLVERR: always low for in-range accesses
	assign PSLVERR = 1'b0;

	// Write to register file
	always @(posedge PCLK or negedge PRESETn) begin
		if (!PRESETn) begin
			// Reset: clear all registers (optional, but good practice)
			// For simplicity, we'll just let them be uninitialized
			// In a real design, you might want to initialize them
		end else if (write_en) begin
			// Write with byte strobes
			integer i;
			for (i = 0; i < DW/8; i = i + 1) begin
				if (PWSTRB[i]) begin
					mem[reg_index][i*8 +: 8] <= PWDATA[i*8 +: 8];
				end
			end
		end
	end

	// Initialize PREADY and PRDATA
	always @(posedge PCLK or negedge PRESETn) begin
		if (!PRESETn) begin
			PREADY <= 1'b0;
			PRDATA <= {DW{1'b0}};
		end else begin
			PREADY <= PSEL;
			PRDATA <= mem[reg_index];
		end
	end

endmodule
