module apbslave #(
		parameter	C_APB_ADDR_WIDTH = 12,
		parameter	C_APB_DATA_WIDTH = 32,
		localparam	AW = C_APB_ADDR_WIDTH,
		localparam	DW = C_APB_DATA_WIDTH,
		localparam	APBLSB = $clog2(C_APB_DATA_WIDTH)-3
	) (
		input	wire			PCLK, PRESETn,
		input	wire			PSEL,
		input	wire			PENABLE,
		output	reg			PREADY,
		input	wire	[AW-1:0]	PADDR,
		input	wire			PWRITE,
		input	wire	[DW-1:0]	PWDATA,
		input	wire	[DW/8-1:0]	PWSTRB,
		input	wire	[2:0]		PPROT,
		output	reg	[DW-1:0]	PRDATA,
		output	wire			PSLVERR
	);

	// Internal register file: 256 bytes (64 words)
	reg [DW-1:0] mem [0:63];

	// State machine
	localparam S_IDLE = 1'b0;
	localparam S_SETUP = 1'b1;

	reg state;

	// PSLVERR is always 0 for this simple slave
	assign PSLVERR = 1'b0;

	// PREADY is always 1 (no wait states)
	always @(*) begin
		PREADY = 1'b1;
	end

	// State machine
	always @(posedge PCLK or negedge PRESETn) begin
		if (!PRESETn) begin
			state <= S_IDLE;
		end else begin
			case (state)
				S_IDLE: begin
					if (PSEL)
						state <= S_SETUP;
					else
						state <= S_IDLE;
				end
				S_SETUP: begin
					if (PSEL && PENABLE)
						state <= S_IDLE;
					else if (!PSEL)
						state <= S_IDLE;
					else
						state <= S_SETUP;
				end
				default: state <= S_IDLE;
			endcase
		end
	end

	// Read data output
	always @(posedge PCLK or negedge PRESETn) begin
		if (!PRESETn) begin
			PRDATA <= {DW{1'b0}};
		end else begin
			if (state == S_SETUP && PSEL && !PWRITE) begin
				PRDATA <= mem[PADDR[AW-1:APBLSB]];
			end
		end
	end

	// Write data input
	always @(posedge PCLK or negedge PRESETn) begin
		if (!PRESETn) begin
			// No reset action needed for memory
		end else begin
			if (state == S_SETUP && PSEL && PWRITE) begin
				for (integer i = 0; i < DW/8; i = i + 1) begin
					if (PWSTRB[i]) begin
						mem[PADDR[AW-1:APBLSB]][i*8 +: 8] <= PWDATA[i*8 +: 8];
					end
				end
			end
		end
	end

endmodule
