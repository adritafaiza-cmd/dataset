module sync #(
    parameter int unsigned STAGES = 2,
    parameter bit ResetValue = 1'b0
) (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic serial_i,
    output logic serial_o
);

    localparam int unsigned MIN_STAGES = 2;
    localparam int unsigned EFFECTIVE_STAGES = (STAGES < MIN_STAGES) ? MIN_STAGES : STAGES;

    logic [EFFECTIVE_STAGES-1:0] sync_ff;

    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            sync_ff <= {EFFECTIVE_STAGES{ResetValue}};
        end else begin
            sync_ff[0] <= serial_i;
            for (int unsigned i = 1; i < EFFECTIVE_STAGES; i++) begin
                sync_ff[i] <= sync_ff[i-1];
            end
        end
    end

    assign serial_o = sync_ff[EFFECTIVE_STAGES-1];

endmodule
