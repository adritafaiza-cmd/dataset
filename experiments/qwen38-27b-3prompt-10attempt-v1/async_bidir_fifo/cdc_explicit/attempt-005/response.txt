module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;
  localparam PTR_W = ASIZE + 1;
  localparam AFULL_THRESH  = DEPTH - 2;
  localparam AEMPTY_THRESH = 1;

  // Internal FIFO storage
  reg [DSIZE-1:0] mem [0:DEPTH-1];

  // Side A pointers (in A clock domain)
  reg [PTR_W-1:0] a_wr_ptr;
  reg [PTR_W-1:0] a_rd_ptr;

  // Side B pointers (in B clock domain)
  reg [PTR_W-1:0] b_wr_ptr;
  reg [PTR_W-1:0] b_rd_ptr;

  // Synchronized pointers
  reg [PTR_W-1:0] a_wr_ptr_sync1, a_wr_ptr_sync2;
  reg [PTR_W-1:0] a_rd_ptr_sync1, a_rd_ptr_sync2;
  reg [PTR_W-1:0] b_wr_ptr_sync1, b_wr_ptr_sync2;
  reg [PTR_W-1:0] b_rd_ptr_sync1, b_rd_ptr_sync2;

  // Synchronized direction signals
  reg a_dir_sync1, a_dir_sync2;
  reg b_dir_sync1, b_dir_sync2;

  // Synchronized write enable signals
  reg a_winc_sync1, a_winc_sync2;
  reg b_winc_sync1, b_winc_sync2;

  // Synchronized read enable signals
  reg a_rinc_sync1, a_rinc_sync2;
  reg b_rinc_sync1, b_rinc_sync2;

  // Synchronized write data
  reg [DSIZE-1:0] a_wdata_sync1, a_wdata_sync2;
  reg [DSIZE-1:0] b_wdata_sync1, b_wdata_sync2;

  // Effective write/read enables
  wire a_write_en = a_dir & a_winc;
  wire a_read_en  = ~a_dir & a_rinc;
  wire b_write_en = b_dir & b_winc;
  wire b_read_en  = ~b_dir & b_rinc;

  // Full/empty calculations
  wire [PTR_W-1:0] a_wr_ptr_sync = a_wr_ptr_sync2;
  wire [PTR_W-1:0] a_rd_ptr_sync = a_rd_ptr_sync2;
  wire [PTR_W-1:0] b_wr_ptr_sync = b_wr_ptr_sync2;
  wire [PTR_W-1:0] b_rd_ptr_sync = b_rd_ptr_sync2;

  // A side: full when write pointer has lapped read pointer by DEPTH
  wire a_full  = (a_wr_ptr[ASIZE-1:0] == a_rd_ptr_sync[ASIZE-1:0]) &&
                 (a_wr_ptr[ASIZE] != a_rd_ptr_sync[ASIZE]);
  wire a_afull = (a_wr_ptr[ASIZE-1:0] == a_rd_ptr_sync[ASIZE-1:0]) &&
                 (a_wr_ptr[ASIZE] != a_rd_ptr_sync[ASIZE]) ||
                 ((a_wr_ptr[ASIZE-1:0] + 1) == a_rd_ptr_sync[ASIZE-1:0]) &&
                 (a_wr_ptr[ASIZE] != a_rd_ptr_sync[ASIZE]);

  // A side: empty when write pointer equals read pointer
  wire a_empty  = (a_wr_ptr_sync[ASIZE-1:0] == a_rd_ptr[ASIZE-1:0]) &&
                  (a_wr_ptr_sync[ASIZE] == a_rd_ptr[ASIZE]);
  wire a_aempty = (a_wr_ptr_sync[ASIZE-1:0] == a_rd_ptr[ASIZE-1:0]) &&
                  (a_wr_ptr_sync[ASIZE] == a_rd_ptr[ASIZE]) ||
                  ((a_wr_ptr_sync[ASIZE-1:0] + 1) == a_rd_ptr[ASIZE-1:0]) &&
                  (a_wr_ptr_sync[ASIZE] == a_rd_ptr[ASIZE]);

  // B side: full when write pointer has lapped read pointer by DEPTH
  wire b_full  = (b_wr_ptr[ASIZE-1:0] == b_rd_ptr_sync[ASIZE-1:0]) &&
                 (b_wr_ptr[ASIZE] != b_rd_ptr_sync[ASIZE]);
  wire b_afull = (b_wr_ptr[ASIZE-1:0] == b_rd_ptr_sync[ASIZE-1:0]) &&
                 (b_wr_ptr[ASIZE] != b_rd_ptr_sync[ASIZE]) ||
                 ((b_wr_ptr[ASIZE-1:0] + 1) == b_rd_ptr_sync[ASIZE-1:0]) &&
                 (b_wr_ptr[ASIZE] != b_rd_ptr_sync[ASIZE]);

  // B side: empty when write pointer equals read pointer
  wire b_empty  = (b_wr_ptr_sync[ASIZE-1:0] == b_rd_ptr[ASIZE-1:0]) &&
                  (b_wr_ptr_sync[ASIZE] == b_rd_ptr[ASIZE]);
  wire b_aempty = (b_wr_ptr_sync[ASIZE-1:0] == b_rd_ptr[ASIZE-1:0]) &&
                  (b_wr_ptr_sync[ASIZE] == b_rd_ptr[ASIZE]) ||
                  ((b_wr_ptr_sync[ASIZE-1:0] + 1) == b_rd_ptr[ASIZE-1:0]) &&
                  (b_wr_ptr_sync[ASIZE] == b_rd_ptr[ASIZE]);

  // Read data
  wire [DSIZE-1:0] a_rdata_internal = mem[a_rd_ptr[ASIZE-1:0]];
  wire [DSIZE-1:0] b_rdata_internal = mem[b_rd_ptr[ASIZE-1:0]];

  assign a_rdata = a_rdata_internal;
  assign b_rdata = b_rdata_internal;

  assign a_full   = a_full;
  assign a_afull  = a_afull;
  assign a_empty  = a_empty;
  assign a_aempty = a_aempty;
  assign b_full   = b_full;
  assign b_afull  = b_afull;
  assign b_empty  = b_empty;
  assign b_aempty = b_aempty;

  // A clock domain: update pointers
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= {PTR_W{1'b0}};
      a_rd_ptr <= {PTR_W{1'b0}};
    end else begin
      if (a_write_en && !a_full)
        a_wr_ptr <= a_wr_ptr + 1'b1;
      if (a_read_en && !a_empty)
        a_rd_ptr <= a_rd_ptr + 1'b1;
    end
  end

  // B clock domain: update pointers
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= {PTR_W{1'b0}};
      b_rd_ptr <= {PTR_W{1'b0}};
    end else begin
      if (b_write_en && !b_full)
        b_wr_ptr <= b_wr_ptr + 1'b1;
      if (b_read_en && !b_empty)
        b_rd_ptr <= b_rd_ptr + 1'b1;
    end
  end

  // Memory writes
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      // No action needed for memory
    end else if (a_write_en && !a_full) begin
      mem[a_wr_ptr[ASIZE-1:0]] <= a_wdata;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      // No action needed for memory
    end else if (b_write_en && !b_full) begin
      mem[b_wr_ptr[ASIZE-1:0]] <= b_wdata;
    end
  end

  // Synchronize A write pointer to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_wr_ptr_sync1 <= {PTR_W{1'b0}};
      a_wr_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      a_wr_ptr_sync1 <= a_wr_ptr;
      a_wr_ptr_sync2 <= a_wr_ptr_sync1;
    end
  end

  // Synchronize A read pointer to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_rd_ptr_sync1 <= {PTR_W{1'b0}};
      a_rd_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      a_rd_ptr_sync1 <= a_rd_ptr;
      a_rd_ptr_sync2 <= a_rd_ptr_sync1;
    end
  end

  // Synchronize B write pointer to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_wr_ptr_sync1 <= {PTR_W{1'b0}};
      b_wr_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      b_wr_ptr_sync1 <= b_wr_ptr;
      b_wr_ptr_sync2 <= b_wr_ptr_sync1;
    end
  end

  // Synchronize B read pointer to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_rd_ptr_sync1 <= {PTR_W{1'b0}};
      b_rd_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      b_rd_ptr_sync1 <= b_rd_ptr;
      b_rd_ptr_sync2 <= b_rd_ptr_sync1;
    end
  end

endmodule
