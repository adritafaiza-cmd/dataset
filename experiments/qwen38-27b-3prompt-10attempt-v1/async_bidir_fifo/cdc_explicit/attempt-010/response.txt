```verilog
module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;
  localparam PTR_W = ASIZE + 1;
  localparam AFULL_THRESH  = DEPTH - 2;
  localparam AEMPTY_THRESH = 1;

  // Internal FIFO storage
  reg [DSIZE-1:0] mem [0:DEPTH-1];

  // Side A pointers (in A clock domain)
  reg [PTR_W-1:0] a_wr_ptr;
  reg [PTR_W-1:0] a_rd_ptr;

  // Side B pointers (in B clock domain)
  reg [PTR_W-1:0] b_wr_ptr;
  reg [PTR_W-1:0] b_rd_ptr;

  // CDC synchronizers for write pointers
  // a_wr_ptr -> B domain
  reg [PTR_W-1:0] a_wr_ptr_sync1, a_wr_ptr_sync2;
  // b_wr_ptr -> A domain
  reg [PTR_W-1:0] b_wr_ptr_sync1, b_wr_ptr_sync2;

  // CDC synchronizers for read pointers
  // a_rd_ptr -> B domain
  reg [PTR_W-1:0] a_rd_ptr_sync1, a_rd_ptr_sync2;
  // b_rd_ptr -> A domain
  reg [PTR_W-1:0] b_rd_ptr_sync1, b_rd_ptr_sync2;

  // Synchronizer logic
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr_sync1 <= {PTR_W{1'b0}};
      a_wr_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      a_wr_ptr_sync1 <= a_wr_ptr;
      a_wr_ptr_sync2 <= a_wr_ptr_sync1;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr_sync1 <= {PTR_W{1'b0}};
      b_wr_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      b_wr_ptr_sync1 <= b_wr_ptr;
      b_wr_ptr_sync2 <= b_wr_ptr_sync1;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_rd_ptr_sync1 <= {PTR_W{1'b0}};
      a_rd_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      a_rd_ptr_sync1 <= a_rd_ptr;
      a_rd_ptr_sync2 <= a_rd_ptr_sync1;
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_rd_ptr_sync1 <= {PTR_W{1'b0}};
      b_rd_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      b_rd_ptr_sync1 <= b_rd_ptr;
      b_rd_ptr_sync2 <= b_rd_ptr_sync1;
    end
  end

  // Synchronized pointer values
  wire [PTR_W-1:0] a_wr_ptr_b = a_wr_ptr_sync2;
  wire [PTR_W-1:0] b_wr_ptr_a = b_wr_ptr_sync2;
  wire [PTR_W-1:0] a_rd_ptr_b = a_rd_ptr_sync2;
  wire [PTR_W-1:0] b_rd_ptr_a = b_rd_ptr_sync2;

  // Effective write/read enables
  wire a_write_en = a_dir & a_winc;
  wire a_read_en  = ~a_dir & a_rinc;
  wire b_write_en = b_dir & b_winc;
  wire b_read_en  = ~b_dir & b_rinc;

  // Full/Empty calculations
  // A side: writes go to FIFO, reads come from FIFO
  // A is full when a_wr_ptr - b_rd_ptr_a == DEPTH
  // A is empty when a_wr_ptr == b_rd_ptr_a
  wire [PTR_W-1:0] a_fill = a_wr_ptr - b_rd_ptr_a;
  wire a_full  = (a_fill == DEPTH);
  wire a_empty = (a_fill == 0);

  // B side: writes go to FIFO, reads come from FIFO
  // B is full when b_wr_ptr - a_rd_ptr_b == DEPTH
  // B is empty when b_wr_ptr == a_rd_ptr_b
  wire [PTR_W-1:0] b_fill = b_wr_ptr - a_rd_ptr_b;
  wire b_full  = (b_fill == DEPTH);
  wire b_empty = (b_fill == 0);

  // Almost full/empty
  wire a_afull  = (a_fill >= AFULL_THRESH);
  wire a_aempty = (a_fill <= AEMPTY_THRESH);
  wire b_afull  = (b_fill >= AFULL_THRESH);
  wire b_aempty = (b_fill <= AEMPTY_THRESH);

  // Pointer updates
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= {PTR_W{1'b0}};
      a_rd_ptr <= {PTR_W{1'b0}};
    end else begin
      if (a_write_en && !a_full)
        a_wr_ptr <= a_wr_ptr + 1'b1;
      if (a_read_en && !a_empty)
        a_rd_ptr <= a_rd_ptr + 1'b1;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= {PTR_W{1'b0}};
      b_rd_ptr <= {PTR_W{1'b0}};
    end else begin
      if (b_write_en && !b_full)
        b_wr_ptr <= b_wr_ptr + 1'b1;
      if (b_read_en && !b_empty)
        b_rd_ptr <= b_rd_ptr + 1'b1;
    end
  end

  // Memory write
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      // No action needed for memory
    end else begin
      if (a_write_en && !a_full)
        mem[a_wr_ptr[ASIZE-1:0]] <= a_wdata;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      // No action needed for memory
    end else begin
      if (b_write_en && !b_full)
        mem[b_wr_ptr[ASIZE-1:0]] <= b_wdata;
    end
  end

  // Read data
  // A reads from a_rd_ptr
  // B reads from b_rd_ptr
  // With fall-through, the read data is the word at the read pointer
  // Without fall-through, the read data is the word at the read pointer (same for async FIFO)
  // The key difference is when the data is valid: with fall-through, data is valid when not empty
  // and the read pointer points to valid data. Without fall-through, same.
  // For async FIFO, the read data is simply the memory at the read pointer.
  
  wire [DSIZE-1:0] a_rdata_internal = mem[a_rd_ptr[ASIZE-1:0]];
  wire [DSIZE-1:0] b_rdata_internal = mem[b_rd_ptr[ASIZE-1:0]];

  assign a_rdata = a_rdata_internal;
  assign b_rdata = b_rdata_internal;

  // Output assignments
  assign a_full   = a_full;
  assign a_afull  = a_afull;
  assign a_empty  = a_empty;
  assign a_aempty = a_aempty;
  assign b_full   = b_full;
  assign b_afull  = b_afull;
  assign b_empty  = b_empty;
  assign b_aempty = b_aempty;

endmodule
