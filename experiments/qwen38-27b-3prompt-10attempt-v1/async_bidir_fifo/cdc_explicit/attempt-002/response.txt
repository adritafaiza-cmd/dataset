```verilog
module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;
  localparam PTR_W = ASIZE + 1;
  localparam AFULL_THRESH  = DEPTH - 2;
  localparam AEMPTY_THRESH = 1;

  // Internal FIFO storage
  reg [DSIZE-1:0] mem [0:DEPTH-1];

  // Pointers in A domain
  reg [PTR_W-1:0] a_wr_ptr;
  reg [PTR_W-1:0] a_rd_ptr;

  // Pointers in B domain
  reg [PTR_W-1:0] b_wr_ptr;
  reg [PTR_W-1:0] b_rd_ptr;

  // Synchronized pointers
  reg [PTR_W-1:0] a_wr_ptr_sync1, a_wr_ptr_sync2;
  reg [PTR_W-1:0] a_rd_ptr_sync1, a_rd_ptr_sync2;
  reg [PTR_W-1:0] b_wr_ptr_sync1, b_wr_ptr_sync2;
  reg [PTR_W-1:0] b_rd_ptr_sync1, b_rd_ptr_sync2;

  // Synchronized reset signals
  reg a_rst_n_sync1, a_rst_n_sync2;
  reg b_rst_n_sync1, b_rst_n_sync2;

  // Direction signals synchronized to opposite domain
  reg a_dir_sync1, a_dir_sync2;
  reg b_dir_sync1, b_dir_sync2;

  // Write enable signals
  wire a_write_en;
  wire b_write_en;
  wire a_read_en;
  wire b_read_en;

  // Full/Empty flags
  wire a_full_int;
  wire a_empty_int;
  wire b_full_int;
  wire b_empty_int;

  // Almost full/empty
  wire a_afull_int;
  wire a_aempty_int;
  wire b_afull_int;
  wire b_aempty_int;

  // Read data
  wire [DSIZE-1:0] a_rdata_int;
  wire [DSIZE-1:0] b_rdata_int;

  // Determine write/read enables based on direction
  assign a_write_en = a_dir & a_winc;
  assign a_read_en  = ~a_dir & a_rinc;
  assign b_write_en = b_dir & b_winc;
  assign b_read_en  = ~b_dir & b_rinc;

  // Full/Empty detection
  // A side: full when a_wr_ptr - a_rd_ptr_sync2 == DEPTH
  // A side: empty when a_wr_ptr_sync2 == a_rd_ptr
  // B side: full when b_wr_ptr - b_rd_ptr_sync2 == DEPTH
  // B side: empty when b_wr_ptr_sync2 == b_rd_ptr

  wire [PTR_W-1:0] a_wr_diff;
  wire [PTR_W-1:0] b_wr_diff;

  assign a_wr_diff = a_wr_ptr - a_rd_ptr_sync2;
  assign b_wr_diff = b_wr_ptr - b_rd_ptr_sync2;

  assign a_full_int  = (a_wr_diff == DEPTH);
  assign a_empty_int = (a_wr_ptr_sync2 == a_rd_ptr);
  assign b_full_int  = (b_wr_diff == DEPTH);
  assign b_empty_int = (b_wr_ptr_sync2 == b_rd_ptr);

  assign a_afull_int  = (a_wr_diff >= AFULL_THRESH);
  assign a_aempty_int = (a_wr_diff <= AEMPTY_THRESH);
  assign b_afull_int  = (b_wr_diff >= AFULL_THRESH);
  assign b_aempty_int = (b_wr_diff <= AEMPTY_THRESH);

  // Read data
  assign a_rdata_int = mem[a_rd_ptr[ASIZE-1:0]];
  assign b_rdata_int = mem[b_rd_ptr[ASIZE-1:0]];

  // Output assignments
  assign a_rdata  = a_rdata_int;
  assign a_full   = a_full_int;
  assign a_afull  = a_afull_int;
  assign a_empty  = a_empty_int;
  assign a_aempty = a_aempty_int;
  assign b_rdata  = b_rdata_int;
  assign b_full   = b_full_int;
  assign b_afull  = b_afull_int;
  assign b_empty  = b_empty_int;
  assign b_aempty = b_aempty_int;

  // Reset synchronization for A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_rst_n_sync1 <= 1'b0;
      a_rst_n_sync2 <= 1'b0;
    end else begin
      a_rst_n_sync1 <= 1'b1;
      a_rst_n_sync2 <= a_rst_n_sync1;
    end
  end

  // Reset synchronization for B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_rst_n_sync1 <= 1'b0;
      b_rst_n_sync2 <= 1'b0;
    end else begin
      b_rst_n_sync1 <= 1'b1;
      b_rst_n_sync2 <= b_rst_n_sync1;
    end
  end

  // Direction synchronization: a_dir to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_dir_sync1 <= 1'b0;
      a_dir_sync2 <= 1'b0;
    end else begin
      a_dir_sync1 <= a_dir;
      a_dir_sync2 <= a_dir_sync1;
    end
  end

  // Direction synchronization: b_dir to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_dir_sync1 <= 1'b0;
      b_dir_sync2 <= 1'b0;
    end else begin
      b_dir_sync1 <= b_dir;
      b_dir_sync2 <= b_dir_sync1;
    end
  end

  // Pointer synchronization: a_wr_ptr to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_wr_ptr_sync1 <= {PTR_W{1'b0}};
      a_wr_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      a_wr_ptr_sync1 <= a_wr_ptr;
      a_wr_ptr_sync2 <= a_wr_ptr_sync1;
    end
  end

  // Pointer synchronization: a_rd_ptr to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_rd_ptr_sync1 <= {PTR_W{1'b0}};
      a_rd_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      a_rd_ptr_sync1 <= a_rd_ptr;
      a_rd_ptr_sync2 <= a_rd_ptr_sync1;
    end
  end

  // Pointer synchronization: b_wr_ptr to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_wr_ptr_sync1 <= {PTR_W{1'b0}};
      b_wr_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      b_wr_ptr_sync1 <= b_wr_ptr;
      b_wr_ptr_sync2 <= b_wr_ptr_sync1;
    end
  end

  // Pointer synchronization: b_rd_ptr to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_rd_ptr_sync1 <= {PTR_W-1:0}'(0);
      b_rd_ptr_sync2 <= {PTR_W-1:0}'(0);
    end else begin
      b_rd_ptr_sync1 <= b_rd_ptr;
      b_rd_ptr_sync2 <= b_rd_ptr_sync1;
    end
  end

  // A domain: write pointer and memory write
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= {PTR_W{1'b0}};
    end else if (a_write_en && !a_full_int) begin
      mem[a_wr_ptr[ASIZE-1:0]] <= a_wdata;
      a_wr_ptr <= a_wr_ptr + 1'b1;
    end
  end

  // A domain: read pointer
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_rd_ptr <= {PTR_W{1'b0}};
    end else if (a_read_en && !a_empty_int) begin
      a_rd_ptr <= a_rd_ptr + 1'b1;
    end
  end

  // B domain: write pointer and memory write
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= {PTR_W{1'b0}};
    end else if (b_write_en && !b_full_int) begin
      mem[b_wr_ptr[ASIZE-1:0]] <= b_wdata;
      b_wr_ptr <= b_wr_ptr + 1'b1;
    end
  end

  // B domain: read pointer
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_rd_ptr <= {PTR_W{1'b0}};
    end else if (b_read_en && !b_empty_int) begin
      b_rd_ptr <= b_rd_ptr + 1'b1;
    end
  end

endmodule
