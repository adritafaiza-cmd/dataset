```verilog
module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;
  localparam PTR_W = ASIZE + 1;
  localparam AFULL_THRESH  = DEPTH - 2;
  localparam AEMPTY_THRESH = 1;

  // Internal FIFO storage
  reg [DSIZE-1:0] mem [0:DEPTH-1];

  // Side A pointers (in A clock domain)
  reg [PTR_W-1:0] a_wr_ptr;
  reg [PTR_W-1:0] a_rd_ptr;

  // Side B pointers (in B clock domain)
  reg [PTR_W-1:0] b_wr_ptr;
  reg [PTR_W-1:0] b_rd_ptr;

  // Gray code pointers for CDC
  wire [PTR_W-1:0] a_wr_ptr_gray;
  wire [PTR_W-1:0] a_rd_ptr_gray;
  wire [PTR_W-1:0] b_wr_ptr_gray;
  wire [PTR_W-1:0] b_rd_ptr_gray;

  assign a_wr_ptr_gray = a_wr_ptr ^ {1'b0, a_wr_ptr[ASIZE-1:0]};
  assign a_rd_ptr_gray = a_rd_ptr ^ {1'b0, a_rd_ptr[ASIZE-1:0]};
  assign b_wr_ptr_gray = b_wr_ptr ^ {1'b0, b_wr_ptr[ASIZE-1:0]};
  assign b_rd_ptr_gray = b_rd_ptr ^ {1'b0, b_rd_ptr[ASIZE-1:0]};

  // Synchronizers for A-side pointers into B domain
  reg [PTR_W-1:0] a_wr_ptr_gray_b_sync1, a_wr_ptr_gray_b_sync2;
  reg [PTR_W-1:0] a_rd_ptr_gray_b_sync1, a_rd_ptr_gray_b_sync2;

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_wr_ptr_gray_b_sync1 <= {PTR_W{1'b0}};
      a_wr_ptr_gray_b_sync2 <= {PTR_W{1'b0}};
      a_rd_ptr_gray_b_sync1 <= {PTR_W{1'b0}};
      a_rd_ptr_gray_b_sync2 <= {PTR_W{1'b0}};
    end else begin
      a_wr_ptr_gray_b_sync1 <= a_wr_ptr_gray;
      a_wr_ptr_gray_b_sync2 <= a_wr_ptr_gray_b_sync1;
      a_rd_ptr_gray_b_sync1 <= a_rd_ptr_gray;
      a_rd_ptr_gray_b_sync2 <= a_rd_ptr_gray_b_sync1;
    end
  end

  // Synchronizers for B-side pointers into A domain
  reg [PTR_W-1:0] b_wr_ptr_gray_a_sync1, b_wr_ptr_gray_a_sync2;
  reg [PTR_W-1:0] b_rd_ptr_gray_a_sync1, b_rd_ptr_gray_a_sync2;

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_wr_ptr_gray_a_sync1 <= {PTR_W{1'b0}};
      b_wr_ptr_gray_a_sync2 <= {PTR_W{1'b0}};
      b_rd_ptr_gray_a_sync1 <= {PTR_W{1'b0}};
      b_rd_ptr_gray_a_sync2 <= {PTR_W{1'b0}};
    end else begin
      b_wr_ptr_gray_a_sync1 <= b_wr_ptr_gray;
      b_wr_ptr_gray_a_sync2 <= b_wr_ptr_gray_a_sync1;
      b_rd_ptr_gray_a_sync1 <= b_rd_ptr_gray;
      b_rd_ptr_gray_a_sync2 <= b_rd_ptr_gray_a_sync1;
    end
  end

  // Convert gray to binary
  function [PTR_W-1:0] gray2bin;
    input [PTR_W-1:0] gray;
    integer i;
    begin
      gray2bin[PTR_W-1] = gray[PTR_W-1];
      for (i = PTR_W-2; i >= 0; i = i - 1)
        gray2bin[i] = gray2bin[i+1] ^ gray[i];
    end
  endfunction

  wire [PTR_W-1:0] a_wr_ptr_b_bin = gray2bin(a_wr_ptr_gray_b_sync2);
  wire [PTR_W-1:0] a_rd_ptr_b_bin = gray2bin(a_rd_ptr_gray_b_sync2);
  wire [PTR_W-1:0] b_wr_ptr_a_bin = gray2bin(b_wr_ptr_gray_a_sync2);
  wire [PTR_W-1:0] b_rd_ptr_a_bin = gray2bin(b_rd_ptr_gray_a_sync2);

  // Determine effective write/read operations on each side
  // Side A: a_dir=1 means write, a_dir=0 means read
  wire a_do_write = a_dir & a_winc & ~a_full;
  wire a_do_read  = ~a_dir & a_rinc & ~a_empty;

  // Side B: b_dir=1 means write, b_dir=0 means read
  wire b_do_write = b_dir & b_winc & ~b_full;
  wire b_do_read  = ~b_dir & b_rinc & ~b_empty;

  // Write to memory
  always @(posedge a_clk) begin
    if (a_do_write)
      mem[a_wr_ptr[ASIZE-1:0]] <= a_wdata;
  end

  always @(posedge b_clk) begin
    if (b_do_write)
      mem[b_wr_ptr[ASIZE-1:0]] <= b_wdata;
  end

  // Update pointers
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= {PTR_W{1'b0}};
      a_rd_ptr <= {PTR_W{1'b0}};
    end else begin
      if (a_do_write)
        a_wr_ptr <= a_wr_ptr + 1'b1;
      if (a_do_read)
        a_rd_ptr <= a_rd_ptr + 1'b1;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= {PTR_W{1'b0}};
      b_rd_ptr <= {PTR_W{1'b0}};
    end else begin
      if (b_do_write)
        b_wr_ptr <= b_wr_ptr + 1'b1;
      if (b_do_read)
        b_rd_ptr <= b_rd_ptr + 1'b1;
    end
  end

  // Read data
  reg [DSIZE-1:0] a_rdata_reg;
  reg [DSIZE-1:0] b_rdata_reg;

  always @(posedge a_clk) begin
    if (a_do_read)
      a_rdata_reg <= mem[a_rd_ptr[ASIZE-1:0]];
  end

  always @(posedge b_clk) begin
    if (b_do_read)
      b_rdata_reg <= mem[b_rd_ptr[ASIZE-1:0]];
  end

  // Full/Empty flags
  // A side: full when a_wr_ptr - b_rd_ptr_a_bin == DEPTH
  // A side: empty when a_wr_ptr == b_rd_ptr_a_bin
  wire a_full_internal  = (a_wr_ptr == (b_rd_ptr_a_bin + DEPTH));
  wire a_empty_internal = (a_wr_ptr == b_rd_ptr_a_bin);

  // B side: full when b_wr_ptr - a_rd_ptr_b_bin == DEPTH
  // B side: empty when b_wr_ptr == a_rd_ptr_b_bin
  wire b_full_internal  = (b_wr_ptr == (a_rd_ptr_b_bin + DEPTH));
  wire b_empty_internal = (b_wr_ptr == a_rd_ptr_b_bin);

  assign a_full  = a_full_internal;
  assign a_empty = a_empty_internal;
  assign b_full  = b_full_internal;
  assign b_empty = b_empty_internal;

  // Almost full/empty
  wire [PTR_W-1:0] a_count = a_wr_ptr - b_rd_ptr_a_bin;
  wire [PTR_W-1:0] b_count = b_wr_ptr - a_rd_ptr_b_bin;

  assign a_afull   = (a_count >= AFULL_THRESH);
  assign a_aempty  = (a_count <= AEMPTY_THRESH);
  assign b_afull   = (b_count >= AFULL_THRESH);
  assign b_aempty  = (b_count <= AEMPTY_THRESH);

  // Output data
  assign a_rdata = a_rdata_reg;
  assign b_rdata = b_rdata_reg;

endmodule
