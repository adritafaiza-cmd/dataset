```verilog
module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;
  localparam PTR_W = ASIZE + 1;
  localparam AFULL_THRESH = DEPTH - 2;
  localparam AEMPTY_THRESH = 2;

  // Internal FIFO storage
  reg [DSIZE-1:0] mem [0:DEPTH-1];

  // Side A pointers (in A clock domain)
  reg [PTR_W-1:0] a_wr_ptr;
  reg [PTR_W-1:0] a_rd_ptr;

  // Side B pointers (in B clock domain)
  reg [PTR_W-1:0] b_wr_ptr;
  reg [PTR_W-1:0] b_rd_ptr;

  // Gray code pointers for CDC
  wire [ASIZE-1:0] a_wr_ptr_gray;
  wire [ASIZE-1:0] a_rd_ptr_gray;
  wire [ASIZE-1:0] b_wr_ptr_gray;
  wire [ASIZE-1:0] b_rd_ptr_gray;

  // Synchronized gray code pointers
  reg [ASIZE-1:0] a_wr_ptr_gray_sync1, a_wr_ptr_gray_sync2;
  reg [ASIZE-1:0] a_rd_ptr_gray_sync1, a_rd_ptr_gray_sync2;
  reg [ASIZE-1:0] b_wr_ptr_gray_sync1, b_wr_ptr_gray_sync2;
  reg [ASIZE-1:0] b_rd_ptr_gray_sync1, b_rd_ptr_gray_sync2;

  // Binary conversion of synchronized gray codes
  wire [PTR_W-1:0] a_wr_ptr_bin_sync;
  wire [PTR_W-1:0] a_rd_ptr_bin_sync;
  wire [PTR_W-1:0] b_wr_ptr_bin_sync;
  wire [PTR_W-1:0] b_rd_ptr_bin_sync;

  // Write enable signals
  wire a_write_en;
  wire b_write_en;
  wire a_read_en;
  wire b_read_en;

  assign a_write_en = a_dir & a_winc & ~a_full;
  assign b_write_en = b_dir & b_winc & ~b_full;
  assign a_read_en  = ~a_dir & a_rinc & ~a_empty;
  assign b_read_en  = ~b_dir & b_rinc & ~b_empty;

  // Gray code conversion
  function [ASIZE-1:0] bin2gray;
    input [PTR_W-1:0] bin;
    begin
      bin2gray = bin ^ (bin >> 1);
    end
  endfunction

  function [PTR_W-1:0] gray2bin;
    input [ASIZE-1:0] gray;
    reg [PTR_W-1:0] bin;
    integer i;
    begin
      bin[PTR_W-1] = gray[ASIZE-1];
      for (i = ASIZE-2; i >= 0; i = i - 1)
        bin[i] = bin[i+1] ^ gray[i];
      gray2bin = bin;
    end
  endfunction

  assign a_wr_ptr_gray = bin2gray(a_wr_ptr);
  assign a_rd_ptr_gray = bin2gray(a_rd_ptr);
  assign b_wr_ptr_gray = bin2gray(b_wr_ptr);
  assign b_rd_ptr_gray = bin2gray(b_rd_ptr);

  // Synchronize A write pointer to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_wr_ptr_gray_sync1 <= {ASIZE{1'b0}};
      a_wr_ptr_gray_sync2 <= {ASIZE{1'b0}};
    end else begin
      a_wr_ptr_gray_sync1 <= a_wr_ptr_gray;
      a_wr_ptr_gray_sync2 <= a_wr_ptr_gray_sync1;
    end
  end

  // Synchronize A read pointer to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_rd_ptr_gray_sync1 <= {ASIZE{1'b0}};
      a_rd_ptr_gray_sync2 <= {ASIZE{1'b0}};
    end else begin
      a_rd_ptr_gray_sync1 <= a_rd_ptr_gray;
      a_rd_ptr_gray_sync2 <= a_rd_ptr_gray_sync1;
    end
  end

  // Synchronize B write pointer to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_wr_ptr_gray_sync1 <= {ASIZE{1'b0}};
      b_wr_ptr_gray_sync2 <= {ASIZE{1'b0}};
    end else begin
      b_wr_ptr_gray_sync1 <= b_wr_ptr_gray;
      b_wr_ptr_gray_sync2 <= b_wr_ptr_gray_sync1;
    end
  end

  // Synchronize B read pointer to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_rd_ptr_gray_sync1 <= {ASIZE{1'b0}};
      b_rd_ptr_gray_sync2 <= {ASIZE{1'b0}};
    end else begin
      b_rd_ptr_gray_sync1 <= b_rd_ptr_gray;
      b_rd_ptr_gray_sync2 <= b_rd_ptr_gray_sync1;
    end
  end

  // Convert synchronized gray codes to binary
  assign a_wr_ptr_bin_sync = gray2bin(a_wr_ptr_gray_sync2);
  assign a_rd_ptr_bin_sync = gray2bin(a_rd_ptr_gray_sync2);
  assign b_wr_ptr_bin_sync = gray2bin(b_wr_ptr_gray_sync2);
  assign b_rd_ptr_bin_sync = gray2bin(b_rd_ptr_gray_sync2);

  // Update pointers
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= {PTR_W{1'b0}};
      a_rd_ptr <= {PTR_W{1'b0}};
    end else begin
      if (a_write_en)
        a_wr_ptr <= a_wr_ptr + 1'b1;
      if (a_read_en)
        a_rd_ptr <= a_rd_ptr + 1'b1;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= {PTR_W{1'b0}};
      b_rd_ptr <= {PTR_W{1'b0}};
    end else begin
      if (b_write_en)
        b_wr_ptr <= b_wr_ptr + 1'b1;
      if (b_read_en)
        b_rd_ptr <= b_rd_ptr + 1'b1;
    end
  end

  // Memory write
  always @(posedge a_clk) begin
    if (a_write_en)
      mem[a_wr_ptr[ASIZE-1:0]] <= a_wdata;
  end

  always @(posedge b_clk) begin
    if (b_write_en)
      mem[b_wr_ptr[ASIZE-1:0]] <= b_wdata;
  end

  // Memory read
  wire [DSIZE-1:0] a_mem_rdata;
  wire [DSIZE-1:0] b_mem_rdata;

  assign a_mem_rdata = mem[a_rd_ptr[ASIZE-1:0]];
  assign b_mem_rdata = mem[b_rd_ptr[ASIZE-1:0]];

  // Output data
  assign a_rdata = a_mem_rdata;
  assign b_rdata = b_mem_rdata;

  // Status flags
  // A side: full when A write pointer catches up to B read pointer (synchronized)
  // Empty when A read pointer catches up to B write pointer (synchronized)
  wire a_full_internal;
  wire a_empty_internal;
  wire b_full_internal;
  wire b_empty_internal;

  assign a_full_internal  = (a_wr_ptr == b_rd_ptr_bin_sync);
  assign a_empty_internal = (a_rd_ptr == b_wr_ptr_bin_sync);
  assign b_full_internal  = (b_wr_ptr == a_rd_ptr_bin_sync);
  assign b_empty_internal = (b_rd_ptr == a_wr_ptr_bin_sync);

  assign a_full   = a_full_internal;
  assign b_full   = b_full_internal;
  assign a_empty  = a_empty_internal;
  assign b_empty  = b_empty_internal;

  // Almost full/empty
  wire [PTR_W-1:0] a_fill_level;
  wire [PTR_W-1:0] b_fill_level;

  assign a_fill_level = a_wr_ptr - b_rd_ptr_bin_sync;
  assign b_fill_level = b_wr_ptr - a_rd_ptr_bin_sync;

  assign a_afull   = (a_fill_level >= AFULL_THRESH);
  assign b_afull   = (b_fill_level >= AFULL_THRESH);
  assign a_aempty  = (a_fill_level <= AEMPTY_THRESH);
  assign b_aempty  = (b_fill_level <= AEMPTY_THRESH);

endmodule
