  wire b_empty_raw;

  assign a_empty  = a_empty_raw;

  assign a_aempty = a_aempty;

  assign b_empty  = b_empty_raw;
  assign b_afull  = b_afull;

  // B reads when a_dir=0 and b_rinc and not empty (i.e., b_dir=1)
