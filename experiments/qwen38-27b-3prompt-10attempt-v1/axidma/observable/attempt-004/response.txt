```verilog
module axidma #(
	parameter	C_AXI_ID_WIDTH = 1,
	parameter	C_AXI_ADDR_WIDTH = 32,
	parameter	C_AXI_DATA_WIDTH = 32,
	localparam	C_AXIL_ADDR_WIDTH = 5,
	localparam	C_AXIL_DATA_WIDTH = 32,
	parameter [0:0]	OPT_UNALIGNED = 1'b1,
	parameter [0:0]	OPT_WRAPMEM = 1'b1,
	parameter	LGFIFO = LGMAXBURST+1,
	parameter	LGLEN = C_AXI_ADDR_WIDTH,
	parameter [0:0]	OPT_LOWPOWER = 1'b0,
	parameter [0:0]	OPT_CLKGATE = OPT_LOWPOWER,
	parameter	[C_AXI_ID_WIDTH-1:0]	AXI_READ_ID = 0,
	parameter	[C_AXI_ID_WIDTH-1:0]	AXI_WRITE_ID = 0,
	parameter	[7:0]			ABORT_KEY  = 8'h6d,
	localparam	ADDRLSB= $clog2(C_AXI_DATA_WIDTH)-3,
	localparam	AXILLSB= $clog2(C_AXIL_DATA_WIDTH)-3,
	localparam	LGLENW= LGLEN-ADDRLSB
) (
	input	wire	S_AXI_ACLK,
	input	wire	S_AXI_ARESETN,
	input	wire				S_AXIL_AWVALID,
	output	wire				S_AXIL_AWREADY,
	input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_AWADDR,
	input	wire 	[2:0]			S_AXIL_AWPROT,
	input	wire				S_AXIL_WVALID,
	output	wire				S_AXIL_WREADY,
	input	wire [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_WDATA,
	input	wire [C_AXIL_DATA_WIDTH/8-1:0]	S_AXIL_WSTRB,
	output	reg				S_AXIL_BVALID,
	input	wire				S_AXIL_BREADY,
	output	wire	[1:0]			S_AXIL_BRESP,
	input	wire				S_AXIL_ARVALID,
	output	wire				S_AXIL_ARREADY,
	input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_ARADDR,
	input	wire 	[2:0]			S_AXIL_ARPROT,
	output	reg				S_AXIL_RVALID,
	input	wire				S_AXIL_RREADY,
	output	reg [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_RDATA,
	output	wire	[1:0]			S_AXIL_RRESP,
	output	reg				M_AXI_AWVALID,
	input	wire				M_AXI_AWREADY,
	output	reg	[C_AXI_ID_WIDTH-1:0]	M_AXI_AWID,
	output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_AWADDR,
	output	reg	[2:0]			M_AXI_AWSIZE,
	output	reg	[1:0]			M_AXI_AWBURST,
	output	reg				M_AXI_AWLOCK,
	output	reg	[3:0]			M_AXI_AWCACHE,
	output	reg	[2:0]			M_AXI_AWPROT,
	output	reg	[3:0]			M_AXI_AWQOS,
	output	reg				M_AXI_WVALID,
	input	wire				M_AXI_WREADY,
	output	reg	[C_AXI_DATA_WIDTH-1:0]	M_AXI_WDATA,
	output	reg [C_AXI_DATA_WIDTH/8-1:0]	M_AXI_WSTRB,
	output	reg				M_AXI_WLAST,
	input	wire				M_AXI_BVALID,
	output	reg				M_AXI_BREADY,
	input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_BID,
	input	wire	[1:0]			M_AXI_BRESP,
	output	reg				M_AXI_ARVALID,
	input	wire				M_AXI_ARREADY,
	output	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_ARID,
	output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_ARADDR,
	output	wire	[2:0]			M_AXI_ARSIZE,
	output	wire	[1:0]			M_AXI_ARBURST,
	output	wire				M_AXI_ARLOCK,
	output	wire	[3:0]			M_AXI_ARCACHE,
	output	wire	[2:0]			M_AXI_ARPROT,
	output	wire	[3:0]			M_AXI_ARQOS,
	input	wire				M_AXI_RVALID,
	output	wire				M_AXI_RREADY,
	input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_RID,
	input	wire	[C_AXI_DATA_WIDTH-1:0]	M_AXI_RDATA,
	input	wire				M_AXI_RLAST,
	input	wire	[1:0]			M_AXI_RRESP,
	output	reg				o_int
);

	// Register definitions
	localparam REG_SRC_ADDR = 0;
	localparam REG_DST_ADDR = 1;
	localparam REG_LENGTH   = 2;
	localparam REG_CONTROL  = 3;
	localparam REG_STATUS   = 4;

	// Control register bits
	localparam CTRL_START = 0;
	localparam CTRL_ABORT = 1;

	// Status register bits
	localparam STAT_BUSY  = 0;
	localparam STAT_DONE  = 1;
	localparam STAT_ERROR = 2;

	// State machine states
	localparam ST_IDLE    = 2'd0;
	localparam ST_READ    = 2'd1;
	localparam ST_WRITE   = 2'd2;
	localparam ST_DONE    = 2'd3;

	// Internal signals
	reg [C_AXI_ADDR_WIDTH-1:0] src_addr_reg;
	reg [C_AXI_ADDR_WIDTH-1:0] dst_addr_reg;
	reg [C_AXI_ADDR_WIDTH-1:0] length_reg;
	reg [7:0] control_reg;
	reg [7:0] status_reg;

	reg [1:0] state;
	reg [C_AXI_ADDR_WIDTH-1:0] cur_src_addr;
	reg [C_AXI_ADDR_WIDTH-1:0] cur_dst_addr;
	reg [C_AXI_ADDR_WIDTH-1:0] remaining_len;

	reg [C_AXI_DATA_WIDTH-1:0] rdata_buf;
	reg rdata_valid;
	reg wdata_valid;
	reg [C_AXI_DATA_WIDTH-1:0] wdata_buf;

	reg [C_AXI_ADDR_WIDTH-1:0] next_src_addr;
	reg [C_AXI_ADDR_WIDTH-1:0] next_dst_addr;

	// AXI-Lite control interface
	reg aw_ready;
	reg w_ready;
	reg ar_ready;
	reg b_valid;
	reg r_valid;
	reg [C_AXIL_DATA_WIDTH-1:0] rdata_out;

	// AXI-Lite write address decode
	wire [C_AXIL_ADDR_WIDTH-1:0] awaddr = S_AXIL_AWADDR;
	wire [1:0] awreg = awaddr[1:0];

	// AXI-Lite read address decode
	wire [C_AXIL_ADDR_WIDTH-1:0] araddr = S_AXIL_ARADDR;
	wire [1:0] arreg = araddr[1:0];

	// AXI-Lite write data
	wire [C_AXIL_DATA_WIDTH-1:0] wdata = S_AXIL_WDATA;

	// AXI-Lite write ready
	assign S_AXIL_AWREADY = aw_ready;
	assign S_AXIL_WREADY = w_ready;
	assign S_AXIL_ARREADY = ar_ready;
	assign S_AXIL_BVALID = b_valid;
	assign S_AXIL_BRESP = 2'b00;
	assign S_AXIL_RVALID = r_valid;
	assign S_AXIL_RDATA = rdata_out;
	assign S_AXIL_RRESP = 2'b00;

	// AXI master interface
	assign M_AXI_ARID = AXI_READ_ID;
	assign M_AXI_ARSIZE = 2'b010; // 4 bytes
	assign M_AXI_ARBURST = 2'b001; // INCR
	assign M_AXI_ARLOCK = 1'b0;
	assign M_AXI_ARCACHE = 4'b0000;
	assign M_AXI_ARPROT = 3'b000;
	assign M_AXI_ARQOS = 4'b0000;

	assign M_AXI_AWID = AXI_WRITE_ID;
	assign M_AXI_AWSIZE = 2'b010; // 4 bytes
	assign M_AXI_AWBURST = 2'b001; // INCR
	assign M_AXI_AWLOCK = 1'b0;
	assign M_AXI_AWCACHE = 4'b0000;
	assign M_AXI_AWPROT = 3'b000;
	assign M_AXI_AWQOS = 4'b0000;

	// AXI write data
	assign M_AXI_WSTRB = {(C_AXI_DATA_WIDTH/8){1'b1}};

	// AXI read ready
	assign M_AXI_RREADY = (state == ST_READ) && !rdata_valid;

	// AXI write ready
	assign M_AXI_BREADY = (state == ST_WRITE) && wdata_valid;

	// Interrupt
	assign o_int = status_reg[STAT_DONE] || status_reg[STAT_ERROR];

	// AXI-Lite write address phase
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			aw_ready <= 1'b1;
			b_valid <= 1'b0;
		end else begin
			if (aw_ready && S_AXIL_AWVALID) begin
				aw_ready <= 1'b0;
				b_valid <= 1'b1;
			end else if (b_valid && S_AXIL_BREADY) begin
				b_valid <= 1'b0;
				aw_ready <= 1'b1;
			end
		end
	end

	// AXI-Lite write data phase
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			w_ready <= 1'b0;
		end else begin
			if (!aw_ready && S_AXIL_AWVALID) begin
				w_ready <= 1'b1;
			end else if (w_ready && S_AXIL_WVALID) begin
				w_ready <= 1'b0;
				case (awreg)
					2'd0: src_addr_reg <= wdata[C_AXI_ADDR_WIDTH-1:0];
					2'd1: dst_addr_reg <= wdata[C_AXI_ADDR_WIDTH-1:0];
					2'd2: length_reg <= wdata[C_AXI_ADDR_WIDTH-1:0];
					2'd3: begin
						control_reg <= wdata[7:0];
						if (wdata[CTRL_START]) begin
							cur_src_addr <= src_addr_reg;
							cur_dst_addr <= dst_addr_reg;
							remaining_len <= length_reg;
							state <= ST_READ;
							status_reg[STAT_BUSY] <= 1'b1;
							status_reg[STAT_DONE] <= 1'b0;
							status_reg[STAT_ERROR] <= 1'b0;
						end
						if (wdata[CTRL_ABORT]) begin
							state <= ST_IDLE;
							status_reg[STAT_BUSY] <= 1'b0;
							status_reg[STAT_ERROR] <= 1'b1;
						end
					end
					default: ;
				endcase
			end
		end
	end

	// AXI-Lite read address phase
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			ar_ready <= 1'b1;
			r_valid <= 1'b0;
			rdata_out <= {C_AXIL_DATA_WIDTH{1'b0}};
		end else begin
			if (ar_ready && S_AXIL_ARVALID) begin
				ar_ready <= 1'b0;
				r_valid <= 1'b1;
				case (arreg)
					2'd0: rdata_out <= src_addr_reg[C_AXIL_DATA_WIDTH-1:0];
					2'd1: rdata_out <= dst_addr_reg[C_AXIL_DATA_WIDTH-1:0];
					2'd2: rdata_out <= length_reg[C_AXIL_DATA_WIDTH-1:0];
					2'd3: rdata_out <= control_reg;
					2'd4: rdata_out <= status_reg;
					default: rdata_out <= {C_AXIL_DATA_WIDTH{1'b0}};
				endcase
			end else if (r_valid && S_AXIL_RREADY) begin
				r_valid <= 1'b0;
				ar_ready <= 1'b1;
			end
		end
	end

	// AXI read address
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			M_AXI_ARVALID <= 1'b0;
			M_AXI_ARADDR <= {C_AXI_ADDR_WIDTH{1'b0}};
		end else begin
			if (state == ST_IDLE) begin
				M_AXI_ARVALID <= 1'b0;
			end else if (state == ST_READ && !rdata_valid && remaining_len > 0) begin
				if (!M_AXI_ARVALID) begin
					M_AXI_ARVALID <= 1'b1;
					M_AXI_ARADDR <= cur_src_addr;
				end
			end else if (M_AXI_ARVALID && M_AXI_ARREADY) begin
				M_AXI_ARVALID <= 1'b0;
			end
		end
	end

	// AXI read data handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			rdata_valid <= 1'b0;
			rdata_buf <= {C_AXI_DATA_WIDTH{1'b0}};
		end else begin
			if (M_AXI_RVALID && M_AXI_RREADY) begin
				rdata_buf <= M_AXI_RDATA;
				rdata_valid <= 1'b1;
			end else if (state == ST_WRITE && wdata_valid) begin
				rdata_valid <= 1'b0;
			end
		end
	end

	// AXI write address
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			M_AXI_AWVALID <= 1'b0;
			M_AXI_AWADDR <= {C_AXI_ADDR_WIDTH{1'b0}};
		end else begin
			if (state == ST_IDLE) begin
				M_AXI_AWVALID <= 1'b0;
			end else if (state == ST_WRITE && rdata_valid && !wdata_valid) begin
				if (!M_AXI_AWVALID) begin
					M_AXI_AWVALID <= 1'b1;
					M_AXI_AWADDR <= cur_dst_addr;
				end
			end else if (M_AXI_AWVALID && M_AXI_AWREADY) begin
				M_AXI_AWVALID <= 1'b0;
			end
		end
	end

	// AXI write data
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			M_AXI_WVALID <= 1'b0;
			M_AXI_WDATA <= {C_AXI_DATA_WIDTH{1'b0}};
			M_AXI_WLAST <= 1'b0;
			wdata_valid <= 1'b0;
			wdata_buf <= {C_AXI_DATA_WIDTH{1'b0}};
		end else begin
			if (state == ST_IDLE) begin
				M_AXI_WVALID <= 1'b0;
				wdata_valid <= 1'b0;
			end else if (state == ST_WRITE && rdata_valid && !wdata_valid) begin
				if (!M_AXI_WVALID) begin
					M_AXI_WVALID <= 1'b1;
					M_AXI_WDATA <= rdata_buf;
					M_AXI_WLAST <= 1'b1;
					wdata_valid <= 1'b1;
					wdata_buf <= rdata_buf;
				end
			end else if (M_AXI_WVALID && M_AXI_WREADY) begin
				M_AXI_WVALID <= 1'b0;
			end
		end
	end

	// State machine
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			state <= ST_IDLE;
			cur_src_addr <= {C_AXI_ADDR_WIDTH{1'b0}};
			cur_dst_addr <= {C_AXI_ADDR_WIDTH{1'b0}};
			remaining_len <= {C_AXI_ADDR_WIDTH{1'b0}};
			status_reg <= 8'b0;
		end else begin
			case (state)
				ST_IDLE: begin
					if (control_reg[CTRL_START]) begin
						state <= ST_READ;
						cur_src_addr <= src_addr_reg;
						cur_dst_addr <= dst_addr_reg;
						remaining_len <= length_reg;
						status_reg[STAT_BUSY] <= 1'b1;
						status_reg[STAT_DONE] <= 1'b0;
						status_reg[STAT_ERROR] <= 1'b0;
					end
				end
				ST_READ: begin
					if (rdata_valid) begin
						state <= ST_WRITE;
					end
				end
				ST_WRITE: begin
					if (M_AXI_BVALID && M_AXI_BREADY) begin
						if (remaining_len > 4) begin
							remaining_len <= remaining_len - 4;
							cur_src_addr <= cur_src_addr + 4;
							cur_dst_addr <= cur_dst_addr + 4;
							state <= ST_READ;
						end else begin
							state <= ST_DONE;
							status_reg[STAT_BUSY] <= 1'b0;
							status_reg[STAT_DONE] <= 1'b1;
						end
					end
				end
				ST_DONE: begin
					state <= ST_IDLE;
				end
				default: state <= ST_IDLE;
			endcase
		end
	end

endmodule
