```verilog
module axidma #(
	parameter	C_AXI_ID_WIDTH = 1,
	parameter	C_AXI_ADDR_WIDTH = 32,
	parameter	C_AXI_DATA_WIDTH = 32,
	localparam	C_AXIL_ADDR_WIDTH = 5,
	localparam	C_AXIL_DATA_WIDTH = 32,
	parameter [0:0]	OPT_UNALIGNED = 1'b1,
	parameter [0:0]	OPT_WRAPMEM = 1'b1,
	parameter	LGFIFO = LGMAXBURST+1,
	parameter	LGLEN = C_AXI_ADDR_WIDTH,
	parameter [0:0]	OPT_LOWPOWER = 1'b0,
	parameter [0:0]	OPT_CLKGATE = OPT_LOWPOWER,
	parameter	[C_AXI_ID_WIDTH-1:0]	AXI_READ_ID = 0,
	parameter	[C_AXI_ID_WIDTH-1:0]	AXI_WRITE_ID = 0,
	parameter	[7:0]			ABORT_KEY  = 8'h6d,
	localparam	ADDRLSB= $clog2(C_AXI_DATA_WIDTH)-3,
	localparam	AXILLSB= $clog2(C_AXIL_DATA_WIDTH)-3,
	localparam	LGLENW= LGLEN-ADDRLSB
) (
	input	wire	S_AXI_ACLK,
	input	wire	S_AXI_ARESETN,
	input	wire				S_AXIL_AWVALID,
	output	wire				S_AXIL_AWREADY,
	input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_AWADDR,
	input	wire 	[2:0]			S_AXIL_AWPROT,
	input	wire				S_AXIL_WVALID,
	output	wire				S_AXIL_WREADY,
	input	wire [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_WDATA,
	input	wire [C_AXIL_DATA_WIDTH/8-1:0]	S_AXIL_WSTRB,
	output	reg				S_AXIL_BVALID,
	input	wire				S_AXIL_BREADY,
	output	wire	[1:0]			S_AXIL_BRESP,
	input	wire				S_AXIL_ARVALID,
	output	wire				S_AXIL_ARREADY,
	input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_ARADDR,
	input	wire 	[2:0]			S_AXIL_ARPROT,
	output	reg				S_AXIL_RVALID,
	input	wire				S_AXIL_RREADY,
	output	reg [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_RDATA,
	output	wire	[1:0]			S_AXIL_RRESP,
	output	reg				M_AXI_AWVALID,
	input	wire				M_AXI_AWREADY,
	output	reg	[C_AXI_ID_WIDTH-1:0]	M_AXI_AWID,
	output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_AWADDR,
	output	reg	[2:0]			M_AXI_AWSIZE,
	output	reg	[1:0]			M_AXI_AWBURST,
	output	reg				M_AXI_AWLOCK,
	output	reg	[3:0]			M_AXI_AWCACHE,
	output	reg	[2:0]			M_AXI_AWPROT,
	output	reg	[3:0]			M_AXI_AWQOS,
	output	reg				M_AXI_WVALID,
	input	wire				M_AXI_WREADY,
	output	reg	[C_AXI_DATA_WIDTH-1:0]	M_AXI_WDATA,
	output	reg [C_AXI_DATA_WIDTH/8-1:0]	M_AXI_WSTRB,
	output	reg				M_AXI_WLAST,
	input	wire				M_AXI_BVALID,
	output	reg				M_AXI_BREADY,
	input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_BID,
	input	wire	[1:0]			M_AXI_BRESP,
	output	reg				M_AXI_ARVALID,
	input	wire				M_AXI_ARREADY,
	output	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_ARID,
	output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_ARADDR,
	output	wire	[2:0]			M_AXI_ARSIZE,
	output	wire	[1:0]			M_AXI_ARBURST,
	output	wire				M_AXI_ARLOCK,
	output	wire	[3:0]			M_AXI_ARCACHE,
	output	wire	[2:0]			M_AXI_ARPROT,
	output	wire	[3:0]			M_AXI_ARQOS,
	input	wire				M_AXI_RVALID,
	output	wire				M_AXI_RREADY,
	input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_RID,
	input	wire	[C_AXI_DATA_WIDTH-1:0]	M_AXI_RDATA,
	input	wire				M_AXI_RLAST,
	input	wire	[1:0]			M_AXI_RRESP,
	output	reg				o_int
);

	// Register file for AXI-Lite control
	reg [C_AXI_ADDR_WIDTH-1:0] src_addr;
	reg [C_AXI_ADDR_WIDTH-1:0] dst_addr;
	reg [C_AXI_ADDR_WIDTH-1:0] xfer_len;
	reg [7:0] ctrl_reg;
	reg [31:0] status_reg;

	// State machine
	localparam S_IDLE = 2'd0;
	localparam S_READ = 2'd1;
	localparam S_WRITE = 2'd2;
	localparam S_DONE = 2'd3;

	reg [1:0] state;
	reg [C_AXI_ADDR_WIDTH-1:0] cur_src_addr;
	reg [C_AXI_ADDR_WIDTH-1:0] cur_dst_addr;
	reg [C_AXI_ADDR_WIDTH-1:0] remaining;
	reg [C_AXI_ADDR_WIDTH-1:0] burst_len;
	reg [C_AXI_ADDR_WIDTH-1:0] burst_count;
	reg [C_AXI_ADDR_WIDTH-1:0] read_offset;
	reg [C_AXI_ADDR_WIDTH-1:0] write_offset;
	reg [C_AXI_DATA_WIDTH-1:0] rdata_buf;
	reg [C_AXI_DATA_WIDTH/8-1:0] wstrb_buf;
	reg rdata_valid;
	reg wdata_valid;
	reg [C_AXI_ADDR_WIDTH-1:0] rdata_addr;
	reg [C_AXI_ADDR_WIDTH-1:0] wdata_addr;

	// AXI-Lite write address
	reg [C_AXIL_ADDR_WIDTH-1:0] awaddr_reg;
	reg awaddr_valid;

	// AXI-Lite write data
	reg [C_AXIL_DATA_WIDTH-1:0] wdata_reg;
	reg [C_AXIL_DATA_WIDTH/8-1:0] wstrb_reg;
	reg wdata_valid_axil;

	// AXI-Lite read address
	reg [C_AXIL_ADDR_WIDTH-1:0] araddr_reg;
	reg araddr_valid;

	// AXI-Lite read data
	reg [C_AXIL_DATA_WIDTH-1:0] rdata_axil;
	reg rdata_valid_axil;

	// AXI-Lite response
	reg [1:0] bresp_reg;
	reg [1:0] rresp_reg;

	// Burst size calculation
	wire [2:0] axi_size = $clog2(C_AXI_DATA_WIDTH) - 3;
	wire [1:0] axi_burst = 2'b01; // INCR

	// Calculate max burst length
	localparam MAX_BURST = 16;
	reg [C_AXI_ADDR_WIDTH-1:0] max_burst_len;

	always @(*) begin
		if (remaining > MAX_BURST)
			max_burst_len = MAX_BURST;
		else
			max_burst_len = remaining;
	end

	// AXI-Lite control register offsets
	localparam CTRL_OFFSET = 0;
	localparam SRC_OFFSET = 4;
	localparam DST_OFFSET = 8;
	localparam LEN_OFFSET = 12;
	localparam STATUS_OFFSET = 16;

	// AXI-Lite write address handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			awaddr_reg <= 0;
			awaddr_valid <= 0;
		end else begin
			if (S_AXIL_AWVALID && S_AXIL_AWREADY) begin
				awaddr_reg <= S_AXIL_AWADDR;
				awaddr_valid <= 1;
			end else begin
				awaddr_valid <= 0;
			end
		end
	end

	// AXI-Lite write data handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			wdata_reg <= 0;
			wstrb_reg <= 0;
			wdata_valid_axil <= 0;
		end else begin
			if (S_AXIL_WVALID && S_AXIL_WREADY) begin
				wdata_reg <= S_AXIL_WDATA;
				wstrb_reg <= S_AXIL_WSTRB;
				wdata_valid_axil <= 1;
			end else begin
				wdata_valid_axil <= 0;
			end
		end
	end

	// AXI-Lite read address handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			araddr_reg <= 0;
			araddr_valid <= 0;
		end else begin
			if (S_AXIL_ARVALID && S_AXIL_ARREADY) begin
				araddr_reg <= S_AXIL_ARADDR;
				araddr_valid <= 1;
			end else begin
				araddr_valid <= 0;
			end
		end
	end

	// AXI-Lite read data handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			rdata_axil <= 0;
			rdata_valid_axil <= 0;
		end else begin
			if (araddr_valid && S_AXIL_RREADY) begin
				rdata_valid_axil <= 1;
			end else begin
				rdata_valid_axil <= 0;
			end
		end
	end

	// AXI-Lite ready signals
	assign S_AXIL_AWREADY = !awaddr_valid;
	assign S_AXIL_WREADY = !wdata_valid_axil;
	assign S_AXIL_ARREADY = !araddr_valid;

	// AXI-Lite response
	assign S_AXIL_BRESP = bresp_reg;
	assign S_AXIL_RRESP = rresp_reg;

	// AXI-Lite B channel
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			S_AXIL_BVALID <= 0;
			bresp_reg <= 0;
		end else begin
			if (S_AXIL_BVALID && S_AXIL_BREADY) begin
				S_AXIL_BVALID <= 0;
			end else if (wdata_valid_axil && awaddr_valid) begin
				S_AXIL_BVALID <= 1;
				bresp_reg <= 2'b00;
			end
		end
	end

	// AXI-Lite R channel
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			S_AXIL_RVALID <= 0;
			S_AXIL_RDATA <= 0;
			rresp_reg <= 0;
		end else begin
			if (S_AXIL_RVALID && S_AXIL_RREADY) begin
				S_AXIL_RVALID <= 0;
			end else if (araddr_valid) begin
				S_AXIL_RVALID <= 1;
				rresp_reg <= 2'b00;
				case (araddr_reg)
					CTRL_OFFSET: S_AXIL_RDATA <= {24'b0, ctrl_reg};
					SRC_OFFSET: S_AXIL_RDATA <= src_addr[C_AXIL_DATA_WIDTH-1:0];
					DST_OFFSET: S_AXIL_RDATA <= dst_addr[C_AXIL_DATA_WIDTH-1:0];
					LEN_OFFSET: S_AXIL_RDATA <= xfer_len[C_AXIL_DATA_WIDTH-1:0];
					STATUS_OFFSET: S_AXIL_RDATA <= status_reg;
					default: S_AXIL_RDATA <= 0;
				endcase
			end
		end
	end

	// AXI-Lite write register update
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			ctrl_reg <= 0;
			src_addr <= 0;
			dst_addr <= 0;
			xfer_len <= 0;
			status_reg <= 0;
		end else begin
			if (wdata_valid_axil && awaddr_valid) begin
				case (awaddr_reg)
					CTRL_OFFSET: begin
						if (wstrb_reg[0]) ctrl_reg[0] <= wdata_reg[0];
						if (wstrb_reg[1]) ctrl_reg[1] <= wdata_reg[1];
						if (wstrb_reg[2]) ctrl_reg[2] <= wdata_reg[2];
						if (wstrb_reg[3]) ctrl_reg[3] <= wdata_reg[3];
						if (wstrb_reg[4]) ctrl_reg[4] <= wdata_reg[4];
						if (wstrb_reg[5]) ctrl_reg[5] <= wdata_reg[5];
						if (wstrb_reg[6]) ctrl_reg[6] <= wdata_reg[6];
						if (wstrb_reg[7]) ctrl_reg[7] <= wdata_reg[7];
					end
					SRC_OFFSET: begin
						if (wstrb_reg[0]) src_addr[7:0] <= wdata_reg[7:0];
						if (wstrb_reg[1]) src_addr[15:8] <= wdata_reg[15:8];
						if (wstrb_reg[2]) src_addr[23:16] <= wdata_reg[23:16];
						if (wstrb_reg[3]) src_addr[31:24] <= wdata_reg[31:24];
					end
					DST_OFFSET: begin
						if (wstrb_reg[0]) dst_addr[7:0] <= wdata_reg[7:0];
						if (wstrb_reg[1]) dst_addr[15:8] <= wdata_reg[15:8];
						if (wstrb_reg[2]) dst_addr[23:16] <= wdata_reg[23:16];
						if (wstrb_reg[3]) dst_addr[31:24] <= wdata_reg[31:24];
					end
					LEN_OFFSET: begin
						if (wstrb_reg[0]) xfer_len[7:0] <= wdata_reg[7:0];
						if (wstrb_reg[1]) xfer_len[15:8] <= wdata_reg[15:8];
						if (wstrb_reg[2]) xfer_len[23:16] <= wdata_reg[23:16];
						if (wstrb_reg[3]) xfer_len[31:24] <= wdata_reg[31:24];
					end
					STATUS_OFFSET: begin
						// Status is read-only, ignore writes
					end
					default: ;
				endcase
			end
		end
	end

	// DMA state machine
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			state <= S_IDLE;
			cur_src_addr <= 0;
			cur_dst_addr <= 0;
			remaining <= 0;
			burst_len <= 0;
			burst_count <= 0;
			read_offset <= 0;
			write_offset <= 0;
			rdata_buf <= 0;
			wstrb_buf <= 0;
			rdata_valid <= 0;
			wdata_valid <= 0;
			rdata_addr <= 0;
			wdata_addr <= 0;
			o_int <= 0;
		end else begin
			case (state)
				S_IDLE: begin
					if (ctrl_reg[0] && xfer_len > 0) begin
						state <= S_READ;
						cur_src_addr <= src_addr;
						cur_dst_addr <= dst_addr;
						remaining <= xfer_len;
						burst_len <= max_burst_len;
						burst_count <= 0;
						read_offset <= 0;
						write_offset <= 0;
						o_int <= 0;
					end
				end
				S_READ: begin
					if (M_AXI_ARVALID && M_AXI_ARREADY) begin
						if (burst_count == burst_len - 1) begin
							state <= S_WRITE;
							burst_count <= 0;
							read_offset <= 0;
						end else begin
							burst_count <= burst_count + 1;
						end
					end
				end
				S_WRITE: begin
					if (M_AXI_AWVALID && M_AXI_AWREADY) begin
						if (burst_count == burst_len - 1) begin
							if (remaining <= burst_len) begin
								state <= S_DONE;
								o_int <= 1;
							end else begin
								state <= S_READ;
								remaining <= remaining - burst_len;
								cur_src_addr <= cur_src_addr + burst_len * (C_AXI_DATA_WIDTH/8);
								cur_dst_addr <= cur_dst_addr + burst_len * (C_AXI_DATA_WIDTH/8);
								burst_len <= max_burst_len;
								burst_count <= 0;
								read_offset <= 0;
								write_offset <= 0;
							end
						end else begin
							burst_count <= burst_count + 1;
						end
					end
				end
				S_DONE: begin
					if (ctrl_reg[0] == 0) begin
						state <= S_IDLE;
						o_int <= 0;
					end
				end
				default: state <= S_IDLE;
			endcase
		end
	end

	// AXI Read Address Channel
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			M_AXI_ARVALID <= 0;
			M_AXI_ARADDR <= 0;
		end else begin
			if (state == S_READ && !M_AXI_ARVALID) begin
				M_AXI_ARVALID <= 1;
				M_AXI_ARADDR <= cur_src_addr + read_offset * (C_AXI_DATA_WIDTH/8);
			end else if (M_AXI_ARVALID && M_AXI_ARREADY) begin
				M_AXI_ARVALID <= 0;
			end
		end
	end

	assign M_AXI_ARID = AXI_READ_ID;
	assign M_AXI_ARSIZE = axi_size;
	assign M_AXI_ARBURST = axi_burst;
	assign M_AXI_ARLOCK = 1'b0;
	assign M_AXI_ARCACHE = 4'b0011;
	assign M_AXI_ARPROT = 3'b000;
	assign M_AXI_ARQOS = 4'b0000;

	// AXI Read Data Channel
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			M_AXI_RREADY <= 0;
			rdata_buf <= 0;
			rdata_valid <= 0;
			rdata_addr <= 0;
		end else begin
			if (M_AXI_RVALID) begin
				M_AXI_RREADY <= 1;
				rdata_buf <= M_AXI_RDATA;
				rdata_valid <= 1;
				rdata_addr <= cur_src_addr + read_offset * (C_AXI_DATA_WIDTH/8);
			end else begin
				M_AXI_RREADY <= 0;
				rdata_valid <= 0;
			end
		end
	end

	// AXI Write Address Channel
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			M_AXI_AWVALID <= 0;
			M_AXI_AWADDR <= 0;
		end else begin
			if (state == S_WRITE && !M_AXI_AWVALID) begin
				M_AXI_AWVALID <= 1;
				M_AXI_AWADDR <= cur_dst_addr + write_offset * (C_AXI_DATA_WIDTH/8);
			end else if (M_AXI_AWVALID && M_AXI_AWREADY) begin
				M_AXI_AWVALID <= 0;
			end
		end
	end

	assign M_AXI_AWID = AXI_WRITE_ID;
	assign M_AXI_AWSIZE = axi_size;
	assign M_AXI_AWBURST = axi_burst;
	assign M_AXI_AWLOCK = 1'b0;
	assign M_AXI_AWCACHE = 4'b0011;
	assign M_AXI_AWPROT = 3'b000;
	assign M_AXI_AWQOS = 4'b0000;

	// AXI Write Data Channel
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			M_AXI_WVALID <= 0;
			M_AXI_WDATA <= 0;
			M_AXI_WSTRB <= 0;
			M_AXI_WLAST <= 0;
			wdata_valid <= 0;
			wdata_addr <= 0;
		end else begin
			if (rdata_valid && !M_AXI_WVALID) begin
				M_AXI_WVALID <= 1;
				M_AXI_WDATA <= rdata_buf;
				M_AXI_WSTRB <= {C_AXI_DATA_WIDTH/8{1'b1}};
				M_AXI_WLAST <= (burst_count == burst_len - 1);
				wdata_valid <= 1;
				wdata_addr <= cur_dst_addr + write_offset * (C_AXI_DATA_WIDTH/8);
			end else if (M_AXI_WVALID && M_AXI_WREADY) begin
				M_AXI_WVALID <= 0;
				wdata_valid <= 0;
			end
		end
	end

	// AXI Write Response Channel
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			M_AXI_BREADY <= 0;
		end else begin
			if (M_AXI_BVALID) begin
				M_AXI_BREADY <= 1;
			end else begin
				M_AXI_BREADY <= 0;
			end
		end
	end

	// Update status register
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			status_reg <= 0;
		end else begin
			status_reg[0] <= (state != S_IDLE);
			status_reg[1] <= o_int;
			status_reg[2] <= 1'b0;
			status_reg[3] <= 1'b0;
			status_reg[31:4] <= 0;
		end
	end

endmodule
