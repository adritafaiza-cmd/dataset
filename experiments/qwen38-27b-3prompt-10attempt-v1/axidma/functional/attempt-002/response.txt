```verilog
module axidma #(
		parameter	C_AXI_ID_WIDTH = 1,
		parameter	C_AXI_ADDR_WIDTH = 32,
		parameter	C_AXI_DATA_WIDTH = 32,
		localparam	C_AXIL_ADDR_WIDTH = 5,
		localparam	C_AXIL_DATA_WIDTH = 32,
		parameter [0:0]	OPT_UNALIGNED = 1'b1,
		parameter [0:0]	OPT_WRAPMEM = 1'b1,
		parameter	LGFIFO = 4,
		parameter	LGLEN = C_AXI_ADDR_WIDTH,
		parameter [0:0]	OPT_LOWPOWER = 1'b0,
		parameter [0:0]	OPT_CLKGATE = OPT_LOWPOWER,
		parameter	[C_AXI_ID_WIDTH-1:0]	AXI_READ_ID = 0,
		parameter	[C_AXI_ID_WIDTH-1:0]	AXI_WRITE_ID = 0,
		parameter	[7:0]			ABORT_KEY  = 8'h6d,
		localparam	ADDRLSB= $clog2(C_AXI_DATA_WIDTH)-3,
		localparam	AXILLSB= $clog2(C_AXIL_DATA_WIDTH)-3,
		localparam	LGLENW= LGLEN-ADDRLSB
	) (
		input	wire	S_AXI_ACLK,
		input	wire	S_AXI_ARESETN,
		input	wire				S_AXIL_AWVALID,
		output	wire				S_AXIL_AWREADY,
		input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_AWADDR,
		input	wire 	[2:0]			S_AXIL_AWPROT,
		input	wire				S_AXIL_WVALID,
		output	wire				S_AXIL_WREADY,
		input	wire [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_WDATA,
		input	wire [C_AXIL_DATA_WIDTH/8-1:0]	S_AXIL_WSTRB,
		output	reg				S_AXIL_BVALID,
		input	wire				S_AXIL_BREADY,
		output	wire	[1:0]			S_AXIL_BRESP,
		input	wire				S_AXIL_ARVALID,
		output	wire				S_AXIL_ARREADY,
		input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_ARADDR,
		input	wire 	[2:0]			S_AXIL_ARPROT,
		output	reg				S_AXIL_RVALID,
		input	wire				S_AXIL_RREADY,
		output	reg [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_RDATA,
		output	wire	[1:0]			S_AXIL_RRESP,
		output	reg				M_AXI_AWVALID,
		input	wire				M_AXI_AWREADY,
		output	reg	[C_AXI_ID_WIDTH-1:0]	M_AXI_AWID,
		output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_AWADDR,
		output	reg	[2:0]			M_AXI_AWSIZE,
		output	reg	[1:0]			M_AXI_AWBURST,
		output	reg				M_AXI_AWLOCK,
		output	reg	[3:0]			M_AXI_AWCACHE,
		output	reg	[2:0]			M_AXI_AWPROT,
		output	reg	[3:0]			M_AXI_AWQOS,
		output	reg				M_AXI_WVALID,
		input	wire				M_AXI_WREADY,
		output	reg	[C_AXI_DATA_WIDTH-1:0]	M_AXI_WDATA,
		output	reg [C_AXI_DATA_WIDTH/8-1:0]	M_AXI_WSTRB,
		output	reg				M_AXI_WLAST,
		input	wire				M_AXI_BVALID,
		output	reg				M_AXI_BREADY,
		input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_BID,
		input	wire	[1:0]			M_AXI_BRESP,
		output	reg				M_AXI_ARVALID,
		input	wire				M_AXI_ARREADY,
		output	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_ARID,
		output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_ARADDR,
		output	wire	[2:0]			M_AXI_ARSIZE,
		output	wire	[1:0]			M_AXI_ARBURST,
		output	wire				M_AXI_ARLOCK,
		output	wire	[3:0]			M_AXI_ARCACHE,
		output	wire	[2:0]			M_AXI_ARPROT,
		output	wire	[3:0]			M_AXI_ARQOS,
		input	wire				M_AXI_RVALID,
		output	wire				M_AXI_RREADY,
		input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_RID,
		input	wire	[C_AXI_DATA_WIDTH-1:0]	M_AXI_RDATA,
		input	wire				M_AXI_RLAST,
		input	wire	[1:0]			M_AXI_RRESP,
		output	reg				o_int
	);

	// Register addresses
	localparam REG_SRC_ADDR = 0;
	localparam REG_DST_ADDR = 1;
	localparam REG_LENGTH   = 2;
	localparam REG_CONTROL  = 3;
	localparam REG_STATUS   = 4;
	localparam REG_ABORT    = 5;

	// Control register bits
	localparam CTRL_START = 0;
	localparam CTRL_ABORT = 1;

	// Status register bits
	localparam STATUS_BUSY = 0;
	localparam STATUS_DONE = 1;
	localparam STATUS_ERROR = 2;

	// Internal signals
	reg [C_AXI_ADDR_WIDTH-1:0] src_addr_reg;
	reg [C_AXI_ADDR_WIDTH-1:0] dst_addr_reg;
	reg [LGLEN-1:0] length_reg;
	reg [31:0] control_reg;
	reg [31:0] status_reg;

	reg [C_AXI_ADDR_WIDTH-1:0] cur_src_addr;
	reg [C_AXI_ADDR_WIDTH-1:0] cur_dst_addr;
	reg [LGLEN-1:0] remaining_len;

	reg dma_active;
	reg dma_done;
	reg dma_error;

	// AXI write channel state
	reg aw_valid;
	reg w_valid;
	reg w_last;
	reg b_ready;
	reg b_valid;
	reg [C_AXI_DATA_WIDTH-1:0] w_data_reg;
	reg [C_AXI_DATA_WIDTH/8-1:0] w_strb_reg;
	reg [C_AXI_ADDR_WIDTH-1:0] aw_addr_reg;
	reg [2:0] aw_size_reg;
	reg [1:0] aw_burst_reg;
	reg [C_AXI_ID_WIDTH-1:0] aw_id_reg;
	reg [2:0] aw_prot_reg;

	// AXI read channel state
	reg ar_valid;
	reg r_ready;
	reg r_valid;
	reg [C_AXI_ADDR_WIDTH-1:0] ar_addr_reg;
	reg [2:0] ar_size_reg;
	reg [1:0] ar_burst_reg;
	reg [C_AXI_ID_WIDTH-1:0] ar_id_reg;
	reg [2:0] ar_prot_reg;

	// FIFO for read data
	localparam FIFO_DEPTH = 1 << LGFIFO;
	localparam FIFO_PTR_WIDTH = LGFIFO;
	reg [C_AXI_DATA_WIDTH-1:0] fifo_mem [0:FIFO_DEPTH-1];
	reg [FIFO_PTR_WIDTH-1:0] fifo_wr_ptr;
	reg [FIFO_PTR_WIDTH-1:0] fifo_rd_ptr;
	reg fifo_full;
	reg fifo_empty;

	// AXI Lite write address channel
	reg aw_ready;
	reg w_ready;
	reg b_valid_reg;
	reg [1:0] b_resp_reg;

	// AXI Lite read address channel
	reg ar_ready;
	reg r_valid_reg;
	reg [C_AXIL_DATA_WIDTH-1:0] r_data_reg;
	reg [1:0] r_resp_reg;

	// Write strobe generation
	wire [C_AXI_DATA_WIDTH/8-1:0] full_strb = {C_AXI_DATA_WIDTH/8{1'b1}};

	// Calculate number of beats for current burst
	// For simplicity, we do single-beat transfers or handle bursts based on alignment
	// Here we assume aligned transfers for simplicity, but handle unaligned if OPT_UNALIGNED
	// For a robust implementation, we would need to handle partial beats.
	// Let's assume aligned 32-bit transfers for the core logic, with length in bytes.
	
	// State machine for DMA
	localparam S_IDLE = 0;
	localparam S_READ_ADDR = 1;
	localparam S_READ_DATA = 2;
	localparam S_WRITE_ADDR = 3;
	localparam S_WRITE_DATA = 4;
	localparam S_WRITE_RESP = 5;
	localparam S_DONE = 6;

	reg [2:0] state;

	// AXI Lite write address decode
	wire [C_AXIL_ADDR_WIDTH-1:0] aw_addr = S_AXIL_AWADDR;
	wire [C_AXIL_ADDR_WIDTH-1:0] ar_addr = S_AXIL_ARADDR;

	// AXI Lite write data
	wire [C_AXIL_DATA_WIDTH-1:0] w_data = S_AXIL_WDATA;
	wire [C_AXIL_DATA_WIDTH/8-1:0] w_strb = S_AXIL_WSTRB;

	// AXI Lite write address ready
	assign S_AXIL_AWREADY = aw_ready;
	assign S_AXIL_WREADY = w_ready;
	assign S_AXIL_BVALID = b_valid_reg;
	assign S_AXIL_BRESP = b_resp_reg;
	assign S_AXIL_ARREADY = ar_ready;
	assign S_AXIL_RVALID = r_valid_reg;
	assign S_AXIL_RDATA = r_data_reg;
	assign S_AXIL_RRESP = r_resp_reg;

	// AXI Master write address
	assign M_AXI_AWVALID = aw_valid;
	assign M_AXI_AWID = aw_id_reg;
	assign M_AXI_AWADDR = aw_addr_reg;
	assign M_AXI_AWSIZE = aw_size_reg;
	assign M_AXI_AWBURST = aw_burst_reg;
	assign M_AXI_AWLOCK = 1'b0;
	assign M_AXI_AWCACHE = 4'b0000;
	assign M_AXI_AWPROT = aw_prot_reg;
	assign M_AXI_AWQOS = 4'b0000;

	// AXI Master write data
	assign M_AXI_WVALID = w_valid;
	assign M_AXI_WDATA = w_data_reg;
	assign M_AXI_WSTRB = w_strb_reg;
	assign M_AXI_WLAST = w_last;
	assign M_AXI_BREADY = b_ready;

	// AXI Master read address
	assign M_AXI_ARVALID = ar_valid;
	assign M_AXI_ARID = ar_id_reg;
	assign M_AXI_ARADDR = ar_addr_reg;
	assign M_AXI_ARSIZE = ar_size_reg;
	assign M_AXI_ARBURST = ar_burst_reg;
	assign M_AXI_ARLOCK = 1'b0;
	assign M_AXI_ARCACHE = 4'b0000;
	assign M_AXI_ARPROT = ar_prot_reg;
	assign M_AXI_ARQOS = 4'b0000;

	// AXI Master read data
	assign M_AXI_RREADY = r_ready;

	// Interrupt
	assign o_int = dma_done | dma_error;

	// FIFO logic
	wire fifo_wr_en = r_valid && M_AXI_RVALID && !fifo_full;
	wire fifo_rd_en = !fifo_empty && (state == S_WRITE_DATA || state == S_WRITE_ADDR);

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			fifo_wr_ptr <= 0;
			fifo_rd_ptr <= 0;
			fifo_full <= 1'b0;
			fifo_empty <= 1'b1;
		end else begin
			if (fifo_wr_en && !fifo_rd_en) begin
				fifo_wr_ptr <= fifo_wr_ptr + 1;
				if (fifo_wr_ptr == fifo_rd_ptr) begin
					fifo_full <= 1'b1;
					fifo_empty <= 1'b0;
				end
			end else if (!fifo_wr_en && fifo_rd_en) begin
				fifo_rd_ptr <= fifo_rd_ptr + 1;
				if (fifo_rd_ptr == fifo_wr_ptr) begin
					fifo_empty <= 1'b1;
					fifo_full <= 1'b0;
				end
			end else if (fifo_wr_en && fifo_rd_en) begin
				// Both write and read, pointers advance
				fifo_wr_ptr <= fifo_wr_ptr + 1;
				fifo_rd_ptr <= fifo_rd_ptr + 1;
				// Full/empty status remains unchanged
			end
		end
	end

	// FIFO write
	always @(posedge S_AXI_ACLK) begin
		if (fifo_wr_en) begin
			fifo_mem[fifo_wr_ptr] <= M_AXI_RDATA;
		end
	end

	// AXI Lite register write logic
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			aw_ready <= 1'b0;
			w_ready <= 1'b0;
			b_valid_reg <= 1'b0;
			b_resp_reg <= 2'b00;
			ar_ready <= 1'b0;
			r_valid_reg <= 1'b0;
			r_data_reg <= 0;
			r_resp_reg <= 2'b00;
			src_addr_reg <= 0;
			dst_addr_reg <= 0;
			length_reg <= 0;
			control_reg <= 0;
			status_reg <= 0;
		end else begin
			// Default: not ready
			aw_ready <= 1'b0;
			w_ready <= 1'b0;
			ar_ready <= 1'b0;
			r_valid_reg <= 1'b0;

			// Handle write address
			if (S_AXIL_AWVALID && !aw_ready) begin
				aw_ready <= 1'b1;
			end

			// Handle write data
			if (aw_ready && S_AXIL_WVALID) begin
				w_ready <= 1'b1;
				case (aw_addr[4:2])
					REG_SRC_ADDR: src_addr_reg <= w_data;
					REG_DST_ADDR: dst_addr_reg <= w_data;
					REG_LENGTH:   length_reg <= w_data[LGLEN-1:0];
					REG_CONTROL:  control_reg <= w_data;
					REG_ABORT:    begin
						if (w_data[7:0] == ABORT_KEY) begin
							control_reg[CTRL_ABORT] <= 1'b1;
						end
					end
					default: ;
				endcase
				aw_ready <= 1'b0;
			end

			// Handle write response
			if (w_ready && S_AXIL_BREADY) begin
				b_valid_reg <= 1'b1;
				b_resp_reg <= 2'b00;
				w_ready <= 1'b0;
			end

			if (b_valid_reg && S_AXIL_BREADY) begin
				b_valid_reg <= 1'b0;
			end

			// Handle read address
			if (S_AXIL_ARVALID && !ar_ready) begin
				ar_ready <= 1'b1;
			end

			// Handle read data
			if (ar_ready) begin
				case (ar_addr[4:2])
					REG_SRC_ADDR: r_data_reg <= src_addr_reg;
					REG_DST_ADDR: r_data_reg <= dst_addr_reg;
					REG_LENGTH:   r_data_reg <= {16'b0, length_reg};
					REG_CONTROL:  r_data_reg <= control_reg;
					REG_STATUS:   r_data_reg <= status_reg;
					default:      r_data_reg <= 0;
				endcase
				r_valid_reg <= 1'b1;
				r_resp_reg <= 2'b00;
				ar_ready <= 1'b0;
			end

			if (r_valid_reg && S_AXIL_RREADY) begin
				r_valid_reg <= 1'b0;
			end
		end
	end

	// DMA state machine
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			state <= S_IDLE;
			dma_active <= 1'b0;
			dma_done <= 1'b0;
			dma_error <= 1'b0;
			cur_src_addr <= 0;
			cur_dst_addr <= 0;
			remaining_len <= 0;
			aw_valid <= 1'b0;
			w_valid <= 1'b0;
			w_last <= 1'b0;
			b_ready <= 1'b0;
			ar_valid <= 1'b0;
			r_ready <= 1'b0;
			aw_addr_reg <= 0;
			aw_size_reg <= 0;
			aw_burst_reg <= 0;
			aw_id_reg <= 0;
			aw_prot_reg <= 0;
			ar_addr_reg <= 0;
			ar_size_reg <= 0;
			ar_burst_reg <= 0;
			ar_id_reg <= 0;
			ar_prot_reg <= 0;
			w_data_reg <= 0;
			w_strb_reg <= 0;
			status_reg <= 0;
		end else begin
			// Default: deassert valids
			aw_valid <= 1'b0;
			w_valid <= 1'b0;
			ar_valid <= 1'b0;
			b_ready <= 1'b0;
			r_ready <= 1'b0;

			// Check for abort
			if (control_reg[CTRL_ABORT]) begin
				state <= S_IDLE;
				dma_active <= 1'b0;
				dma_done <= 1'b0;
				dma_error <= 1'b1;
				status_reg[STATUS_ERROR] <= 1'b1;
				status_reg[STATUS_BUSY] <= 1'b0;
				control_reg[CTRL_ABORT] <= 1'b0;
			end else begin
				case (state)
					S_IDLE: begin
						if (control_reg[CTRL_START] && !dma_active) begin
							dma_active <= 1'b1;
							dma_done <= 1'b0;
							dma_error <= 1'b0;
							cur_src_addr <= src_addr_reg;
							cur_dst_addr <= dst_addr_reg;
							remaining_len <= length_reg;
							status_reg[STATUS_BUSY] <= 1'b1;
							status_reg[STATUS_DONE] <= 1'b0;
							status_reg[STATUS_ERROR] <= 1'b0;
							control_reg[CTRL_START] <= 1'b0;
							state <= S_READ_ADDR;
						end
					end

					S_READ_ADDR: begin
						if (remaining_len > 0) begin
							// Issue read address
							ar_valid <= 1'b1;
							ar_addr_reg <= cur_src_addr;
							ar_size_reg <= ADDRLSB; // 32-bit transfer
							ar_burst_reg <= 2'b01; // INCR
							ar_id_reg <= AXI_READ_ID;
							ar_prot_reg <= 3'b000;
							state <= S_READ_DATA;
						end else begin
							state <= S_DONE;
						end
					end

					S_READ_DATA: begin
						if (ar_valid && M_AXI_ARREADY) begin
							ar_valid <= 1'b0;
							r_ready <= 1'b1;
						end
						if (r_ready && M_AXI_RVALID) begin
							r_ready <= 1'b0;
							// Data is written to FIFO by separate logic
							// Check if we have enough data to write
							if (!fifo_empty) begin
								state <= S_WRITE_ADDR;
							end
						end
					end

					S_WRITE_ADDR: begin
						if (!fifo_empty) begin
							// Issue write address
							aw_valid <= 1'b1;
							aw_addr_reg <= cur_dst_addr;
							aw_size_reg <= ADDRLSB; // 32-bit transfer
							aw_burst_reg <= 2'b01; // INCR
							aw_id_reg <= AXI_WRITE_ID;
							aw_prot_reg <= 3'b000;
							state <= S_WRITE_DATA;
						end
					end

					S_WRITE_DATA: begin
						if (aw_valid && M_AXI_AWREADY) begin
							aw_valid <= 1'b0;
							w_valid <= 1'b1;
							w_data_reg <= fifo_mem[fifo_rd_ptr];
							w_strb_reg <= full_strb;
							w_last <= 1'b1; // Single beat burst
						end
						if (w_valid && M_AXI_WREADY) begin
							w_valid <= 1'b0;
							b_ready <= 1'b1;
							state <= S_WRITE_RESP;
						end
					end

					S_WRITE_RESP: begin
						if (b_ready && M_AXI_BVALID) begin
							b_ready <= 1'b0;
							// Update addresses and length
							cur_src_addr <= cur_src_addr + C_AXI_DATA_WIDTH/8;
							cur_dst_addr <= cur_dst_addr + C_AXI_DATA_WIDTH/8;
							remaining_len <= remaining_len - (C_AXI_DATA_WIDTH/8);
							state <= S_READ_ADDR;
						end
					end

					S_DONE: begin
						dma_active <= 1'b0;
						dma_done <= 1'b1;
						status_reg[STATUS_BUSY] <= 1'b0;
						status_reg[STATUS_DONE] <= 1'b1;
						state <= S_IDLE;
					end

					default: state <= S_IDLE;
				endcase
			end
		end
	end

	// Update status register
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			status_reg <= 0;
		end else begin
			status_reg[STATUS_BUSY] <= dma_active;
			status_reg[STATUS_DONE] <= dma_done;
			status_reg[STATUS_ERROR] <= dma_error;
		end
	end

endmodule
