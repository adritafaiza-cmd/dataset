module axidma #(
		parameter	C_AXI_ID_WIDTH = 1,
		parameter	C_AXI_ADDR_WIDTH = 32,
		parameter	C_AXI_DATA_WIDTH = 32,
		localparam	C_AXIL_ADDR_WIDTH = 5,
		localparam	C_AXIL_DATA_WIDTH = 32,
		parameter [0:0]	OPT_UNALIGNED = 1'b1,
		parameter [0:0]	OPT_WRAPMEM = 1'b1,
		parameter	LGFIFO = 4,
		parameter	LGLEN = C_AXI_ADDR_WIDTH,
		parameter [0:0]	OPT_LOWPOWER = 1'b0,
		parameter [0:0]	OPT_CLKGATE = OPT_LOWPOWER,
		parameter	[C_AXI_ID_WIDTH-1:0]	AXI_READ_ID = 0,
		parameter	[C_AXI_ID_WIDTH-1:0]	AXI_WRITE_ID = 0,
		parameter	[7:0]			ABORT_KEY  = 8'h6d,
		localparam	ADDRLSB= $clog2(C_AXI_DATA_WIDTH)-3,
		localparam	AXILLSB= $clog2(C_AXIL_DATA_WIDTH)-3,
		localparam	LGLENW= LGLEN-ADDRLSB
	) (
		input	wire	S_AXI_ACLK,
		input	wire	S_AXI_ARESETN,
		input	wire				S_AXIL_AWVALID,
		output	wire				S_AXIL_AWREADY,
		input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_AWADDR,
		input	wire 	[2:0]			S_AXIL_AWPROT,
		input	wire				S_AXIL_WVALID,
		output	wire				S_AXIL_WREADY,
		input	wire [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_WDATA,
		input	wire [C_AXIL_DATA_WIDTH/8-1:0]	S_AXIL_WSTRB,
		output	reg				S_AXIL_BVALID,
		input	wire				S_AXIL_BREADY,
		output	wire	[1:0]			S_AXIL_BRESP,
		input	wire				S_AXIL_ARVALID,
		output	wire				S_AXIL_ARREADY,
		input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_ARADDR,
		input	wire 	[2:0]			S_AXIL_ARPROT,
		output	reg				S_AXIL_RVALID,
		input	wire				S_AXIL_RREADY,
		output	reg [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_RDATA,
		output	wire	[1:0]			S_AXIL_RRESP,
		output	reg				M_AXI_AWVALID,
		input	wire				M_AXI_AWREADY,
		output	reg	[C_AXI_ID_WIDTH-1:0]	M_AXI_AWID,
		output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_AWADDR,
		output	reg	[2:0]			M_AXI_AWSIZE,
		output	reg	[1:0]			M_AXI_AWBURST,
		output	reg				M_AXI_AWLOCK,
		output	reg	[3:0]			M_AXI_AWCACHE,
		output	reg	[2:0]			M_AXI_AWPROT,
		output	reg	[3:0]			M_AXI_AWQOS,
		output	reg				M_AXI_WVALID,
		input	wire				M_AXI_WREADY,
		output	reg	[C_AXI_DATA_WIDTH-1:0]	M_AXI_WDATA,
		output	reg [C_AXI_DATA_WIDTH/8-1:0]	M_AXI_WSTRB,
		output	reg				M_AXI_WLAST,
		input	wire				M_AXI_BVALID,
		output	reg				M_AXI_BREADY,
		input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_BID,
		input	wire	[1:0]			M_AXI_BRESP,
		output	reg				M_AXI_ARVALID,
		input	wire				M_AXI_ARREADY,
		output	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_ARID,
		output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_ARADDR,
		output	wire	[2:0]			M_AXI_ARSIZE,
		output	wire	[1:0]			M_AXI_ARBURST,
		output	wire				M_AXI_ARLOCK,
		output	wire	[3:0]			M_AXI_ARCACHE,
		output	wire	[2:0]			M_AXI_ARPROT,
		output	wire	[3:0]			M_AXI_ARQOS,
		input	wire				M_AXI_RVALID,
		output	wire				M_AXI_RREADY,
		input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_RID,
		input	wire	[C_AXI_DATA_WIDTH-1:0]	M_AXI_RDATA,
		input	wire				M_AXI_RLAST,
		input	wire	[1:0]			M_AXI_RRESP,
		output	reg				o_int
	);

	// Register addresses
	localparam REG_SRC_ADDR = 4'h0;
	localparam REG_DST_ADDR = 4'h4;
	localparam REG_LENGTH   = 4'h8;
	localparam REG_CONTROL  = 4'hC;
	localparam REG_STATUS   = 4'h10;
	localparam REG_DOORBELL = 4'h14;

	// Control register bits
	localparam CTRL_START = 0;
	localparam CTRL_ABORT = 1;

	// Status register bits
	localparam STATUS_BUSY  = 0;
	localparam STATUS_DONE  = 1;
	localparam STATUS_ERROR = 2;

	// Internal signals
	reg [C_AXI_ADDR_WIDTH-1:0] src_addr_reg;
	reg [C_AXI_ADDR_WIDTH-1:0] dst_addr_reg;
	reg [LGLEN-1:0]            length_reg;
	reg [1:0]                  control_reg;
	reg [2:0]                  status_reg;

	reg [C_AXI_ADDR_WIDTH-1:0] src_addr_cur;
	reg [C_AXI_ADDR_WIDTH-1:0] dst_addr_cur;
	reg [LGLEN-1:0]            length_cur;

	reg                        dma_active;
	reg                        dma_done;
	reg                        dma_error;

	// AXI4-Lite write address channel
	reg                        awaddr_valid;
	reg [C_AXIL_ADDR_WIDTH-1:0] awaddr_reg;
	reg [2:0]                  awprot_reg;

	// AXI4-Lite write data channel
	reg                        wdata_valid;
	reg [C_AXIL_DATA_WIDTH-1:0] wdata_reg;
	reg [C_AXIL_DATA_WIDTH/8-1:0] wstrb_reg;

	// AXI4-Lite read address channel
	reg                        araddr_valid;
	reg [C_AXIL_ADDR_WIDTH-1:0] araddr_reg;
	reg [2:0]                  arprot_reg;

	// AXI4-Lite read data channel
	reg                        rdata_valid;
	reg [C_AXIL_DATA_WIDTH-1:0] rdata_reg;
	reg [1:0]                  rresp_reg;

	// DMA state machine
	localparam S_IDLE = 2'b00;
	localparam S_READ = 2'b01;
	localparam S_WRITE = 2'b10;
	localparam S_DONE = 2'b11;

	reg [1:0] dma_state;
	reg [1:0] next_dma_state;

	// AXI master write address channel
	reg                        m_aw_valid;
	reg [C_AXI_ID_WIDTH-1:0]  m_aw_id;
	reg [C_AXI_ADDR_WIDTH-1:0] m_aw_addr;
	reg [2:0]                  m_aw_size;
	reg [1:0]                  m_aw_burst;
	reg                        m_aw_lock;
	reg [3:0]                  m_aw_cache;
	reg [2:0]                  m_aw_prot;
	reg [3:0]                  m_aw_qos;

	// AXI master write data channel
	reg                        m_w_valid;
	reg [C_AXI_DATA_WIDTH-1:0] m_w_data;
	reg [C_AXI_DATA_WIDTH/8-1:0] m_w_strb;
	reg                        m_w_last;

	// AXI master write response channel
	reg                        m_b_ready;

	// AXI master read address channel
	reg                        m_ar_valid;
	reg [C_AXI_ID_WIDTH-1:0]  m_ar_id;
	reg [C_AXI_ADDR_WIDTH-1:0] m_ar_addr;
	reg [2:0]                  m_ar_size;
	reg [1:0]                  m_ar_burst;
	reg                        m_ar_lock;
	reg [3:0]                  m_ar_cache;
	reg [2:0]                  m_ar_prot;
	reg [3:0]                  m_ar_qos;

	// AXI master read data channel
	reg                        m_r_ready;

	// FIFO for read data
	reg [C_AXI_DATA_WIDTH-1:0] fifo_mem [0:(1<<LGFIFO)-1];
	reg [LGFIFO-1:0] fifo_wr_ptr;
	reg [LGFIFO-1:0] fifo_rd_ptr;
	reg [LGFIFO:0] fifo_count;

	wire fifo_full = (fifo_count == (1<<LGFIFO));
	wire fifo_empty = (fifo_count == 0);

	// AXI4-Lite write address ready
	assign S_AXIL_AWREADY = !awaddr_valid;

	// AXI4-Lite write data ready
	assign S_AXIL_WREADY = !wdata_valid;

	// AXI4-Lite write response
	assign S_AXIL_BRESP = 2'b00;

	// AXI4-Lite read address ready
	assign S_AXIL_ARREADY = !araddr_valid;

	// AXI4-Lite read response
	assign S_AXIL_RRESP = rresp_reg;

	// AXI master write address
	assign M_AXI_AWVALID = m_aw_valid;
	assign M_AXI_AWID = m_aw_id;
	assign M_AXI_AWADDR = m_aw_addr;
	assign M_AXI_AWSIZE = m_aw_size;
	assign M_AXI_AWBURST = m_aw_burst;
	assign M_AXI_AWLOCK = m_aw_lock;
	assign M_AXI_AWCACHE = m_aw_cache;
	assign M_AXI_AWPROT = m_aw_prot;
	assign M_AXI_AWQOS = m_aw_qos;

	// AXI master write data
	assign M_AXI_WVALID = m_w_valid;
	assign M_AXI_WDATA = m_w_data;
	assign M_AXI_WSTRB = m_w_strb;
	assign M_AXI_WLAST = m_w_last;

	// AXI master write response
	assign M_AXI_BREADY = m_b_ready;

	// AXI master read address
	assign M_AXI_ARVALID = m_ar_valid;
	assign M_AXI_ARID = m_ar_id;
	assign M_AXI_ARADDR = m_ar_addr;
	assign M_AXI_ARSIZE = m_ar_size;
	assign M_AXI_ARBURST = m_ar_burst;
	assign M_AXI_ARLOCK = m_ar_lock;
	assign M_AXI_ARCACHE = m_ar_cache;
	assign M_AXI_ARPROT = m_ar_prot;
	assign M_AXI_ARQOS = m_ar_qos;

	// AXI master read data
	assign M_AXI_RREADY = m_r_ready;

	// Interrupt
	assign o_int = dma_done | dma_error;

	// AXI4-Lite write address handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			awaddr_valid <= 1'b0;
			awaddr_reg <= {C_AXIL_ADDR_WIDTH{1'b0}};
			awprot_reg <= 3'b000;
		end else begin
			if (S_AXIL_AWVALID && S_AXIL_AWREADY) begin
				awaddr_valid <= 1'b1;
				awaddr_reg <= S_AXIL_AWADDR;
				awprot_reg <= S_AXIL_AWPROT;
			end else if (awaddr_valid && wdata_valid && S_AXIL_WVALID && S_AXIL_WREADY) begin
				awaddr_valid <= 1'b0;
			end
		end
	end

	// AXI4-Lite write data handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			wdata_valid <= 1'b0;
			wdata_reg <= {C_AXIL_DATA_WIDTH{1'b0}};
			wstrb_reg <= {C_AXIL_DATA_WIDTH/8{1'b0}};
		end else begin
			if (S_AXIL_WVALID && S_AXIL_WREADY) begin
				wdata_valid <= 1'b1;
				wdata_reg <= S_AXIL_WDATA;
				wstrb_reg <= S_AXIL_WSTRB;
			end else if (awaddr_valid && wdata_valid && S_AXIL_WVALID && S_AXIL_WREADY) begin
				wdata_valid <= 1'b0;
			end
		end
	end

	// AXI4-Lite write response
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			S_AXIL_BVALID <= 1'b0;
		end else begin
			if (awaddr_valid && wdata_valid && S_AXIL_WVALID && S_AXIL_WREADY) begin
				S_AXIL_BVALID <= 1'b1;
			end else if (S_AXIL_BVALID && S_AXIL_BREADY) begin
				S_AXIL_BVALID <= 1'b0;
			end
		end
	end

	// AXI4-Lite read address handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			araddr_valid <= 1'b0;
			araddr_reg <= {C_AXIL_ADDR_WIDTH{1'b0}};
			arprot_reg <= 3'b000;
		end else begin
			if (S_AXIL_ARVALID && S_AXIL_ARREADY) begin
				araddr_valid <= 1'b1;
				araddr_reg <= S_AXIL_ARADDR;
				arprot_reg <= S_AXIL_ARPROT;
			end else if (araddr_valid && rdata_valid && S_AXIL_RVALID && S_AXIL_RREADY) begin
				araddr_valid <= 1'b0;
			end
		end
	end

	// AXI4-Lite read data handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			rdata_valid <= 1'b0;
			rdata_reg <= {C_AXIL_DATA_WIDTH{1'b0}};
			rresp_reg <= 2'b00;
		end else begin
			if (araddr_valid && rdata_valid && S_AXIL_RVALID && S_AXIL_RREADY) begin
				rdata_valid <= 1'b0;
			end
		end
	end

	// AXI4-Lite read response
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			S_AXIL_RVALID <= 1'b0;
			S_AXIL_RDATA <= {C_AXIL_DATA_WIDTH{1'b0}};
		end else begin
			if (araddr_valid && rdata_valid && S_AXIL_RVALID && S_AXIL_RREADY) begin
				S_AXIL_RVALID <= 1'b0;
			end
		end
	end

	// Register write logic
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			src_addr_reg <= {C_AXI_ADDR_WIDTH{1'b0}};
			dst_addr_reg <= {C_AXI_ADDR_WIDTH{1'b0}};
			length_reg <= {LGLEN{1'b0}};
			control_reg <= 2'b00;
			status_reg <= 3'b000;
		end else begin
			if (awaddr_valid && wdata_valid && S_AXIL_WVALID && S_AXIL_WREADY) begin
				case (awaddr_reg)
					REG_SRC_ADDR: src_addr_reg <= wdata_reg;
					REG_DST_ADDR: dst_addr_reg <= wdata_reg;
					REG_LENGTH:   length_reg <= wdata_reg[LGLEN-1:0];
					REG_CONTROL:  control_reg <= wdata_reg[1:0];
					default: ;
				endcase
			end
		end
	end

	// Register read logic
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			S_AXIL_RDATA <= {C_AXIL_DATA_WIDTH{1'b0}};
		end else begin
			if (araddr_valid && S_AXIL_ARVALID && S_AXIL_ARREADY) begin
				case (araddr_reg)
					REG_SRC_ADDR: S_AXIL_RDATA <= src_addr_reg;
					REG_DST_ADDR: S_AXIL_RDATA <= dst_addr_reg;
					REG_LENGTH:   S_AXIL_RDATA <= length_reg;
					REG_CONTROL:  S_AXIL_RDATA <= {30'b0, control_reg};
					REG_STATUS:   S_AXIL_RDATA <= {29'b0, status_reg};
					default:      S_AXIL_RDATA <= {C_AXIL_DATA_WIDTH{1'b0}};
				endcase
			end
		end
	end

	// DMA state machine
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			dma_state <= S_IDLE;
			dma_active <= 1'b0;
			dma_done <= 1'b0;
			dma_error <= 1'b0;
			src_addr_cur <= {C_AXI_ADDR_WIDTH{1'b0}};
			dst_addr_cur <= {C_AXI_ADDR_WIDTH{1'b0}};
			length_cur <= {LGLEN{1'b0}};
		end else begin
			// Start DMA
			if (control_reg[CTRL_START] && !dma_active) begin
				dma_active <= 1'b1;
				dma_done <= 1'b0;
				dma_error <= 1'b0;
				src_addr_cur <= src_addr_reg;
				dst_addr_cur <= dst_addr_reg;
				length_cur <= length_reg;
				dma_state <= S_READ;
			end

			// Abort DMA
			if (control_reg[CTRL_ABORT] && dma_active) begin
				dma_active <= 1'b0;
				dma_error <= 1'b1;
				dma_state <= S_IDLE;
			end

			// State machine
			if (dma_active) begin
				case (dma_state)
					S_READ: begin
						if (length_cur > 0) begin
							// Issue read
							m_ar_valid <= 1'b1;
							m_ar_id <= AXI_READ_ID;
							m_ar_addr <= src_addr_cur;
							m_ar_size <= $clog2(C_AXI_DATA_WIDTH)-3;
							m_ar_burst <= 2'b01; // INCR
							m_ar_lock <= 1'b0;
							m_ar_cache <= 4'b0011;
							m_ar_prot <= 3'b000;
							m_ar_qos <= 4'b0000;
						end else begin
							dma_state <= S_WRITE;
						end
					end
					S_WRITE: begin
						if (!fifo_empty) begin
							// Issue write
							m_aw_valid <= 1'b1;
							m_aw_id <= AXI_WRITE_ID;
							m_aw_addr <= dst_addr_cur;
							m_aw_size <= $clog2(C_AXI_DATA_WIDTH)-3;
							m_aw_burst <= 2'b01; // INCR
							m_aw_lock <= 1'b0;
							m_aw_cache <= 4'b0011;
							m_aw_prot <= 3'b000;
							m_aw_qos <= 4'b0000;
						end else if (length_cur == 0) begin
							dma_state <= S_DONE;
						end
					end
					S_DONE: begin
						dma_active <= 1'b0;
						dma_done <= 1'b1;
						dma_state <= S_IDLE;
					end
					default: dma_state <= S_IDLE;
				endcase
			end
		end
	end

	// AXI master read address handshake
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			m_ar_valid <= 1'b0;
		end else begin
			if (m_ar_valid && M_AXI_ARREADY) begin
				m_ar_valid <= 1'b0;
			end
		end
	end

	// AXI master read data handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			m_r_ready <= 1'b0;
			fifo_wr_ptr <= {LGFIFO{1'b0}};
			fifo_rd_ptr <= {LGFIFO{1'b0}};
			fifo_count <= {(LGFIFO+1){1'b0}};
		end else begin
			// Read data ready
			m_r_ready <= !fifo_full;

			// FIFO write
			if (M_AXI_RVALID && M_AXI_RREADY) begin
				fifo_mem[fifo_wr_ptr] <= M_AXI_RDATA;
				fifo_wr_ptr <= fifo_wr_ptr + 1'b1;
				fifo_count <= fifo_count + 1'b1;
			end

			// FIFO read
			if (!fifo_empty && m_w_valid && M_AXI_WREADY) begin
				fifo_rd_ptr <= fifo_rd_ptr + 1'b1;
				fifo_count <= fifo_count - 1'b1;
			end
		end
	end

	// AXI master write address handshake
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			m_aw_valid <= 1'b0;
		end else begin
			if (m_aw_valid && M_AXI_AWREADY) begin
				m_aw_valid <= 1'b0;
			end
		end
	end

	// AXI master write data handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			m_w_valid <= 1'b0;
			m_w_data <= {C_AXI_DATA_WIDTH{1'b0}};
			m_w_strb <= {C_AXI_DATA_WIDTH/8{1'b0}};
			m_w_last <= 1'b0;
		end else begin
			// Write data valid
			m_w_valid <= !fifo_empty;

			// Write data
			if (!fifo_empty) begin
				m_w_data <= fifo_mem[fifo_rd_ptr];
				m_w_strb <= {C_AXI_DATA_WIDTH/8{1'b1}};
				m_w_last <= (length_cur == 1);
			end

			// Update addresses and length
			if (m_w_valid && M_AXI_WREADY) begin
				dst_addr_cur <= dst_addr_cur + C_AXI_DATA_WIDTH/8;
				length_cur <= length_cur - 1'b1;
			end
		end
	end

	// AXI master write response handling
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			m_b_ready <= 1'b0;
		end else begin
			m_b_ready <= 1'b1;
		end
	end

	// Update source address
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			src_addr_cur <= {C_AXI_ADDR_WIDTH{1'b0}};
		end else begin
			if (M_AXI_RVALID && M_AXI_RREADY) begin
				src_addr_cur <= src_addr_cur + C_AXI_DATA_WIDTH/8;
			end
		end
	end

endmodule
