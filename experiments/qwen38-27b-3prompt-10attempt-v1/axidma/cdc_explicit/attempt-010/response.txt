```verilog
module axidma #(
		parameter	C_AXI_ID_WIDTH = 1,
		parameter	C_AXI_ADDR_WIDTH = 32,
		parameter	C_AXI_DATA_WIDTH = 32,
		localparam	C_AXIL_ADDR_WIDTH = 5,
		localparam	C_AXIL_DATA_WIDTH = 32,
		parameter [0:0]	OPT_UNALIGNED = 1'b1,
		parameter [0:0]	OPT_WRAPMEM = 1'b1,
		parameter	LGFIFO = 4,
		parameter	LGLEN = C_AXI_ADDR_WIDTH,
		parameter [0:0]	OPT_LOWPOWER = 1'b0,
		parameter [0:0]	OPT_CLKGATE = OPT_LOWPOWER,
		parameter	[C_AXI_ID_WIDTH-1:0]	AXI_READ_ID = 0,
		parameter	[C_AXI_ID_WIDTH-1:0]	AXI_WRITE_ID = 0,
		parameter	[7:0]			ABORT_KEY  = 8'h6d,
		localparam	ADDRLSB= $clog2(C_AXI_DATA_WIDTH)-3,
		localparam	AXILLSB= $clog2(C_AXIL_DATA_WIDTH)-3,
		localparam	LGLENW= LGLEN-ADDRLSB
	) (
		input	wire	S_AXI_ACLK,
		input	wire	S_AXI_ARESETN,
		input	wire				S_AXIL_AWVALID,
		output	wire				S_AXIL_AWREADY,
		input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_AWADDR,
		input	wire 	[2:0]			S_AXIL_AWPROT,
		input	wire				S_AXIL_WVALID,
		output	wire				S_AXIL_WREADY,
		input	wire [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_WDATA,
		input	wire [C_AXIL_DATA_WIDTH/8-1:0]	S_AXIL_WSTRB,
		output	reg				S_AXIL_BVALID,
		input	wire				S_AXIL_BREADY,
		output	wire	[1:0]			S_AXIL_BRESP,
		input	wire				S_AXIL_ARVALID,
		output	wire				S_AXIL_ARREADY,
		input	wire [C_AXIL_ADDR_WIDTH-1:0]	S_AXIL_ARADDR,
		input	wire 	[2:0]			S_AXIL_ARPROT,
		output	reg				S_AXIL_RVALID,
		input	wire				S_AXIL_RREADY,
		output	reg [C_AXIL_DATA_WIDTH-1:0]	S_AXIL_RDATA,
		output	wire	[1:0]			S_AXIL_RRESP,
		output	reg				M_AXI_AWVALID,
		input	wire				M_AXI_AWREADY,
		output	reg	[C_AXI_ID_WIDTH-1:0]	M_AXI_AWID,
		output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_AWADDR,
		output	reg	[2:0]			M_AXI_AWSIZE,
		output	reg	[1:0]			M_AXI_AWBURST,
		output	reg				M_AXI_AWLOCK,
		output	reg	[3:0]			M_AXI_AWCACHE,
		output	reg	[2:0]			M_AXI_AWPROT,
		output	reg	[3:0]			M_AXI_AWQOS,
		output	reg				M_AXI_WVALID,
		input	wire				M_AXI_WREADY,
		output	reg	[C_AXI_DATA_WIDTH-1:0]	M_AXI_WDATA,
		output	reg [C_AXI_DATA_WIDTH/8-1:0]	M_AXI_WSTRB,
		output	reg				M_AXI_WLAST,
		input	wire				M_AXI_BVALID,
		output	reg				M_AXI_BREADY,
		input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_BID,
		input	wire	[1:0]			M_AXI_BRESP,
		output	reg				M_AXI_ARVALID,
		input	wire				M_AXI_ARREADY,
		output	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_ARID,
		output	reg	[C_AXI_ADDR_WIDTH-1:0]	M_AXI_ARADDR,
		output	wire	[2:0]			M_AXI_ARSIZE,
		output	wire	[1:0]			M_AXI_ARBURST,
		output	wire				M_AXI_ARLOCK,
		output	wire	[3:0]			M_AXI_ARCACHE,
		output	wire	[2:0]			M_AXI_ARPROT,
		output	wire	[3:0]			M_AXI_ARQOS,
		input	wire				M_AXI_RVALID,
		output	wire				M_AXI_RREADY,
		input	wire	[C_AXI_ID_WIDTH-1:0]	M_AXI_RID,
		input	wire	[C_AXI_DATA_WIDTH-1:0]	M_AXI_RDATA,
		input	wire				M_AXI_RLAST,
		input	wire	[1:0]			M_AXI_RRESP,
		output	reg				o_int
	);

	// Register addresses
	localparam REG_SRC_ADDR = 0;
	localparam REG_DST_ADDR = 1;
	localparam REG_LENGTH   = 2;
	localparam REG_CONTROL  = 3;
	localparam REG_STATUS   = 4;

	// Control register bits
	localparam CTRL_START = 0;
	localparam CTRL_ABORT = 1;

	// Status register bits
	localparam STATUS_BUSY  = 0;
	localparam STATUS_DONE  = 1;
	localparam STATUS_ERROR = 2;

	// Internal signals
	reg [C_AXI_ADDR_WIDTH-1:0] src_addr_reg;
	reg [C_AXI_ADDR_WIDTH-1:0] dst_addr_reg;
	reg [LGLEN-1:0]            length_reg;
	reg [1:0]                  control_reg;
	reg [2:0]                  status_reg;

	reg                        dma_active;
	reg [C_AXI_ADDR_WIDTH-1:0] cur_src_addr;
	reg [C_AXI_ADDR_WIDTH-1:0] cur_dst_addr;
	reg [LGLEN-1:0]            cur_length;
	reg [LGLEN-1:0]            bytes_done;

	// AXI write channel
	reg                        aw_valid;
	reg [C_AXI_ID_WIDTH-1:0]   aw_id;
	reg [C_AXI_ADDR_WIDTH-1:0] aw_addr;
	reg [2:0]                  aw_size;
	reg [1:0]                  aw_burst;
	reg                        aw_lock;
	reg [3:0]                  aw_cache;
	reg [2:0]                  aw_prot;
	reg [3:0]                  aw_qos;

	reg                        w_valid;
	reg [C_AXI_DATA_WIDTH-1:0] w_data;
	reg [C_AXI_DATA_WIDTH/8-1:0] w_strb;
	reg                        w_last;

	reg                        b_ready;

	// AXI read channel
	reg                        ar_valid;
	reg [C_AXI_ID_WIDTH-1:0]   ar_id;
	reg [C_AXI_ADDR_WIDTH-1:0] ar_addr;
	reg [2:0]                  ar_size;
	reg [1:0]                  ar_burst;
	reg                        ar_lock;
	reg [3:0]                  ar_cache;
	reg [2:0]                  ar_prot;
	reg [3:0]                  ar_qos;

	reg                        r_ready;

	// State machine
	localparam S_IDLE = 2'd0;
	localparam S_READ = 2'd1;
	localparam S_WRITE = 2'd2;
	localparam S_DONE = 2'd3;

	reg [1:0] state;

	// AXI4-Lite write address channel
	reg aw_ready;
	reg w_ready;
	reg b_valid;
	reg [1:0] b_resp;

	// AXI4-Lite read address channel
	reg ar_ready;
	reg r_valid;
	reg [C_AXIL_DATA_WIDTH-1:0] r_data;
	reg [1:0] r_resp;

	// Assign outputs
	assign S_AXIL_AWREADY = aw_ready;
	assign S_AXIL_WREADY  = w_ready;
	assign S_AXIL_BVALID  = b_valid;
	assign S_AXIL_BRESP   = b_resp;
	assign S_AXIL_ARREADY = ar_ready;
	assign S_AXIL_RVALID  = r_valid;
	assign S_AXIL_RDATA   = r_data;
	assign S_AXIL_RRESP   = r_resp;

	assign M_AXI_AWVALID = aw_valid;
	assign M_AXI_AWID    = aw_id;
	assign M_AXI_AWADDR  = aw_addr;
	assign M_AXI_AWSIZE  = aw_size;
	assign M_AXI_AWBURST = aw_burst;
	assign M_AXI_AWLOCK  = aw_lock;
	assign M_AXI_AWCACHE = aw_cache;
	assign M_AXI_AWPROT  = aw_prot;
	assign M_AXI_AWQOS   = aw_qos;

	assign M_AXI_WVALID  = w_valid;
	assign M_AXI_WDATA   = w_data;
	assign M_AXI_WSTRB   = w_strb;
	assign M_AXI_WLAST   = w_last;
	assign M_AXI_BREADY  = b_ready;

	assign M_AXI_ARVALID = ar_valid;
	assign M_AXI_ARID    = ar_id;
	assign M_AXI_ARADDR  = ar_addr;
	assign M_AXI_ARSIZE  = ar_size;
	assign M_AXI_ARBURST = ar_burst;
	assign M_AXI_ARLOCK  = ar_lock;
	assign M_AXI_ARCACHE = ar_cache;
	assign M_AXI_ARPROT  = ar_prot;
	assign M_AXI_ARQOS   = ar_qos;
	assign M_AXI_RREADY  = r_ready;

	// AXI4-Lite write address channel logic
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			aw_ready <= 1'b1;
			w_ready  <= 1'b1;
			b_valid  <= 1'b0;
			b_resp   <= 2'b00;
		end else begin
			if (S_AXIL_AWVALID && aw_ready) begin
				aw_ready <= 1'b0;
				w_ready  <= 1'b1;
			end else if (S_AXIL_WVALID && w_ready) begin
				w_ready  <= 1'b0;
				b_valid  <= 1'b1;
				b_resp   <= 2'b00;
			end else if (b_valid && S_AXIL_BREADY) begin
				b_valid  <= 1'b0;
				aw_ready <= 1'b1;
			end
		end
	end

	// AXI4-Lite read address channel logic
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			ar_ready <= 1'b1;
			r_valid  <= 1'b0;
			r_data   <= {C_AXIL_DATA_WIDTH{1'b0}};
			r_resp   <= 2'b00;
		end else begin
			if (S_AXIL_ARVALID && ar_ready) begin
				ar_ready <= 1'b0;
				r_valid  <= 1'b1;
				r_data   <= S_AXIL_WDATA; // Placeholder, actual read data comes from DMA
				r_resp   <= 2'b00;
			end else if (r_valid && S_AXIL_RREADY) begin
				r_valid  <= 1'b0;
				ar_ready <= 1'b1;
			end
		end
	end

	// Register file and control logic
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			src_addr_reg <= {C_AXI_ADDR_WIDTH{1'b0}};
			dst_addr_reg <= {C_AXI_ADDR_WIDTH{1'b0}};
			length_reg   <= {LGLEN{1'b0}};
			control_reg  <= 2'b00;
			status_reg   <= 3'b000;
			dma_active   <= 1'b0;
			cur_src_addr <= {C_AXI_ADDR_WIDTH{1'b0}};
			cur_dst_addr <= {C_AXI_ADDR_WIDTH{1'b0}};
			cur_length   <= {LGLEN{1'b0}};
			bytes_done   <= {LGLEN{1'b0}};
			state        <= S_IDLE;
			o_int        <= 1'b0;
		end else begin
			// Handle register writes
			if (S_AXIL_AWVALID && S_AXIL_WVALID && aw_ready && w_ready) begin
				case (S_AXIL_AWADDR[C_AXIL_ADDR_WIDTH-1:0])
					REG_SRC_ADDR: src_addr_reg <= S_AXIL_WDATA;
					REG_DST_ADDR: dst_addr_reg <= S_AXIL_WDATA;
					REG_LENGTH:   length_reg   <= S_AXIL_WDATA[LGLEN-1:0];
					REG_CONTROL: begin
						control_reg <= S_AXIL_WDATA[1:0];
						if (S_AXIL_WDATA[CTRL_START] && !dma_active) begin
							dma_active   <= 1'b1;
							cur_src_addr <= src_addr_reg;
							cur_dst_addr <= dst_addr_reg;
							cur_length   <= length_reg;
							bytes_done   <= {LGLEN{1'b0}};
							state        <= S_READ;
							status_reg   <= 3'b000;
							o_int        <= 1'b0;
						end
						if (S_AXIL_WDATA[CTRL_ABORT] && dma_active) begin
							dma_active <= 1'b0;
							state      <= S_IDLE;
							status_reg <= 3'b000;
							o_int      <= 1'b0;
						end
					end
					default: ;
				endcase
			end

			// DMA state machine
			if (dma_active) begin
				case (state)
					S_READ: begin
						if (ar_valid && M_AXI_ARREADY) begin
							ar_valid <= 1'b0;
							state    <= S_WRITE;
						end else begin
							ar_valid <= 1'b1;
							ar_id    <= AXI_READ_ID;
							ar_addr  <= cur_src_addr;
							ar_size  <= ADDRLSB;
							ar_burst <= 2'b01; // INCR
							ar_lock  <= 1'b0;
							ar_cache <= 4'b0011;
							ar_prot  <= 3'b000;
							ar_qos   <= 4'b0000;
						end
					end
					S_WRITE: begin
						if (aw_valid && M_AXI_AWREADY && w_valid && M_AXI_WREADY) begin
							aw_valid <= 1'b0;
							w_valid  <= 1'b0;
							b_ready  <= 1'b1;
						end else begin
							aw_valid <= 1'b1;
							aw_id    <= AXI_WRITE_ID;
							aw_addr  <= cur_dst_addr;
							aw_size  <= ADDRLSB;
							aw_burst <= 2'b01; // INCR
							aw_lock  <= 1'b0;
							aw_cache <= 4'b0011;
							aw_prot  <= 3'b000;
							aw_qos   <= 4'b0000;
							w_valid  <= 1'b1;
							w_data   <= M_AXI_RDATA;
							w_strb   <= {C_AXI_DATA_WIDTH/8{1'b1}};
							w_last   <= 1'b1;
						end
					end
					S_DONE: begin
						dma_active <= 1'b0;
						state      <= S_IDLE;
						status_reg <= 3'b010; // DONE
						o_int      <= 1'b1;
					end
					default: state <= S_IDLE;
				endcase
			end

			// Handle AXI read response
			if (M_AXI_RVALID && r_ready) begin
				r_ready <= 1'b0;
				if (M_AXI_RLAST) begin
					// Update addresses and length
					cur_src_addr <= cur_src_addr + C_AXI_DATA_WIDTH/8;
					cur_dst_addr <= cur_dst_addr + C_AXI_DATA_WIDTH/8;
					bytes_done   <= bytes_done + C_AXI_DATA_WIDTH/8;
					if (bytes_done + C_AXI_DATA_WIDTH/8 >= cur_length) begin
						state <= S_DONE;
					end else begin
						state <= S_READ;
					end
				end
			end else begin
				r_ready <= 1'b1;
			end

			// Handle AXI write response
			if (M_AXI_BVALID && b_ready) begin
				b_ready <= 1'b0;
			end else begin
				b_ready <= 1'b1;
			end
		end
	end

	// Status register output
	assign status_reg = {1'b0, dma_active, 1'b0};

endmodule
