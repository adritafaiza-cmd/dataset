```verilog
module i2c_master (
    input  wire        clk,
    input  wire        rst,
    /*
     * Host interface
     */
    input  wire [6:0]  s_axis_cmd_address,
    input  wire        s_axis_cmd_start,
    input  wire        s_axis_cmd_read,
    input  wire        s_axis_cmd_write,
    input  wire        s_axis_cmd_write_multiple,
    input  wire        s_axis_cmd_stop,
    input  wire        s_axis_cmd_valid,
    output wire        s_axis_cmd_ready,
    input  wire [7:0]  s_axis_data_tdata,
    input  wire        s_axis_data_tvalid,
    output wire        s_axis_data_tready,
    input  wire        s_axis_data_tlast,
    output wire [7:0]  m_axis_data_tdata,
    output wire        m_axis_data_tvalid,
    input  wire        m_axis_data_tready,
    output wire        m_axis_data_tlast,
    /*
     * I2C interface
     */
    input  wire        scl_i,
    output wire        scl_o,
    output wire        scl_t,
    input  wire        sda_i,
    output wire        sda_o,
    output wire        sda_t,
    /*
     * Status
     */
    output wire        busy,
    output wire        bus_control,
    output wire        bus_active,
    output wire        missed_ack,
    /*
     * Configuration
     */
    input  wire [15:0] prescale,
    input  wire        stop_on_idle
);

    // State machine states
    localparam S_IDLE = 4'd0;
    localparam S_START = 4'd1;
    localparam S_ADDR = 4'd2;
    localparam S_DATA = 4'd3;
    localparam S_STOP = 4'd4;
    localparam S_ACK_CHECK = 4'd5;
    localparam S_READ_DATA = 4'd6;
    localparam S_READ_ACK = 4'd7;

    reg [3:0] state;
    reg [3:0] next_state;

    // Timing counter
    reg [15:0] timer;
    reg [15:0] timer_limit;

    // Bit counter
    reg [2:0] bit_count;

    // Data registers
    reg [7:0] tx_data;
    reg [7:0] rx_data;
    reg [6:0] addr_reg;
    reg       addr_read;
    reg       write_multiple;
    reg       stop_cmd;

    // Open-drain control
    reg scl_drive;
    reg sda_drive;
    reg scl_release;
    reg sda_release;

    // Bus control
    reg bus_ctrl;
    reg bus_act;

    // Missed ACK
    reg missed_ack_reg;

    // Command interface
    reg cmd_ready;
    reg data_tready;

    // Read data output
    reg [7:0] m_data;
    reg m_valid;
    reg m_last;

    // Prescale handling
    wire [15:0] prescale_eff = (prescale == 16'd0) ? 16'd1 : prescale;

    // Timer limit based on prescale
    always @(*) begin
        if (state == S_START || state == S_STOP)
            timer_limit = prescale_eff;
        else if (state == S_ADDR || state == S_DATA || state == S_READ_DATA || state == S_READ_ACK)
            timer_limit = prescale_eff;
        else
            timer_limit = prescale_eff;
    end

    // State machine
    always @(*) begin
        next_state = state;
        case (state)
            S_IDLE: begin
                if (s_axis_cmd_valid && s_axis_cmd_ready)
                    next_state = S_START;
            end
            S_START: begin
                if (timer == 16'd0)
                    next_state = S_ADDR;
            end
            S_ADDR: begin
                if (timer == 16'd0 && bit_count == 3'd0)
                    next_state = S_ACK_CHECK;
            end
            S_ACK_CHECK: begin
                if (timer == 16'd0) begin
                    if (sda_i == 1'b0) begin
                        // ACK received
                        if (addr_read)
                            next_state = S_READ_DATA;
                        else if (write_multiple)
                            next_state = S_DATA;
                        else
                            next_state = S_STOP;
                    end else begin
                        // NACK
                        next_state = S_STOP;
                    end
                end
            end
            S_DATA: begin
                if (timer == 16'd0 && bit_count == 3'd0)
                    next_state = S_ACK_CHECK;
            end
            S_READ_DATA: begin
                if (timer == 16'd0 && bit_count == 3'd0)
                    next_state = S_READ_ACK;
            end
            S_READ_ACK: begin
                if (timer == 16'd0) begin
                    if (s_axis_data_tlast)
                        next_state = S_STOP;
                    else
                        next_state = S_READ_DATA;
                end
            end
            S_STOP: begin
                if (timer == 16'd0)
                    next_state = S_IDLE;
            end
            default: next_state = S_IDLE;
        endcase
    end

    // Main sequential logic
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= S_IDLE;
            timer <= 16'd0;
            bit_count <= 3'd0;
            tx_data <= 8'd0;
            rx_data <= 8'd0;
            addr_reg <= 7'd0;
            addr_read <= 1'b0;
            write_multiple <= 1'b0;
            stop_cmd <= 1'b0;
            scl_drive <= 1'b1;
            sda_drive <= 1'b1;
            scl_release <= 1'b1;
            sda_release <= 1'b1;
            bus_ctrl <= 1'b0;
            bus_act <= 1'b0;
            missed_ack_reg <= 1'b0;
            cmd_ready <= 1'b1;
            data_tready <= 1'b0;
            m_data <= 8'd0;
            m_valid <= 1'b0;
            m_last <= 1'b0;
        end else begin
            // Default assignments
            cmd_ready <= 1'b0;
            data_tready <= 1'b0;
            m_valid <= 1'b0;
            m_last <= 1'b0;

            case (state)
                S_IDLE: begin
                    cmd_ready <= 1'b1;
                    scl_drive <= 1'b1;
                    sda_drive <= 1'b1;
                    scl_release <= 1'b1;
                    sda_release <= 1'b1;
                    bus_ctrl <= 1'b0;
                    bus_act <= 1'b0;
                    missed_ack_reg <= 1'b0;
                    timer <= 16'd0;
                    bit_count <= 3'd0;

                    if (s_axis_cmd_valid) begin
                        addr_reg <= s_axis_cmd_address;
                        addr_read <= s_axis_cmd_read;
                        write_multiple <= s_axis_cmd_write_multiple;
                        stop_cmd <= s_axis_cmd_stop;
                        state <= S_START;
                        timer <= timer_limit - 1;
                        bus_ctrl <= 1'b1;
                        bus_act <= 1'b1;
                        // START condition: SDA high to low while SCL high
                        sda_drive <= 1'b0;
                        sda_release <= 1'b0;
                        scl_drive <= 1'b1;
                        scl_release <= 1'b1;
                    end
                end

                S_START: begin
                    if (timer == 16'd0) begin
                        state <= S_ADDR;
                        timer <= timer_limit - 1;
                        bit_count <= 3'd3;
                        // After START, SCL goes low, SDA holds address MSB
                        scl_drive <= 1'b0;
                        scl_release <= 1'b0;
                        sda_drive <= 1'b0; // Will be set based on addr bit
                        sda_release <= 1'b0;
                        // Set SDA to address MSB
                        sda_drive <= ~addr_reg[6];
                        sda_release <= 1'b0;
                    end else begin
                        timer <= timer - 1;
                    end
                end

                S_ADDR: begin
                    if (timer == 16'd0) begin
                        if (bit_count == 3'd0) begin
                            state <= S_ACK_CHECK;
                            timer <= timer_limit - 1;
                            // Release SDA for ACK
                            sda_drive <= 1'b1;
                            sda_release <= 1'b1;
                            scl_drive <= 1'b1;
                            scl_release <= 1'b1;
                        end else begin
                            bit_count <= bit_count - 1;
                            timer <= timer_limit - 1;
                            // SCL high, SDA holds current bit
                            scl_drive <= 1'b1;
                            scl_release <= 1'b1;
                            // Next bit
                            case (bit_count)
                                3'd3: sda_drive <= ~addr_reg[5];
                                3'd2: sda_drive <= ~addr_reg[4];
                                3'd1: sda_drive <= ~addr_reg[3];
                                3'd0: sda_drive <= ~addr_reg[2];
                                default: sda_drive <= 1'b1;
                            endcase
                            sda_release <= 1'b0;
                        end
                    end else begin
                        timer <= timer - 1;
                    end
                end

                S_ACK_CHECK: begin
                    if (timer == 16'd0) begin
                        if (sda_i == 1'b0) begin
                            // ACK
                            missed_ack_reg <= 1'b0;
                            if (addr_read) begin
                                state <= S_READ_DATA;
                                timer <= timer_limit - 1;
                                bit_count <= 3'd3;
                                // Release SDA for read
                                sda_drive <= 1'b1;
                                sda_release <= 1'b1;
                                scl_drive <= 1'b1;
                                scl_release <= 1'b1;
                            end else if (write_multiple) begin
                                state <= S_DATA;
                                timer <= timer_limit - 1;
                                bit_count <= 3'd3;
                                data_tready <= 1'b1;
                                scl_drive <= 1'b0;
                                scl_release <= 1'b0;
                                sda_drive <= 1'b1;
                                sda_release <= 1'b1;
                            end else begin
                                state <= S_STOP;
                                timer <= timer_limit - 1;
                                scl_drive <= 1'b1;
                                scl_release <= 1'b1;
                                sda_drive <= 1'b0;
                                sda_release <= 1'b0;
                            end
                        end else begin
                            // NACK
                            missed_ack_reg <= 1'b1;
                            state <= S_STOP;
                            timer <= timer_limit - 1;
                            scl_drive <= 1'b1;
                            scl_release <= 1'b1;
                            sda_drive <= 1'b0;
                            sda_release <= 1'b0;
                        end
                    end else begin
                        timer <= timer - 1;
                    end
                end

                S_DATA: begin
                    if (timer == 16'd0) begin
                        if (bit_count == 3'd0) begin
                            state <= S_ACK_CHECK;
                            timer <= timer_limit - 1;
                            // Release SDA for ACK
                            sda_drive <= 1'b1;
                            sda_release <= 1'b1;
                            scl_drive <= 1'b1;
                            scl_release <= 1'b1;
                        end else begin
                            bit_count <= bit_count - 1;
                            timer <= timer_limit - 1;
                            // SCL high, SDA holds current bit
                            scl_drive <= 1'b1;
                            scl_release <= 1'b1;
                            // Next bit
                            case (bit_count)
                                3'd3: sda_drive <= ~tx_data[6];
                                3'd2: sda_drive <= ~tx_data[5];
                                3'd1: sda_drive <= ~tx_data[4];
                                3'd0: sda_drive <= ~tx_data[3];
                                default: sda_drive <= 1'b1;
                            endcase
                            sda_release <= 1'b0;
                        end
                    end else begin
                        timer <= timer - 1;
                    end

                    // Load data when ready
                    if (s_axis_data_tvalid && data_tready) begin
                        tx_data <= s_axis_data_tdata;
                        data_tready <= 1'b0;
                    end
                end

                S_READ_DATA: begin
                    if (timer == 16'd0) begin
                        if (bit_count == 3'd0) begin
                            state <= S_READ_ACK;
                            timer <= timer_limit - 1;
                            // SCL high, SDA released for ACK
                            scl_drive <= 1'b1;
                            scl_release <= 1'b1;
                            sda_drive <= 1'b1;
                            sda_release <= 1'b1;
                            // Output read data
                            m_data <= rx_data;
                            m_valid <= 1'b1;
                            m_last <= s_axis_data_tlast;
                        end else begin
                            bit_count <= bit_count - 1;
                            timer <= timer_limit - 1;
                            // SCL high, sample SDA
                            scl_drive <= 1'b1;
                            scl_release <= 1'b1;
                            sda_drive <= 1'b1;
                            sda_release <= 1'b1;
                            // Shift in data
                            rx_data <= {rx_data[6:0], sda_i};
                        end
                    end else begin
                        timer <= timer - 1;
                    end
                end

                S_READ_ACK: begin
                    if (timer == 16'd0) begin
                        if (s_axis_data_tlast) begin
                            state <= S_STOP;
                            timer <= timer_limit - 1;
                            scl_drive <= 1'b1;
                            scl_release <= 1'b1;
                            sda_drive <= 1'b0;
                            sda_release <= 1'b0;
                        end else begin
                            state <= S_READ_DATA;
                            timer <= timer_limit - 1;
                            bit_count <= 3'd3;
                            scl_drive <= 1'b0;
                            scl_release <= 1'b0;
                            sda_drive <= 1'b1;
                            sda_release <= 1'b1;
                        end
                    end else begin
                        timer <= timer - 1;
                    end
                end

                S_STOP: begin
                    if (timer == 16'd0) begin
                        state <= S_IDLE;
                        timer <= 16'd0;
                        bit_count <= 3'd0;
                        scl_drive <= 1'b1;
                        scl_release <= 1'b1;
                        sda_drive <= 1'b1;
                        sda_release <= 1'b1;
                        bus_ctrl <= 1'b0;
                        bus_act <= 1'b0;
                    end else begin
                        timer <= timer - 1;
                    end
                end

                default: begin
                    state <= S_IDLE;
                    timer <= 16'd0;
                    bit_count <= 3'd0;
                    scl_drive <= 1'b1;
                    sda_drive <= 1'b1;
                    scl_release <= 1'b1;
                    sda_release <= 1'b1;
                    bus_ctrl <= 1'b0;
                    bus_act <= 1'b0;
                end
            endcase
        end
    end

    // Open-drain outputs
    assign scl_o = scl_drive;
    assign scl_t = ~scl_release;
    assign sda_o = sda_drive;
    assign sda_t = ~sda_release;

    // Status outputs
    assign busy = (state != S_IDLE);
    assign bus_control = bus_ctrl;
    assign bus_active = bus_act;
    assign missed_ack = missed_ack_reg;

    // Command ready
    assign s_axis_cmd_ready = cmd_ready;

    // Data ready
    assign s_axis_data_tready = data_tready;

    // Read data output
    assign m_axis_data_tdata = m_data;
    assign m_axis_data_tvalid = m_valid;
    assign m_axis_data_tlast = m_last;

endmodule
