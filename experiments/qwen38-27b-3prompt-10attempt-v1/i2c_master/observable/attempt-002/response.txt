```verilog
module i2c_master (
    input  wire        clk,
    input  wire        rst,
    /*
     * Host interface
     */
    input  wire [6:0]  s_axis_cmd_address,
    input  wire        s_axis_cmd_start,
    input  wire        s_axis_cmd_read,
    input  wire        s_axis_cmd_write,
    input  wire        s_axis_cmd_write_multiple,
    input  wire        s_axis_cmd_stop,
    input  wire        s_axis_cmd_valid,
    output wire        s_axis_cmd_ready,
    input  wire [7:0]  s_axis_data_tdata,
    input  wire        s_axis_data_tvalid,
    output wire        s_axis_data_tready,
    input  wire        s_axis_data_tlast,
    output wire [7:0]  m_axis_data_tdata,
    output wire        m_axis_data_tvalid,
    input  wire        m_axis_data_tready,
    output wire        m_axis_data_tlast,
    /*
     * I2C interface
     */
    input  wire        scl_i,
    output wire        scl_o,
    output wire        scl_t,
    input  wire        sda_i,
    output wire        sda_o,
    output wire        sda_t,
    /*
     * Status
     */
    output wire        busy,
    output wire        bus_control,
    output wire        bus_active,
    output wire        missed_ack,
    /*
     * Configuration
     */
    input  wire [15:0] prescale,
    input  wire        stop_on_idle
);

    // State machine states
    localparam S_IDLE          = 4'd0;
    localparam S_START         = 4'd1;
    localparam S_SEND_BIT      = 4'd2;
    localparam S_CHECK_ACK     = 4'd3;
    localparam S_RECV_BIT      = 4'd4;
    localparam S_RECV_ACK      = 4'd5;
    localparam S_STOP          = 4'd6;
    localparam S_WAIT_DATA     = 4'd7;
    localparam S_DONE          = 4'd8;

    reg [3:0]  state;
    reg [3:0]  bit_count;
    reg [7:0]  tx_byte;
    reg [7:0]  rx_byte;
    reg        is_read;
    reg        is_write;
    reg        is_write_multiple;
    reg        need_stop;
    reg        ack_received;
    reg        missed_ack_reg;
    reg        bus_active_reg;
    reg        bus_control_reg;

    // Timing counter
    reg [15:0] timer;
    reg        timer_done;

    // SDA/SCL control
    reg        sda_drive;
    reg        sda_output;
    reg        scl_drive;
    reg        scl_output;

    // Command latching
    reg [6:0]  cmd_address;
    reg        cmd_start;
    reg        cmd_read;
    reg        cmd_write;
    reg        cmd_write_multiple;
    reg        cmd_stop;

    // Data buffer for write multiple
    reg [7:0]  data_buffer [0:255];
    reg [7:0]  data_count;
    reg [7:0]  data_index;
    reg        data_valid;

    // Mux for SDA/SCL
    assign sda_o = sda_drive ? sda_output : 1'b0;
    assign sda_t = sda_drive;
    assign scl_o = scl_drive ? scl_output : 1'b0;
    assign scl_t = scl_drive;

    // Status outputs
    assign busy          = (state != S_IDLE);
    assign bus_control   = bus_control_reg;
    assign bus_active    = bus_active_reg;
    assign missed_ack    = missed_ack_reg;

    // AXI command ready
    assign s_axis_cmd_ready = (state == S_IDLE) && !rst;

    // AXI data ready
    assign s_axis_data_tready = (state == S_WAIT_DATA) && !rst;

    // M axis data output
    assign m_axis_data_tdata  = rx_byte;
    assign m_axis_data_tvalid = (state == S_DONE) && is_read && !rst;
    assign m_axis_data_tlast  = (state == S_DONE) && is_read && !rst;

    // Timer logic
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            timer <= 16'd0;
            timer_done <= 1'b0;
        end else begin
            if (timer_done) begin
                timer <= 16'd0;
                timer_done <= 1'b0;
            end else if (timer < prescale) begin
                timer <= timer + 16'd1;
            end else begin
                timer <= 16'd0;
                timer_done <= 1'b1;
            end
        end
    end

    // Main state machine
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state             <= S_IDLE;
            bit_count         <= 4'd0;
            tx_byte           <= 8'd0;
            rx_byte           <= 8'd0;
            is_read           <= 1'b0;
            is_write          <= 1'b0;
            is_write_multiple <= 1'b0;
            need_stop         <= 1'b0;
            ack_received      <= 1'b0;
            missed_ack_reg    <= 1'b0;
            bus_active_reg    <= 1'b0;
            bus_control_reg   <= 1'b0;
            sda_drive         <= 1'b0;
            sda_output        <= 1'b1;
            scl_drive         <= 1'b0;
            scl_output        <= 1'b1;
            cmd_address       <= 7'd0;
            cmd_start         <= 1'b0;
            cmd_read          <= 1'b0;
            cmd_write         <= 1'b0;
            cmd_write_multiple<= 1'b0;
            cmd_stop          <= 1'b0;
            data_count        <= 8'd0;
            data_index        <= 8'd0;
            data_valid        <= 1'b0;
        end else begin
            case (state)
                S_IDLE: begin
                    sda_drive   <= 1'b0;
                    sda_output  <= 1'b1;
                    scl_drive   <= 1'b0;
                    scl_output  <= 1'b1;
                    bus_active_reg    <= 1'b0;
                    bus_control_reg   <= 1'b0;
                    missed_ack_reg    <= 1'b0;
                    need_stop         <= 1'b0;
                    data_count        <= 8'd0;
                    data_index        <= 8'd0;
                    data_valid        <= 1'b0;

                    if (s_axis_cmd_valid) begin
                        cmd_address       <= s_axis_cmd_address;
                        cmd_start         <= s_axis_cmd_start;
                        cmd_read          <= s_axis_cmd_read;
                        cmd_write         <= s_axis_cmd_write;
                        cmd_write_multiple<= s_axis_cmd_write_multiple;
                        cmd_stop          <= s_axis_cmd_stop;
                        is_read           <= s_axis_cmd_read;
                        is_write          <= s_axis_cmd_write;
                        is_write_multiple <= s_axis_cmd_write_multiple;
                        bus_control_reg   <= 1'b1;
                        bus_active_reg    <= 1'b1;
                        state             <= S_START;
                    end
                end

                S_START: begin
                    // Generate START condition: SDA high to low while SCL high
                    if (!timer_done) begin
                        // Wait for timer
                    end else begin
                        // Phase 1: SCL high, SDA high
                        scl_drive   <= 1'b1;
                        scl_output  <= 1'b1;
                        sda_drive   <= 1'b1;
                        sda_output  <= 1'b1;
                        state       <= S_START;
                        // Use a sub-state approach via bit_count
                        if (bit_count == 4'd0) begin
                            bit_count <= 4'd1;
                        end else if (bit_count == 4'd1) begin
                            // Phase 2: SDA goes low while SCL high
                            sda_drive   <= 1'b1;
                            sda_output  <= 1'b0;
                            bit_count   <= 4'd2;
                        end else if (bit_count == 4'd2) begin
                            // Phase 3: SCL goes low
                            scl_drive   <= 1'b1;
                            scl_output  <= 1'b0;
                            bit_count   <= 4'd3;
                        end else begin
                            // Phase 4: Prepare to send address
                            scl_drive   <= 1'b0;
                            scl_output  <= 1'b0;
                            sda_drive   <= 1'b0;
                            sda_output  <= 1'b1;
                            bit_count   <= 4'd0;
                            tx_byte     <= {cmd_address, cmd_read};
                            state       <= S_SEND_BIT;
                        end
                    end
                end

                S_SEND_BIT: begin
                    if (!timer_done) begin
                        // Wait
                    end else begin
                        if (bit_count < 4'd8) begin
                            // Drive SDA with current bit
                            sda_drive   <= 1'b1;
                            sda_output  <= tx_byte[7];
                            scl_drive   <= 1'b0;
                            scl_output  <= 1'b0;
                            bit_count   <= bit_count + 4'd1;
                        end else begin
                            // All 8 bits sent, release SDA for ACK
                            sda_drive   <= 1'b0;
                            sda_output  <= 1'b1;
                            scl_drive   <= 1'b0;
                            scl_output  <= 1'b0;
                            state       <= S_CHECK_ACK;
                            bit_count   <= 4'd0;
                        end
                    end
                end

                S_CHECK_ACK: begin
                    if (!timer_done) begin
                        // Wait
                    end else begin
                        // Clock SCL high to sample ACK
                        scl_drive   <= 1'b1;
                        scl_output  <= 1'b1;
                        sda_drive   <= 1'b0;
                        sda_output  <= 1'b1;
                        state       <= S_CHECK_ACK;
                        if (bit_count == 4'd0) begin
                            bit_count <= 4'd1;
                        end else begin
                            // Sample SDA
                            ack_received <= ~sda_i;
                            scl_drive   <= 1'b1;
                            scl_output  <= 1'b0;
                            sda_drive   <= 1'b0;
                            sda_output  <= 1'b1;
                            bit_count   <= 4'd0;

                            if (ack_received) begin
                                if (is_read) begin
                                    state <= S_RECV_BIT;
                                end else if (is_write_multiple) begin
                                    state <= S_WAIT_DATA;
                                end else if (is_write) begin
                                    state <= S_WAIT_DATA;
                                end else begin
                                    state <= S_STOP;
                                end
                            end else begin
                                missed_ack_reg <= 1'b1;
                                state <= S_STOP;
                            end
                        end
                    end
                end

                S_RECV_BIT: begin
                    if (!timer_done) begin
                        // Wait
                    end else begin
                        if (bit_count < 4'd8) begin
                            // Clock SCL high to sample data
                            scl_drive   <= 1'b1;
                            scl_output  <= 1'b1;
                            sda_drive   <= 1'b0;
                            sda_output  <= 1'b1;
                            state       <= S_RECV_BIT;
                            if (bit_count == 4'd0) begin
                                bit_count <= 4'd1;
                            end else begin
                                // Sample SDA
                                rx_byte[7 - bit_count + 4'd1] <= sda_i;
                                scl_drive   <= 1'b1;
                                scl_output  <= 1'b0;
                                sda_drive   <= 1'b0;
                                sda_output  <= 1'b1;
                                bit_count   <= bit_count + 4'd1;
                            end
                        end else begin
                            // All 8 bits received, send ACK/NACK
                            scl_drive   <= 1'b0;
                            scl_output  <= 1'b0;
                            sda_drive   <= 1'b1;
                            sda_output  <= 1'b0; // ACK
                            state       <= S_RECV_ACK;
                            bit_count   <= 4'd0;
                        end
                    end
                end

                S_RECV_ACK: begin
                    if (!timer_done) begin
                        // Wait
                    end else begin
                        // Clock SCL high to complete ACK
                        scl_drive   <= 1'b1;
                        scl_output  <= 1'b1;
                        sda_drive   <= 1'b1;
                        sda_output  <= 1'b0;
                        state       <= S_RECV_ACK;
                        if (bit_count == 4'd0) begin
                            bit_count <= 4'd1;
                        end else begin
                            scl_drive   <= 1'b1;
                            scl_output  <= 1'b0;
                            sda_drive   <= 1'b0;
                            sda_output  <= 1'b1;
                            bit_count   <= 4'd0;
                            state       <= S_DONE;
                        end
                    end
                end

                S_WAIT_DATA: begin
                    if (s_axis_data_tvalid) begin
                        if (is_write_multiple) begin
                            data_buffer[data_index] <= s_axis_data_tdata;
                            data_index <= data_index + 8'd1;
                            data_count <= data_count + 8'd1;
                            if (s_axis_data_tlast) begin
                                data_valid <= 1'b1;
                                state <= S_SEND_BIT;
                                tx_byte <= s_axis_data_tdata;
                                bit_count <= 4'd0;
                            end else begin
                                state <= S_SEND_BIT;
                                tx_byte <= s_axis_data_tdata;
                                bit_count <= 4'd0;
                            end
                        end else begin
                            // Single write
                            state <= S_SEND_BIT;
                            tx_byte <= s_axis_data_tdata;
                            bit_count <= 4'd0;
                        end
                    end
                end

                S_DONE: begin
                    if (m_axis_data_tready) begin
                        if (is_read) begin
                            state <= S_STOP;
                        end else begin
                            state <= S_STOP;
                        end
                    end
                end

                S_STOP: begin
                    // Generate STOP condition: SDA low to high while SCL high
                    if (!timer_done) begin
                        // Wait
                    end else begin
                        // Phase 1: SCL high, SDA low
                        scl_drive   <= 1'b1;
                        scl_output  <= 1'b1;
                        sda_drive   <= 1'b1;
                        sda_output  <= 1'b0;
                        state       <= S_STOP;
                        if (bit_count == 4'd0) begin
                            bit_count <= 4'd1;
                        end else if (bit_count == 4'd1) begin
                            // Phase 2: SDA goes high while SCL high
                            sda_drive   <= 1'b1;
                            sda_output  <= 1'b1;
                            bit_count   <= 4'd2;
                        end else begin
                            // Done
                            scl_drive   <= 1'b0;
                            scl_output  <= 1'b1;
                            sda_drive   <= 1'b0;
                            sda_output  <= 1'b1;
                            bus_active_reg    <= 1'b0;
                            bus_control_reg   <= 1'b0;
                            state             <= S_IDLE;
                            bit_count         <= 4'd0;
                        end
                    end
                end

                default: begin
                    state <= S_IDLE;
                end
            end
        end
    end

endmodule
