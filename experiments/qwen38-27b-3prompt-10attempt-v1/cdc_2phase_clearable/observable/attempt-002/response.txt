module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1

  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,

  output src_ready_o,

  output dst_valid_o,
  input  dst_ready_i

  wire dst_req;
  wire dst_ack;

  // We use a 2-phase handshake: request toggles on each transfer
  // The source holds data and valid until ack toggles back

  reg [WIDTH-1:0] src_data_r;
  reg src_valid_r;
  reg src_clear_pending_r;

  reg [WIDTH-1:0] dst_data_r;
  reg dst_valid_r;
  reg dst_clear_pending_r;

  reg [SYNC_STAGES-1:0] src_req_sync;
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni)
      src_req_sync <= {SYNC_STAGES{1'b0}};
    else
      src_req_sync <= {src_req_sync[SYNC_STAGES-2:0], src_req};
  end
  wire src_req_dst = src_req_sync[SYNC_STAGES-1];

  // Synchronizers for dst_ack to src domain
  reg [SYNC_STAGES-1:0] dst_ack_sync;

    else
      dst_ack_sync <= {dst_ack_sync[SYNC_STAGES-2:0], dst_ack};
  end

      src_clear_sync <= {SYNC_STAGES{1'b0}};
    else

  // src_ready is high when we can accept new data

  // src_req starts at 0. When we send, we toggle it.

  // In IDLE: if valid && !clear, capture data, toggle req, go to WAIT_ACK
  // In WAIT_ACK: wait for ack to toggle, then go to IDLE

  // When source sees ack=1, it knows transfer complete, req goes to 0
  // When destination sees req=0, it knows source is done, ack goes to 0
  // This is the standard 2-phase handshake

  // "A clear request should flush the channel and let both sides return to idle without a stray transfer."
  // "A clear isolates both ends, then drops the one outstanding beat by returning request and acknowledge to zero."

  // "Each clear step reaches the other clock one handshake later."

  // - When src_valid_i is high and src_ready_o is high, we accept data
  // - src_ready_o is high when we can accept new data
  // - We can accept new data when: not in clear, and (src_req_r == 0, meaning no outstanding request)

  // src_clear_pending_r is cleared when the clear has been acknowledged by the destination
  // The acknowledgment is that dst_ack_src has gone to 0 (after being possibly 1)

  // The clear signal is synchronized to the other domain.
  // When clear is active, the handshake is reset.

  // Source side:
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_req_r <= 1'b0;
      src_data_r <= {WIDTH{1'b0}};
      src_clear_pending_r <= 1'b0;
    end else begin
      // Handle clear
      if (src_clear_i) begin
        src_clear_pending_r <= 1'b1;
        src_req_r <= 1'b0;
      end else if (src_clear_pending_r) begin

        // Clear is acknowledged when dst_ack_src is 0 and was previously 1, or just when dst_ack_src is 0
        // Actually, let's say clear is done when dst_ack_src is 0
        if (dst_ack_src == 1'b0) begin
          src_clear_pending_r <= 1'b0;
        end
      end else begin
        // Normal operation
        if (src_valid_i && src_ready_o) begin
          src_data_r <= src_data_i;
          src_req_r <= 1'b1;
        end else if (ack_toggled && src_req_r == 1'b1) begin
          src_req_r <= 1'b0;
        end
      end
    end
  end

  assign src_ready_o = ~src_clear_pending_r && (src_req_r == 1'b0);
  assign src_clear_pending_o = src_clear_pending_r;

  // Destination side:
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_ack_r <= 1'b0;
      dst_data_r <= {WIDTH{1'b0}};
      dst_valid_r <= 1'b0;
      dst_clear_pending_r <= 1'b0;
    end else begin
      // Handle clear
      if (dst_clear_i) begin
        dst_clear_pending_r <= 1'b1;
        dst_ack_r <= 1'b0;
        dst_valid_r <= 1'b0;
      end else if (dst_clear_pending_r) begin
        // Wait for clear to be acknowledged
        if (src_req_dst == 1'b0) begin
          dst_clear_pending_r <= 1'b0;
        end
      end else begin
        // Normal operation
        if (src_req_dst == 1'b1 && dst_ack_r == 1'b0) begin
          dst_data_r <= src_data_r; // Wait, we can't access src_data_r directly!
          dst_valid_r <= 1'b1;
          dst_ack_r <= 1'b1;
        end else if (dst_valid_r && dst_ready_i) begin
          dst_valid_r <= 1'b0;
        end
        // Deassert ack when req goes to 0
        if (src_req_dst == 1'b0 && dst_ack_r == 1'b1) begin
          dst_ack_r <= 1'b0;
        end
      end
    end
  end

  assign dst_data_o = dst_data_r;
  assign dst_valid_o = dst_valid_r;
  assign dst_clear_pending_o = dst_clear_pending_r;

endmodule
