```systemverilog
module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Internal signals
  wire src_req;
  wire dst_ack;
  wire dst_req;
  wire src_ack;

  // Source domain state
  reg src_valid_reg;
  reg [WIDTH-1:0] src_data_reg;
  reg src_clear_pending_reg;
  reg src_clear_active_reg;

  // Destination domain state
  reg dst_valid_reg;
  reg [WIDTH-1:0] dst_data_reg;
  reg dst_clear_pending_reg;
  reg dst_clear_active_reg;

  // CDC synchronizers
  // src_req -> dst domain
  reg [SYNC_STAGES-1:0] src_req_sync;
  // dst_ack -> src domain
  reg [SYNC_STAGES-1:0] dst_ack_sync;
  // dst_req -> src domain
  reg [SYNC_STAGES-1:0] dst_req_sync;
  // src_ack -> dst domain
  reg [SYNC_STAGES-1:0] src_ack_sync;

  // Synchronizer logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni)
      src_req_sync <= {SYNC_STAGES{1'b0}};
    else
      src_req_sync <= {src_req_sync[SYNC_STAGES-2:0], src_req};
  end

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni)
      dst_ack_sync <= {SYNC_STAGES{1'b0}};
    else
      dst_ack_sync <= {dst_ack_sync[SYNC_STAGES-2:0], dst_ack};
  end

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni)
      dst_req_sync <= {SYNC_STAGES{1'b0}};
    else
      dst_req_sync <= {dst_req_sync[SYNC_STAGES-2:0], dst_req};
  end

  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni)
      src_ack_sync <= {SYNC_STAGES{1'b0}};
    else
      src_ack_sync <= {src_ack_sync[SYNC_STAGES-2:0], src_ack};
  end

  // Synchronized signals
  wire src_req_synced = src_req_sync[SYNC_STAGES-1];
  wire dst_ack_synced = dst_ack_sync[SYNC_STAGES-1];
  wire dst_req_synced = dst_req_sync[SYNC_STAGES-1];
  wire src_ack_synced = src_ack_sync[SYNC_STAGES-1];

  // Source domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_valid_reg <= 1'b0;
      src_data_reg <= {WIDTH{1'b0}};
      src_clear_pending_reg <= 1'b0;
      src_clear_active_reg <= 1'b0;
    end else begin
      // Clear handling
      if (src_clear_i) begin
        src_clear_active_reg <= 1'b1;
        src_clear_pending_reg <= 1'b1;
      end

      if (src_clear_active_reg) begin
        // Wait for destination to acknowledge clear
        if (dst_ack_synced) begin
          src_clear_active_reg <= 1'b0;
          src_clear_pending_reg <= 1'b0;
          src_valid_reg <= 1'b0;
          src_data_reg <= {WIDTH{1'b0}};
        end
      end else begin
        // Normal operation
        if (src_valid_i && src_ready_o) begin
          src_valid_reg <= 1'b1;
          src_data_reg <= src_data_i;
        end else if (dst_ack_synced) begin
          src_valid_reg <= 1'b0;
        end
      end
    end
  end

  // Source ready: can accept new data if not holding valid data and not in clear
  assign src_ready_o = !src_valid_reg && !src_clear_active_reg;

  // Source clear pending output
  assign src_clear_pending_o = src_clear_pending_reg;

  // Source request: high when we have valid data to send
  assign src_req = src_valid_reg && !src_clear_active_reg;

  // Source acknowledge: high when we acknowledge destination's clear request
  assign src_ack = src_clear_active_reg;

  // Destination domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid_reg <= 1'b0;
      dst_data_reg <= {WIDTH{1'b0}};
      dst_clear_pending_reg <= 1'b0;
      dst_clear_active_reg <= 1'b0;
    end else begin
      // Clear handling
      if (dst_clear_i) begin
        dst_clear_active_reg <= 1'b1;
        dst_clear_pending_reg <= 1'b1;
      end

      if (dst_clear_active_reg) begin
        // Wait for source to acknowledge clear
        if (src_ack_synced) begin
          dst_clear_active_reg <= 1'b0;
          dst_clear_pending_reg <= 1'b0;
          dst_valid_reg <= 1'b0;
          dst_data_reg <= {WIDTH{1'b0}};
        end
      end else begin
        // Normal operation
        if (src_req_synced && !dst_valid_reg) begin
          dst_valid_reg <= 1'b1;
          dst_data_reg <= src_data_reg;
        end else if (dst_valid_reg && dst_ready_i) begin
          dst_valid_reg <= 1'b0;
        end
      end
    end
  end

  // Destination valid output
  assign dst_valid_o = dst_valid_reg && !dst_clear_active_reg;

  // Destination data output
  assign dst_data_o = dst_data_reg;

  // Destination clear pending output
  assign dst_clear_pending_o = dst_clear_pending_reg;

  // Destination request: high when we request clear
  assign dst_req = dst_clear_active_reg;

  // Destination acknowledge: high when we acknowledge source's clear request
  assign dst_ack = dst_clear_active_reg;

endmodule
