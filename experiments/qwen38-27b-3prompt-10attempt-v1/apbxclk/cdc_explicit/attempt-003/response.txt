```verilog
module apbxclk #(
	parameter	C_APB_ADDR_WIDTH = 12,
	parameter	C_APB_DATA_WIDTH = 32,
	parameter [0:0]	OPT_REGISTERED = 1'b0,
	localparam	AW = C_APB_ADDR_WIDTH,
	localparam	DW = C_APB_DATA_WIDTH
) (
	input	wire			S_APB_PCLK, S_PRESETn,
	input	wire			S_APB_PSEL,
	input	wire			S_APB_PENABLE,
	output	reg			S_APB_PREADY,
	input	wire	[AW-1:0]	S_APB_PADDR,
	input	wire			S_APB_PWRITE,
	input	wire	[DW-1:0]	S_APB_PWDATA,
	input	wire	[DW/8-1:0]	S_APB_PWSTRB,
	input	wire	[2:0]		S_APB_PPROT,
	output	wire	[DW-1:0]	S_APB_PRDATA,
	output	wire			S_APB_PSLVERR,
	input	wire			M_APB_PCLK,
	output	reg			M_PRESETn,
	output	reg			M_APB_PSEL,
	output	reg			M_APB_PENABLE,
	input	wire			M_APB_PREADY,
	output	wire	[AW-1:0]	M_APB_PADDR,
	output	wire			M_APB_PWRITE,
	output	wire	[DW-1:0]	M_APB_PWDATA,
	output	wire	[DW/8-1:0]	M_APB_PWSTRB,
	output	wire	[2:0]		M_APB_PPROT,
	input	wire	[DW-1:0]	M_APB_PRDATA,
	input	wire			M_APB_PSLVERR
);

	// Internal signals
	reg	[AW-1:0]	m_addr_reg;
	reg			m_write_reg;
	reg	[DW-1:0]	m_wdata_reg;
	reg	[DW/8-1:0]	m_wstrb_reg;
	reg	[2:0]		m_pprot_reg;

	reg			m_req_valid;
	reg			m_req_ack;

	reg			m_done_valid;
	reg			m_done_ack;

	reg	[DW-1:0]	m_rdata_reg;
	reg			m_slverr_reg;

	// S-domain: request capture and response
	reg			s_req_pending;
	reg			s_req_valid;
	reg			s_req_ack;

	reg			s_done_valid;
	reg			s_done_ack;

	reg	[DW-1:0]	s_rdata_reg;
	reg			s_slverr_reg;

	// M-domain: APB state machine
	reg			m_setup;
	reg			m_access;
	reg			m_busy;

	// S-domain: APB state machine
	reg			s_setup;
	reg			s_access;
	reg			s_busy;

	// Reset synchronizers
	// S-domain reset synchronizer for M_PRESETn generation
	reg [1:0] s_rst_sync;
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			s_rst_sync <= 2'b00;
		else
			s_rst_sync <= {s_rst_sync[0], 1'b1};
	end

	// M-domain reset synchronizer for M_PRESETn
	reg [1:0] m_rst_sync;
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn)
			m_rst_sync <= 2'b00;
		else
			m_rst_sync <= {m_rst_sync[0], 1'b1};
	end

	// M_PRESETn is derived from S_PRESETn synchronized to M domain
	// We use a simple synchronizer chain from S_PRESETn to M domain
	// But since M_PRESETn is an output, we need to generate it in M domain
	// based on S_PRESETn. We'll use a 2-stage synchronizer in M domain.
	
	// Actually, let's use a proper reset synchronizer.
	// S_PRESETn is active-low. We need to synchronize its release to M domain.
	// We'll create a synchronized version of S_PRESETn in M domain.
	
	reg [1:0] m_s_presetn_sync;
	always @(posedge M_APB_PCLK) begin
		m_s_presetn_sync <= {m_s_presetn_sync[0], S_PRESETn};
	end

	// M_PRESETn is active-low, so it's low when S_PRESETn is low (after sync)
	// We'll assert M_PRESETn when the synchronized S_PRESETn is low
	always @(posedge M_APB_PCLK) begin
		M_PRESETn <= m_s_presetn_sync[1];
	end

	// S-domain: Capture request when S_APB_PSEL is asserted
	// and not busy
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_req_pending <= 1'b0;
			s_req_valid <= 1'b0;
			s_req_ack <= 1'b0;
			s_done_valid <= 1'b0;
			s_done_ack <= 1'b0;
			s_rdata_reg <= {DW{1'b0}};
			s_slverr_reg <= 1'b0;
			s_setup <= 1'b0;
			s_access <= 1'b0;
			s_busy <= 1'b0;
			S_APB_PREADY <= 1'b0;
		end else begin
			// Handle done ack from M domain
			if (s_done_valid && s_done_ack) begin
				s_done_valid <= 1'b0;
				s_done_ack <= 1'b0;
			end

			// Handle request ack from M domain
			if (s_req_valid && s_req_ack) begin
				s_req_valid <= 1'b0;
				s_req_ack <= 1'b0;
			end

			// APB state machine
			if (!s_busy) begin
				if (S_APB_PSEL) begin
					s_setup <= 1'b1;
					s_busy <= 1'b1;
					// Capture request
					s_req_valid <= 1'b1;
					m_addr_reg <= S_APB_PADDR;
					m_write_reg <= S_APB_PWRITE;
					m_wdata_reg <= S_APB_PWDATA;
					m_wstrb_reg <= S_APB_PWSTRB;
					m_pprot_reg <= S_APB_PPROT;
				end else begin
					s_setup <= 1'b0;
					s_access <= 1'b0;
				end
			end else begin
				if (s_setup) begin
					if (S_APB_PENABLE) begin
						s_setup <= 1'b0;
						s_access <= 1'b1;
					end
				end else if (s_access) begin
					if (S_APB_PREADY) begin
						s_access <= 1'b0;
						s_busy <= 1'b0;
						// Capture response
						if (s_done_valid) begin
							s_rdata_reg <= m_rdata_reg;
							s_slverr_reg <= m_slverr_reg;
						end
					end
				end
			end

			// S_APB_PREADY is asserted when in access phase and done is valid
			if (s_access && s_done_valid)
				S_APB_PREADY <= 1'b1;
			else
				S_APB_PREADY <= 1'b0;
		end
	end

	// M-domain: APB state machine
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_req_valid <= 1'b0;
			m_req_ack <= 1'b0;
			m_done_valid <= 1'b0;
			m_done_ack <= 1'b0;
			m_rdata_reg <= {DW{1'b0}};
			m_slverr_reg <= 1'b0;
			m_setup <= 1'b0;
			m_access <= 1'b0;
			m_busy <= 1'b0;
			M_APB_PSEL <= 1'b0;
			M_APB_PENABLE <= 1'b0;
		end else begin
			// Handle request ack from S domain
			if (m_req_valid && m_req_ack) begin
				m_req_valid <= 1'b0;
				m_req_ack <= 1'b0;
			end

			// Handle done ack from S domain
			if (m_done_valid && m_done_ack) begin
				m_done_valid <= 1'b0;
				m_done_ack <= 1'b0;
			end

			// APB state machine
			if (!m_busy) begin
				if (m_req_valid) begin
					m_setup <= 1'b1;
					m_busy <= 1'b1;
					M_APB_PSEL <= 1'b1;
					M_APB_PENABLE <= 1'b0;
				end else begin
					m_setup <= 1'b0;
					m_access <= 1'b0;
					M_APB_PSEL <= 1'b0;
					M_APB_PENABLE <= 1'b0;
				end
			end else begin
				if (m_setup) begin
					m_access <= 1'b1;
					M_APB_PSEL <= 1'b1;
					M_APB_PENABLE <= 1'b1;
				end else if (m_access) begin
					if (M_APB_PREADY) begin
						m_access <= 1'b0;
						m_busy <= 1'b0;
						M_APB_PSEL <= 1'b0;
						M_APB_PENABLE <= 1'b0;
						// Capture response and signal done
						m_rdata_reg <= M_APB_PRDATA;
						m_slverr_reg <= M_APB_PSLVERR;
						m_done_valid <= 1'b1;
					end
				end
			end
		end
	end

	// Handshake: S-domain request to M-domain
	// s_req_valid is set in S domain, m_req_ack is set in M domain
	// We need to synchronize these across domains.
	
	// S-domain: m_req_ack synchronized to S domain
	reg [1:0] s_m_req_ack_sync;
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			s_m_req_ack_sync <= 2'b00;
		else
			s_m_req_ack_sync <= {s_m_req_ack_sync[0], m_req_ack};
	end
	assign s_req_ack = s_m_req_ack_sync[1];

	// M-domain: s_req_valid synchronized to M domain
	reg [1:0] m_s_req_valid_sync;
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn)
			m_s_req_valid_sync <= 2'b00;
		else
			m_s_req_valid_sync <= {m_s_req_valid_sync[0], s_req_valid};
	end
	assign m_req_valid = m_s_req_valid_sync[1];

	// Handshake: M-domain done to S-domain
	// m_done_valid is set in M domain, s_done_ack is set in S domain
	
	// S-domain: m_done_valid synchronized to S domain
	reg [1:0] s_m_done_valid_sync;
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			s_m_done_valid_sync <= 2'b00;
		else
			s_m_done_valid_sync <= {s_m_done_valid_sync[0], m_done_valid};
	end
	assign s_done_valid = s_m_done_valid_sync[1];

	// M-domain: s_done_ack synchronized to M domain
	reg [1:0] m_s_done_ack_sync;
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn)
			m_s_done_ack_sync <= 2'b00;
		else
			m_s_done_ack_sync <= {m_s_done_ack_sync[0], s_done_ack};
	end
	assign m_done_ack = m_s_done_ack_sync[1];

	// Data crossing: S to M (request data)
	// The data is captured in S domain and held stable.
	// We need to synchronize the data to M domain.
	// Since the data is held stable until ack, we can use a simple synchronizer
	// but for multi-bit data, we need to ensure coherence.
	// We'll use a 2-stage synchronizer for each bit, but that's not ideal for multi-bit.
	// Instead, we'll use a handshake-based approach where the data is captured
	// in M domain when the request is accepted.
	
	// Actually, the data is already registered in S domain (m_addr_reg, etc.)
	// and held stable. We can use a simple 2-stage synchronizer for each signal
	// since they are held stable during the transfer.
	
	// For OPT_REGISTERED, we add an extra register stage.
	// For simplicity, we'll use 2-stage synchronizers for all crossing signals.
	
	// S to M: address
	reg [AW-1:0] m_addr_sync1, m_addr_sync2;
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_addr_sync1 <= {AW{1'b0}};
			m_addr_sync2 <= {AW{1'b0}};
		end else begin
			m_addr_sync1 <= m_addr_reg;
			m_addr_sync2 <= m_addr_sync1;
		end
	end
	assign M_APB_PADDR = m_addr_sync2;

	// S to M: write
	reg m_write_sync1, m_write_sync2;
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_write_sync1 <= 1'b0;
			m_write_sync2 <= 1'b0;
		end else begin
			m_write_sync1 <= m_write_reg;
			m_write_sync2 <= m_write_sync1;
		end
	end
	assign M_APB_PWRITE = m_write_sync2;

	// S to M: wdata
	reg [DW-1:0] m_wdata_sync1, m_wdata_sync2;
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_wdata_sync1 <= {DW{1'b0}};
			m_wdata_sync2 <= {DW{1'b0}};
		end else begin
			m_wdata_sync1 <= m_wdata_reg;
			m_wdata_sync2 <= m_wdata_sync1;
		end
	end
	assign M_APB_PWDATA = m_wdata_sync2;

	// S to M: wstrb
	reg [DW/8-1:0] m_wstrb_sync1, m_wstrb_sync2;
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_wstrb_sync1 <= {DW/8{1'b0}};
			m_wstrb_sync2 <= {DW/8{1'b0}};
		end else begin
			m_wstrb_sync1 <= m_wstrb_reg;
			m_wstrb_sync2 <= m_wstrb_sync1;
		end
	end
	assign M_APB_PWSTRB = m_wstrb_sync2;

	// S to M: pprot
	reg [2:0] m_pprot_sync1, m_pprot_sync2;
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_pprot_sync1 <= 3'b000;
			m_pprot_sync2 <= 3'b000;
		end else begin
			m_pprot_sync1 <= m_pprot_reg;
			m_pprot_sync2 <= m_pprot_sync1;
		end
	end
	assign M_APB_PPROT = m_pprot_sync2;

	// M to S: rdata
	reg [DW-1:0] s_rdata_sync1, s_rdata_sync2;
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_rdata_sync1 <= {DW{1'b0}};
			s_rdata_sync2 <= {DW{1'b0}};
		end else begin
			s_rdata_sync1 <= m_rdata_reg;
			s_rdata_sync2 <= s_rdata_sync1;
		end
	end
	assign S_APB_PRDATA = s_rdata_sync2;

	// M to S: slverr
	reg s_slverr_sync1, s_slverr_sync2;
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_slverr_sync1 <= 1'b0;
			s_slverr_sync2 <= 1'b0;
		end else begin
			s_slverr_sync1 <= m_slverr_reg;
			s_slverr_sync2 <= s_slverr_sync1;
		end
	end
	assign S_APB_PSLVERR = s_slverr_sync2;

endmodule
