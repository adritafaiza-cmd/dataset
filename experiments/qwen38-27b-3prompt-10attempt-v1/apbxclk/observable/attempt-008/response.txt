module apbxclk #(
	parameter	C_APB_ADDR_WIDTH = 12,
	parameter	C_APB_DATA_WIDTH = 32,
	parameter [0:0]	OPT_REGISTERED = 1'b0,
	localparam	AW = C_APB_ADDR_WIDTH,
	localparam	DW = C_APB_DATA_WIDTH
) (
	input	wire			S_APB_PCLK, S_PRESETn,
	input	wire			S_APB_PSEL,
	input	wire			S_APB_PENABLE,
	output	reg			S_APB_PREADY,
	input	wire	[AW-1:0]	S_APB_PADDR,
	input	wire			S_APB_PWRITE,
	input	wire	[DW-1:0]	S_APB_PWDATA,
	input	wire	[DW/8-1:0]	S_APB_PWSTRB,
	input	wire	[2:0]		S_APB_PPROT,
	output	wire	[DW-1:0]	S_APB_PRDATA,
	output	wire			S_APB_PSLVERR,
	input	wire			M_APB_PCLK,
	output	reg			M_PRESETn,
	output	reg			M_APB_PSEL,
	output	reg			M_APB_PENABLE,
	input	wire			M_APB_PREADY,
	output	wire	[AW-1:0]	M_APB_PADDR,
	output	wire			M_APB_PWRITE,
	output	wire	[DW-1:0]	M_APB_PWDATA,
	output	wire	[DW/8-1:0]	M_APB_PWSTRB,
	output	wire	[2:0]		M_APB_PPROT,
	input	wire	[DW-1:0]	M_APB_PRDATA,
	input	wire			M_APB_PSLVERR
);

	// Slave-side state
	reg [1:0] s_state;
	localparam S_IDLE = 2'b00;
	localparam S_SETUP = 2'b01;
	localparam S_ACCESS = 2'b10;

	// Master-side state
	reg [1:0] m_state;
	localparam M_IDLE = 2'b00;
	localparam M_SETUP = 2'b01;
	localparam M_ACCESS = 2'b10;

	// Synchronization signals
	reg [1:0] s2m_req_sync;
	reg [1:0] m2s_done_sync;

	// Payload registers (master side)
	reg [AW-1:0] m_addr_reg;
	reg m_write_reg;
	reg [DW-1:0] m_wdata_reg;
	reg [DW/8-1:0] m_wstrb_reg;
	reg [2:0] m_pprot_reg;

	// Response registers (slave side)
	reg [DW-1:0] s_rdata_reg;
	reg s_pslverr_reg;

	// Master-side request pending
	reg m_req_pending;

	// Slave-side response pending
	reg s_resp_pending;

	// Synchronizer for request from slave to master
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn)
			s2m_req_sync <= 2'b00;
		else
			s2m_req_sync <= {s2m_req_sync[0], s_state == S_SETUP};
	end

	// Synchronizer for done from master to slave
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			m2s_done_sync <= 2'b00;
		else
			m2s_done_sync <= {m2s_done_sync[0], m_state == M_IDLE && m_req_pending};
	end

	// Master-side reset
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn)
			M_PRESETn <= 1'b0;
		else
			M_PRESETn <= 1'b1;
	end

	// Master-side state machine
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_state <= M_IDLE;
			M_APB_PSEL <= 1'b0;
			M_APB_PENABLE <= 1'b0;
			m_req_pending <= 1'b0;
		end else begin
			case (m_state)
				M_IDLE: begin
					M_APB_PSEL <= 1'b0;
					M_APB_PENABLE <= 1'b0;
					if (s2m_req_sync[1]) begin
						m_state <= M_SETUP;
						m_req_pending <= 1'b1;
					end
				end
				M_SETUP: begin
					M_APB_PSEL <= 1'b1;
					M_APB_PENABLE <= 1'b0;
					m_state <= M_ACCESS;
				end
				M_ACCESS: begin
					M_APB_PSEL <= 1'b1;
					M_APB_PENABLE <= 1'b1;
					if (M_APB_PREADY) begin
						m_state <= M_IDLE;
						m_req_pending <= 1'b0;
					end
				end
				default: m_state <= M_IDLE;
			endcase
		end
	end

	// Master-side payload registers
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_addr_reg <= {AW{1'b0}};
			m_write_reg <= 1'b0;
			m_wdata_reg <= {DW{1'b0}};
			m_wstrb_reg <= {(DW/8){1'b0}};
			m_pprot_reg <= 3'b000;
		end else if (m_state == M_IDLE && s2m_req_sync[1]) begin
			m_addr_reg <= S_APB_PADDR;
			m_write_reg <= S_APB_PWRITE;
			m_wdata_reg <= S_APB_PWDATA;
			m_wstrb_reg <= S_APB_PWSTRB;
			m_pprot_reg <= S_APB_PPROT;
		end
	end

	// Master-side outputs
	assign M_APB_PADDR = m_addr_reg;
	assign M_APB_PWRITE = m_write_reg;
	assign M_APB_PWDATA = m_wdata_reg;
	assign M_APB_PWSTRB = m_wstrb_reg;
	assign M_APB_PPROT = m_pprot_reg;

	// Slave-side state machine
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_state <= S_IDLE;
			S_APB_PREADY <= 1'b0;
			s_resp_pending <= 1'b0;
		end else begin
			case (s_state)
				S_IDLE: begin
					S_APB_PREADY <= 1'b0;
					if (S_APB_PSEL) begin
						s_state <= S_SETUP;
					end
				end
				S_SETUP: begin
					S_APB_PREADY <= 1'b0;
					s_state <= S_ACCESS;
				end
				S_ACCESS: begin
					if (m2s_done_sync[1]) begin
						S_APB_PREADY <= 1'b1;
						s_resp_pending <= 1'b1;
						s_state <= S_IDLE;
					end else begin
						S_APB_PREADY <= 1'b0;
					end
				end
				default: s_state <= S_IDLE;
			endcase
		end
	end

	// Slave-side response registers
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_rdata_reg <= {DW{1'b0}};
			s_pslverr_reg <= 1'b0;
		end else if (m2s_done_sync[1]) begin
			s_rdata_reg <= M_APB_PRDATA;
			s_pslverr_reg <= M_APB_PSLVERR;
		end
	end

	// Slave-side outputs
	assign S_APB_PRDATA = s_rdata_reg;
	assign S_APB_PSLVERR = s_pslverr_reg;

endmodule
