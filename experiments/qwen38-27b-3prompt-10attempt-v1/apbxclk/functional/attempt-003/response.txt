```verilog
module apbxclk #(
	parameter	C_APB_ADDR_WIDTH = 12,
	parameter	C_APB_DATA_WIDTH = 32,
	parameter [0:0]	OPT_REGISTERED = 1'b0,
	localparam	AW = C_APB_ADDR_WIDTH,
	localparam	DW = C_APB_DATA_WIDTH
) (
	input	wire			S_APB_PCLK, S_PRESETn,
	input	wire			S_APB_PSEL,
	input	wire			S_APB_PENABLE,
	output	reg			S_APB_PREADY,
	input	wire	[AW-1:0]	S_APB_PADDR,
	input	wire			S_APB_PWRITE,
	input	wire	[DW-1:0]	S_APB_PWDATA,
	input	wire	[DW/8-1:0]	S_APB_PWSTRB,
	input	wire	[2:0]		S_APB_PPROT,
	output	wire	[DW-1:0]	S_APB_PRDATA,
	output	wire			S_APB_PSLVERR,
	input	wire			M_APB_PCLK,
	output	reg			M_PRESETn,
	output	reg			M_APB_PSEL,
	output	reg			M_APB_PENABLE,
	input	wire			M_APB_PREADY,
	output	wire	[AW-1:0]	M_APB_PADDR,
	output	wire			M_APB_PWRITE,
	output	wire	[DW-1:0]	M_APB_PWDATA,
	output	wire	[DW/8-1:0]	M_APB_PWSTRB,
	output	wire	[2:0]		M_APB_PPROT,
	input	wire	[DW-1:0]	M_APB_PRDATA,
	input	wire			M_APB_PSLVERR
);

	// Synchronous reset for M domain
	reg [1:0] m_reset_sync;
	reg m_reset_n;

	always @(posedge M_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			m_reset_sync <= 2'b00;
			m_reset_n <= 1'b0;
		end else begin
			m_reset_sync <= {m_reset_sync[0], 1'b1};
			m_reset_n <= m_reset_sync[1];
		end
	end

	assign M_PRESETn = m_reset_n;

	// Handshake signals
	reg s_req;
	reg m_ack;
	reg m_req;
	reg s_ack;

	// S domain: detect new APB transaction start
	// A new transaction starts when PSEL is high and PENABLE is low (setup phase)
	// We need to capture the request when it's valid and not already in progress
	reg s_busy;

	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_busy <= 1'b0;
			s_req <= 1'b0;
			s_ack <= 1'b0;
			S_APB_PREADY <= 1'b0;
		end else begin
			// Detect start of a new APB transaction
			// PSEL high, PENABLE low means setup phase
			// We start a new transaction if not busy
			if (!s_busy && S_APB_PSEL && !S_APB_PENABLE) begin
				s_busy <= 1'b1;
				s_req <= 1'b1;
				S_APB_PREADY <= 1'b0;
			end else if (s_busy) begin
				// Wait for m_ack to complete the transaction
				if (m_ack) begin
					s_busy <= 1'b0;
					s_req <= 1'b0;
					S_APB_PREADY <= 1'b1;
				end
			end else begin
				S_APB_PREADY <= 1'b0;
			end
		end
	end

	// M domain: generate APB signals
	reg [AW-1:0] m_addr;
	reg m_write;
	reg [DW-1:0] m_wdata;
	reg [DW/8-1:0] m_wstrb;
	reg [2:0] m_pprot;
	reg m_sel;
	reg m_enable;

	always @(posedge M_APB_PCLK or negedge m_reset_n) begin
		if (!m_reset_n) begin
			m_sel <= 1'b0;
			m_enable <= 1'b0;
			m_req <= 1'b0;
			m_ack <= 1'b0;
			m_addr <= {AW{1'b0}};
			m_write <= 1'b0;
			m_wdata <= {DW{1'b0}};
			m_wstrb <= {(DW/8){1'b0}};
			m_pprot <= 3'b000;
		end else begin
			// Handle handshake
			if (s_req && !m_req) begin
				// New request from S domain
				m_req <= 1'b1;
				m_sel <= 1'b1;
				m_enable <= 1'b0;
				m_addr <= S_APB_PADDR;
				m_write <= S_APB_PWRITE;
				m_wdata <= S_APB_PWDATA;
				m_wstrb <= S_APB_PWSTRB;
				m_pprot <= S_APB_PPROT;
			end else if (m_req) begin
				// In progress
				if (!m_enable) begin
					// Setup phase, move to access phase
					m_enable <= 1'b1;
				end else begin
					// Access phase, wait for PREADY
					if (M_APB_PREADY) begin
						m_req <= 1'b0;
						m_sel <= 1'b0;
						m_enable <= 1'b0;
						m_ack <= 1'b1;
					end
				end
			end else begin
				m_ack <= 1'b0;
			end
		end
	end

	assign M_APB_PSEL = m_sel;
	assign M_APB_PENABLE = m_enable;
	assign M_APB_PADDR = m_addr;
	assign M_APB_PWRITE = m_write;
	assign M_APB_PWDATA = m_wdata;
	assign M_APB_PWSTRB = m_wstrb;
	assign M_APB_PPROT = m_pprot;

	// Response path: M domain to S domain
	// Capture read data and error when transaction completes
	reg [DW-1:0] s_rdata;
	reg s_pslverr;

	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_rdata <= {DW{1'b0}};
			s_pslverr <= 1'b0;
		end else begin
			// Capture response when m_ack is asserted
			if (m_ack) begin
				s_rdata <= M_APB_PRDATA;
				s_pslverr <= M_APB_PSLVERR;
			end
		end
	end

	assign S_APB_PRDATA = s_rdata;
	assign S_APB_PSLVERR = s_pslverr;

endmodule
