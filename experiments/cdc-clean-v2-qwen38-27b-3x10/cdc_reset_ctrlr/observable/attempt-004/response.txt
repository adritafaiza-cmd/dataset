module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o,
    input  b_isolate_ack_i
);

    // State encoding
    localparam S_IDLE      = 3'd0;
    localparam S_ISOLATE   = 3'd1;
    localparam S_CLEAR     = 3'd2;
    localparam S_RELEASE   = 3'd3;
    localparam S_DEISOLATE = 3'd4;

    // A-side state
    reg [2:0] a_state;
    reg       a_isolate_o_r;
    reg       a_clear_o_r;

    // B-side state
    reg [2:0] b_state;
    reg       b_isolate_o_r;
    reg       b_clear_o_r;

    // CDC: A -> B (A's progress)
    reg [2:0] a_state_to_b;
    reg [2:0] a_state_to_b_sync1;
    reg [2:0] a_state_to_b_sync2;

    // CDC: B -> A (B's progress)
    reg [2:0] b_state_to_a;
    reg [2:0] b_state_to_a_sync1;
    reg [2:0] b_state_to_a_sync2;

    // A-side outputs
    assign a_isolate_o = a_isolate_o_r;
    assign a_clear_o   = a_clear_o_r;

    // B-side outputs
    assign b_isolate_o = b_isolate_o_r;
    assign b_clear_o   = b_clear_o_r;

    // A-side state machine
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_state       <= S_IDLE;
            a_isolate_o_r <= 1'b0;
            a_clear_o_r   <= 1'b0;
            a_state_to_b  <= S_IDLE;
        end else begin
            case (a_state)
                S_IDLE: begin
                    a_isolate_o_r <= 1'b0;
                    a_clear_o_r   <= 1'b0;
                    if (a_clear_i || b_state_to_a_sync2 != S_IDLE) begin
                        a_state      <= S_ISOLATE;
                        a_isolate_o_r <= 1'b1;
                        a_state_to_b  <= S_ISOLATE;
                    end
                end
                S_ISOLATE: begin
                    a_isolate_o_r <= 1'b1;
                    a_clear_o_r   <= 1'b0;
                    if (a_isolate_ack_i && b_state_to_a_sync2 == S_ISOLATE) begin
                        a_state      <= S_CLEAR;
                        a_clear_o_r  <= 1'b1;
                        a_state_to_b <= S_CLEAR;
                    end
                end
                S_CLEAR: begin
                    a_isolate_o_r <= 1'b1;
                    a_clear_o_r   <= 1'b1;
                    if (a_clear_ack_i && b_state_to_a_sync2 == S_CLEAR) begin
                        a_state      <= S_RELEASE;
                        a_clear_o_r  <= 1'b0;
                        a_state_to_b <= S_RELEASE;
                    end
                end
                S_RELEASE: begin
                    a_isolate_o_r <= 1'b1;
                    a_clear_o_r   <= 1'b0;
                    if (b_state_to_a_sync2 == S_RELEASE) begin
                        a_state      <= S_DEISOLATE;
                        a_isolate_o_r <= 1'b0;
                        a_state_to_b  <= S_DEISOLATE;
                    end
                end
                S_DEISOLATE: begin
                    a_isolate_o_r <= 1'b0;
                    a_clear_o_r   <= 1'b0;
                    if (b_state_to_a_sync2 == S_IDLE) begin
                        a_state      <= S_IDLE;
                        a_state_to_b <= S_IDLE;
                    end
                end
                default: begin
                    a_state       <= S_IDLE;
                    a_isolate_o_r <= 1'b0;
                    a_clear_o_r   <= 1'b0;
                    a_state_to_b  <= S_IDLE;
                end
            endcase
        end
    end

    // B-side state machine
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_state       <= S_IDLE;
            b_isolate_o_r <= 1'b0;
            b_clear_o_r   <= 1'b0;
            b_state_to_a  <= S_IDLE;
        end else begin
            case (b_state)
                S_IDLE: begin
                    b_isolate_o_r <= 1'b0;
                    b_clear_o_r   <= 1'b0;
                    if (b_clear_i || a_state_to_b_sync2 != S_IDLE) begin
                        b_state      <= S_ISOLATE;
                        b_isolate_o_r <= 1'b1;
                        b_state_to_a  <= S_ISOLATE;
                    end
                end
                S_ISOLATE: begin
                    b_isolate_o_r <= 1'b1;
                    b_clear_o_r   <= 1'b0;
                    if (b_isolate_ack_i && a_state_to_b_sync2 == S_ISOLATE) begin
                        b_state      <= S_CLEAR;
                        b_clear_o_r  <= 1'b1;
                        b_state_to_a <= S_CLEAR;
                    end
                end
                S_CLEAR: begin
                    b_isolate_o_r <= 1'b1;
                    b_clear_o_r   <= 1'b1;
                    if (b_clear_ack_i && a_state_to_b_sync2 == S_CLEAR) begin
                        b_state      <= S_RELEASE;
                        b_clear_o_r  <= 1'b0;
                        b_state_to_a <= S_RELEASE;
                    end
                end
                S_RELEASE: begin
                    b_isolate_o_r <= 1'b1;
                    b_clear_o_r   <= 1'b0;
                    if (a_state_to_b_sync2 == S_RELEASE) begin
                        b_state      <= S_DEISOLATE;
                        b_isolate_o_r <= 1'b0;
                        b_state_to_a  <= S_DEISOLATE;
                    end
                end
                S_DEISOLATE: begin
                    b_isolate_o_r <= 1'b0;
                    b_clear_o_r   <= 1'b0;
                    if (a_state_to_b_sync2 == S_IDLE) begin
                        b_state      <= S_IDLE;
                        b_state_to_a <= S_IDLE;
                    end
                end
                default: begin
                    b_state       <= S_IDLE;
                    b_isolate_o_r <= 1'b0;
                    b_clear_o_r   <= 1'b0;
                    b_state_to_a  <= S_IDLE;
                end
            endcase
        end
    end

    // CDC synchronizers: A -> B
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            a_state_to_b_sync1 <= S_IDLE;
            a_state_to_b_sync2 <= S_IDLE;
        end else begin
            a_state_to_b_sync1 <= a_state_to_b;
            a_state_to_b_sync2 <= a_state_to_b_sync1;
        end
    end

    // CDC synchronizers: B -> A
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            b_state_to_a_sync1 <= S_IDLE;
            b_state_to_a_sync2 <= S_IDLE;
        end else begin
            b_state_to_a_sync1 <= b_state_to_a;
            b_state_to_a_sync2 <= b_state_to_a_sync1;
        end
    end

endmodule
