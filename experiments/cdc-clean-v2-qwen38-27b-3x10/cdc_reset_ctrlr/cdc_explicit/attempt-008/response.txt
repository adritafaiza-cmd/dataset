module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o,
    input  b_isolate_ack_i
);

    // State encoding
    localparam S_IDLE      = 3'd0;
    localparam S_ISOLATE   = 3'd1;
    localparam S_CLEAR     = 3'd2;
    localparam S_RELEASE   = 3'd3;
    localparam S_DONE      = 3'd4;

    // Internal signals for domain A
    reg [2:0] a_state;
    reg       a_isolate_reg;
    reg       a_clear_reg;
    reg       a_peer_isolate_ack;
    reg       a_peer_clear_ack;
    reg       a_peer_release;
    reg       a_req_isolate;
    reg       a_req_clear;
    reg       a_req_release;

    // Internal signals for domain B
    reg [2:0] b_state;
    reg       b_isolate_reg;
    reg       b_clear_reg;
    reg       b_peer_isolate_ack;
    reg       b_peer_clear_ack;
    reg       b_peer_release;
    reg       b_req_isolate;
    reg       b_req_clear;
    reg       b_req_release;

    // CDC synchronizers for A->B signals
    reg [SYNC_STAGES-1:0] a_to_b_isolate_sync;
    reg [SYNC_STAGES-1:0] a_to_b_clear_sync;
    reg [SYNC_STAGES-1:0] a_to_b_release_sync;

    // CDC synchronizers for B->A signals
    reg [SYNC_STAGES-1:0] b_to_a_isolate_sync;
    reg [SYNC_STAGES-1:0] b_to_a_clear_sync;
    reg [SYNC_STAGES-1:0] b_to_a_release_sync;

    // Synchronized versions
    wire a_to_b_isolate_synced = a_to_b_isolate_sync[SYNC_STAGES-1];
    wire a_to_b_clear_synced   = a_to_b_clear_sync[SYNC_STAGES-1];
    wire a_to_b_release_synced = a_to_b_release_sync[SYNC_STAGES-1];

    wire b_to_a_isolate_synced = b_to_a_isolate_sync[SYNC_STAGES-1];
    wire b_to_a_clear_synced   = b_to_a_clear_sync[SYNC_STAGES-1];
    wire b_to_a_release_synced = b_to_a_release_sync[SYNC_STAGES-1];

    // Synchronizer for A->B isolate
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni)
            a_to_b_isolate_sync <= {SYNC_STAGES{1'b0}};
        else
            a_to_b_isolate_sync <= {a_to_b_isolate_sync[SYNC_STAGES-2:0], a_isolate_reg};
    end

    // Synchronizer for A->B clear
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni)
            a_to_b_clear_sync <= {SYNC_STAGES{1'b0}};
        else
            a_to_b_clear_sync <= {a_to_b_clear_sync[SYNC_STAGES-2:0], a_clear_reg};
    end

    // Synchronizer for A->B release
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni)
            a_to_b_release_sync <= {SYNC_STAGES{1'b0}};
        else
            a_to_b_release_sync <= {a_to_b_release_sync[SYNC_STAGES-2:0], a_req_release};
    end

    // Synchronizer for B->A isolate
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni)
            b_to_a_isolate_sync <= {SYNC_STAGES{1'b0}};
        else
            b_to_a_isolate_sync <= {b_to_a_isolate_sync[SYNC_STAGES-2:0], b_isolate_reg};
    end

    // Synchronizer for B->A clear
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni)
            b_to_a_clear_sync <= {SYNC_STAGES{1'b0}};
        else
            b_to_a_clear_sync <= {b_to_a_clear_sync[SYNC_STAGES-2:0], b_clear_reg};
    end

    // Synchronizer for B->A release
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni)
            b_to_a_release_sync <= {SYNC_STAGES{1'b0}};
        else
            b_to_a_release_sync <= {b_to_a_release_sync[SYNC_STAGES-2:0], b_req_release};
    end

    // Domain A state machine
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_state            <= S_IDLE;
            a_isolate_reg      <= 1'b0;
            a_clear_reg        <= 1'b0;
            a_peer_isolate_ack <= 1'b0;
            a_peer_clear_ack   <= 1'b0;
            a_peer_release     <= 1'b0;
            a_req_isolate      <= 1'b0;
            a_req_clear        <= 1'b0;
            a_req_release      <= 1'b0;
        end else begin
            case (a_state)
                S_IDLE: begin
                    a_isolate_reg      <= 1'b0;
                    a_clear_reg        <= 1'b0;
                    a_peer_isolate_ack <= 1'b0;
                    a_peer_clear_ack   <= 1'b0;
                    a_peer_release     <= 1'b0;
                    a_req_isolate      <= 1'b0;
                    a_req_clear        <= 1'b0;
                    a_req_release      <= 1'b0;
                    if (a_clear_i) begin
                        a_state       <= S_ISOLATE;
                        a_isolate_reg <= 1'b1;
                        a_req_isolate <= 1'b1;
                    end
                end

                S_ISOLATE: begin
                    a_isolate_reg <= 1'b1;
                    a_req_isolate <= 1'b1;
                    // Wait for local isolate ack and peer isolate ack
                    if (a_isolate_ack_i && b_to_a_isolate_synced) begin
                        a_state            <= S_CLEAR;
                        a_clear_reg        <= 1'b1;
                        a_req_clear        <= 1'b1;
                        a_peer_isolate_ack <= 1'b1;
                    end
                end

                S_CLEAR: begin
                    a_isolate_reg <= 1'b1;
                    a_clear_reg   <= 1'b1;
                    a_req_clear   <= 1'b1;
                    // Wait for local clear ack and peer clear ack
                    if (a_clear_ack_i && b_to_a_clear_synced) begin
                        a_state          <= S_RELEASE;
                        a_req_release    <= 1'b1;
                        a_peer_clear_ack <= 1'b1;
                    end
                end

                S_RELEASE: begin
                    a_isolate_reg <= 1'b1;
                    a_clear_reg   <= 1'b0;
                    a_req_release <= 1'b1;
                    // Wait for peer release
                    if (b_to_a_release_synced) begin
                        a_state        <= S_DONE;
                        a_isolate_reg  <= 1'b0;
                        a_req_release  <= 1'b0;
                        a_peer_release <= 1'b1;
                    end
                end

                S_DONE: begin
                    a_isolate_reg <= 1'b0;
                    a_clear_reg   <= 1'b0;
                    a_req_release <= 1'b0;
                    a_state       <= S_IDLE;
                end

                default: begin
                    a_state       <= S_IDLE;
                    a_isolate_reg <= 1'b0;
                    a_clear_reg   <= 1'b0;
                end
            endcase
        end
    end

    // Domain B state machine
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_state            <= S_IDLE;
            b_isolate_reg      <= 1'b0;
            b_clear_reg        <= 1'b0;
            b_peer_isolate_ack <= 1'b0;
            b_peer_clear_ack   <= 1'b0;
            b_peer_release     <= 1'b0;
            b_req_isolate      <= 1'b0;
            b_req_clear        <= 1'b0;
            b_req_release      <= 1'b0;
        end else begin
            case (b_state)
                S_IDLE: begin
                    b_isolate_reg      <= 1'b0;
                    b_clear_reg        <= 1'b0;
                    b_peer_isolate_ack <= 1'b0;
                    b_peer_clear_ack   <= 1'b0;
                    b_peer_release     <= 1'b0;
                    b_req_isolate      <= 1'b0;
                    b_req_clear        <= 1'b0;
                    b_req_release      <= 1'b0;
                    if (b_clear_i) begin
                        b_state       <= S_ISOLATE;
                        b_isolate_reg <= 1'b1;
                        b_req_isolate <= 1'b1;
                    end
                end

                S_ISOLATE: begin
                    b_isolate_reg <= 1'b1;
                    b_req_isolate <= 1'b1;
                    // Wait for local isolate ack and peer isolate ack
                    if (b_isolate_ack_i && a_to_b_isolate_synced) begin
                        b_state            <= S_CLEAR;
                        b_clear_reg        <= 1'b1;
                        b_req_clear        <= 1'b1;
                        b_peer_isolate_ack <= 1'b1;
                    end
                end

                S_CLEAR: begin
                    b_isolate_reg <= 1'b1;
                    b_clear_reg   <= 1'b1;
                    b_req_clear   <= 1'b1;
                    // Wait for local clear ack and peer clear ack
                    if (b_clear_ack_i && a_to_b_clear_synced) begin
                        b_state          <= S_RELEASE;
                        b_req_release    <= 1'b1;
                        b_peer_clear_ack <= 1'b1;
                    end
                end

                S_RELEASE: begin
                    b_isolate_reg <= 1'b1;
                    b_clear_reg   <= 1'b0;
                    b_req_release <= 1'b1;
                    // Wait for peer release
                    if (a_to_b_release_synced) begin
                        b_state        <= S_DONE;
                        b_isolate_reg  <= 1'b0;
                        b_req_release  <= 1'b0;
                        b_peer_release <= 1'b1;
                    end
                end

                S_DONE: begin
                    b_isolate_reg <= 1'b0;
                    b_clear_reg   <= 1'b0;
                    b_req_release <= 1'b0;
                    b_state       <= S_IDLE;
                end

                default: begin
                    b_state       <= S_IDLE;
                    b_isolate_reg <= 1'b0;
                    b_clear_reg   <= 1'b0;
                end
            endcase
        end
    end

    // Outputs
    assign a_isolate_o = a_isolate_reg;
    assign a_clear_o   = a_clear_reg;
    assign b_isolate_o = b_isolate_reg;
    assign b_clear_o   = b_clear_reg;

endmodule
