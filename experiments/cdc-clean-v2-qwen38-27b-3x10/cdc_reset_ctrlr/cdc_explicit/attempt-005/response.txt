module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o,
    input  b_isolate_ack_i
);

    // State encoding
    localparam S_IDLE      = 3'd0;
    localparam S_ISOLATE   = 3'd1;
    localparam S_CLEAR     = 3'd2;
    localparam S_RELEASE   = 3'd3;
    localparam S_DONE      = 3'd4;

    // A-side state
    reg [2:0] a_state;
    reg [2:0] a_state_next;

    // B-side state
    reg [2:0] b_state;
    reg [2:0] b_state_next;

    // A-side outputs
    reg a_isolate_o_reg;
    reg a_clear_o_reg;

    // B-side outputs
    reg b_isolate_o_reg;
    reg b_clear_o_reg;

    // Synchronizers for A->B state
    reg [SYNC_STAGES-1:0] a_state_sync;
    wire [2:0] a_state_synced = a_state_sync[SYNC_STAGES-1:0];

    // Synchronizers for B->A state
    reg [SYNC_STAGES-1:0] b_state_sync;
    wire [2:0] b_state_synced = b_state_sync[SYNC_STAGES-1:0];

    // A-side state machine
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_state <= S_IDLE;
        end else begin
            a_state <= a_state_next;
        end
    end

    always @(*) begin
        a_state_next = a_state;
        case (a_state)
            S_IDLE: begin
                if (a_clear_i || b_clear_i) begin
                    a_state_next = S_ISOLATE;
                end
            end
            S_ISOLATE: begin
                if (a_isolate_ack_i && b_isolate_ack_i) begin
                    a_state_next = S_CLEAR;
                end
            end
            S_CLEAR: begin
                if (a_clear_ack_i && b_clear_ack_i) begin
                    a_state_next = S_RELEASE;
                end
            end
            S_RELEASE: begin
                // Wait for peer to observe our release
                if (b_state_synced == S_RELEASE || b_state_synced == S_DONE || b_state_synced == S_IDLE) begin
                    a_state_next = S_DONE;
                end
            end
            S_DONE: begin
                // Wait for peer to return to idle
                if (b_state_synced == S_IDLE) begin
                    a_state_next = S_IDLE;
                end
            end
            default: a_state_next = S_IDLE;
        endcase
    end

    // A-side outputs
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_isolate_o_reg <= 1'b0;
            a_clear_o_reg   <= 1'b0;
        end else begin
            case (a_state)
                S_ISOLATE: begin
                    a_isolate_o_reg <= 1'b1;
                    a_clear_o_reg   <= 1'b0;
                end
                S_CLEAR: begin
                    a_isolate_o_reg <= 1'b1;
                    a_clear_o_reg   <= 1'b1;
                end
                S_RELEASE: begin
                    a_isolate_o_reg <= 1'b1;
                    a_clear_o_reg   <= 1'b0;
                end
                S_DONE: begin
                    a_isolate_o_reg <= 1'b0;
                    a_clear_o_reg   <= 1'b0;
                end
                default: begin
                    a_isolate_o_reg <= 1'b0;
                    a_clear_o_reg   <= 1'b0;
                end
            endcase
        end
    end

    assign a_isolate_o = a_isolate_o_reg;
    assign a_clear_o   = a_clear_o_reg;

    // B-side state machine
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_state <= S_IDLE;
        end else begin
            b_state <= b_state_next;
        end
    end

    always @(*) begin
        b_state_next = b_state;
        case (b_state)
            S_IDLE: begin
                if (a_clear_i || b_clear_i) begin
                    b_state_next = S_ISOLATE;
                end
            end
            S_ISOLATE: begin
                if (a_isolate_ack_i && b_isolate_ack_i) begin
                    b_state_next = S_CLEAR;
                end
            end
            S_CLEAR: begin
                if (a_clear_ack_i && b_clear_ack_i) begin
                    b_state_next = S_RELEASE;
                end
            end
            S_RELEASE: begin
                // Wait for peer to observe our release
                if (a_state_synced == S_RELEASE || a_state_synced == S_DONE || a_state_synced == S_IDLE) begin
                    b_state_next = S_DONE;
                end
            end
            S_DONE: begin
                // Wait for peer to return to idle
                if (a_state_synced == S_IDLE) begin
                    b_state_next = S_IDLE;
                end
            end
            default: b_state_next = S_IDLE;
        endcase
    end

    // B-side outputs
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_isolate_o_reg <= 1'b0;
            b_clear_o_reg   <= 1'b0;
        end else begin
            case (b_state)
                S_ISOLATE: begin
                    b_isolate_o_reg <= 1'b1;
                    b_clear_o_reg   <= 1'b0;
                end
                S_CLEAR: begin
                    b_isolate_o_reg <= 1'b1;
                    b_clear_o_reg   <= 1'b1;
                end
                S_RELEASE: begin
                    b_isolate_o_reg <= 1'b1;
                    b_clear_o_reg   <= 1'b0;
                end
                S_DONE: begin
                    b_isolate_o_reg <= 1'b0;
                    b_clear_o_reg   <= 1'b0;
                end
                default: begin
                    b_isolate_o_reg <= 1'b0;
                    b_clear_o_reg   <= 1'b0;
                end
            endcase
        end
    end

    assign b_isolate_o = b_isolate_o_reg;
    assign b_clear_o   = b_clear_o_reg;

    // A->B state synchronizer
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            a_state_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            a_state_sync <= {a_state_sync[SYNC_STAGES-2:0], a_state};
        end
    end

    // B->A state synchronizer
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            b_state_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            b_state_sync <= {b_state_sync[SYNC_STAGES-2:0], b_state};
        end
    end

endmodule
