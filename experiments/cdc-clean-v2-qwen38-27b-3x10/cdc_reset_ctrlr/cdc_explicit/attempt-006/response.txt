```verilog
module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o,
    input  b_isolate_ack_i
);

    // State encoding
    localparam S_IDLE       = 3'd0;
    localparam S_ISOLATE    = 3'd1;
    localparam S_ISOLATE_W  = 3'd2;
    localparam S_CLEAR      = 3'd3;
    localparam S_CLEAR_W    = 3'd4;
    localparam S_RELEASE    = 3'd5;
    localparam S_RELEASE_W  = 3'd6;

    // Domain A state
    reg [2:0] a_state;
    reg       a_isolate_o_r;
    reg       a_clear_o_r;

    // Domain B state
    reg [2:0] b_state;
    reg       b_isolate_o_r;
    reg       b_clear_o_r;

    // CDC signals: A -> B
    // a_isolate_req: A requests isolation (goes high when A enters S_ISOLATE)
    // a_clear_req:   A requests clear (goes high when A enters S_CLEAR)
    // a_release_req: A requests release (goes high when A enters S_RELEASE)
    reg a_isolate_req_r;
    reg a_clear_req_r;
    reg a_release_req_r;

    // CDC signals: B -> A
    reg b_isolate_req_r;
    reg b_clear_req_r;
    reg b_release_req_r;

    // Synchronizers for A->B signals
    reg [SYNC_STAGES-1:0] a_isolate_req_sync;
    reg [SYNC_STAGES-1:0] a_clear_req_sync;
    reg [SYNC_STAGES-1:0] a_release_req_sync;

    // Synchronizers for B->A signals
    reg [SYNC_STAGES-1:0] b_isolate_req_sync;
    reg [SYNC_STAGES-1:0] b_clear_req_sync;
    reg [SYNC_STAGES-1:0] b_release_req_sync;

    // Synchronized versions
    wire a_isolate_req_synced = a_isolate_req_sync[SYNC_STAGES-1];
    wire a_clear_req_synced   = a_clear_req_sync[SYNC_STAGES-1];
    wire a_release_req_synced = a_release_req_sync[SYNC_STAGES-1];

    wire b_isolate_req_synced = b_isolate_req_sync[SYNC_STAGES-1];
    wire b_clear_req_synced   = b_clear_req_sync[SYNC_STAGES-1];
    wire b_release_req_synced = b_release_req_sync[SYNC_STAGES-1];

    // Synchronizers for acks: B acks to A
    reg [SYNC_STAGES-1:0] b_isolate_ack_sync;
    reg [SYNC_STAGES-1:0] b_clear_ack_sync;
    reg [SYNC_STAGES-1:0] b_release_ack_sync;

    // Synchronizers for acks: A acks to B
    reg [SYNC_STAGES-1:0] a_isolate_ack_sync;
    reg [SYNC_STAGES-1:0] a_clear_ack_sync;
    reg [SYNC_STAGES-1:0] a_release_ack_sync;

    wire b_isolate_ack_synced = b_isolate_ack_sync[SYNC_STAGES-1];
    wire b_clear_ack_synced   = b_clear_ack_sync[SYNC_STAGES-1];
    wire b_release_ack_synced = b_release_ack_sync[SYNC_STAGES-1];

    wire a_isolate_ack_synced = a_isolate_ack_sync[SYNC_STAGES-1];
    wire a_clear_ack_synced   = a_clear_ack_sync[SYNC_STAGES-1];
    wire a_release_ack_synced = a_release_ack_sync[SYNC_STAGES-1];

    // Synchronizer for local clear request from peer
    // Actually, the clear request from peer is the *_clear_i input
    // We need to synchronize the peer's clear request to know when to start
    // But the behavior says: when either side requests clear, first raise both isolate outputs
    // So each side independently detects its own clear request and initiates the sequence
    // The peer's clear request is synchronized to know when the peer has also requested

    // Synchronize peer clear requests
    reg [SYNC_STAGES-1:0] b_clear_i_sync;
    reg [SYNC_STAGES-1:0] a_clear_i_sync;

    wire b_clear_i_synced = b_clear_i_sync[SYNC_STAGES-1];
    wire a_clear_i_synced = a_clear_i_sync[SYNC_STAGES-1];

    // Synchronize peer isolate acks (local acks are already in local domain)
    // a_isolate_ack_i is in A domain, b_isolate_ack_i is in B domain
    // We need to synchronize b_isolate_ack_i to A domain and a_isolate_ack_i to B domain
    // But wait, the acks are generated by the external logic in each domain
    // a_isolate_ack_i is in A domain, so it's already synchronized for A
    // b_isolate_ack_i is in B domain, so it's already synchronized for B
    // We need to synchronize them across domains for the peer to know

    // Actually, let me reconsider the architecture.
    // Each domain has its own state machine.
    // The state machine in domain A:
    // - Detects a_clear_i
    // - Raises a_isolate_o
    // - Waits for a_isolate_ack_i (local) AND b_isolate_ack_synced (peer's isolate ack)
    // - Raises a_clear_o
    // - Waits for a_clear_ack_i (local) AND b_clear_ack_synced (peer's clear ack)
    // - Releases a_clear_o
    // - Waits for b_release_ack_synced (peer's release ack)
    // - Releases a_isolate_o
    // - Returns to idle

    // But we also need to synchronize the isolate/clear/release requests to the peer
    // so the peer knows to do the same.

    // Let me redesign:
    // Each side has a state machine that goes through the sequence.
    // The requests (isolate, clear, release) are synchronized to the peer.
    // The acks (isolate_ack, clear_ack, release_ack) are synchronized to the peer.
    // Each side waits for both local and peer acks before proceeding.

    // Domain A state machine
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_state <= S_IDLE;
            a_isolate_o_r <= 1'b0;
            a_clear_o_r <= 1'b0;
            a_isolate_req_r <= 1'b0;
            a_clear_req_r <= 1'b0;
            a_release_req_r <= 1'b0;
        end else begin
            case (a_state)
                S_IDLE: begin
                    a_isolate_o_r <= 1'b0;
                    a_clear_o_r <= 1'b0;
                    a_isolate_req_r <= 1'b0;
                    a_clear_req_r <= 1'b0;
                    a_release_req_r <= 1'b0;
                    if (a_clear_i) begin
                        a_state <= S_ISOLATE;
                        a_isolate_o_r <= 1'b1;
                        a_isolate_req_r <= 1'b1;
                    end
                end
                S_ISOLATE: begin
                    // Wait for both local and peer isolate acks
                    if (a_isolate_ack_i && b_isolate_ack_synced) begin
                        a_state <= S_CLEAR;
                        a_clear_o_r <= 1'b1;
                        a_clear_req_r <= 1'b1;
                    end
                end
                S_CLEAR: begin
                    // Wait for both local and peer clear acks
                    if (a_clear_ack_i && b_clear_ack_synced) begin
                        a_state <= S_RELEASE;
                        a_clear_o_r <= 1'b0;
                        a_release_req_r <= 1'b1;
                    end
                end
                S_RELEASE: begin
                    // Wait for peer release ack
                    if (b_release_ack_synced) begin
                        a_state <= S_IDLE;
                        a_isolate_o_r <= 1'b0;
                        a_isolate_req_r <= 1'b0;
                        a_clear_req_r <= 1'b0;
                        a_release_req_r <= 1'b0;
                    end
                end
                default: begin
                    a_state <= S_IDLE;
                    a_isolate_o_r <= 1'b0;
                    a_clear_o_r <= 1'b0;
                    a_isolate_req_r <= 1'b0;
                    a_clear_req_r <= 1'b0;
                    a_release_req_r <= 1'b0;
                end
            endcase
        end
    end

    // Domain B state machine
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_state <= S_IDLE;
            b_isolate_o_r <= 1'b0;
            b_clear_o_r <= 1'b0;
            b_isolate_req_r <= 1'b0;
            b_clear_req_r <= 1'b0;
            b_release_req_r <= 1'b0;
        end else begin
            case (b_state)
                S_IDLE: begin
                    b_isolate_o_r <= 1'b0;
                    b_clear_o_r <= 1'b0;
                    b_isolate_req_r <= 1'b0;
                    b_clear_req_r <= 1'b0;
                    b_release_req_r <= 1'b0;
                    if (b_clear_i) begin
                        b_state <= S_ISOLATE;
                        b_isolate_o_r <= 1'b1;
                        b_isolate_req_r <= 1'b1;
                    end
                end
                S_ISOLATE: begin
                    // Wait for both local and peer isolate acks
                    if (b_isolate_ack_i && a_isolate_ack_synced) begin
                        b_state <= S_CLEAR;
                        b_clear_o_r <= 1'b1;
                        b_clear_req_r <= 1'b1;
                    end
                end
                S_CLEAR: begin
                    // Wait for both local and peer clear acks
                    if (b_clear_ack_i && a_clear_ack_synced) begin
                        b_state <= S_RELEASE;
                        b_clear_o_r <= 1'b0;
                        b_release_req_r <= 1'b1;
                    end
                end
                S_RELEASE: begin
                    // Wait for peer release ack
                    if (a_release_ack_synced) begin
                        b_state <= S_IDLE;
                        b_isolate_o_r <= 1'b0;
                        b_isolate_req_r <= 1'b0;
                        b_clear_req_r <= 1'b0;
                        b_release_req_r <= 1'b0;
                    end
                end
                default: begin
                    b_state <= S_IDLE;
                    b_isolate_o_r <= 1'b0;
                    b_clear_o_r <= 1'b0;
                    b_isolate_req_r <= 1'b0;
                    b_clear_req_r <= 1'b0;
                    b_release_req_r <= 1'b0;
                end
            endcase
        end
    end

    // Synchronizers for A->B signals
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            a_isolate_req_sync <= {SYNC_STAGES{1'b0}};
            a_clear_req_sync <= {SYNC_STAGES{1'b0}};
            a_release_req_sync <= {SYNC_STAGES{1'b0}};
            a_isolate_ack_sync <= {SYNC_STAGES{1'b0}};
            a_clear_ack_sync <= {SYNC_STAGES{1'b0}};
            a_release_ack_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            a_isolate_req_sync <= {a_isolate_req_sync[SYNC_STAGES-2:0], a_isolate_req_r};
            a_clear_req_sync <= {a_clear_req_sync[SYNC_STAGES-2:0], a_clear_req_r};
            a_release_req_sync <= {a_release_req_sync[SYNC_STAGES-2:0], a_release_req_r};
            a_isolate_ack_sync <= {a_isolate_ack_sync[SYNC_STAGES-2:0], a_isolate_ack_i};
            a_clear_ack_sync <= {a_clear_ack_sync[SYNC_STAGES-2:0], a_clear_ack_i};
            a_release_ack_sync <= {a_release_ack_sync[SYNC_STAGES-2:0], 1'b1}; // A release ack is implicit when A completes
        end
    end

    // Synchronizers for B->A signals
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            b_isolate_req_sync <= {SYNC_STAGES{1'b0}};
            b_clear_req_sync <= {SYNC_STAGES{1'b0}};
            b_release_req_sync <= {SYNC_STAGES{1'b0}};
            b_isolate_ack_sync <= {SYNC_STAGES{1'b0}};
            b_clear_ack_sync <= {SYNC_STAGES{1'b0}};
            b_release_ack_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            b_isolate_req_sync <= {b_isolate_req_sync[SYNC_STAGES-2:0], b_isolate_req_r};
            b_clear_req_sync <= {b_clear_req_sync[SYNC_STAGES-2:0], b_clear_req_r};
            b_release_req_sync <= {b_release_req_sync[SYNC_STAGES-2:0], b_release_req_r};
            b_isolate_ack_sync <= {b_isolate_ack_sync[SYNC_STAGES-2:0], b_isolate_ack_i};
            b_clear_ack_sync <= {b_clear_ack_sync[SYNC_STAGES-2:0], b_clear_ack_i};
            b_release_ack_sync <= {b_release_ack_sync[SYNC_STAGES-2:0], 1'b1}; // B release ack is implicit when B completes
        end
    end

    // Synchronizers for peer clear requests
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            a_clear_i_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            a_clear_i_sync <= {a_clear_i_sync[SYNC_STAGES-2:0], a_clear_i};
        end
    end

    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            b_clear_i_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            b_clear_i_sync <= {b_clear_i_sync[SYNC_STAGES-2:0], b_clear_i};
        end
    end

    // Outputs
    assign a_isolate_o = a_isolate_o_r;
    assign a_clear_o = a_clear_o_r;
    assign b_isolate_o = b_isolate_o_r;
    assign b_clear_o = b_clear_o_r;

endmodule
