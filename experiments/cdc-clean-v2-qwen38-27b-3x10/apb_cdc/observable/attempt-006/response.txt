```verilog
module apb_cdc #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_pclk_i,
    input                      src_preset_ni,
    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,
    input  [ADDR_WIDTH-1:0]    src_paddr_i,
    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam AW = LOG_DEPTH;

    // Request FIFO (src -> dst)
    reg [ADDR_WIDTH-1:0] req_addr [0:DEPTH-1];
    reg [DATA_WIDTH-1:0] req_wdata [0:DEPTH-1];
    reg [DATA_WIDTH/8-1:0] req_strb [0:DEPTH-1];
    reg [2:0] req_prot [0:DEPTH-1];
    reg req_write [0:DEPTH-1];

    reg [AW:0] req_wr_ptr;
    reg [AW:0] req_rd_ptr;

    // Response FIFO (dst -> src)
    reg [DATA_WIDTH-1:0] rsp_data [0:DEPTH-1];
    reg rsp_err [0:DEPTH-1];

    reg [AW:0] rsp_wr_ptr;
    reg [AW:0] rsp_rd_ptr;

    // Gray code pointers for CDC
    wire [AW:0] req_wr_ptr_gray = req_wr_ptr ^ (req_wr_ptr >> 1);
    wire [AW:0] req_rd_ptr_gray = req_rd_ptr ^ (req_rd_ptr >> 1);
    wire [AW:0] rsp_wr_ptr_gray = rsp_wr_ptr ^ (rsp_wr_ptr >> 1);
    wire [AW:0] rsp_rd_ptr_gray = rsp_rd_ptr ^ (rsp_rd_ptr >> 1);

    // Synchronizers for request pointers
    reg [AW:0] req_wr_ptr_gray_dst_s1;
    reg [AW:0] req_wr_ptr_gray_dst_s2;
    reg [AW:0] req_rd_ptr_gray_src_s1;
    reg [AW:0] req_rd_ptr_gray_src_s2;

    // Synchronizers for response pointers
    reg [AW:0] rsp_wr_ptr_gray_src_s1;
    reg [AW:0] rsp_wr_ptr_gray_src_s2;
    reg [AW:0] rsp_rd_ptr_gray_dst_s1;
    reg [AW:0] rsp_rd_ptr_gray_dst_s2;

    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            req_wr_ptr_gray_dst_s1 <= {AW+1{1'b0}};
            req_wr_ptr_gray_dst_s2 <= {AW+1{1'b0}};
        end else begin
            req_wr_ptr_gray_dst_s1 <= req_wr_ptr_gray;
            req_wr_ptr_gray_dst_s2 <= req_wr_ptr_gray_dst_s1;
        end
    end

    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            req_rd_ptr_gray_src_s1 <= {AW+1{1'b0}};
            req_rd_ptr_gray_src_s2 <= {AW+1{1'b0}};
        end else begin
            req_rd_ptr_gray_src_s1 <= req_rd_ptr_gray;
            req_rd_ptr_gray_src_s2 <= req_rd_ptr_gray_src_s1;
        end
    end

    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            rsp_wr_ptr_gray_src_s1 <= {AW+1{1'b0}};
            rsp_wr_ptr_gray_src_s2 <= {AW+1{1'b0}};
        end else begin
            rsp_wr_ptr_gray_src_s1 <= rsp_wr_ptr_gray;
            rsp_wr_ptr_gray_src_s2 <= rsp_wr_ptr_gray_src_s1;
        end
    end

    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            rsp_rd_ptr_gray_dst_s1 <= {AW+1{1'b0}};
            rsp_rd_ptr_gray_dst_s2 <= {AW+1{1'b0}};
        end else begin
            rsp_rd_ptr_gray_dst_s1 <= rsp_rd_ptr_gray;
            rsp_rd_ptr_gray_dst_s2 <= rsp_rd_ptr_gray_dst_s1;
        end
    end

    // Convert gray to binary
    function [AW:0] gray2bin;
        input [AW:0] gray;
        reg [AW:0] bin;
        integer i;
        begin
            bin[AW] = gray[AW];
            for (i = AW-1; i >= 0; i = i - 1)
                bin[i] = bin[i+1] ^ gray[i];
            gray2bin = bin;
        end
    endfunction

    wire [AW:0] req_wr_ptr_dst = gray2bin(req_wr_ptr_gray_dst_s2);
    wire [AW:0] req_rd_ptr_src = gray2bin(req_rd_ptr_gray_src_s2);
    wire [AW:0] rsp_wr_ptr_src = gray2bin(rsp_wr_ptr_gray_src_s2);
    wire [AW:0] rsp_rd_ptr_dst = gray2bin(rsp_rd_ptr_gray_dst_s2);

    // FIFO full/empty conditions
    wire req_full = (req_wr_ptr[AW] != req_rd_ptr_src[AW]) &&
                    (req_wr_ptr[AW-1:0] == req_rd_ptr_src[AW-1:0]);
    wire req_empty = (req_wr_ptr == req_rd_ptr);

    wire rsp_full = (rsp_wr_ptr[AW] != rsp_rd_ptr_dst[AW]) &&
                    (rsp_wr_ptr[AW-1:0] == rsp_rd_ptr_dst[AW-1:0]);
    wire rsp_empty = (rsp_wr_ptr == rsp_rd_ptr);

    // Source side: APB slave interface
    reg src_pready_r;
    reg [DATA_WIDTH-1:0] src_prdata_r;
    reg src_pslverr_r;

    // Detect setup phase: PSEL high, PENABLE low
    wire src_setup = src_psel_i && !src_penable_i;
    // Detect access phase: PSEL high, PENABLE high
    wire src_access = src_psel_i && src_penable_i;

    // Can we accept a new request?
    wire can_accept = src_setup && !req_full;

    // Write to request FIFO
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            req_wr_ptr <= {AW+1{1'b0}};
        end else if (can_accept) begin
            req_addr[req_wr_ptr[AW-1:0]] <= src_paddr_i;
            req_wdata[req_wr_ptr[AW-1:0]] <= src_pwdata_i;
            req_strb[req_wr_ptr[AW-1:0]] <= src_pstrb_i;
            req_prot[req_wr_ptr[AW-1:0]] <= src_pprot_i;
            req_write[req_wr_ptr[AW-1:0]] <= src_pwrite_i;
            req_wr_ptr <= req_wr_ptr + 1'b1;
        end
    end

    // Read from response FIFO
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            rsp_rd_ptr <= {AW+1{1'b0}};
            src_pready_r <= 1'b0;
            src_prdata_r <= {DATA_WIDTH{1'b0}};
            src_pslverr_r <= 1'b0;
        end else begin
            if (!rsp_empty) begin
                src_prdata_r <= rsp_data[rsp_rd_ptr[AW-1:0]];
                src_pslverr_r <= rsp_err[rsp_rd_ptr[AW-1:0]];
                rsp_rd_ptr <= rsp_rd_ptr + 1'b1;
            end
            // Pready is high when we have a response ready and are in access phase
            // Actually, pready should be asserted when the response is available
            // We assert pready when we have a response and the source is in access phase
            // But we need to be careful: pready should be stable during the access phase
            // Let's assert pready when rsp is not empty
            src_pready_r <= !rsp_empty;
        end
    end

    assign src_pready_o = src_pready_r;
    assign src_prdata_o = src_prdata_r;
    assign src_pslverr_o = src_pslverr_r;

    // Destination side: APB master interface
    reg dst_psel_r;
    reg dst_penable_r;
    reg dst_pwrite_r;
    reg [ADDR_WIDTH-1:0] dst_paddr_r;
    reg [DATA_WIDTH-1:0] dst_pwdata_r;
    reg [DATA_WIDTH/8-1:0] dst_pstrb_r;
    reg [2:0] dst_pprot_r;

    // State machine for destination APB master
    // States: IDLE, SETUP, ACCESS
    localparam S_IDLE = 2'b00;
    localparam S_SETUP = 2'b01;
    localparam S_ACCESS = 2'b10;

    reg [1:0] dst_state;

    // Read from request FIFO
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            req_rd_ptr <= {AW+1{1'b0}};
            dst_state <= S_IDLE;
            dst_psel_r <= 1'b0;
            dst_penable_r <= 1'b0;
            dst_pwrite_r <= 1'b0;
            dst_paddr_r <= {ADDR_WIDTH{1'b0}};
            dst_pwdata_r <= {DATA_WIDTH{1'b0}};
            dst_pstrb_r <= {DATA_WIDTH/8{1'b0}};
            dst_pprot_r <= 3'b0;
        end else begin
            case (dst_state)
                S_IDLE: begin
                    dst_psel_r <= 1'b0;
                    dst_penable_r <= 1'b0;
                    if (!req_empty) begin
                        dst_paddr_r <= req_addr[req_rd_ptr[AW-1:0]];
                        dst_pwdata_r <= req_wdata[req_rd_ptr[AW-1:0]];
                        dst_pstrb_r <= req_strb[req_rd_ptr[AW-1:0]];
                        dst_pprot_r <= req_prot[req_rd_ptr[AW-1:0]];
                        dst_pwrite_r <= req_write[req_rd_ptr[AW-1:0]];
                        dst_state <= S_SETUP;
                    end
                end
                S_SETUP: begin
                    dst_psel_r <= 1'b1;
                    dst_penable_r <= 1'b0;
                    dst_state <= S_ACCESS;
                end
                S_ACCESS: begin
                    dst_psel_r <= 1'b1;
                    dst_penable_r <= 1'b1;
                    if (dst_pready_i) begin
                        // Write response to FIFO
                        if (!rsp_full) begin
                            rsp_data[rsp_wr_ptr[AW-1:0]] <= dst_prdata_i;
                            rsp_err[rsp_wr_ptr[AW-1:0]] <= dst_pslverr_i;
                            rsp_wr_ptr <= rsp_wr_ptr + 1'b1;
                        end
                        // Advance request read pointer
                        req_rd_ptr <= req_rd_ptr + 1'b1;
                        dst_state <= S_IDLE;
                    end
                end
                default: begin
                    dst_state <= S_IDLE;
                end
            endcase
        end
    end

    assign dst_psel_o = dst_psel_r;
    assign dst_penable_o = dst_penable_r;
    assign dst_pwrite_o = dst_pwrite_r;
    assign dst_paddr_o = dst_paddr_r;
    assign dst_pwdata_o = dst_pwdata_r;
    assign dst_pstrb_o = dst_pstrb_r;
    assign dst_pprot_o = dst_pprot_r;

endmodule
