```verilog
module apb_cdc #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_pclk_i,
    input                      src_preset_ni,
    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,
    input  [ADDR_WIDTH-1:0]    src_paddr_i,
    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam AW = LOG_DEPTH;

    // Source domain state
    reg  [ADDR_WIDTH-1:0] src_addr_reg;
    reg  [DATA_WIDTH-1:0] src_wdata_reg;
    reg  [DATA_WIDTH/8-1:0] src_strb_reg;
    reg  [2:0] src_pprot_reg;
    reg  src_write_reg;
    reg  src_valid_reg;
    reg  src_busy_reg;

    // Destination domain state
    reg  [ADDR_WIDTH-1:0] dst_addr_reg;
    reg  [DATA_WIDTH-1:0] dst_wdata_reg;
    reg  [DATA_WIDTH/8-1:0] dst_strb_reg;
    reg  [2:0] dst_pprot_reg;
    reg  dst_write_reg;
    reg  dst_valid_reg;
    reg  dst_busy_reg;
    reg  dst_phase_reg; // 0: setup, 1: access

    // FIFO pointers
    reg [AW-1:0] src_wr_ptr;
    reg [AW-1:0] dst_rd_ptr;

    // Synchronization signals
    // src_valid -> dst: 2-flop sync
    reg [1:0] src_valid_sync;
    // dst_done -> src: 2-flop sync
    reg [1:0] dst_done_sync;

    // Data path: use FIFO to store payload
    // For simplicity with LOG_DEPTH, we use a simple register-based approach
    // Since one outstanding transaction is enough, we can use a single buffer
    // but the spec mentions FIFO, so let's implement a small FIFO

    // Actually, with one outstanding transaction, we can simplify:
    // Source captures request, sends valid to dst
    // Dst captures, performs APB, sends done back to src
    // Src returns data

    // Let's use a simpler approach: single buffer with handshake

    // Source side: capture request when src_psel & ~src_penable & ~src_busy
    wire src_req_valid = src_psel_i & ~src_penable_i & ~src_busy_reg;

    // Destination side: capture when dst_valid_reg & ~dst_busy_reg
    // Actually, let's use a proper handshake

    // Simplified: 
    // 1. Src sees request, latches it, sets src_valid_reg
    // 2. Dst sees src_valid (synced), latches data, starts APB
    // 3. Dst completes APB, sets dst_done
    // 4. Src sees dst_done (synced), returns data, clears src_valid

    // But we need to handle the case where dst is not ready
    // Let's use a proper FIFO-based approach

    // For LOG_DEPTH=1, we have 1 entry
    // Let's implement a simple 1-entry FIFO

    // Source writes to FIFO when it has a request
    // Destination reads from FIFO when it's ready

    // FIFO: src_wr_ptr, dst_rd_ptr
    // Valid when src_wr_ptr != dst_rd_ptr

    // But we need to sync the pointers

    // Let's use a different approach: 
    // Source latches request, asserts src_valid
    // Dst syncs src_valid, when it sees it, latches data from a shared register
    // But shared register across domains is not safe

    // Better: use a FIFO for data

    // Let's implement a simple FIFO with LOG_DEPTH entries

    // For now, let's assume LOG_DEPTH=1 and use a single buffer
    // with proper handshake

    // Source domain
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_addr_reg  <= {ADDR_WIDTH{1'b0}};
            src_wdata_reg <= {DATA_WIDTH{1'b0}};
            src_strb_reg  <= {(DATA_WIDTH/8){1'b0}};
            src_pprot_reg <= 3'b000;
            src_write_reg <= 1'b0;
            src_valid_reg <= 1'b0;
            src_busy_reg  <= 1'b0;
            src_wr_ptr    <= {AW{1'b0}};
        end else begin
            // Capture request
            if (src_req_valid) begin
                src_addr_reg  <= src_paddr_i;
                src_wdata_reg <= src_pwdata_i;
                src_strb_reg  <= src_pstrb_i;
                src_pprot_reg <= src_pprot_i;
                src_write_reg <= src_pwrite_i;
                src_valid_reg <= 1'b1;
                src_busy_reg  <= 1'b1;
                src_wr_ptr    <= src_wr_ptr + 1'b1;
            end

            // When dst_done is synced back, clear valid and busy
            if (dst_done_sync[1]) begin
                src_valid_reg <= 1'b0;
                src_busy_reg  <= 1'b0;
            end
        end
    end

    // Source outputs
    assign src_pready_o  = ~src_busy_reg;
    assign src_prdata_o  = dst_prdata_sync[1];
    assign src_pslverr_o = dst_pslverr_sync[1];

    // Sync dst_prdata and dst_pslverr to src domain
    reg [DATA_WIDTH-1:0] dst_prdata_sync [0:1];
    reg                   dst_pslverr_sync [0:1];

    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            dst_prdata_sync[0]  <= {DATA_WIDTH{1'b0}};
            dst_prdata_sync[1]  <= {DATA_WIDTH{1'b0}};
            dst_pslverr_sync[0] <= 1'b0;
            dst_pslverr_sync[1] <= 1'b0;
        end else begin
            dst_prdata_sync[0]  <= dst_prdata_i;
            dst_prdata_sync[1]  <= dst_prdata_sync[0];
            dst_pslverr_sync[0] <= dst_pslverr_i;
            dst_pslverr_sync[1] <= dst_pslverr_sync[0];
        end
    end

    // Sync src_valid to dst domain
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            src_valid_sync[0] <= 1'b0;
            src_valid_sync[1] <= 1'b0;
        end else begin
            src_valid_sync[0] <= src_valid_reg;
            src_valid_sync[1] <= src_valid_sync[0];
        end
    end

    // Sync dst_done to src domain
    reg dst_done_reg;
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            dst_done_reg <= 1'b0;
        end else begin
            // Set done when APB transfer completes
            if (dst_phase_reg == 1'b1 && dst_pready_i) begin
                dst_done_reg <= 1'b1;
            end else if (src_valid_sync[1]) begin
                dst_done_reg <= 1'b0;
            end
        end
    end

    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            dst_done_sync[0] <= 1'b0;
            dst_done_sync[1] <= 1'b0;
        end else begin
            dst_done_sync[0] <= dst_done_reg;
            dst_done_sync[1] <= dst_done_sync[0];
        end
    end

    // Destination domain
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            dst_addr_reg  <= {ADDR_WIDTH{1'b0}};
            dst_wdata_reg <= {DATA_WIDTH{1'b0}};
            dst_strb_reg  <= {(DATA_WIDTH/8){1'b0}};
            dst_pprot_reg <= 3'b000;
            dst_write_reg <= 1'b0;
            dst_valid_reg <= 1'b0;
            dst_busy_reg  <= 1'b0;
            dst_phase_reg <= 1'b0;
            dst_rd_ptr    <= {AW{1'b0}};
        end else begin
            // When src_valid is synced and we're not busy, capture data
            if (src_valid_sync[1] && ~dst_busy_reg) begin
                dst_addr_reg  <= src_addr_reg;
                dst_wdata_reg <= src_wdata_reg;
                dst_strb_reg  <= src_strb_reg;
                dst_pprot_reg <= src_pprot_reg;
                dst_write_reg <= src_write_reg;
                dst_valid_reg <= 1'b1;
                dst_busy_reg  <= 1'b1;
                dst_phase_reg <= 1'b0;
                dst_rd_ptr    <= dst_rd_ptr + 1'b1;
            end

            // APB state machine
            if (dst_busy_reg) begin
                if (dst_phase_reg == 1'b0) begin
                    // Setup phase: wait for dst_pready
                    if (dst_pready_i) begin
                        dst_phase_reg <= 1'b1;
                    end
                end else begin
                    // Access phase: wait for dst_pready
                    if (dst_pready_i) begin
                        dst_busy_reg  <= 1'b0;
                        dst_valid_reg <= 1'b0;
                        dst_phase_reg <= 1'b0;
                    end
                end
            end
        end
    end

    // Destination outputs
    assign dst_psel_o   = dst_valid_reg;
    assign dst_penable_o = dst_phase_reg;
    assign dst_pwrite_o = dst_write_reg;
    assign dst_paddr_o  = dst_addr_reg;
    assign dst_pwdata_o = dst_wdata_reg;
    assign dst_pstrb_o  = dst_strb_reg;
    assign dst_pprot_o  = dst_pprot_reg;

endmodule
