```verilog
module apb_cdc #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_pclk_i,
    input                      src_preset_ni,
    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,
    input  [ADDR_WIDTH-1:0]    src_paddr_i,
    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam AW    = LOG_DEPTH;

    // Source-side FIFO
    reg [DATA_WIDTH-1:0] src_fifo_data [0:DEPTH-1];
    reg [ADDR_WIDTH-1:0] src_fifo_addr [0:DEPTH-1];
    reg [DATA_WIDTH/8-1:0] src_fifo_strb [0:DEPTH-1];
    reg [2:0] src_fifo_prot [0:DEPTH-1];
    reg src_fifo_write_flag [0:DEPTH-1];
    reg src_fifo_read_flag  [0:DEPTH-1];

    reg [AW-1:0] src_wr_ptr;
    reg [AW-1:0] src_rd_ptr;

    // Destination-side FIFO
    reg [DATA_WIDTH-1:0] dst_fifo_data [0:DEPTH-1];
    reg [ADDR_WIDTH-1:0] dst_fifo_addr [0:DEPTH-1];
    reg [DATA_WIDTH/8-1:0] dst_fifo_strb [0:DEPTH-1];
    reg [2:0] dst_fifo_prot [0:DEPTH-1];
    reg dst_fifo_write_flag [0:DEPTH-1];
    reg dst_fifo_read_flag  [0:DEPTH-1];

    reg [AW-1:0] dst_wr_ptr;
    reg [AW-1:0] dst_rd_ptr;

    // Synchronization signals
    reg [AW:0] src_wr_ptr_sync0, src_wr_ptr_sync1, src_wr_ptr_sync2;
    reg [AW:0] dst_rd_ptr_sync0, dst_rd_ptr_sync1, dst_rd_ptr_sync2;
    reg [AW:0] dst_wr_ptr_sync0, dst_wr_ptr_sync1, dst_wr_ptr_sync2;
    reg [AW:0] src_rd_ptr_sync0, src_rd_ptr_sync1, src_rd_ptr_sync2;

    // Source-side state
    reg src_busy;
    reg src_pready_r;
    reg [DATA_WIDTH-1:0] src_prdata_r;
    reg src_pslverr_r;

    // Destination-side state
    reg dst_busy;
    reg dst_psel_r;
    reg dst_penable_r;
    reg dst_pwrite_r;
    reg [ADDR_WIDTH-1:0] dst_paddr_r;
    reg [DATA_WIDTH-1:0] dst_pwdata_r;
    reg [DATA_WIDTH/8-1:0] dst_pstrb_r;
    reg [2:0] dst_pprot_r;

    // Synchronization of dst_pready
    reg dst_pready_sync0, dst_pready_sync1, dst_pready_sync2;
    reg dst_pslverr_sync0, dst_pslverr_sync1, dst_pslverr_sync2;
    reg [DATA_WIDTH-1:0] dst_prdata_sync0, dst_prdata_sync1, dst_prdata_sync2;

    // Source-side: detect APB access phase
    wire src_access = src_psel_i & src_penable_i;
    wire src_idle   = !src_psel_i;

    // Source-side: FIFO full/empty
    wire [AW:0] src_wr_ptr_full = {src_wr_ptr, 1'b0};
    wire [AW:0] src_rd_ptr_full = {src_rd_ptr, 1'b0};
    wire src_fifo_full  = (src_wr_ptr_full == src_rd_ptr_full) && (src_wr_ptr[AW-1] != src_rd_ptr[AW-1]);
    wire src_fifo_empty = (src_wr_ptr_full == src_rd_ptr_full) && (src_wr_ptr[AW-1] == src_rd_ptr[AW-1]);

    // Destination-side: FIFO full/empty
    wire [AW:0] dst_wr_ptr_full = {dst_wr_ptr, 1'b0};
    wire [AW:0] dst_rd_ptr_full = {dst_rd_ptr, 1'b0};
    wire dst_fifo_full  = (dst_wr_ptr_full == dst_rd_ptr_full) && (dst_wr_ptr[AW-1] != dst_rd_ptr[AW-1]);
    wire dst_fifo_empty = (dst_wr_ptr_full == dst_rd_ptr_full) && (dst_wr_ptr[AW-1] == dst_rd_ptr[AW-1]);

    // Source-side: can accept new request
    wire src_can_accept = src_idle && !src_fifo_full;

    // Source-side: write to FIFO
    wire src_fifo_wr_en = src_access && src_can_accept;

    // Source-side: read from FIFO (when dst is ready to take)
    wire src_fifo_rd_en = !src_fifo_empty && !dst_fifo_full;

    // Destination-side: write to FIFO
    wire dst_fifo_wr_en = src_fifo_rd_en;

    // Destination-side: read from FIFO (when dst transfer completes)
    wire dst_fifo_rd_en = dst_busy && dst_pready_i;

    // Source-side: ready to source
    assign src_pready_o = src_pready_r;
    assign src_prdata_o = src_prdata_r;
    assign src_pslverr_o = src_pslverr_r;

    // Destination-side: outputs
    assign dst_psel_o    = dst_psel_r;
    assign dst_penable_o = dst_penable_r;
    assign dst_pwrite_o  = dst_pwrite_r;
    assign dst_paddr_o   = dst_paddr_r;
    assign dst_pwdata_o  = dst_pwdata_r;
    assign dst_pstrb_o   = dst_pstrb_r;
    assign dst_pprot_o   = dst_pprot_r;

    // Source clock domain
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_wr_ptr    <= {AW{1'b0}};
            src_rd_ptr    <= {AW{1'b0}};
            src_busy      <= 1'b0;
            src_pready_r  <= 1'b0;
            src_prdata_r  <= {DATA_WIDTH{1'b0}};
            src_pslverr_r <= 1'b0;
            for (integer i = 0; i < DEPTH; i = i + 1) begin
                src_fifo_write_flag[i] <= 1'b0;
                src_fifo_read_flag[i]  <= 1'b0;
            end
        end else begin
            // Update FIFO write pointer
            if (src_fifo_wr_en) begin
                src_fifo_data[src_wr_ptr[AW-1:0]] <= src_pwdata_i;
                src_fifo_addr[src_wr_ptr[AW-1:0]] <= src_paddr_i;
                src_fifo_strb[src_wr_ptr[AW-1:0]] <= src_pstrb_i;
                src_fifo_prot[src_wr_ptr[AW-1:0]] <= src_pprot_i;
                src_fifo_write_flag[src_wr_ptr[AW-1:0]] <= 1'b1;
                src_wr_ptr <= src_wr_ptr + 1'b1;
            end

            // Update FIFO read pointer
            if (src_fifo_rd_en) begin
                src_fifo_read_flag[src_rd_ptr[AW-1:0]] <= 1'b0;
                src_rd_ptr <= src_rd_ptr + 1'b1;
            end

            // Source-side state machine
            if (src_access) begin
                // In access phase
                if (src_pready_r) begin
                    // Transfer complete, go idle
                    src_busy     <= 1'b0;
                    src_pready_r <= 1'b0;
                end
            end else begin
                // In setup phase or idle
                if (src_psel_i && !src_fifo_full) begin
                    // Setup phase, can proceed to access
                    src_busy     <= 1'b1;
                    src_pready_r <= 1'b0;
                end else begin
                    src_pready_r <= 1'b0;
                end
            end
        end
    end

    // Synchronize source write pointer to destination domain
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            src_wr_ptr_sync0 <= {AW+1{1'b0}};
            src_wr_ptr_sync1 <= {AW+1{1'b0}};
            src_wr_ptr_sync2 <= {AW+1{1'b0}};
        end else begin
            src_wr_ptr_sync0 <= {src_wr_ptr, 1'b0};
            src_wr_ptr_sync1 <= src_wr_ptr_sync0;
            src_wr_ptr_sync2 <= src_wr_ptr_sync1;
        end
    end

    // Synchronize destination read pointer to source domain
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            dst_rd_ptr_sync0 <= {AW+1{1'b0}};
            dst_rd_ptr_sync1 <= {AW+1{1'b0}};
            dst_rd_ptr_sync2 <= {AW+1{1'b0}};
        end else begin
            dst_rd_ptr_sync0 <= {dst_rd_ptr, 1'b0};
            dst_rd_ptr_sync1 <= dst_rd_ptr_sync0;
            dst_rd_ptr_sync2 <= dst_rd_ptr_sync1;
        end
    end

    // Synchronize destination write pointer to source domain
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            dst_wr_ptr_sync0 <= {AW+1{1'b0}};
            dst_wr_ptr_sync1 <= {AW+1{1'b0}};
            dst_wr_ptr_sync2 <= {AW+1{1'b0}};
        end else begin
            dst_wr_ptr_sync0 <= {dst_wr_ptr, 1'b0};
            dst_wr_ptr_sync1 <= dst_wr_ptr_sync0;
            dst_wr_ptr_sync2 <= dst_wr_ptr_sync1;
        end
    end

    // Synchronize source read pointer to destination domain
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            src_rd_ptr_sync0 <= {AW+1{1'b0}};
            src_rd_ptr_sync1 <= {AW+1{1'b0}};
            src_rd_ptr_sync2 <= {AW+1{1'b0}};
        end else begin
            src_rd_ptr_sync0 <= {src_rd_ptr, 1'b0};
            src_rd_ptr_sync1 <= src_rd_ptr_sync0;
            src_rd_ptr_sync2 <= src_rd_ptr_sync1;
        end
    end

    // Destination clock domain
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            dst_wr_ptr    <= {AW{1'b0}};
            dst_rd_ptr    <= {AW{1'b0}};
            dst_busy      <= 1'b0;
            dst_psel_r    <= 1'b0;
            dst_penable_r <= 1'b0;
            dst_pwrite_r  <= 1'b0;
            dst_paddr_r   <= {ADDR_WIDTH{1'b0}};
            dst_pwdata_r  <= {DATA_WIDTH{1'b0}};
            dst_pstrb_r   <= {DATA_WIDTH/8{1'b0}};
            dst_pprot_r   <= 3'b0;
            for (integer i = 0; i < DEPTH; i = i + 1) begin
                dst_fifo_write_flag[i] <= 1'b0;
                dst_fifo_read_flag[i]  <= 1'b0;
            end
        end else begin
            // Update FIFO write pointer
            if (dst_fifo_wr_en) begin
                dst_fifo_data[dst_wr_ptr[AW-1:0]] <= src_fifo_data[src_rd_ptr[AW-1:0]];
                dst_fifo_addr[dst_wr_ptr[AW-1:0]] <= src_fifo_addr[src_rd_ptr[AW-1:0]];
                dst_fifo_strb[dst_wr_ptr[AW-1:0]] <= src_fifo_strb[src_rd_ptr[AW-1:0]];
                dst_fifo_prot[dst_wr_ptr[AW-1:0]] <= src_fifo_prot[src_rd_ptr[AW-1:0]];
                dst_fifo_write_flag[dst_wr_ptr[AW-1:0]] <= 1'b1;
                dst_wr_ptr <= dst_wr_ptr + 1'b1;
            end

            // Update FIFO read pointer
            if (dst_fifo_rd_en) begin
                dst_fifo_read_flag[dst_rd_ptr[AW-1:0]] <= 1'b0;
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
            end

            // Destination-side state machine
            if (dst_busy) begin
                if (dst_pready_i) begin
                    // Transfer complete
                    dst_busy      <= 1'b0;
                    dst_psel_r    <= 1'b0;
                    dst_penable_r <= 1'b0;
                end
            end else begin
                if (!dst_fifo_empty) begin
                    // Start new transfer
                    dst_busy      <= 1'b1;
                    dst_psel_r    <= 1'b1;
                    dst_penable_r <= 1'b0;
                    dst_pwrite_r  <= dst_fifo_write_flag[dst_rd_ptr[AW-1:0]] ? 1'b1 : 1'b0;
                    dst_paddr_r   <= dst_fifo_addr[dst_rd_ptr[AW-1:0]];
                    dst_pwdata_r  <= dst_fifo_data[dst_rd_ptr[AW-1:0]];
                    dst_pstrb_r   <= dst_fifo_strb[dst_rd_ptr[AW-1:0]];
                    dst_pprot_r   <= dst_fifo_prot[dst_rd_ptr[AW-1:0]];
                end else begin
                    dst_psel_r    <= 1'b0;
                    dst_penable_r <= 1'b0;
                end
            end

            // Update enable phase
            if (dst_busy && dst_psel_r && !dst_penable_r) begin
                dst_penable_r <= 1'b1;
            end
        end
    end

    // Synchronize dst_pready to source domain
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            dst_pready_sync0  <= 1'b0;
            dst_pready_sync1  <= 1'b0;
            dst_pready_sync2  <= 1'b0;
            dst_pslverr_sync0 <= 1'b0;
            dst_pslverr_sync1 <= 1'b0;
            dst_pslverr_sync2 <= 1'b0;
            dst_prdata_sync0  <= {DATA_WIDTH{1'b0}};
            dst_prdata_sync1  <= {DATA_WIDTH{1'b0}};
            dst_prdata_sync2  <= {DATA_WIDTH{1'b0}};
        end else begin
            dst_pready_sync0  <= dst_pready_i;
            dst_pready_sync1  <= dst_pready_sync0;
            dst_pready_sync2  <= dst_pready_sync1;
            dst_pslverr_sync0 <= dst_pslverr_i;
            dst_pslverr_sync1 <= dst_pslverr_sync0;
            dst_pslverr_sync2 <= dst_pslverr_sync1;
            dst_prdata_sync0  <= dst_prdata_i;
            dst_prdata_sync1  <= dst_prdata_sync0;
            dst_prdata_sync2  <= dst_prdata_sync1;
        end
    end

    // Source-side: update ready and response based on synchronized dst_pready
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_pready_r  <= 1'b0;
            src_prdata_r  <= {DATA_WIDTH{1'b0}};
            src_pslverr_r <= 1'b0;
        end else begin
            if (src_busy && dst_pready_sync2) begin
                src_pready_r  <= 1'b1;
                src_prdata_r  <= dst_prdata_sync2;
                src_pslverr_r <= dst_pslverr_sync2;
            end else if (!src_psel_i) begin
                src_pready_r  <= 1'b0;
            end
        end
    end

endmodule
