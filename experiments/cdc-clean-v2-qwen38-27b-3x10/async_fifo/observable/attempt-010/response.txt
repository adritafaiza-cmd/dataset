module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    localparam AWIDTH = ASIZE + 1;
    localparam DEPTH  = 1 << ASIZE;
    localparam AWFULL_THRESH = DEPTH - 1;
    localparam AEMPTY_THRESH = 1;

    // Write domain
    reg [AWIDTH-1:0] wptr;
    reg [AWIDTH-1:0] wptr_gray;
    reg [AWIDTH-1:0] rptr_gray_w;
    reg [AWIDTH-1:0] rptr_gray_w_d1;
    reg [AWIDTH-1:0] rptr_gray_w_d2;

    // Read domain
    reg [AWIDTH-1:0] rptr;
    reg [AWIDTH-1:0] rptr_gray;
    reg [AWIDTH-1:0] wptr_gray_r;
    reg [AWIDTH-1:0] wptr_gray_r_d1;
    reg [AWIDTH-1:0] wptr_gray_r_d2;

    // Memory
    reg [DSIZE-1:0] mem [0:DEPTH-1];

    // Write side: gray code conversion
    function [AWIDTH-1:0] bin2gray;
        input [AWIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Read side: gray code conversion
    function [AWIDTH-1:0] bin2gray_r;
        input [AWIDTH-1:0] bin;
        begin
            bin2gray_r = bin ^ (bin >> 1);
        end
    endfunction

    // Write domain: update write pointer and gray code
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr      <= {AWIDTH{1'b0}};
            wptr_gray <= {AWIDTH{1'b0}};
        end else if (winc && !wfull) begin
            wptr      <= wptr + 1'b1;
            wptr_gray <= bin2gray(wptr + 1'b1);
        end
    end

    // Write domain: synchronize read pointer gray code
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_w_d1 <= {AWIDTH{1'b0}};
            rptr_gray_w_d2 <= {AWIDTH{1'b0}};
        end else begin
            rptr_gray_w_d1 <= rptr_gray;
            rptr_gray_w_d2 <= rptr_gray_w_d1;
        end
    end

    // Write domain: compute full and almost full
    wire [AWIDTH-1:0] wptr_gray_sync = wptr_gray;
    wire [AWIDTH-1:0] rptr_gray_sync = rptr_gray_w_d2;

    // Full: write pointer gray equals read pointer gray with MSB different
    // Actually, full condition: wptr_gray == rptr_gray_sync with the top bit inverted
    // Standard: full when wptr_gray == rptr_gray_sync ^ (1 << (AWIDTH-1))
    wire wfull_internal = (wptr_gray == (rptr_gray_sync ^ (1 << (AWIDTH-1))));

    // Almost full: write pointer is one ahead of read pointer in binary
    // Convert rptr_gray_sync back to binary for comparison
    function [AWIDTH-1:0] gray2bin;
        input [AWIDTH-1:0] gray;
        integer i;
        begin
            gray2bin[AWIDTH-1] = gray[AWIDTH-1];
            for (i = AWIDTH-2; i >= 0; i = i - 1)
                gray2bin[i] = gray2bin[i+1] ^ gray[i];
        end
    endfunction

    wire [AWIDTH-1:0] rptr_bin_sync = gray2bin(rptr_gray_sync);
    wire [AWIDTH-1:0] wptr_bin      = wptr;

    wire awfull_internal = (wptr_bin - rptr_bin_sync) == AWFULL_THRESH[AWIDTH-1:0];

    assign wfull  = wfull_internal;
    assign awfull = awfull_internal;

    // Read domain: update read pointer and gray code
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr      <= {AWIDTH{1'b0}};
            rptr_gray <= {AWIDTH{1'b0}};
        end else if (rinc && !rempty) begin
            rptr      <= rptr + 1'b1;
            rptr_gray <= bin2gray_r(rptr + 1'b1);
        end
    end

    // Read domain: synchronize write pointer gray code
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_r_d1 <= {AWIDTH{1'b0}};
            wptr_gray_r_d2 <= {AWIDTH{1'b0}};
        end else begin
            wptr_gray_r_d1 <= wptr_gray;
            wptr_gray_r_d2 <= wptr_gray_r_d1;
        end
    end

    // Read domain: compute empty and almost empty
    wire [AWIDTH-1:0] wptr_gray_sync_r = wptr_gray_r_d2;
    wire [AWIDTH-1:0] rptr_gray_sync_r = rptr_gray;

    // Empty: write pointer gray equals read pointer gray
    wire rempty_internal = (wptr_gray_sync_r == rptr_gray_sync_r);

    // Almost empty: read pointer is one behind write pointer in binary
    function [AWIDTH-1:0] gray2bin_r;
        input [AWIDTH-1:0] gray;
        integer i;
        begin
            gray2bin_r[AWIDTH-1] = gray[AWIDTH-1];
            for (i = AWIDTH-2; i >= 0; i = i - 1)
                gray2bin_r[i] = gray2bin_r[i+1] ^ gray[i];
        end
    endfunction

    wire [AWIDTH-1:0] wptr_bin_sync_r = gray2bin_r(wptr_gray_sync_r);
    wire [AWIDTH-1:0] rptr_bin_r      = rptr;

    wire arempty_internal = (wptr_bin_sync_r - rptr_bin_r) == AEMPTY_THRESH[AWIDTH-1:0];

    assign rempty  = rempty_internal;
    assign arempty = arempty_internal;

    // Memory write
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            // No action needed for memory on reset
        end else if (winc && !wfull) begin
            mem[wptr[ASIZE-1:0]] <= wdata;
        end
    end

    // Memory read
    assign rdata = mem[rptr[ASIZE-1:0]];

endmodule
