module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    localparam AWIDTH = ASIZE + 1;
    localparam DEPTH  = 1 << ASIZE;
    localparam AWFULL_THRESH = DEPTH - 1;
    localparam AEMPTY_THRESH = 1;

    // Write domain
    reg [AWIDTH-1:0] wptr;
    reg [AWIDTH-1:0] rptr_w;
    reg [AWIDTH-1:0] rptr_w_sync1;
    reg [AWIDTH-1:0] rptr_w_sync2;
    reg [AWIDTH-1:0] rptr_w_sync3;

    // Read domain
    reg [AWIDTH-1:0] rptr;
    reg [AWIDTH-1:0] wptr_r;
    reg [AWIDTH-1:0] wptr_r_sync1;
    reg [AWIDTH-1:0] wptr_r_sync2;
    reg [AWIDTH-1:0] wptr_r_sync3;

    // Memory
    reg [DSIZE-1:0] mem [0:DEPTH-1];

    // Write side logic
    wire [AWIDTH-1:0] wptr_next = wptr + 1'b1;
    wire              winc_eff  = winc & ~wfull;

    // Read side logic
    wire [AWIDTH-1:0] rptr_next = rptr + 1'b1;
    wire              rinc_eff  = rinc & ~rempty;

    // Full/Empty flags in write domain
    wire [AWIDTH-1:0] wptr_minus_rptr_w = wptr - rptr_w_sync3;
    wire              wfull_int  = (wptr_minus_rptr_w == DEPTH);
    wire              awfull_int = (wptr_minus_rptr_w == AWFULL_THRESH);

    // Full/Empty flags in read domain
    wire [AWIDTH-1:0] wptr_r_minus_rptr = wptr_r_sync3 - rptr;
    wire              rempty_int  = (wptr_r_minus_rptr == 0);
    wire              arempty_int = (wptr_r_minus_rptr == AEMPTY_THRESH);

    // Outputs
    assign wfull   = wfull_int;
    assign awfull  = awfull_int;
    assign rempty  = rempty_int;
    assign arempty = arempty_int;

    // Read data output
    assign rdata = mem[rptr[ASIZE-1:0]];

    // Write clock domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {AWIDTH{1'b0}};
            rptr_w <= {AWIDTH{1'b0}};
            rptr_w_sync1 <= {AWIDTH{1'b0}};
            rptr_w_sync2 <= {AWIDTH{1'b0}};
            rptr_w_sync3 <= {AWIDTH{1'b0}};
        end else begin
            if (winc_eff)
                mem[wptr[ASIZE-1:0]] <= wdata;
            if (winc_eff)
                wptr <= wptr_next;
            rptr_w_sync1 <= rptr_w;
            rptr_w_sync2 <= rptr_w_sync1;
            rptr_w_sync3 <= rptr_w_sync2;
        end
    end

    // Read clock domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {AWIDTH{1'b0}};
            wptr_r <= {AWIDTH{1'b0}};
            wptr_r_sync1 <= {AWIDTH{1'b0}};
            wptr_r_sync2 <= {AWIDTH{1'b0}};
            wptr_r_sync3 <= {AWIDTH{1'b0}};
        end else begin
            if (rinc_eff)
                rptr <= rptr_next;
            wptr_r_sync1 <= wptr_r;
            wptr_r_sync2 <= wptr_r_sync1;
            wptr_r_sync3 <= wptr_r_sync2;
        end
    end

    // Cross-domain pointer synchronization
    // Write pointer to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n)
            wptr_r <= {AWIDTH{1'b0}};
        else
            wptr_r <= wptr;
    end

    // Read pointer to write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n)
            rptr_w <= {AWIDTH{1'b0}};
        else
            rptr_w <= rptr;
    end

endmodule
