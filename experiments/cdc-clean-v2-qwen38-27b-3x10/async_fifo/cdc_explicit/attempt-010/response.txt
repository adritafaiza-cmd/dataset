module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    localparam DEPTH = 1 << ASIZE;
    localparam AWFULL_THRESH = DEPTH - 2;
    localparam AEMPTY_THRESH = 2;

    // Write domain
    reg [ASIZE:0] wptr;
    reg [ASIZE:0] rptr_w;
    reg [ASIZE:0] rptr_w_sync1;
    reg [ASIZE:0] rptr_w_sync2;

    // Read domain
    reg [ASIZE:0] rptr;
    reg [ASIZE:0] wptr_r;
    reg [ASIZE:0] wptr_r_sync1;
    reg [ASIZE:0] wptr_r_sync2;

    // Memory
    reg [DSIZE-1:0] mem [0:DEPTH-1];

    // Read data pipeline
    reg [DSIZE-1:0] rdata_pipe1;
    reg [DSIZE-1:0] rdata_pipe2;
    reg             rdata_valid1;
    reg             rdata_valid2;

    // Write enable
    wire wvalid = winc & ~wfull;

    // Read enable
    wire rvalid = rinc & ~rempty;

    // Write pointer increment
    wire wptr_inc = wvalid;

    // Read pointer increment
    wire rptr_inc = rvalid;

    // Write domain logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {1'b0, {ASIZE{1'b0}}};
            rptr_w <= {1'b0, {ASIZE{1'b0}}};
            rptr_w_sync1 <= {1'b0, {ASIZE{1'b0}}};
            rptr_w_sync2 <= {1'b0, {ASIZE{1'b0}}};
        end else begin
            if (wptr_inc)
                wptr <= wptr + 1'b1;
            rptr_w_sync1 <= rptr_w;
            rptr_w_sync2 <= rptr_w_sync1;
        end
    end

    // Read domain logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {1'b0, {ASIZE{1'b0}}};
            wptr_r <= {1'b0, {ASIZE{1'b0}}};
            wptr_r_sync1 <= {1'b0, {ASIZE{1'b0}}};
            wptr_r_sync2 <= {1'b0, {ASIZE{1'b0}}};
            rdata_pipe1 <= {DSIZE{1'b0}};
            rdata_pipe2 <= {DSIZE{1'b0}};
            rdata_valid1 <= 1'b0;
            rdata_valid2 <= 1'b0;
        end else begin
            if (rptr_inc)
                rptr <= rptr + 1'b1;
            wptr_r_sync1 <= wptr_r;
            wptr_r_sync2 <= wptr_r_sync1;

            // Pipeline data
            if (rvalid) begin
                rdata_pipe1 <= mem[rptr[ASIZE-1:0]];
                rdata_valid1 <= 1'b1;
            end else begin
                rdata_valid1 <= 1'b0;
            end

            if (rdata_valid1) begin
                rdata_pipe2 <= rdata_pipe1;
                rdata_valid2 <= 1'b1;
            end else begin
                rdata_valid2 <= 1'b0;
            end
        end
    end

    // Memory write
    always @(posedge wclk) begin
        if (wvalid)
            mem[wptr[ASIZE-1:0]] <= wdata;
    end

    // Full and empty flags
    assign wfull = (wptr - rptr_w_sync2) == DEPTH;
    assign rempty = (rptr - wptr_r_sync2) == 0;

    // Almost full and almost empty
    assign awfull = (wptr - rptr_w_sync2) >= AWFULL_THRESH;
    assign arempty = (rptr - wptr_r_sync2) <= AEMPTY_THRESH;

    // Output data
    assign rdata = rdata_pipe2;

endmodule
