```verilog
module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   s_pause_req,
    output wire                   s_pause_ack,
    input  wire                   m_pause_req,
    output wire                   m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] s_status_depth,
    output wire [$clog2(DEPTH):0] s_status_depth_commit,
    output wire                   s_status_overflow,
    output wire                   s_status_bad_frame,
    output wire                   s_status_good_frame,
    output wire [$clog2(DEPTH):0] m_status_depth,
    output wire [$clog2(DEPTH):0] m_status_depth_commit,
    output wire                   m_status_overflow,
    output wire                   m_status_bad_frame,
    output wire                   m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam DEPTH_MASK = DEPTH - 1;
    localparam DEPTH_PLUS_ONE = DEPTH + 1;

    // Total width of the data bus stored in RAM
    localparam TOTAL_DATA_WIDTH = DATA_WIDTH + 
                                  (KEEP_ENABLE ? KEEP_WIDTH : 0) +
                                  (LAST_ENABLE ? 1 : 0) +
                                  (ID_ENABLE ? ID_WIDTH : 0) +
                                  (DEST_ENABLE ? DEST_WIDTH : 0) +
                                  (USER_ENABLE ? USER_WIDTH : 0);

    // Internal signals
    reg [ADDR_WIDTH:0] s_wr_ptr;
    reg [ADDR_WIDTH:0] m_rd_ptr;
    
    // Gray codes for CDC
    wire [ADDR_WIDTH:0] s_wr_ptr_gray;
    wire [ADDR_WIDTH:0] m_rd_ptr_gray;
    
    // Synchronizers
    reg [ADDR_WIDTH:0] m_rd_ptr_gray_s1, m_rd_ptr_gray_s2;
    reg [ADDR_WIDTH:0] s_wr_ptr_gray_m1, s_wr_ptr_gray_m2;
    
    // RAM
    reg [TOTAL_DATA_WIDTH-1:0] ram [0:DEPTH-1];
    
    // Write side
    wire s_wr_en;
    wire [ADDR_WIDTH-1:0] s_wr_addr;
    wire [TOTAL_DATA_WIDTH-1:0] s_wr_data;
    
    // Read side
    wire m_rd_en;
    wire [ADDR_WIDTH-1:0] m_rd_addr;
    wire [TOTAL_DATA_WIDTH-1:0] m_rd_data;
    
    // Status
    wire s_full;
    wire m_empty;
    wire [$clog2(DEPTH):0] s_depth;
    wire [$clog2(DEPTH):0] m_depth;
    
    // Frame tracking
    reg s_in_frame;
    reg m_in_frame;
    reg s_bad_frame;
    reg m_bad_frame;
    reg s_overflow;
    reg m_overflow;
    reg s_good_frame;
    reg m_good_frame;
    
    // Pause
    reg s_paused;
    reg m_paused;

    // Gray code conversion
    function [ADDR_WIDTH:0] bin2gray;
        input [ADDR_WIDTH:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    assign s_wr_ptr_gray = bin2gray(s_wr_ptr);
    assign m_rd_ptr_gray = bin2gray(m_rd_ptr);

    // Synchronizers
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_wr_ptr_gray_m1 <= {(ADDR_WIDTH+1){1'b0}};
            s_wr_ptr_gray_m2 <= {(ADDR_WIDTH+1){1'b0}};
        end else begin
            s_wr_ptr_gray_m1 <= s_wr_ptr_gray;
            s_wr_ptr_gray_m2 <= s_wr_ptr_gray_m1;
        end
    end

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_rd_ptr_gray_s1 <= {(ADDR_WIDTH+1){1'b0}};
            m_rd_ptr_gray_s2 <= {(ADDR_WIDTH+1){1'b0}};
        end else begin
            m_rd_ptr_gray_s1 <= m_rd_ptr_gray;
            m_rd_ptr_gray_s2 <= m_rd_ptr_gray_s1;
        end
    end

    // Convert synchronized gray codes back to binary
    wire [ADDR_WIDTH:0] s_wr_ptr_gray_m2_bin;
    wire [ADDR_WIDTH:0] m_rd_ptr_gray_s2_bin;
    
    genvar i;
    generate
        for (i = 0; i <= ADDR_WIDTH; i = i + 1) begin : gray_to_bin_m
            if (i == 0) begin
                assign s_wr_ptr_gray_m2_bin[i] = s_wr_ptr_gray_m2[i];
            end else begin
                assign s_wr_ptr_gray_m2_bin[i] = s_wr_ptr_gray_m2[i] ^ s_wr_ptr_gray_m2_bin[i-1];
            end
        end
        for (i = 0; i <= ADDR_WIDTH; i = i + 1) begin : gray_to_bin_s
            if (i == 0) begin
                assign m_rd_ptr_gray_s2_bin[i] = m_rd_ptr_gray_s2[i];
            end else begin
                assign m_rd_ptr_gray_s2_bin[i] = m_rd_ptr_gray_s2[i] ^ m_rd_ptr_gray_s2_bin[i-1];
            end
        end
    endgenerate

    // Depth calculation
    assign s_depth = s_wr_ptr - m_rd_ptr_gray_s2_bin;
    assign m_depth = s_wr_ptr_gray_m2_bin - m_rd_ptr;

    // Full and empty
    assign s_full = (s_depth >= DEPTH);
    assign m_empty = (m_depth == 0);

    // Write enable
    wire s_accept = s_axis_tvalid & s_axis_tready;
    assign s_wr_en = s_accept;
    assign s_wr_addr = s_wr_ptr[ADDR_WIDTH-1:0];

    // Build write data
    wire [TOTAL_DATA_WIDTH-1:0] s_wr_data_internal;
    assign s_wr_data_internal = {
        (USER_ENABLE ? s_axis_tuser : {USER_WIDTH{1'b0}}),
        (DEST_ENABLE ? s_axis_tdest : {DEST_WIDTH{1'b0}}),
        (ID_ENABLE ? s_axis_tid : {ID_WIDTH{1'b0}}),
        (LAST_ENABLE ? s_axis_tlast : 1'b0),
        (KEEP_ENABLE ? s_axis_tkeep : {KEEP_WIDTH{1'b0}}),
        s_axis_tdata
    };
    assign s_wr_data = s_wr_data_internal;

    // Write to RAM
    always @(posedge s_clk) begin
        if (s_wr_en) begin
            ram[s_wr_addr] <= s_wr_data;
        end
    end

    // Read address
    assign m_rd_addr = m_rd_ptr[ADDR_WIDTH-1:0];

    // Read from RAM
    wire [TOTAL_DATA_WIDTH-1:0] m_rd_data_internal;
    assign m_rd_data_internal = ram[m_rd_addr];
    assign m_rd_data = m_rd_data_internal;

    // Extract fields from read data
    wire [DATA_WIDTH-1:0] m_axis_tdata_internal;
    wire [KEEP_WIDTH-1:0] m_axis_tkeep_internal;
    wire m_axis_tlast_internal;
    wire [ID_WIDTH-1:0] m_axis_tid_internal;
    wire [DEST_WIDTH-1:0] m_axis_tdest_internal;
    wire [USER_WIDTH-1:0] m_axis_tuser_internal;

    assign m_axis_tdata_internal = m_rd_data[DATA_WIDTH-1:0];
    assign m_axis_tkeep_internal = m_rd_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) - 1 : DATA_WIDTH];
    assign m_axis_tlast_internal = m_rd_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) - 1];
    assign m_axis_tid_internal = m_rd_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0) - 1 : DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0)];
    assign m_axis_tdest_internal = m_rd_data[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0) + (DEST_ENABLE ? DEST_WIDTH : 0) - 1 : DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0)];
    assign m_axis_tuser_internal = m_rd_data[TOTAL_DATA_WIDTH-1 : DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0) + (DEST_ENABLE ? DEST_WIDTH : 0)];

    // Output FIFO logic (simplified: direct output if not enabled)
    reg [TOTAL_DATA_WIDTH-1:0] out_fifo [0:DEPTH-1];
    reg [ADDR_WIDTH:0] out_wr_ptr;
    reg [ADDR_WIDTH:0] out_rd_ptr;
    wire out_full;
    wire out_empty;
    wire [$clog2(DEPTH):0] out_depth;

    if (OUTPUT_FIFO_ENABLE) begin
        assign out_depth = out_wr_ptr - out_rd_ptr;
        assign out_full = (out_depth >= DEPTH);
        assign out_empty = (out_depth == 0);
    end else begin
        assign out_depth = 0;
        assign out_full = 1'b0;
        assign out_empty = 1'b1;
    end

    // Read enable from main FIFO
    wire m_rd_en_internal = m_axis_tvalid & m_axis_tready;
    assign m_rd_en = m_rd_en_internal;

    // Update read pointer
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr <= {(ADDR_WIDTH+1){1'b0}};
        end else if (m_rd_en) begin
            m_rd_ptr <= m_rd_ptr + 1'b1;
        end
    end

    // Update write pointer
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= {(ADDR_WIDTH+1){1'b0}};
        end else if (s_wr_en) begin
            s_wr_ptr <= s_wr_ptr + 1'b1;
        end
    end

    // tready logic
    wire s_tready_internal;
    if (DROP_WHEN_FULL || MARK_WHEN_FULL) begin
        assign s_tready_internal = 1'b1;
    end else begin
        assign s_tready_internal = ~s_full;
    end

    // Pause logic
    if (PAUSE_ENABLE) begin
        always @(posedge s_clk or posedge s_rst) begin
            if (s_rst) begin
                s_paused <= 1'b0;
            end else if (s_pause_req) begin
                s_paused <= 1'b1;
            end else if (s_axis_tvalid & s_axis_tready & s_axis_tlast) begin
                s_paused <= 1'b0;
            end
        end
        assign s_pause_ack = s_paused;
        assign s_axis_tready = s_tready_internal & ~s_paused;
    end else begin
        assign s_pause_ack = 1'b0;
        assign s_axis_tready = s_tready_internal;
    end

    if (PAUSE_ENABLE) begin
        always @(posedge m_clk or posedge m_rst) begin
            if (m_rst) begin
                m_paused <= 1'b0;
            end else if (m_pause_req) begin
                m_paused <= 1'b1;
            end else if (m_axis_tvalid & m_axis_tready & m_axis_tlast) begin
                m_paused <= 1'b0;
            end
        end
        assign m_pause_ack = m_paused;
    end else begin
        assign m_pause_ack = 1'b0;
    end

    // m_axis_tvalid
    wire m_tvalid_internal;
    if (OUTPUT_FIFO_ENABLE) begin
        assign m_tvalid_internal = ~out_empty;
    end else begin
        assign m_tvalid_internal = ~m_empty;
    end

    assign m_axis_tvalid = m_tvalid_internal & ~m_paused;

    // Output data
    if (OUTPUT_FIFO_ENABLE) begin
        // Simplified: just use the main RAM read data for now
        assign m_axis_tdata = m_axis_tdata_internal;
        assign m_axis_tkeep = m_axis_tkeep_internal;
        assign m_axis_tlast = m_axis_tlast_internal;
        assign m_axis_tid = m_axis_tid_internal;
        assign m_axis_tdest = m_axis_tdest_internal;
        assign m_axis_tuser = m_axis_tuser_internal;
    end else begin
        assign m_axis_tdata = m_axis_tdata_internal;
        assign m_axis_tkeep = m_axis_tkeep_internal;
        assign m_axis_tlast = m_axis_tlast_internal;
        assign m_axis_tid = m_axis_tid_internal;
        assign m_axis_tdest = m_axis_tdest_internal;
        assign m_axis_tuser = m_axis_tuser_internal;
    end

    // Frame tracking
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_in_frame <= 1'b0;
            s_bad_frame <= 1'b0;
            s_overflow <= 1'b0;
            s_good_frame <= 1'b0;
        end else begin
            if (s_wr_en) begin
                if (s_axis_tlast) begin
                    s_in_frame <= 1'b0;
                    if (s_full && !DROP_WHEN_FULL && !MARK_WHEN_FULL) begin
                        s_overflow <= 1'b1;
                    end
                    if (USER_ENABLE && (s_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                        s_bad_frame <= 1'b1;
                    end else begin
                        s_good_frame <= 1'b1;
                    end
                end else begin
                    s_in_frame <= 1'b1;
                end
            end
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_in_frame <= 1'b0;
            m_bad_frame <= 1'b0;
            m_overflow <= 1'b0;
            m_good_frame <= 1'b0;
        end else begin
            if (m_rd_en) begin
                if (m_axis_tlast) begin
                    m_in_frame <= 1'b0;
                    if (USER_ENABLE && (m_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                        m_bad_frame <= 1'b1;
                    end else begin
                        m_good_frame <= 1'b1;
                    end
                end else begin
                    m_in_frame <= 1'b1;
                end
            end
        end
    end

    // Status outputs
    assign s_status_depth = s_depth;
    assign s_status_depth_commit = s_depth;
    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;

    assign m_status_depth = m_depth;
    assign m_status_depth_commit = m_depth;
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;

endmodule
