```verilog
module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   s_pause_req,
    output wire                   s_pause_ack,
    input  wire                   m_pause_req,
    output wire                   m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] s_status_depth,
    output wire [$clog2(DEPTH):0] s_status_depth_commit,
    output wire                   s_status_overflow,
    output wire                   s_status_bad_frame,
    output wire                   s_status_good_frame,
    output wire [$clog2(DEPTH):0] m_status_depth,
    output wire [$clog2(DEPTH):0] m_status_depth_commit,
    output wire                   m_status_overflow,
    output wire                   m_status_bad_frame,
    output wire                   m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam DEPTH_MASK = DEPTH - 1;
    
    // Total width of the data word to be stored in RAM
    localparam TOTAL_DATA_WIDTH = DATA_WIDTH + 
                                  (KEEP_ENABLE ? KEEP_WIDTH : 0) +
                                  (LAST_ENABLE ? 1 : 0) +
                                  (ID_ENABLE ? ID_WIDTH : 0) +
                                  (DEST_ENABLE ? DEST_WIDTH : 0) +
                                  (USER_ENABLE ? USER_WIDTH : 0);

    // Internal signals
    wire [TOTAL_DATA_WIDTH-1:0] s_axis_tdata_packed;
    wire [TOTAL_DATA_WIDTH-1:0] m_axis_tdata_unpacked;
    
    // Pack input data
    assign s_axis_tdata_packed = {
        (USER_ENABLE ? s_axis_tuser : {USER_WIDTH{1'b0}}),
        (DEST_ENABLE ? s_axis_tdest : {DEST_WIDTH{1'b0}}),
        (ID_ENABLE ? s_axis_tid : {ID_WIDTH{1'b0}}),
        (LAST_ENABLE ? s_axis_tlast : 1'b0),
        (KEEP_ENABLE ? s_axis_tkeep : {KEEP_WIDTH{1'b0}}),
        s_axis_tdata
    };

    // Unpack output data
    assign m_axis_tdata = m_axis_tdata_unpacked[DATA_WIDTH-1:0];
    assign m_axis_tkeep = m_axis_tdata_unpacked[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) - 1 : DATA_WIDTH];
    assign m_axis_tlast = m_axis_tdata_unpacked[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) - 1];
    assign m_axis_tid = m_axis_tdata_unpacked[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0) - 1 : DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0)];
    assign m_axis_tdest = m_axis_tdata_unpacked[DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0) + (DEST_ENABLE ? DEST_WIDTH : 0) - 1 : DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0)];
    assign m_axis_tuser = m_axis_tdata_unpacked[TOTAL_DATA_WIDTH-1 : DATA_WIDTH + (KEEP_ENABLE ? KEEP_WIDTH : 0) + (LAST_ENABLE ? 1 : 0) + (ID_ENABLE ? ID_WIDTH : 0) + (DEST_ENABLE ? DEST_WIDTH : 0)];

    // FIFO pointers
    reg [ADDR_WIDTH:0] s_wr_ptr;
    reg [ADDR_WIDTH:0] m_rd_ptr;
    
    // Synchronized pointers
    reg [ADDR_WIDTH:0] s_wr_ptr_m_sync1;
    reg [ADDR_WIDTH:0] s_wr_ptr_m_sync2;
    reg [ADDR_WIDTH:0] m_rd_ptr_s_sync1;
    reg [ADDR_WIDTH:0] m_rd_ptr_s_sync2;
    
    // RAM
    reg [TOTAL_DATA_WIDTH-1:0] ram [0:DEPTH-1];
    
    // Read data
    reg [TOTAL_DATA_WIDTH-1:0] ram_rd_data;
    reg [TOTAL_DATA_WIDTH-1:0] ram_rd_data_reg;
    
    // Control signals
    wire s_wr_en;
    wire m_rd_en;
    wire s_full;
    wire m_empty;
    wire s_overflow;
    wire m_overflow;
    
    // Pause signals
    wire s_pause_active;
    wire m_pause_active;
    
    // Frame tracking
    reg s_in_frame;
    reg m_in_frame;
    reg s_bad_frame;
    reg m_bad_frame;
    reg s_good_frame;
    reg m_good_frame;
    
    // Depth calculation
    wire [$clog2(DEPTH):0] s_depth;
    wire [$clog2(DEPTH):0] m_depth;
    wire [$clog2(DEPTH):0] s_depth_commit;
    wire [$clog2(DEPTH):0] m_depth_commit;
    
    // Synchronize write pointer to master clock domain
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_wr_ptr_m_sync1 <= {({ADDR_WIDTH+1}{1'b0}});
            s_wr_ptr_m_sync2 <= {({ADDR_WIDTH+1}{1'b0}});
        end else begin
            s_wr_ptr_m_sync1 <= s_wr_ptr;
            s_wr_ptr_m_sync2 <= s_wr_ptr_m_sync1;
        end
    end
    
    // Synchronize read pointer to slave clock domain
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_rd_ptr_s_sync1 <= {({ADDR_WIDTH+1}{1'b0}});
            m_rd_ptr_s_sync2 <= {({ADDR_WIDTH+1}{1'b0}});
        end else begin
            m_rd_ptr_s_sync1 <= m_rd_ptr;
            m_rd_ptr_s_sync2 <= m_rd_ptr_s_sync1;
        end
    end
    
    // Calculate depth
    assign s_depth = s_wr_ptr - m_rd_ptr_s_sync2;
    assign m_depth = s_wr_ptr_m_sync2 - m_rd_ptr;
    
    // Commit depth (same as depth for this implementation)
    assign s_depth_commit = s_depth;
    assign m_depth_commit = m_depth;
    
    // Full and empty conditions
    assign s_full = (s_depth == DEPTH);
    assign m_empty = (m_depth == 0);
    
    // Pause handling
    assign s_pause_active = (PAUSE_ENABLE && s_pause_req);
    assign m_pause_active = (PAUSE_ENABLE && m_pause_req);
    
    // Overflow detection
    assign s_overflow = (s_axis_tvalid && s_full && !DROP_WHEN_FULL && !MARK_WHEN_FULL);
    assign m_overflow = 1'b0;
    
    // Write enable
    assign s_wr_en = s_axis_tvalid && s_axis_tready && !s_pause_active;
    
    // Read enable
    assign m_rd_en = m_axis_tvalid && m_axis_tready && !m_pause_active;
    
    // Frame tracking - slave side
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_in_frame <= 1'b0;
            s_bad_frame <= 1'b0;
            s_good_frame <= 1'b0;
        end else begin
            if (s_wr_en) begin
                if (s_axis_tlast) begin
                    s_in_frame <= 1'b0;
                    // Check for bad frame
                    if (USER_ENABLE && (s_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                        s_bad_frame <= 1'b1;
                    end else begin
                        s_good_frame <= 1'b1;
                    end
                end else begin
                    s_in_frame <= 1'b1;
                end
            end
        end
    end
    
    // Frame tracking - master side
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_in_frame <= 1'b0;
            m_bad_frame <= 1'b0;
            m_good_frame <= 1'b0;
        end else begin
            if (m_rd_en) begin
                if (m_axis_tlast) begin
                    m_in_frame <= 1'b0;
                    // Check for bad frame
                    if (USER_ENABLE && (m_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                        m_bad_frame <= 1'b1;
                    end else begin
                        m_good_frame <= 1'b1;
                    end
                end else begin
                    m_in_frame <= 1'b1;
                end
            end
        end
    end
    
    // RAM write
    always @(posedge s_clk) begin
        if (s_wr_en) begin
            ram[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tdata_packed;
        end
    end
    
    // RAM read
    always @(posedge m_clk) begin
        ram_rd_data <= ram[m_rd_ptr[ADDR_WIDTH-1:0]];
    end
    
    // Pipeline stage
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            ram_rd_data_reg <= {({TOTAL_DATA_WIDTH}{1'b0}});
        end else begin
            ram_rd_data_reg <= ram_rd_data;
        end
    end
    
    // Output data
    assign m_axis_tdata_unpacked = ram_rd_data_reg;
    
    // Valid signal
    assign m_axis_tvalid = !m_empty && !m_pause_active;
    
    // Ready signal
    assign s_axis_tready = !s_full || DROP_WHEN_FULL || MARK_WHEN_FULL;
    
    // Pointer updates
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= {({ADDR_WIDTH+1}{1'b0}});
        end else if (s_wr_en) begin
            s_wr_ptr <= s_wr_ptr + 1'b1;
        end
    end
    
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr <= {({ADDR_WIDTH+1}{1'b0}});
        end else if (m_rd_en) begin
            m_rd_ptr <= m_rd_ptr + 1'b1;
        end
    end
    
    // Pause acknowledgments
    assign s_pause_ack = s_pause_active;
    assign m_pause_ack = m_pause_active;
    
    // Status outputs
    assign s_status_depth = s_depth;
    assign s_status_depth_commit = s_depth_commit;
    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;
    
    assign m_status_depth = m_depth;
    assign m_status_depth_commit = m_depth_commit;
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;

endmodule
