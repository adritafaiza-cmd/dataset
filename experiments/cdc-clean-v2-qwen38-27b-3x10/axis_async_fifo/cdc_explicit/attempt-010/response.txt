```verilog
module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   s_pause_req,
    output wire                   s_pause_ack,
    input  wire                   m_pause_req,
    output wire                   m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] s_status_depth,
    output wire [$clog2(DEPTH):0] s_status_depth_commit,
    output wire                   s_status_overflow,
    output wire                   s_status_bad_frame,
    output wire                   s_status_good_frame,
    output wire [$clog2(DEPTH):0] m_status_depth,
    output wire [$clog2(DEPTH):0] m_status_depth_commit,
    output wire                   m_status_overflow,
    output wire                   m_status_bad_frame,
    output wire                   m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam DEPTH_POW2 = (DEPTH == 1) ? 1 : (1 << ADDR_WIDTH);
    localparam DEPTH_MASK = DEPTH_POW2 - 1;
    localparam DEPTH_LIMIT = DEPTH;
    
    // Total width for storage
    localparam TOTAL_WIDTH = DATA_WIDTH + 
                             (KEEP_ENABLE ? KEEP_WIDTH : 0) +
                             (LAST_ENABLE ? 1 : 0) +
                             (ID_ENABLE ? ID_WIDTH : 0) +
                             (DEST_ENABLE ? DEST_WIDTH : 0) +
                             (USER_ENABLE ? USER_WIDTH : 0);

    // Internal signals
    wire [ADDR_WIDTH:0] s_wr_ptr;
    wire [ADDR_WIDTH:0] s_rd_ptr_sync;
    wire [ADDR_WIDTH:0] m_rd_ptr;
    wire [ADDR_WIDTH:0] m_wr_ptr_sync;
    
    wire s_full;
    wire s_empty;
    wire m_full;
    wire m_empty;
    
    wire s_wr_en;
    wire m_rd_en;
    
    wire [TOTAL_WIDTH-1:0] s_wr_data;
    wire [TOTAL_WIDTH-1:0] m_rd_data;
    
    // Status signals
    reg s_overflow_reg;
    reg m_overflow_reg;
    reg s_bad_frame_reg;
    reg s_good_frame_reg;
    reg m_bad_frame_reg;
    reg m_good_frame_reg;
    
    // Pause signals
    reg s_pause_ack_reg;
    reg m_pause_ack_reg;
    
    // Frame tracking
    reg s_in_frame;
    reg m_in_frame;
    
    // Depth calculation
    wire [$clog2(DEPTH):0] s_depth;
    wire [$clog2(DEPTH):0] m_depth;
    
    // Commit depth (synchronized)
    wire [$clog2(DEPTH):0] s_depth_commit;
    wire [$clog2(DEPTH):0] m_depth_commit;

    // Data packing
    assign s_wr_data = {
        (USER_ENABLE ? s_axis_tuser : {USER_WIDTH{1'b0}}),
        (DEST_ENABLE ? s_axis_tdest : {DEST_WIDTH{1'b0}}),
        (ID_ENABLE ? s_axis_tid : {ID_WIDTH{1'b0}}),
        (LAST_ENABLE ? s_axis_tlast : 1'b0),
        (KEEP_ENABLE ? s_axis_tkeep : {KEEP_WIDTH{1'b0}}),
        s_axis_tdata
    };

    // Data unpacking
    assign m_axis_tdata = m_rd_data[DATA_WIDTH-1:0];
    assign m_axis_tkeep = KEEP_ENABLE ? m_rd_data[DATA_WIDTH + KEEP_WIDTH-1:DATA_WIDTH] : {KEEP_WIDTH{1'b0}};
    assign m_axis_tlast = LAST_ENABLE ? m_rd_data[DATA_WIDTH + KEEP_WIDTH] : 1'b0;
    assign m_axis_tid = ID_ENABLE ? m_rd_data[DATA_WIDTH + KEEP_WIDTH + (LAST_ENABLE?1:0) + ID_WIDTH-1:DATA_WIDTH + KEEP_WIDTH + (LAST_ENABLE?1:0)] : {ID_WIDTH{1'b0}};
    assign m_axis_tdest = DEST_ENABLE ? m_rd_data[DATA_WIDTH + KEEP_WIDTH + (LAST_ENABLE?1:0) + ID_WIDTH + DEST_WIDTH-1:DATA_WIDTH + KEEP_WIDTH + (LAST_ENABLE?1:0) + ID_WIDTH] : {DEST_WIDTH{1'b0}};
    assign m_axis_tuser = USER_ENABLE ? m_rd_data[TOTAL_WIDTH-1:DATA_WIDTH + KEEP_WIDTH + (LAST_ENABLE?1:0) + ID_WIDTH + DEST_WIDTH] : {USER_WIDTH{1'b0}};

    // Write pointer
    reg [ADDR_WIDTH:0] s_wr_ptr_reg;
    assign s_wr_ptr = s_wr_ptr_reg;
    
    // Read pointer
    reg [ADDR_WIDTH:0] m_rd_ptr_reg;
    assign m_rd_ptr = m_rd_ptr_reg;

    // Synchronize read pointer to source domain
    reg [ADDR_WIDTH:0] s_rd_ptr_sync_1, s_rd_ptr_sync_2;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_rd_ptr_sync_1 <= {(ADDR_WIDTH+1){1'b0}};
            s_rd_ptr_sync_2 <= {(ADDR_WIDTH+1){1'b0}};
        end else begin
            s_rd_ptr_sync_1 <= m_rd_ptr;
            s_rd_ptr_sync_2 <= s_rd_ptr_sync_1;
        end
    end
    assign s_rd_ptr_sync = s_rd_ptr_sync_2;

    // Synchronize write pointer to destination domain
    reg [ADDR_WIDTH:0] m_wr_ptr_sync_1, m_wr_ptr_sync_2;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_wr_ptr_sync_1 <= {(ADDR_WIDTH+1){1'b0}};
            m_wr_ptr_sync_2 <= {(ADDR_WIDTH+1){1'b0}};
        end else begin
            m_wr_ptr_sync_1 <= s_wr_ptr;
            m_wr_ptr_sync_2 <= m_wr_ptr_sync_1;
        end
    end
    assign m_wr_ptr_sync = m_wr_ptr_sync_2;

    // Full/Empty logic
    assign s_full = (s_wr_ptr_reg[ADDR_WIDTH] != s_rd_ptr_sync[ADDR_WIDTH]) && 
                    (s_wr_ptr_reg[ADDR_WIDTH-1:0] == s_rd_ptr_sync[ADDR_WIDTH-1:0]);
    assign s_empty = (s_wr_ptr_reg == s_rd_ptr_sync);
    
    assign m_full = (m_wr_ptr_sync[ADDR_WIDTH] != m_rd_ptr_reg[ADDR_WIDTH]) && 
                    (m_wr_ptr_sync[ADDR_WIDTH-1:0] == m_rd_ptr_reg[ADDR_WIDTH-1:0]);
    assign m_empty = (m_wr_ptr_sync == m_rd_ptr_reg);

    // Depth calculation
    wire [$clog2(DEPTH):0] s_depth_raw = s_wr_ptr_reg - s_rd_ptr_sync;
    wire [$clog2(DEPTH):0] m_depth_raw = m_wr_ptr_sync - m_rd_ptr_reg;
    
    assign s_depth = (s_depth_raw > DEPTH_LIMIT) ? DEPTH_LIMIT[$clog2(DEPTH):0] : s_depth_raw[$clog2(DEPTH):0];
    assign m_depth = (m_depth_raw > DEPTH_LIMIT) ? DEPTH_LIMIT[$clog2(DEPTH):0] : m_depth_raw[$clog2(DEPTH):0];

    // Commit depth (synchronized version)
    reg [$clog2(DEPTH):0] s_depth_commit_reg;
    reg [$clog2(DEPTH):0] m_depth_commit_reg;
    
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_depth_commit_reg <= {($clog2(DEPTH)+1){1'b0}};
        end else begin
            s_depth_commit_reg <= s_depth;
        end
    end
    assign s_status_depth_commit = s_depth_commit_reg;
    
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_depth_commit_reg <= {($clog2(DEPTH)+1){1'b0}};
        end else begin
            m_depth_commit_reg <= m_depth;
        end
    end
    assign m_status_depth_commit = m_depth_commit_reg;

    // Write enable
    wire s_write_allowed = s_axis_tvalid && s_axis_tready;
    assign s_wr_en = s_write_allowed;

    // Read enable
    wire m_read_allowed = m_axis_tvalid && m_axis_tready;
    assign m_rd_en = m_read_allowed;

    // Tready logic
    wire s_pause_active = PAUSE_ENABLE ? (s_pause_req && s_pause_ack_reg) : 1'b0;
    wire m_pause_active = PAUSE_ENABLE ? (m_pause_req && m_pause_ack_reg) : 1'b0;
    
    wire s_can_accept = !s_full && !s_pause_active;
    wire m_can_read = !m_empty && !m_pause_active;
    
    assign s_axis_tready = (DROP_WHEN_FULL || MARK_WHEN_FULL) ? 1'b1 : s_can_accept;
    assign m_axis_tvalid = m_can_read;

    // Write pointer update
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr_reg <= {(ADDR_WIDTH+1){1'b0}};
        end else if (s_wr_en) begin
            s_wr_ptr_reg <= s_wr_ptr_reg + 1'b1;
        end
    end

    // Read pointer update
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr_reg <= {(ADDR_WIDTH+1){1'b0}};
        end else if (m_rd_en) begin
            m_rd_ptr_reg <= m_rd_ptr_reg + 1'b1;
        end
    end

    // RAM implementation
    reg [TOTAL_WIDTH-1:0] ram [0:DEPTH_POW2-1];
    
    // Write to RAM
    always @(posedge s_clk) begin
        if (s_wr_en) begin
            ram[s_wr_ptr_reg[ADDR_WIDTH-1:0]] <= s_wr_data;
        end
    end

    // Read from RAM
    reg [TOTAL_WIDTH-1:0] m_rd_data_reg;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_data_reg <= {TOTAL_WIDTH{1'b0}};
        end else if (m_rd_en) begin
            m_rd_data_reg <= ram[m_rd_ptr_reg[ADDR_WIDTH-1:0]];
        end
    end
    assign m_rd_data = m_rd_data_reg;

    // Status depth
    assign s_status_depth = s_depth;
    assign m_status_depth = m_depth;

    // Overflow detection
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_overflow_reg <= 1'b0;
        end else if (s_axis_tvalid && s_full && !DROP_WHEN_FULL && !MARK_WHEN_FULL) begin
            s_overflow_reg <= 1'b1;
        end
    end
    assign s_status_overflow = s_overflow_reg;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_overflow_reg <= 1'b0;
        end else if (m_axis_tvalid && m_full) begin
            m_overflow_reg <= 1'b1;
        end
    end
    assign m_status_overflow = m_overflow_reg;

    // Frame tracking and status
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_in_frame <= 1'b0;
            s_bad_frame_reg <= 1'b0;
            s_good_frame_reg <= 1'b0;
        end else begin
            if (s_wr_en) begin
                if (s_axis_tlast) begin
                    s_in_frame <= 1'b0;
                    if (USER_ENABLE && (s_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                        s_bad_frame_reg <= 1'b1;
                    end else begin
                        s_good_frame_reg <= 1'b1;
                    end
                end else begin
                    s_in_frame <= 1'b1;
                end
            end
        end
    end
    assign s_status_bad_frame = s_bad_frame_reg;
    assign s_status_good_frame = s_good_frame_reg;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_in_frame <= 1'b0;
            m_bad_frame_reg <= 1'b0;
            m_good_frame_reg <= 1'b0;
        end else begin
            if (m_rd_en) begin
                if (m_axis_tlast) begin
                    m_in_frame <= 1'b0;
                    if (USER_ENABLE && (m_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                        m_bad_frame_reg <= 1'b1;
                    end else begin
                        m_good_frame_reg <= 1'b1;
                    end
                end else begin
                    m_in_frame <= 1'b1;
                end
            end
        end
    end
    assign m_status_bad_frame = m_bad_frame_reg;
    assign m_status_good_frame = m_good_frame_reg;

    // Pause acknowledgment
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_pause_ack_reg <= 1'b0;
        end else if (PAUSE_ENABLE) begin
            s_pause_ack_reg <= s_pause_req;
        end
    end
    assign s_pause_ack = s_pause_ack_reg;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_pause_ack_reg <= 1'b0;
        end else if (PAUSE_ENABLE) begin
            m_pause_ack_reg <= m_pause_req;
        end
    end
    assign m_pause_ack = m_pause_ack_reg;

endmodule
