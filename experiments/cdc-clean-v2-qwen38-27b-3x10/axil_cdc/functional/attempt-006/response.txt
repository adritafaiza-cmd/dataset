    reg [2:0]            aw_prot_reg;

    always @(posedge s_clk or posedge s_rst) begin

            aw_ready_reg <= 1'b0;
            aw_addr_reg  <= {ADDR_WIDTH{1'b0}};

        end else begin

                // Slave side captures AW when s_axil_awvalid && s_axil_awready
                // Then sets a flag that gets synchronized to master
            end

    // 2. The pending flag is synchronized to master clock domain.
    // 3. Master side: When pending flag is high, drive m_axil_awvalid. When m_axil_awready, clear pending.
    // 4. Similarly for W, B, AR, R channels.

    // For simplicity and correctness, I'll use a simple handshake with synchronizers.

            b_pending  <= 1'b1;
            b_resp_reg <= m_axil_bresp;
        end
    end

    reg [1:0] b_pending_sync;
    always @(posedge s_clk or posedge s_rst) begin

            b_pending_sync <= 2'b00;
        else
            b_pending_sync <= {b_pending_sync[0], b_pending};
