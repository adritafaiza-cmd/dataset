        else
            aw_s2m_sync_r <= {aw_s2m_sync_r[0], aw_s2m_req};
    end
    assign aw_s2m_sync = aw_s2m_sync_r;

    // M to S synchronizer
    wire [1:0] aw_m2s_sync;
    reg [1:0] aw_m2s_sync_r;

        if (m_rst)
            aw_m2s_sync_r <= 2'b00;

            aw_m2s_sync_r <= {aw_m2s_sync_r[0], aw_m2s_ack};
    end
    assign aw_m2s_sync = aw_m2s_sync_r;

    // S-side AW FSM
    reg aw_s_req;
    reg aw_s_ack;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            aw_s_req <= 1'b0;
            aw_s_ack <= 1'b0;
        end else begin
            if (aw_s_req && aw_m2s_sync[1]) begin
                aw_s_req <= 1'b0;
                aw_s_ack <= 1'b0;
            end else if (!aw_s_req && s_axil_awvalid && s_axil_awready) begin
                aw_s_req <= 1'b1;
            end
        end
    end
    assign aw_s2m_req = aw_s_req;
    assign s_axil_awready = !aw_s_req;

    // M-side AW FSM
    reg aw_m_req;

            aw_m_req <= 1'b0;
            aw_m_ack <= 1'b0;
        end else begin
            if (aw_m_req && aw_s2m_sync[1]) begin
                aw_m_req <= 1'b0;
                aw_m_ack <= 1'b0;
            end else if (!aw_m_req && aw_m2s_sync[1]) begin
                aw_m_req <= 1'b1;
            end
        end
    end
    assign aw_m2s_ack = aw_m_ack;
    assign m_axil_awvalid = aw_m_req;

    // ========================================================================
    // Write Data Channel (W) - S to M
    // ========================================================================
    wire [DATA_WIDTH+STRB_WIDTH:0] w_payload;
    assign w_payload = {s_axil_wdata, s_axil_wstrb};

    wire w_s2m_ack;
    wire w_m2s_req;
    wire w_m2s_ack;

    // S to M synchronizer
    wire [1:0] w_s2m_sync;
    reg [1:0] w_s2m_sync_r;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            w_s2m_sync_r <= 2'b00;
        else
            w_s2m_sync_r <= {w_s2m_sync_r[0], w_s2m_req};
    end
    assign w_s2m_sync = w_s2m_sync_r;

    // M to S synchronizer
    wire [1:0] w_m2s_sync;
    reg [1:0] w_m2s_sync_r;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            w_m2s_sync_r <= 2'b00;
        else
            w_m2s_sync_r <= {w_m2s_sync_r[0], w_m2s_ack};

            if (w_s_req && w_m2s_sync[1]) begin

            end
        end

    assign s_axil_wready = !w_s_req;

    reg w_m_req;
    reg w_m_ack;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            w_m_req <= 1'b0;
            w_m_ack <= 1'b0;
        end else begin
            if (w_m_req && w_s2m_sync[1]) begin
                w_m_req <= 1'b0;
                w_m_ack <= 1'b0;

                w_m_req <= 1'b1;
            end
        end
    end
    assign w_m2s_ack = w_m_ack;
    assign m_axil_wvalid = w_m_req;

    // ========================================================================
    // Write Response Channel (B) - M to S
    // ========================================================================
    wire [1:0] b_payload;
    assign b_payload = m_axil_bresp;

    wire b_m2s_req;
    wire b_m2s_ack;
    wire b_s2m_req;
    wire b_s2m_ack;

    // M to S synchronizer
    wire [1:0] b_m2s_sync;
    reg [1:0] b_m2s_sync_r;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            b_m2s_sync_r <= 2'b00;
        else
            b_m2s_sync_r <= {b_m2s_sync_r[0], b_m2s_req};
    end
    assign b_m2s_sync = b_m2s_sync_r;

    // S to M synchronizer
    wire [1:0] b_s2m_sync;
    reg [1:0] b_s2m_sync_r;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)

    reg b_m_req;

            b_m_ack <= 1'b0;

                b_m_ack <= 1'b0;

    reg [DATA_WIDTH-1:0] m_wdata_r;
    reg [STRB_WIDTH-1:0] m_wstrb_r;

    reg [2:0]            m_arprot_r;

    always @(posedge m_clk or posedge m_rst) begin
