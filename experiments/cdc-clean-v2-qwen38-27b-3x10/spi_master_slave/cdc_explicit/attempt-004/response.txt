module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

    // ========================================================================
    // Reset Synchronization (RDC safe)
    // ========================================================================
    // Sclk domain reset
    reg [2:0] sclk_rst_sync;
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i)
            sclk_rst_sync <= 3'b000;
        else
            sclk_rst_sync <= {sclk_rst_sync[1:0], 1'b1};
    end
    wire sclk_rst = ~sclk_rst_sync[2];

    // Pclk domain reset
    reg [2:0] pclk_rst_sync;
    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i)
            pclk_rst_sync <= 3'b000;
        else
            pclk_rst_sync <= {pclk_rst_sync[1:0], 1'b1};
    end
    wire pclk_rst = ~pclk_rst_sync[2];

    // ========================================================================
    // Pclk Domain: Parallel Interface & Handshake
    // ========================================================================
    // State machine for parallel side
    // IDLE: waiting for wren_i
    // ACK:  pulse wr_ack_o, capture di_i
    // WAIT: wait for sclk domain to signal word accepted (tx_start)
    // REQ:  pulse di_req_o, wait for wren_i to go low (return-to-zero)
    // IDLE: back to idle

    localparam S_IDLE = 2'd0;
    localparam S_ACK  = 2'd1;
    localparam S_WAIT = 2'd2;
    localparam S_REQ  = 2'd3;

    reg [1:0] p_state;
    reg [N-1:0] tx_word_reg;
    reg [N-1:0] rx_word_reg;
    reg         tx_start_pulse;
    reg         rx_done_pulse;
    reg         di_req_pulse;
    reg         wr_ack_pulse;
    reg         do_valid_pulse;

    // Synchronize single-bit signals from sclk domain to pclk domain
    // tx_start: sclk -> pclk
    reg [2:0] tx_start_sync;
    always @(posedge pclk_i or posedge pclk_rst) begin
        if (pclk_rst)
            tx_start_sync <= 3'b000;
        else
            tx_start_sync <= {tx_start_sync[1:0], tx_start_pulse};
    end
    wire tx_start_pclk = tx_start_sync[2];

    // rx_done: sclk -> pclk
    reg [2:0] rx_done_sync;
    always @(posedge pclk_i or posedge pclk_rst) begin
        if (pclk_rst)
            rx_done_sync <= 3'b000;
        else
            rx_done_sync <= {rx_done_sync[1:0], rx_done_pulse};
    end
    wire rx_done_pclk = rx_done_sync[2];

    // Synchronize rx_word from sclk domain to pclk domain (multi-bit, use handshake)
    // We use a simple approach: when rx_done is asserted, the rx_word is stable.
    // We synchronize the data using a 2-stage synchronizer per bit is not ideal for multi-bit.
    // Instead, we use a valid/data handshake: sclk domain holds rx_word stable until pclk acknowledges.
    // For simplicity and CDC safety, we'll use a 2-phase handshake for the data.
    
    // Actually, let's use a simpler approach: since rx_done is a single pulse, and we know
    // the data is stable when rx_done is high, we can synchronize the data bits individually
    // but that's not coherent. Better to use a handshake.
    
    // Let's use a 2-phase handshake for rx_word transfer:
    // sclk domain: rx_word_valid (high when data ready)
    // pclk domain: rx_word_ack (high when data captured)
    
    reg rx_word_valid_sclk;  // in sclk domain
    reg rx_word_ack_pclk;    // in pclk domain
    
    // Synchronize rx_word_valid from sclk to pclk
    reg [2:0] rx_word_valid_sync;
    always @(posedge pclk_i or posedge pclk_rst) begin
        if (pclk_rst)
            rx_word_valid_sync <= 3'b000;
        else
            rx_word_valid_sync <= {rx_word_valid_sync[1:0], rx_word_valid_sclk};
    end
    wire rx_word_valid_pclk = rx_word_valid_sync[2];

    // Synchronize rx_word_ack from pclk to sclk
    reg [2:0] rx_word_ack_sync;
    always @(posedge sclk_i or posedge sclk_rst) begin
        if (sclk_rst)
            rx_word_ack_sync <= 3'b000;
        else
            rx_word_ack_sync <= {rx_word_ack_sync[1:0], rx_word_ack_pclk};
    end
    wire rx_word_ack_sclk = rx_word_ack_sync[2];

    // Synchronize tx_word from pclk to sclk domain (multi-bit, use handshake)
    // pclk domain: tx_word_valid (high when data ready)
    // sclk domain: tx_word_ack (high when data captured)
    
    reg tx_word_valid_pclk;  // in pclk domain
    reg tx_word_ack_sclk;    // in sclk domain
    
    // Synchronize tx_word_valid from pclk to sclk
    reg [2:0] tx_word_valid_sync;
    always @(posedge sclk_i or posedge sclk_rst) begin
        if (sclk_rst)
            tx_word_valid_sync <= 3'b000;
        else
            tx_word_valid_sync <= {tx_word_valid_sync[1:0], tx_word_valid_pclk};
    end
    wire tx_word_valid_sclk = tx_word_valid_sync[2];

    // Synchronize tx_word_ack from sclk to pclk
    reg [2:0] tx_word_ack_sync;
    always @(posedge pclk_i or posedge pclk_rst) begin
        if (pclk_rst)
            tx_word_ack_sync <= 3'b000;
        else
            tx_word_ack_sync <= {tx_word_ack_sync[1:0], tx_word_ack_sclk};
    end
    wire tx_word_ack_pclk = tx_word_ack_sync[2];

    // Pclk domain state machine
    always @(posedge pclk_i or posedge pclk_rst) begin
        if (pclk_rst) begin
            p_state        <= S_IDLE;
            tx_word_reg    <= {N{1'b0}};
            rx_word_reg    <= {N{1'b0}};
            tx_start_pulse <= 1'b0;
            rx_done_pulse  <= 1'b0;
            di_req_pulse   <= 1'b0;
            wr_ack_pulse   <= 1'b0;
            do_valid_pulse <= 1'b0;
            tx_word_valid_pclk <= 1'b0;
            rx_word_ack_pclk   <= 1'b0;
            di_req_o     <= 1'b0;
            wr_ack_o     <= 1'b0;
            do_valid_o   <= 1'b0;
            do_o         <= {N{1'b0}};
        end else begin
            // Default: deassert pulses
            tx_start_pulse <= 1'b0;
            rx_done_pulse  <= 1'b0;
            di_req_pulse   <= 1'b0;
            wr_ack_pulse   <= 1'b0;
            do_valid_pulse <= 1'b0;
            di_req_o       <= 1'b0;
            wr_ack_o       <= 1'b0;
            do_valid_o     <= 1'b0;

            case (p_state)
                S_IDLE: begin
                    if (wren_i) begin
                        // Capture data and acknowledge
                        tx_word_reg <= di_i;
                        wr_ack_o    <= 1'b1;
                        tx_word_valid_pclk <= 1'b1;
                        p_state     <= S_ACK;
                    end
                end

                S_ACK: begin
                    // Wait for sclk domain to acknowledge tx_word
                    if (tx_word_ack_pclk) begin
                        tx_word_valid_pclk <= 1'b0;
                        p_state            <= S_WAIT;
                    end
                end

                S_WAIT: begin
                    // Wait for rx_done from sclk domain
                    if (rx_done_pclk) begin
                        // Capture rx_word
                        rx_word_reg  <= rx_word_reg; // placeholder, actual capture below
                        do_valid_o   <= 1'b1;
                        do_o         <= rx_word_reg;
                        p_state      <= S_REQ;
                    end
                end

                S_REQ: begin
                    // Pulse di_req_o and wait for wren_i to return low
                    di_req_o <= 1'b1;
                    if (!wren_i) begin
                        di_req_o <= 1'b0;
                        p_state  <= S_IDLE;
                    end
                end

                default: p_state <= S_IDLE;
            endcase
        end
    end

    // ========================================================================
    // Sclk Domain: SPI Serial Interface
    // ========================================================================
    // State machine for SPI side
    // IDLE: ssel high (inactive), sck low
    // START: ssel low (active), prepare to shift
    // SHIFT: shift out tx_word, shift in rx_word
    // DONE:  ssel high, signal rx_done

    localparam SPI_IDLE  = 2'd0;
    localparam SPI_START = 2'd1;
    localparam SPI_SHIFT = 2'd2;
    localparam SPI_DONE  = 2'd3;

    reg [1:0] spi_state;
    reg [N-1:0] spi_tx_shift;
    reg [N-1:0] spi_rx_shift;
    reg [N-1:0] spi_rx_word;
    reg [N-1:0] spi_tx_word;
    reg         spi_tx_word_valid;
    reg         spi_tx_word_ack;
    reg         spi_rx_word_valid;
    reg         spi_rx_word_ack;
    reg         spi_tx_start;
    reg         spi_rx_done;
    reg [N-1:0] spi_mosi_reg;
    reg         spi_ssel_reg;
    reg         spi_sck_reg;
    reg [3:0]   spi_clk_div_cnt;
    reg         spi_clk_phase;

    // Sclk domain state machine
    always @(posedge sclk_i or posedge sclk_rst) begin
        if (sclk_rst) begin
            spi_state       <= SPI_IDLE;
            spi_tx_shift    <= {N{1'b0}};
            spi_rx_shift    <= {N{1'b0}};
            spi_rx_word     <= {N{1'b0}};
            spi_tx_word     <= {N{1'b0}};
            spi_tx_word_valid <= 1'b0;
            spi_tx_word_ack   <= 1'b0;
            spi_rx_word_valid <= 1'b0;
            spi_rx_word_ack   <= 1'b0;
            spi_tx_start    <= 1'b0;
            spi_rx_done     <= 1'b0;
            spi_mosi_reg    <= {N{1'b0}};
            spi_ssel_reg    <= 1'b1;
            spi_sck_reg     <= 1'b0;
            spi_clk_div_cnt <= 4'd0;
            spi_clk_phase   <= 1'b0;
            spi_ssel_o      <= 1'b1;
            spi_sck_o       <= 1'b0;
            spi_mosi_o      <= 1'b0;
        end else begin
            // Default: deassert pulses
            spi_tx_start  <= 1'b0;
            spi_rx_done   <= 1'b0;

            case (spi_state)
                SPI_IDLE: begin
                    spi_ssel_o <= 1'b1;
                    spi_sck_o  <= 1'b0;
                    spi_mosi_o <= 1'b0;
                    if (tx_word_valid_sclk) begin
                        // Capture tx_word
                        spi_tx_word     <= tx_word_reg;
                        spi_tx_word_ack <= 1'b1;
                        spi_state       <= SPI_START;
                    end
                end

                SPI_START: begin
                    spi_ssel_o <= 1'b0;
                    spi_sck_o  <= 1'b0;
                    spi_mosi_o <= spi_tx_word[N-1];
                    spi_tx_shift <= spi_tx_word;
                    spi_rx_shift <= {N{1'b0}};
                    spi_clk_div_cnt <= 4'd0;
                    spi_clk_phase   <= 1'b0;
                    spi_state       <= SPI_SHIFT;
                end

                SPI_SHIFT: begin
                    // Clock divider
                    if (spi_clk_div_cnt == SPI_2X_CLK_DIV - 1) begin
                        spi_clk_div_cnt <= 4'd0;
                        spi_clk_phase   <= ~spi_clk_phase;
                        
                        if (spi_clk_phase) begin
                            // Rising edge of SCK: capture MISO
                            spi_rx_shift <= {spi_rx_shift[N-2:0], spi_miso_i};
                            
                            // Check if we've shifted all bits
                            if (spi_clk_div_cnt == SPI_2X_CLK_DIV - 1 && 
                                spi_clk_phase && 
                                // This is the last bit
                                1'b0) begin
                                // Handled below
                            end
                        end else begin
                            // Falling edge of SCK: update MOSI
                            spi_mosi_o <= spi_tx_shift[N-1];
                            spi_tx_shift <= {spi_tx_shift[N-2:0], 1'b0};
                        end
                    end else begin
                        spi_clk_div_cnt <= spi_clk_div_cnt + 1'b1;
                    end

                    // Check for completion: after N bits have been shifted
                    // We need a bit counter
                    // Let's add a bit counter
                end

                SPI_DONE: begin
                    spi_ssel_o <= 1'b1;
                    spi_sck_o  <= 1'b0;
                    spi_mosi_o <= 1'b0;
                    spi_rx_word     <= spi_rx_shift;
                    spi_rx_word_valid <= 1'b1;
                    spi_rx_done     <= 1'b1;
                    spi_state       <= SPI_IDLE;
                end

                default: spi_state <= SPI_IDLE;
            endcase
        end
    end

    // Bit counter for SPI shift
    reg [N-1:0] spi_bit_cnt;
    always @(posedge sclk_i or posedge sclk_rst) begin
        if (sclk_rst)
            spi_bit_cnt <= {N{1'b0}};
        else if (spi_state == SPI_START)
            spi_bit_cnt <= {N{1'b0}};
        else if (spi_state == SPI_SHIFT && spi_clk_phase == 1'b0 && spi_clk_div_cnt == SPI_2X_CLK_DIV - 1)
            spi_bit_cnt <= spi_bit_cnt + 1'b1;
    end
