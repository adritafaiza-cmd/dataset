```verilog
module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

    // SPI clock domain signals
    reg [N-1:0] tx_shift_reg;
    reg [N-1:0] rx_shift_reg;
    reg [N-1:0] bit_counter;
    reg spi_active;
    reg spi_sck_reg;
    reg [N-1:0] spi_clk_divider;
    reg spi_clk_edge;

    // Parallel clock domain signals
    reg [N-1:0] tx_word_reg;
    reg [N-1:0] rx_word_reg;
    reg tx_word_valid;
    reg rx_word_valid;
    reg di_req_reg;
    reg wr_ack_reg;
    reg do_valid_reg;
    reg [N-1:0] do_reg;

    // Synchronizers for clock domain crossings
    reg [1:0] tx_word_valid_sync;
    reg [1:0] rx_word_valid_sync;
    reg [1:0] di_req_sync;
    reg [1:0] wr_ack_sync;
    reg [1:0] do_valid_sync;

    // Synchronizer for tx_word
    reg [N-1:0] tx_word_sync;
    reg tx_word_sync_valid;
    reg [1:0] tx_word_sync_valid_sync;

    // Synchronizer for rx_word
    reg [N-1:0] rx_word_sync;
    reg rx_word_sync_valid;
    reg [1:0] rx_word_sync_valid_sync;

    // Synchronizer for do_valid
    reg [1:0] do_valid_sync_pclk;
    reg [N-1:0] do_sync;
    reg do_sync_valid;
    reg [1:0] do_sync_valid_sync;

    // Synchronizer for di_req
    reg [1:0] di_req_sync_sclk;

    // Synchronizer for wr_ack
    reg [1:0] wr_ack_sync_sclk;

    // SPI clock generation
    wire spi_clk = spi_sck_reg;

    // SPI state machine
    localparam SPI_IDLE = 2'd0;
    localparam SPI_SHIFT = 2'd1;
    localparam SPI_DONE = 2'd2;

    reg [1:0] spi_state;
    reg [1:0] spi_state_next;

    // SPI clock divider
    reg [N-1:0] spi_clk_div;

    // SPI shift logic
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            spi_ssel_o <= 1'b1;
            spi_sck_o <= 1'b0;
            tx_shift_reg <= {N{1'b0}};
            rx_shift_reg <= {N{1'b0}};
            bit_counter <= {N{1'b0}};
            spi_active <= 1'b0;
            spi_sck_reg <= 1'b0;
            spi_clk_div <= {N{1'b0}};
            spi_state <= SPI_IDLE;
        end else begin
            spi_ssel_o <= (spi_state == SPI_IDLE) ? 1'b1 : 1'b0;
            spi_sck_o <= spi_sck_reg;

            case (spi_state)
                SPI_IDLE: begin
                    spi_sck_reg <= 1'b0;
                    spi_clk_div <= {N{1'b0}};
                    if (tx_word_sync_valid_sync[1]) begin
                        tx_shift_reg <= tx_word_sync;
                        rx_shift_reg <= {N{1'b0}};
                        bit_counter <= {N{1'b0}};
                        spi_state <= SPI_SHIFT;
                    end
                end
                SPI_SHIFT: begin
                    // Generate SPI clock
                    spi_clk_div <= spi_clk_div + 1'b1;
                    if (spi_clk_div == SPI_2X_CLK_DIV - 1) begin
                        spi_clk_div <= {N{1'b0}};
                        spi_sck_reg <= ~spi_sck_reg;
                    end

                    // On rising edge of SPI clock, shift out and shift in
                    if (spi_sck_reg == 1'b1 && spi_clk_div == SPI_2X_CLK_DIV - 1) begin
                        // This is the rising edge
                        tx_shift_reg <= {tx_shift_reg[N-2:0], 1'b0};
                        rx_shift_reg <= {spi_miso_i, rx_shift_reg[N-1:0]};
                        bit_counter <= bit_counter + 1'b1;
                        if (bit_counter == N - 1) begin
                            spi_state <= SPI_DONE;
                        end
                    end
                end
                SPI_DONE: begin
                    spi_sck_reg <= 1'b0;
                    spi_state <= SPI_IDLE;
                end
                default: spi_state <= SPI_IDLE;
            endcase
        end
    end

    // SPI MOSI output
    assign spi_mosi_o = tx_shift_reg[N-1];

    // Parallel clock domain - handshake logic
    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            di_req_o <= 1'b1;
            wr_ack_o <= 1'b0;
            do_valid_o <= 1'b0;
            do_o <= {N{1'b0}};
            tx_word_reg <= {N{1'b0}};
            rx_word_reg <= {N{1'b0}};
            tx_word_valid <= 1'b0;
            rx_word_valid <= 1'b0;
            di_req_reg <= 1'b1;
            wr_ack_reg <= 1'b0;
            do_valid_reg <= 1'b0;
            do_reg <= {N{1'b0}};
        end else begin
            // Default: no pulses
            wr_ack_o <= 1'b0;
            do_valid_o <= 1'b0;

            // Accept new word if di_req is high and wren is high
            if (di_req_o && wren_i) begin
                tx_word_reg <= di_i;
                tx_word_valid <= 1'b1;
                wr_ack_o <= 1'b1;
                di_req_o <= 1'b0;
            end

            // When rx_word_valid is set, output the received word
            if (rx_word_valid) begin
                do_o <= rx_word_reg;
                do_valid_o <= 1'b1;
                rx_word_valid <= 1'b0;
                di_req_o <= 1'b1;
            end
        end
    end

    // Synchronizers in pclk domain
    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            tx_word_valid_sync <= 2'b00;
            rx_word_valid_sync <= 2'b00;
            di_req_sync <= 2'b00;
            wr_ack_sync <= 2'b00;
            do_valid_sync <= 2'b00;
            tx_word_sync <= {N{1'b0}};
            tx_word_sync_valid <= 1'b0;
            tx_word_sync_valid_sync <= 2'b00;
            rx_word_sync <= {N{1'b0}};
            rx_word_sync_valid <= 1'b0;
            rx_word_sync_valid_sync <= 2'b00;
            do_valid_sync_pclk <= 2'b00;
            do_sync <= {N{1'b0}};
            do_sync_valid <= 1'b0;
            do_sync_valid_sync <= 2'b00;
        end else begin
            // Synchronize tx_word_valid from sclk domain
            tx_word_valid_sync <= {tx_word_valid_sync[0], tx_word_valid};

            // Synchronize rx_word_valid from sclk domain
            rx_word_valid_sync <= {rx_word_valid_sync[0], rx_word_valid};

            // Synchronize di_req from sclk domain
            di_req_sync <= {di_req_sync[0], di_req_reg};

            // Synchronize wr_ack from sclk domain
            wr_ack_sync <= {wr_ack_sync[0], wr_ack_reg};

            // Synchronize do_valid from sclk domain
            do_valid_sync <= {do_valid_sync[0], do_valid_reg};

            // Synchronize tx_word data
            if (tx_word_valid_sync[1] && !tx_word_valid_sync[0]) begin
                tx_word_sync <= tx_word_reg;
                tx_word_sync_valid <= 1'b1;
            end else if (tx_word_sync_valid_sync[1]) begin
                tx_word_sync_valid <= 1'b0;
            end
            tx_word_sync_valid_sync <= {tx_word_sync_valid_sync[0], tx_word_sync_valid};

            // Synchronize rx_word data
            if (rx_word_valid_sync[1] && !rx_word_valid_sync[0]) begin
                rx_word_sync <= rx_word_reg;
                rx_word_sync_valid <= 1'b1;
            end else if (rx_word_sync_valid_sync[1]) begin
                rx_word_sync_valid <= 1'b0;
            end
            rx_word_sync_valid_sync <= {rx_word_sync_valid_sync[0], rx_word_sync_valid};

            // Synchronize do_valid and do data
            if (do_valid_sync[1] && !do_valid_sync[0]) begin
                do_sync <= do_reg;
                do_sync_valid <= 1'b1;
            end else if (do_sync_valid_sync[1]) begin
                do_sync_valid <= 1'b0;
            end
            do_sync_valid_sync <= {do_sync_valid_sync[0], do_sync_valid};
        end
    end

    // Synchronizers in sclk domain
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            di_req_sync_sclk <= 2'b00;
            wr_ack_sync_sclk <= 2'b00;
            tx_word_valid <= 1'b0;
            rx_word_valid <= 1'b0;
            di_req_reg <= 1'b1;
            wr_ack_reg <= 1'b0;
            do_valid_reg <= 1'b0;
            do_reg <= {N{1'b0}};
            tx_word_reg <= {N{1'b0}};
            rx_word_reg <= {N{1'b0}};
        end else begin
            // Synchronize di_req from pclk domain
            di_req_sync_sclk <= {di_req_sync_sclk[0], di_req_o};

            // Synchronize wr_ack from pclk domain
            wr_ack_sync_sclk <= {wr_ack_sync_sclk[0], wr_ack_o};

            // When tx_word_valid_sync indicates a new word is available
            if (tx_word_sync_valid_sync[1] && !tx_word_sync_valid_sync[0]) begin
                tx_word_reg <= tx_word_sync;
                tx_word_valid <= 1'b1;
            end else begin
                tx_word_valid <= 1'b0;
            end

            // When rx_word_valid_sync indicates a received word is available
            if (rx_word_sync_valid_sync[1] && !rx_word_sync_valid_sync[0]) begin
                rx_word_reg <= rx_word_sync;
                rx_word_valid <= 1'b1;
            end else begin
                rx_word_valid <= 1'b0;
            end

            // When do_valid_sync indicates a valid output
            if (do_valid_sync[1] && !do_valid_sync[0]) begin
                do_reg <= do_sync;
                do_valid_reg <= 1'b1;
            end else begin
                do_valid_reg <= 1'b0;
            end

            // Update di_req_reg based on synchronized di_req
            di_req_reg <= di_req_sync_sclk[1];

            // Update wr_ack_reg based on synchronized wr_ack
            wr_ack_reg <= wr_ack_sync_sclk[1];
        end
    end

    // When SPI completes a transaction, signal rx_word_valid
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            rx_word_valid <= 1'b0;
        end else begin
            if (spi_state == SPI_DONE) begin
                rx_word_reg <= rx_shift_reg;
                rx_word_valid <= 1'b1;
            end else begin
                rx_word_valid <= 1'b0;
            end
        end
    end

endmodule
