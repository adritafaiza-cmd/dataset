module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

    // Internal signals
    reg [N-1:0] tx_word;
    reg [N-1:0] rx_word;
    reg [3:0]   bit_cnt;
    reg         busy;
    reg         tx_valid;
    reg         rx_valid;

    // SPI clock generation
    reg [3:0]   sclk_div_cnt;
    reg         sclk_phase;
    reg         sclk_out;

    // MOSI output
    assign spi_mosi_o = tx_word[N-1];

    // SPI clock generation logic
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            sclk_div_cnt <= 4'd0;
            sclk_phase   <= 1'b0;
            sclk_out     <= 1'b0;
        end else if (busy) begin
            if (sclk_div_cnt == SPI_2X_CLK_DIV - 1) begin
                sclk_div_cnt <= 4'd0;
                sclk_phase   <= ~sclk_phase;
                sclk_out     <= sclk_phase;
            end else begin
                sclk_div_cnt <= sclk_div_cnt + 4'd1;
            end
        end else begin
            sclk_div_cnt <= 4'd0;
            sclk_phase   <= 1'b0;
            sclk_out     <= 1'b0;
        end
    end

    assign spi_sck_o = sclk_out;

    // SPI data shift logic
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            tx_word   <= {N{1'b0}};
            rx_word   <= {N{1'b0}};
            bit_cnt   <= 4'd0;
            busy      <= 1'b0;
            tx_valid  <= 1'b0;
            rx_valid  <= 1'b0;
            spi_ssel_o <= 1'b1;
        end else if (busy) begin
            // On rising edge of sclk_out, shift out MSB and shift in MISO
            if (sclk_out == 1'b1 && sclk_phase == 1'b1) begin
                // This is the rising edge of the SPI clock
                tx_word <= {tx_word[N-2:0], 1'b0};
                rx_word <= {rx_word[N-2:0], spi_miso_i};
                if (bit_cnt == N - 1) begin
                    bit_cnt  <= 4'd0;
                    busy     <= 1'b0;
                    rx_valid <= 1'b1;
                    spi_ssel_o <= 1'b1;
                end else begin
                    bit_cnt <= bit_cnt + 4'd1;
                end
            end
        end else if (tx_valid) begin
            // Start a new transaction
            tx_word    <= di_i;
            rx_word    <= {N{1'b0}};
            bit_cnt    <= 4'd0;
            busy       <= 1'b1;
            tx_valid   <= 1'b0;
            spi_ssel_o <= 1'b0;
        end
    end

    // Parallel side logic
    reg        wr_ack_reg;
    reg        do_valid_reg;
    reg [N-1:0] do_reg;
    reg        di_req_reg;

    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            wr_ack_reg   <= 1'b0;
            do_valid_reg <= 1'b0;
            do_reg       <= {N{1'b0}};
            di_req_reg   <= 1'b1;
        end else begin
            // Default: deassert one-cycle pulses
            wr_ack_reg   <= 1'b0;
            do_valid_reg <= 1'b0;

            // Accept new word when not busy and wren is asserted
            if (di_req_reg && wren_i) begin
                wr_ack_reg <= 1'b1;
                di_req_reg <= 1'b0;
                tx_valid   <= 1'b1;
            end

            // When receive is valid, output the word and re-enable di_req
            if (rx_valid) begin
                do_valid_reg <= 1'b1;
                do_reg       <= rx_word;
                di_req_reg   <= 1'b1;
                rx_valid     <= 1'b0;
            end
        end
    end

    assign wr_ack_o   = wr_ack_reg;
    assign do_valid_o = do_valid_reg;
    assign do_o       = do_reg;
    assign di_req_o   = di_req_reg;

endmodule
