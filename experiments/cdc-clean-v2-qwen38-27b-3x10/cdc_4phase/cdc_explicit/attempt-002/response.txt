      src_req_sync <= {src_req_sync[0], src_req_r};
  end

  // Synchronize dst_ack to src domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni)
      dst_ack_sync <= 2'b00;
    else
      dst_ack_sync <= {dst_ack_sync[0], dst_ack_r};
  end

  // Source domain logic
  assign src_ready_o = !src_pending_r;

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_req_r <= 1'b0;
      src_data_hold_r <= {WIDTH{1'b0}};
      src_pending_r <= 1'b0;
    end else begin
      if (src_req_r && dst_ack_sync[1]) begin
        src_req_r <= 1'b0;
        src_pending_r <= 1'b0;
      end
      else if (!src_pending_r && src_valid_i) begin
        src_data_hold_r <= src_data_i;
        src_req_r <= 1'b1;
        src_pending_r <= 1'b1;
      end
    end
  end

  // Destination domain logic
  assign dst_ack = dst_ack_r;

  // Detect rising edge of src_req in dst domain
  wire src_req_rise = src_req_sync[1] && !src_req_sync[0];

  // Data synchronization: 2-stage pipeline
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      data_sync_1 <= {WIDTH{1'b0}};
      data_sync_2 <= {WIDTH{1'b0}};
      data_capture_en <= 1'b0;
      dst_ack_r <= 1'b0;
      dst_data_hold_r <= {WIDTH{1'b0}};
      dst_pending_r <= 1'b0;
    end else begin
      // If we have a pending ack and src_req has fallen, clear it
      if (dst_ack_r && !src_req_sync[1]) begin
        dst_ack_r <= 1'b0;
        dst_pending_r <= 1'b0;
        data_capture_en <= 1'b0;
      end
      // If src_req rises, start data capture
      else if (src_req_rise) begin
        data_sync_1 <= src_data_hold_r;
        data_capture_en <= 1'b1;
      end
      // If data capture is enabled, advance the pipeline
      else if (data_capture_en) begin
        data_sync_2 <= data_sync_1;
        data_capture_en <= 1'b0;
        dst_data_hold_r <= data_sync_2;
        dst_ack_r <= 1'b1;
        dst_pending_r <= 1'b1;
      end
    end
  end

  // Output
  assign dst_data_o = dst_data_hold_r;
  assign dst_valid_o = dst_pending_r;

endmodule
