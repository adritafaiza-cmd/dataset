module cdc_4phase #(
  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Internal signals
  wire [WIDTH-1:0] src_payload;
  wire src_req;
  wire dst_ack;
  wire dst_req;
  wire src_ack;
  wire [WIDTH-1:0] dst_payload;

  // Source domain: capture data and generate request
  reg [WIDTH-1:0] src_data_reg;
  reg src_valid_reg;
  reg src_ready_int;
  reg src_req_reg;

  // Destination domain: acknowledge and generate request back
  reg dst_ack_reg;
  reg dst_req_reg;
  reg [WIDTH-1:0] dst_data_reg;
  reg dst_valid_reg;

  // Synchronizers for control signals
  // src_req -> dst domain
  reg [1:0] src_req_sync;
  // dst_ack -> src domain
  reg [1:0] dst_ack_sync;
  // dst_req -> src domain
  reg [1:0] dst_req_sync;
  // src_ack -> dst domain
  reg [1:0] src_ack_sync;

  // Synchronizers for payload (multi-bit, must be stable during transfer)
  // src_payload -> dst domain
  reg [WIDTH-1:0] src_payload_sync1;
  reg [WIDTH-1:0] src_payload_sync2;
  // dst_payload -> src domain
  reg [WIDTH-1:0] dst_payload_sync1;
  reg [WIDTH-1:0] dst_payload_sync2;

  // Source domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_reg <= {WIDTH{1'b0}};
      src_valid_reg <= 1'b0;
      src_ready_int <= 1'b1;
      src_req_reg <= 1'b0;
    end else begin
      // Accept data when valid and ready
      if (src_valid_i && src_ready_int) begin
        src_data_reg <= src_data_i;
        src_valid_reg <= 1'b1;
        src_ready_int <= 1'b0;
        src_req_reg <= 1'b1;
      end else if (dst_ack_sync[1]) begin
        // Acknowledge received from destination
        src_valid_reg <= 1'b0;
        src_ready_int <= 1'b1;
        src_req_reg <= 1'b0;
      end
    end
  end

  // Destination domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_ack_reg <= 1'b0;
      dst_req_reg <= 1'b0;
      dst_data_reg <= {WIDTH{1'b0}};
      dst_valid_reg <= 1'b0;
    end else begin
      // When request from source is detected
      if (src_req_sync[1] && !dst_ack_reg) begin
        dst_ack_reg <= 1'b1;
        dst_data_reg <= src_payload_sync2;
        dst_valid_reg <= 1'b1;
      end else if (dst_valid_reg && dst_ready_i) begin
        // Destination accepts the data
        dst_valid_reg <= 1'b0;
        dst_ack_reg <= 1'b0;
        dst_req_reg <= 1'b1;
      end else if (src_ack_sync[1]) begin
        // Acknowledge from source received
        dst_req_reg <= 1'b0;
      end
    end
  end

  // Synchronizers
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_req_sync <= 2'b00;
    end else begin
      src_req_sync <= {src_req_sync[0], src_req_reg};
    end
  end

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_ack_sync <= 2'b00;
    end else begin
      dst_ack_sync <= {dst_ack_sync[0], dst_ack_reg};
    end
  end

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_req_sync <= 2'b00;
    end else begin
      dst_req_sync <= {dst_req_sync[0], dst_req_reg};
    end
  end

  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_ack_sync <= 2'b00;
    end else begin
      src_ack_sync <= {src_ack_sync[0], src_req_reg};
    end
  end

  // Payload synchronizers - only update when request is active
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_payload_sync1 <= {WIDTH{1'b0}};
      src_payload_sync2 <= {WIDTH{1'b0}};
    end else begin
      if (src_req_sync[0]) begin
        src_payload_sync1 <= src_data_reg;
      end
      src_payload_sync2 <= src_payload_sync1;
    end
  end

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_payload_sync1 <= {WIDTH{1'b0}};
      dst_payload_sync2 <= {WIDTH{1'b0}};
    end else begin
      if (dst_req_sync[0]) begin
        dst_payload_sync1 <= dst_data_reg;
      end
      dst_payload_sync2 <= dst_payload_sync1;
    end
  end

  // Output assignments
  assign src_ready_o = src_ready_int;
  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

endmodule
