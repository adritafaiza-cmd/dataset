module cdc_4phase #(
  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Internal signals
  wire [WIDTH-1:0] src_payload;
  wire src_req;
  wire dst_ack;
  wire dst_req;
  wire src_ack;
  wire [WIDTH-1:0] dst_payload;

  // Source domain: capture data and generate request
  // Destination domain: acknowledge request and generate new request for next transfer

  // Source side: hold payload stable while request is pending
  reg [WIDTH-1:0] src_hold_data;
  reg src_hold_valid;
  reg src_req_reg;

  // Destination side: hold payload and generate valid
  reg [WIDTH-1:0] dst_hold_data;
  reg dst_valid_reg;
  reg dst_req_reg;

  // Synchronizers for request/acknowledge signals
  // src_req goes from src to dst
  reg [1:0] src_req_sync;
  // dst_ack goes from dst to src
  reg [1:0] dst_ack_sync;
  // dst_req goes from dst to src
  reg [1:0] dst_req_sync;
  // src_ack goes from src to dst
  reg [1:0] src_ack_sync;

  // Synchronizers for payload data (multi-bit, must be stable during transfer)
  // We use a handshake-based approach where payload is held stable
  // and synchronized through 2 stages after the control signal settles

  // Source domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_hold_data <= {WIDTH{1'b0}};
      src_hold_valid <= 1'b0;
      src_req_reg <= 1'b0;
    end else begin
      // If we have a pending request, wait for ack
      if (src_req_reg) begin
        if (dst_ack_sync[1]) begin
          // Ack received, clear request
          src_req_reg <= 1'b0;
          src_hold_valid <= 1'b0;
        end
      end else begin
        // No pending request, can accept new data
        if (src_valid_i && src_ready_o) begin
          src_hold_data <= src_data_i;
          src_hold_valid <= 1'b1;
          src_req_reg <= 1'b1;
        end
      end
    end
  end

  // Source ready: can accept new data if no pending request
  // In DECOUPLED mode, we can accept new data before the previous ack returns
  // But for simplicity and safety, we use a standard 4-phase handshake
  // where we wait for ack before accepting new data
  assign src_ready_o = !src_req_reg;

  // Destination domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_hold_data <= {WIDTH{1'b0}};
      dst_valid_reg <= 1'b0;
      dst_req_reg <= 1'b0;
    end else begin
      // If we have a pending valid, wait for ready
      if (dst_valid_reg) begin
        if (dst_ready_i) begin
          // Handshake complete, clear valid
          dst_valid_reg <= 1'b0;
          // Generate ack to source
          dst_req_reg <= 1'b1;
        end
      end else begin
        // No pending valid, check for new request from source
        if (src_req_sync[1]) begin
          // New request received, capture payload
          dst_hold_data <= src_hold_data;
          dst_valid_reg <= 1'b1;
          // Clear the request by sending ack
          dst_req_reg <= 1'b1;
        end
      end
    end
  end

  // Output signals
  assign dst_data_o = dst_hold_data;
  assign dst_valid_o = dst_valid_reg;

  // Synchronizers
  // src_req: src -> dst
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni)
      src_req_sync <= 2'b00;
    else
      src_req_sync <= {src_req_sync[0], src_req_reg};
  end

  // dst_ack: dst -> src (dst_req_reg is the ack signal)
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni)
      dst_ack_sync <= 2'b00;
    else
      dst_ack_sync <= {dst_ack_sync[0], dst_req_reg};
  end

  // dst_req: dst -> src (for decoupled mode, but we don't use it in basic mode)
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni)
      dst_req_sync <= 2'b00;
    else
      dst_req_sync <= {dst_req_sync[0], dst_req_reg};
  end

  // src_ack: src -> dst (not used in basic mode)
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni)
      src_ack_sync <= 2'b00;
    else
      src_ack_sync <= {src_ack_sync[0], src_req_reg};
  end

endmodule
