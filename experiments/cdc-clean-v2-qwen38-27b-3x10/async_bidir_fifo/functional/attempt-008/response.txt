```verilog
module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;
  localparam PTR_W = ASIZE + 1;
  localparam AFULL_THRESH  = DEPTH - 2;
  localparam AEMPTY_THRESH = 1;

  // Internal FIFO storage
  reg [DSIZE-1:0] mem [0:DEPTH-1];

  // Side A: write pointer (when a_dir=1), read pointer (when a_dir=0)
  reg [PTR_W-1:0] a_wr_ptr;
  reg [PTR_W-1:0] a_rd_ptr;

  // Side B: write pointer (when b_dir=1), read pointer (when b_dir=0)
  reg [PTR_W-1:0] b_wr_ptr;
  reg [PTR_W-1:0] b_rd_ptr;

  // Cross-domain pointer synchronization
  // A's write pointer synchronized to B domain
  reg [PTR_W-1:0] a_wr_ptr_sync1, a_wr_ptr_sync2, a_wr_ptr_sync3;
  // A's read pointer synchronized to B domain
  reg [PTR_W-1:0] a_rd_ptr_sync1, a_rd_ptr_sync2, a_rd_ptr_sync3;
  // B's write pointer synchronized to A domain
  reg [PTR_W-1:0] b_wr_ptr_sync1, b_wr_ptr_sync2, b_wr_ptr_sync3;
  // B's read pointer synchronized to A domain
  reg [PTR_W-1:0] b_rd_ptr_sync1, b_rd_ptr_sync2, b_rd_ptr_sync3;

  // Synchronization logic
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_wr_ptr_sync1 <= {PTR_W{1'b0}};
      a_wr_ptr_sync2 <= {PTR_W{1'b0}};
      a_wr_ptr_sync3 <= {PTR_W{1'b0}};
      a_rd_ptr_sync1 <= {PTR_W{1'b0}};
      a_rd_ptr_sync2 <= {PTR_W{1'b0}};
      a_rd_ptr_sync3 <= {PTR_W{1'b0}};
    end else begin
      a_wr_ptr_sync1 <= a_wr_ptr;
      a_wr_ptr_sync2 <= a_wr_ptr_sync1;
      a_wr_ptr_sync3 <= a_wr_ptr_sync2;
      a_rd_ptr_sync1 <= a_rd_ptr;
      a_rd_ptr_sync2 <= a_rd_ptr_sync1;
      a_rd_ptr_sync3 <= a_rd_ptr_sync2;
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_wr_ptr_sync1 <= {PTR_W{1'b0}};
      b_wr_ptr_sync2 <= {PTR_W{1'b0}};
      b_wr_ptr_sync3 <= {PTR_W{1'b0}};
      b_rd_ptr_sync1 <= {PTR_W{1'b0}};
      b_rd_ptr_sync2 <= {PTR_W{1'b0}};
      b_rd_ptr_sync3 <= {PTR_W{1'b0}};
    end else begin
      b_wr_ptr_sync1 <= b_wr_ptr;
      b_wr_ptr_sync2 <= b_wr_ptr_sync1;
      b_wr_ptr_sync3 <= b_wr_ptr_sync2;
      b_rd_ptr_sync1 <= b_rd_ptr;
      b_rd_ptr_sync2 <= b_rd_ptr_sync1;
      b_rd_ptr_sync3 <= b_rd_ptr_sync2;
    end
  end

  // Side A: write to FIFO when a_dir=1 and a_winc
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= {PTR_W{1'b0}};
    end else if (a_dir && a_winc) begin
      a_wr_ptr <= a_wr_ptr + 1'b1;
      mem[a_wr_ptr[ASIZE-1:0]] <= a_wdata;
    end
  end

  // Side A: read from FIFO when a_dir=0 and a_rinc
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_rd_ptr <= {PTR_W{1'b0}};
    end else if (!a_dir && a_rinc) begin
      a_rd_ptr <= a_rd_ptr + 1'b1;
    end
  end

  // Side B: write to FIFO when b_dir=1 and b_winc
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= {PTR_W{1'b0}};
    end else if (b_dir && b_winc) begin
      b_wr_ptr <= b_wr_ptr + 1'b1;
      mem[b_wr_ptr[ASIZE-1:0]] <= b_wdata;
    end
  end

  // Side B: read from FIFO when b_dir=0 and b_rinc
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_rd_ptr <= {PTR_W{1'b0}};
    end else if (!b_dir && b_rinc) begin
      b_rd_ptr <= b_rd_ptr + 1'b1;
    end
  end

  // Side A: read data
  // When a_dir=0, a is the reader. Read data comes from the FIFO.
  // The read pointer a_rd_ptr points to the next word to read.
  // We need to handle the case where writes come from B side.
  // The FIFO is shared: writes from A go to mem, writes from B go to mem.
  // Reads from A read from mem, reads from B read from mem.
  // The pointers track the global write and read positions.
  
  // Actually, let me reconsider the architecture.
  // This is a bidirectional FIFO. Both sides can write and read.
  // The FIFO has a single memory.
  // When a_dir=1, A writes to the FIFO. When a_dir=0, A reads from the FIFO.
  // When b_dir=1, B writes to the FIFO. When b_dir=0, B reads from the FIFO.
  
  // The write pointer is the global write position.
  // The read pointer is the global read position.
  // But each side has its own view.
  
  // Let me think about this differently.
  // The FIFO has a head (write) pointer and a tail (read) pointer.
  // When A writes, it increments the head pointer.
  // When B writes, it increments the head pointer.
  // When A reads, it increments the tail pointer.
  // When B reads, it increments the tail pointer.
  
  // But the head and tail pointers need to be shared across domains.
  // This is complex. Let me use a simpler approach.
  
  // Actually, the problem says "bidirectional asynchronous FIFO".
  // I think the intent is that data can flow in either direction.
  // When a_dir=1, A is writing to the FIFO, and B will read from it.
  // When a_dir=0, A is reading from the FIFO, and B will write to it.
  // Similarly for B.
  
  // So the FIFO is used in one direction at a time? Or can both sides write simultaneously?
  // The problem says "a_dir / b_dir select the direction on that side".
  // I think both sides can independently be in write or read mode.
  // But if both are writing, that's a conflict. If both are reading, that's also a conflict.
  
  // Let me re-read: "Accepted writes must appear in order on the opposite side when that side reads."
  // This suggests that when A writes, B reads, and vice versa.
  // So the FIFO is used in a ping-pong fashion.
  
  // I'll assume that at any given time, one side is writing and the other is reading.
  // The write pointer is maintained by the writing side.
  // The read pointer is maintained by the reading side.
  
  // Let me redefine:
  // wr_ptr: global write pointer, incremented by whichever side is writing.
  // rd_ptr: global read pointer, incremented by whichever side is reading.
  
  // But these pointers need to be synchronized across domains.
  
  // Let me use a different approach. I'll have two separate FIFOs:
  // One for A-to-B transfers, one for B-to-A transfers.
  // When a_dir=1, A writes to the A-to-B FIFO.
  // When b_dir=0, B reads from the A-to-B FIFO.
  // When b_dir=1, B writes to the B-to-A FIFO.
  // When a_dir=0, A reads from the B-to-A FIFO.
  
  // This makes more sense for a bidirectional FIFO.

  // Let me restart with this approach.

  // A-to-B FIFO
  reg [DSIZE-1:0] ab_mem [0:DEPTH-1];
  reg [PTR_W-1:0] ab_wr_ptr;  // Write pointer for A-to-B FIFO (in A domain)
  reg [PTR_W-1:0] ab_rd_ptr;  // Read pointer for A-to-B FIFO (in B domain)

  // B-to-A FIFO
  reg [DSIZE-1:0] ba_mem [0:DEPTH-1];
  reg [PTR_W-1:0] ba_wr_ptr;  // Write pointer for B-to-A FIFO (in B domain)
  reg [PTR_W-1:0] ba_rd_ptr;  // Read pointer for B-to-A FIFO (in A domain)

  // Synchronize A-to-B write pointer to B domain
  reg [PTR_W-1:0] ab_wr_ptr_sync1, ab_wr_ptr_sync2, ab_wr_ptr_sync3;
  // Synchronize A-to-B read pointer to A domain
  reg [PTR_W-1:0] ab_rd_ptr_sync1, ab_rd_ptr_sync2, ab_rd_ptr_sync3;
  // Synchronize B-to-A write pointer to A domain
  reg [PTR_W-1:0] ba_wr_ptr_sync1, ba_wr_ptr_sync2, ba_wr_ptr_sync3;
  // Synchronize B-to-A read pointer to B domain
  reg [PTR_W-1:0] ba_rd_ptr_sync1, ba_rd_ptr_sync2, ba_rd_ptr_sync3;

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ab_wr_ptr_sync1 <= {PTR_W{1'b0}};
      ab_wr_ptr_sync2 <= {PTR_W{1'b0}};
      ab_wr_ptr_sync3 <= {PTR_W{1'b0}};
      ba_rd_ptr_sync1 <= {PTR_W{1'b0}};
      ba_rd_ptr_sync2 <= {PTR_W{1'b0}};
      ba_rd_ptr_sync3 <= {PTR_W{1'b0}};
    end else begin
      ab_wr_ptr_sync1 <= ab_wr_ptr;
      ab_wr_ptr_sync2 <= ab_wr_ptr_sync1;
      ab_wr_ptr_sync3 <= ab_wr_ptr_sync2;
      ba_rd_ptr_sync1 <= ba_rd_ptr;
      ba_rd_ptr_sync2 <= ba_rd_ptr_sync1;
      ba_rd_ptr_sync3 <= ba_rd_ptr_sync2;
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ab_rd_ptr_sync1 <= {PTR_W{1'b0}};
      ab_rd_ptr_sync2 <= {PTR_W{1'b0}};
      ab_rd_ptr_sync3 <= {PTR_W{1'b0}};
      ba_wr_ptr_sync1 <= {PTR_W{1'b0}};
      ba_wr_ptr_sync2 <= {PTR_W{1'b0}};
      ba_wr_ptr_sync3 <= {PTR_W{1'b0}};
    end else begin
      ab_rd_ptr_sync1 <= ab_rd_ptr;
      ab_rd_ptr_sync2 <= ab_rd_ptr_sync1;
      ab_rd_ptr_sync3 <= ab_rd_ptr_sync2;
      ba_wr_ptr_sync1 <= ba_wr_ptr;
      ba_wr_ptr_sync2 <= ba_wr_ptr_sync1;
      ba_wr_ptr_sync3 <= ba_wr_ptr_sync2;
    end
  end

  // A-to-B FIFO: Write in A domain when a_dir=1
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ab_wr_ptr <= {PTR_W{1'b0}};
    end else if (a_dir && a_winc) begin
      ab_wr_ptr <= ab_wr_ptr + 1'b1;
      ab_mem[ab_wr_ptr[ASIZE-1:0]] <= a_wdata;
    end
  end

  // A-to-B FIFO: Read in B domain when b_dir=0
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ab_rd_ptr <= {PTR_W{1'b0}};
    end else if (!b_dir && b_rinc) begin
      ab_rd_ptr <= ab_rd_ptr + 1'b1;
    end
  end

  // B-to-A FIFO: Write in B domain when b_dir=1
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ba_wr_ptr <= {PTR_W{1'b0}};
    end else if (b_dir && b_winc) begin
      ba_wr_ptr <= ba_wr_ptr + 1'b1;
      ba_mem[ba_wr_ptr[ASIZE-1:0]] <= b_wdata;
    end
  end

  // B-to-A FIFO: Read in A domain when a_dir=0
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ba_rd_ptr <= {PTR_W{1'b0}};
    end else if (!a_dir && a_rinc) begin
      ba_rd_ptr <= ba_rd_ptr + 1'b1;
    end
  end

  // Side A: Read data
  // When a_dir=0, A reads from B-to-A FIFO
  wire [DSIZE-1:0] ba_rd_data = ba_mem[ba_rd_ptr[ASIZE-1:0]];
  assign a_rdata = ba_rd_data;

  // Side B: Read data
  // When b_dir=0, B reads from A-to-B FIFO
  wire [DSIZE-1:0] ab_rd_data = ab_mem[ab_rd_ptr[ASIZE-1:0]];
  assign b_rdata = ab_rd_data;

  // Side A: Full/Empty flags
  // A is full when A-to-B FIFO is full (A is writing)
  // A is empty when B-to-A FIFO is empty (A is reading)
  
  // A-to-B FIFO full: ab_wr_ptr - ab_rd_ptr_sync3 == DEPTH
  wire [PTR_W-1:0] ab_count = ab_wr_ptr - ab_rd_ptr_sync3;
  wire a_full_ab = (ab_count == DEPTH);
  wire a_afull_ab = (ab_count >= AFULL_THRESH);
  
  // B-to-A FIFO empty: ba_wr_ptr_sync3 - ba_rd_ptr == 0
  wire [PTR_W-1:0] ba_count = ba_wr_ptr_sync3 - ba_rd_ptr;
  wire a_empty_ba = (ba_count == 0);
  wire a_aempty_ba = (ba_count <= AEMPTY_THRESH);

  assign a_full   = a_full_ab;
  assign a_afull  = a_afull_ab;
  assign a_empty  = a_empty_ba;
  assign a_aempty = a_aempty_ba;

  // Side B: Full/Empty flags
  // B is full when B-to-A FIFO is full (B is writing)
  // B is empty when A-to-B FIFO is empty (B is reading)
  
  // B-to-A FIFO full: ba_wr_ptr - ba_rd_ptr_sync3 == DEPTH
  wire [PTR_W-1:0] ba_count_b = ba_wr_ptr - ba_rd_ptr_sync3;
  wire b_full_ba = (ba_count_b == DEPTH);
  wire b_afull_ba = (ba_count_b >= AFULL_THRESH);
  
  // A-to-B FIFO empty: ab_wr_ptr_sync3 - ab_rd_ptr == 0
  wire [PTR_W-1:0] ab_count_b = ab_wr_ptr_sync3 - ab_rd_ptr;
  wire b_empty_ab = (ab_count_b == 0);
  wire b_aempty_ab = (ab_count_b <= AEMPTY_THRESH);

  assign b_full   = b_full_ba;
  assign b_afull  = b_afull_ba;
  assign b_empty  = b_empty_ab;
  assign b_aempty = b_aempty_ab;

endmodule
