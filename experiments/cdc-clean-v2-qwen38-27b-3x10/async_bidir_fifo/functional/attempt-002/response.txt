```verilog
module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;
  localparam PTR_W = ASIZE + 1;
  localparam AFULL_THRESH  = DEPTH - 1;
  localparam AEMPTY_THRESH = 1;

  // Internal FIFO storage
  reg [DSIZE-1:0] mem [0:DEPTH-1];

  // Side A pointers (A is writer when a_dir=1, reader when a_dir=0)
  // We maintain two independent FIFOs: A->B and B->A
  // But the problem says "bidirectional" with direction select per side.
  // Re-reading: a_dir=1 means A writes, a_dir=0 means A reads.
  // So A can be writing to B or reading from B, but not both simultaneously.
  // Similarly for B.
  // This is a bidirectional FIFO where each side can be in write or read mode.
  // When A writes and B reads, data flows A->B.
  // When B writes and A reads, data flows B->A.
  // We need two separate FIFOs: one for A->B, one for B->A.

  // FIFO A->B: A writes, B reads
  reg [PTR_W-1:0] ab_wr_ptr;
  reg [PTR_W-1:0] ab_rd_ptr;
  reg [PTR_W-1:0] ab_wr_ptr_sync1, ab_wr_ptr_sync2;
  reg [PTR_W-1:0] ab_rd_ptr_sync1, ab_rd_ptr_sync2;

  // FIFO B->A: B writes, A reads
  reg [PTR_W-1:0] ba_wr_ptr;
  reg [PTR_W-1:0] ba_rd_ptr;
  reg [PTR_W-1:0] ba_wr_ptr_sync1, ba_wr_ptr_sync2;
  reg [PTR_W-1:0] ba_rd_ptr_sync1, ba_rd_ptr_sync2;

  // Storage for A->B FIFO
  reg [DSIZE-1:0] ab_mem [0:DEPTH-1];
  // Storage for B->A FIFO
  reg [DSIZE-1:0] ba_mem [0:DEPTH-1];

  // Synchronizers for A->B write pointer (to B side)
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ab_wr_ptr_sync1 <= {PTR_W{1'b0}};
      ab_wr_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      ab_wr_ptr_sync1 <= ab_wr_ptr;
      ab_wr_ptr_sync2 <= ab_wr_ptr_sync1;
    end
  end

  // Synchronizers for A->B read pointer (to A side)
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ab_rd_ptr_sync1 <= {PTR_W{1'b0}};
      ab_rd_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      ab_rd_ptr_sync1 <= ab_rd_ptr;
      ab_rd_ptr_sync2 <= ab_rd_ptr_sync1;
    end
  end

  // Synchronizers for B->A write pointer (to A side)
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ba_wr_ptr_sync1 <= {PTR_W{1'b0}};
      ba_wr_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      ba_wr_ptr_sync1 <= ba_wr_ptr;
      ba_wr_ptr_sync2 <= ba_wr_ptr_sync1;
    end
  end

  // Synchronizers for B->A read pointer (to B side)
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ba_rd_ptr_sync1 <= {PTR_W{1'b0}};
      ba_rd_ptr_sync2 <= {PTR_W{1'b0}};
    end else begin
      ba_rd_ptr_sync1 <= ba_rd_ptr;
      ba_rd_ptr_sync2 <= ba_rd_ptr_sync1;
    end
  end

  // A->B FIFO: A writes when a_dir=1 and a_winc=1
  wire ab_write_en = a_dir & a_winc;
  wire ab_read_en  = !b_dir & b_rinc;

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ab_wr_ptr <= {PTR_W{1'b0}};
    end else if (ab_write_en) begin
      ab_mem[ab_wr_ptr[ASIZE-1:0]] <= a_wdata;
      ab_wr_ptr <= ab_wr_ptr + 1'b1;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ab_rd_ptr <= {PTR_W{1'b0}};
    end else if (ab_read_en) begin
      ab_rd_ptr <= ab_rd_ptr + 1'b1;
    end
  end

  // B->A FIFO: B writes when b_dir=1 and b_winc=1
  wire ba_write_en = b_dir & b_winc;
  wire ba_read_en  = !a_dir & a_rinc;

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ba_wr_ptr <= {PTR_W{1'b0}};
    end else if (ba_write_en) begin
      ba_mem[ba_wr_ptr[ASIZE-1:0]] <= b_wdata;
      ba_wr_ptr <= ba_wr_ptr + 1'b1;
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ba_rd_ptr <= {PTR_W{1'b0}};
    end else if (ba_read_en) begin
      ba_rd_ptr <= ba_rd_ptr + 1'b1;
    end
  end

  // A->B FIFO status on A side (A is writer)
  wire [PTR_W-1:0] ab_wr_ptr_a = ab_wr_ptr;
  wire [PTR_W-1:0] ab_rd_ptr_a = ab_rd_ptr_sync2;
  wire [PTR_W-1:0] ab_count_a  = ab_wr_ptr_a - ab_rd_ptr_a;
  wire ab_full_a  = (ab_count_a == DEPTH);
  wire ab_afull_a = (ab_count_a >= AFULL_THRESH);
  wire ab_empty_a = (ab_count_a == 0);
  wire ab_aempty_a = (ab_count_a <= AEMPTY_THRESH);

  // A->B FIFO status on B side (B is reader)
  wire [PTR_W-1:0] ab_wr_ptr_b = ab_wr_ptr_sync2;
  wire [PTR_W-1:0] ab_rd_ptr_b = ab_rd_ptr;
  wire [PTR_W-1:0] ab_count_b  = ab_wr_ptr_b - ab_rd_ptr_b;
  wire ab_full_b  = (ab_count_b == DEPTH);
  wire ab_afull_b = (ab_count_b >= AFULL_THRESH);
  wire ab_empty_b = (ab_count_b == 0);
  wire ab_aempty_b = (ab_count_b <= AEMPTY_THRESH);

  // B->A FIFO status on B side (B is writer)
  wire [PTR_W-1:0] ba_wr_ptr_b = ba_wr_ptr;
  wire [PTR_W-1:0] ba_rd_ptr_b = ba_rd_ptr_sync2;
  wire [PTR_W-1:0] ba_count_b  = ba_wr_ptr_b - ba_rd_ptr_b;
  wire ba_full_b  = (ba_count_b == DEPTH);
  wire ba_afull_b = (ba_count_b >= AFULL_THRESH);
  wire ba_empty_b = (ba_count_b == 0);
  wire ba_aempty_b = (ba_count_b <= AEMPTY_THRESH);

  // B->A FIFO status on A side (A is reader)
  wire [PTR_W-1:0] ba_wr_ptr_a = ba_wr_ptr_sync2;
  wire [PTR_W-1:0] ba_rd_ptr_a = ba_rd_ptr;
  wire [PTR_W-1:0] ba_count_a  = ba_wr_ptr_a - ba_rd_ptr_a;
  wire ba_full_a  = (ba_count_a == DEPTH);
  wire ba_afull_a = (ba_count_a >= AFULL_THRESH);
  wire ba_empty_a = (ba_count_a == 0);
  wire ba_aempty_a = (ba_count_a <= AEMPTY_THRESH);

  // A side outputs:
  // When a_dir=1 (A is writer): full/afull refer to A->B FIFO
  // When a_dir=0 (A is reader): empty/aempty refer to B->A FIFO
  assign a_full   = a_dir ? ab_full_a  : ba_full_a;
  assign a_afull  = a_dir ? ab_afull_a : ba_afull_a;
  assign a_empty  = a_dir ? ab_empty_a : ba_empty_a;
  assign a_aempty = a_dir ? ab_aempty_a : ba_aempty_a;

  // A read data: when a_dir=0, read from B->A FIFO
  reg [DSIZE-1:0] a_rdata_reg;
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_rdata_reg <= {DSIZE{1'b0}};
    end else if (ba_read_en) begin
      a_rdata_reg <= ba_mem[ba_rd_ptr[ASIZE-1:0]];
    end
  end
  assign a_rdata = a_rdata_reg;

  // B side outputs:
  // When b_dir=1 (B is writer): full/afull refer to B->A FIFO
  // When b_dir=0 (B is reader): empty/aempty refer to A->B FIFO
  assign b_full   = b_dir ? ba_full_b  : ab_full_b;
  assign b_afull  = b_dir ? ba_afull_b : ab_afull_b;
  assign b_empty  = b_dir ? ba_empty_b : ab_empty_b;
  assign b_aempty = b_dir ? ba_aempty_b : ab_aempty_b;

  // B read data: when b_dir=0, read from A->B FIFO
  reg [DSIZE-1:0] b_rdata_reg;
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_rdata_reg <= {DSIZE{1'b0}};
    end else if (ab_read_en) begin
      b_rdata_reg <= ab_mem[ab_rd_ptr[ASIZE-1:0]];
    end
  end
  assign b_rdata = b_rdata_reg;

endmodule
