```verilog
module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;
  localparam AWIDTH = ASIZE + 1;
  localparam AFULL_THRESH = DEPTH - 2;
  localparam AEMPTY_THRESH = 2;

  // Internal FIFO storage for A->B direction
  reg [DSIZE-1:0] fifo_ab [0:DEPTH-1];
  // Internal FIFO storage for B->A direction
  reg [DSIZE-1:0] fifo_ba [0:DEPTH-1];

  // Pointers for A->B FIFO
  // Write pointer in A domain, read pointer in B domain
  reg [AWIDTH-1:0] ab_wr_ptr_a;
  reg [AWIDTH-1:0] ab_rd_ptr_b;
  // Synchronized versions
  reg [AWIDTH-1:0] ab_wr_ptr_b_sync0, ab_wr_ptr_b_sync1, ab_wr_ptr_b_sync2;
  reg [AWIDTH-1:0] ab_rd_ptr_a_sync0, ab_rd_ptr_a_sync1, ab_rd_ptr_a_sync2;

  // Pointers for B->A FIFO
  // Write pointer in B domain, read pointer in A domain
  reg [AWIDTH-1:0] ba_wr_ptr_b;
  reg [AWIDTH-1:0] ba_rd_ptr_a;
  // Synchronized versions
  reg [AWIDTH-1:0] ba_wr_ptr_a_sync0, ba_wr_ptr_a_sync1, ba_wr_ptr_a_sync2;
  reg [AWIDTH-1:0] ba_rd_ptr_b_sync0, ba_rd_ptr_b_sync1, ba_rd_ptr_b_sync2;

  // Data staging for A->B: two destination-clock stages
  reg [DSIZE-1:0] ab_data_stage0, ab_data_stage1;
  reg ab_valid_stage0, ab_valid_stage1;

  // Data staging for B->A: two destination-clock stages
  reg [DSIZE-1:0] ba_data_stage0, ba_data_stage1;
  reg ba_valid_stage0, ba_valid_stage1;

  // Direction change tracking
  // For A side: a_dir=1 means write to A->B FIFO, a_dir=0 means read from B->A FIFO
  // For B side: b_dir=1 means write to B->A FIFO, b_dir=0 means read from A->B FIFO

  // A side: write to A->B FIFO when a_dir=1 and a_winc
  wire a_write_ab = a_dir & a_winc;
  // A side: read from B->A FIFO when a_dir=0 and a_rinc
  wire a_read_ba = ~a_dir & a_rinc;

  // B side: write to B->A FIFO when b_dir=1 and b_winc
  wire b_write_ba = b_dir & b_winc;
  // B side: read from A->B FIFO when b_dir=0 and b_rinc
  wire b_read_ab = ~b_dir & b_rinc;

  // Compute full/empty flags
  // A->B FIFO: full when write pointer (A) catches up to read pointer (B)
  wire [AWIDTH-1:0] ab_wr_ptr_b_sync = ab_wr_ptr_b_sync2;
  wire [AWIDTH-1:0] ab_rd_ptr_a_sync = ab_rd_ptr_a_sync2;
  wire [AWIDTH-1:0] ab_diff = ab_wr_ptr_a - ab_rd_ptr_a_sync;
  wire a_full_ab = (ab_diff == 0) && (ab_wr_ptr_a[ASIZE] != ab_rd_ptr_a_sync[ASIZE]);
  wire a_afull_ab = (ab_diff == AFULL_THRESH);

  // B->A FIFO: full when write pointer (B) catches up to read pointer (A)
  wire [AWIDTH-1:0] ba_wr_ptr_a_sync = ba_wr_ptr_a_sync2;
  wire [AWIDTH-1:0] ba_rd_ptr_b_sync = ba_rd_ptr_b_sync2;
  wire [AWIDTH-1:0] ba_diff = ba_wr_ptr_b - ba_rd_ptr_b_sync;
  wire b_full_ba = (ba_diff == 0) && (ba_wr_ptr_b[ASIZE] != ba_rd_ptr_b_sync[ASIZE]);
  wire b_afull_ba = (ba_diff == AFULL_THRESH);

  // Empty flags
  // A->B FIFO empty: read pointer (B) equals write pointer (A)
  wire b_empty_ab = (ab_rd_ptr_b == ab_wr_ptr_b_sync);
  wire b_aempty_ab = (ab_diff == 0) || (ab_diff == 1);

  // B->A FIFO empty: read pointer (A) equals write pointer (B)
  wire a_empty_ba = (ba_rd_ptr_a == ba_wr_ptr_a_sync);
  wire a_aempty_ba = (ba_diff == 0) || (ba_diff == 1);

  // Local full/empty for each side based on direction
  // A side: if a_dir=1 (write), full is a_full_ab; if a_dir=0 (read), empty is a_empty_ba
  assign a_full = a_dir ? a_full_ab : 1'b0;
  assign a_afull = a_dir ? a_afull_ab : 1'b0;
  assign a_empty = ~a_dir ? a_empty_ba : 1'b0;
  assign a_aempty = ~a_dir ? a_aempty_ba : 1'b0;

  // B side: if b_dir=1 (write), full is b_full_ba; if b_dir=0 (read), empty is b_empty_ab
  assign b_full = b_dir ? b_full_ba : 1'b0;
  assign b_afull = b_dir ? b_afull_ba : 1'b0;
  assign b_empty = ~b_dir ? b_empty_ab : 1'b0;
  assign b_aempty = ~b_dir ? b_aempty_ab : 1'b0;

  // A->B FIFO write in A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ab_wr_ptr_a <= {AWIDTH{1'b0}};
      ab_data_stage0 <= {DSIZE{1'b0}};
      ab_valid_stage0 <= 1'b0;
    end else begin
      if (a_write_ab && !a_full_ab) begin
        fifo_ab[ab_wr_ptr_a[ASIZE-1:0]] <= a_wdata;
        ab_wr_ptr_a <= ab_wr_ptr_a + 1'b1;
        ab_data_stage0 <= a_wdata;
        ab_valid_stage0 <= 1'b1;
      end else begin
        ab_valid_stage0 <= 1'b0;
      end
    end
  end

  // A->B FIFO read in B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ab_rd_ptr_b <= {AWIDTH{1'b0}};
      ab_data_stage1 <= {DSIZE{1'b0}};
      ab_valid_stage1 <= 1'b0;
    end else begin
      if (b_read_ab && !b_empty_ab) begin
        ab_rd_ptr_b <= ab_rd_ptr_b + 1'b1;
        ab_data_stage1 <= fifo_ab[ab_rd_ptr_b[ASIZE-1:0]];
        ab_valid_stage1 <= 1'b1;
      end else begin
        ab_valid_stage1 <= 1'b0;
      end
    end
  end

  // B->A FIFO write in B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ba_wr_ptr_b <= {AWIDTH{1'b0}};
      ba_data_stage0 <= {DSIZE{1'b0}};
      ba_valid_stage0 <= 1'b0;
    end else begin
      if (b_write_ba && !b_full_ba) begin
        fifo_ba[ba_wr_ptr_b[ASIZE-1:0]] <= b_wdata;
        ba_wr_ptr_b <= ba_wr_ptr_b + 1'b1;
        ba_data_stage0 <= b_wdata;
        ba_valid_stage0 <= 1'b1;
      end else begin
        ba_valid_stage0 <= 1'b0;
      end
    end
  end

  // B->A FIFO read in A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ba_rd_ptr_a <= {AWIDTH{1'b0}};
      ba_data_stage1 <= {DSIZE{1'b0}};
      ba_valid_stage1 <= 1'b0;
    end else begin
      if (a_read_ba && !a_empty_ba) begin
        ba_rd_ptr_a <= ba_rd_ptr_a + 1'b1;
        ba_data_stage1 <= fifo_ba[ba_rd_ptr_a[ASIZE-1:0]];
        ba_valid_stage1 <= 1'b1;
      end else begin
        ba_valid_stage1 <= 1'b0;
      end
    end
  end

  // Synchronize A->B write pointer to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ab_wr_ptr_b_sync0 <= {AWIDTH{1'b0}};
      ab_wr_ptr_b_sync1 <= {AWIDTH{1'b0}};
      ab_wr_ptr_b_sync2 <= {AWIDTH{1'b0}};
    end else begin
      ab_wr_ptr_b_sync0 <= ab_wr_ptr_a;
      ab_wr_ptr_b_sync1 <= ab_wr_ptr_b_sync0;
      ab_wr_ptr_b_sync2 <= ab_wr_ptr_b_sync1;
    end
  end

  // Synchronize A->B read pointer to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ab_rd_ptr_a_sync0 <= {AWIDTH{1'b0}};
      ab_rd_ptr_a_sync1 <= {AWIDTH{1'b0}};
      ab_rd_ptr_a_sync2 <= {AWIDTH{1'b0}};
    end else begin
      ab_rd_ptr_a_sync0 <= ab_rd_ptr_b;
      ab_rd_ptr_a_sync1 <= ab_rd_ptr_a_sync0;
      ab_rd_ptr_a_sync2 <= ab_rd_ptr_a_sync1;
    end
  end

  // Synchronize B->A write pointer to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ba_wr_ptr_a_sync0 <= {AWIDTH{1'b0}};
      ba_wr_ptr_a_sync1 <= {AWIDTH{1'b0}};
      ba_wr_ptr_a_sync2 <= {AWIDTH{1'b0}};
    end else begin
      ba_wr_ptr_a_sync0 <= ba_wr_ptr_b;
      ba_wr_ptr_a_sync1 <= ba_wr_ptr_a_sync0;
      ba_wr_ptr_a_sync2 <= ba_wr_ptr_a_sync1;
    end
  end

  // Synchronize B->A read pointer to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ba_rd_ptr_b_sync0 <= {AWIDTH{1'b0}};
      ba_rd_ptr_b_sync1 <= {AWIDTH{1'b0}};
      ba_rd_ptr_b_sync2 <= {AWIDTH{1'b0}};
    end else begin
      ba_rd_ptr_b_sync0 <= ba_rd_ptr_a;
      ba_rd_ptr_b_sync1 <= ba_rd_ptr_b_sync0;
      ba_rd_ptr_b_sync2 <= ba_rd_ptr_b_sync1;
    end
  end

  // Read data outputs
  assign a_rdata = ba_data_stage1;
  assign b_rdata = ab_data_stage1;

endmodule
