module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;
  localparam PTR_W = ASIZE + 1;
  localparam AFULL_THRESH  = DEPTH - 2;
  localparam AEMPTY_THRESH = 2;

  // Internal FIFO state for A->B direction
  reg [DSIZE-1:0] fifo_ab [0:DEPTH-1];
  reg [PTR_W-1:0] ab_wr_ptr_a;
  reg [PTR_W-1:0] ab_rd_ptr_b;
  reg [PTR_W-1:0] ab_wr_ptr_b_sync;
  reg [PTR_W-1:0] ab_rd_ptr_a_sync;

  // Internal FIFO state for B->A direction
  reg [DSIZE-1:0] fifo_ba [0:DEPTH-1];
  reg [PTR_W-1:0] ba_wr_ptr_b;
  reg [PTR_W-1:0] ba_rd_ptr_a;
  reg [PTR_W-1:0] ba_wr_ptr_a_sync;
  reg [PTR_W-1:0] ba_rd_ptr_b_sync;

  // Synchronization registers for A->B pointers
  reg [PTR_W-1:0] ab_wr_ptr_b_sync_r1, ab_wr_ptr_b_sync_r2;
  reg [PTR_W-1:0] ab_rd_ptr_a_sync_r1, ab_rd_ptr_a_sync_r2;

  // Synchronization registers for B->A pointers
  reg [PTR_W-1:0] ba_wr_ptr_a_sync_r1, ba_wr_ptr_a_sync_r2;
  reg [PTR_W-1:0] ba_rd_ptr_b_sync_r1, ba_rd_ptr_b_sync_r2;

  // Data synchronization for A->B (two-stage pipeline in B domain)
  reg [DSIZE-1:0] ab_data_b_r1, ab_data_b_r2;
  reg             ab_valid_b_r1, ab_valid_b_r2;

  // Data synchronization for B->A (two-stage pipeline in A domain)
  reg [DSIZE-1:0] ba_data_a_r1, ba_data_a_r2;
  reg             ba_valid_a_r1, ba_valid_a_r2;

  // Write enable signals
  wire ab_wr_en_a;
  wire ab_rd_en_b;
  wire ba_wr_en_b;
  wire ba_rd_en_a;

  // Direction-based enables
  assign ab_wr_en_a = a_dir & a_winc & ~a_full;
  assign ab_rd_en_b = ~b_dir & b_rinc & ~b_empty;
  assign ba_wr_en_b = b_dir & b_winc & ~b_full;
  assign ba_rd_en_a = ~a_dir & a_rinc & ~a_empty;

  // A domain: A->B write
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ab_wr_ptr_a <= {PTR_W{1'b0}};
    end else if (ab_wr_en_a) begin
      ab_wr_ptr_a <= ab_wr_ptr_a + 1'b1;
    end
  end

  // A domain: B->A read
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ba_rd_ptr_a <= {PTR_W{1'b0}};
    end else if (ba_rd_en_a) begin
      ba_rd_ptr_a <= ba_rd_ptr_a + 1'b1;
    end
  end

  // B domain: A->B read
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ab_rd_ptr_b <= {PTR_W{1'b0}};
    end else if (ab_rd_en_b) begin
      ab_rd_ptr_b <= ab_rd_ptr_b + 1'b1;
    end
  end

  // B domain: B->A write
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ba_wr_ptr_b <= {PTR_W{1'b0}};
    end else if (ba_wr_en_b) begin
      ba_wr_ptr_b <= ba_wr_ptr_b + 1'b1;
    end
  end

  // A domain: store data for A->B
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      // No reset needed for data, but we can leave it
    end else if (ab_wr_en_a) begin
      fifo_ab[ab_wr_ptr_a[ASIZE-1:0]] <= a_wdata;
    end
  end

  // B domain: store data for B->A
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      // No reset needed for data
    end else if (ba_wr_en_b) begin
      fifo_ba[ba_wr_ptr_b[ASIZE-1:0]] <= b_wdata;
    end
  end

  // A domain: read data for B->A (output to A side)
  wire [DSIZE-1:0] ba_rd_data_a;
  assign ba_rd_data_a = fifo_ba[ba_rd_ptr_a[ASIZE-1:0]];

  // B domain: read data for A->B (output to B side)
  wire [DSIZE-1:0] ab_rd_data_b;
  assign ab_rd_data_b = fifo_ab[ab_rd_ptr_b[ASIZE-1:0]];

  // Synchronize A->B write pointer to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ab_wr_ptr_b_sync_r1 <= {PTR_W{1'b0}};
      ab_wr_ptr_b_sync_r2 <= {PTR_W{1'b0}};
    end else begin
      ab_wr_ptr_b_sync_r1 <= ab_wr_ptr_a;
      ab_wr_ptr_b_sync_r2 <= ab_wr_ptr_b_sync_r1;
    end
  end
  assign ab_wr_ptr_b_sync = ab_wr_ptr_b_sync_r2;

  // Synchronize A->B read pointer to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ab_rd_ptr_a_sync_r1 <= {PTR_W{1'b0}};
      ab_rd_ptr_a_sync_r2 <= {PTR_W{1'b0}};
    end else begin
      ab_rd_ptr_a_sync_r1 <= ab_rd_ptr_b;
      ab_rd_ptr_a_sync_r2 <= ab_rd_ptr_a_sync_r1;
    end
  end
  assign ab_rd_ptr_a_sync = ab_rd_ptr_a_sync_r2;

  // Synchronize B->A write pointer to A domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ba_wr_ptr_a_sync_r1 <= {PTR_W{1'b0}};
      ba_wr_ptr_a_sync_r2 <= {PTR_W{1'b0}};
    end else begin
      ba_wr_ptr_a_sync_r1 <= ba_wr_ptr_b;
      ba_wr_ptr_a_sync_r2 <= ba_wr_ptr_a_sync_r1;
    end
  end
  assign ba_wr_ptr_a_sync = ba_wr_ptr_a_sync_r2;

  // Synchronize B->A read pointer to B domain
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ba_rd_ptr_b_sync_r1 <= {PTR_W{1'b0}};
      ba_rd_ptr_b_sync_r2 <= {PTR_W{1'b0}};
    end else begin
      ba_rd_ptr_b_sync_r1 <= ba_rd_ptr_a;
      ba_rd_ptr_b_sync_r2 <= ba_rd_ptr_b_sync_r1;
    end
  end
  assign ba_rd_ptr_b_sync = ba_rd_ptr_b_sync_r2;

  // Data pipeline for A->B: capture data in B domain when read
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ab_data_b_r1 <= {DSIZE{1'b0}};
      ab_data_b_r2 <= {DSIZE{1'b0}};
      ab_valid_b_r1 <= 1'b0;
      ab_valid_b_r2 <= 1'b0;
    end else begin
      if (ab_rd_en_b) begin
        ab_data_b_r1 <= ab_rd_data_b;
        ab_valid_b_r1 <= 1'b1;
      end else begin
        ab_valid_b_r1 <= 1'b0;
      end
      ab_data_b_r2 <= ab_data_b_r1;
      ab_valid_b_r2 <= ab_valid_b_r1;
    end
  end

  // Data pipeline for B->A: capture data in A domain when read
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ba_data_a_r1 <= {DSIZE{1'b0}};
      ba_data_a_r2 <= {DSIZE{1'b0}};
      ba_valid_a_r1 <= 1'b0;
      ba_valid_a_r2 <= 1'b0;
    end else begin
      if (ba_rd_en_a) begin
        ba_data_a_r1 <= ba_rd_data_a;
        ba_valid_a_r1 <= 1'b1;
      end else begin
        ba_valid_a_r1 <= 1'b0;
      end
      ba_data_a_r2 <= ba_data_a_r1;
      ba_valid_a_r2 <= ba_valid_a_r1;
    end
  end

  // Output data
  assign a_rdata = ba_data_a_r2;
  assign b_rdata = ab_data_b_r2;

  // Status flags
  wire [PTR_W-1:0] ab_count_a;
  wire [PTR_W-1:0] ab_count_b;
  wire [PTR_W-1:0] ba_count_a;
  wire [PTR_W-1:0] ba_count_b;

  assign ab_count_a = ab_wr_ptr_a - ab_rd_ptr_a_sync;
  assign ab_count_b = ab_wr_ptr_b_sync - ab_rd_ptr_b;
  assign ba_count_a = ba_wr_ptr_a_sync - ba_rd_ptr_a;
  assign ba_count_b = ba_wr_ptr_b - ba_rd_ptr_b_sync;

  // A side: A is writer for A->B, reader for B->A
  assign a_full   = (ab_count_a == DEPTH);
  assign a_afull  = (ab_count_a >= AFULL_THRESH);
  assign a_empty  = (ba_count_a == 0);
  assign a_aempty = (ba_count_a <= AEMPTY_THRESH);

  // B side: B is reader for A->B, writer for B->A
  assign b_full   = (ba_count_b == DEPTH);
  assign b_afull  = (ba_count_b >= AFULL_THRESH);
  assign b_empty  = (ab_count_b == 0);
  assign b_aempty = (ab_count_b <= AEMPTY_THRESH);

endmodule
