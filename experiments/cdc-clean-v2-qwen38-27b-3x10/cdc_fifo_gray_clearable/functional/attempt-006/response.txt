```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;
    localparam GRAY_W = LOG_DEPTH + 1;

    // Source domain
    reg [PTR_W-1:0] src_wr_ptr;
    reg [GRAY_W-1:0] src_wr_ptr_gray;
    reg [GRAY_W-1:0] src_rd_ptr_gray_synced;
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg src_clear_pending;
    reg src_clear_active;

    // Destination domain
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [GRAY_W-1:0] dst_rd_ptr_gray;
    reg [GRAY_W-1:0] dst_wr_ptr_gray_synced;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;
    reg dst_clear_pending;
    reg dst_clear_active;

    // Synchronizer registers for source read pointer gray
    reg [GRAY_W-1:0] src_rd_ptr_gray_sync [0:SYNC_STAGES-1];

    // Synchronizer registers for destination write pointer gray
    reg [GRAY_W-1:0] dst_wr_ptr_gray_sync [0:SYNC_STAGES-1];

    // Clear synchronizers: src_clear to dst, dst_clear to src
    reg [SYNC_STAGES-1:0] src_clear_sync;
    reg [SYNC_STAGES-1:0] dst_clear_sync;

    // Convert binary to gray
    function [GRAY_W-1:0] bin2gray;
        input [PTR_W-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Convert gray to binary
    function [PTR_W-1:0] gray2bin;
        input [GRAY_W-1:0] gray;
        integer i;
        begin
            gray2bin[GRAY_W-1] = gray[GRAY_W-1];
            for (i = GRAY_W-2; i >= 0; i = i - 1)
                gray2bin[i] = gray2bin[i+1] ^ gray[i];
        end
    endfunction

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
            src_wr_ptr_gray <= {GRAY_W{1'b0}};
            src_clear_pending <= 1'b0;
            src_clear_active <= 1'b0;
        end else begin
            if (src_clear_i) begin
                src_clear_active <= 1'b1;
                src_clear_pending <= 1'b1;
            end

            if (src_clear_active) begin
                // Wait for dst to acknowledge clear
                if (dst_clear_sync[SYNC_STAGES-1]) begin
                    src_clear_active <= 1'b0;
                    src_clear_pending <= 1'b0;
                    src_wr_ptr <= {PTR_W{1'b0}};
                    src_wr_ptr_gray <= {GRAY_W{1'b0}};
                end
            end else begin
                if (src_valid_i && src_ready_o) begin
                    mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
                    src_wr_ptr <= src_wr_ptr + 1'b1;
                    src_wr_ptr_gray <= bin2gray(src_wr_ptr + 1'b1);
                end
            end
        end
    end

    // Source ready: FIFO not full and not clearing
    wire src_full = (src_wr_ptr_gray == src_rd_ptr_gray_synced) && 
                    (src_wr_ptr[LOG_DEPTH] != src_rd_ptr_gray_synced[LOG_DEPTH]);
    // Actually, full condition for gray pointers:
    // Full when gray write pointer equals gray read pointer and MSB differs
    // Let me recalculate:
    // For a FIFO of depth 2^LOG_DEPTH, with pointers of width LOG_DEPTH+1:
    // Empty: gray_wr == gray_rd
    // Full: gray_wr == gray_rd ^ (1 << LOG_DEPTH) i.e., MSB differs, rest same

    wire src_empty = (src_wr_ptr_gray == src_rd_ptr_gray_synced);
    wire src_full = (src_wr_ptr_gray == (src_rd_ptr_gray_synced ^ (1 << LOG_DEPTH)));

    assign src_ready_o = !src_full && !src_clear_active;
    assign src_clear_pending_o = src_clear_pending;

    // Synchronize src_rd_ptr_gray to source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rd_ptr_gray_sync[0] <= {GRAY_W{1'b0}};
            src_rd_ptr_gray_sync[1] <= {GRAY_W{1'b0}};
            src_rd_ptr_gray_sync[2] <= {GRAY_W{1'b0}};
        end else begin
            src_rd_ptr_gray_sync[0] <= dst_rd_ptr_gray;
            src_rd_ptr_gray_sync[1] <= src_rd_ptr_gray_sync[0];
            src_rd_ptr_gray_sync[2] <= src_rd_ptr_gray_sync[1];
        end
    end

    assign src_rd_ptr_gray_synced = src_rd_ptr_gray_sync[SYNC_STAGES-1];

    // Synchronize src_clear to destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            src_clear_sync[0] <= src_clear_i;
            src_clear_sync[1] <= src_clear_sync[0];
            src_clear_sync[2] <= src_clear_sync[1];
        end
    end

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_rd_ptr_gray <= {GRAY_W{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
            dst_clear_pending <= 1'b0;
            dst_clear_active <= 1'b0;
        end else begin
            if (dst_clear_i) begin
                dst_clear_active <= 1'b1;
                dst_clear_pending <= 1'b1;
            end

            if (dst_clear_active) begin
                // Wait for src to acknowledge clear
                if (dst_clear_sync[SYNC_STAGES-1]) begin
                    dst_clear_active <= 1'b0;
                    dst_clear_pending <= 1'b0;
                    dst_rd_ptr <= {PTR_W{1'b0}};
                    dst_rd_ptr_gray <= {GRAY_W{1'b0}};
                    dst_data_reg <= {WIDTH{1'b0}};
                    dst_valid_reg <= 1'b0;
                end
            end else begin
                if (dst_valid_reg && dst_ready_i) begin
                    dst_rd_ptr <= dst_rd_ptr + 1'b1;
                    dst_rd_ptr_gray <= bin2gray(dst_rd_ptr + 1'b1);
                    dst_valid_reg <= 1'b0;
                end
            end
        end
    end

    // Destination empty/full check
    wire dst_empty = (dst_rd_ptr_gray == dst_wr_ptr_gray_synced);
    wire dst_full = (dst_rd_ptr_gray == (dst_wr_ptr_gray_synced ^ (1 << LOG_DEPTH)));

    // Read from memory when valid and ready
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_clear_active) begin
                dst_data_reg <= {WIDTH{1'b0}};
                dst_valid_reg <= 1'b0;
            end else begin
                if (!dst_empty && !dst_valid_reg) begin
                    dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                    dst_valid_reg <= 1'b1;
                end
            end
        end
    end

    assign dst_valid_o = dst_valid_reg && !dst_clear_active;
    assign dst_data_o = dst_data_reg;
    assign dst_clear_pending_o = dst_clear_pending;

    // Synchronize dst_rd_ptr_gray to destination domain (for src to see)
    // Actually, dst_rd_ptr_gray is in dst domain, needs to go to src domain
    // That's already handled by src_rd_ptr_gray_sync above

    // Synchronize dst_wr_ptr_gray to destination domain (for dst to see)
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync[0] <= {GRAY_W{1'b0}};
            dst_wr_ptr_gray_sync[1] <= {GRAY_W{1'b0}};
            dst_wr_ptr_gray_sync[2] <= {GRAY_W{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync[0] <= src_wr_ptr_gray;
            dst_wr_ptr_gray_sync[1] <= dst_wr_ptr_gray_sync[0];
            dst_wr_ptr_gray_sync[2] <= dst_wr_ptr_gray_sync[1];
        end
    end

    assign dst_wr_ptr_gray_synced = dst_wr_ptr_gray_sync[SYNC_STAGES-1];

    // Synchronize dst_clear to source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            dst_clear_sync[0] <= dst_clear_i;
            dst_clear_sync[1] <= dst_clear_sync[0];
            dst_clear_sync[2] <= dst_clear_sync[1];
        end
    end

endmodule
