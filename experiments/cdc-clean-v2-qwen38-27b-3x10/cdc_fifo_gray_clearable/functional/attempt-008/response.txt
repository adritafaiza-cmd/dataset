```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Source domain signals
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_W-1:0] src_wr_ptr;
    reg [PTR_W-1:0] src_wr_ptr_gray;
    reg [PTR_W-1:0] dst_rd_ptr_gray_src;
    reg [PTR_W-1:0] dst_rd_ptr_gray_src_sync [0:SYNC_STAGES-1];
    reg src_clear_pending;
    reg src_clear_in_progress;

    // Destination domain signals
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [PTR_W-1:0] dst_rd_ptr_gray;
    reg [PTR_W-1:0] src_wr_ptr_gray_dst;
    reg [PTR_W-1:0] src_wr_ptr_gray_dst_sync [0:SYNC_STAGES-1];
    reg dst_clear_pending;
    reg dst_clear_in_progress;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;

    // Binary to Gray conversion function
    function [PTR_W-1:0] bin2gray;
        input [PTR_W-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Gray to Binary conversion function
    function [PTR_W-1:0] gray2bin;
        input [PTR_W-1:0] gray;
        integer i;
        begin
            gray2bin[PTR_W-1] = gray[PTR_W-1];
            for (i = PTR_W-2; i >= 0; i = i - 1)
                gray2bin[i] = gray[i] ^ gray2bin[i+1];
        end
    endfunction

    // Source domain: write pointer and memory
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
            src_wr_ptr_gray <= {PTR_W{1'b0}};
            src_clear_pending <= 1'b0;
            src_clear_in_progress <= 1'b0;
        end else begin
            if (src_clear_i) begin
                src_clear_in_progress <= 1'b1;
                src_clear_pending <= 1'b1;
            end

            if (src_clear_in_progress) begin
                // Wait for destination to acknowledge clear
                if (dst_clear_pending_sync_src[SYNC_STAGES-1] == 1'b0) begin
                    src_clear_in_progress <= 1'b0;
                    src_clear_pending <= 1'b0;
                    src_wr_ptr <= {PTR_W{1'b0}};
                    src_wr_ptr_gray <= {PTR_W{1'b0}};
                end
            end else begin
                if (src_valid_i && src_ready_o) begin
                    mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
                    src_wr_ptr <= src_wr_ptr + 1'b1;
                    src_wr_ptr_gray <= bin2gray(src_wr_ptr + 1'b1);
                end
            end
        end
    end

    // Source domain: synchronize destination read pointer gray
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_gray_src <= {PTR_W{1'b0}};
            for (integer i = 0; i < SYNC_STAGES; i = i + 1)
                dst_rd_ptr_gray_src_sync[i] <= {PTR_W{1'b0}};
        end else begin
            dst_rd_ptr_gray_src <= dst_rd_ptr_gray;
            for (integer i = 0; i < SYNC_STAGES; i = i + 1) begin
                if (i == 0)
                    dst_rd_ptr_gray_src_sync[i] <= dst_rd_ptr_gray_src;
                else
                    dst_rd_ptr_gray_src_sync[i] <= dst_rd_ptr_gray_src_sync[i-1];
            end
        end
    end

    // Source domain: synchronize destination clear pending
    reg dst_clear_pending_sync_src [0:SYNC_STAGES-1];
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            for (integer i = 0; i < SYNC_STAGES; i = i + 1)
                dst_clear_pending_sync_src[i] <= 1'b0;
        end else begin
            dst_clear_pending_sync_src[0] <= dst_clear_pending;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                dst_clear_pending_sync_src[i] <= dst_clear_pending_sync_src[i-1];
        end
    end

    // Source ready: FIFO not full and not in clear
    wire src_full = (dst_rd_ptr_gray_src_sync[SYNC_STAGES-1] == src_wr_ptr_gray) ? 1'b0 : 
                    (dst_rd_ptr_gray_src_sync[SYNC_STAGES-1][PTR_W-1] != src_wr_ptr_gray[PTR_W-1]) &&
                    (dst_rd_ptr_gray_src_sync[SYNC_STAGES-1][PTR_W-2:0] == src_wr_ptr_gray[PTR_W-2:0]);
    assign src_ready_o = !src_full && !src_clear_in_progress;

    assign src_clear_pending_o = src_clear_pending;

    // Destination domain: read pointer and output
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_rd_ptr_gray <= {PTR_W{1'b0}};
            dst_clear_pending <= 1'b0;
            dst_clear_in_progress <= 1'b0;
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_clear_i) begin
                dst_clear_in_progress <= 1'b1;
                dst_clear_pending <= 1'b1;
            end

            if (dst_clear_in_progress) begin
                // Wait for source to acknowledge clear
                if (src_clear_pending_sync_dst[SYNC_STAGES-1] == 1'b0) begin
                    dst_clear_in_progress <= 1'b0;
                    dst_clear_pending <= 1'b0;
                    dst_rd_ptr <= {PTR_W{1'b0}};
                    dst_rd_ptr_gray <= {PTR_W{1'b0}};
                    dst_valid_reg <= 1'b0;
                end
            end else begin
                if (dst_valid_reg && dst_ready_i) begin
                    dst_rd_ptr <= dst_rd_ptr + 1'b1;
                    dst_rd_ptr_gray <= bin2gray(dst_rd_ptr + 1'b1);
                    dst_valid_reg <= 1'b0;
                end else if (!dst_valid_reg && !dst_empty) begin
                    dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                    dst_valid_reg <= 1'b1;
                end
            end
        end
    end

    // Destination domain: synchronize source write pointer gray
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_gray_dst <= {PTR_W{1'b0}};
            for (integer i = 0; i < SYNC_STAGES; i = i + 1)
                src_wr_ptr_gray_dst_sync[i] <= {PTR_W{1'b0}};
        end else begin
            src_wr_ptr_gray_dst <= src_wr_ptr_gray;
            for (integer i = 0; i < SYNC_STAGES; i = i + 1) begin
                if (i == 0)
                    src_wr_ptr_gray_dst_sync[i] <= src_wr_ptr_gray_dst;
                else
                    src_wr_ptr_gray_dst_sync[i] <= src_wr_ptr_gray_dst_sync[i-1];
            end
        end
    end

    // Destination domain: synchronize source clear pending
    reg src_clear_pending_sync_dst [0:SYNC_STAGES-1];
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            for (integer i = 0; i < SYNC_STAGES; i = i + 1)
                src_clear_pending_sync_dst[i] <= 1'b0;
        end else begin
            src_clear_pending_sync_dst[0] <= src_clear_pending;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                src_clear_pending_sync_dst[i] <= src_clear_pending_sync_dst[i-1];
        end
    end

    // Destination empty: FIFO empty
    wire dst_empty = (src_wr_ptr_gray_dst_sync[SYNC_STAGES-1] == dst_rd_ptr_gray);

    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;
    assign dst_clear_pending_o = dst_clear_pending;

endmodule
