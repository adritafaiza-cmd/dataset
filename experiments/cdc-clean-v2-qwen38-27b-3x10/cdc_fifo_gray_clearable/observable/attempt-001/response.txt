```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;
    localparam GRAY_W = PTR_W;

    // Gray code conversion
    function [GRAY_W-1:0] bin2gray;
        input [PTR_W-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Source domain
    reg [PTR_W-1:0] src_wr_ptr;
    reg [GRAY_W-1:0] src_wr_gray;
    reg [GRAY_W-1:0] src_rd_gray_synced;
    reg [WIDTH-1:0] mem [0:DEPTH-1];

    // Destination domain
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [GRAY_W-1:0] dst_rd_gray;
    reg [GRAY_W-1:0] dst_wr_gray_synced;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;

    // Clear pending signals
    reg src_clear_pending;
    reg dst_clear_pending;

    // Synchronizer for src_wr_gray to dst domain
    reg [SYNC_STAGES-1:0] src_wr_gray_sync;
    // Synchronizer for dst_rd_gray to src domain
    reg [SYNC_STAGES-1:1] dst_rd_gray_sync;

    // Clear synchronizers
    reg [SYNC_STAGES-1:0] src_clear_sync;
    reg [SYNC_STAGES-1:0] dst_clear_sync;

    // Source domain logic
    wire [PTR_W-1:0] src_wr_ptr_next;
    wire src_full;
    wire src_can_write;

    // Destination domain logic
    wire [PTR_W-1:0] dst_rd_ptr_next;
    wire dst_empty;
    wire dst_can_read;

    // Source full: compare src_wr_gray with src_rd_gray_synced
    // Full when the two MSBs differ and the rest are equal
    assign src_full = (src_wr_gray == {~src_rd_gray_synced[GRAY_W-1:GRAY_W-2], src_rd_gray_synced[GRAY_W-3:0]}) & (src_wr_gray[GRAY_W-1:GRAY_W-2] != src_rd_gray_synced[GRAY_W-1:GRAY_W-2]);

    // Destination empty: compare dst_rd_gray with dst_wr_gray_synced
    assign dst_empty = (dst_rd_gray == dst_wr_gray_synced);

    // Clear pending logic
    // src_clear_pending is high when src_clear_i is asserted or when dst_clear has propagated
    // dst_clear_pending is high when dst_clear_i is asserted or when src_clear has propagated

    // Synchronizers
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_gray_sync <= {SYNC_STAGES{1'b0}};
            src_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            src_wr_gray_sync <= {src_wr_gray_sync[SYNC_STAGES-2:0], dst_rd_gray};
            src_clear_sync <= {src_clear_sync[SYNC_STAGES-2:0], dst_clear_sync[SYNC_STAGES-1]};
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_gray_sync <= {SYNC_STAGES{1'b0}};
            dst_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            dst_rd_gray_sync <= {dst_rd_gray_sync[SYNC_STAGES-2:0], src_wr_gray};
            dst_clear_sync <= {dst_clear_sync[SYNC_STAGES-2:0], src_clear_sync[SYNC_STAGES-1]};
        end
    end

    // Clear pending outputs
    assign src_clear_pending_o = src_clear_i | src_clear_sync[SYNC_STAGES-1];
    assign dst_clear_pending_o = dst_clear_i | dst_clear_sync[SYNC_STAGES-1];

    // Source write logic
    assign src_can_write = src_valid_i & ~src_full & ~src_clear_pending_o;
    assign src_ready_o = ~src_full & ~src_clear_pending_o;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
            src_wr_gray <= {GRAY_W{1'b0}};
        end else if (src_clear_i | src_clear_sync[SYNC_STAGES-1]) begin
            src_wr_ptr <= {PTR_W{1'b0}};
            src_wr_gray <= {GRAY_W{1'b0}};
        end else if (src_can_write) begin
            src_wr_ptr <= src_wr_ptr + 1'b1;
            src_wr_gray <= bin2gray(src_wr_ptr + 1'b1);
            mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
        end
    end

    // Destination read logic
    assign dst_can_read = ~dst_empty & dst_ready_i & ~dst_clear_pending_o;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_rd_gray <= {GRAY_W{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else if (dst_clear_i | dst_clear_sync[SYNC_STAGES-1]) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_rd_gray <= {GRAY_W{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_can_read) begin
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_rd_gray <= bin2gray(dst_rd_ptr + 1'b1);
                dst_valid_reg <= 1'b0;
            end
            if (~dst_empty) begin
                dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                dst_valid_reg <= 1'b1;
            end
        end
    end

    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;

endmodule
