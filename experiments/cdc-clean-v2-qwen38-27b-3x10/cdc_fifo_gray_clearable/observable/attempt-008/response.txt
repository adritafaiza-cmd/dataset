```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;
    localparam GRAY_W = LOG_DEPTH + 1;

    // Source domain
    reg [PTR_W-1:0] src_wr_ptr;
    reg [GRAY_W-1:0] src_wr_gray;
    reg [GRAY_W-1:0] src_rd_gray_sync;
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg src_clear_pending;

    // Destination domain
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [GRAY_W-1:0] dst_rd_gray;
    reg [GRAY_W-1:0] dst_wr_gray_sync;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;
    reg dst_clear_pending;

    // Gray code conversion functions
    function [GRAY_W-1:0] bin2gray;
        input [PTR_W-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_W-1:0] gray2bin;
        input [GRAY_W-1:0] gray;
        reg [PTR_W-1:0] bin;
        integer i;
        begin
            bin[PTR_W-1] = gray[PTR_W-1];
            for (i = PTR_W-2; i >= 0; i = i - 1)
                bin[i] = gray[i] ^ bin[i+1];
            gray2bin = bin;
        end
    endfunction

    // Source domain: write pointer and gray
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
            src_wr_gray <= {GRAY_W{1'b0}};
            src_clear_pending <= 1'b0;
        end else begin
            if (src_clear_i) begin
                src_wr_ptr <= {PTR_W{1'b0}};
                src_wr_gray <= {GRAY_W{1'b0}};
                src_clear_pending <= 1'b1;
            end else begin
                if (src_valid_i && src_ready_o) begin
                    src_wr_ptr <= src_wr_ptr + 1'b1;
                    src_wr_gray <= bin2gray(src_wr_ptr + 1'b1);
                end
                if (src_clear_pending && src_rd_gray_sync == {GRAY_W{1'b0}})
                    src_clear_pending <= 1'b0;
            end
        end
    end

    // Source domain: read gray synchronization
    reg [GRAY_W-1:0] src_rd_gray_sync_r [0:SYNC_STAGES-1];
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            for (integer i = 0; i < SYNC_STAGES; i = i + 1)
                src_rd_gray_sync_r[i] <= {GRAY_W{1'b0}};
        end else begin
            src_rd_gray_sync_r[0] <= dst_rd_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                src_rd_gray_sync_r[i] <= src_rd_gray_sync_r[i-1];
        end
    end
    assign src_rd_gray_sync = src_rd_gray_sync_r[SYNC_STAGES-1];

    // Source domain: ready signal
    wire [PTR_W-1:0] src_wr_bin = src_wr_ptr;
    wire [PTR_W-1:0] src_rd_bin = gray2bin(src_rd_gray_sync);
    wire src_full = (src_wr_bin[PTR_W-1] != src_rd_bin[PTR_W-1]) &&
                    (src_wr_bin[PTR_W-2:0] == src_rd_bin[PTR_W-2:0]);
    assign src_ready_o = !src_full && !src_clear_pending;

    // Source domain: memory write
    always @(posedge src_clk_i) begin
        if (src_valid_i && src_ready_o)
            mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
    end

    // Destination domain: read pointer and gray
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_rd_gray <= {GRAY_W{1'b0}};
            dst_clear_pending <= 1'b0;
            dst_valid_reg <= 1'b0;
            dst_data_reg <= {WIDTH{1'b0}};
        end else begin
            if (dst_clear_i) begin
                dst_rd_ptr <= {PTR_W{1'b0}};
                dst_rd_gray <= {GRAY_W{1'b0}};
                dst_clear_pending <= 1'b1;
                dst_valid_reg <= 1'b0;
            end else begin
                if (dst_valid_reg && dst_ready_i) begin
                    dst_rd_ptr <= dst_rd_ptr + 1'b1;
                    dst_rd_gray <= bin2gray(dst_rd_ptr + 1'b1);
                    dst_valid_reg <= 1'b0;
                end
                if (dst_clear_pending && dst_wr_gray_sync == {GRAY_W{1'b0}})
                    dst_clear_pending <= 1'b0;
            end
        end
    end

    // Destination domain: write gray synchronization
    reg [GRAY_W-1:0] dst_wr_gray_sync_r [0:SYNC_STAGES-1];
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            for (integer i = 0; i < SYNC_STAGES; i = i + 1)
                dst_wr_gray_sync_r[i] <= {GRAY_W{1'b0}};
        end else begin
            dst_wr_gray_sync_r[0] <= src_wr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1)
                dst_wr_gray_sync_r[i] <= dst_wr_gray_sync_r[i-1];
        end
    end
    assign dst_wr_gray_sync = dst_wr_gray_sync_r[SYNC_STAGES-1];

    // Destination domain: empty check and data read
    wire [PTR_W-1:0] dst_rd_bin = dst_rd_ptr;
    wire [PTR_W-1:0] dst_wr_bin = gray2bin(dst_wr_gray_sync);
    wire dst_empty = (dst_rd_bin == dst_wr_bin);

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_valid_reg <= 1'b0;
            dst_data_reg <= {WIDTH{1'b0}};
        end else begin
            if (dst_clear_i) begin
                dst_valid_reg <= 1'b0;
            end else if (!dst_valid_reg && !dst_empty && !dst_clear_pending) begin
                dst_valid_reg <= 1'b1;
                dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
            end else if (dst_valid_reg && dst_ready_i) begin
                dst_valid_reg <= 1'b0;
            end
        end
    end

    assign dst_valid_o = dst_valid_reg;
    assign dst_data_o = dst_data_reg;
    assign src_clear_pending_o = src_clear_pending;
    assign dst_clear_pending_o = dst_clear_pending;

endmodule
