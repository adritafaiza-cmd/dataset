module apbxclk #(
    parameter C_APB_ADDR_WIDTH = 12,
    parameter C_APB_DATA_WIDTH = 32,
    parameter [0:0] OPT_REGISTERED = 1'b0,
    localparam AW = C_APB_ADDR_WIDTH,
    localparam DW = C_APB_DATA_WIDTH
) (
    input wire S_APB_PCLK, S_PRESETn,
    input wire S_APB_PSEL,
    input wire S_APB_PENABLE,
    output reg S_APB_PREADY,
    input wire [AW-1:0] S_APB_PADDR,
    input wire S_APB_PWRITE,
    input wire [DW-1:0] S_APB_PWDATA,
    input wire [DW/8-1:0] S_APB_PWSTRB,
    input wire [2:0] S_APB_PPROT,
    output wire [DW-1:0] S_APB_PRDATA,
    output wire S_APB_PSLVERR,
    input wire M_APB_PCLK,
    output reg M_PRESETn,
    output reg M_APB_PSEL,
    output reg M_APB_PENABLE,
    input wire M_APB_PREADY,
    output wire [AW-1:0] M_APB_PADDR,
    output wire M_APB_PWRITE,
    output wire [DW-1:0] M_APB_PWDATA,
    output wire [DW/8-1:0] M_APB_PWSTRB,
    output wire [2:0] M_APB_PPROT,
    input wire [DW-1:0] M_APB_PRDATA,
    input wire M_APB_PSLVERR
);

    // Synchronization stages for request (slave -> master)
    reg [1:0] req_sync;
    reg req_valid_m;
    reg req_pending_m;

    // Synchronization stages for response (master -> slave)
    reg [1:0] rsp_sync;
    reg rsp_valid_s;
    reg rsp_pending_s;

    // Master-side captured request fields
    reg [AW-1:0] m_addr;
    reg m_write;
    reg [DW-1:0] m_wdata;
    reg [DW/8-1:0] m_wstrb;
    reg [2:0] m_pprot;

    // Slave-side captured response fields
    reg [DW-1:0] s_rdata;
    reg s_pslverr;

    // Master-side state machine
    // States: IDLE, SETUP, ACCESS
    localparam [1:0] M_IDLE = 2'b00,
                     M_SETUP = 2'b01,
                     M_ACCESS = 2'b10;
    reg [1:0] m_state;

    // Slave-side state machine
    // States: S_IDLE, S_SETUP, S_ACCESS
    localparam [1:0] S_IDLE = 2'b00,
                     S_SETUP = 2'b01,
                     S_ACCESS = 2'b10;
    reg [1:0] s_state;

    // Master clock domain logic
    always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
        if (!M_PRESETn) begin
            req_sync <= 2'b00;
            req_valid_m <= 1'b0;
            req_pending_m <= 1'b0;
            m_state <= M_IDLE;
            m_addr <= {AW{1'b0}};
            m_write <= 1'b0;
            m_wdata <= {DW{1'b0}};
            m_wstrb <= {(DW/8){1'b0}};
            m_pprot <= 3'b000;
            M_APB_PSEL <= 1'b0;
            M_APB_PENABLE <= 1'b0;
        end else begin
            // Synchronize request valid
            req_sync <= {req_sync[0], S_APB_PSEL & S_APB_PENABLE & S_APB_PREADY};
            
            // Detect rising edge of synchronized request
            if (req_sync[1] && !req_sync[0]) begin
                req_valid_m <= 1'b1;
            end else if (req_valid_m && m_state == M_IDLE) begin
                req_valid_m <= 1'b0;
            end

            // State machine
            case (m_state)
                M_IDLE: begin
                    M_APB_PSEL <= 1'b0;
                    M_APB_PENABLE <= 1'b0;
                    if (req_valid_m) begin
                        m_state <= M_SETUP;
                        M_APB_PSEL <= 1'b1;
                        M_APB_PENABLE <= 1'b0;
                        // Capture request fields
                        m_addr <= S_APB_PADDR;
                        m_write <= S_APB_PWRITE;
                        m_wdata <= S_APB_PWDATA;
                        m_wstrb <= S_APB_PWSTRB;
                        m_pprot <= S_APB_PPROT;
                    end
                end
                M_SETUP: begin
                    M_APB_PSEL <= 1'b1;
                    M_APB_PENABLE <= 1'b1;
                    m_state <= M_ACCESS;
                end
                M_ACCESS: begin
                    M_APB_PSEL <= 1'b1;
                    M_APB_PENABLE <= 1'b1;
                    if (M_APB_PREADY) begin
                        m_state <= M_IDLE;
                        M_APB_PSEL <= 1'b0;
                        M_APB_PENABLE <= 1'b0;
                        // Signal response to slave side
                        rsp_sync <= 2'b00;
                    end
                end
                default: m_state <= M_IDLE;
            endcase
        end
    end

    // Master-side output assignments
    assign M_APB_PADDR = m_addr;
    assign M_APB_PWRITE = m_write;
    assign M_APB_PWDATA = m_wdata;
    assign M_APB_PWSTRB = m_wstrb;
    assign M_APB_PPROT = m_pprot;

    // Synchronize response from master to slave
    // We need to synchronize the completion of the master transfer
    // Use a simple 2-stage synchronizer for the response valid
    reg [1:0] m_done_sync;
    reg m_done;

    always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
        if (!M_PRESETn) begin
            m_done_sync <= 2'b00;
            m_done <= 1'b0;
        end else begin
            m_done_sync <= {m_done_sync[0], (m_state == M_ACCESS) && M_APB_PREADY};
            if (m_done_sync[1] && !m_done_sync[0]) begin
                m_done <= 1'b1;
            end else if (m_done && rsp_sync[1]) begin
                m_done <= 1'b0;
            end
        end
    end

    // Slave clock domain logic
    always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
        if (!S_PRESETn) begin
            rsp_sync <= 2'b00;
            rsp_valid_s <= 1'b0;
            rsp_pending_s <= 1'b0;
            s_state <= S_IDLE;
            s_rdata <= {DW{1'b0}};
            s_pslverr <= 1'b0;
            S_APB_PREADY <= 1'b0;
        end else begin
            // Synchronize response valid from master
            rsp_sync <= {rsp_sync[0], m_done};
            
            // Detect rising edge of synchronized response
            if (rsp_sync[1] && !rsp_sync[0]) begin
                rsp_valid_s <= 1'b1;
            end else if (rsp_valid_s && s_state == S_ACCESS) begin
                rsp_valid_s <= 1'b0;
            end

            // State machine
            case (s_state)
                S_IDLE: begin
                    S_APB_PREADY <= 1'b0;
                    if (S_APB_PSEL && !S_APB_PENABLE) begin
                        s_state <= S_SETUP;
                    end
                end
                S_SETUP: begin
                    S_APB_PREADY <= 1'b0;
                    s_state <= S_ACCESS;
                end
                S_ACCESS: begin
                    if (rsp_valid_s) begin
                        // Capture response
                        s_rdata <= M_APB_PRDATA;
                        s_pslverr <= M_APB_PSLVERR;
                        S_APB_PREADY <= 1'b1;
                        s_state <= S_IDLE;
                    end else begin
                        S_APB_PREADY <= 1'b0;
                    end
                end
                default: begin
                    S_APB_PREADY <= 1'b0;
                    s_state <= S_IDLE;
                end
            endcase
        end
    end

    // Slave-side output assignments
    assign S_APB_PRDATA = s_rdata;
    assign S_APB_PSLVERR = s_pslverr;

endmodule
