module apbxclk #(
	parameter	C_APB_ADDR_WIDTH = 12,
	parameter	C_APB_DATA_WIDTH = 32,
	parameter [0:0]	OPT_REGISTERED = 1'b0,
	localparam	AW = C_APB_ADDR_WIDTH,
	localparam	DW = C_APB_DATA_WIDTH
) (
	input	wire			S_APB_PCLK, S_PRESETn,
	input	wire			S_APB_PSEL,
	input	wire			S_APB_PENABLE,
	output	reg			S_APB_PREADY,
	input	wire	[AW-1:0]	S_APB_PADDR,
	input	wire			S_APB_PWRITE,
	input	wire	[DW-1:0]	S_APB_PWDATA,
	input	wire	[DW/8-1:0]	S_APB_PWSTRB,
	input	wire	[2:0]		S_APB_PPROT,
	output	wire	[DW-1:0]	S_APB_PRDATA,
	output	wire			S_APB_PSLVERR,
	input	wire			M_APB_PCLK,
	output	reg			M_PRESETn,
	output	reg			M_APB_PSEL,
	output	reg			M_APB_PENABLE,
	input	wire			M_APB_PREADY,
	output	wire	[AW-1:0]	M_APB_PADDR,
	output	wire			M_APB_PWRITE,
	output	wire	[DW-1:0]	M_APB_PWDATA,
	output	wire	[DW/8-1:0]	M_APB_PWSTRB,
	output	wire	[2:0]		M_APB_PPROT,
	input	wire	[DW-1:0]	M_APB_PRDATA,
	input	wire			M_APB_PSLVERR
);

	// Slave-side state
	reg [1:0] s_state;
	localparam S_IDLE = 2'b00;
	localparam S_SETUP = 2'b01;
	localparam S_ACCESS = 2'b10;

	// Master-side state
	reg [1:0] m_state;
	localparam M_IDLE = 2'b00;
	localparam M_SETUP = 2'b01;
	localparam M_ACCESS = 2'b10;

	// Request valid from slave to master (synchronized)
	reg s_req_valid;
	reg [AW-1:0] s_req_addr;
	reg s_req_write;
	reg [DW-1:0] s_req_wdata;
	reg [DW/8-1:0] s_req_wstrb;
	reg [2:0] s_req_pprot;

	// Synchronization of request valid
	reg [2:0] req_valid_sync;
	wire req_valid_m = req_valid_sync[2];

	// Response valid from master to slave (synchronized)
	reg m_resp_valid;
	reg [DW-1:0] m_resp_rdata;
	reg m_resp_slverr;

	// Synchronization of response valid
	reg [2:0] resp_valid_sync;
	wire resp_valid_s = resp_valid_sync[2];

	// Synchronization of response data and error
	reg [DW-1:0] resp_rdata_sync;
	reg m_resp_slverr_sync;

	// Slave clock domain: capture request fields
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_state <= S_IDLE;
			S_APB_PREADY <= 1'b0;
			s_req_valid <= 1'b0;
			s_req_addr <= {AW{1'b0}};
			s_req_write <= 1'b0;
			s_req_wdata <= {DW{1'b0}};
			s_req_wstrb <= {(DW/8){1'b0}};
			s_req_pprot <= 3'b0;
		end else begin
			case (s_state)
				S_IDLE: begin
					S_APB_PREADY <= 1'b0;
					if (S_APB_PSEL && !S_APB_PENABLE) begin
						s_state <= S_SETUP;
						s_req_addr <= S_APB_PADDR;
						s_req_write <= S_APB_PWRITE;
						s_req_wdata <= S_APB_PWDATA;
						s_req_wstrb <= S_APB_PWSTRB;
						s_req_pprot <= S_APB_PPROT;
						s_req_valid <= 1'b1;
					end
				end
				S_SETUP: begin
					S_APB_PREADY <= 1'b0;
					if (S_APB_PSEL && S_APB_PENABLE) begin
						s_state <= S_ACCESS;
					end else begin
						s_state <= S_IDLE;
						s_req_valid <= 1'b0;
					end
				end
				S_ACCESS: begin
					if (resp_valid_s) begin
						S_APB_PREADY <= 1'b1;
						s_state <= S_IDLE;
						s_req_valid <= 1'b0;
					end else begin
						S_APB_PREADY <= 1'b0;
					end
				end
				default: begin
					s_state <= S_IDLE;
					S_APB_PREADY <= 1'b0;
					s_req_valid <= 1'b0;
				end
			endcase
		end
	end

	// Synchronize request valid to master clock
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			req_valid_sync <= 3'b0;
		end else begin
			req_valid_sync <= {req_valid_sync[1:0], s_req_valid};
		end
	end

	// Master clock domain: drive APB master
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			M_PRESETn <= 1'b0;
			m_state <= M_IDLE;
			M_APB_PSEL <= 1'b0;
			M_APB_PENABLE <= 1'b0;
			M_APB_PADDR <= {AW{1'b0}};
			M_APB_PWRITE <= 1'b0;
			M_APB_PWDATA <= {DW{1'b0}};
			M_APB_PWSTRB <= {(DW/8){1'b0}};
			M_APB_PPROT <= 3'b0;
			m_resp_valid <= 1'b0;
			m_resp_rdata <= {DW{1'b0}};
			m_resp_slverr <= 1'b0;
		end else begin
			M_PRESETn <= 1'b1;
			case (m_state)
				M_IDLE: begin
					M_APB_PSEL <= 1'b0;
					M_APB_PENABLE <= 1'b0;
					if (req_valid_m) begin
						m_state <= M_SETUP;
						M_APB_PSEL <= 1'b1;
						M_APB_PADDR <= s_req_addr;
						M_APB_PWRITE <= s_req_write;
						M_APB_PWDATA <= s_req_wdata;
						M_APB_PWSTRB <= s_req_wstrb;
						M_APB_PPROT <= s_req_pprot;
					end
				end
				M_SETUP: begin
					M_APB_PSEL <= 1'b1;
					M_APB_PENABLE <= 1'b1;
					m_state <= M_ACCESS;
				end
				M_ACCESS: begin
					M_APB_PSEL <= 1'b1;
					M_APB_PENABLE <= 1'b1;
					if (M_APB_PREADY) begin
						m_state <= M_IDLE;
						M_APB_PSEL <= 1'b0;
						M_APB_PENABLE <= 1'b0;
						m_resp_valid <= 1'b1;
						m_resp_rdata <= M_APB_PRDATA;
						m_resp_slverr <= M_APB_PSLVERR;
					end
				end
				default: begin
					m_state <= M_IDLE;
					M_APB_PSEL <= 1'b0;
					M_APB_PENABLE <= 1'b0;
					m_resp_valid <= 1'b0;
				end
			endcase
		end
	end

	// Synchronize response valid to slave clock
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			resp_valid_sync <= 3'b0;
			resp_rdata_sync <= {DW{1'b0}};
			m_resp_slverr_sync <= 1'b0;
		end else begin
			resp_valid_sync <= {resp_valid_sync[1:0], m_resp_valid};
			resp_rdata_sync <= m_resp_rdata;
			m_resp_slverr_sync <= m_resp_slverr;
		end
	end

	// Slave-side outputs
	assign S_APB_PRDATA = resp_rdata_sync;
	assign S_APB_PSLVERR = m_resp_slverr_sync;

endmodule
