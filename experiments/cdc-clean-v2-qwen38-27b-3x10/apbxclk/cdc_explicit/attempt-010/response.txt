module apbxclk #(
		parameter	C_APB_ADDR_WIDTH = 12,
		parameter	C_APB_DATA_WIDTH = 32,
		parameter [0:0]	OPT_REGISTERED = 1'b0,
		localparam	AW = C_APB_ADDR_WIDTH,
		localparam	DW = C_APB_DATA_WIDTH
	) (
		input	wire			S_APB_PCLK, S_PRESETn,
		input	wire			S_APB_PSEL,
		input	wire			S_APB_PENABLE,
		output	reg			S_APB_PREADY,
		input	wire	[AW-1:0]	S_APB_PADDR,
		input	wire			S_APB_PWRITE,
		input	wire	[DW-1:0]	S_APB_PWDATA,
		input	wire	[DW/8-1:0]	S_APB_PWSTRB,
		input	wire	[2:0]		S_APB_PPROT,
		output	wire	[DW-1:0]	S_APB_PRDATA,
		output	wire			S_APB_PSLVERR,
		input	wire			M_APB_PCLK,
		output	reg			M_PRESETn,
		output	reg			M_APB_PSEL,
		output	reg			M_APB_PENABLE,
		input	wire			M_APB_PREADY,
		output	wire	[AW-1:0]	M_APB_PADDR,
		output	wire			M_APB_PWRITE,
		output	wire	[DW-1:0]	M_APB_PWDATA,
		output	wire	[DW/8-1:0]	M_APB_PWSTRB,
		output	wire	[2:0]		M_APB_PPROT,
		input	wire	[DW-1:0]	M_APB_PRDATA,
		input	wire			M_APB_PSLVERR
	);

	// ------------------------------------------------------------------
	// Reset synchronization for M_APB domain
	// ------------------------------------------------------------------
	reg [1:0] m_reset_sync;
	reg       m_reset_int;

	always @(posedge M_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			m_reset_sync <= 2'b00;
		end else begin
			m_reset_sync <= {m_reset_sync[0], 1'b1};
		end
	end

	assign m_reset_int = ~m_reset_sync[1];
	assign M_PRESETn   = ~m_reset_int;

	// ------------------------------------------------------------------
	// Source-side: capture request on setup phase, hold until ack
	// ------------------------------------------------------------------
	reg       s_req_valid;
	reg [AW-1:0] s_addr;
	reg       s_write;
	reg [DW-1:0] s_wdata;
	reg [DW/8-1:0] s_wstrb;
	reg [2:0]  s_pprot;

	// Synchronization of s_req_valid into M domain
	reg [1:0] req_valid_sync;
	reg       m_req_valid;

	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_req_valid <= 1'b0;
			s_addr      <= {AW{1'b0}};
			s_write     <= 1'b0;
			s_wdata     <= {DW{1'b0}};
			s_wstrb     <= {(DW/8){1'b0}};
			s_pprot     <= 3'b000;
		end else if (S_APB_PSEL && !S_APB_PENABLE && !s_req_valid) begin
			s_req_valid <= 1'b1;
			s_addr      <= S_APB_PADDR;
			s_write     <= S_APB_PWRITE;
			s_wdata     <= S_APB_PWDATA;
			s_wstrb     <= S_APB_PWSTRB;
			s_pprot     <= S_APB_PPROT;
		end else if (S_APB_PSEL && S_APB_PENABLE && S_APB_PREADY) begin
			s_req_valid <= 1'b0;
		end
	end

	// Synchronize request valid to M domain
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			req_valid_sync <= 2'b00;
		end else begin
			req_valid_sync <= {req_valid_sync[0], s_req_valid};
		end
	end
	assign m_req_valid = req_valid_sync[1];

	// ------------------------------------------------------------------
	// Destination-side: APB state machine
	// ------------------------------------------------------------------
	localparam ST_IDLE   = 2'b00;
	localparam ST_SETUP  = 2'b01;
	localparam ST_ACCESS = 2'b10;

	reg [1:0] m_state;
	reg       m_sel;
	reg       m_enable;

	// Latched request fields in M domain (captured when m_req_valid is seen)
	reg [AW-1:0] m_addr;
	reg       m_write;
	reg [DW-1:0] m_wdata;
	reg [DW/8-1:0] m_wstrb;
	reg [2:0]  m_pprot;

	// Synchronization of completion back to S domain
	reg [1:0] done_sync;
	reg       s_done;

	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_state  <= ST_IDLE;
			m_sel    <= 1'b0;
			m_enable <= 1'b0;
			m_addr   <= {AW{1'b0}};
			m_write  <= 1'b0;
			m_wdata  <= {DW{1'b0}};
			m_wstrb  <= {(DW/8){1'b0}};
			m_pprot  <= 3'b000;
		end else begin
			case (m_state)
				ST_IDLE: begin
					m_sel    <= 1'b0;
					m_enable <= 1'b0;
					if (m_req_valid) begin
						m_state  <= ST_SETUP;
						m_sel    <= 1'b1;
						m_enable <= 1'b0;
						m_addr   <= s_addr;
						m_write  <= s_write;
						m_wdata  <= s_wdata;
						m_wstrb  <= s_wstrb;
						m_pprot  <= s_pprot;
					end
				end
				ST_SETUP: begin
					m_sel    <= 1'b1;
					m_enable <= 1'b1;
					m_state  <= ST_ACCESS;
				end
				ST_ACCESS: begin
					m_sel    <= 1'b1;
					m_enable <= 1'b1;
					if (M_APB_PREADY) begin
						m_state  <= ST_IDLE;
						m_sel    <= 1'b0;
						m_enable <= 1'b0;
					end
				end
				default: begin
					m_state  <= ST_IDLE;
					m_sel    <= 1'b0;
					m_enable <= 1'b0;
				end
			endcase
		end
	end

	assign M_APB_PSEL    = m_sel;
	assign M_APB_PENABLE = m_enable;
	assign M_APB_PADDR   = m_addr;
	assign M_APB_PWRITE  = m_write;
	assign M_APB_PWDATA  = m_wdata;
	assign M_APB_PWSTRB  = m_wstrb;
	assign M_APB_PPROT   = m_pprot;

	// ------------------------------------------------------------------
	// Synchronize completion (M_APB_PREADY in access phase) to S domain
	// ------------------------------------------------------------------
	wire m_done_pulse = (m_state == ST_ACCESS) && M_APB_PREADY;

	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			done_sync <= 2'b00;
		end else begin
			done_sync <= {done_sync[0], m_done_pulse};
		end
	end
	assign s_done = done_sync[1];

	// ------------------------------------------------------------------
	// Source-side: generate S_APB_PREADY and capture response
	// ------------------------------------------------------------------
	reg [DW-1:0] s_rdata;
	reg          s_pslverr;

	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			S_APB_PREADY <= 1'b0;
			s_rdata      <= {DW{1'b0}};
			s_pslverr    <= 1'b0;
		end else if (s_req_valid && s_done) begin
			S_APB_PREADY <= 1'b1;
			s_rdata      <= M_APB_PRDATA;
			s_pslverr    <= M_APB_PSLVERR;
		end else if (S_APB_PSEL && S_APB_PENABLE && S_APB_PREADY) begin
			S_APB_PREADY <= 1'b0;
		end else begin
			S_APB_PREADY <= 1'b0;
		end
	end

	assign S_APB_PRDATA  = s_rdata;
	assign S_APB_PSLVERR = s_pslverr;

endmodule
