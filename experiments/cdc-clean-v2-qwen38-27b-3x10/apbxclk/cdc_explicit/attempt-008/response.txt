module apbxclk #(
		parameter	C_APB_ADDR_WIDTH = 12,
		parameter	C_APB_DATA_WIDTH = 32,
		parameter [0:0]	OPT_REGISTERED = 1'b0,
		localparam	AW = C_APB_ADDR_WIDTH,
		localparam	DW = C_APB_DATA_WIDTH
	) (
		input	wire			S_APB_PCLK, S_PRESETn,
		input	wire			S_APB_PSEL,
		input	wire			S_APB_PENABLE,
		output	reg			S_APB_PREADY,
		input	wire	[AW-1:0]	S_APB_PADDR,
		input	wire			S_APB_PWRITE,
		input	wire	[DW-1:0]	S_APB_PWDATA,
		input	wire	[DW/8-1:0]	S_APB_PWSTRB,
		input	wire	[2:0]		S_APB_PPROT,
		output	wire	[DW-1:0]	S_APB_PRDATA,
		output	wire			S_APB_PSLVERR,
		input	wire			M_APB_PCLK,
		output	reg			M_PRESETn,
		output	reg			M_APB_PSEL,
		output	reg			M_APB_PENABLE,
		input	wire			M_APB_PREADY,
		output	wire	[AW-1:0]	M_APB_PADDR,
		output	wire			M_APB_PWRITE,
		output	wire	[DW-1:0]	M_APB_PWDATA,
		output	wire	[DW/8-1:0]	M_APB_PWSTRB,
		output	wire	[2:0]		M_APB_PPROT,
		input	wire	[DW-1:0]	M_APB_PRDATA,
		input	wire			M_APB_PSLVERR
	);

	// Internal signals
	reg [AW-1:0] m_addr_reg;
	reg m_write_reg;
	reg [DW-1:0] m_wdata_reg;
	reg [DW/8-1:0] m_wstrb_reg;
	reg [2:0] m_pprot_reg;

	reg m_sel_reg;
	reg m_enable_reg;

	reg m_req_valid;
	reg m_req_pending;

	reg [DW-1:0] m_rdata_reg;
	reg m_rdata_valid;
	reg m_rdata_pending;

	reg m_pslverr_reg;
	reg m_pslverr_valid;
	reg m_pslverr_pending;

	reg m_reset_sync1;
	reg m_reset_sync2;

	reg s_ack_sync1;
	reg s_ack_sync2;
	reg s_ack_sync3;

	reg s_rdata_valid_sync1;
	reg s_rdata_valid_sync2;
	reg s_rdata_valid_sync3;

	reg s_pslverr_valid_sync1;
	reg s_pslverr_valid_sync2;
	reg s_pslverr_valid_sync3;

	// M_APB_PCLK domain: reset synchronization
	always @(posedge M_APB_PCLK) begin
		m_reset_sync1 <= 1'b1;
		m_reset_sync2 <= m_reset_sync1;
	end

	assign M_PRESETn = m_reset_sync2;

	// M_APB_PCLK domain: capture request payload when valid
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_addr_reg <= {AW{1'b0}};
			m_write_reg <= 1'b0;
			m_wdata_reg <= {DW{1'b0}};
			m_wstrb_reg <= {(DW/8){1'b0}};
			m_pprot_reg <= 3'b000;
			m_sel_reg <= 1'b0;
			m_enable_reg <= 1'b0;
			m_req_valid <= 1'b0;
			m_req_pending <= 1'b0;
			m_rdata_reg <= {DW{1'b0}};
			m_rdata_valid <= 1'b0;
			m_rdata_pending <= 1'b0;
			m_pslverr_reg <= 1'b0;
			m_pslverr_valid <= 1'b0;
			m_pslverr_pending <= 1'b0;
		end else begin
			// Capture request when source indicates valid
			if (m_req_valid && !m_req_pending) begin
				m_addr_reg <= S_APB_PADDR;
				m_write_reg <= S_APB_PWRITE;
				m_wdata_reg <= S_APB_PWDATA;
				m_wstrb_reg <= S_APB_PWSTRB;
				m_pprot_reg <= S_APB_PPROT;
				m_req_pending <= 1'b1;
			end

			// APB state machine on M side
			if (m_req_pending) begin
				if (!m_sel_reg) begin
					// Setup phase
					m_sel_reg <= 1'b1;
					m_enable_reg <= 1'b0;
				end else if (!m_enable_reg) begin
					// Access phase
					m_sel_reg <= 1'b1;
					m_enable_reg <= 1'b1;
				end else begin
					// Access phase with PREADY
					if (M_APB_PREADY) begin
						m_sel_reg <= 1'b0;
						m_enable_reg <= 1'b0;
						m_req_pending <= 1'b0;
						// Capture response
						m_rdata_reg <= M_APB_PRDATA;
						m_rdata_valid <= 1'b1;
						m_rdata_pending <= 1'b1;
						m_pslverr_reg <= M_APB_PSLVERR;
						m_pslverr_valid <= 1'b1;
						m_pslverr_pending <= 1'b1;
					end
				end
			end else begin
				m_sel_reg <= 1'b0;
				m_enable_reg <= 1'b0;
			end

			// Clear response valid after it has been synchronized back
			if (m_rdata_pending && s_rdata_valid_sync3) begin
				m_rdata_valid <= 1'b0;
				m_rdata_pending <= 1'b0;
			end
			if (m_pslverr_pending && s_pslverr_valid_sync3) begin
				m_pslverr_valid <= 1'b0;
				m_pslverr_pending <= 1'b0;
			end
		end
	end

	// M_APB_PCLK domain: drive M_APB outputs
	assign M_APB_PSEL = m_sel_reg;
	assign M_APB_PENABLE = m_enable_reg;
	assign M_APB_PADDR = m_addr_reg;
	assign M_APB_PWRITE = m_write_reg;
	assign M_APB_PWDATA = m_wdata_reg;
	assign M_APB_PWSTRB = m_wstrb_reg;
	assign M_APB_PPROT = m_pprot_reg;

	// S_APB_PCLK domain: synchronize M_APB_PREADY back
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_ack_sync1 <= 1'b0;
			s_ack_sync2 <= 1'b0;
			s_ack_sync3 <= 1'b0;
		end else begin
			s_ack_sync1 <= m_enable_reg;
			s_ack_sync2 <= s_ack_sync1;
			s_ack_sync3 <= s_ack_sync2;
		end
	end

	// S_APB_PCLK domain: synchronize read data valid back
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_rdata_valid_sync1 <= 1'b0;
			s_rdata_valid_sync2 <= 1'b0;
			s_rdata_valid_sync3 <= 1'b0;
		end else begin
			s_rdata_valid_sync1 <= m_rdata_valid;
			s_rdata_valid_sync2 <= s_rdata_valid_sync1;
			s_rdata_valid_sync3 <= s_rdata_valid_sync2;
		end
	end

	// S_APB_PCLK domain: synchronize pslverr valid back
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_pslverr_valid_sync1 <= 1'b0;
			s_pslverr_valid_sync2 <= 1'b0;
			s_pslverr_valid_sync3 <= 1'b0;
		end else begin
			s_pslverr_valid_sync1 <= m_pslverr_valid;
			s_pslverr_valid_sync2 <= s_pslverr_valid_sync1;
			s_pslverr_valid_sync3 <= s_pslverr_valid_sync2;
		end
	end

	// S_APB_PCLK domain: capture read data and pslverr when valid
	reg [DW-1:0] s_rdata_reg;
	reg s_pslverr_reg;

	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_rdata_reg <= {DW{1'b0}};
			s_pslverr_reg <= 1'b0;
		end else begin
			if (s_rdata_valid_sync3) begin
				s_rdata_reg <= m_rdata_reg;
			end
			if (s_pslverr_valid_sync3) begin
				s_pslverr_reg <= m_pslverr_reg;
			end
		end
	end

	assign S_APB_PRDATA = s_rdata_reg;
	assign S_APB_PSLVERR = s_pslverr_reg;

	// S_APB_PCLK domain: APB source state machine
	reg s_busy;
	reg s_in_access;

	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			S_APB_PREADY <= 1'b0;
			s_busy <= 1'b0;
			s_in_access <= 1'b0;
			m_req_valid <= 1'b0;
		end else begin
			if (!s_busy) begin
				// Idle: wait for PSEL
				if (S_APB_PSEL) begin
					s_busy <= 1'b1;
					s_in_access <= 1'b0;
					m_req_valid <= 1'b1;
					S_APB_PREADY <= 1'b0;
				end else begin
					S_APB_PREADY <= 1'b0;
					m_req_valid <= 1'b0;
				end
			end else begin
				if (!s_in_access) begin
					// Setup phase: wait for PENABLE
					if (S_APB_PENABLE) begin
						s_in_access <= 1'b1;
						S_APB_PREADY <= 1'b0;
					end else begin
						S_APB_PREADY <= 1'b0;
					end
				end else begin
					// Access phase: wait for downstream completion
					if (s_ack_sync3) begin
						S_APB_PREADY <= 1'b1;
						s_busy <= 1'b0;
						s_in_access <= 1'b0;
						m_req_valid <= 1'b0;
					end else begin
						S_APB_PREADY <= 1'b0;
					end
				end
			end
		end
	end

endmodule
