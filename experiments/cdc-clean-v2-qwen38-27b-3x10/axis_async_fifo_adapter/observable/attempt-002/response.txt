module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam SYNC_STAGES = 2;

    // Internal data width for storage: data + keep + tlast + sidebands
    localparam SIDE_WIDTH = (ID_ENABLE ? ID_WIDTH : 0) + 
                            (DEST_ENABLE ? DEST_WIDTH : 0) + 
                            (USER_ENABLE ? USER_WIDTH : 0);
    localparam STORE_WIDTH = S_DATA_WIDTH + (S_KEEP_ENABLE ? S_KEEP_WIDTH : 0) + 1 + SIDE_WIDTH;

    // RAM storage
    reg [STORE_WIDTH-1:0] ram [0:DEPTH-1];

    // Write pointer (s_clk domain)
    reg [PTR_WIDTH-1:0] w_ptr;
    // Read pointer (m_clk domain)
    reg [PTR_WIDTH-1:0] r_ptr;

    // Synchronized read pointer to s_clk domain
    reg [PTR_WIDTH-1:0] r_ptr_sync [0:SYNC_STAGES-1];
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            r_ptr_sync[0] <= {PTR_WIDTH{1'b0}};
            r_ptr_sync[1] <= {PTR_WIDTH{1'b0}};
        end else begin
            r_ptr_sync[0] <= r_ptr;
            r_ptr_sync[1] <= r_ptr_sync[0];
        end
    end

    // Synchronized write pointer to m_clk domain
    reg [PTR_WIDTH-1:0] w_ptr_sync [0:SYNC_STAGES-1];
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            w_ptr_sync[0] <= {PTR_WIDTH{1'b0}};
            w_ptr_sync[1] <= {PTR_WIDTH{1'b0}};
        end else begin
            w_ptr_sync[0] <= w_ptr;
            w_ptr_sync[1] <= w_ptr_sync[0];
        end
    end

    // Compute full/empty
    wire [PTR_WIDTH-1:0] r_ptr_s = r_ptr_sync[SYNC_STAGES-1];
    wire [PTR_WIDTH-1:0] w_ptr_m = w_ptr_sync[SYNC_STAGES-1];

    // Depth calculation
    wire [PTR_WIDTH-1:0] s_depth_raw = w_ptr - r_ptr_s;
    wire [PTR_WIDTH-1:0] m_depth_raw = w_ptr_m - r_ptr;

    // Full condition: depth == DEPTH
    wire s_full = (s_depth_raw == DEPTH);
    wire m_full = (m_depth_raw == DEPTH);
    wire m_empty = (m_depth_raw == 0);

    // Pause logic
    reg s_pause_active;
    reg m_pause_active;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) s_pause_active <= 1'b0;
        else if (PAUSE_ENABLE) s_pause_active <= s_pause_req;
        else s_pause_active <= 1'b0;
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) m_pause_active <= 1'b0;
        else if (PAUSE_ENABLE) m_pause_active <= m_pause_req;
        else m_pause_active <= 1'b0;
    end

    assign s_pause_ack = s_pause_active;
    assign m_pause_ack = m_pause_active;

    // Write enable
    wire s_write_en = s_axis_tvalid && s_axis_tready && !s_pause_active;
    assign s_axis_tready = !s_full && !s_pause_active;

    // Build store data
    reg [STORE_WIDTH-1:0] store_data;
    integer i;
    always @* begin
        store_data = {STORE_WIDTH{1'b0}};
        // Pack: [tlast][keep][data][sidebands]
        // We'll use a simple concatenation approach
        // For simplicity, let's define the layout:
        // bits [0:S_DATA_WIDTH-1] = tdata
        // bits [S_DATA_WIDTH:S_DATA_WIDTH+S_KEEP_WIDTH-1] = tkeep (if enabled)
        // next bit = tlast
        // remaining = sidebands
        store_data[0:S_DATA_WIDTH-1] = s_axis_tdata;
        if (S_KEEP_ENABLE) begin
            store_data[S_DATA_WIDTH:S_DATA_WIDTH+S_KEEP_WIDTH-1] = s_axis_tkeep;
        end
        store_data[S_DATA_WIDTH + (S_KEEP_ENABLE ? S_KEEP_WIDTH : 0)] = s_axis_tlast;
        // Sidebands
        if (SIDE_WIDTH > 0) begin
            // Pack sidebands
            reg [SIDE_WIDTH-1:0] side_data;
            integer si;
            side_data = {SIDE_WIDTH{1'b0}};
            si = 0;
            if (ID_ENABLE) begin
                side_data[si:si+ID_WIDTH-1] = s_axis_tid;
                si = si + ID_WIDTH;
            end
            if (DEST_ENABLE) begin
                side_data[si:si+DEST_WIDTH-1] = s_axis_tdest;
                si = si + DEST_WIDTH;
            end
            if (USER_ENABLE) begin
                side_data[si:si+USER_WIDTH-1] = s_axis_tuser;
                si = si + USER_WIDTH;
            end
            store_data[S_DATA_WIDTH + (S_KEEP_ENABLE ? S_KEEP_WIDTH : 0) + 1 : 
                       S_DATA_WIDTH + (S_KEEP_ENABLE ? S_KEEP_WIDTH : 0) + SIDE_WIDTH] = side_data;
        end
    end

    // Write to RAM
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            w_ptr <= {PTR_WIDTH{1'b0}};
        end else if (s_write_en) begin
            ram[w_ptr[ADDR_WIDTH-1:0]] <= store_data;
            w_ptr <= w_ptr + 1'b1;
        end
    end

    // Read from RAM (m_clk domain)
    reg [STORE_WIDTH-1:0] read_data;
    wire [ADDR_WIDTH-1:0] r_addr = r_ptr[ADDR_WIDTH-1:0];
    
    // Simple read: combinational read from RAM
    assign read_data = ram[r_addr];

    // Read enable
    wire m_read_en = m_axis_tvalid && m_axis_tready && !m_pause_active;
    assign m_axis_tvalid = !m_empty && !m_pause_active;

    // Unpack read data
    wire [S_DATA_WIDTH-1:0] r_data = read_data[0:S_DATA_WIDTH-1];
    wire [S_KEEP_WIDTH-1:0] r_keep = S_KEEP_ENABLE ? read_data[S_DATA_WIDTH:S_DATA_WIDTH+S_KEEP_WIDTH-1] : {S_KEEP_WIDTH{1'b0}};
    wire r_tlast = read_data[S_DATA_WIDTH + (S_KEEP_ENABLE ? S_KEEP_WIDTH : 0)];

    // Unpack sidebands
    wire [SIDE_WIDTH-1:0] r_side = (SIDE_WIDTH > 0) ? 
        read_data[S_DATA_WIDTH + (S_KEEP_ENABLE ? S_KEEP_WIDTH : 0) + 1 : 
                  S_DATA_WIDTH + (S_KEEP_ENABLE ? S_KEEP_WIDTH : 0) + SIDE_WIDTH] : {SIDE_WIDTH{1'b0}};
    
    wire [ID_WIDTH-1:0] r_tid;
    wire [DEST_WIDTH-1:0] r_tdest;
    wire [USER_WIDTH-1:0] r_tuser;
    
    wire [SIDE_WIDTH-1:0] side_bits = r_side;
    wire [ID_WIDTH-1:0] tid_val = ID_ENABLE ? side_bits[0:ID_WIDTH-1] : {ID_WIDTH{1'b0}};
    wire [DEST_WIDTH-1:0] dest_val = DEST_ENABLE ? side_bits[ID_ENABLE ? ID_WIDTH : 0 : 
                                                              ID_ENABLE ? ID_WIDTH+DEST_WIDTH-1 : DEST_WIDTH-1] : {DEST_WIDTH{1'b0}};
    wire [USER_WIDTH-1:0] user_val = USER_ENABLE ? side_bits[(ID_ENABLE ? ID_WIDTH : 0) + (DEST_ENABLE ? DEST_WIDTH : 0) : 
                                                              (ID_ENABLE ? ID_WIDTH : 0) + (DEST_ENABLE ? DEST_WIDTH : 0) + USER_WIDTH-1] : {USER_WIDTH{1'b0}};

    assign m_axis_tid = ID_ENABLE ? tid_val : {ID_WIDTH{1'b0}};
    assign m_axis_tdest = DEST_ENABLE ? dest_val : {DEST_WIDTH{1'b0}};
    assign m_axis_tuser = USER_ENABLE ? user_val : {USER_WIDTH{1'b0}};

    // Width adaptation: S_DATA_WIDTH to M_DATA_WIDTH
    // For simplicity, assume S_DATA_WIDTH == M_DATA_WIDTH or handle basic case
    // In a real implementation, this would be more complex
    // Here we do a simple pass-through with keep adjustment
    
    reg [M_DATA_WIDTH-1:0] m_data_out;
    reg [M_KEEP_WIDTH-1:0] m_keep_out;
    reg m_last_out;

    always @* begin
        // Simple width adaptation: if widths are equal, pass through
        // If different, we'd need a proper width converter
        // For this implementation, we assume the widths are compatible
        // and do a basic mapping
        
        if (S_DATA_WIDTH == M_DATA_WIDTH) begin
            m_data_out = r_data[0:M_DATA_WIDTH-1];
            m_keep_out = S_KEEP_ENABLE && M_KEEP_ENABLE ? r_keep[0:M_KEEP_WIDTH-1] : {M_KEEP_WIDTH{1'b0}};
            m_last_out = r_tlast;
        end else if (S_DATA_WIDTH > M_DATA_WIDTH) begin
            // Take the lower M_DATA_WIDTH bits
            m_data_out = r_data[0:M_DATA_WIDTH-1];
            m_keep_out = M_KEEP_ENABLE ? {M_KEEP_WIDTH{1'b1}} : {M_KEEP_WIDTH{1'b0}};
            m_last_out = r_tlast;
        end else begin
            // S_DATA_WIDTH < M_DATA_WIDTH
            m_data_out = {M_DATA_WIDTH-1:S_DATA_WIDTH{1'b0}} | r_data;
            m_keep_out = M_KEEP_ENABLE ? {M_KEEP_WIDTH{1'b1}} : {M_KEEP_WIDTH{1'b0}};
            m_last_out = r_tlast;
        end
    end

    assign m_axis_tdata = m_data_out;
    assign m_axis_tkeep = m_keep_out;
    assign m_axis_tlast = m_last_out;

    // Update read pointer
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            r_ptr <= {PTR_WIDTH{1'b0}};
        end else if (m_read_en) begin
            r_ptr <= r_ptr + 1'b1;
        end
    end

    // Status outputs
    assign s_status_depth = s_depth_raw;
    assign s_status_depth_commit = s_depth_raw;
    assign s_status_overflow = s_full;
    assign s_status_bad_frame = 1'b0;
    assign s_status_good_frame = 1'b0;

    assign m_status_depth = m_depth_raw;
    assign m_status_depth_commit = m_depth_raw;
    assign m_status_overflow = m_full;
    assign m_status_bad_frame = 1'b0;
    assign m_status_good_frame = 1'b0;

endmodule
