module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Internal data width for storage: data + keep + last + user
    localparam STORE_DATA_WIDTH = S_DATA_WIDTH;
    localparam STORE_KEEP_WIDTH = S_KEEP_WIDTH;
    localparam STORE_USER_WIDTH = USER_WIDTH;
    localparam STORE_TOTAL_WIDTH = STORE_DATA_WIDTH + STORE_KEEP_WIDTH + 1 + STORE_USER_WIDTH;

    // RAM storage
    reg [STORE_TOTAL_WIDTH-1:0] ram [0:DEPTH-1];

    // Write pointer (s_clk domain)
    reg [PTR_WIDTH-1:0] w_ptr;
    // Read pointer (m_clk domain)
    reg [PTR_WIDTH-1:0] r_ptr;

    // Gray code pointers for CDC
    reg [PTR_WIDTH-1:0] w_ptr_gray;
    reg [PTR_WIDTH-1:0] r_ptr_gray;

    // Synchronizers
    reg [PTR_WIDTH-1:0] w_ptr_gray_sync1, w_ptr_gray_sync2;
    reg [PTR_WIDTH-1:0] r_ptr_gray_sync1, r_ptr_gray_sync2;

    // Synchronizers for pause
    reg s_pause_req_sync1, s_pause_req_sync2;
    reg m_pause_req_sync1, m_pause_req_sync2;

    // Status flags
    reg s_overflow;
    reg m_overflow;
    reg s_bad_frame;
    reg m_bad_frame;
    reg s_good_frame;
    reg m_good_frame;

    // Pause logic
    reg s_pause_active;
    reg m_pause_active;

    // Write side logic
    wire s_write_enable = s_axis_tvalid && s_axis_tready;
    wire s_full = (w_ptr[PTR_WIDTH-1] == r_ptr_gray_sync2[PTR_WIDTH-1]) &&
                  (w_ptr[ADDR_WIDTH-1:0] == r_ptr_gray_sync2[ADDR_WIDTH-1:0]);
    wire s_almost_full = (w_ptr[PTR_WIDTH-1] == r_ptr_gray_sync2[PTR_WIDTH-1]) &&
                         (w_ptr[ADDR_WIDTH-1:0] == (r_ptr_gray_sync2[ADDR_WIDTH-1:0] + 1'b1));

    // Read side logic
    wire m_read_enable = m_axis_tvalid && m_axis_tready;
    wire m_empty = (r_ptr[PTR_WIDTH-1] == w_ptr_gray_sync2[PTR_WIDTH-1]) &&
                   (r_ptr[ADDR_WIDTH-1:0] == w_ptr_gray_sync2[ADDR_WIDTH-1:0]);

    // Depth calculation
    wire [PTR_WIDTH-1:0] s_depth = w_ptr - r_ptr_gray_sync2;
    wire [PTR_WIDTH-1:0] m_depth = w_ptr_gray_sync2 - r_ptr;

    // tready generation
    assign s_axis_tready = !s_full && !s_pause_active && !s_rst;
    assign m_axis_tvalid = !m_empty && !m_rst;

    // Pause ack
    assign s_pause_ack = s_pause_active;
    assign m_pause_ack = m_pause_active;

    // Status outputs
    assign s_status_depth = s_depth[ADDR_WIDTH:0];
    assign s_status_depth_commit = s_depth[ADDR_WIDTH:0];
    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;
    assign m_status_depth = m_depth[ADDR_WIDTH:0];
    assign m_status_depth_commit = m_depth[ADDR_WIDTH:0];
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;

    // Gray code conversion
    function [PTR_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    // Write pointer update
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            w_ptr <= {PTR_WIDTH{1'b0}};
            w_ptr_gray <= {PTR_WIDTH{1'b0}};
            s_overflow <= 1'b0;
            s_bad_frame <= 1'b0;
            s_good_frame <= 1'b0;
            s_pause_active <= 1'b0;
        end else begin
            if (s_write_enable) begin
                w_ptr <= w_ptr + 1'b1;
                w_ptr_gray <= bin2gray(w_ptr + 1'b1);
            end
            if (s_write_enable && s_full) begin
                s_overflow <= 1'b1;
            end
            if (s_write_enable && s_axis_tlast) begin
                if (DROP_BAD_FRAME && s_axis_tuser[0]) begin
                    s_bad_frame <= 1'b1;
                end else begin
                    s_good_frame <= 1'b1;
                end
            end
            if (PAUSE_ENABLE) begin
                if (s_pause_req_sync2 && !s_pause_active) begin
                    s_pause_active <= 1'b1;
                end else if (!s_pause_req_sync2 && s_pause_active) begin
                    s_pause_active <= 1'b0;
                end
            end
        end
    end

    // RAM write
    always @(posedge s_clk) begin
        if (s_write_enable) begin
            ram[w_ptr[ADDR_WIDTH-1:0]] <= {s_axis_tdata, s_axis_tkeep, s_axis_tlast, s_axis_tuser};
        end
    end

    // Read pointer update
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            r_ptr <= {PTR_WIDTH{1'b0}};
            r_ptr_gray <= {PTR_WIDTH{1'b0}};
            m_overflow <= 1'b0;
            m_bad_frame <= 1'b0;
            m_good_frame <= 1'b0;
            m_pause_active <= 1'b0;
        end else begin
            if (m_read_enable) begin
                r_ptr <= r_ptr + 1'b1;
                r_ptr_gray <= bin2gray(r_ptr + 1'b1);
            end
            if (m_read_enable && m_axis_tlast) begin
                if (DROP_BAD_FRAME && m_axis_tuser[0]) begin
                    m_bad_frame <= 1'b1;
                end else begin
                    m_good_frame <= 1'b1;
                end
            end
            if (PAUSE_ENABLE) begin
                if (m_pause_req_sync2 && !m_pause_active) begin
                    m_pause_active <= 1'b1;
                end else if (!m_pause_req_sync2 && m_pause_active) begin
                    m_pause_active <= 1'b0;
                end
            end
        end
    end

    // RAM read
    reg [STORE_TOTAL_WIDTH-1:0] ram_read_data;
    always @(posedge m_clk) begin
        ram_read_data <= ram[r_ptr[ADDR_WIDTH-1:0]];
    end

    // Output data
    assign m_axis_tdata = ram_read_data[STORE_DATA_WIDTH-1:0];
    assign m_axis_tkeep = ram_read_data[STORE_DATA_WIDTH + STORE_KEEP_WIDTH-1:STORE_DATA_WIDTH];
    assign m_axis_tlast = ram_read_data[STORE_DATA_WIDTH + STORE_KEEP_WIDTH];
    assign m_axis_tuser = ram_read_data[STORE_TOTAL_WIDTH-1:STORE_DATA_WIDTH + STORE_KEEP_WIDTH + 1];
    assign m_axis_tid = {ID_WIDTH{1'b0}};
    assign m_axis_tdest = {DEST_WIDTH{1'b0}};

    // Synchronizers
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            w_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            w_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
            s_pause_req_sync1 <= 1'b0;
            s_pause_req_sync2 <= 1'b0;
        end else begin
            w_ptr_gray_sync1 <= w_ptr_gray;
            w_ptr_gray_sync2 <= w_ptr_gray_sync1;
            s_pause_req_sync1 <= s_pause_req;
            s_pause_req_sync2 <= s_pause_req_sync1;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            r_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            r_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
            m_pause_req_sync1 <= 1'b0;
            m_pause_req_sync2 <= 1'b0;
        end else begin
            r_ptr_gray_sync1 <= r_ptr_gray;
            r_ptr_gray_sync2 <= r_ptr_gray_sync1;
            m_pause_req_sync1 <= m_pause_req;
            m_pause_req_sync2 <= m_pause_req_sync1;
        end
    end

endmodule
