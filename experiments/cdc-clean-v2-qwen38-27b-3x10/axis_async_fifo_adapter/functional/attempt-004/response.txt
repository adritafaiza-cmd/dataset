module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Internal signals
    wire s_wr_en;
    wire m_rd_en;
    wire s_full;
    wire m_empty;
    wire [S_DATA_WIDTH-1:0] s_wr_data;
    wire [S_KEEP_WIDTH-1:0] s_wr_keep;
    wire s_wr_last;
    wire [ID_WIDTH-1:0] s_wr_id;
    wire [DEST_WIDTH-1:0] s_wr_dest;
    wire [USER_WIDTH-1:0] s_wr_user;

    wire [M_DATA_WIDTH-1:0] m_rd_data;
    wire [M_KEEP_WIDTH-1:0] m_rd_keep;
    wire m_rd_last;
    wire [ID_WIDTH-1:0] m_rd_id;
    wire [DEST_WIDTH-1:0] m_rd_dest;
    wire [USER_WIDTH-1:0] m_rd_user;

    // Pause logic
    reg s_paused = 1'b0;
    reg m_paused = 1'b0;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_paused <= 1'b0;
        else if (PAUSE_ENABLE)
            s_paused <= s_pause_req;
        else
            s_paused <= 1'b0;
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_paused <= 1'b0;
        else if (PAUSE_ENABLE)
            m_paused <= m_pause_req;
        else
            m_paused <= 1'b0;
    end

    assign s_pause_ack = s_paused;
    assign m_pause_ack = m_paused;

    // Write enable: valid, not full, not paused
    assign s_wr_en = s_axis_tvalid & ~s_full & ~s_paused;

    // Read enable: not empty, ready, not paused
    assign m_rd_en = ~m_empty & m_axis_tready & ~m_paused;

    // Write data
    assign s_wr_data = s_axis_tdata;
    assign s_wr_keep = s_axis_tkeep;
    assign s_wr_last = s_axis_tlast;
    assign s_wr_id = s_axis_tid;
    assign s_wr_dest = s_axis_tdest;
    assign s_wr_user = s_axis_tuser;

    // Read data
    assign m_rd_data = m_axis_tdata;
    assign m_rd_keep = m_axis_tkeep;
    assign m_rd_last = m_axis_tlast;
    assign m_rd_id = m_axis_tid;
    assign m_rd_dest = m_axis_tdest;
    assign m_rd_user = m_axis_tuser;

    // Status
    assign s_status_overflow = s_axis_tvalid & s_full;
    assign m_status_overflow = 1'b0; // Not applicable for read side in this context
    assign s_status_bad_frame = 1'b0;
    assign s_status_good_frame = 1'b0;
    assign m_status_bad_frame = 1'b0;
    assign m_status_good_frame = 1'b0;

    // Depth calculation
    wire [PTR_WIDTH-1:0] s_wr_ptr;
    wire [PTR_WIDTH-1:0] m_rd_ptr;
    wire [PTR_WIDTH-1:0] s_wr_ptr_m;
    wire [PTR_WIDTH-1:0] m_rd_ptr_s;

    // Async FIFO core
    async_fifo #(
        .DEPTH(DEPTH),
        .DATA_WIDTH(S_DATA_WIDTH + S_KEEP_WIDTH + 1 + ID_WIDTH + DEST_WIDTH + USER_WIDTH)
    ) u_fifo (
        .wr_clk(s_clk),
        .wr_rst(s_rst),
        .wr_en(s_wr_en),
        .wr_data({s_wr_user, s_wr_dest, s_wr_id, s_wr_last, s_wr_keep, s_wr_data}),
        .full(s_full),
        .rd_clk(m_clk),
        .rd_rst(m_rst),
        .rd_en(m_rd_en),
        .rd_data({m_rd_user, m_rd_dest, m_rd_id, m_rd_last, m_rd_keep, m_rd_data}),
        .empty(m_empty)
    );

    // Pointer synchronization for depth
    // Write pointer to master clock
    wire [PTR_WIDTH-1:0] s_wr_ptr_sync1;
    wire [PTR_WIDTH-1:0] s_wr_ptr_sync2;
    wire [PTR_WIDTH-1:0] s_wr_ptr_sync3;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_wr_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            s_wr_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            s_wr_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            s_wr_ptr_sync1 <= s_wr_ptr;
            s_wr_ptr_sync2 <= s_wr_ptr_sync1;
            s_wr_ptr_sync3 <= s_wr_ptr_sync2;
        end
    end
    assign s_wr_ptr_m = s_wr_ptr_sync3;

    // Read pointer to slave clock
    wire [PTR_WIDTH-1:0] m_rd_ptr_sync1;
    wire [PTR_WIDTH-1:0] m_rd_ptr_sync2;
    wire [PTR_WIDTH-1:0] m_rd_ptr_sync3;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_rd_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            m_rd_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            m_rd_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            m_rd_ptr_sync1 <= m_rd_ptr;
            m_rd_ptr_sync2 <= m_rd_ptr_sync1;
            m_rd_ptr_sync3 <= m_rd_ptr_sync2;
        end
    end
    assign m_rd_ptr_s = m_rd_ptr_sync3;

    // Depth in slave clock domain
    wire [PTR_WIDTH-1:0] s_depth_raw = s_wr_ptr - m_rd_ptr_s;
    assign s_status_depth = s_depth_raw[PTR_WIDTH-1:0];
    assign s_status_depth_commit = s_depth_raw[PTR_WIDTH-1:0];

    // Depth in master clock domain
    wire [PTR_WIDTH-1:0] m_depth_raw = s_wr_ptr_m - m_rd_ptr;
    assign m_status_depth = m_depth_raw[PTR_WIDTH-1:0];
    assign m_status_depth_commit = m_depth_raw[PTR_WIDTH-1:0];

    // tready
    assign s_axis_tready = ~s_full & ~s_paused;

    // tvalid
    assign m_axis_tvalid = ~m_empty & ~m_paused;

endmodule

// Simple async FIFO implementation
module async_fifo #(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8
) (
    input wire wr_clk,
    input wire wr_rst,
    input wire wr_en,
    input wire [DATA_WIDTH-1:0] wr_data,
    output wire full,
    input wire rd_clk,
    input wire rd_rst,
    input wire rd_en,
    output wire [DATA_WIDTH-1:0] rd_data,
    output wire empty
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];

    // Write pointer
    reg [PTR_WIDTH-1:0] wr_ptr = {PTR_WIDTH{1'b0}};
    wire [ADDR_WIDTH-1:0] wr_addr = wr_ptr[ADDR_WIDTH-1:0];

    // Read pointer
    reg [PTR_WIDTH-1:0] rd_ptr = {PTR_WIDTH{1'b0}};
    wire [ADDR_WIDTH-1:0] rd_addr = rd_ptr[ADDR_WIDTH-1:0];

    // Write pointer synchronization to read clock
    reg [PTR_WIDTH-1:0] wr_ptr_sync1 = {PTR_WIDTH{1'b0}};
    reg [PTR_WIDTH-1:0] wr_ptr_sync2 = {PTR_WIDTH{1'b0}};
    reg [PTR_WIDTH-1:0] wr_ptr_sync3 = {PTR_WIDTH{1'b0}};

    always @(posedge rd_clk or posedge rd_rst) begin
        if (rd_rst) begin
            wr_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            wr_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            wr_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            wr_ptr_sync1 <= wr_ptr;
            wr_ptr_sync2 <= wr_ptr_sync1;
            wr_ptr_sync3 <= wr_ptr_sync2;
        end
    end

    // Read pointer synchronization to write clock
    reg [PTR_WIDTH-1:0] rd_ptr_sync1 = {PTR_WIDTH{1'b0}};
    reg [PTR_WIDTH-1:0] rd_ptr_sync2 = {PTR_WIDTH{1'b0}};
    reg [PTR_WIDTH-1:0] rd_ptr_sync3 = {PTR_WIDTH{1'b0}};

    always @(posedge wr_clk or posedge wr_rst) begin
        if (wr_rst) begin
            rd_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            rd_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            rd_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            rd_ptr_sync1 <= rd_ptr;
            rd_ptr_sync2 <= rd_ptr_sync1;
            rd_ptr_sync3 <= rd_ptr_sync2;
        end
    end

    // Full and empty detection
    wire [PTR_WIDTH-1:0] wr_ptr_rd = wr_ptr_sync3;
    wire [PTR_WIDTH-1:0] rd_ptr_wr = rd_ptr_sync3;

    assign full = (wr_ptr[ADDR_WIDTH] != rd_ptr_wr[ADDR_WIDTH]) && 
                  (wr_ptr[ADDR_WIDTH-1:0] == rd_ptr_wr[ADDR_WIDTH-1:0]);
    assign empty = (rd_ptr == wr_ptr_rd);

    // Write logic
    always @(posedge wr_clk or posedge wr_rst) begin
        if (wr_rst)
            wr_ptr <= {PTR_WIDTH{1'b0}};
        else if (wr_en && !full)
            wr_ptr <= wr_ptr + 1'b1;
    end

    always @(posedge wr_clk) begin
        if (wr_en && !full)
            mem[wr_addr] <= wr_data;
    end

    // Read logic
    always @(posedge rd_clk or posedge rd_rst) begin
        if (rd_rst)
            rd_ptr <= {PTR_WIDTH{1'b0}};
        else if (rd_en && !empty)
            rd_ptr <= rd_ptr + 1'b1;
    end

    assign rd_data = mem[rd_addr];

endmodule
