module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,

    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Internal signals
    wire s_wr_en;
    wire m_rd_en;
    wire s_full;
    wire m_empty;
    wire s_overflow;
    wire m_overflow;
    wire s_bad_frame;
    wire m_bad_frame;
    wire s_good_frame;
    wire m_good_frame;
    wire s_pause_active;
    wire m_pause_active;

    // Synchronous FIFO core
    wire [S_DATA_WIDTH-1:0] fifo_tdata;
    wire [S_KEEP_WIDTH-1:0] fifo_tkeep;
    wire fifo_tlast;
    wire [ID_WIDTH-1:0] fifo_tid;
    wire [DEST_WIDTH-1:0] fifo_tdest;
    wire [USER_WIDTH-1:0] fifo_tuser;

    // Width adapter signals
    wire [M_DATA_WIDTH-1:0] adapter_tdata;
    wire [M_KEEP_WIDTH-1:0] adapter_tkeep;
    wire adapter_tlast;
    wire [ID_WIDTH-1:0] adapter_tid;
    wire [DEST_WIDTH-1:0] adapter_tdest;
    wire [USER_WIDTH-1:0] adapter_tuser;
    wire adapter_valid;
    wire adapter_ready;

    // Pause logic
    assign s_pause_active = PAUSE_ENABLE ? s_pause_req : 1'b0;
    assign m_pause_active = PAUSE_ENABLE ? m_pause_req : 1'b0;
    assign s_pause_ack = PAUSE_ENABLE ? s_pause_req : 1'b0;
    assign m_pause_ack = PAUSE_ENABLE ? m_pause_req : 1'b0;

    // Synchronous FIFO instantiation

    reg [PTR_WIDTH-1:0] m_rd_ptr_reg;

        else if (s_wr_en)

    always @(posedge m_clk or posedge m_rst) begin

        else if (m_rd_en)

    assign m_rd_ptr = m_rd_ptr_reg;

    // Depth calculation
    wire [PTR_WIDTH-1:0] s_depth_raw = s_wr_ptr - m_rd_ptr_sync;

    assign m_status_depth = m_depth_raw[ADDR_WIDTH:0];
    assign m_status_depth_commit = m_depth_raw[ADDR_WIDTH:0];

    width_adapter #(
        .S_DATA_WIDTH(S_DATA_WIDTH),
        .S_KEEP_ENABLE(S_KEEP_ENABLE),
        .S_KEEP_WIDTH(S_KEEP_WIDTH),
        .M_DATA_WIDTH(M_DATA_WIDTH),
        .M_KEEP_ENABLE(M_KEEP_ENABLE),
        .M_KEEP_WIDTH(M_KEEP_WIDTH),
        .ID_ENABLE(ID_ENABLE),
        .ID_WIDTH(ID_WIDTH),
        .DEST_ENABLE(DEST_ENABLE),
        .DEST_WIDTH(DEST_WIDTH),
        .USER_ENABLE(USER_ENABLE),
        .USER_WIDTH(USER_WIDTH)

        .rst(m_rst),
        .s_valid(!m_empty),
        .s_ready(adapter_ready),
        .s_tdata(fifo_tdata),
        .s_tkeep(fifo_tkeep),
        .s_tlast(fifo_tlast),
        .s_tid(fifo_tid),
        .s_tdest(fifo_tdest),
        .s_tuser(fifo_tuser),
        .m_valid(adapter_valid),
        .m_ready(m_axis_tready),
        .m_tdata(adapter_tdata),
        .m_tkeep(adapter_tkeep),
        .m_tlast(adapter_tlast),
        .m_tid(adapter_tid),
        .m_tdest(adapter_tdest),
        .m_tuser(adapter_tuser)

    assign m_axis_tdata = adapter_tdata;
    assign m_axis_tkeep = adapter_tkeep;
    assign m_axis_tvalid = adapter_valid && !m_pause_active;
    assign m_axis_tlast = adapter_tlast;

// Synchronous FIFO
module sync_fifo #(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8
) (
    input  wire                  clk,
    input  wire                  rst,
    input  wire                  wr_en,
    input  wire                  rd_en,
    input  wire [DATA_WIDTH-1:0] din,
    output wire [DATA_WIDTH-1:0] dout,
    output wire                  full,
    output wire                  empty
);
    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];
    reg [ADDR_WIDTH-1:0] wr_addr;
    reg [ADDR_WIDTH-1:0] rd_addr;
    reg [PTR_WIDTH-1:0] wr_ptr;
    reg [PTR_WIDTH-1:0] rd_ptr;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            wr_addr <= {ADDR_WIDTH{1'b0}};
            rd_addr <= {ADDR_WIDTH{1'b0}};
            wr_ptr <= {PTR_WIDTH{1'b0}};
            rd_ptr <= {PTR_WIDTH{1'b0}};
        end else begin
            if (wr_en) begin
                mem[wr_addr] <= din;
                wr_addr <= wr_addr + 1'b1;
                wr_ptr <= wr_ptr + 1'b1;
            end
            if (rd_en) begin
                rd_addr <= rd_addr + 1'b1;
                rd_ptr <= rd_ptr + 1'b1;
            end
        end
    end

    assign dout = mem[rd_addr];
    assign full = (wr_ptr - rd_ptr) == DEPTH;
    assign empty = (wr_ptr == rd_ptr);

endmodule

// Pointer synchronizer
module sync_ptr #(
    parameter WIDTH = 8
) (
    input  wire              clk,
    input  wire              rst,
    input  wire [WIDTH-1:0] din,
    output wire [WIDTH-1:0] dout
);
    reg [WIDTH-1:0] sync_reg1;
    reg [WIDTH-1:0] sync_reg2;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            sync_reg1 <= {WIDTH{1'b0}};
            sync_reg2 <= {WIDTH{1'b0}};
        end else begin
            sync_reg1 <= din;
            sync_reg2 <= sync_reg1;
        end
    end

    assign dout = sync_reg2;

endmodule
