```verilog
module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Internal signals
    wire s_wr_en;
    wire m_rd_en;
    wire s_full;
    wire m_empty;
    wire s_overflow;
    wire m_overflow;
    wire s_bad_frame;
    wire m_bad_frame;
    wire s_good_frame;
    wire m_good_frame;
    wire s_pause_active;
    wire m_pause_active;

    // Write side (s_clk domain)
    reg [PTR_WIDTH-1:0] s_wr_ptr;
    reg [PTR_WIDTH-1:0] s_rd_ptr_sync;
    reg [PTR_WIDTH-1:0] s_rd_ptr_sync2;
    reg [PTR_WIDTH-1:0] s_rd_ptr_sync3;
    reg [ADDR_WIDTH-1:0] s_wr_addr;
    reg [S_DATA_WIDTH-1:0] s_wr_data;
    reg [S_KEEP_WIDTH-1:0] s_wr_keep;
    reg s_wr_tlast;
    reg [ID_WIDTH-1:0] s_wr_tid;
    reg [DEST_WIDTH-1:0] s_wr_tdest;
    reg [USER_WIDTH-1:0] s_wr_tuser;
    reg s_wr_overflow;
    reg s_wr_bad_frame;
    reg s_wr_good_frame;

    // Read side (m_clk domain)
    reg [PTR_WIDTH-1:0] m_rd_ptr;
    reg [PTR_WIDTH-1:0] m_wr_ptr_sync;
    reg [PTR_WIDTH-1:0] m_wr_ptr_sync2;
    reg [PTR_WIDTH-1:0] m_wr_ptr_sync3;
    reg [ADDR_WIDTH-1:0] m_rd_addr;
    reg [M_DATA_WIDTH-1:0] m_rd_data;
    reg [M_KEEP_WIDTH-1:0] m_rd_keep;
    reg m_rd_tlast;
    reg [ID_WIDTH-1:0] m_rd_tid;
    reg [DEST_WIDTH-1:0] m_rd_tdest;
    reg [USER_WIDTH-1:0] m_rd_tuser;
    reg m_rd_overflow;
    reg m_rd_bad_frame;
    reg m_rd_good_frame;

    // RAM storage
    reg [S_DATA_WIDTH-1:0] ram_data [0:DEPTH-1];
    reg [S_KEEP_WIDTH-1:0] ram_keep [0:DEPTH-1];
    reg ram_tlast [0:DEPTH-1];
    reg [ID_WIDTH-1:0] ram_tid [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] ram_tdest [0:DEPTH-1];
    reg [USER_WIDTH-1:0] ram_tuser [0:DEPTH-1];
    reg ram_overflow [0:DEPTH-1];
    reg ram_bad_frame [0:DEPTH-1];
    reg ram_good_frame [0:DEPTH-1];

    // Write address
    assign s_wr_addr = s_wr_ptr[ADDR_WIDTH-1:0];

    // Read address
    assign m_rd_addr = m_rd_ptr[ADDR_WIDTH-1:0];

    // Synchronization of read pointer to write domain
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_rd_ptr_sync <= {PTR_WIDTH{1'b0}};
            s_rd_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            s_rd_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            s_rd_ptr_sync <= m_rd_ptr;
            s_rd_ptr_sync2 <= s_rd_ptr_sync;
            s_rd_ptr_sync3 <= s_rd_ptr_sync2;
        end
    end

    // Synchronization of write pointer to read domain
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_wr_ptr_sync <= {PTR_WIDTH{1'b0}};
            m_wr_ptr_sync2 <= {PTR_WIDTH{1'b0}};
            m_wr_ptr_sync3 <= {PTR_WIDTH{1'b0}};
        end else begin
            m_wr_ptr_sync <= s_wr_ptr;
            m_wr_ptr_sync2 <= m_wr_ptr_sync;
            m_wr_ptr_sync3 <= m_wr_ptr_sync2;
        end
    end

    // Full/Empty detection
    wire s_full = (s_wr_ptr == s_rd_ptr_sync3) && (s_wr_ptr[PTR_WIDTH-1] != s_rd_ptr_sync3[PTR_WIDTH-1]);
    wire m_empty = (m_rd_ptr == m_wr_ptr_sync3);

    // Pause logic
    wire s_pause = PAUSE_ENABLE ? s_pause_req : 1'b0;
    wire m_pause = PAUSE_ENABLE ? m_pause_req : 1'b0;

    // Write enable
    assign s_wr_en = s_axis_tvalid && s_axis_tready && !s_pause;

    // Read enable
    assign m_rd_en = m_axis_tvalid && m_axis_tready && !m_pause;

    // Write pointer update
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= {PTR_WIDTH{1'b0}};
        end else if (s_wr_en) begin
            s_wr_ptr <= s_wr_ptr + 1'b1;
        end
    end

    // Read pointer update
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr <= {PTR_WIDTH{1'b0}};
        end else if (m_rd_en) begin
            m_rd_ptr <= m_rd_ptr + 1'b1;
        end
    end

    // Write data to RAM
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_overflow <= 1'b0;
            s_wr_bad_frame <= 1'b0;
            s_wr_good_frame <= 1'b0;
        end else if (s_wr_en) begin
            ram_data[s_wr_addr] <= s_axis_tdata;
            ram_keep[s_wr_addr] <= s_axis_tkeep;
            ram_tlast[s_wr_addr] <= s_axis_tlast;
            ram_tid[s_wr_addr] <= s_axis_tid;
            ram_tdest[s_wr_addr] <= s_axis_tdest;
            ram_tuser[s_wr_addr] <= s_axis_tuser;
            
            // Overflow detection
            if (s_full && DROP_WHEN_FULL) begin
                s_wr_overflow <= 1'b1;
            end else begin
                s_wr_overflow <= 1'b0;
            end
            
            // Frame detection
            if (s_axis_tlast) begin
                // Check for bad frame
                if (USER_ENABLE && (s_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                    s_wr_bad_frame <= 1'b1;
                    s_wr_good_frame <= 1'b0;
                end else begin
                    s_wr_bad_frame <= 1'b0;
                    s_wr_good_frame <= 1'b1;
                end
            end else begin
                s_wr_bad_frame <= 1'b0;
                s_wr_good_frame <= 1'b0;
            end
        end
    end

    // Read data from RAM
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_overflow <= 1'b0;
            m_rd_bad_frame <= 1'b0;
            m_rd_good_frame <= 1'b0;
        end else if (m_rd_en) begin
            m_rd_overflow <= ram_overflow[m_rd_addr];
            m_rd_bad_frame <= ram_bad_frame[m_rd_addr];
            m_rd_good_frame <= ram_good_frame[m_rd_addr];
        end
    end

    // Width adapter: Convert S_DATA_WIDTH to M_DATA_WIDTH
    // For simplicity, we assume S_DATA_WIDTH == M_DATA_WIDTH or handle basic cases
    // In a real implementation, this would be more complex
    wire [M_DATA_WIDTH-1:0] adapted_data;
    wire [M_KEEP_WIDTH-1:0] adapted_keep;
    
    // Simple passthrough if widths match, otherwise truncate/extend
    generate
        if (S_DATA_WIDTH == M_DATA_WIDTH) begin
            assign adapted_data = ram_data[m_rd_addr];
            assign adapted_keep = ram_keep[m_rd_addr];
        end else if (S_DATA_WIDTH > M_DATA_WIDTH) begin
            assign adapted_data = ram_data[m_rd_addr][M_DATA_WIDTH-1:0];
            assign adapted_keep = ram_keep[m_rd_addr][M_KEEP_WIDTH-1:0];
        end else begin
            assign adapted_data = {ram_data[m_rd_addr], {(M_DATA_WIDTH-S_DATA_WIDTH){1'b0}}};
            assign adapted_keep = {ram_keep[m_rd_addr], {(M_KEEP_WIDTH-S_KEEP_WIDTH){1'b0}}};
        end
    endgenerate

    // Output signals
    assign m_axis_tdata = adapted_data;
    assign m_axis_tkeep = adapted_keep;
    assign m_axis_tvalid = !m_empty;
    assign m_axis_tlast = ram_tlast[m_rd_addr];
    assign m_axis_tid = ram_tid[m_rd_addr];
    assign m_axis_tdest = ram_tdest[m_rd_addr];
    assign m_axis_tuser = ram_tuser[m_rd_addr];

    // Input ready
    assign s_axis_tready = !s_full && !s_pause;

    // Pause ack
    assign s_pause_ack = s_pause;
    assign m_pause_ack = m_pause;

    // Status signals
    wire [$clog2(DEPTH):0] s_depth = s_wr_ptr - s_rd_ptr_sync3;
    wire [$clog2(DEPTH):0] m_depth = m_wr_ptr_sync3 - m_rd_ptr;

    assign s_status_depth = s_depth;
    assign s_status_depth_commit = s_depth;
    assign s_status_overflow = s_wr_overflow;
    assign s_status_bad_frame = s_wr_bad_frame;
    assign s_status_good_frame = s_wr_good_frame;

    assign m_status_depth = m_depth;
    assign m_status_depth_commit = m_depth;
    assign m_status_overflow = m_rd_overflow;
    assign m_status_bad_frame = m_rd_bad_frame;
    assign m_status_good_frame = m_rd_good_frame;

endmodule
