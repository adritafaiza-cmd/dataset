module axis_async_fifo #(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = 0,
    parameter KEEP_WIDTH = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 0,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1
)
(
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire                     s_wr_en,
    input  wire [DATA_WIDTH-1:0]    s_wr_data,
    input  wire [KEEP_WIDTH-1:0]    s_wr_keep,
    input  wire                     s_wr_tlast,
    input  wire [ID_WIDTH-1:0]      s_wr_tid,
    input  wire [DEST_WIDTH-1:0]    s_wr_tdest,
    input  wire [USER_WIDTH-1:0]    s_wr_tuser,
    output wire                     s_full,
    output wire                     s_overflow,
    output wire                     s_bad_frame,
    output wire                     s_good_frame,
    output wire [$clog2(DEPTH):0]   s_depth,
    output wire [$clog2(DEPTH):0]   s_depth_commit,
    input  wire                     m_clk,
    input  wire                     m_rst,
    input  wire                     m_rd_en,
    output wire [DATA_WIDTH-1:0]    m_rd_data,
    output wire [KEEP_WIDTH-1:0]    m_rd_keep,
    output wire                     m_rd_tlast,
    output wire [ID_WIDTH-1:0]      m_rd_tid,
    output wire [DEST_WIDTH-1:0]    m_rd_tdest,
    output wire [USER_WIDTH-1:0]    m_rd_tuser,
    output wire                     m_empty,
    output wire                     m_overflow,
    output wire                     m_bad_frame,
    output wire                     m_good_frame,
    output wire [$clog2(DEPTH):0]   m_depth,
    output wire [$clog2(DEPTH):0]   m_depth_commit
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Write pointer (S domain)
    reg [PTR_WIDTH-1:0] s_wr_ptr;
    reg [PTR_WIDTH-1:0] s_wr_ptr_gray;
    reg [PTR_WIDTH-1:0] s_wr_ptr_gray_sync;

    // Read pointer (M domain)
    reg [PTR_WIDTH-1:0] m_rd_ptr;
    reg [PTR_WIDTH-1:0] m_rd_ptr_gray;
    reg [PTR_WIDTH-1:0] m_rd_ptr_gray_sync;

    // RAM
    reg [DATA_WIDTH-1:0] ram_data [0:DEPTH-1];
    reg [KEEP_WIDTH-1:0] ram_keep [0:DEPTH-1];
    reg                  ram_tlast [0:DEPTH-1];
    reg [ID_WIDTH-1:0]   ram_tid [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] ram_tdest [0:DEPTH-1];
    reg [USER_WIDTH-1:0] ram_tuser [0:DEPTH-1];

    // Write logic (S domain)
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= {PTR_WIDTH{1'b0}};
            s_wr_ptr_gray <= {PTR_WIDTH{1'b0}};
        end else if (s_wr_en) begin
            s_wr_ptr <= s_wr_ptr + 1'b1;
            s_wr_ptr_gray <= s_wr_ptr ^ (s_wr_ptr >> 1);
        end
    end

    // Write to RAM
    always @(posedge s_clk) begin
        if (s_wr_en) begin
            ram_data[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_wr_data;
            ram_keep[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_wr_keep;
            ram_tlast[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_wr_tlast;
            ram_tid[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_wr_tid;
            ram_tdest[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_wr_tdest;
            ram_tuser[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_wr_tuser;
        end
    end

    // Read pointer (M domain)
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr <= {PTR_WIDTH{1'b0}};
            m_rd_ptr_gray <= {PTR_WIDTH{1'b0}};
        end else if (m_rd_en) begin
            m_rd_ptr <= m_rd_ptr + 1'b1;
            m_rd_ptr_gray <= m_rd_ptr ^ (m_rd_ptr >> 1);
        end
    end

    // Read from RAM
    wire [ADDR_WIDTH-1:0] m_rd_addr = m_rd_ptr[ADDR_WIDTH-1:0];
    assign m_rd_data = ram_data[m_rd_addr];
    assign m_rd_keep = ram_keep[m_rd_addr];
    assign m_rd_tlast = ram_tlast[m_rd_addr];
    assign m_rd_tid = ram_tid[m_rd_addr];
    assign m_rd_tdest = ram_tdest[m_rd_addr];
    assign m_rd_tuser = ram_tuser[m_rd_addr];

    // Synchronize write pointer to M domain
    reg [PTR_WIDTH-1:0] s_wr_ptr_gray_sync1;
    reg [PTR_WIDTH-1:0] s_wr_ptr_gray_sync2;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_wr_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            s_wr_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            s_wr_ptr_gray_sync1 <= s_wr_ptr_gray;
            s_wr_ptr_gray_sync2 <= s_wr_ptr_gray_sync1;
        end
    end

    // Synchronize read pointer to S domain
    reg [PTR_WIDTH-1:0] m_rd_ptr_gray_sync1;
    reg [PTR_WIDTH-1:0] m_rd_ptr_gray_sync2;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_rd_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            m_rd_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            m_rd_ptr_gray_sync1 <= m_rd_ptr_gray;
            m_rd_ptr_gray_sync2 <= m_rd_ptr_gray_sync1;
        end
    end

    // Convert gray to binary
    function [PTR_WIDTH-1:0] gray_to_binary;
        input [PTR_WIDTH-1:0] gray;
        integer i;
        begin
            gray_to_binary[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1)
                gray_to_binary[i] = gray_to_binary[i+1] ^ gray[i];
        end
    endfunction

    wire [PTR_WIDTH-1:0] s_wr_ptr_bin_m = gray_to_binary(s_wr_ptr_gray_sync2);
    wire [PTR_WIDTH-1:0] m_rd_ptr_bin_s = gray_to_binary(m_rd_ptr_gray_sync2);

    // Full and empty
    assign s_full = (s_wr_ptr_gray_sync2 == m_rd_ptr_gray_sync2) && (s_wr_ptr[PTR_WIDTH-1] != m_rd_ptr_bin_s[PTR_WIDTH-1]);
    assign m_empty = (s_wr_ptr_gray_sync2 == m_rd_ptr_gray_sync2) && (m_rd_ptr[PTR_WIDTH-1] != s_wr_ptr_bin_m[PTR_WIDTH-1]);

    // Depth calculation
    wire [$clog2(DEPTH):0] s_depth_calc = s_wr_ptr - m_rd_ptr_bin_s;
    wire [$clog2(DEPTH):0] m_depth_calc = s_wr_ptr_bin_m - m_rd_ptr;

    assign s_depth = s_depth_calc;
    assign m_depth = m_depth_calc;
    assign s_depth_commit = s_depth_calc;
    assign m_depth_commit = m_depth_calc;

    // Status signals (simplified)
    assign s_overflow = 1'b0;
    assign m_overflow = 1'b0;
    assign s_bad_frame = 1'b0;
    assign m_bad_frame = 1'b0;
    assign s_good_frame = 1'b0;
    assign m_good_frame = 1'b0;

endmodule
