```verilog
module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Internal signals
    wire s_wr_en;
    wire m_rd_en;
    wire s_full;
    wire m_empty;
    wire s_overflow;
    wire m_overflow;
    wire s_bad_frame;
    wire m_bad_frame;
    wire s_good_frame;
    wire m_good_frame;
    wire s_pause_active;
    wire m_pause_active;

    // Synchronous FIFO with async read/write
    wire [S_DATA_WIDTH-1:0] fifo_rd_data;
    wire [S_KEEP_WIDTH-1:0] fifo_rd_keep;
    wire fifo_rd_tlast;
    wire [ID_WIDTH-1:0] fifo_rd_tid;
    wire [DEST_WIDTH-1:0] fifo_rd_tdest;
    wire [USER_WIDTH-1:0] fifo_rd_tuser;

    // Width adapter signals
    wire [M_DATA_WIDTH-1:0] adapter_tdata;
    wire [M_KEEP_WIDTH-1:0] adapter_tkeep;
    wire adapter_tlast;
    wire [ID_WIDTH-1:0] adapter_tid;
    wire [DEST_WIDTH-1:0] adapter_tdest;
    wire [USER_WIDTH-1:0] adapter_tuser;
    wire adapter_valid;
    wire adapter_ready;

    // Output FIFO signals
    wire [M_DATA_WIDTH-1:0] out_fifo_rd_data;
    wire [M_KEEP_WIDTH-1:0] out_fifo_rd_keep;
    wire out_fifo_rd_tlast;
    wire [ID_WIDTH-1:0] out_fifo_rd_tid;
    wire [DEST_WIDTH-1:0] out_fifo_rd_tdest;
    wire [USER_WIDTH-1:0] out_fifo_rd_tuser;
    wire out_fifo_valid;
    wire out_fifo_ready;
    wire out_fifo_empty;
    wire out_fifo_full;

    // Synchronous FIFO core
    wire [ADDR_WIDTH-1:0] s_wr_ptr;
    wire [ADDR_WIDTH-1:0] m_rd_ptr;
    wire [PTR_WIDTH-1:0] s_wr_ptr_sync;
    wire [PTR_WIDTH-1:0] m_rd_ptr_sync;

    // Write side (s_clk domain)
    wire s_wr_en_internal = s_axis_tvalid & s_axis_tready & ~s_pause_active;
    assign s_wr_en = s_wr_en_internal;

    // Read side (m_clk domain)
    wire m_rd_en_internal = m_axis_tvalid & m_axis_tready & ~m_pause_active;
    assign m_rd_en = m_rd_en_internal;

    // Synchronous FIFO implementation
    reg [S_DATA_WIDTH-1:0] mem [0:DEPTH-1];
    reg [S_KEEP_WIDTH-1:0] mem_keep [0:DEPTH-1];
    reg mem_tlast [0:DEPTH-1];
    reg [ID_WIDTH-1:0] mem_tid [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] mem_tdest [0:DEPTH-1];
    reg [USER_WIDTH-1:0] mem_tuser [0:DEPTH-1];

    reg [PTR_WIDTH-1:0] wr_ptr;
    reg [PTR_WIDTH-1:0] rd_ptr;

    // Write pointer
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            wr_ptr <= {PTR_WIDTH{1'b0}};
        else if (s_wr_en)
            wr_ptr <= wr_ptr + 1'b1;
    end

    // Read pointer
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            rd_ptr <= {PTR_WIDTH{1'b0}};
        else if (m_rd_en)
            rd_ptr <= rd_ptr + 1'b1;
    end

    // Write data
    always @(posedge s_clk) begin
        if (s_wr_en) begin
            mem[wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tdata;
            mem_keep[wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tkeep;
            mem_tlast[wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tlast;
            mem_tid[wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tid;
            mem_tdest[wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tdest;
            mem_tuser[wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tuser;
        end
    end

    // Read data
    always @(posedge m_clk) begin
        if (m_rd_en) begin
            fifo_rd_data <= mem[rd_ptr[ADDR_WIDTH-1:0]];
            fifo_rd_keep <= mem_keep[rd_ptr[ADDR_WIDTH-1:0]];
            fifo_rd_tlast <= mem_tlast[rd_ptr[ADDR_WIDTH-1:0]];
            fifo_rd_tid <= mem_tid[rd_ptr[ADDR_WIDTH-1:0]];
            fifo_rd_tdest <= mem_tdest[rd_ptr[ADDR_WIDTH-1:0]];
            fifo_rd_tuser <= mem_tuser[rd_ptr[ADDR_WIDTH-1:0]];
        end
    end

    // Synchronize write pointer to m_clk domain
    reg [PTR_WIDTH-1:0] wr_ptr_sync1, wr_ptr_sync2;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            wr_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            wr_ptr_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            wr_ptr_sync1 <= wr_ptr;
            wr_ptr_sync2 <= wr_ptr_sync1;
        end
    end
    assign s_wr_ptr_sync = wr_ptr_sync2;

    // Synchronize read pointer to s_clk domain
    reg [PTR_WIDTH-1:0] rd_ptr_sync1, rd_ptr_sync2;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            rd_ptr_sync1 <= {PTR_WIDTH{1'b0}};
            rd_ptr_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            rd_ptr_sync1 <= rd_ptr;
            rd_ptr_sync2 <= rd_ptr_sync1;
        end
    end
    assign m_rd_ptr_sync = rd_ptr_sync2;

    // Full and empty detection
    wire [PTR_WIDTH-1:0] wr_ptr_m = wr_ptr_sync2;
    wire [PTR_WIDTH-1:0] rd_ptr_s = rd_ptr_sync2;

    assign s_full = (wr_ptr[ADDR_WIDTH-1:0] == rd_ptr_s[ADDR_WIDTH-1:0]) &&
                    (wr_ptr[ADDR_WIDTH] != rd_ptr_s[ADDR_WIDTH]);
    assign m_empty = (rd_ptr == wr_ptr_m);

    // Overflow detection
    reg s_overflow_reg;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_overflow_reg <= 1'b0;
        else if (s_axis_tvalid & s_full & ~s_axis_tready)
            s_overflow_reg <= 1'b1;
    end
    assign s_overflow = s_overflow_reg;

    reg m_overflow_reg;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_overflow_reg <= 1'b0;
        else if (m_axis_tvalid & m_empty & ~m_axis_tready)
            m_overflow_reg <= 1'b1;
    end
    assign m_overflow = m_overflow_reg;

    // Bad frame detection
    reg s_bad_frame_reg;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_bad_frame_reg <= 1'b0;
        else if (s_wr_en & (s_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE)
            s_bad_frame_reg <= 1'b1;
    end
    assign s_bad_frame = s_bad_frame_reg;

    reg m_bad_frame_reg;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_bad_frame_reg <= 1'b0;
        else if (m_rd_en & (fifo_rd_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE)
            m_bad_frame_reg <= 1'b1;
    end
    assign m_bad_frame = m_bad_frame_reg;

    // Good frame detection
    reg s_good_frame_reg;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_good_frame_reg <= 1'b0;
        else if (s_wr_en & s_axis_tlast & ~s_bad_frame)
            s_good_frame_reg <= 1'b1;
    end
    assign s_good_frame = s_good_frame_reg;

    reg m_good_frame_reg;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_good_frame_reg <= 1'b0;
        else if (m_rd_en & fifo_rd_tlast & ~m_bad_frame)
            m_good_frame_reg <= 1'b1;
    end
    assign m_good_frame = m_good_frame_reg;

    // Pause logic
    reg s_pause_ack_reg;
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_pause_ack_reg <= 1'b0;
        else if (s_pause_req)
            s_pause_ack_reg <= 1'b1;
        else
            s_pause_ack_reg <= 1'b0;
    end
    assign s_pause_ack = s_pause_ack_reg;
    assign s_pause_active = s_pause_req & s_pause_ack;

    reg m_pause_ack_reg;
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_pause_ack_reg <= 1'b0;
        else if (m_pause_req)
            m_pause_ack_reg <= 1'b1;
        else
            m_pause_ack_reg <= 1'b0;
    end
    assign m_pause_ack = m_pause_ack_reg;
    assign m_pause_active = m_pause_req & m_pause_ack;

    // Width adapter
    // For simplicity, assume S_DATA_WIDTH == M_DATA_WIDTH or handle basic cases
    // In a real implementation, this would be more complex
    assign adapter_tdata = fifo_rd_data[M_DATA_WIDTH-1:0];
    assign adapter_tkeep = fifo_rd_keep[M_KEEP_WIDTH-1:0];
    assign adapter_tlast = fifo_rd_tlast;
    assign adapter_tid = fifo_rd_tid;
    assign adapter_tdest = fifo_rd_tdest;
    assign adapter_tuser = fifo_rd_tuser;
    assign adapter_valid = ~m_empty;
    assign adapter_ready = m_axis_tready & ~m_pause_active;

    // Output FIFO
    if (OUTPUT_FIFO_ENABLE) begin
        // Simple output FIFO
        reg [M_DATA_WIDTH-1:0] out_mem [0:DEPTH-1];
        reg [M_KEEP_WIDTH-1:0] out_mem_keep [0:DEPTH-1];
        reg out_mem_tlast [0:DEPTH-1];
        reg [ID_WIDTH-1:0] out_mem_tid [0:DEPTH-1];
        reg [DEST_WIDTH-1:0] out_mem_tdest [0:DEPTH-1];
        reg [USER_WIDTH-1:0] out_mem_tuser [0:DEPTH-1];

        reg [PTR_WIDTH-1:0] out_wr_ptr;
        reg [PTR_WIDTH-1:0] out_rd_ptr;

        wire out_wr_en = adapter_valid & adapter_ready;
        wire out_rd_en = m_axis_tvalid & m_axis_tready & ~m_pause_active;

        always @(posedge m_clk or posedge m_rst) begin
            if (m_rst) begin
                out_wr_ptr <= {PTR_WIDTH{1'b0}};
                out_rd_ptr <= {PTR_WIDTH{1'b0}};
            end else begin
                if (out_wr_en)
                    out_wr_ptr <= out_wr_ptr + 1'b1;
                if (out_rd_en)
                    out_rd_ptr <= out_rd_ptr + 1'b1;
            end
        end

        always @(posedge m_clk) begin
            if (out_wr_en) begin
                out_mem[out_wr_ptr[ADDR_WIDTH-1:0]] <= adapter_tdata;
                out_mem_keep[out_wr_ptr[ADDR_WIDTH-1:0]] <= adapter_tkeep;
                out_mem_tlast[out_wr_ptr[ADDR_WIDTH-1:0]] <= adapter_tlast;
                out_mem_tid[out_wr_ptr[ADDR_WIDTH-1:0]] <= adapter_tid;
                out_mem_tdest[out_wr_ptr[ADDR_WIDTH-1:0]] <= adapter_tdest;
                out_mem_tuser[out_wr_ptr[ADDR_WIDTH-1:0]] <= adapter_tuser;
            end
        end

        always @(posedge m_clk) begin
            if (out_rd_en) begin
                out_fifo_rd_data <= out_mem[out_rd_ptr[ADDR_WIDTH-1:0]];
                out_fifo_rd_keep <= out_mem_keep[out_rd_ptr[ADDR_WIDTH-1:0]];
                out_fifo_rd_tlast <= out_mem_tlast[out_rd_ptr[ADDR_WIDTH-1:0]];
                out_fifo_rd_tid <= out_mem_tid[out_rd_ptr[ADDR_WIDTH-1:0]];
                out_fifo_rd_tdest <= out_mem_tdest[out_rd_ptr[ADDR_WIDTH-1:0]];
                out_fifo_rd_tuser <= out_mem_tuser[out_rd_ptr[ADDR_WIDTH-1:0]];
            end
        end

        assign out_fifo_empty = (out_rd_ptr == out_wr_ptr);
        assign out_fifo_full = (out_wr_ptr[ADDR_WIDTH-1:0] == out_rd_ptr[ADDR_WIDTH-1:0]) &&
                               (out_wr_ptr[ADDR_WIDTH] != out_rd_ptr[ADDR_WIDTH]);
        assign out_fifo_valid = ~out_fifo_empty;
        assign out_fifo_ready = m_axis_tready & ~m_pause_active;
    end else begin
        assign out_fifo_rd_data = adapter_tdata;
        assign out_fifo_rd_keep = adapter_tkeep;
        assign out_fifo_rd_tlast = adapter_tlast;
        assign out_fifo_rd_tid = adapter_tid;
        assign out_fifo_rd_tdest = adapter_tdest;
        assign out_fifo_rd_tuser = adapter_tuser;
        assign out_fifo_valid = adapter_valid;
        assign out_fifo_ready = adapter_ready;
        assign out_fifo_empty = m_empty;
        assign out_fifo_full = 1'b0;
    end

    // Output signals
    assign m_axis_tdata = out_fifo_rd_data;
    assign m_axis_tkeep = out_fifo_rd_keep;
    assign m_axis_tvalid = out_fifo_valid;
    assign m_axis_tlast = out_fifo_rd_tlast;
    assign m_axis_tid = out_fifo_rd_tid;
    assign m_axis_tdest = out_fifo_rd_tdest;
    assign m_axis_tuser = out_fifo_rd_tuser;

    // Input ready
    assign s_axis_tready = ~s_full & ~s_pause_active;

    // Status depth
    wire [PTR_WIDTH-1:0] depth_s = wr_ptr - rd_ptr_s;
    wire [PTR_WIDTH-1:0] depth_m = wr_ptr_m - rd_ptr;
    assign s_status_depth = depth_s[ADDR_WIDTH:0];
    assign s_status_depth_commit = depth_s[ADDR_WIDTH:0];
    assign m_status_depth = depth_m[ADDR_WIDTH:0];
    assign m_status_depth_commit = depth_m[ADDR_WIDTH:0];

endmodule
