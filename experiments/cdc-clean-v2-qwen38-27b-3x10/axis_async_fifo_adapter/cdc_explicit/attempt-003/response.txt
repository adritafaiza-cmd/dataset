```verilog
module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);
    localparam PTR_WIDTH = ADDR_WIDTH + 1;
    localparam DEPTH_MASK = DEPTH - 1;

    // Internal signals
    wire [S_DATA_WIDTH-1:0] s_data_reg;
    wire [S_KEEP_WIDTH-1:0] s_keep_reg;
    wire s_last_reg;
    wire [ID_WIDTH-1:0] s_id_reg;
    wire [DEST_WIDTH-1:0] s_dest_reg;
    wire [USER_WIDTH-1:0] s_user_reg;

    // Source side FIFO control
    reg [PTR_WIDTH-1:0] s_wr_ptr;
    reg [PTR_WIDTH-1:0] s_rd_ptr_sync;
    reg [PTR_WIDTH-1:0] s_rd_ptr_commit;
    reg s_overflow;
    reg s_bad_frame;
    reg s_good_frame;
    reg s_pause_ack_reg;

    // Destination side FIFO control
    reg [PTR_WIDTH-1:0] m_rd_ptr;
    reg [PTR_WIDTH-1:0] m_wr_ptr_sync;
    reg [PTR_WIDTH-1:0] m_wr_ptr_commit;
    reg m_overflow;
    reg m_bad_frame;
    reg m_good_frame;
    reg m_pause_ack_reg;

    // RAM storage
    reg [S_DATA_WIDTH-1:0] ram_data [0:DEPTH-1];
    reg [S_KEEP_WIDTH-1:0] ram_keep [0:DEPTH-1];
    reg ram_last [0:DEPTH-1];
    reg [ID_WIDTH-1:0] ram_id [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] ram_dest [0:DEPTH-1];
    reg [USER_WIDTH-1:0] ram_user [0:DEPTH-1];

    // Read data from RAM
    wire [S_DATA_WIDTH-1:0] rd_data;
    wire [S_KEEP_WIDTH-1:0] rd_keep;
    wire rd_last;
    wire [ID_WIDTH-1:0] rd_id;
    wire [DEST_WIDTH-1:0] rd_dest;
    wire [USER_WIDTH-1:0] rd_user;

    assign rd_data = ram_data[m_rd_ptr[ADDR_WIDTH-1:0]];
    assign rd_keep = ram_keep[m_rd_ptr[ADDR_WIDTH-1:0]];
    assign rd_last = ram_last[m_rd_ptr[ADDR_WIDTH-1:0]];
    assign rd_id = ram_id[m_rd_ptr[ADDR_WIDTH-1:0]];
    assign rd_dest = ram_dest[m_rd_ptr[ADDR_WIDTH-1:0]];
    assign rd_user = ram_user[m_rd_ptr[ADDR_WIDTH-1:0]];

    // Width adapter: S_DATA_WIDTH to M_DATA_WIDTH
    // For simplicity, we assume S_DATA_WIDTH == M_DATA_WIDTH or handle basic cases
    // In a real implementation, this would be more complex
    wire [M_DATA_WIDTH-1:0] m_data_out;
    wire [M_KEEP_WIDTH-1:0] m_keep_out;
    wire m_last_out;
    wire [ID_WIDTH-1:0] m_id_out;
    wire [DEST_WIDTH-1:0] m_dest_out;
    wire [USER_WIDTH-1:0] m_user_out;

    // Simple width adapter - pass through if widths match, otherwise truncate/extend
    generate
        if (S_DATA_WIDTH == M_DATA_WIDTH) begin : gen_width_match
            assign m_data_out = rd_data;
            assign m_keep_out = rd_keep;
        end
        else if (S_DATA_WIDTH > M_DATA_WIDTH) begin : gen_s_wider
            assign m_data_out = rd_data[M_DATA_WIDTH-1:0];
            assign m_keep_out = rd_keep[M_KEEP_WIDTH-1:0];
        end
        else begin : gen_m_wider
            assign m_data_out = { {(M_DATA_WIDTH-S_DATA_WIDTH){1'b0}}, rd_data };
            assign m_keep_out = { {(M_KEEP_WIDTH-S_KEEP_WIDTH){1'b0}}, rd_keep };
        end
    endgenerate

    assign m_last_out = rd_last;
    assign m_id_out = rd_id;
    assign m_dest_out = rd_dest;
    assign m_user_out = rd_user;

    // Handshake signals
    wire s_write_en;
    wire m_read_en;

    assign s_write_en = s_axis_tvalid && s_axis_tready;
    assign m_read_en = m_axis_tvalid && m_axis_tready;

    // FIFO full/empty logic
    wire s_fifo_full;
    wire m_fifo_empty;
    wire s_fifo_empty;
    wire m_fifo_full;

    assign s_fifo_full = (s_wr_ptr - s_rd_ptr_sync) == DEPTH;
    assign m_fifo_empty = (m_rd_ptr == m_wr_ptr_sync);
    assign s_fifo_empty = (s_wr_ptr == s_rd_ptr_commit);
    assign m_fifo_full = (m_wr_ptr_sync - m_rd_ptr) == DEPTH;

    // Ready signals
    wire s_ready_internal;
    wire m_valid_internal;

    assign s_ready_internal = !s_fifo_full && (!s_pause_ack_reg || !s_pause_req);
    assign m_valid_internal = !m_fifo_empty && (!m_pause_ack_reg || !m_pause_req);

    assign s_axis_tready = s_ready_internal;
    assign m_axis_tvalid = m_valid_internal;

    // Output data
    assign m_axis_tdata = m_data_out;
    assign m_axis_tkeep = m_keep_out;
    assign m_axis_tlast = m_last_out;
    assign m_axis_tid = m_id_out;
    assign m_axis_tdest = m_dest_out;
    assign m_axis_tuser = m_user_out;

    // Pause ack
    assign s_pause_ack = s_pause_ack_reg;
    assign m_pause_ack = m_pause_ack_reg;

    // Status depth
    wire [PTR_WIDTH-1:0] s_depth;
    wire [PTR_WIDTH-1:0] m_depth;

    assign s_depth = s_wr_ptr - s_rd_ptr_sync;
    assign m_depth = m_wr_ptr_sync - m_rd_ptr;

    assign s_status_depth = s_depth;
    assign s_status_depth_commit = s_wr_ptr - s_rd_ptr_commit;
    assign m_status_depth = m_depth;
    assign m_status_depth_commit = m_wr_ptr_commit - m_rd_ptr;

    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;

    // Source clock domain logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= {PTR_WIDTH{1'b0}};
            s_rd_ptr_sync <= {PTR_WIDTH{1'b0}};
            s_rd_ptr_commit <= {PTR_WIDTH{1'b0}};
            s_overflow <= 1'b0;
            s_bad_frame <= 1'b0;
            s_good_frame <= 1'b0;
            s_pause_ack_reg <= 1'b0;
        end else begin
            // Write to RAM
            if (s_write_en) begin
                ram_data[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tdata;
                ram_keep[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tkeep;
                ram_last[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tlast;
                ram_id[s_wr_ptr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tid;
                ram_dest[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tdest;
                ram_user[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tuser;
                s_wr_ptr <= s_wr_ptr + 1'b1;
            end

            // Update read pointer from destination domain
            s_rd_ptr_sync <= m_rd_ptr;
            s_rd_ptr_commit <= m_rd_ptr;

            // Overflow detection
            if (s_write_en && s_fifo_full) begin
                s_overflow <= 1'b1;
            end

            // Frame status
            if (s_write_en && s_axis_tlast) begin
                s_good_frame <= 1'b1;
            end
        end
    end

    // Destination clock domain logic
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr <= {PTR_WIDTH{1'b0}};
            m_wr_ptr_sync <= {PTR_WIDTH{1'b0}};
            m_wr_ptr_commit <= {PTR_WIDTH{1'b0}};
            m_overflow <= 1'b0;
            m_bad_frame <= 1'b0;
            m_good_frame <= 1'b0;
            m_pause_ack_reg <= 1'b0;
        end else begin
            // Read from RAM
            if (m_read_en) begin
                m_rd_ptr <= m_rd_ptr + 1'b1;
            end

            // Update write pointer from source domain
            m_wr_ptr_sync <= s_wr_ptr;
            m_wr_ptr_commit <= s_wr_ptr;

            // Overflow detection
            if (m_read_en && m_fifo_full) begin
                m_overflow <= 1'b1;
            end

            // Frame status
            if (m_read_en && m_last_out) begin
                m_good_frame <= 1'b1;
            end
        end
    end

    // Pause logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_pause_ack_reg <= 1'b0;
        end else if (PAUSE_ENABLE) begin
            s_pause_ack_reg <= s_pause_req;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_pause_ack_reg <= 1'b0;
        end else if (PAUSE_ENABLE) begin
            m_pause_ack_reg <= m_pause_req;
        end
    end

endmodule
