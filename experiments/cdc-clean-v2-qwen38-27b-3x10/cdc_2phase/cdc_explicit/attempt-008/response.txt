module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain: hold the payload and issue a request
  reg [WIDTH-1:0] src_payload_reg;
  reg             src_req_reg;

  // Destination domain: acknowledge the request
  reg             dst_ack_reg;

  // Synchronize the request from source to destination
  reg             req_sync1;
  reg             req_sync2;

  // Synchronize the acknowledge from destination to source
  reg             ack_sync1;
  reg             ack_sync2;

  // Destination domain: hold the payload and synchronize it
  reg [WIDTH-1:0] dst_payload_sync1;
  reg [WIDTH-1:0] dst_payload_sync2;

  // Destination domain: valid signal derived from synchronized request
  reg             dst_valid_reg;

  // Source ready: high when no outstanding request
  assign src_ready_o = ~src_req_reg;

  // Destination valid: high when a request has been synchronized and not yet acknowledged
  assign dst_valid_o = dst_valid_reg;

  // Destination data: from the synchronized payload
  assign dst_data_o = dst_payload_sync2;

  // Source domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_req_reg     <= 1'b0;
      src_payload_reg <= {WIDTH{1'b0}};
    end else begin
      if (src_valid_i && src_ready_o) begin
        src_req_reg     <= 1'b1;
        src_payload_reg <= src_data_i;
      end else if (src_req_reg && ack_sync2) begin
        src_req_reg <= 1'b0;
      end
    end
  end

  // Synchronize request to destination domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      req_sync1 <= 1'b0;
      req_sync2 <= 1'b0;
    end else begin
      req_sync1 <= src_req_reg;
      req_sync2 <= req_sync1;
    end
  end

  // Destination domain: capture payload when request is synchronized
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_payload_sync1 <= {WIDTH{1'b0}};
      dst_payload_sync2 <= {WIDTH{1'b0}};
      dst_valid_reg     <= 1'b0;
      dst_ack_reg       <= 1'b0;
    end else begin
      // Capture payload when a new request arrives (req_sync2 rising edge)
      if (req_sync2 && !req_sync1) begin
        dst_payload_sync1 <= src_payload_reg;
        dst_valid_reg     <= 1'b1;
      end

      // Synchronize payload
      dst_payload_sync2 <= dst_payload_sync1;

      // Acknowledge when valid and ready
      if (dst_valid_reg && dst_ready_i) begin
        dst_ack_reg   <= 1'b1;
        dst_valid_reg <= 1'b0;
      end
    end
  end

  // Synchronize acknowledge to source domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      ack_sync1 <= 1'b0;
      ack_sync2 <= 1'b0;
    end else begin
      ack_sync1 <= dst_ack_reg;
      ack_sync2 <= ack_sync1;
    end
  end

endmodule
