module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain: hold the data and generate request
  reg [WIDTH-1:0] src_data_hold;
  reg             src_req;
  wire            src_ack;

  // Destination domain: synchronize request, capture data, generate ack
  reg             dst_req_sync1;
  reg             dst_req_sync2;
  reg             dst_ack;
  reg [WIDTH-1:0] dst_data_reg;
  reg             dst_valid_reg;

  // Synchronize ack from dst to src
  reg             src_ack_sync1;
  reg             src_ack_sync2;

  // Synchronize dst_ready from dst to src (for backpressure)
  // Actually, we need to know when dst is ready to accept a new item.
  // The protocol: src sends req, dst captures data, dst sends ack.
  // src can send next req only after receiving ack.
  // dst_valid_o is high when dst has a valid item and is waiting for dst_ready_i.
  // dst_valid_o must remain high until dst_ready_i is high.

  // Let's use a simple handshake:
  // src: when src_valid_i && src_ready_o, latch src_data_i into src_data_hold, set src_req=1.
  // src_ready_o = !src_req || src_ack_sync2 (i.e., can accept new if no outstanding req, or previous ack received)
  // Actually, src_ready_o should be low when src_req is high and ack not yet received.
  // src_ready_o = !src_req || src_ack_sync2;
  // But we need to be careful: when src_req is set, we wait for ack.
  // When ack arrives, we clear src_req and can accept new.

  // dst: when dst_req_sync2 rises, capture src_data_hold (synchronized) into dst_data_reg, set dst_valid_reg=1.
  // dst_valid_o = dst_valid_reg.
  // When dst_valid_reg && dst_ready_i, clear dst_valid_reg, set dst_ack=1.
  // dst_ack is then synchronized back to src.
  // When src receives ack, clear src_req.

  // But wait: the data must be stable during the request. So src_data_hold must be stable while src_req is high.
  // The data is synchronized through 2 stages in dst domain.

  // Let's implement:

  // Source domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_hold <= {WIDTH{1'b0}};
      src_req       <= 1'b0;
    end else begin
      if (src_valid_i && src_ready_o) begin
        src_data_hold <= src_data_i;
        src_req       <= 1'b1;
      end else if (src_ack_sync2) begin
        src_req       <= 1'b0;
      end
    end
  end

  assign src_ready_o = !src_req || src_ack_sync2;

  // Synchronize ack from dst to src
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ack_sync1 <= 1'b0;
      src_ack_sync2 <= 1'b0;
    end else begin
      src_ack_sync1 <= dst_ack;
      src_ack_sync2 <= src_ack_sync1;
    end
  end

  // Destination domain
  // Synchronize req from src to dst
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_req_sync1 <= 1'b0;
      dst_req_sync2 <= 1'b0;
    end else begin
      dst_req_sync1 <= src_req;
      dst_req_sync2 <= dst_req_sync1;
    end
  end

  // Synchronize data from src to dst (2 stages)
  reg [WIDTH-1:0] dst_data_sync1;
  reg [WIDTH-1:0] dst_data_sync2;

  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_sync1 <= {WIDTH{1'b0}};
      dst_data_sync2 <= {WIDTH{1'b0}};
    end else begin
      dst_data_sync1 <= src_data_hold;
      dst_data_sync2 <= dst_data_sync1;
    end
  end

  // Detect rising edge of dst_req_sync2
  wire dst_req_rise = dst_req_sync2 && !dst_req_sync1;

  // Destination logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg  <= {WIDTH{1'b0}};
      dst_valid_reg <= 1'b0;
      dst_ack       <= 1'b0;
    end else begin
      if (dst_req_rise) begin
        dst_data_reg  <= dst_data_sync2;
        dst_valid_reg <= 1'b1;
        dst_ack       <= 1'b0;
      end else if (dst_valid_reg && dst_ready_i) begin
        dst_valid_reg <= 1'b0;
        dst_ack       <= 1'b1;
      end
    end
  end

  assign dst_data_o  = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

endmodule
