module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain: hold data when a transfer is accepted
  reg [WIDTH-1:0] src_data_hold;
  reg             src_data_valid;
  reg             src_ack_sync0;
  reg             src_ack_sync1;

  // Destination domain: synchronized request and data
  reg             dst_req_sync0;
  reg             dst_req_sync1;
  reg             dst_req_sync2;
  reg [WIDTH-1:0] dst_data_sync0;
  reg [WIDTH-1:0] dst_data_sync1;
  reg [WIDTH-1:0] dst_data_sync2;
  reg             dst_ack;

  // Source domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_hold  <= {WIDTH{1'b0}};
      src_data_valid <= 1'b0;
      src_ack_sync0  <= 1'b0;
      src_ack_sync1  <= 1'b0;
    end else begin
      // Synchronize ack from destination
      src_ack_sync0 <= dst_ack;
      src_ack_sync1 <= src_ack_sync0;

      // Accept new data when valid and ready
      if (src_valid_i && src_ready_o) begin
        src_data_hold  <= src_data_i;
        src_data_valid <= 1'b1;
      end else if (src_data_valid && src_ack_sync1) begin
        // Ack received, clear valid
        src_data_valid <= 1'b0;
      end
    end
  end

  // Source ready: low when we have an outstanding item not yet acked
  assign src_ready_o = !src_data_valid || src_ack_sync1;

  // Destination domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_req_sync0  <= 1'b0;
      dst_req_sync1  <= 1'b0;
      dst_req_sync2  <= 1'b0;
      dst_data_sync0 <= {WIDTH{1'b0}};
      dst_data_sync1 <= {WIDTH{1'b0}};
      dst_data_sync2 <= {WIDTH{1'b0}};
      dst_ack        <= 1'b0;
    end else begin
      // Synchronize request from source
      dst_req_sync0 <= src_data_valid;
      dst_req_sync1 <= dst_req_sync0;
      dst_req_sync2 <= dst_req_sync1;

      // Synchronize data (hold stable during request)
      dst_data_sync0 <= src_data_hold;
      dst_data_sync1 <= dst_data_sync0;
      dst_data_sync2 <= dst_data_sync1;

      // Acknowledge when request is fully synchronized and destination is ready
      if (dst_req_sync2 && dst_ready_i) begin
        dst_ack <= 1'b1;
      end else if (!dst_req_sync2) begin
        dst_ack <= 1'b0;
      end
    end
  end

  // Destination outputs
  assign dst_valid_o = dst_req_sync2;
  assign dst_data_o  = dst_data_sync2;

endmodule
