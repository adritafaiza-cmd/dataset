module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source-side state
  reg src_valid_r;
  reg [WIDTH-1:0] src_data_r;

  // Destination-side state
  reg dst_valid_r;
  reg [WIDTH-1:0] dst_data_r;

  // Handshake signals
  // src_to_dst_valid: source has data to send
  // dst_to_src_ready: destination is ready to accept (i.e., not holding pending data)

  // Source side: track if we have pending data
  // We use a simple approach:
  // - src_valid_r indicates we have data to send
  // - When dst is ready (dst_to_src_ready), we can transfer

  // Let's use a 2-phase handshake with proper CDC
  // Source side:
  //   - If src_valid_i and src_ready_o, capture data
  //   - src_ready_o is high when we don't have pending data OR when dst accepts

  // Actually, let me think about this more carefully.
  // We need to transfer data from src to dst across clock domains.
  // The data must be stable while valid.

  // Approach:
  // 1. Source captures data when src_valid_i && src_ready_o
  // 2. Source holds the data and asserts a "valid" signal to dst domain
  // 3. Dst domain sees valid, and when dst_ready_i is high, it accepts the data
  // 4. Dst domain sends back an "ack" to src domain
  // 5. Src domain deasserts valid when it sees ack

  // This is a standard 2-phase handshake CDC.

  // Source side registers
  reg src_valid_r;
  reg [WIDTH-1:0] src_data_r;

  // Destination side registers
  reg dst_valid_r;
  reg [WIDTH-1:0] dst_data_r;

  // Handshake signals (synchronized)
  // src_valid_to_dst: source has data
  // dst_ack_to_src: destination has accepted data

  // Let's define the signals:
  // src_valid_r: source has pending data
  // dst_ack_r: destination has acknowledged (in src domain)

  // Source domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_valid_r <= 1'b0;
      src_data_r  <= {WIDTH{1'b0}};
    end else begin
      if (src_valid_i && src_ready_o) begin
        src_valid_r <= 1'b1;
        src_data_r  <= src_data_i;
      end else if (dst_ack_r) begin
        src_valid_r <= 1'b0;
      end
    end
  end

  // src_ready_o: high when we can accept new data
  // We can accept new data if we don't have pending data
  assign src_ready_o = ~src_valid_r;

  // dst_ack_r: synchronized ack from dst domain
  reg dst_ack_r;
  reg dst_ack_sync1;
  reg dst_ack_sync2;

  // dst_ack from dst domain
  wire dst_ack;

  // Destination domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid_r <= 1'b0;
      dst_data_r  <= {WIDTH{1'b0}};
      dst_ack     <= 1'b0;
    end else begin
      if (src_valid_sync2) begin
        dst_valid_r <= 1'b1;
        dst_data_r  <= src_data_sync2;
      end else if (dst_valid_r && dst_ready_i) begin
        dst_valid_r <= 1'b0;
        dst_ack     <= 1'b1;
      end else if (!dst_valid_r) begin
        dst_ack     <= 1'b0;
      end
    end
  end

  // Synchronize src_valid and src_data to dst domain
  reg src_valid_sync1;
  reg src_valid_sync2;
  reg [WIDTH-1:0] src_data_sync1;
  reg [WIDTH-1:0] src_data_sync2;

  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_valid_sync1 <= 1'b0;
      src_valid_sync2 <= 1'b0;
      src_data_sync1  <= {WIDTH{1'b0}};
      src_data_sync2  <= {WIDTH{1'b0}};
    end else begin
      src_valid_sync1 <= src_valid_r;
      src_valid_sync2 <= src_valid_sync1;
      src_data_sync1  <= src_data_r;
      src_data_sync2  <= src_data_sync1;
    end
  end

  // Synchronize dst_ack to src domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_ack_sync1 <= 1'b0;
      dst_ack_sync2 <= 1'b0;
      dst_ack_r     <= 1'b0;
    end else begin
      dst_ack_sync1 <= dst_ack;
      dst_ack_sync2 <= dst_ack_sync1;
      dst_ack_r     <= dst_ack_sync2;
    end
  end

  // Outputs
  assign dst_data_o  = dst_data_r;
  assign dst_valid_o = dst_valid_r;

endmodule
