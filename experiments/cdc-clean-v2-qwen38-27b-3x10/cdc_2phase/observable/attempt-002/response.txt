module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source-side state
  reg src_pending;
  reg [WIDTH-1:0] src_data_reg;
  reg src_sent;

  // Destination-side state
  reg dst_pending;
  reg [WIDTH-1:0] dst_data_reg;
  reg dst_sent;

  // Synchronizers for src_sent (src -> dst)
  reg src_sent_sync1;
  reg src_sent_sync2;

  // Synchronizers for dst_sent (dst -> src)
  reg dst_sent_sync1;
  reg dst_sent_sync2;

  // Synchronizers for dst_pending (dst -> src)
  reg dst_pending_sync1;
  reg dst_pending_sync2;

  // Synchronizers for src_pending (src -> dst)
  reg src_pending_sync1;
  reg src_pending_sync2;

  // Source clock domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_pending <= 1'b0;
      src_data_reg <= {WIDTH{1'b0}};
      src_sent <= 1'b0;
    end else begin
      // If we have a pending item and the destination has consumed the previous one
      // (dst_sent_sync2 indicates dst has finished with the last item), we can accept new data
      // Actually, let's think about this differently.

      // The protocol:
      // 1. Source has data, sets src_pending=1, sends data
      // 2. Destination sees src_pending=1 (after sync), captures data, sets dst_pending=1
      // 3. Destination presents dst_valid=1 to its consumer
      // 4. When dst_ready_i is high, destination accepts the data, sets dst_sent=1
      // 5. Source sees dst_sent=1 (after sync), clears src_pending, can accept new data

      // Let's use a simpler handshake:
      // - src_ready_o is high when source can accept new data
      // - Source accepts data when src_valid_i && src_ready_o
      // - After accepting, source holds data until destination confirms receipt

      // State machine:
      // IDLE: src_ready_o=1, waiting for valid data
      // PENDING: src_ready_o=0, waiting for dst to confirm

      if (src_pending) begin
        // Waiting for destination to confirm
        if (dst_sent_sync2) begin
          // Destination has consumed the data
          src_pending <= 1'b0;
          src_sent <= 1'b0;
        end
      end else begin
        // Idle, can accept new data
        if (src_valid_i) begin
          src_pending <= 1'b1;
          src_data_reg <= src_data_i;
          src_sent <= 1'b1;
        end
      end
    end
  end

  assign src_ready_o = ~src_pending;

  // Synchronize src_sent to dst domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_sent_sync1 <= 1'b0;
      src_sent_sync2 <= 1'b0;
    end else begin
      src_sent_sync1 <= src_sent;
      src_sent_sync2 <= src_sent_sync1;
    end
  end

  // Synchronize src_pending to dst domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_pending_sync1 <= 1'b0;
      src_pending_sync2 <= 1'b0;
    end else begin
      src_pending_sync1 <= src_pending;
      src_pending_sync2 <= src_pending_sync1;
    end
  end

  // Destination clock domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_pending <= 1'b0;
      dst_data_reg <= {WIDTH{1'b0}};
      dst_sent <= 1'b0;
    end else begin
      if (dst_pending) begin
        // Waiting for consumer to accept
        if (dst_ready_i) begin
          dst_pending <= 1'b0;
          dst_sent <= 1'b1;
        end
      end else begin
        // Idle, check if source has new data
        if (src_pending_sync2) begin
          dst_pending <= 1'b1;
          dst_data_reg <= src_data_reg;
          dst_sent <= 1'b0;
        end
      end
    end
  end

  assign dst_valid_o = dst_pending;
  assign dst_data_o = dst_data_reg;

  // Synchronize dst_sent to src domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_sent_sync1 <= 1'b0;
      dst_sent_sync2 <= 1'b0;
    end else begin
      dst_sent_sync1 <= dst_sent;
      dst_sent_sync2 <= dst_sent_sync1;
    end
  end

  // Synchronize dst_pending to src domain (not strictly needed but for completeness)
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_pending_sync1 <= 1'b0;
      dst_pending_sync2 <= 1'b0;
    end else begin
      dst_pending_sync1 <= dst_pending;
      dst_pending_sync2 <= dst_pending_sync1;
    end
  end

endmodule
