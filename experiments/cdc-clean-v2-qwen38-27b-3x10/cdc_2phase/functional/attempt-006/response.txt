module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain state
  reg src_busy;
  reg [WIDTH-1:0] src_data_reg;

  // Destination domain state
  reg dst_busy;
  reg [WIDTH-1:0] dst_data_reg;

  // Synchronization signals
  // src_to_dst: source indicates it has data to send
  reg [2:0] src_to_dst_sync;
  reg src_to_dst;
  
  // dst_to_src: destination indicates it is ready to accept new data
  reg [2:0] dst_to_src_sync;
  reg dst_to_src;

  // Source domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_busy <= 1'b0;
      src_data_reg <= {WIDTH{1'b0}};
      src_to_dst <= 1'b0;
    end else begin
      if (src_valid_i && !src_busy && dst_to_src) begin
        // Accept new data
        src_data_reg <= src_data_i;
        src_busy <= 1'b1;
        src_to_dst <= 1'b1;
      end else if (!src_valid_i && !src_busy) begin
        // Idle, ensure src_to_dst is low
        src_to_dst <= 1'b0;
      end
    end
  end

  // Source ready: can accept new data when not busy and destination is ready
  assign src_ready_o = !src_busy && dst_to_src;

  // Synchronize src_to_dst to destination domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_to_dst_sync <= 3'b000;
    end else begin
      src_to_dst_sync <= {src_to_dst_sync[1:0], src_to_dst};
    end
  end

  // Synchronize dst_to_src to source domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_to_src_sync <= 3'b000;
    end else begin
      dst_to_src_sync <= {dst_to_src_sync[1:0], dst_to_src};
    end
  end

  assign dst_to_src = dst_to_src_sync[2];

  // Destination domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_busy <= 1'b0;
      dst_data_reg <= {WIDTH{1'b0}};
      dst_to_src <= 1'b1;
    end else begin
      if (dst_busy) begin
        if (dst_ready_i) begin
          // Transfer complete, ready for next
          dst_busy <= 1'b0;
          dst_to_src <= 1'b1;
        end
        // If not ready, stay busy, keep data stable
      end else begin
        if (src_to_dst_sync[2]) begin
          // New data available from source
          dst_data_reg <= src_data_i;
          dst_busy <= 1'b1;
          dst_to_src <= 1'b0;
        end else begin
          // Ensure dst_to_src is high when idle
          dst_to_src <= 1'b1;
        end
      end
    end
  end

  assign dst_valid_o = dst_busy;
  assign dst_data_o = dst_data_reg;

endmodule
