module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain: capture data and generate request pulse
  reg [WIDTH-1:0] src_data_reg;
  reg             src_req;
  wire            src_accept;

  // Destination domain: track request and generate response pulse
  reg             dst_req;
  reg             dst_ack;
  reg [WIDTH-1:0] dst_data_reg;
  reg             dst_valid_reg;

  // Synchronizers
  reg [1:0] req_sync;
  reg [1:0] ack_sync;

  // Source domain logic
  assign src_accept = src_valid_i & src_ready_o;

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_reg <= {WIDTH{1'b0}};
      src_req      <= 1'b0;
    end else begin
      if (src_accept) begin
        src_data_reg <= src_data_i;
        src_req      <= 1'b1;
      end else if (ack_sync[1]) begin
        src_req <= 1'b0;
      end
    end
  end

  // Synchronize request to destination domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      req_sync <= 2'b00;
    end else begin
      req_sync <= {req_sync[0], src_req};
    end
  end

  // Synchronize acknowledge to source domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      ack_sync <= 2'b00;
    end else begin
      ack_sync <= {ack_sync[0], dst_ack};
    end
  end

  // Destination domain logic
  wire dst_req_sync = req_sync[1];
  wire dst_ack_sync = ack_sync[1];

  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_req       <= 1'b0;
      dst_ack       <= 1'b0;
      dst_data_reg  <= {WIDTH{1'b0}};
      dst_valid_reg <= 1'b0;
    end else begin
      // Handle request
      if (dst_req_sync && !dst_req) begin
        dst_req      <= 1'b1;
        dst_data_reg <= src_data_reg;
        dst_valid_reg <= 1'b1;
      end

      // Handle acknowledge
      if (dst_ack_sync && !dst_ack) begin
        dst_ack <= 1'b1;
      end

      // Clear request when acknowledged
      if (dst_req && dst_ack) begin
        dst_req <= 1'b0;
      end

      // Clear acknowledge when request is low
      if (dst_ack && !dst_req) begin
        dst_ack <= 1'b0;
      end

      // Clear valid when destination accepts
      if (dst_valid_reg && dst_ready_i) begin
        dst_valid_reg <= 1'b0;
      end
    end
  end

  // Source ready: can accept new data when no outstanding request
  assign src_ready_o = !src_req;

  // Destination outputs
  assign dst_data_o  = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

endmodule
