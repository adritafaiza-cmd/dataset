		output	wire [3 : 0]			M_AXI_ARQOS,
		output	wire				M_AXI_ARVALID,
		input	wire				M_AXI_ARREADY,
		input	wire [C_S_AXI_ID_WIDTH-1 : 0]	M_AXI_RID,

		input	wire [1 : 0]			M_AXI_RRESP,
		input	wire				M_AXI_RLAST,
		input	wire				M_AXI_RVALID,

	// FIFO depth
	localparam FIFO_DEPTH = (1 << LGFIFO);
	localparam FIFO_ADDR_WIDTH = LGFIFO;

	// AW channel: S clock domain
	reg [C_S_AXI_ID_WIDTH-1:0] s_awid_r;
	reg [C_S_AXI_ADDR_WIDTH-1:0] s_awaddr_r;
	reg [7:0] s_awlen_r;
	reg [2:0] s_awsize_r;
	reg [1:0] s_awburst_r;
	reg s_awlock_r;
	reg [3:0] s_awcache_r;
	reg [2:0] s_awprot_r;
	reg [3:0] s_awqos_r;
	reg s_awvalid_r;
	reg s_awready_r;

	// AW FIFO
	reg [C_S_AXI_ID_WIDTH-1:0] aw_fifo_data [0:FIFO_DEPTH-1];
	reg [C_S_AXI_ADDR_WIDTH-1:0] aw_fifo_addr [0:FIFO_DEPTH-1];
	reg [7:0] aw_fifo_len [0:FIFO_DEPTH-1];
	reg [2:0] aw_fifo_size [0:FIFO_DEPTH-1];
	reg [1:0] aw_fifo_burst [0:FIFO_DEPTH-1];
	reg aw_fifo_lock [0:FIFO_DEPTH-1];
	reg [3:0] aw_fifo_cache [0:FIFO_DEPTH-1];
	reg [2:0] aw_fifo_prot [0:FIFO_DEPTH-1];
	reg [3:0] aw_fifo_qos [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH-1:0] aw_wr_ptr, aw_rd_ptr;
	reg [FIFO_ADDR_WIDTH:0] aw_count;

	// W channel: S clock domain
	reg [C_S_AXI_DATA_WIDTH-1:0] s_wdata_r;
	reg [(C_S_AXI_DATA_WIDTH/8)-1:0] s_wstrb_r;
	reg s_wlast_r;
	reg s_wvalid_r;
	reg s_wready_r;

	// W FIFO
	reg [C_S_AXI_DATA_WIDTH-1:0] w_fifo_data [0:FIFO_DEPTH-1];
	reg [(C_S_AXI_DATA_WIDTH/8)-1:0] w_fifo_strb [0:FIFO_DEPTH-1];
	reg w_fifo_last [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH-1:0] w_wr_ptr, w_rd_ptr;
	reg [FIFO_ADDR_WIDTH:0] w_count;

	// B channel: M clock domain
	reg [C_S_AXI_ID_WIDTH-1:0] m_bid_r;
	reg [1:0] m_bresp_r;
	reg m_bvalid_r;
	reg m_bready_r;

	// B FIFO
	reg [C_S_AXI_ID_WIDTH-1:0] b_fifo_id [0:FIFO_DEPTH-1];
	reg [1:0] b_fifo_resp [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH-1:0] b_wr_ptr, b_rd_ptr;
	reg [FIFO_ADDR_WIDTH:0] b_count;

	// AR channel: S clock domain
	reg [C_S_AXI_ID_WIDTH-1:0] s_arid_r;
	reg [C_S_AXI_ADDR_WIDTH-1:0] s_araddr_r;
	reg [7:0] s_arlen_r;
	reg [2:0] s_arsize_r;
	reg [1:0] s_arburst_r;
	reg s_arlock_r;
	reg [3:0] s_arcache_r;
	reg [2:0] s_arprot_r;
	reg [3:0] s_arqos_r;
	reg s_arvalid_r;
	reg s_arready_r;

	// AR FIFO
	reg [C_S_AXI_ID_WIDTH-1:0] ar_fifo_id [0:FIFO_DEPTH-1];
	reg [C_S_AXI_ADDR_WIDTH-1:0] ar_fifo_addr [0:FIFO_DEPTH-1];
	reg [7:0] ar_fifo_len [0:FIFO_DEPTH-1];
	reg [2:0] ar_fifo_size [0:FIFO_DEPTH-1];
	reg [1:0] ar_fifo_burst [0:FIFO_DEPTH-1];
	reg ar_fifo_lock [0:FIFO_DEPTH-1];
	reg [3:0] ar_fifo_cache [0:FIFO_DEPTH-1];
	reg [2:0] ar_fifo_prot [0:FIFO_DEPTH-1];
	reg [3:0] ar_fifo_qos [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH-1:0] ar_wr_ptr, ar_rd_ptr;
	reg [FIFO_ADDR_WIDTH:0] ar_count;

	// R channel: M clock domain
	reg [C_S_AXI_ID_WIDTH-1:0] m_rid_r;
	reg [C_S_AXI_DATA_WIDTH-1:0] m_rdata_r;
	reg [1:0] m_rresp_r;
	reg m_rlast_r;
	reg m_rvalid_r;
	reg m_rready_r;

	// R FIFO
	reg [C_S_AXI_ID_WIDTH-1:0] r_fifo_id [0:FIFO_DEPTH-1];
	reg [C_S_AXI_DATA_WIDTH-1:0] r_fifo_data [0:FIFO_DEPTH-1];
	reg [1:0] r_fifo_resp [0:FIFO_DEPTH-1];
	reg r_fifo_last [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH-1:0] r_wr_ptr, r_rd_ptr;
	reg [FIFO_ADDR_WIDTH:0] r_count;

	// Reset synchronization
	reg [1:0] m_reset_sync;
	reg m_reset_n;

	always @(posedge M_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			m_reset_sync <= 2'b00;
			m_reset_n <= 1'b0;
		end else begin
			m_reset_sync <= {m_reset_sync[0], 1'b1};
			m_reset_n <= m_reset_sync[1];
		end
	end

	assign M_AXI_ARESETN = m_reset_n;

	// AW channel S-side
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			s_awid_r <= {C_S_AXI_ID_WIDTH{1'b0}};
			s_awaddr_r <= {C_S_AXI_ADDR_WIDTH{1'b0}};
			s_awlen_r <= 8'b0;
			s_awsize_r <= 3'b0;
			s_awburst_r <= 2'b0;
			s_awlock_r <= 1'b0;
			s_awcache_r <= 4'b0;
			s_awprot_r <= 3'b0;
			s_awqos_r <= 4'b0;
			s_awvalid_r <= 1'b0;
			s_awready_r <= 1'b0;
		end else begin
			if (S_AXI_AWVALID && s_awready_r) begin
				s_awid_r <= S_AXI_AWID;
				s_awaddr_r <= S_AXI_AWADDR;
				s_awlen_r <= S_AXI_AWLEN;
				s_awsize_r <= S_AXI_AWSIZE;
				s_awburst_r <= S_AXI_AWBURST;
				s_awlock_r <= S_AXI_AWLOCK;
				s_awcache_r <= S_AXI_AWCACHE;
				s_awprot_r <= S_AXI_AWPROT;
				s_awqos_r <= S_AXI_AWQOS;
				s_awvalid_r <= 1'b1;
			end else if (s_awvalid_r && (aw_count < FIFO_DEPTH[0])) begin
				s_awvalid_r <= 1'b0;
			end
			if (s_awvalid_r && (aw_count < FIFO_DEPTH[0])) begin
				s_awready_r <= 1'b1;
			end else begin
				s_awready_r <= 1'b0;
			end
		end
	end

	assign S_AXI_AWREADY = s_awready_r;

	// AW FIFO write (S clock)
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			aw_wr_ptr <= {FIFO_ADDR_WIDTH{1'b0}};
			aw_count <= {(FIFO_ADDR_WIDTH+1){1'b0}};
		end else begin
			if (s_awvalid_r && (aw_count < FIFO_DEPTH[0])) begin
				aw_fifo_data[aw_wr_ptr] <= s_awid_r;
				aw_fifo_addr[aw_wr_ptr] <= s_awaddr_r;
				aw_fifo_len[aw_wr_ptr] <= s_awlen_r;
				aw_fifo_size[aw_wr_ptr] <= s_awsize_r;
				aw_fifo_burst[aw_wr_ptr] <= s_awburst_r;
				aw_fifo_lock[aw_wr_ptr] <= s_awlock_r;
				aw_fifo_cache[aw_wr_ptr] <= s_awcache_r;
				aw_fifo_prot[aw_wr_ptr] <= s_awprot_r;
				aw_fifo_qos[aw_wr_ptr] <= s_awqos_r;
				aw_wr_ptr <= aw_wr_ptr + 1'b1;
				aw_count <= aw_count + 1'b1;
			end
		end
	end

	// AW FIFO read (M clock)
	always @(posedge M_AXI_ACLK or negedge m_reset_n) begin
		if (!m_reset_n) begin
			aw_rd_ptr <= {FIFO_ADDR_WIDTH{1'b0}};
		end else begin
			if (M_AXI_AWREADY && (aw_count > 0)) begin
				aw_rd_ptr <= aw_rd_ptr + 1'b1;
			end
		end
	end

	assign M_AXI_AWID = aw_fifo_data[aw_rd_ptr];
	assign M_AXI_AWADDR = aw_fifo_addr[aw_rd_ptr];
	assign M_AXI_AWLEN = aw_fifo_len[aw_rd_ptr];
	assign M_AXI_AWSIZE = aw_fifo_size[aw_rd_ptr];
	assign M_AXI_AWBURST = aw_fifo_burst[aw_rd_ptr];
	assign M_AXI_AWLOCK = aw_fifo_lock[aw_rd_ptr];
	assign M_AXI_AWCACHE = aw_fifo_cache[aw_rd_ptr];
	assign M_AXI_AWPROT = aw_fifo_prot[aw_rd_ptr];
	assign M_AXI_AWQOS = aw_fifo_qos[aw_rd_ptr];
	assign M_AXI_AWVALID = (aw_count > 0);

	// W channel S-side
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			s_wdata_r <= {C_S_AXI_DATA_WIDTH{1'b0}};
			s_wstrb_r <= {(C_S_AXI_DATA_WIDTH/8){1'b0}};
			s_wlast_r <= 1'b0;
			s_wvalid_r <= 1'b0;
			s_wready_r <= 1'b0;
		end else begin
			if (S_AXI_WVALID && s_wready_r) begin
				s_wdata_r <= S_AXI_WDATA;
				s_wstrb_r <= S_AXI_WSTRB;
				s_wlast_r <= S_AXI_WLAST;
				s_wvalid_r <= 1'b1;
			end else if (s_wvalid_r && (w_count < FIFO_DEPTH[0])) begin
				s_wvalid_r <= 1'b0;
			end
			if (s_wvalid_r && (w_count < FIFO_DEPTH[0])) begin
				s_wready_r <= 1'b1;
			end else begin
				s_wready_r <= 1'b0;
			end
		end
	end

	assign S_AXI_WREADY = s_wready_r;

	// W FIFO write (S clock)
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			w_wr_ptr <= {FIFO_ADDR_WIDTH{1'b0}};
			w_count <= {(FIFO_ADDR_WIDTH+1){1'b0}};
		end else begin
			if (s_wvalid_r && (w_count < FIFO_DEPTH[0])) begin
				w_fifo_data[w_wr_ptr] <= s_wdata_r;
				w_fifo_strb[w_wr_ptr] <= s_wstrb_r;
				w_fifo_last[w_wr_ptr] <= s_wlast_r;
				w_wr_ptr <= w_wr_ptr + 1'b1;
				w_count <= w_count + 1'b1;
			end
		end
	end

	// W FIFO read (M clock)
	always @(posedge M_AXI_ACLK or negedge m_reset_n) begin
		if (!m_reset_n) begin
			w_rd_ptr <= {FIFO_ADDR_WIDTH{1'b0}};
		end else begin
			if (M_AXI_WREADY && (w_count > 0)) begin
				w_rd_ptr <= w_rd_ptr + 1'b1;
			end
		end
	end

	// Extra register for W data (one extra M_AXI clock before M_AXI_WVALID)
	reg [C_S_AXI_DATA_WIDTH-1:0] m_wdata_r;
	reg [(C_S_AXI_DATA_WIDTH/8)-1:0] m_wstrb_r;
	reg m_wlast_r;
	reg m_wvalid_r;

	always @(posedge M_AXI_ACLK or negedge m_reset_n) begin
		if (!m_reset_n) begin
			m_wdata_r <= {C_S_AXI_DATA_WIDTH{1'b0}};
			m_wstrb_r <= {(C_S_AXI_DATA_WIDTH/8){1'b0}};
			m_wlast_r <= 1'b0;
			m_wvalid_r <= 1'b0;
		end else begin
			if ((w_count > 0) && !m_wvalid_r) begin
				m_wdata_r <= w_fifo_data[w_rd_ptr];
				m_wstrb_r <= w_fifo_strb[w_rd_ptr];
				m_wlast_r <= w_fifo_last[w_rd_ptr];
				m_wvalid_r <= 1'b1;
			end else if (m_wvalid_r && M_AXI_WREADY) begin
				m_wvalid_r <= 1'b0;
			end
		end
	end

	assign M_AXI_WDATA = m_wdata_r;
	assign M_AXI_WSTRB = m_wstrb_r;
	assign M_AXI_WLAST = m_wlast_r;
	assign M_AXI_WVALID = m_wvalid_r;

	// B channel M-side
	always @(posedge M_AXI_ACLK or negedge m_reset_n) begin
		if (!m_reset_n) begin
			m_bid_r <= {C_S_AXI_ID_WIDTH{1'b0}};
			m_bresp_r <= 2'b0;
			m_bvalid_r <= 1'b0;
			m_bready_r <= 1'b0;
		end else begin
			if (M_AXI_BVALID && m_bready_r) begin
				m_bid_r <= M_AXI_BID;
				m_bresp_r <= M_AXI_BRESP;
				m_bvalid_r <= 1'b1;
			end else if (m_bvalid_r && (b_count < FIFO_DEPTH[0])) begin
				m_bvalid_r <= 1'b0;
			end
			if (m_bvalid_r && (b_count < FIFO_DEPTH[0])) begin
				m_bready_r <= 1'b1;
			end else begin
				m_bready_r <= 1'b0;
			end
		end
	end

	assign M_AXI_BREADY = m_bready_r;

	// B FIFO write (M clock)
	always @(posedge M_AXI_ACLK or negedge m_reset_n) begin
		if (!m_reset_n) begin
			b_wr_ptr <= {FIFO_ADDR_WIDTH{1'b0}};
			b_count <= {(FIFO_ADDR_WIDTH+1){1'b0}};
		end else begin
			if (m_bvalid_r && (b_count < FIFO_DEPTH[0])) begin
				b_fifo_id[b_wr_ptr] <= m_bid_r;
				b_fifo_resp[b_wr_ptr] <= m_bresp_r;
				b_wr_ptr <= b_wr_ptr + 1'b1;
				b_count <= b_count + 1'b1;
			end
		end
	end

	// B FIFO read (S clock)
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			b_rd_ptr <= {FIFO_ADDR_WIDTH{1'b0}};
		end else begin
			if (S_AXI_BREADY && (b_count > 0)) begin
				b_rd_ptr <= b_rd_ptr + 1'b1;
			end
		end
	end

	assign S_AXI_BID = b_fifo_id[b_rd_ptr];
	assign S_AXI_BRESP = b_fifo_resp[b_rd_ptr];
	assign S_AXI_BVALID = (b_count > 0);

	// AR channel S-side
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			s_arid_r <= {C_S_AXI_ID_WIDTH{1'b0}};
			s_araddr_r <= {C_S_AXI_ADDR_WIDTH{1'b0}};
			s_arlen_r <= 8'b0;
			s_arsize_r <= 3'b0;
			s_arburst_r <= 2'b0;
			s_arlock_r <= 1'b0;
			s_arcache_r <= 4'b0;
			s_arprot_r <= 3'b0;
			s_arqos_r <= 4'b0;
			s_arvalid_r <= 1'b0;
			s_arready_r <= 1'b0;
		end else begin
			if (S_AXI_ARVALID && s_arready_r) begin
				s_arid_r <= S_AXI_ARID;
				s_araddr_r <= S_AXI_ARADDR;
				s_arlen_r <= S_AXI_ARLEN;
				s_arsize_r <= S_AXI_ARSIZE;
				s_arburst_r <= S_AXI_ARBURST;
				s_arlock_r <= S_AXI_ARLOCK;
				s_arcache_r <= S_AXI_ARCACHE;
				s_arprot_r <= S_AXI_ARPROT;
				s_arqos_r <= S_AXI_ARQOS;
				s_arvalid_r <= 1'b1;
			end else if (s_arvalid_r && (ar_count < FIFO_DEPTH[0])) begin
				s_arvalid_r <= 1'b0;
			end
			if (s_arvalid_r && (ar_count < FIFO_DEPTH[0])) begin
				s_arready_r <= 1'b1;
			end else begin
				s_arready_r <= 1'b0;
			end
		end
	end

	assign S_AXI_ARREADY = s_arready_r;

	// AR FIFO write (S clock)
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			ar_wr_ptr <= {FIFO_ADDR_WIDTH{1'b0}};
			ar_count <= {(FIFO_ADDR_WIDTH+1){1'b0}};
		end else begin
