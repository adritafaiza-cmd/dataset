	localparam integer FIFO_ADDR_WIDTH = LGFIFO;

		else
			m_reset_sync <= {m_reset_sync[FFS_STAGES-2:0], 1'b1};

	assign aw_empty = (aw_wr_ptr == aw_rd_ptr);

		else if (aw_wr_en)

	end

			aw_fifo_mem[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= {
				S_AXI_AWID,
				S_AXI_AWADDR,
				S_AXI_AWLEN,
				S_AXI_AWSIZE,
				S_AXI_AWBURST,
				S_AXI_AWLOCK,
				S_AXI_AWCACHE,
				S_AXI_AWPROT,
				S_AXI_AWQOS
			};
		end

		if (!M_AXI_ARESETN)
			aw_rd_ptr <= {FIFO_ADDR_WIDTH+1{1'b0}};
		else if (aw_rd_en)

				 m_awqos_reg} <= aw_fifo_mem[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				m_awvalid_reg <= 1'b1;
			end else if (m_awvalid_reg && M_AXI_AWREADY) begin
				m_awvalid_reg <= 1'b0;
			end
		end

	// R read side (S_AXI_ACLK domain)
	assign r_rd_en = S_AXI_RREADY && !r_empty;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN)
			r_rd_ptr <= {FIFO_ADDR_WIDTH+1{1'b0}};
		else if (r_rd_en)
			r_rd_ptr <= r_rd_ptr + 1'b1;
	end

	// R output registers
	reg [C_S_AXI_ID_WIDTH-1:0] s_rid_reg;
	reg [C_S_AXI_DATA_WIDTH-1:0] s_rdata_reg;
	reg [1:0] s_rresp_reg;
	reg s_rlast_reg;
	reg s_rvalid_reg;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			s_rid_reg    <= {C_S_AXI_ID_WIDTH{1'b0}};
			s_rdata_reg  <= {C_S_AXI_DATA_WIDTH{1'b0}};
			s_rresp_reg  <= 2'b0;
			s_rlast_reg  <= 1'b0;
			s_rvalid_reg <= 1'b0;
		end else begin
			if (r_rd_en) begin
				{s_rid_reg, s_rdata_reg, s_rresp_reg, s_rlast_reg} <= r_fifo_mem[r_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				s_rvalid_reg <= 1'b1;
			end else if (s_rvalid_reg && S_AXI_RREADY) begin
				s_rvalid_reg <= 1'b0;
			end
		end
	end

	assign S_AXI_RID    = s_rid_reg;
	assign S_AXI_RDATA  = s_rdata_reg;
	assign S_AXI_RRESP  = s_rresp_reg;
	assign S_AXI_RLAST  = s_rlast_reg;
	assign S_AXI_RVALID = s_rvalid_reg;

endmodule
