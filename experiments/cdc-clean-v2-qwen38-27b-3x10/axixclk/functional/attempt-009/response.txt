	localparam integer FIFO_DEPTH = 1 << LGFIFO;
	localparam integer FIFO_ADDR_WIDTH = LGFIFO;
	localparam integer FFS_STAGES = XCLOCK_FFS;

	reg [FFS_STAGES-1:0] m_reset_sync;
	always @(posedge M_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN)
			m_reset_sync <= {FFS_STAGES{1'b0}};
		else
			m_reset_sync <= {m_reset_sync[FFS_STAGES-2:0], 1'b1};
	end
	assign M_AXI_ARESETN = m_reset_sync[FFS_STAGES-1];

	reg [C_S_AXI_ID_WIDTH-1:0] aw_fifo_id [0:FIFO_DEPTH-1];
	reg [C_S_AXI_ADDR_WIDTH-1:0] aw_fifo_addr [0:FIFO_DEPTH-1];
	reg [7:0] aw_fifo_len [0:FIFO_DEPTH-1];
	reg [2:0] aw_fifo_size [0:FIFO_DEPTH-1];
	reg [1:0] aw_fifo_burst [0:FIFO_DEPTH-1];
	reg aw_fifo_lock [0:FIFO_DEPTH-1];
	reg [3:0] aw_fifo_cache [0:FIFO_DEPTH-1];
	reg [2:0] aw_fifo_prot [0:FIFO_DEPTH-1];
	reg [3:0] aw_fifo_qos [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] aw_wr_ptr, aw_rd_ptr;
	wire aw_full = (aw_wr_ptr[FIFO_ADDR_WIDTH] != aw_rd_ptr[FIFO_ADDR_WIDTH]) && 
	               (aw_wr_ptr[FIFO_ADDR_WIDTH-1:0] == aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
	wire aw_empty = (aw_wr_ptr == aw_rd_ptr);
	wire aw_wr_en = S_AXI_AWVALID && S_AXI_AWREADY;
	wire aw_rd_en = M_AXI_AWVALID && M_AXI_AWREADY;
	assign S_AXI_AWREADY = !aw_full;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN)
			aw_wr_ptr <= 0;
		else if (aw_wr_en)
			aw_wr_ptr <= aw_wr_ptr + 1;
	end

	always @(posedge S_AXI_ACLK) begin
		if (aw_wr_en) begin
			aw_fifo_id[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWID;
			aw_fifo_addr[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWADDR;
			aw_fifo_len[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWLEN;
			aw_fifo_size[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWSIZE;
			aw_fifo_burst[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWBURST;
			aw_fifo_lock[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWLOCK;
			aw_fifo_cache[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWCACHE;
			aw_fifo_prot[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWPROT;
			aw_fifo_qos[aw_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWQOS;
		end
	end

	// AW read side in M clock domain
	reg [C_S_AXI_ID_WIDTH-1:0] aw_rd_id;
	reg [C_S_AXI_ADDR_WIDTH-1:0] aw_rd_addr;
	reg [7:0] aw_rd_len;
	reg [2:0] aw_rd_size;
	reg [1:0] aw_rd_burst;

			aw_rd_ptr <= 0;
			aw_rd_valid <= 1'b0;
		end else begin
			if (aw_rd_en) begin
				aw_rd_ptr <= aw_rd_ptr + 1;
			end
			if (!aw_empty) begin
				aw_rd_id <= aw_fifo_id[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				aw_rd_addr <= aw_fifo_addr[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				aw_rd_len <= aw_fifo_len[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				aw_rd_size <= aw_fifo_size[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				aw_rd_burst <= aw_fifo_burst[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				aw_rd_lock <= aw_fifo_lock[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				aw_rd_cache <= aw_fifo_cache[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				aw_rd_prot <= aw_fifo_prot[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				aw_rd_qos <= aw_fifo_qos[aw_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				aw_rd_valid <= 1'b1;
			end else if (aw_rd_en) begin
				aw_rd_valid <= 1'b0;
			end
		end
	end

	assign M_AXI_AWID = aw_rd_id;
	assign M_AXI_AWADDR = aw_rd_addr;
	assign M_AXI_AWLEN = aw_rd_len;
	assign M_AXI_AWSIZE = aw_rd_size;
	assign M_AXI_AWBURST = aw_rd_burst;
	assign M_AXI_AWLOCK = aw_rd_lock;
	assign M_AXI_AWCACHE = aw_rd_cache;
	assign M_AXI_AWPROT = aw_rd_prot;
	assign M_AXI_AWQOS = aw_rd_qos;
	assign M_AXI_AWVALID = aw_rd_valid;

	reg [C_S_AXI_DATA_WIDTH-1:0] w_fifo_data [0:FIFO_DEPTH-1];
	reg [(C_S_AXI_DATA_WIDTH/8)-1:0] w_fifo_strb [0:FIFO_DEPTH-1];
	reg w_fifo_last [0:FIFO_DEPTH-1];

			w_fifo_strb[w_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_WSTRB;
			w_fifo_last[w_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_WLAST;
		end
	end

	reg [C_S_AXI_DATA_WIDTH-1:0] w_rd_data;
	reg [(C_S_AXI_DATA_WIDTH/8)-1:0] w_rd_strb;
	reg w_rd_last;
	reg w_rd_valid;
	reg w_rd_valid_d1;

	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			w_rd_ptr <= 0;
			w_rd_valid <= 1'b0;
			w_rd_valid_d1 <= 1'b0;
		end else begin
			if (w_rd_en) begin
				w_rd_ptr <= w_rd_ptr + 1;
			end
			if (!w_empty) begin

				w_rd_strb <= w_fifo_strb[w_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				w_rd_last <= w_fifo_last[w_rd_ptr[FIFO_ADDR_WIDTH-1:0]];

	assign M_AXI_WVALID = w_rd_valid_d1;

	reg [1:0] b_fifo_resp [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] b_wr_ptr, b_rd_ptr;
	wire b_full = (b_wr_ptr[FIFO_ADDR_WIDTH] != b_rd_ptr[FIFO_ADDR_WIDTH]) && 
	               (b_wr_ptr[FIFO_ADDR_WIDTH-1:0] == b_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
	wire b_empty = (b_wr_ptr == b_rd_ptr);
	wire b_wr_en = M_AXI_BVALID && M_AXI_BREADY;
	wire b_rd_en = S_AXI_BVALID && S_AXI_BREADY;

			b_fifo_id[b_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= M_AXI_BID;
			b_fifo_resp[b_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= M_AXI_BRESP;

	end

	reg [C_S_AXI_ID_WIDTH-1:0] b_rd_id;
	reg [1:0] b_rd_resp;
	reg b_rd_valid;

				b_rd_resp <= b_fifo_resp[b_rd_ptr[FIFO_ADDR_WIDTH-1:0]];

			end else if (b_rd_en) begin
				b_rd_valid <= 1'b0;
			end
		end

	assign S_AXI_BRESP = b_rd_resp;
	assign S_AXI_BVALID = b_rd_valid;

	reg [C_S_AXI_ID_WIDTH-1:0] ar_fifo_id [0:FIFO_DEPTH-1];
	reg [C_S_AXI_ADDR_WIDTH-1:0] ar_fifo_addr [0:FIFO_DEPTH-1];

	reg [2:0] ar_fifo_prot [0:FIFO_DEPTH-1];

	wire ar_empty = (ar_wr_ptr == ar_rd_ptr);
	wire ar_wr_en = S_AXI_ARVALID && S_AXI_ARREADY;
	wire ar_rd_en = M_AXI_ARVALID && M_AXI_ARREADY;
	assign S_AXI_ARREADY = !ar_full;

	reg [2:0] ar_rd_prot;

			r_fifo_id[r_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= M_AXI_RID;
			r_fifo_data[r_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= M_AXI_RDATA;
			r_fifo_resp[r_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= M_AXI_RRESP;
			r_fifo_last[r_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= M_AXI_RLAST;
		end
	end

	// R read side in S clock domain
	reg [C_S_AXI_ID_WIDTH-1:0] r_rd_id;
	reg [C_S_AXI_DATA_WIDTH-1:0] r_rd_data;
	reg [1:0] r_rd_resp;
	reg r_rd_last;
	reg r_rd_valid;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			r_rd_ptr <= 0;
			r_rd_valid <= 1'b0;
		end else begin
			if (r_rd_en) begin
				r_rd_ptr <= r_rd_ptr + 1;
			end
			if (!r_empty) begin
				r_rd_id <= r_fifo_id[r_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				r_rd_data <= r_fifo_data[r_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				r_rd_resp <= r_fifo_resp[r_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				r_rd_last <= r_fifo_last[r_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				r_rd_valid <= 1'b1;
			end else if (r_rd_en) begin
				r_rd_valid <= 1'b0;
			end
		end
	end

	assign S_AXI_RID = r_rd_id;
	assign S_AXI_RDATA = r_rd_data;
	assign S_AXI_RRESP = r_rd_resp;
	assign S_AXI_RLAST = r_rd_last;
	assign S_AXI_RVALID = r_rd_valid;

endmodule
