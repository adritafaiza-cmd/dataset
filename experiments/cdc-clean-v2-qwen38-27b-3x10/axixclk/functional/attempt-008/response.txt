	localparam integer FIFO_DEPTH = 1 << LGFIFO;
	localparam integer FIFO_ADDR_WIDTH = LGFIFO;
	localparam integer FFS_STAGES = XCLOCK_FFS;

	// Internal signals
	wire [C_S_AXI_ID_WIDTH-1:0] s_awid;
	wire [C_S_AXI_ADDR_WIDTH-1:0] s_awaddr;
	wire [7:0] s_awlen;
	wire [2:0] s_awsize;
	wire [1:0] s_awburst;
	wire s_awlock;
	wire [3:0] s_awcache;
	wire [2:0] s_awprot;
	wire [3:0] s_awqos;
	wire s_awvalid;
	wire s_awready;

	wire [C_S_AXI_DATA_WIDTH-1:0] s_wdata;
	wire [(C_S_AXI_DATA_WIDTH/8)-1:0] s_wstrb;
	wire s_wlast;
	wire s_wvalid;
	wire s_wready;

	wire [C_S_AXI_ID_WIDTH-1:0] s_bid;
	wire [1:0] s_bresp;
	wire s_bvalid;

	wire [C_S_AXI_ID_WIDTH-1:0] s_arid;
	wire [C_S_AXI_ADDR_WIDTH-1:0] s_araddr;

	wire [3:0] s_arcache;
	wire [2:0] s_arprot;
	wire [3:0] s_arqos;
	wire s_arvalid;
	wire s_arready;

	wire [C_S_AXI_ID_WIDTH-1:0] s_rid;
	wire [C_S_AXI_DATA_WIDTH-1:0] s_rdata;

	wire s_rlast;
	wire s_rvalid;
	wire s_rready;

	wire [C_S_AXI_ADDR_WIDTH-1:0] m_awaddr;

	wire [2:0] m_awsize;

	wire m_awlock;
	wire [3:0] m_awcache;

	wire [3:0] m_awqos;

	wire [1:0] m_arburst;

	wire [3:0] m_arcache;

	reg [FFS_STAGES-1:0] m_reset_sync;
	always @(posedge M_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN)
			m_reset_sync <= {FFS_STAGES{1'b0}};
		else
			m_reset_sync <= {m_reset_sync[FFS_STAGES-2:0], 1'b1};
	end
	assign M_AXI_ARESETN = m_reset_sync[FFS_STAGES-1];

	reg [C_S_AXI_ID_WIDTH-1:0] aw_fifo_data [0:FIFO_DEPTH-1];
	reg [C_S_AXI_ADDR_WIDTH-1:0] aw_fifo_addr [0:FIFO_DEPTH-1];

	reg [2:0] aw_fifo_size [0:FIFO_DEPTH-1];
	reg [1:0] aw_fifo_burst [0:FIFO_DEPTH-1];
	reg aw_fifo_lock [0:FIFO_DEPTH-1];
	reg [3:0] aw_fifo_cache [0:FIFO_DEPTH-1];
	reg [2:0] aw_fifo_prot [0:FIFO_DEPTH-1];
	reg [3:0] aw_fifo_qos [0:FIFO_DEPTH-1];

	reg [FIFO_ADDR_WIDTH:0] aw_fifo_rd_ptr;

	reg aw_fifo_empty;

			aw_fifo_full <= 1'b0;
		end else begin
			if (s_awvalid && s_awready && !aw_fifo_full) begin
				aw_fifo_data[aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWID;
				aw_fifo_addr[aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWADDR;
				aw_fifo_len[aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWLEN;
				aw_fifo_size[aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWSIZE;
				aw_fifo_burst[aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWBURST;
				aw_fifo_lock[aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWLOCK;
				aw_fifo_cache[aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWCACHE;
				aw_fifo_prot[aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWPROT;
				aw_fifo_qos[aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_AWQOS;
				aw_fifo_wr_ptr <= aw_fifo_wr_ptr + 1;
			end
		end
	end

	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			aw_fifo_rd_ptr <= 0;
			aw_fifo_empty <= 1'b1;
		end else begin
			if (m_awvalid && m_awready && !aw_fifo_empty) begin
				aw_fifo_rd_ptr <= aw_fifo_rd_ptr + 1;
			end
		end
	end

	always @(*) begin

			aw_fifo_full = (aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0] == aw_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
			aw_fifo_empty = 1'b0;
		end else begin
			aw_fifo_full = 1'b0;
			aw_fifo_empty = (aw_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0] == aw_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
		end

			m_awaddr = {C_S_AXI_ADDR_WIDTH{1'b0}};
			m_awlen = 8'b0;
			m_awsize = 3'b0;
			m_awburst = 2'b0;
			m_awlock = 1'b0;
			m_awcache = 4'b0;
			m_awprot = 3'b0;
			m_awqos = 4'b0;
			m_awvalid = 1'b0;
		end else begin

			m_awaddr = aw_fifo_addr[aw_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];

			m_awcache = aw_fifo_cache[aw_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_awprot = aw_fifo_prot[aw_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_awqos = aw_fifo_qos[aw_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_awvalid = 1'b1;
		end
	end

	assign s_awready = !aw_fifo_full;
	assign S_AXI_AWREADY = s_awready;

	// W channel: S to M
	// FIFO for W channel
	reg [C_S_AXI_DATA_WIDTH-1:0] w_fifo_data [0:FIFO_DEPTH-1];
	reg [(C_S_AXI_DATA_WIDTH/8)-1:0] w_fifo_strb [0:FIFO_DEPTH-1];
	reg w_fifo_last [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] w_fifo_wr_ptr;

	reg w_fifo_full;

	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin

			w_fifo_wr_ptr <= 0;
			w_fifo_full <= 1'b0;
		end else begin
			if (s_wvalid && s_wready && !w_fifo_full) begin
				w_fifo_data[w_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_WDATA;
				w_fifo_strb[w_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_WSTRB;
				w_fifo_last[w_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_WLAST;
				w_fifo_wr_ptr <= w_fifo_wr_ptr + 1;
			end
		end

			w_fifo_full = 1'b0;

			w_out_strb <= {(C_S_AXI_DATA_WIDTH/8){1'b0}};
			w_out_last <= 1'b0;
			w_out_valid <= 1'b0;
		end else begin
			if (!w_fifo_empty) begin
				w_out_data <= w_fifo_data[w_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				w_out_strb <= w_fifo_strb[w_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
				w_out_last <= w_fifo_last[w_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];

	reg [3:0] ar_fifo_cache [0:FIFO_DEPTH-1];
	reg [2:0] ar_fifo_prot [0:FIFO_DEPTH-1];
	reg [3:0] ar_fifo_qos [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] ar_fifo_wr_ptr;
	reg [FIFO_ADDR_WIDTH:0] ar_fifo_rd_ptr;

				ar_fifo_burst[ar_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_ARBURST;
				ar_fifo_lock[ar_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_ARLOCK;
				ar_fifo_cache[ar_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_ARCACHE;
				ar_fifo_prot[ar_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_ARPROT;
				ar_fifo_qos[ar_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= S_AXI_ARQOS;
				ar_fifo_wr_ptr <= ar_fifo_wr_ptr + 1;
			end
		end
	end

	// AR read in M clock domain
	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			ar_fifo_rd_ptr <= 0;
			ar_fifo_empty <= 1'b1;
		end else begin
			if (m_arvalid && m_arready && !ar_fifo_empty) begin
				ar_fifo_rd_ptr <= ar_fifo_rd_ptr + 1;
			end
		end
	end

	// AR FIFO status
	always @(*) begin
		if (ar_fifo_wr_ptr[FIFO_ADDR_WIDTH] != ar_fifo_rd_ptr[FIFO_ADDR_WIDTH]) begin
			ar_fifo_full = (ar_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0] == ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
			ar_fifo_empty = 1'b0;
		end else begin
			ar_fifo_full = 1'b0;
			ar_fifo_empty = (ar_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0] == ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
		end
	end

	// AR output in M clock domain
	always @(*) begin
		if (ar_fifo_empty) begin
			m_arid = {C_S_AXI_ID_WIDTH{1'b0}};
			m_araddr = {C_S_AXI_ADDR_WIDTH{1'b0}};
			m_arlen = 8'b0;
			m_arsize = 3'b0;
			m_arburst = 2'b0;
			m_arlock = 1'b0;
			m_arcache = 4'b0;
			m_arprot = 3'b0;
			m_arqos = 4'b0;
			m_arvalid = 1'b0;
		end else begin
			m_arid = ar_fifo_id[ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_araddr = ar_fifo_addr[ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_arlen = ar_fifo_len[ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_arsize = ar_fifo_size[ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_arburst = ar_fifo_burst[ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_arlock = ar_fifo_lock[ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_arcache = ar_fifo_cache[ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_arprot = ar_fifo_prot[ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_arqos = ar_fifo_qos[ar_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			m_arvalid = 1'b1;
		end
	end

	// AR ready in S clock domain
	assign s_arready = !ar_fifo_full;
	assign S_AXI_ARREADY = s_arready;

	// R channel: M to S
	// FIFO for R channel
	reg [C_S_AXI_ID_WIDTH-1:0] r_fifo_id [0:FIFO_DEPTH-1];
	reg [C_S_AXI_DATA_WIDTH-1:0] r_fifo_data [0:FIFO_DEPTH-1];
	reg [1:0] r_fifo_resp [0:FIFO_DEPTH-1];
	reg r_fifo_last [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] r_fifo_wr_ptr;
	reg [FIFO_ADDR_WIDTH:0] r_fifo_rd_ptr;
	reg r_fifo_full;
	reg r_fifo_empty;

	// R write in M clock domain
	always @(posedge M_AXI_ACLK or negedge M_AXI_ARESETN) begin
		if (!M_AXI_ARESETN) begin
			r_fifo_wr_ptr <= 0;
			r_fifo_full <= 1'b0;
		end else begin
			if (M_AXI_RVALID && M_AXI_RREADY && !r_fifo_full) begin
				r_fifo_id[r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= M_AXI_RID;
				r_fifo_data[r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= M_AXI_RDATA;
				r_fifo_resp[r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= M_AXI_RRESP;
				r_fifo_last[r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= M_AXI_RLAST;
				r_fifo_wr_ptr <= r_fifo_wr_ptr + 1;
			end
		end
	end

	// R read in S clock domain
	always @(posedge S_AXI_ACLK or negedge S_AXI_ARESETN) begin
		if (!S_AXI_ARESETN) begin
			r_fifo_rd_ptr <= 0;
			r_fifo_empty <= 1'b1;
		end else begin
			if (s_rvalid && s_rready && !r_fifo_empty) begin
				r_fifo_rd_ptr <= r_fifo_rd_ptr + 1;
			end
		end
	end

	// R FIFO status
	always @(*) begin
		if (r_fifo_wr_ptr[FIFO_ADDR_WIDTH] != r_fifo_rd_ptr[FIFO_ADDR_WIDTH]) begin
			r_fifo_full = (r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0] == r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
			r_fifo_empty = 1'b0;
		end else begin
			r_fifo_full = 1'b0;
			r_fifo_empty = (r_fifo_wr_ptr[FIFO_ADDR_WIDTH-1:0] == r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
		end
	end

	// R output in S clock domain
	always @(*) begin
		if (r_fifo_empty) begin
			s_rid = {C_S_AXI_ID_WIDTH{1'b0}};
			s_rdata = {C_S_AXI_DATA_WIDTH{1'b0}};
			s_rresp = 2'b0;
			s_rlast = 1'b0;
			s_rvalid = 1'b0;
		end else begin
			s_rid = r_fifo_id[r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			s_rdata = r_fifo_data[r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			s_rresp = r_fifo_resp[r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			s_rlast = r_fifo_last[r_fifo_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
			s_rvalid = 1'b1;
		end
	end

	// R ready in M clock domain
	assign m_rready = !r_fifo_full;
	assign M_AXI_RREADY = m_rready;

	// R output to S
	assign S_AXI_RID = s_rid;
	assign S_AXI_RDATA = s_rdata;
	assign S_AXI_RRESP = s_rresp;
	assign S_AXI_RLAST = s_rlast;
	assign S_AXI_RVALID = s_rvalid;

	// Master outputs
	assign M_AXI_AWID = m_awid;
	assign M_AXI_AWADDR = m_awaddr;
	assign M_AXI_AWLEN = m_awlen;
	assign M_AXI_AWSIZE = m_awsize;
	assign M_AXI_AWBURST = m_awburst;
	assign M_AXI_AWLOCK = m_awlock;
	assign M_AXI_AWCACHE = m_awcache;
	assign M_AXI_AWPROT = m_awprot;
	assign M_AXI_AWQOS = m_awqos;
	assign M_AXI_AWVALID = m_awvalid;

	assign M_AXI_WDATA = m_wdata;
	assign M_AXI_WSTRB = m_wstrb;
	assign M_AXI_WLAST = m_wlast;
	assign M_AXI_WVALID = m_wvalid;

	assign M_AXI_ARID = m_arid;
	assign M_AXI_ARADDR = m_araddr;
	assign M_AXI_ARLEN = m_arlen;
	assign M_AXI_ARSIZE = m_arsize;
	assign M_AXI_ARBURST = m_arburst;
	assign M_AXI_ARLOCK = m_arlock;
	assign M_AXI_ARCACHE = m_arcache;
	assign M_AXI_ARPROT = m_arprot;
	assign M_AXI_ARQOS = m_arqos;
	assign M_AXI_ARVALID = m_arvalid;
