module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Source domain: write pointer
    reg [PTR_W-1:0] wptr;
    reg [PTR_W-1:0] wptr_sync; // synchronized to dst domain

    // Destination domain: read pointer
    reg [PTR_W-1:0] rptr;
    reg [PTR_W-1:0] rptr_sync; // synchronized to src domain

    // FIFO storage
    reg [WIDTH-1:0] mem [0:DEPTH-1];

    // Write enable
    wire wfull = (wptr[PTR_W-1] == rptr_sync[PTR_W-1]) &&
                 (wptr[PTR_W-2:0] == rptr_sync[PTR_W-2:0]);
    wire wempty = (wptr == rptr_sync);

    wire src_ready_o = !wfull;
    wire wr_en = src_valid_i & src_ready_o;

    // Write to memory
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            wptr <= {PTR_W{1'b0}};
        end else if (wr_en) begin
            mem[wptr[PTR_W-2:0]] <= src_data_i;
            wptr <= wptr + 1'b1;
        end
    end

    // Synchronize write pointer to destination domain
    reg [PTR_W-1:0] wptr_sync_r1, wptr_sync_r2;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            wptr_sync_r1 <= {PTR_W{1'b0}};
            wptr_sync_r2 <= {PTR_W{1'b0}};
        end else begin
            wptr_sync_r1 <= wptr;
            wptr_sync_r2 <= wptr_sync_r1;
        end
    end
    assign wptr_sync = wptr_sync_r2;

    // Synchronize read pointer to source domain
    reg [PTR_W-1:0] rptr_sync_r1, rptr_sync_r2;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            rptr_sync_r1 <= {PTR_W{1'b0}};
            rptr_sync_r2 <= {PTR_W{1'b0}};
        end else begin
            rptr_sync_r1 <= rptr;
            rptr_sync_r2 <= rptr_sync_r1;
        end
    end
    assign rptr_sync = rptr_sync_r2;

    // Destination domain: read logic
    wire rfull = (wptr_sync[PTR_W-1] != rptr[PTR_W-1]) ||
                 (wptr_sync[PTR_W-2:0] != rptr[PTR_W-2:0]);

    // Two-stage pipeline for destination output
    reg [WIDTH-1:0] stage1_data;
    reg             stage1_valid;
    reg [WIDTH-1:0] stage2_data;
    reg             stage2_valid;

    wire rd_en = dst_ready_i & dst_valid_o;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            rptr <= {PTR_W{1'b0}};
            stage1_data <= {WIDTH{1'b0}};
            stage1_valid <= 1'b0;
            stage2_data <= {WIDTH{1'b0}};
            stage2_valid <= 1'b0;
        end else begin
            // Stage 2: output stage
            stage2_data <= stage1_data;
            stage2_valid <= stage1_valid;

            // Stage 1: read from memory
            if (rfull && !stage1_valid) begin
                stage1_data <= mem[rptr[PTR_W-2:0]];
                stage1_valid <= 1'b1;
                rptr <= rptr + 1'b1;
            end else if (rd_en) begin
                stage1_valid <= 1'b0;
            end
        end
    end

    assign dst_data_o = stage2_data;
    assign dst_valid_o = stage2_valid;

endmodule
