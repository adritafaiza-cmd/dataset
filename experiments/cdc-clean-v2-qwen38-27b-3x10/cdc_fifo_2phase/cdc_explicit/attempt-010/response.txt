module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Source domain: write pointer and FIFO storage
    reg [PTR_W-1:0] wr_ptr;
    reg [WIDTH-1:0] mem [0:DEPTH-1];

    // Destination domain: read pointer
    reg [PTR_W-1:0] rd_ptr;

    // Synchronization of write pointer to destination domain
    reg [PTR_W-1:0] wr_ptr_sync1, wr_ptr_sync2;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            wr_ptr_sync1 <= {PTR_W{1'b0}};
            wr_ptr_sync2 <= {PTR_W{1'b0}};
        end else begin
            wr_ptr_sync1 <= wr_ptr;
            wr_ptr_sync2 <= wr_ptr_sync1;
        end
    end

    // Synchronization of read pointer to source domain
    reg [PTR_W-1:0] rd_ptr_sync1, rd_ptr_sync2;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            rd_ptr_sync1 <= {PTR_W{1'b0}};
            rd_ptr_sync2 <= {PTR_W{1'b0}};
        end else begin
            rd_ptr_sync1 <= rd_ptr;
            rd_ptr_sync2 <= rd_ptr_sync1;
        end
    end

    // Source domain: write logic
    wire src_wr_en = src_valid_i & src_ready_o;
    wire [PTR_W-1:0] wr_ptr_next = wr_ptr + 1'b1;
    wire [PTR_W-1:0] rd_ptr_src = rd_ptr_sync2;
    wire src_full = (wr_ptr_next == rd_ptr_src);
    assign src_ready_o = !src_full;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            wr_ptr <= {PTR_W{1'b0}};
        end else if (src_wr_en) begin
            mem[wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
            wr_ptr <= wr_ptr_next;
        end
    end

    // Destination domain: read logic
    wire [PTR_W-1:0] wr_ptr_dst = wr_ptr_sync2;
    wire [PTR_W-1:0] rd_ptr_next = rd_ptr + 1'b1;
    wire dst_empty = (rd_ptr == wr_ptr_dst);
    wire dst_rd_en = dst_valid_o & dst_ready_i;

    // Two-stage pipeline for destination output
    reg [WIDTH-1:0] data_pipe1, data_pipe2;
    reg             valid_pipe1, valid_pipe2;

    // Stage 1: read from memory
    wire [WIDTH-1:0] mem_rd_data = mem[rd_ptr[LOG_DEPTH-1:0]];

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            rd_ptr <= {PTR_W{1'b0}};
            data_pipe1 <= {WIDTH{1'b0}};
            valid_pipe1 <= 1'b0;
            data_pipe2 <= {WIDTH{1'b0}};
            valid_pipe2 <= 1'b0;
        end else begin
            // Advance read pointer when we consume from stage 2
            if (dst_rd_en) begin
                rd_ptr <= rd_ptr_next;
            end

            // Stage 1: load from memory when not empty and stage 2 is ready to accept
            // We load stage 1 when the FIFO is not empty.
            // Stage 2 accepts from stage 1 when stage 2 is not valid or is being consumed.
            if (!dst_empty) begin
                data_pipe1 <= mem_rd_data;
                valid_pipe1 <= 1'b1;
            end else begin
                valid_pipe1 <= 1'b0;
            end

            // Stage 2: accept from stage 1
            if (valid_pipe1) begin
                data_pipe2 <= data_pipe1;
                valid_pipe2 <= 1'b1;
            end else begin
                valid_pipe2 <= 1'b0;
            end
        end
    end

    assign dst_data_o = data_pipe2;
    assign dst_valid_o = valid_pipe2;

endmodule
