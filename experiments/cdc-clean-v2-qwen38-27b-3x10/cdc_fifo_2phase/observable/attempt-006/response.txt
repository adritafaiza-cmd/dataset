module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;

    // Source domain registers
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_WIDTH-1:0] src_wr_ptr;
    reg [PTR_WIDTH-1:0] dst_rd_ptr_src;

    // Destination domain registers
    reg [PTR_WIDTH-1:0] dst_rd_ptr;
    reg [PTR_WIDTH-1:0] src_wr_ptr_dst;
    reg [WIDTH-1:0] dst_data_reg;
    reg dst_valid_reg;

    // Gray code pointers
    wire [PTR_WIDTH-1:0] src_wr_ptr_gray;
    wire [PTR_WIDTH-1:0] dst_rd_ptr_gray;

    // Synchronizers
    reg [PTR_WIDTH-1:0] src_wr_ptr_gray_sync1, src_wr_ptr_gray_sync2;
    reg [PTR_WIDTH-1:0] dst_rd_ptr_gray_sync1, dst_rd_ptr_gray_sync2;

    // Full and empty flags
    wire src_full;
    wire dst_empty;

    // Source ready
    assign src_ready_o = ~src_full;

    // Destination outputs
    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;

    // Gray code conversion
    function [PTR_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    assign src_wr_ptr_gray = bin2gray(src_wr_ptr);
    assign dst_rd_ptr_gray = bin2gray(dst_rd_ptr);

    // Full condition: write pointer gray equals read pointer gray with MSB different
    // In binary: full when src_wr_ptr[MSB] != dst_rd_ptr_src[MSB] and src_wr_ptr[MSB-1:0] == dst_rd_ptr_src[MSB-1:0]
    assign src_full = (src_wr_ptr[PTR_WIDTH-1] != dst_rd_ptr_src[PTR_WIDTH-1]) && 
                      (src_wr_ptr[PTR_WIDTH-2:0] == dst_rd_ptr_src[PTR_WIDTH-2:0]);

    // Empty condition: read pointer gray equals write pointer gray
    assign dst_empty = (dst_rd_ptr == src_wr_ptr_dst);

    // Source clock domain: write to memory
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_src <= {PTR_WIDTH{1'b0}};
        end else begin
            if (src_valid_i && src_ready_o) begin
                mem[src_wr_ptr[PTR_WIDTH-2:0]] <= src_data_i;
                src_wr_ptr <= src_wr_ptr + 1'b1;
            end
        end
    end

    // Synchronize src_wr_ptr_gray to destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            src_wr_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            src_wr_ptr_gray_sync1 <= src_wr_ptr_gray;
            src_wr_ptr_gray_sync2 <= src_wr_ptr_gray_sync1;
        end
    end

    // Convert synchronized gray code back to binary in destination domain
    reg [PTR_WIDTH-1:0] src_wr_ptr_dst;
    always @(*) begin
        src_wr_ptr_dst = src_wr_ptr_gray_sync2;
        // Gray to binary conversion
        begin : gray2bin
            integer i;
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1) begin
                src_wr_ptr_dst[i] = src_wr_ptr_dst[i+1] ^ src_wr_ptr_gray_sync2[i];
            end
        end
    end

    // Synchronize dst_rd_ptr_gray to source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            dst_rd_ptr_gray_sync1 <= dst_rd_ptr_gray;
            dst_rd_ptr_gray_sync2 <= dst_rd_ptr_gray_sync1;
        end
    end

    // Convert synchronized gray code back to binary in source domain
    reg [PTR_WIDTH-1:0] dst_rd_ptr_src_calc;
    always @(*) begin
        dst_rd_ptr_src_calc = dst_rd_ptr_gray_sync2;
        begin : gray2bin_src
            integer i;
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1) begin
                dst_rd_ptr_src_calc[i] = dst_rd_ptr_src_calc[i+1] ^ dst_rd_ptr_gray_sync2[i];
            end
        end
    end

    // Update dst_rd_ptr_src in source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_src <= {PTR_WIDTH{1'b0}};
        end else begin
            dst_rd_ptr_src <= dst_rd_ptr_src_calc;
        end
    end

    // Destination clock domain: read from memory
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_valid_reg && dst_ready_i) begin
                // Current data is being consumed, prepare next
                if (!dst_empty) begin
                    dst_data_reg <= mem[dst_rd_ptr[PTR_WIDTH-2:0]];
                    dst_rd_ptr <= dst_rd_ptr + 1'b1;
                    dst_valid_reg <= 1'b1;
                end else begin
                    dst_valid_reg <= 1'b0;
                end
            end else if (!dst_valid_reg && !dst_empty) begin
                // No valid data, load next
                dst_data_reg <= mem[dst_rd_ptr[PTR_WIDTH-2:0]];
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_valid_reg <= 1'b1;
            end
        end
    end

endmodule
