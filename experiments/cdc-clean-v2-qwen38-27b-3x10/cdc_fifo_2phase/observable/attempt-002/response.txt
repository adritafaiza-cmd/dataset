module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH;
    localparam FULL_THRESH = DEPTH - 1;

    // Source domain registers
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_W-1:0] src_wr_ptr;
    reg [PTR_W-1:0] dst_rd_ptr_src; // dst read pointer synchronized to src domain

    // Destination domain registers
    reg [WIDTH-1:0] dst_data_reg;
    reg             dst_valid_reg;
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [PTR_W-1:0] src_wr_ptr_dst; // src write pointer synchronized to dst domain

    // Synchronization registers
    reg [PTR_W-1:0] src_wr_ptr_sync1, src_wr_ptr_sync2;
    reg [PTR_W-1:0] dst_rd_ptr_sync1, dst_rd_ptr_sync2;

    // Synchronize src_wr_ptr to dst domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_sync1 <= {PTR_W{1'b0}};
            src_wr_ptr_sync2 <= {PTR_W{1'b0}};
        end else begin
            src_wr_ptr_sync1 <= src_wr_ptr;
            src_wr_ptr_sync2 <= src_wr_ptr_sync1;
        end
    end

    // Synchronize dst_rd_ptr to src domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_sync1 <= {PTR_W{1'b0}};
            dst_rd_ptr_sync2 <= {PTR_W{1'b0}};
        end else begin
            dst_rd_ptr_sync1 <= dst_rd_ptr;
            dst_rd_ptr_sync2 <= dst_rd_ptr_sync1;
        end
    end

    // Compute full/empty in respective domains
    wire [PTR_W:0] src_wr_ptr_ext = {1'b0, src_wr_ptr};
    wire [PTR_W:0] dst_rd_ptr_src_ext = {1'b0, dst_rd_ptr_src};
    wire [PTR_W:0] src_wr_ptr_dst_ext = {1'b0, src_wr_ptr_dst};
    wire [PTR_W:0] dst_rd_ptr_ext = {1'b0, dst_rd_ptr};

    // In src domain: full when (src_wr_ptr - dst_rd_ptr_src) == DEPTH
    // We use the synchronized dst_rd_ptr (dst_rd_ptr_src)
    wire [PTR_W:0] src_count = src_wr_ptr_ext - dst_rd_ptr_src_ext;
    wire src_full = (src_count == DEPTH);

    // In dst domain: empty when (src_wr_ptr_dst - dst_rd_ptr) == 0
    wire [PTR_W:0] dst_count = src_wr_ptr_dst_ext - dst_rd_ptr_ext;
    wire dst_empty = (dst_count == 0);

    // Source side: write to memory
    wire src_write = src_valid_i & ~src_full;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
        end else if (src_write) begin
            mem[src_wr_ptr] <= src_data_i;
            src_wr_ptr <= src_wr_ptr + 1'b1;
        end
    end

    // Destination side: read from memory
    wire dst_read = dst_valid_reg & dst_ready_i;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_read) begin
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_valid_reg <= 1'b0;
            end
            if (!dst_empty && !dst_valid_reg) begin
                dst_data_reg <= mem[dst_rd_ptr];
                dst_valid_reg <= 1'b1;
            end
        end
    end

    // Outputs
    assign src_ready_o = ~src_full;
    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;

endmodule
