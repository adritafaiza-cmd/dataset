module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_W = LOG_DEPTH + 1;

    // Source domain registers
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_W-1:0] src_wr_ptr;
    reg [PTR_W-1:0] dst_rd_ptr_src; // dst read pointer synchronized to src domain

    // Destination domain registers
    reg [WIDTH-1:0] dst_data_reg;
    reg             dst_valid_reg;
    reg [PTR_W-1:0] dst_rd_ptr;
    reg [PTR_W-1:0] src_wr_ptr_dst; // src write pointer synchronized to dst domain

    // Synchronizer registers
    reg [PTR_W-1:0] src_wr_ptr_sync1, src_wr_ptr_sync2;
    reg [PTR_W-1:0] dst_rd_ptr_sync1, dst_rd_ptr_sync2;

    // Synchronize src_wr_ptr to dst domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_sync1 <= {PTR_W{1'b0}};
            src_wr_ptr_sync2 <= {PTR_W{1'b0}};
        end else begin
            src_wr_ptr_sync1 <= src_wr_ptr;
            src_wr_ptr_sync2 <= src_wr_ptr_sync1;
        end
    end

    // Synchronize dst_rd_ptr to src domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_sync1 <= {PTR_W{1'b0}};
            dst_rd_ptr_sync2 <= {PTR_W{1'b0}};
        end else begin
            dst_rd_ptr_sync1 <= dst_rd_ptr;
            dst_rd_ptr_sync2 <= dst_rd_ptr_sync1;
        end
    end

    // Compute full/empty conditions
    wire src_full;
    wire dst_empty;

    // Full: write pointer has lapped read pointer by DEPTH
    // In Gray code or binary, we compare the upper bit and the lower LOG_DEPTH bits
    // For binary pointers with PTR_W = LOG_DEPTH+1:
    // full when src_wr_ptr[LOG_DEPTH] != dst_rd_ptr_src[LOG_DEPTH] && src_wr_ptr[LOG_DEPTH-1:0] == dst_rd_ptr_src[LOG_DEPTH-1:0]
    assign src_full = (src_wr_ptr[LOG_DEPTH] != dst_rd_ptr_src[LOG_DEPTH]) &&
                      (src_wr_ptr[LOG_DEPTH-1:0] == dst_rd_ptr_src[LOG_DEPTH-1:0]);

    // Empty: write pointer equals read pointer
    assign dst_empty = (src_wr_ptr_dst == dst_rd_ptr);

    // Source ready: not full
    assign src_ready_o = ~src_full;

    // Write to memory in source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_W{1'b0}};
        end else if (src_valid_i && src_ready_o) begin
            mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
            src_wr_ptr <= src_wr_ptr + 1'b1;
        end
    end

    // Destination domain: read from memory and present data
    // dst_valid_reg indicates there is data available to be consumed
    // We use a simple approach: dst_valid_reg is set when we read from memory
    // and cleared when dst_ready_i is asserted.

    // Read pointer and data register in destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_W{1'b0}};
            dst_data_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_valid_reg && dst_ready_i) begin
                // Data is consumed, clear valid
                dst_valid_reg <= 1'b0;
            end
            if (!dst_valid_reg && !dst_empty) begin
                // Load next word from memory
                dst_data_reg <= mem[dst_rd_ptr[LOG_DEPTH-1:0]];
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_valid_reg <= 1'b1;
            end
        end
    end

    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;

endmodule
