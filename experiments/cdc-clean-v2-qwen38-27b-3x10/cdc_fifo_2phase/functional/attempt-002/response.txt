module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;

    // Source domain
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_WIDTH-1:0] src_wr_ptr;
    reg [PTR_WIDTH-1:0] src_rd_ptr_sync;
    reg [PTR_WIDTH-1:0] src_rd_ptr_gray_sync;

    // Destination domain
    reg [PTR_WIDTH-1:0] dst_rd_ptr;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_sync;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync;
    reg [WIDTH-1:0] dst_reg;
    reg dst_valid_reg;

    // Gray code conversion functions
    function [PTR_WIDTH-1:0] bin2gray;
        input [PTR_WIDTH-1:0] bin;
        begin
            bin2gray = bin ^ (bin >> 1);
        end
    endfunction

    function [PTR_WIDTH-1:0] gray2bin;
        input [PTR_WIDTH-1:0] gray;
        reg [PTR_WIDTH-1:0] bin;
        integer i;
        begin
            bin[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1)
                bin[i] = bin[i+1] ^ gray[i];
            gray2bin = bin;
        end
    endfunction

    // Source domain: write pointer and memory
    wire src_wr_ptr_gray = bin2gray(src_wr_ptr);
    wire src_rd_ptr_bin = gray2bin(src_rd_ptr_gray_sync);
    wire src_full = (src_wr_ptr[PTR_WIDTH-1] != src_rd_ptr_bin[PTR_WIDTH-1]) &&
                    (src_wr_ptr[PTR_WIDTH-2:0] == src_rd_ptr_bin[PTR_WIDTH-2:0]);

    assign src_ready_o = ~src_full;

    wire src_wr_en = src_valid_i & src_ready_o;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
        else if (src_wr_en)
            src_wr_ptr <= src_wr_ptr + 1'b1;
    end

    always @(posedge src_clk_i) begin
        if (src_wr_en)
            mem[src_wr_ptr[PTR_WIDTH-2:0]] <= src_data_i;
    end

    // Source domain: synchronize destination read pointer (gray)
    reg [PTR_WIDTH-1:0] src_rd_ptr_gray_sync_r1;
    reg [PTR_WIDTH-1:0] src_rd_ptr_gray_sync_r2;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rd_ptr_gray_sync_r1 <= {PTR_WIDTH{1'b0}};
            src_rd_ptr_gray_sync_r2 <= {PTR_WIDTH{1'b0}};
        end else begin
            src_rd_ptr_gray_sync_r1 <= dst_rd_ptr_gray_sync;
            src_rd_ptr_gray_sync_r2 <= src_rd_ptr_gray_sync_r1;
        end
    end

    assign src_rd_ptr_gray_sync = src_rd_ptr_gray_sync_r2;

    // Destination domain: read pointer and output
    wire dst_rd_ptr_gray = bin2gray(dst_rd_ptr);
    wire dst_wr_ptr_bin = gray2bin(dst_wr_ptr_gray_sync);
    wire dst_empty = (dst_rd_ptr == dst_wr_ptr_bin);

    wire dst_rd_en = dst_valid_reg & dst_ready_i;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= {PTR_WIDTH{1'b0}};
            dst_reg <= {WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
        end else begin
            if (dst_rd_en) begin
                dst_rd_ptr <= dst_rd_ptr + 1'b1;
                dst_valid_reg <= 1'b0;
            end else if (!dst_empty && !dst_valid_reg) begin
                dst_reg <= mem[dst_rd_ptr[PTR_WIDTH-2:0]];
                dst_valid_reg <= 1'b1;
            end
        end
    end

    assign dst_data_o = dst_reg;
    assign dst_valid_o = dst_valid_reg;

    // Destination domain: synchronize source write pointer (gray)
    reg [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync_r1;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync_r2;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync_r1 <= {PTR_WIDTH{1'b0}};
            dst_wr_ptr_gray_sync_r2 <= {PTR_WIDTH{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync_r1 <= src_wr_ptr_gray;
            dst_wr_ptr_gray_sync_r2 <= dst_wr_ptr_gray_sync_r1;
        end
    end

    assign dst_wr_ptr_gray_sync = dst_wr_ptr_gray_sync_r2;

endmodule
