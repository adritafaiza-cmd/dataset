module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam PTR_WIDTH = LOG_DEPTH + 1;

    // Source domain
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_WIDTH-1:0] src_wr_ptr;
    reg [PTR_WIDTH-1:0] dst_rd_ptr_src;

    // Destination domain
    reg [WIDTH-1:0] dst_reg;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_dst;
    reg [PTR_WIDTH-1:0] src_wr_ptr_dst;
    reg dst_valid_reg;

    // Gray code pointers
    wire [PTR_WIDTH-1:0] src_wr_ptr_gray = src_wr_ptr ^ {1'b0, src_wr_ptr[PTR_WIDTH-1:1]};
    wire [PTR_WIDTH-1:0] dst_rd_ptr_gray = dst_rd_ptr_src ^ {1'b0, dst_rd_ptr_src[PTR_WIDTH-1:1]};
    wire [PTR_WIDTH-1:0] dst_wr_ptr_gray = dst_wr_ptr_dst ^ {1'b0, dst_wr_ptr_dst[PTR_WIDTH-1:1]};
    wire [PTR_WIDTH-1:0] src_wr_ptr_gray_dst = src_wr_ptr_dst ^ {1'b0, src_wr_ptr_dst[PTR_WIDTH-1:1]};

    // Synchronizers
    reg [PTR_WIDTH-1:0] dst_rd_ptr_gray_sync1, dst_rd_ptr_gray_sync2;
    reg [PTR_WIDTH-1:0] dst_wr_ptr_gray_sync1, dst_wr_ptr_gray_sync2;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            dst_rd_ptr_gray_sync1 <= dst_rd_ptr_gray;
            dst_rd_ptr_gray_sync2 <= dst_rd_ptr_gray_sync1;
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync1 <= {PTR_WIDTH{1'b0}};
            dst_wr_ptr_gray_sync2 <= {PTR_WIDTH{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync1 <= dst_wr_ptr_gray;
            dst_wr_ptr_gray_sync2 <= dst_wr_ptr_gray_sync1;
        end
    end

    // Convert synchronized gray codes back to binary
    function [PTR_WIDTH-1:0] gray_to_bin;
        input [PTR_WIDTH-1:0] gray;
        integer i;
        begin
            gray_to_bin[PTR_WIDTH-1] = gray[PTR_WIDTH-1];
            for (i = PTR_WIDTH-2; i >= 0; i = i - 1)
                gray_to_bin[i] = gray_to_bin[i+1] ^ gray[i];
        end
    endfunction

    wire [PTR_WIDTH-1:0] dst_rd_ptr_src_sync = gray_to_bin(dst_rd_ptr_gray_sync2);
    wire [PTR_WIDTH-1:0] dst_wr_ptr_dst_sync = gray_to_bin(dst_wr_ptr_gray_sync2);

    // Source domain logic
    wire src_full = (src_wr_ptr_gray == dst_rd_ptr_gray_sync2) && (src_wr_ptr[PTR_WIDTH-1] != dst_rd_ptr_gray_sync2[PTR_WIDTH-1]);
    wire src_can_write = src_valid_i && !src_full;

    assign src_ready_o = !src_full;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= {PTR_WIDTH{1'b0}};
            dst_rd_ptr_src <= {PTR_WIDTH{1'b0}};
        end else begin
            if (src_can_write) begin
                mem[src_wr_ptr[LOG_DEPTH-1:0]] <= src_data_i;
                src_wr_ptr <= src_wr_ptr + 1'b1;
            end
            if (dst_valid_reg && dst_ready_i) begin
                dst_rd_ptr_src <= dst_rd_ptr_src + 1'b1;
            end
        end
    end

    // Destination domain logic
    wire dst_empty = (dst_wr_ptr_gray == src_wr_ptr_gray_dst) && (dst_wr_ptr_dst[PTR_WIDTH-1] == src_wr_ptr_gray_dst[PTR_WIDTH-1]);

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_dst <= {PTR_WIDTH{1'b0}};
            src_wr_ptr_dst <= {PTR_WIDTH{1'b0}};
            dst_valid_reg <= 1'b0;
            dst_reg <= {WIDTH{1'b0}};
        end else begin
            if (dst_valid_reg && dst_ready_i) begin
                dst_valid_reg <= 1'b0;
            end
            if (!dst_valid_reg && !dst_empty) begin
                dst_reg <= mem[src_wr_ptr_dst[LOG_DEPTH-1:0]];
                src_wr_ptr_dst <= src_wr_ptr_dst + 1'b1;
                dst_valid_reg <= 1'b1;
            end
        end
    end

    assign dst_data_o = dst_reg;
    assign dst_valid_o = dst_valid_reg;

endmodule
