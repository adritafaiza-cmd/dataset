module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 1 << ADDR_WIDTH;

    // Write domain
    reg [ADDR_WIDTH:0] wptr;
    reg [ADDR_WIDTH:0] rptr_w;
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];

    // Read domain
    reg [ADDR_WIDTH:0] rptr;
    reg [ADDR_WIDTH:0] wptr_r;
    reg [DATA_WIDTH-1:0] rdata_reg;

    // Gray code pointers
    wire [ADDR_WIDTH:0] wptr_gray;
    wire [ADDR_WIDTH:0] rptr_gray;

    // Synchronizers
    reg [ADDR_WIDTH:0] wptr_gray_r [0:SYNC_STAGES-1];
    reg [ADDR_WIDTH:0] rptr_gray_w [0:SYNC_STAGES-1];

    assign wptr_gray = wptr ^ (wptr >> 1);
    assign rptr_gray = rptr ^ (rptr >> 1);

    // Write domain logic
    wire wfull_int = (wptr_gray == rptr_w) && (wptr[ADDR_WIDTH] != rptr_w[ADDR_WIDTH]);
    assign wfull = wfull_int;
    assign waddr = wptr;

    wire winc_eff = winc & ~wfull;

    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {(ADDR_WIDTH+1){1'b0}};
        end else if (winc_eff) begin
            mem[wptr[ADDR_WIDTH-1:0]] <= wdata;
            wptr <= wptr + 1'b1;
        end
    end

    // Read domain logic
    wire rempty_int = (rptr_gray == wptr_r);
    assign rempty = rempty_int;
    assign raddr = rptr;

    wire rinc_eff = rinc & ~rempty;

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {(ADDR_WIDTH+1){1'b0}};
            rdata_reg <= {DATA_WIDTH{1'b0}};
        end else if (rinc_eff) begin
            rdata_reg <= mem[rptr[ADDR_WIDTH-1:0]];
            rptr <= rptr + 1'b1;
        end
    end

    assign rdata = rdata_reg;

    // Synchronize wptr_gray to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_r[0] <= {(ADDR_WIDTH+1){1'b0}};
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                wptr_gray_r[i] <= {(ADDR_WIDTH+1){1'b0}};
        end else begin
            wptr_gray_r[0] <= wptr_gray;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                wptr_gray_r[i] <= wptr_gray_r[i-1];
        end
    end

    assign wptr_r = wptr_gray_r[SYNC_STAGES-1];

    // Synchronize rptr_gray to write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_w[0] <= {(ADDR_WIDTH+1){1'b0}};
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                rptr_gray_w[i] <= {(ADDR_WIDTH+1){1'b0}};
        end else begin
            rptr_gray_w[0] <= rptr_gray;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                rptr_gray_w[i] <= rptr_gray_w[i-1];
        end
    end

    assign rptr_w = rptr_gray_w[SYNC_STAGES-1];

endmodule
