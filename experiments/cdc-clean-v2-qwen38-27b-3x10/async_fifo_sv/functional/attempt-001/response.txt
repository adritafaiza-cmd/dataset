module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 1 << ADDR_WIDTH;
    localparam PTR_WIDTH = ADDR_WIDTH + 1;

    // Write domain
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_WIDTH-1:0]  wptr;
    reg [PTR_WIDTH-1:0]  rptr_w; // read pointer synchronized to write domain

    // Read domain
    reg [PTR_WIDTH-1:0]  rptr;
    reg [PTR_WIDTH-1:0]  wptr_r; // write pointer synchronized to read domain

    // Synchronizer for write pointer to read domain
    reg [PTR_WIDTH-1:0] wptr_sync [0:SYNC_STAGES-1];

    // Synchronizer for read pointer to write domain
    reg [PTR_WIDTH-1:0] rptr_sync [0:SYNC_STAGES-1];

    // Write domain logic
    wire wfull_internal;
    wire winc_valid;

    assign wfull_internal = (wptr[ADDR_WIDTH] != rptr_w[ADDR_WIDTH]) &&
                             (wptr[ADDR_WIDTH-1:0] == rptr_w[ADDR_WIDTH-1:0]);
    assign winc_valid = winc && !wfull_internal;

    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {PTR_WIDTH{1'b0}};
            rptr_w <= {PTR_WIDTH{1'b0}};
        end else begin
            if (winc_valid) begin
                mem[wptr[ADDR_WIDTH-1:0]] <= wdata;
                wptr <= wptr + 1'b1;
            end
            // rptr_w is updated by the synchronizer
        end
    end

    // Synchronize read pointer to write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_sync[0] <= {PTR_WIDTH{1'b0}};
            rptr_sync[1] <= {PTR_WIDTH{1'b0}};
            for (int i = 2; i < SYNC_STAGES; i++)
                rptr_sync[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            rptr_sync[0] <= rptr;
            for (int i = 1; i < SYNC_STAGES; i++)
                rptr_sync[i] <= rptr_sync[i-1];
        end
    end

    assign rptr_w = rptr_sync[SYNC_STAGES-1];

    // Read domain logic
    wire rempty_internal;
    wire rinc_valid;

    assign rempty_internal = (rptr == wptr_r);
    assign rinc_valid = rinc && !rempty_internal;

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {PTR_WIDTH{1'b0}};
            wptr_r <= {PTR_WIDTH{1'b0}};
        end else begin
            if (rinc_valid) begin
                rptr <= rptr + 1'b1;
            end
            // wptr_r is updated by the synchronizer
        end
    end

    // Synchronize write pointer to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_sync[0] <= {PTR_WIDTH{1'b0}};
            wptr_sync[1] <= {PTR_WIDTH{1'b0}};
            for (int i = 2; i < SYNC_STAGES; i++)
                wptr_sync[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            wptr_sync[0] <= wptr;
            for (int i = 1; i < SYNC_STAGES; i++)
                wptr_sync[i] <= wptr_sync[i-1];
        end
    end

    assign wptr_r = wptr_sync[SYNC_STAGES-1];

    // Outputs
    assign wfull = wfull_internal;
    assign waddr = wptr;
    assign rdata = mem[rptr[ADDR_WIDTH-1:0]];
    assign rempty = rempty_internal;
    assign raddr = rptr;

endmodule
