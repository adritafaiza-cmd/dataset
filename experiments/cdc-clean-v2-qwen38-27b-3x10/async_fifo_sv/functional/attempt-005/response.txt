module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 1 << ADDR_WIDTH;
    localparam PTR_WIDTH = ADDR_WIDTH + 1;

    // Write domain
    reg [PTR_WIDTH-1:0] wptr;
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];

    // Read domain
    reg [PTR_WIDTH-1:0] rptr;
    reg [DATA_WIDTH-1:0] rdata_reg;

    // Gray code pointers
    wire [ADDR_WIDTH:0] wptr_gray = wptr ^ (wptr >> 1);
    wire [ADDR_WIDTH:0] rptr_gray = rptr ^ (rptr >> 1);

    // Synchronizers
    reg [ADDR_WIDTH:0] rptr_gray_sync [0:SYNC_STAGES-1];
    reg [ADDR_WIDTH:0] wptr_gray_sync [0:SYNC_STAGES-1];

    // Write domain: synchronize read pointer gray
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_sync[0] <= {ADDR_WIDTH+1{1'b0}};
            rptr_gray_sync[1] <= {ADDR_WIDTH+1{1'b0}};
            for (int i = 2; i < SYNC_STAGES; i = i + 1)
                rptr_gray_sync[i] <= {ADDR_WIDTH+1{1'b0}};
        end else begin
            rptr_gray_sync[0] <= rptr_gray;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                rptr_gray_sync[i] <= rptr_gray_sync[i-1];
        end
    end

    // Read domain: synchronize write pointer gray
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_sync[0] <= {ADDR_WIDTH+1{1'b0}};
            wptr_gray_sync[1] <= {ADDR_WIDTH+1{1'b0}};
            for (int i = 2; i < SYNC_STAGES; i = i + 1)
                wptr_gray_sync[i] <= {ADDR_WIDTH+1{1'b0}};
        end else begin
            wptr_gray_sync[0] <= wptr_gray;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                wptr_gray_sync[i] <= wptr_gray_sync[i-1];
        end
    end

    // Convert synchronized gray to binary
    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        reg [ADDR_WIDTH:0] bin;
        begin
            bin[ADDR_WIDTH] = gray[ADDR_WIDTH];
            for (int i = ADDR_WIDTH-1; i >= 0; i = i - 1)
                bin[i] = bin[i+1] ^ gray[i];
            gray_to_bin = bin;
        end
    endfunction

    wire [ADDR_WIDTH:0] rptr_gray_sync_bin = gray_to_bin(rptr_gray_sync[SYNC_STAGES-1]);
    wire [ADDR_WIDTH:0] wptr_gray_sync_bin = gray_to_bin(wptr_gray_sync[SYNC_STAGES-1]);

    // Full and empty flags
    assign wfull = (wptr[ADDR_WIDTH] == rptr_gray_sync_bin[ADDR_WIDTH]) &&
                   (wptr[ADDR_WIDTH-1:0] == rptr_gray_sync_bin[ADDR_WIDTH-1:0]) &&
                   (wptr[ADDR_WIDTH] != rptr_gray_sync_bin[ADDR_WIDTH]);
    
    // Correct full condition: MSB differs, lower bits same
    assign wfull = (wptr[ADDR_WIDTH] != rptr_gray_sync_bin[ADDR_WIDTH]) &&
                   (wptr[ADDR_WIDTH-1:0] == rptr_gray_sync_bin[ADDR_WIDTH-1:0]);

    assign rempty = (rptr == wptr_gray_sync_bin);

    // Write logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {PTR_WIDTH{1'b0}};
        end else if (winc && !wfull) begin
            mem[wptr[ADDR_WIDTH-1:0]] <= wdata;
            wptr <= wptr + 1'b1;
        end
    end

    // Read logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {PTR_WIDTH{1'b0}};
            rdata_reg <= {DATA_WIDTH{1'b0}};
        end else if (rinc && !rempty) begin
            rdata_reg <= mem[rptr[ADDR_WIDTH-1:0]];
            rptr <= rptr + 1'b1;
        end
    end

    assign rdata = rdata_reg;
    assign waddr = wptr;
    assign raddr = rptr;

endmodule
