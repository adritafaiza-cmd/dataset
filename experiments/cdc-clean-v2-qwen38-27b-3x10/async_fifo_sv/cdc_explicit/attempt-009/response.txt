module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 1 << ADDR_WIDTH;

    // Write domain
    reg [ADDR_WIDTH:0] wptr;
    reg [ADDR_WIDTH:0] rptr_w;
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];

    // Read domain
    reg [ADDR_WIDTH:0] rptr;
    reg [ADDR_WIDTH:0] wptr_r;
    reg [DATA_WIDTH-1:0] rdata_reg;

    // Synchronizer registers for write pointer to read domain
    reg [ADDR_WIDTH:0] wptr_sync [0:SYNC_STAGES];
    // Synchronizer registers for read pointer to write domain
    reg [ADDR_WIDTH:0] rptr_sync [0:SYNC_STAGES];

    // Write pointer update
    wire winc_eff = winc & ~wfull;
    wire [ADDR_WIDTH:0] wptr_next = wptr + (winc_eff ? 1 : 0);

    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n)
            wptr <= {({ADDR_WIDTH+1}{1'b0}});
        else
            wptr <= wptr_next;
    end

    // Read pointer update
    wire rinc_eff = rinc & ~rempty;
    wire [ADDR_WIDTH:0] rptr_next = rptr + (rinc_eff ? 1 : 0);

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n)
            rptr <= {({ADDR_WIDTH+1}{1'b0}});
        else
            rptr <= rptr_next;
    end

    // Write to memory
    always @(posedge wclk) begin
        if (winc_eff)
            mem[wptr[ADDR_WIDTH-1:0]] <= wdata;
    end

    // Read from memory
    always @(posedge rclk) begin
        if (rinc_eff)
            rdata_reg <= mem[rptr[ADDR_WIDTH-1:0]];
    end

    // Synchronize write pointer to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            for (int i = 0; i < SYNC_STAGES; i = i + 1)
                wptr_sync[i] <= {({ADDR_WIDTH+1}{1'b0}});
        end else begin
            wptr_sync[0] <= wptr;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                wptr_sync[i] <= wptr_sync[i-1];
        end
    end

    // Synchronize read pointer to write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            for (int i = 0; i < SYNC_STAGES; i = i + 1)
                rptr_sync[i] <= {({ADDR_WIDTH+1}{1'b0}});
        end else begin
            rptr_sync[0] <= rptr;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                rptr_sync[i] <= rptr_sync[i-1];
        end
    end

    // Full and empty flags
    assign wfull = (wptr[ADDR_WIDTH] != rptr_sync[SYNC_STAGES-1][ADDR_WIDTH]) &&
                   (wptr[ADDR_WIDTH-1:0] == rptr_sync[SYNC_STAGES-1][ADDR_WIDTH-1:0]);

    assign rempty = (rptr == wptr_sync[SYNC_STAGES-1]);

    // Debug outputs
    assign waddr = wptr;
    assign raddr = rptr;
    assign rdata = rdata_reg;

endmodule
