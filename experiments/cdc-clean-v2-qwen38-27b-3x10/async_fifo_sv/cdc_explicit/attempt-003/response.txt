module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 1 << ADDR_WIDTH;
    localparam PTR_WIDTH = ADDR_WIDTH + 1;

    // Write domain
    reg [PTR_WIDTH-1:0] wptr;
    reg [PTR_WIDTH-1:0] wptr_sync; // synchronized read pointer (in write domain)
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];

    // Read domain
    reg [PTR_WIDTH-1:0] rptr;
    reg [PTR_WIDTH-1:0] rptr_sync; // synchronized write pointer (in read domain)
    reg [DATA_WIDTH-1:0] rdata_reg;
    reg rdata_valid;

    // Synchronization registers
    reg [PTR_WIDTH-1:0] wptr_sync_r [0:SYNC_STAGES-1];
    reg [PTR_WIDTH-1:0] rptr_sync_w [0:SYNC_STAGES-1];

    // Full and empty flags
    assign wfull = (wptr == wptr_sync);
    assign rempty = (rptr == rptr_sync);

    // Debug pointer outputs
    assign waddr = wptr;
    assign raddr = rptr;

    // Write pointer increment
    wire winc_eff = winc & ~wfull;
    wire rinc_eff = rinc & ~rempty;

    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {PTR_WIDTH{1'b0}};
        end else if (winc_eff) begin
            wptr <= wptr + 1'b1;
        end
    end

    // Read pointer increment
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {PTR_WIDTH{1'b0}};
        end else if (rinc_eff) begin
            rptr <= rptr + 1'b1;
        end
    end

    // Synchronize write pointer to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            for (int i = 0; i < SYNC_STAGES; i = i + 1)
                wptr_sync_r[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            wptr_sync_r[0] <= wptr;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                wptr_sync_r[i] <= wptr_sync_r[i-1];
        end
    end
    assign wptr_sync = wptr_sync_r[SYNC_STAGES-1];

    // Synchronize read pointer to write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            for (int i = 0; i < SYNC_STAGES; i = i + 1)
                rptr_sync_w[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            rptr_sync_w[0] <= rptr;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                rptr_sync_w[i] <= rptr_sync_w[i-1];
        end
    end
    assign rptr_sync = rptr_sync_w[SYNC_STAGES-1];

    // Write data to memory
    always @(posedge wclk) begin
        if (winc_eff) begin
            mem[wptr[ADDR_WIDTH-1:0]] <= wdata;
        end
    end

    // Read data from memory with two-stage pipeline for CDC safety
    // Stage 1: capture data and address on read clock when read is accepted
    // Stage 2: present data as valid

    reg [DATA_WIDTH-1:0] rdata_stage1;
    reg rdata_stage1_valid;
    reg [ADDR_WIDTH-1:0] raddr_stage1;

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rdata_stage1 <= {DATA_WIDTH{1'b0}};
            rdata_stage1_valid <= 1'b0;
            raddr_stage1 <= {ADDR_WIDTH{1'b0}};
        end else begin
            if (rinc_eff) begin
                rdata_stage1 <= mem[rptr[ADDR_WIDTH-1:0]];
                rdata_stage1_valid <= 1'b1;
                raddr_stage1 <= rptr[ADDR_WIDTH-1:0];
            end else begin
                rdata_stage1_valid <= 1'b0;
            end
        end
    end

    // Stage 2: final output
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rdata_reg <= {DATA_WIDTH{1'b0}};
            rdata_valid <= 1'b0;
        end else begin
            rdata_reg <= rdata_stage1;
            rdata_valid <= rdata_stage1_valid;
        end
    end

    assign rdata = rdata_reg;

endmodule
