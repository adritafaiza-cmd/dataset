module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 1 << ADDR_WIDTH;
    localparam PTR_WIDTH = ADDR_WIDTH + 1;

    // Write domain
    reg [PTR_WIDTH-1:0] wptr;
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];
    reg [PTR_WIDTH-1:0] wptr_sync_reg [0:SYNC_STAGES-1];

    // Read domain
    reg [PTR_WIDTH-1:0] rptr;
    reg [DATA_WIDTH-1:0] rdata_reg;
    reg [PTR_WIDTH-1:0] rptr_sync_reg [0:SYNC_STAGES-1];

    // Synchronized pointers
    wire [PTR_WIDTH-1:0] wptr_synced = wptr_sync_reg[SYNC_STAGES-1];
    wire [PTR_WIDTH-1:0] rptr_synced = rptr_sync_reg[SYNC_STAGES-1];

    // Full/Empty flags
    assign wfull  = (wptr == {rptr_synced[ADDR_WIDTH], ~rptr_synced[ADDR_WIDTH-1:0]});
    assign rempty = (rptr == rptr_synced);

    // Debug address outputs
    assign waddr = wptr;
    assign raddr = rptr;

    // Write logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= {PTR_WIDTH{1'b0}};
            wptr_sync_reg[0] <= {PTR_WIDTH{1'b0}};
            wptr_sync_reg[1] <= {PTR_WIDTH{1'b0}};
            // Initialize remaining sync stages
            for (int i = 2; i < SYNC_STAGES; i = i + 1)
                wptr_sync_reg[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            // Shift sync register
            wptr_sync_reg[0] <= wptr;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                wptr_sync_reg[i] <= wptr_sync_reg[i-1];

            // Write data and increment pointer
            if (winc && !wfull) begin
                mem[wptr[ADDR_WIDTH-1:0]] <= wdata;
                wptr <= wptr + 1'b1;
            end
        end
    end

    // Read logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= {PTR_WIDTH{1'b0}};
            rptr_sync_reg[0] <= {PTR_WIDTH{1'b0}};
            rptr_sync_reg[1] <= {PTR_WIDTH{1'b0}};
            rdata_reg <= {DATA_WIDTH{1'b0}};
            // Initialize remaining sync stages
            for (int i = 2; i < SYNC_STAGES; i = i + 1)
                rptr_sync_reg[i] <= {PTR_WIDTH{1'b0}};
        end else begin
            // Shift sync register
            rptr_sync_reg[0] <= rptr;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                rptr_sync_reg[i] <= rptr_sync_reg[i-1];

            // Read data and increment pointer
            if (rinc && !rempty) begin
                rdata_reg <= mem[rptr[ADDR_WIDTH-1:0]];
                rptr <= rptr + 1'b1;
            end
        end
    end

    assign rdata = rdata_reg;

endmodule
