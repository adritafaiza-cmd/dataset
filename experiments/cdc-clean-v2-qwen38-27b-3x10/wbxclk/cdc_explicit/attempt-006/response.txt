```verilog
module wbxclk #(
		parameter	AW=32,
				DW=32,
				LGFIFO = 5
	) (
		input	wire			i_wb_clk, i_reset,
		input	wire			i_wb_cyc, i_wb_stb, i_wb_we,
		input	wire	[(AW-1):0]	i_wb_addr,
		input	wire	[(DW-1):0]	i_wb_data,
		input	wire	[(DW/8-1):0]	i_wb_sel,
		output	wire			o_wb_stall,
		output	reg			o_wb_ack,
		output	reg	[(DW-1):0]	o_wb_data,
		output	reg			o_wb_err,
		input	wire			i_xclk_clk,
		output	reg			o_xclk_cyc,
		output	reg			o_xclk_stb,
		output	reg			o_xclk_we,
		output	reg	[(AW-1):0]	o_xclk_addr,
		output	reg	[(DW-1):0]	o_xclk_data,
		output	reg	[(DW/8-1):0]	o_xclk_sel,
		input	wire			i_xclk_stall,
		input	wire			i_xclk_ack,
		input	wire	[(DW-1):0]	i_xclk_data,
		input	wire			i_xclk_err
	);

	localparam FIFO_DEPTH = 1 << LGFIFO;
	localparam FIFO_MASK  = FIFO_DEPTH - 1;

	// ------------------------------------------------------------------
	// Reset synchronization
	// ------------------------------------------------------------------
	reg [1:0] rst_sync_wb;
	reg [1:0] rst_sync_xclk;

	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset)
			rst_sync_wb <= 2'b00;
		else
			rst_sync_wb <= {rst_sync_wb[0], 1'b1};
	end

	always @(posedge i_xclk_clk) begin
		rst_sync_xclk <= {rst_sync_xclk[0], rst_sync_wb[1]};
	end

	wire wb_rst_n = rst_sync_wb[1];
	wire xclk_rst_n = rst_sync_xclk[1];

	// ------------------------------------------------------------------
	// Request FIFO (wb_clk domain)
	// ------------------------------------------------------------------
	reg [AW-1:0] req_addr_mem [0:FIFO_DEPTH-1];
	reg [DW-1:0] req_data_mem [0:FIFO_DEPTH-1];
	reg [DW/8-1:0] req_sel_mem [0:FIFO_DEPTH-1];
	reg req_we_mem [0:FIFO_DEPTH-1];

	reg [LGFIFO:0] req_wr_ptr;
	reg [LGFIFO:0] req_rd_ptr;

	wire req_full = (req_wr_ptr == {req_rd_ptr[LGFIFO-1:0], ~req_rd_ptr[LGFIFO]});
	wire req_empty = (req_wr_ptr == req_rd_ptr);

	// Write to request FIFO
	wire req_wr_en = i_wb_cyc & i_wb_stb & wb_rst_n & ~req_full;

	always @(posedge i_wb_clk) begin
		if (req_wr_en) begin
			req_addr_mem[req_wr_ptr[LGFIFO-1:0]] <= i_wb_addr;
			req_data_mem[req_wr_ptr[LGFIFO-1:0]] <= i_wb_data;
			req_sel_mem[req_wr_ptr[LGFIFO-1:0]]  <= i_wb_sel;
			req_we_mem[req_wr_ptr[LGFIFO-1:0]]   <= i_wb_we;
			req_wr_ptr <= req_wr_ptr + 1'b1;
		end
	end

	// ------------------------------------------------------------------
	// Response FIFO (xclk_clk domain)
	// ------------------------------------------------------------------
	reg [DW-1:0] rsp_data_mem [0:FIFO_DEPTH-1];
	reg rsp_err_mem [0:FIFO_DEPTH-1];

	reg [LGFIFO:0] rsp_wr_ptr;
	reg [LGFIFO:0] rsp_rd_ptr;

	wire rsp_full = (rsp_wr_ptr == {rsp_rd_ptr[LGFIFO-1:0], ~rsp_rd_ptr[LGFIFO]});
	wire rsp_empty = (rsp_wr_ptr == rsp_rd_ptr);

	// Write to response FIFO
	wire rsp_wr_en = i_xclk_ack & xclk_rst_n & ~rsp_full;

	always @(posedge i_xclk_clk) begin
		if (rsp_wr_en) begin
			rsp_data_mem[rsp_wr_ptr[LGFIFO-1:0]] <= i_xclk_data;
			rsp_err_mem[rsp_wr_ptr[LGFIFO-1:0]]  <= i_xclk_err;
			rsp_wr_ptr <= rsp_wr_ptr + 1'b1;
		end
	end

	// ------------------------------------------------------------------
	// CDC: Request FIFO pointers
	// ------------------------------------------------------------------
	// wb_clk -> xclk_clk: write pointer
	reg [LGFIFO:0] req_wr_ptr_sync0;
	reg [LGFIFO:0] req_wr_ptr_sync1;

	always @(posedge i_xclk_clk) begin
		req_wr_ptr_sync0 <= req_wr_ptr;
		req_wr_ptr_sync1 <= req_wr_ptr_sync0;
	end

	// xclk_clk -> wb_clk: read pointer
	reg [LGFIFO:0] req_rd_ptr_sync0;
	reg [LGFIFO:0] req_rd_ptr_sync1;

	always @(posedge i_wb_clk) begin
		req_rd_ptr_sync0 <= req_rd_ptr;
		req_rd_ptr_sync1 <= req_rd_ptr_sync0;
	end

	// ------------------------------------------------------------------
	// CDC: Response FIFO pointers
	// ------------------------------------------------------------------
	// xclk_clk -> wb_clk: write pointer
	reg [LGFIFO:0] rsp_wr_ptr_sync0;
	reg [LGFIFO:0] rsp_wr_ptr_sync1;

	always @(posedge i_wb_clk) begin
		rsp_wr_ptr_sync0 <= rsp_wr_ptr;
		rsp_wr_ptr_sync1 <= rsp_wr_ptr_sync0;
	end

	// wb_clk -> xclk_clk: read pointer
	reg [LGFIFO:0] rsp_rd_ptr_sync0;
	reg [LGFIFO:0] rsp_rd_ptr_sync1;

	always @(posedge i_xclk_clk) begin
		rsp_rd_ptr_sync0 <= rsp_rd_ptr;
		rsp_rd_ptr_sync1 <= rsp_rd_ptr_sync0;
	end

	// ------------------------------------------------------------------
	// Destination side (xclk_clk domain)
	// ------------------------------------------------------------------
	// Read from request FIFO
	wire req_rd_en = ~req_empty & ~i_xclk_stall & xclk_rst_n;

	always @(posedge i_xclk_clk) begin
		if (req_rd_en) begin
			req_rd_ptr <= req_rd_ptr + 1'b1;
		end
	end

	// Pipeline stages for destination output
	reg [AW-1:0] pipe_addr_s1, pipe_addr_s2;
	reg [DW-1:0] pipe_data_s1, pipe_data_s2;
	reg [DW/8-1:0] pipe_sel_s1, pipe_sel_s2;
	reg pipe_we_s1, pipe_we_s2;
	reg pipe_valid_s1, pipe_valid_s2;

	always @(posedge i_xclk_clk) begin
		if (xclk_rst_n) begin
			pipe_addr_s1 <= req_addr_mem[req_rd_ptr[LGFIFO-1:0]];
			pipe_data_s1 <= req_data_mem[req_rd_ptr[LGFIFO-1:0]];
			pipe_sel_s1  <= req_sel_mem[req_rd_ptr[LGFIFO-1:0]];
			pipe_we_s1   <= req_we_mem[req_rd_ptr[LGFIFO-1:0]];
			pipe_valid_s1 <= req_rd_en;

			pipe_addr_s2 <= pipe_addr_s1;
			pipe_data_s2 <= pipe_data_s1;
			pipe_sel_s2  <= pipe_sel_s1;
			pipe_we_s2   <= pipe_we_s1;
			pipe_valid_s2 <= pipe_valid_s1;
		end else begin
			pipe_valid_s1 <= 1'b0;
			pipe_valid_s2 <= 1'b0;
		end
	end

	// Destination bus outputs
	always @(posedge i_xclk_clk) begin
		if (xclk_rst_n) begin
			o_xclk_cyc <= pipe_valid_s2;
			o_xclk_stb <= pipe_valid_s2;
			o_xclk_we  <= pipe_we_s2;
			o_xclk_addr <= pipe_addr_s2;
			o_xclk_data <= pipe_data_s2;
			o_xclk_sel  <= pipe_sel_s2;
		end else begin
			o_xclk_cyc <= 1'b0;
			o_xclk_stb <= 1'b0;
			o_xclk_we  <= 1'b0;
		end
	end

	// ------------------------------------------------------------------
	// Source side (wb_clk domain)
	// ------------------------------------------------------------------
	// Read from response FIFO
	wire rsp_rd_en = ~rsp_empty & wb_rst_n;

	always @(posedge i_wb_clk) begin
		if (rsp_rd_en) begin
			rsp_rd_ptr <= rsp_rd_ptr + 1'b1;
		end
	end

	// Source bus outputs
	always @(posedge i_wb_clk) begin
		if (wb_rst_n) begin
			o_wb_ack   <= rsp_rd_en;
			o_wb_data  <= rsp_data_mem[rsp_rd_ptr[LGFIFO-1:0]];
			o_wb_err   <= rsp_err_mem[rsp_rd_ptr[LGFIFO-1:0]];
		end else begin
			o_wb_ack   <= 1'b0;
			o_wb_err   <= 1'b0;
		end
	end

	// ------------------------------------------------------------------
	// Backpressure
	// ------------------------------------------------------------------
	// o_wb_stall: high when request FIFO is full
	assign o_wb_stall = req_full | ~wb_rst_n;

endmodule
