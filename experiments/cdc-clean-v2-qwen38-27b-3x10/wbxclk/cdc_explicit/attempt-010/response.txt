```verilog
module wbxclk #(
		parameter	AW=32,
				DW=32,
				LGFIFO = 5
	) (
		input	wire			i_wb_clk, i_reset,
		input	wire			i_wb_cyc, i_wb_stb, i_wb_we,
		input	wire	[(AW-1):0]	i_wb_addr,
		input	wire	[(DW-1):0]	i_wb_data,
		input	wire	[(DW/8-1):0]	i_wb_sel,
		output	wire			o_wb_stall,
		output	reg			o_wb_ack,
		output	reg	[(DW-1):0]	o_wb_data,
		output	reg			o_wb_err,
		input	wire			i_xclk_clk,
		output	reg			o_xclk_cyc,
		output	reg			o_xclk_stb,
		output	reg			o_xclk_we,
		output	reg	[(AW-1):0]	o_xclk_addr,
		output	reg	[(DW-1):0]	o_xclk_data,
		output	reg	[(DW/8-1):0]	o_xclk_sel,
		input	wire			i_xclk_stall,
		input	wire			i_xclk_ack,
		input	wire	[(DW-1):0]	i_xclk_data,
		input	wire			i_xclk_err
	);

	localparam FIFO_DEPTH = 1 << LGFIFO;
	localparam FIFO_MASK  = FIFO_DEPTH - 1;

	// ------------------------------------------------------------------
	// Reset synchronization
	// ------------------------------------------------------------------
	reg [1:0] rst_sync_wb;
	reg [1:0] rst_sync_xclk;

	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset)
			rst_sync_wb <= 2'b00;
		else
			rst_sync_wb <= {rst_sync_wb[0], 1'b1};
	end

	always @(posedge i_xclk_clk) begin
		if (i_reset)
			rst_sync_xclk <= 2'b00;
		else
			rst_sync_xclk <= {rst_sync_xclk[0], 1'b1};
	end

	wire wb_rst_n = rst_sync_wb[1];
	wire xclk_rst_n = rst_sync_xclk[1];

	// ------------------------------------------------------------------
	// Request FIFO (wb_clk domain)
	// ------------------------------------------------------------------
	reg [AW-1:0] req_addr_mem [0:FIFO_DEPTH-1];
	reg [DW-1:0] req_data_mem [0:FIFO_DEPTH-1];
	reg [DW/8-1:0] req_sel_mem [0:FIFO_DEPTH-1];
	reg req_we_mem [0:FIFO_DEPTH-1];

	reg [LGFIFO:0] req_wr_ptr;
	reg [LGFIFO:0] req_rd_ptr;

	reg [LGFIFO:0] req_rd_ptr_sync;
	reg [LGFIFO:0] req_rd_ptr_sync2;

	always @(posedge i_wb_clk) begin
		if (!wb_rst_n) begin
			req_wr_ptr <= 0;
			req_rd_ptr_sync <= 0;
			req_rd_ptr_sync2 <= 0;
		end else begin
			req_rd_ptr_sync <= req_rd_ptr;
			req_rd_ptr_sync2 <= req_rd_ptr_sync;
		end
	end

	// Synchronize read pointer to xclk domain
	reg [LGFIFO:0] req_rd_ptr_xclk;
	reg [LGFIFO:0] req_rd_ptr_xclk2;

	always @(posedge i_xclk_clk) begin
		if (!xclk_rst_n) begin
			req_rd_ptr_xclk <= 0;
			req_rd_ptr_xclk2 <= 0;
		end else begin
			req_rd_ptr_xclk <= req_rd_ptr_sync2;
			req_rd_ptr_xclk2 <= req_rd_ptr_xclk;
		end
	end

	wire req_full = (req_wr_ptr - req_rd_ptr_sync2) == FIFO_DEPTH;
	wire req_empty = (req_wr_ptr == req_rd_ptr_sync2);

	// Write to request FIFO
	wire req_wr_en = i_wb_cyc & i_wb_stb & ~req_full;

	always @(posedge i_wb_clk) begin
		if (req_wr_en) begin
			req_addr_mem[req_wr_ptr[LGFIFO-1:0]] <= i_wb_addr;
			req_data_mem[req_wr_ptr[LGFIFO-1:0]] <= i_wb_data;
			req_sel_mem[req_wr_ptr[LGFIFO-1:0]] <= i_wb_sel;
			req_we_mem[req_wr_ptr[LGFIFO-1:0]] <= i_wb_we;
			req_wr_ptr <= req_wr_ptr + 1;
		end
	end

	// ------------------------------------------------------------------
	// Response FIFO (xclk_clk domain)
	// ------------------------------------------------------------------
	reg [DW-1:0] rsp_data_mem [0:FIFO_DEPTH-1];
	reg rsp_err_mem [0:FIFO_DEPTH-1];

	reg [LGFIFO:0] rsp_wr_ptr;
	reg [LGFIFO:0] rsp_rd_ptr;

	reg [LGFIFO:0] rsp_rd_ptr_sync;
	reg [LGFIFO:0] rsp_rd_ptr_sync2;

	always @(posedge i_xclk_clk) begin
		if (!xclk_rst_n) begin
			rsp_wr_ptr <= 0;
			rsp_rd_ptr_sync <= 0;
			rsp_rd_ptr_sync2 <= 0;
		end else begin
			rsp_rd_ptr_sync <= rsp_rd_ptr;
			rsp_rd_ptr_sync2 <= rsp_rd_ptr_sync;
		end
	end

	// Synchronize read pointer to wb domain
	reg [LGFIFO:0] rsp_rd_ptr_wb;
	reg [LGFIFO:0] rsp_rd_ptr_wb2;

	always @(posedge i_wb_clk) begin
		if (!wb_rst_n) begin
			rsp_rd_ptr_wb <= 0;
			rsp_rd_ptr_wb2 <= 0;
		end else begin
			rsp_rd_ptr_wb <= rsp_rd_ptr_sync2;
			rsp_rd_ptr_wb2 <= rsp_rd_ptr_wb;
		end
	end

	wire rsp_full = (rsp_wr_ptr - rsp_rd_ptr_wb2) == FIFO_DEPTH;
	wire rsp_empty = (rsp_wr_ptr == rsp_rd_ptr_wb2);

	// ------------------------------------------------------------------
	// XCLK side: read from request FIFO, drive xclk bus
	// ------------------------------------------------------------------
	reg xclk_active;
	reg [LGFIFO-1:0] xclk_rd_addr;

	always @(posedge i_xclk_clk) begin
		if (!xclk_rst_n) begin
			xclk_active <= 0;
			o_xclk_cyc <= 0;
			o_xclk_stb <= 0;
			o_xclk_we <= 0;
			o_xclk_addr <= 0;
			o_xclk_data <= 0;
			o_xclk_sel <= 0;
		end else begin
			if (xclk_active) begin
				// Waiting for ack
				if (i_xclk_ack) begin
					xclk_active <= 0;
					o_xclk_cyc <= 0;
					o_xclk_stb <= 0;
					// Write response to FIFO
					if (!rsp_full) begin
						rsp_data_mem[rsp_wr_ptr[LGFIFO-1:0]] <= i_xclk_data;
						rsp_err_mem[rsp_wr_ptr[LGFIFO-1:0]] <= i_xclk_err;
						rsp_wr_ptr <= rsp_wr_ptr + 1;
					end
				end
			end else begin
				// Check if we can start a new transfer
				if (!req_empty && !i_xclk_stall) begin
					xclk_active <= 1;
					o_xclk_cyc <= 1;
					o_xclk_stb <= 1;
					o_xclk_we <= req_we_mem[req_rd_ptr_xclk2[LGFIFO-1:0]];
					o_xclk_addr <= req_addr_mem[req_rd_ptr_xclk2[LGFIFO-1:0]];
					o_xclk_data <= req_data_mem[req_rd_ptr_xclk2[LGFIFO-1:0]];
					o_xclk_sel <= req_sel_mem[req_rd_ptr_xclk2[LGFIFO-1:0]];
					// Advance read pointer
					req_rd_ptr <= req_rd_ptr + 1;
				end else begin
					o_xclk_cyc <= 0;
					o_xclk_stb <= 0;
				end
			end
		end
	end

	// ------------------------------------------------------------------
	// WB side: read from response FIFO, drive wb outputs
	// ------------------------------------------------------------------
	always @(posedge i_wb_clk) begin
		if (!wb_rst_n) begin
			o_wb_ack <= 0;
			o_wb_data <= 0;
			o_wb_err <= 0;
		end else begin
			if (!rsp_empty) begin
				o_wb_ack <= 1;
				o_wb_data <= rsp_data_mem[rsp_rd_ptr_wb2[LGFIFO-1:0]];
				o_wb_err <= rsp_err_mem[rsp_rd_ptr_wb2[LGFIFO-1:0]];
				rsp_rd_ptr <= rsp_rd_ptr + 1;
			end else begin
				o_wb_ack <= 0;
			end
		end
	end

	// ------------------------------------------------------------------
	// Stall output
	// ------------------------------------------------------------------
	assign o_wb_stall = req_full;

endmodule
