```verilog
module wbxclk #(
		parameter	AW=32,
				DW=32,
				LGFIFO = 5
	) (
		input	wire			i_wb_clk, i_reset,
		input	wire			i_wb_cyc, i_wb_stb, i_wb_we,
		input	wire	[(AW-1):0]	i_wb_addr,
		input	wire	[(DW-1):0]	i_wb_data,
		input	wire	[(DW/8-1):0]	i_wb_sel,
		output	wire			o_wb_stall,
		output	reg			o_wb_ack,
		output	reg	[(DW-1):0]	o_wb_data,
		output	reg			o_wb_err,
		input	wire			i_xclk_clk,
		output	reg			o_xclk_cyc,
		output	reg			o_xclk_stb,
		output	reg			o_xclk_we,
		output	reg	[(AW-1):0]	o_xclk_addr,
		output	reg	[(DW-1):0]	o_xclk_data,
		output	reg	[(DW/8-1):0]	o_xclk_sel,
		input	wire			i_xclk_stall,
		input	wire			i_xclk_ack,
		input	wire	[(DW-1):0]	i_xclk_data,
		input	wire			i_xclk_err
	);

	localparam FIFO_DEPTH = 1 << LGFIFO;
	localparam FIFO_MASK  = FIFO_DEPTH - 1;

	// Reset synchronizers
	reg [1:0] rst_sync_wb;
	reg [1:0] rst_sync_xclk;

	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset)
			rst_sync_wb <= 2'b00;
		else
			rst_sync_wb <= {rst_sync_wb[0], 1'b1};
	end

	always @(posedge i_xclk_clk or posedge i_reset) begin
		if (i_reset)
			rst_sync_xclk <= 2'b00;
		else
			rst_sync_xclk <= {rst_sync_xclk[0], 1'b1};
	end

	wire wb_rst_n = rst_sync_wb[1];
	wire xclk_rst_n = rst_sync_xclk[1];

	// FIFO storage
	reg [DW-1:0] data_fifo [0:FIFO_DEPTH-1];
	reg [AW-1:0] addr_fifo [0:FIFO_DEPTH-1];
	reg [DW/8-1:0] sel_fifo [0:FIFO_DEPTH-1];
	reg we_fifo [0:FIFO_DEPTH-1];

	reg [LGFIFO:0] wr_ptr;
	reg [LGFIFO:0] rd_ptr;

	// Synchronize write pointer to xclk domain
	reg [LGFIFO:0] wr_ptr_sync1;
	reg [LGFIFO:0] wr_ptr_sync2;
	reg [LGFIFO:0] wr_ptr_sync3;

	always @(posedge i_wb_clk) begin
		if (wb_rst_n)
			wr_ptr_sync1 <= wr_ptr;
	end

	always @(posedge i_xclk_clk) begin
		if (xclk_rst_n) begin
			wr_ptr_sync2 <= wr_ptr_sync1;
			wr_ptr_sync3 <= wr_ptr_sync2;
		end
	end

	// Synchronize read pointer to wb domain
	reg [LGFIFO:0] rd_ptr_sync1;
	reg [LGFIFO:0] rd_ptr_sync2;
	reg [LGFIFO:0] rd_ptr_sync3;

	always @(posedge i_xclk_clk) begin
		if (xclk_rst_n)
			rd_ptr_sync1 <= rd_ptr;
	end

	always @(posedge i_wb_clk) begin
		if (wb_rst_n) begin
			rd_ptr_sync2 <= rd_ptr_sync1;
			rd_ptr_sync3 <= rd_ptr_sync2;
		end
	end

	// Write side (wb_clk domain)
	wire wb_write = i_wb_cyc & i_wb_stb & ~o_wb_stall;

	always @(posedge i_wb_clk) begin
		if (wb_rst_n) begin
			if (wb_write) begin
				data_fifo[wr_ptr[LGFIFO-1:0]] <= i_wb_data;
				addr_fifo[wr_ptr[LGFIFO-1:0]] <= i_wb_addr;
				sel_fifo[wr_ptr[LGFIFO-1:0]] <= i_wb_sel;
				we_fifo[wr_ptr[LGFIFO-1:0]] <= i_wb_we;
				wr_ptr <= wr_ptr + 1'b1;
			end
		end
	end

	// Read side (xclk_clk domain)
	// Two-stage pipeline: stage 1 reads FIFO, stage 2 drives outputs
	reg [DW-1:0] stage1_data;
	reg [AW-1:0] stage1_addr;
	reg [DW/8-1:0] stage1_sel;
	reg stage1_we;
	reg stage1_valid;

	reg [DW-1:0] stage2_data;
	reg [AW-1:0] stage2_addr;
	reg [DW/8-1:0] stage2_sel;
	reg stage2_we;
	reg stage2_valid;

	wire fifo_empty = (wr_ptr_sync3 == rd_ptr);
	wire can_read = xclk_rst_n & ~fifo_empty & ~i_xclk_stall;

	always @(posedge i_xclk_clk) begin
		if (xclk_rst_n) begin
			if (can_read) begin
				stage1_data <= data_fifo[rd_ptr[LGFIFO-1:0]];
				stage1_addr <= addr_fifo[rd_ptr[LGFIFO-1:0]];
				stage1_sel <= sel_fifo[rd_ptr[LGFIFO-1:0]];
				stage1_we <= we_fifo[rd_ptr[LGFIFO-1:0]];
				stage1_valid <= 1'b1;
				rd_ptr <= rd_ptr + 1'b1;
			end
			else begin
				stage1_valid <= 1'b0;
			end
		end
	end

	always @(posedge i_xclk_clk) begin
		if (xclk_rst_n) begin
			stage2_data <= stage1_data;
			stage2_addr <= stage1_addr;
			stage2_sel <= stage1_sel;
			stage2_we <= stage1_we;
			stage2_valid <= stage1_valid;
		end
	end

	// Drive xclk outputs
	always @(posedge i_xclk_clk) begin
		if (xclk_rst_n) begin
			if (stage2_valid) begin
				o_xclk_cyc <= 1'b1;
				o_xclk_stb <= 1'b1;
				o_xclk_we <= stage2_we;
				o_xclk_addr <= stage2_addr;
				o_xclk_data <= stage2_data;
				o_xclk_sel <= stage2_sel;
			end
			else begin
				o_xclk_cyc <= 1'b0;
				o_xclk_stb <= 1'b0;
				o_xclk_we <= 1'b0;
				o_xclk_addr <= {AW{1'b0}};
				o_xclk_data <= {DW{1'b0}};
				o_xclk_sel <= {(DW/8){1'b0}};
			end
		end
	end

	// Response FIFO (xclk_clk domain)
	reg [DW-1:0] resp_data_fifo [0:FIFO_DEPTH-1];
	reg resp_err_fifo [0:FIFO_DEPTH-1];
	reg [LGFIFO:0] resp_wr_ptr;
	reg [LGFIFO:0] resp_rd_ptr;

	// Synchronize response write pointer to wb domain
	reg [LGFIFO:0] resp_wr_ptr_sync1;
	reg [LGFIFO:0] resp_wr_ptr_sync2;
	reg [LGFIFO:0] resp_wr_ptr_sync3;

	always @(posedge i_xclk_clk) begin
		if (xclk_rst_n)
			resp_wr_ptr_sync1 <= resp_wr_ptr;
	end

	always @(posedge i_wb_clk) begin
		if (wb_rst_n) begin
			resp_wr_ptr_sync2 <= resp_wr_ptr_sync1;
			resp_wr_ptr_sync3 <= resp_wr_ptr_sync2;
		end
	end

	// Synchronize response read pointer to xclk domain
	reg [LGFIFO:0] resp_rd_ptr_sync1;
	reg [LGFIFO:0] resp_rd_ptr_sync2;
	reg [LGFIFO:0] resp_rd_ptr_sync3;

	always @(posedge i_wb_clk) begin
		if (wb_rst_n)
			resp_rd_ptr_sync1 <= resp_rd_ptr;
	end

	always @(posedge i_xclk_clk) begin
		if (xclk_rst_n) begin
			resp_rd_ptr_sync2 <= resp_rd_ptr_sync1;
			resp_rd_ptr_sync3 <= resp_rd_ptr_sync2;
		end
	end

	// Write response on ack
	wire resp_write = i_xclk_ack & xclk_rst_n;

	always @(posedge i_xclk_clk) begin
		if (xclk_rst_n) begin
		if (resp_write) begin
			resp_data_fifo[resp_wr_ptr[LGFIFO-1:0]] <= i_xclk_data;
			resp_err_fifo[resp_wr_ptr[LGFIFO-1:0]] <= i_xclk_err;
			resp_wr_ptr <= resp_wr_ptr + 1'b1;
		end
	end
	end

	// Read response (wb_clk domain)
	wire resp_empty = (resp_wr_ptr_sync3 == resp_rd_ptr);
	wire resp_read = wb_rst_n & ~resp_empty;

	always @(posedge i_wb_clk) begin
		if (wb_rst_n) begin
			if (resp_read) begin
				o_wb_ack <= 1'b1;
				o_wb_data <= resp_data_fifo[resp_rd_ptr[LGFIFO-1:0]];
				o_wb_err <= resp_err_fifo[resp_rd_ptr[LGFIFO-1:0]];
				resp_rd_ptr <= resp_rd_ptr + 1'b1;
			end
			else begin
				o_wb_ack <= 1'b0;
			end
		end
	end

	// Stall signal
	assign o_wb_stall = (wr_ptr - rd_ptr_sync3) >= (FIFO_DEPTH - 1);

endmodule
