module wbxclk #(
		parameter	AW=32,
				DW=32,
				LGFIFO = 5
	) (
		input	wire			i_wb_clk, i_reset,
		input	wire			i_wb_cyc, i_wb_stb, i_wb_we,
		input	wire	[(AW-1):0]	i_wb_addr,
		input	wire	[(DW-1):0]	i_wb_data,
		input	wire	[(DW/8-1):0]	i_wb_sel,
		output	wire			o_wb_stall,
		output	reg			o_wb_ack,
		output	reg	[(DW-1):0]	o_wb_data,
		output	reg			o_wb_err,
		input	wire			i_xclk_clk,
		output	reg			o_xclk_cyc,
		output	reg			o_xclk_stb,
		output	reg			o_xclk_we,
		output	reg	[(AW-1):0]	o_xclk_addr,
		output	reg	[(DW-1):0]	o_xclk_data,
		output	reg	[(DW/8-1):0]	o_xclk_sel,
		input	wire			i_xclk_stall,
		input	wire			i_xclk_ack,
		input	wire	[(DW-1):0]	i_xclk_data,
		input	wire			i_xclk_err
	);

	localparam FIFO_DEPTH = 1 << LGFIFO;
	localparam FIFO_WIDTH = AW + DW + DW/8 + 1; // addr + data + sel + we

	// Request FIFO (wb -> xclk)
	reg [FIFO_WIDTH-1:0] req_fifo [0:FIFO_DEPTH-1];
	reg [LGFIFO:0] req_wr_ptr, req_rd_ptr;
	wire req_full, req_empty;
	wire req_wr_en, req_rd_en;

	// Response FIFO (xclk -> wb)
	reg [DW+1:0] rsp_fifo [0:FIFO_DEPTH-1]; // data + err
	reg [LGFIFO:0] rsp_wr_ptr, rsp_rd_ptr;
	wire rsp_full, rsp_empty;
	wire rsp_wr_en, rsp_rd_en;

	// Request FIFO full/empty
	assign req_full  = (req_wr_ptr[LGFIFO] != req_rd_ptr[LGFIFO]) &&
	                   (req_wr_ptr[LGFIFO-1:0] == req_rd_ptr[LGFIFO-1:0]);
	assign req_empty = (req_wr_ptr == req_rd_ptr);

	// Response FIFO full/empty
	assign rsp_full  = (rsp_wr_ptr[LGFIFO] != rsp_rd_ptr[LGFIFO]) &&
	                   (rsp_wr_ptr[LGFIFO-1:0] == rsp_rd_ptr[LGFIFO-1:0]);
	assign rsp_empty = (rsp_wr_ptr == rsp_rd_ptr);

	// WB side: accept when not full
	assign req_wr_en = i_wb_cyc && i_wb_stb && !req_full;
	assign o_wb_stall = req_full;

	// XCLK side: read when not empty and not stalled
	assign req_rd_en = !req_empty && !i_xclk_stall;

	// Response FIFO write from xclk side
	assign rsp_wr_en = i_xclk_ack;

	// Response FIFO read from wb side
	assign rsp_rd_en = !rsp_empty;

	// WB side: write to request FIFO
	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset) begin
			req_wr_ptr <= 0;
		end else if (req_wr_en) begin
			req_fifo[req_wr_ptr[LGFIFO-1:0]] <= {i_wb_we, i_wb_addr, i_wb_data, i_wb_sel};
			req_wr_ptr <= req_wr_ptr + 1;
		end
	end

	// XCLK side: read from request FIFO
	always @(posedge i_xclk_clk or posedge i_reset) begin
		if (i_reset) begin
			req_rd_ptr <= 0;
			o_xclk_cyc <= 0;
			o_xclk_stb <= 0;
			o_xclk_we  <= 0;
			o_xclk_addr <= 0;
			o_xclk_data <= 0;
			o_xclk_sel  <= 0;
		end else if (req_rd_en) begin
			req_rd_ptr <= req_rd_ptr + 1;
			o_xclk_cyc  <= 1;
			o_xclk_stb  <= 1;
			o_xclk_we   <= req_fifo[req_rd_ptr[LGFIFO-1:0]][FIFO_WIDTH-1];
			o_xclk_addr <= req_fifo[req_rd_ptr[LGFIFO-1:0]][FIFO_WIDTH-2:FIFO_WIDTH-1-DW];
			o_xclk_data <= req_fifo[req_rd_ptr[LGFIFO-1:0]][FIFO_WIDTH-1-DW:FIFO_WIDTH-1-DW-DW/8];
			o_xclk_sel  <= req_fifo[req_rd_ptr[LGFIFO-1:0]][FIFO_WIDTH-1-DW-DW/8:0];
		end else begin
			o_xclk_cyc <= 0;
			o_xclk_stb <= 0;
		end
	end

	// XCLK side: write to response FIFO
	always @(posedge i_xclk_clk or posedge i_reset) begin
		if (i_reset) begin
			rsp_wr_ptr <= 0;
		end else if (rsp_wr_en) begin
			rsp_fifo[rsp_wr_ptr[LGFIFO-1:0]] <= {i_xclk_err, i_xclk_data};
			rsp_wr_ptr <= rsp_wr_ptr + 1;
		end
	end

	// WB side: read from response FIFO
	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset) begin
			rsp_rd_ptr <= 0;
			o_wb_ack   <= 0;
			o_wb_data  <= 0;
			o_wb_err   <= 0;
		end else if (rsp_rd_en) begin
			rsp_rd_ptr <= rsp_rd_ptr + 1;
			o_wb_ack   <= 1;
			o_wb_data  <= rsp_fifo[rsp_rd_ptr[LGFIFO-1:0]][DW:0];
			o_wb_err   <= rsp_fifo[rsp_rd_ptr[LGFIFO-1:0]][DW+1];
		end else begin
			o_wb_ack   <= 0;
		end
	end

endmodule
