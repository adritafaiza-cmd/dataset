module wbxclk #(
	parameter	AW=32,
			DW=32,
			LGFIFO = 5
) (
	input	wire			i_wb_clk, i_reset,
	input	wire			i_wb_cyc, i_wb_stb, i_wb_we,
	input	wire	[(AW-1):0]	i_wb_addr,
	input	wire	[(DW-1):0]	i_wb_data,
	input	wire	[(DW/8-1):0]	i_wb_sel,
	output	wire			o_wb_stall,
	output	reg			o_wb_ack,
	output	reg	[(DW/8-1):0]	o_wb_data,
	output	reg			o_wb_err,
	input	wire			i_xclk_clk,
	output	reg			o_xclk_cyc,
	output	reg			o_xclk_stb,
	output	reg			o_xclk_we,
	output	reg	[(AW-1):0]	o_xclk_addr,
	output	reg	[(DW/8-1):0]	o_xclk_data,
	output	reg	[(DW/8-1):0]	o_xclk_sel,
	input	wire			i_xclk_stall,
	input	wire			i_xclk_ack,
	input	wire	[(DW/8-1):0]	i_xclk_data,
	input	wire			i_xclk_err
);

	localparam FIFO_DEPTH = 1 << LGFIFO;
	localparam FIFO_WIDTH = AW + DW + DW/8 + 1; // addr + data + sel + we
	localparam FIFO_ADDR_WIDTH = LGFIFO;

	// Request FIFO: WB side writes, XCLK side reads
	reg [FIFO_WIDTH-1:0] req_fifo [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH:0] req_wr_ptr, req_rd_ptr;
	wire req_full = (req_wr_ptr[FIFO_ADDR_WIDTH] != req_rd_ptr[FIFO_ADDR_WIDTH]) &&
	                 (req_wr_ptr[FIFO_ADDR_WIDTH-1:0] == req_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
	wire req_empty = (req_wr_ptr == req_rd_ptr);

	// Response FIFO: XCLK side writes, WB side reads
	reg [DW+1:0] resp_fifo [0:FIFO_DEPTH-1]; // data + err
	reg [FIFO_ADDR_WIDTH:0] resp_wr_ptr, resp_rd_ptr;
	wire resp_full = (resp_wr_ptr[FIFO_ADDR_WIDTH] != resp_rd_ptr[FIFO_ADDR_WIDTH]) &&
	                  (resp_wr_ptr[FIFO_ADDR_WIDTH-1:0] == resp_rd_ptr[FIFO_ADDR_WIDTH-1:0]);
	wire resp_empty = (resp_wr_ptr == resp_rd_ptr);

	// WB side: accept request when not full
	wire wb_accept = i_wb_cyc && i_wb_stb && !req_full;
	assign o_wb_stall = req_full;

	// Write to request FIFO on WB clock
	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset) begin
			req_wr_ptr <= 0;
		end else if (wb_accept) begin
			req_fifo[req_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= {i_wb_we, i_wb_addr, i_wb_data, i_wb_sel};
			req_wr_ptr <= req_wr_ptr + 1;
		end
	end

	// XCLK side: read from request FIFO and drive Wishbone
	reg xclk_active;
	reg [FIFO_WIDTH-1:0] xclk_req_reg;

	always @(posedge i_xclk_clk or posedge i_reset) begin
		if (i_reset) begin
			req_rd_ptr <= 0;
			xclk_active <= 0;
			o_xclk_cyc <= 0;
			o_xclk_stb <= 0;
			o_xclk_we <= 0;
			o_xclk_addr <= 0;
			o_xclk_data <= 0;
			o_xclk_sel <= 0;
		end else begin
			if (!xclk_active) begin
				if (!req_empty) begin
					xclk_req_reg <= req_fifo[req_rd_ptr[FIFO_ADDR_WIDTH-1:0]];
					req_rd_ptr <= req_rd_ptr + 1;
					xclk_active <= 1;
					o_xclk_cyc <= 1;
					o_xclk_stb <= 1;
					o_xclk_we <= xclk_req_reg[FIFO_WIDTH-1];
					o_xclk_addr <= xclk_req_reg[FIFO_WIDTH-2:FIFO_WIDTH-1-AW];
					o_xclk_data <= xclk_req_reg[FIFO_WIDTH-1-AW:FIFO_WIDTH-1-AW-DW];
					o_xclk_sel <= xclk_req_reg[FIFO_WIDTH-1-AW-DW:FIFO_WIDTH-1-AW-DW-DW/8];
				end else begin
					o_xclk_cyc <= 0;
					o_xclk_stb <= 0;
				end
			end else begin
				if (i_xclk_ack) begin
					xclk_active <= 0;
					o_xclk_cyc <= 0;
					o_xclk_stb <= 0;
				end
			end
		end
	end

	// Write to response FIFO on XCLK clock when ack
	always @(posedge i_xclk_clk or posedge i_reset) begin
		if (i_reset) begin
			resp_wr_ptr <= 0;
		end else if (i_xclk_ack) begin
			resp_fifo[resp_wr_ptr[FIFO_ADDR_WIDTH-1:0]] <= {i_xclk_err, i_xclk_data};
			resp_wr_ptr <= resp_wr_ptr + 1;
		end
	end

	// WB side: read from response FIFO and drive ack
	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset) begin
			resp_rd_ptr <= 0;
			o_wb_ack <= 0;
			o_wb_data <= 0;
			o_wb_err <= 0;
		end else begin
			if (!resp_empty) begin
				o_wb_ack <= 1;
				o_wb_data <= resp_fifo[resp_rd_ptr[FIFO_ADDR_WIDTH-1:0]][DW:0];
				o_wb_err <= resp_fifo[resp_rd_ptr[FIFO_ADDR_WIDTH-1:0]][DW+1];
				resp_rd_ptr <= resp_rd_ptr + 1;
			end else begin
				o_wb_ack <= 0;
			end
		end
	end

endmodule
