  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain signals
  reg [WIDTH-1:0] src_data_reg;
  reg             src_valid_reg;
  reg             src_ready_reg;
  reg             src_clear_pending_reg;
  reg             src_clear_ack_reg;
  reg             src_clear_req_reg;
  reg             src_clear_ack_sync;
  reg             src_clear_ack_sync2;
  reg             src_clear_ack_sync3;

  reg [WIDTH-1:0] dst_data_reg;
  reg             dst_valid_reg;
  reg             dst_ready_reg;
  reg             dst_clear_pending_reg;
  reg             dst_clear_ack_reg;
  reg             dst_clear_req_reg;
  reg             dst_clear_ack_sync;
  reg             dst_clear_ack_sync2;
  reg             dst_clear_ack_sync3;

  wire            src_valid_to_dst;
  wire            src_ready_to_dst;
  wire            dst_valid_to_src;
  wire            dst_ready_to_src;
  wire            src_clear_req_to_dst;
  wire            dst_clear_req_to_src;
  wire            src_clear_ack_to_dst;
  wire            dst_clear_ack_to_src;

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_clear_ack_sync  <= 1'b0;
      src_clear_ack_sync2 <= 1'b0;
      src_clear_ack_sync3 <= 1'b0;
    end else begin
      src_clear_ack_sync  <= dst_clear_ack_to_src;
      src_clear_ack_sync2 <= src_clear_ack_sync;
      src_clear_ack_sync3 <= src_clear_ack_sync2;
    end
  end

  reg dst_valid_sync3;

      dst_valid_sync  <= 1'b0;
      dst_valid_sync2 <= 1'b0;
      dst_valid_sync3 <= 1'b0;

      dst_valid_sync  <= dst_valid_to_src;
      dst_valid_sync2 <= dst_valid_sync;
      dst_valid_sync3 <= dst_valid_sync2;
    end
  end

      dst_ready_sync  <= 1'b0;
      dst_ready_sync2 <= 1'b0;
      dst_ready_sync3 <= 1'b0;
    end else begin
      dst_ready_sync  <= dst_ready_to_src;
      dst_ready_sync2 <= dst_ready_sync;
      dst_ready_sync3 <= dst_ready_sync2;
    end
  end

  // Source domain: main logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_reg          <= {WIDTH{1'b0}};
      src_valid_reg         <= 1'b0;
      src_ready_reg         <= 1'b0;
      src_clear_pending_reg <= 1'b0;
      src_clear_ack_reg     <= 1'b0;
      src_clear_req_reg     <= 1'b0;
    end else begin
      // Clear handling
      if (src_clear_i && !src_clear_pending_reg) begin
        src_clear_pending_reg <= 1'b1;
        src_clear_req_reg     <= 1'b1;
      end else if (src_clear_pending_reg && src_clear_ack_sync3) begin
        src_clear_pending_reg <= 1'b0;
        src_clear_req_reg     <= 1'b0;
        src_clear_ack_reg     <= 1'b0;
        src_valid_reg         <= 1'b0;
        src_ready_reg         <= 1'b0;
      end else if (src_clear_pending_reg) begin
        // Wait for ack

        // Normal operation
        if (src_valid_i && src_ready_reg) begin
          src_data_reg  <= src_data_i;
          src_valid_reg <= 1'b1;
          src_ready_reg <= 1'b0;

          src_valid_reg <= 1'b0;
          src_ready_reg <= 1'b1;
        end else if (!src_valid_reg && !src_valid_i) begin
          src_ready_reg <= 1'b1;
        end
      end
    end
  end

  assign src_ready_to_dst     = src_ready_reg;
  assign src_clear_req_to_dst = src_clear_req_reg;
  assign src_clear_ack_to_dst = src_clear_ack_reg;

      src_valid_sync2 <= 1'b0;
      src_valid_sync3 <= 1'b0;
    end else begin
      src_valid_sync  <= src_valid_to_dst;
      src_valid_sync2 <= src_valid_sync;
      src_valid_sync3 <= src_valid_sync2;
    end
  end

  reg src_ready_sync2;
  reg src_ready_sync3;
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_ready_sync  <= 1'b0;
      src_ready_sync2 <= 1'b0;
      src_ready_sync3 <= 1'b0;
    end else begin
      src_ready_sync  <= src_ready_to_dst;
      src_ready_sync2 <= src_ready_sync;
      src_ready_sync3 <= src_ready_sync2;
    end
  end

  // Destination domain: synchronize src_clear_req
  reg src_clear_req_sync;
  reg src_clear_req_sync2;
  reg src_clear_req_sync3;
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin

      src_clear_req_sync  <= 1'b0;

    end
  end

  reg src_clear_ack_sync2;
  reg src_clear_ack_sync3;
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_clear_ack_sync  <= 1'b0;
      src_clear_ack_sync2 <= 1'b0;
      src_clear_ack_sync3 <= 1'b0;

      src_clear_ack_sync  <= src_clear_ack_to_dst;
      src_clear_ack_sync2 <= src_clear_ack_sync;
      src_clear_ack_sync3 <= src_clear_ack_sync2;
    end
  end

  // Destination domain: main logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg          <= {WIDTH{1'b0}};
      dst_valid_reg         <= 1'b0;
      dst_ready_reg         <= 1'b0;
      dst_clear_pending_reg <= 1'b0;
      dst_clear_ack_reg     <= 1'b0;
      dst_clear_req_reg     <= 1'b0;
    end else begin
      // Clear handling
      if (dst_clear_i && !dst_clear_pending_reg) begin
        dst_clear_pending_reg <= 1'b1;
        dst_clear_req_reg     <= 1'b1;
      end else if (dst_clear_pending_reg && src_clear_ack_sync3) begin
        dst_clear_pending_reg <= 1'b0;
        dst_clear_req_reg     <= 1'b0;
        dst_clear_ack_reg     <= 1'b0;
        dst_valid_reg         <= 1'b0;
        dst_ready_reg         <= 1'b0;
      end else if (dst_clear_pending_reg) begin
        // Wait for ack
      end else begin
        // Normal operation
        if (src_valid_sync3 && !dst_valid_reg) begin
          dst_data_reg  <= src_data_reg;
          dst_valid_reg <= 1'b1;
          dst_ready_reg <= 1'b0;
        end else if (dst_valid_reg && dst_ready_i) begin
          dst_valid_reg <= 1'b0;
          dst_ready_reg <= 1'b1;
        end else if (!dst_valid_reg && !src_valid_sync3) begin
          dst_ready_reg <= 1'b1;
        end
      end
    end
  end

  // Destination domain: drive cross-domain signals
  assign dst_valid_to_src     = dst_valid_reg;
  assign dst_ready_to_src     = dst_ready_reg;
  assign dst_clear_req_to_src = dst_clear_req_reg;

  // Destination domain: outputs
  assign dst_data_o          = dst_data_reg;
  assign dst_valid_o         = dst_valid_reg;
  assign dst_clear_pending_o = dst_clear_pending_reg;
