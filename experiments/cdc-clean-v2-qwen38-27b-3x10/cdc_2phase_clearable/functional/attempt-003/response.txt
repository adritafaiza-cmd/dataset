module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Internal signals
  wire src_req;
  wire dst_ack;
  wire dst_req;
  wire src_ack;

  // Source domain: request generation
  reg src_req_reg;
  reg src_ack_reg;
  reg src_clear_pending_reg;
  reg src_isolated_reg;

  // Destination domain: acknowledge generation
  reg dst_ack_reg;
  reg dst_req_reg;
  reg dst_ack_reg_next;
  reg dst_clear_pending_reg;
  reg dst_isolated_reg;
  reg [WIDTH-1:0] dst_data_reg;
  reg dst_valid_reg;

  // Synchronizers
  wire [SYNC_STAGES-1:0] src_req_sync;
  wire [SYNC_STAGES-1:0] dst_ack_sync;
  wire [SYNC_STAGES-1:0] dst_req_sync;
  wire [SYNC_STAGES-1:0] src_ack_sync;

  // Source domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_req_reg <= 1'b0;
      src_ack_reg <= 1'b0;
      src_clear_pending_reg <= 1'b0;
      src_isolated_reg <= 1'b0;
    end else begin
      // Clear handling
      if (src_clear_i && !src_isolated_reg) begin
        src_isolated_reg <= 1'b1;
        src_clear_pending_reg <= 1'b1;
      end

      // Clear completion: when dst_ack is deasserted (synced back)
      if (src_isolated_reg && !dst_ack_sync[SYNC_STAGES-1]) begin
        src_isolated_reg <= 1'b0;
        src_clear_pending_reg <= 1'b0;
        src_req_reg <= 1'b0;
        src_ack_reg <= 1'b0;
      end

      // Normal operation
      if (!src_isolated_reg) begin
        // Two-phase handshake: request toggles when valid and ready
        // Ready is high when not in flight (ack matches req)
        if (src_valid_i && (src_ack_reg == src_req_reg)) begin
          src_req_reg <= ~src_req_reg;
        end
        // Acknowledge from destination
        src_ack_reg <= dst_ack_sync[SYNC_STAGES-1];
      end
    end
  end

  assign src_req = src_req_reg;
  assign src_ready_o = !src_isolated_reg && (src_ack_reg == src_req_reg);
  assign src_clear_pending_o = src_clear_pending_reg;

  // Destination domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_ack_reg <= 1'b0;
      dst_req_reg <= 1'b0;
      dst_clear_pending_reg <= 1'b0;
      dst_isolated_reg <= 1'b0;
      dst_data_reg <= {WIDTH{1'b0}};
      dst_valid_reg <= 1'b0;
    end else begin
      // Clear handling
      if (dst_clear_i && !dst_isolated_reg) begin
        dst_isolated_reg <= 1'b1;
        dst_clear_pending_reg <= 1'b1;
      end

      // Clear completion: when src_req is deasserted (synced back)
      if (dst_isolated_reg && !src_req_sync[SYNC_STAGES-1]) begin
        dst_isolated_reg <= 1'b0;
        dst_clear_pending_reg <= 1'b0;
        dst_ack_reg <= 1'b0;
        dst_req_reg <= 1'b0;
        dst_valid_reg <= 1'b0;
      end

      // Normal operation
      if (!dst_isolated_reg) begin
        // Capture data on request edge
        if (src_req_sync[SYNC_STAGES-1] != dst_req_reg) begin
          dst_req_reg <= src_req_sync[SYNC_STAGES-1];
          dst_data_reg <= src_data_i; // Data is stable during transfer
          dst_valid_reg <= 1'b1;
        end

        // Acknowledge when ready
        if (dst_valid_reg && dst_ready_i) begin
          dst_ack_reg <= ~dst_ack_reg;
          dst_valid_reg <= 1'b0;
        end
      end
    end
  end

  assign dst_ack = dst_ack_reg;
  assign dst_req = dst_req_reg;
  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;
  assign dst_clear_pending_o = dst_clear_pending_reg;

  // Synchronizers: src_req to dst domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_req_sync <= {SYNC_STAGES{1'b0}};
    end else begin
      src_req_sync <= {src_req_sync[SYNC_STAGES-2:0], src_req};
    end
  end

  // Synchronizers: dst_ack to src domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_ack_sync <= {SYNC_STAGES{1'b0}};
    end else begin
      dst_ack_sync <= {dst_ack_sync[SYNC_STAGES-2:0], dst_ack};
    end
  end

  // Synchronizers: dst_req to src domain (not directly used but for completeness)
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_req_sync <= {SYNC_STAGES{1'b0}};
    end else begin
      dst_req_sync <= {dst_req_sync[SYNC_STAGES-2:0], dst_req};
    end
  end

  // Synchronizers: src_ack to dst domain (not directly used but for completeness)
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_ack_sync <= {SYNC_STAGES{1'b0}};
    end else begin
      src_ack_sync <= {src_ack_sync[SYNC_STAGES-2:0], src_ack_reg};
    end
  end

endmodule
