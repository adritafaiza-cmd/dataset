```systemverilog
module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Internal signals
  wire src_req;
  wire src_ack;
  wire dst_req;
  wire dst_ack;

  // Source domain: request generation
  reg src_req_reg;
  reg src_ack_reg;
  reg src_clear_pending_reg;
  reg src_isolated;

  // Destination domain: acknowledge generation
  reg dst_ack_reg;
  reg dst_req_reg;
  reg dst_clear_pending_reg;
  reg dst_isolated;
  reg [WIDTH-1:0] dst_data_reg;
  reg dst_valid_reg;

  // Synchronizers
  wire [SYNC_STAGES-1:0] src_req_sync;
  wire [SYNC_STAGES-1:0] src_ack_sync;
  wire [SYNC_STAGES-1:0] dst_req_sync;
  wire [SYNC_STAGES-1:0] dst_ack_sync;

  // Source domain logic
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_req_reg <= 1'b0;
      src_ack_reg <= 1'b0;
      src_clear_pending_reg <= 1'b0;
      src_isolated <= 1'b0;
    end else begin
      // Clear handling
      if (src_clear_i && !src_clear_pending_reg) begin
        src_clear_pending_reg <= 1'b1;
        src_isolated <= 1'b1;
      end else if (src_clear_pending_reg) begin
        // Wait for ack to go low (clear complete)
        if (!src_ack_sync[SYNC_STAGES-1]) begin
          src_clear_pending_reg <= 1'b0;
          src_isolated <= 1'b0;
          src_req_reg <= 1'b0;
          src_ack_reg <= 1'b0;
        end
      end

      // Normal handshake
      if (!src_isolated && !src_clear_pending_reg) begin
        if (src_valid_i && src_ready_o && !src_req_reg) begin
          src_req_reg <= 1'b1;
        end else if (src_req_reg && src_ack_sync[SYNC_STAGES-1]) begin
          src_req_reg <= 1'b0;
          src_ack_reg <= 1'b1;
        end
      end
    end
  end

  assign src_req = src_req_reg;
  assign src_ack = src_ack_sync[SYNC_STAGES-1];
  assign src_ready_o = !src_req_reg && !src_isolated && !src_clear_pending_reg;
  assign src_clear_pending_o = src_clear_pending_reg;

  // Destination domain logic
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_ack_reg <= 1'b0;
      dst_req_reg <= 1'b0;
      dst_clear_pending_reg <= 1'b0;
      dst_isolated <= 1'b0;
      dst_data_reg <= {WIDTH{1'b0}};
      dst_valid_reg <= 1'b0;
    end else begin
      // Clear handling
      if (dst_clear_i && !dst_clear_pending_reg) begin
        dst_clear_pending_reg <= 1'b1;
        dst_isolated <= 1'b1;
      end else if (dst_clear_pending_reg) begin
        // Wait for req to go low (clear complete)
        if (!dst_req_sync[SYNC_STAGES-1]) begin
          dst_clear_pending_reg <= 1'b0;
          dst_isolated <= 1'b0;
          dst_ack_reg <= 1'b0;
          dst_req_reg <= 1'b0;
          dst_valid_reg <= 1'b0;
        end
      end

      // Normal handshake
      if (!dst_isolated && !dst_clear_pending_reg) begin
        if (dst_req_sync[SYNC_STAGES-1] && !dst_ack_reg) begin
          dst_ack_reg <= 1'b1;
          dst_data_reg <= dst_req_sync[SYNC_STAGES-1] ? dst_data_reg : dst_data_reg; // Data captured via separate path
          dst_valid_reg <= 1'b1;
        end else if (dst_ack_reg && dst_ready_i) begin
          dst_ack_reg <= 1'b0;
          dst_valid_reg <= 1'b0;
        end
      end
    end
  end

  assign dst_ack = dst_ack_reg;
  assign dst_req = dst_req_sync[SYNC_STAGES-1];
  assign dst_valid_o = dst_valid_reg;
  assign dst_data_o = dst_data_reg;
  assign dst_clear_pending_o = dst_clear_pending_reg;

  // Synchronizers for src_req to dst domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_req_sync <= {SYNC_STAGES{1'b0}};
    end else begin
      dst_req_sync <= {dst_req_sync[SYNC_STAGES-2:0], src_req};
    end
  end

  // Synchronizers for src_ack to src domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ack_sync <= {SYNC_STAGES{1'b0}};
    end else begin
      src_ack_sync <= {src_ack_sync[SYNC_STAGES-2:0], dst_ack};
    end
  end

  // Synchronizers for dst_req to dst domain (already done above)
  // Synchronizers for dst_ack to src domain (already done above)

  // Data path: need to synchronize data along with request
  // Use a data synchronizer that captures data when request is high
  reg [WIDTH-1:0] dst_data_sync;
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_sync <= {WIDTH{1'b0}};
    end else begin
      if (dst_req_sync[SYNC_STAGES-1] && !dst_req_sync[SYNC_STAGES-2]) begin
        // Rising edge of synchronized request
        dst_data_sync <= src_data_i;
      end
    end
  end

  // Update dst_data_reg from dst_data_sync
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg <= {WIDTH{1'b0}};
    end else begin
      if (dst_req_sync[SYNC_STAGES-1] && !dst_ack_reg) begin
        dst_data_reg <= dst_data_sync;
      end
    end
  end

endmodule
