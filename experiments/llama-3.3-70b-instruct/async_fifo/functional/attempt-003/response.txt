```verilog
module async_fifo #(
  parameter DSIZE = 8,
  parameter ASIZE = 4,
  parameter FALLTHROUGH = "TRUE"
)(
  input  wire             wclk,
  input  wire             wrst_n,
  input  wire             winc,
  input  wire [DSIZE-1:0] wdata,
  output wire             wfull,
  output wire             awfull,

  input  wire             rclk,
  input  wire             rrst_n,
  input  wire             rinc,
  output wire [DSIZE-1:0] rdata,
  output wire             rempty,
  output wire             arempty
);

  reg [DSIZE-1:0] mem [2**ASIZE-1:0];
  reg [ASIZE-1:0] wptr, rptr;
  reg [ASIZE-1:0] wptr_gray, rptr_gray;
  reg [ASIZE-1:0] wptr_sync, rptr_sync;
  reg [ASIZE:0]  wcnt, rcnt;
  reg             wfull_reg, rempty_reg;
  reg             awfull_reg, arempty_reg;

  assign wfull = wfull_reg;
  assign awfull = awfull_reg;
  assign rempty = rempty_reg;
  assign arempty = arempty_reg;

  always @(posedge wclk or negedge wrst_n) begin
    if (~wrst_n) begin
      wptr <= 0;
      wcnt <= 0;
      wfull_reg <= 0;
      awfull_reg <= 0;
    end else if (winc && ~wfull_reg) begin
      mem[wptr] <= wdata;
      wptr <= wptr + 1;
      wcnt <= wcnt + 1;
      if (wcnt == 2**ASIZE - 1) begin
        wfull_reg <= 1;
      end
      if (wcnt == 2**ASIZE - 2) begin
        awfull_reg <= 1;
      end
    end else if (~winc && wfull_reg) begin
      if (wcnt == 2**ASIZE - 1) begin
        wfull_reg <= 1;
      end
      if (wcnt == 2**ASIZE - 2) begin
        awfull_reg <= 1;
      end
    end else if (~winc && ~wfull_reg) begin
      if (wcnt == 2**ASIZE - 2) begin
        awfull_reg <= 0;
      end
      if (wcnt == 2**ASIZE - 1) begin
        wfull_reg <= 0;
      end
    end
  end

  always @(posedge rclk or negedge rrst_n) begin
    if (~rrst_n) begin
      rptr <= 0;
      rcnt <= 0;
      rempty_reg <= 1;
      arempty_reg <= 1;
    end else if (rinc && ~rempty_reg) begin
      rptr <= rptr + 1;
      rcnt <= rcnt - 1;
      if (rcnt == 1) begin
        rempty_reg <= 1;
      end
      if (rcnt == 2) begin
        arempty_reg <= 1;
      end
    end else if (~rinc && rempty_reg) begin
      if (rcnt == 1) begin
        rempty_reg <= 1;
      end
      if (rcnt == 2) begin
        arempty_reg <= 1;
      end
    end else if (~rinc && ~rempty_reg) begin
      if (rcnt == 2) begin
        arempty_reg <= 0;
      end
      if (rcnt == 1) begin
        rempty_reg <= 0;
      end
    end
  end

  always @(posedge wclk) begin
    wptr_gray <= {wptr[ASIZE-1], wptr[ASIZE-1:1]};
  end

  always @(posedge rclk) begin
    rptr_gray <= {rptr[ASIZE-1], rptr[ASIZE-1:1]};
  end

  always @(posedge wclk) begin
    wptr_sync <= wptr_gray;
  end

  always @(posedge rclk) begin
    rptr_sync <= rptr_gray;
  end

  always @(posedge rclk) begin
    if (FALLTHROUGH == "TRUE") begin
      rdata <= mem[rptr];
    end else if (rinc && ~rempty_reg) begin
      rdata <= mem[rptr];
    end
  end

  always @(posedge wclk) begin
    wcnt <= wcnt;
  end

  always @(posedge rclk) begin
    rcnt <= wcnt - rcnt;
  end

endmodule
```
