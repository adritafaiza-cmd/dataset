module cdc_2phase #(
  parameter WIDTH = 1
)(
  input                  src_rst_ni,
  input                  src_clk_i,
  input      [WIDTH-1:0] src_data_i,
  input                  src_valid_i,
  output                 src_ready_o,

  input                  dst_rst_ni,
  input                  dst_clk_i,
  output     [WIDTH-1:0] dst_data_o,
  output                 dst_valid_o,
  input                  dst_ready_i
);

  logic [WIDTH-1:0] data;
  logic             valid;

  // Synchronize src_rst_ni to src_clk_i
  logic src_rst_sync;
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_rst_sync <= 1'b0;
    end else begin
      src_rst_sync <= 1'b1;
    end
  end

  // Synchronize dst_rst_ni to dst_clk_i
  logic dst_rst_sync;
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_rst_sync <= 1'b0;
    end else begin
      dst_rst_sync <= 1'b1;
    end
  end

  // Reset logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ready_o <= 1'b0;
    end else begin
      if (src_valid_i && src_ready_o) begin
        // Accept new data
        src_ready_o <= 1'b0;
      end else if (!valid) begin
        // Ready to accept new data
        src_ready_o <= 1'b1;
      end
    end
  end

  always_ff @(posedge src_clk_i) begin
    if (src_rst_sync) begin
      if (src_valid_i && src_ready_o) begin
        // Capture new data
        data <= src_data_i;
        valid <= 1'b1;
      end
    end else begin
      valid <= 1'b0;
    end
  end

  // Synchronize data and valid to dst_clk_i
  logic [WIDTH-1:0] data_sync;
  logic             valid_sync;
  always_ff @(posedge dst_clk_i) begin
    data_sync <= data;
    valid_sync <= valid;
  end

  // Synchronize dst_ready_i to src_clk_i
  logic dst_ready_sync;
  always_ff @(posedge src_clk_i) begin
    dst_ready_sync <= dst_ready_i;
  end

  // Reset logic for dst side
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid_o <= 1'b0;
    end else begin
      if (valid_sync && dst_ready_i) begin
        // Transfer complete
        dst_valid_o <= 1'b0;
      end else if (valid_sync && !dst_valid_o) begin
        // Start transfer
        dst_valid_o <= 1'b1;
      end
    end
  end

  always_ff @(posedge dst_clk_i) begin
    if (dst_rst_sync) begin
      if (valid_sync && dst_ready_i) begin
        // Clear valid
        valid <= 1'b0;
      end
    end else begin
      valid <= 1'b0;
    end
  end

  assign dst_data_o = data_sync;

endmodule
