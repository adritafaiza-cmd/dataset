module cdc_2phase #(
  parameter WIDTH = 1
)(
  input                  src_rst_ni,
  input                  src_clk_i,
  input      [WIDTH-1:0] src_data_i,
  input                  src_valid_i,
  output                 src_ready_o,

  input                  dst_rst_ni,
  input                  dst_clk_i,
  output     [WIDTH-1:0] dst_data_o,
  output                 dst_valid_o,
  input                  dst_ready_i
);

  logic [WIDTH-1:0] fifo_data;
  logic             fifo_valid;
  logic             fifo_ready;

  logic             src_rst_sync;
  logic             dst_rst_sync;

  logic             src_clk_sync;
  logic             dst_clk_sync;

  logic             src_valid_sync;
  logic             src_data_sync [WIDTH-1:0];

  logic             dst_valid_sync;
  logic             dst_data_sync [WIDTH-1:0];

  logic             src_valid_pulse;
  logic             dst_valid_pulse;

  // Synchronize src_rst_ni to src_clk_i
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (~src_rst_ni) begin
      src_rst_sync <= 1'b1;
    end else begin
      src_rst_sync <= 1'b0;
    end
  end

  // Synchronize dst_rst_ni to dst_clk_i
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (~dst_rst_ni) begin
      dst_rst_sync <= 1'b1;
    end else begin
      dst_rst_sync <= 1'b0;
    end
  end

  // Synchronize src_valid_i and src_data_i to src_clk_i
  always_ff @(posedge src_clk_i) begin
    src_valid_sync <= src_valid_i;
    src_data_sync  <= src_data_i;
  end

  // Synchronize dst_valid_o and dst_data_o to dst_clk_i
  always_ff @(posedge dst_clk_i) begin
    dst_valid_sync <= dst_valid_o;
    dst_data_sync  <= dst_data_o;
  end

  // Generate src_valid_pulse
  always_ff @(posedge src_clk_i) begin
    if (src_rst_sync) begin
      src_valid_pulse <= 1'b0;
    end else if (src_valid_i && src_ready_o) begin
      src_valid_pulse <= 1'b1;
    end else begin
      src_valid_pulse <= 1'b0;
    end
  end

  // Generate dst_valid_pulse
  always_ff @(posedge dst_clk_i) begin
    if (dst_rst_sync) begin
      dst_valid_pulse <= 1'b0;
    end else if (dst_valid_o && dst_ready_i) begin
      dst_valid_pulse <= 1'b1;
    end else begin
      dst_valid_pulse <= 1'b0;
    end
  end

  // FIFO
  always_ff @(posedge src_clk_i) begin
    if (src_rst_sync) begin
      fifo_valid <= 1'b0;
      fifo_data  <= '0;
    end else if (src_valid_pulse) begin
      fifo_valid <= 1'b1;
      fifo_data  <= src_data_sync;
    end else if (fifo_ready) begin
      fifo_valid <= 1'b0;
    end
  end

  // FIFO ready
  always_ff @(posedge dst_clk_i) begin
    if (dst_rst_sync) begin
      fifo_ready <= 1'b0;
    end else if (dst_valid_pulse) begin
      fifo_ready <= 1'b1;
    end else if (~fifo_valid) begin
      fifo_ready <= 1'b0;
    end
  end

  // src_ready_o
  always_comb begin
    src_ready_o = ~fifo_valid;
  end

  // dst_valid_o and dst_data_o
  always_comb begin
    dst_valid_o = fifo_valid;
    dst_data_o  = fifo_data;
  end

endmodule
