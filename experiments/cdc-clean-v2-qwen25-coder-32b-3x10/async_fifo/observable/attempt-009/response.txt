module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    localparam DEPTH = 1 << ASIZE;
    localparam AWIDTH = ASIZE + 1;

    reg [DSIZE-1:0] mem [DEPTH-1:0];

    reg [AWIDTH-1:0] wptr, rptr;
    reg [AWIDTH-1:0] wptr_gray, rptr_gray;
    reg [AWIDTH-1:0] wptr_gray_sync, rptr_gray_sync;

    // Write side logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= 0;
        end else if (winc && !wfull) begin
            mem[wptr[ASIZE-1:0]] <= wdata;
            wptr <= wptr + 1;
        end
    end

    // Read side logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= 0;
        end else if (rinc && !rempty) begin
            rptr <= rptr + 1;
        end
    end

    // Gray code conversion for pointers
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr_gray <= 0;
        end else begin
            wptr_gray <= (wptr >> 1) ^ wptr;
        end
    end

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr_gray <= 0;
        end else begin
            rptr_gray <= (rptr >> 1) ^ rptr;
        end
    end

    // Cross-domain synchronization
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_sync <= 0;
        end else begin
            wptr_gray_sync <= wptr_gray;
        end
    end

    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_sync <= 0;
        end else begin
            rptr_gray_sync <= rptr_gray;
        end
    end

    // Binary conversion for synchronized pointers
    reg [AWIDTH-1:0] wptr_sync_bin, rptr_sync_bin;

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_sync_bin <= 0;
        end else begin
            wptr_sync_bin <= (wptr_gray_sync ^ (wptr_gray_sync >> 1));
        end
    end

    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_sync_bin <= 0;
        end else begin
            rptr_sync_bin <= (rptr_gray_sync ^ (rptr_gray_sync >> 1));
        end
    end

    // Full and empty flags
    wire [AWIDTH-1:0] wptr_plus1;
    assign wptr_plus1 = wptr + 1;

    wire [AWIDTH-1:0] wptr_plus1_sync_bin;
    assign wptr_plus1_sync_bin = (wptr_plus1 ^ (wptr_plus1 >> 1));

    assign wfull = (wptr_plus1_sync_bin == rptr_sync_bin);
    assign awfull = (wptr_gray == rptr_gray_sync);

    assign rempty = (wptr_sync_bin == rptr);
    assign arempty = (wptr_gray_sync == rptr_gray);

    // Read data output
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rdata <= 0;
        end else if (rinc && !rempty) begin
            rdata <= mem[rptr[ASIZE-1:0]];
        end
    end

endmodule
