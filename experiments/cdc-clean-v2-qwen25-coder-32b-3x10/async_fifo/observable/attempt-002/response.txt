module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    localparam DEPTH = 2**ASIZE;
    localparam AWIDTH = ASIZE + 1;

    reg [DSIZE-1:0] mem [0:DEPTH-1];

    reg [AWIDTH-1:0] wptr, rptr;
    reg [AWIDTH-1:0] wptr_gray, rptr_gray;
    reg [AWIDTH-1:0] rptr_gray_synced, wptr_gray_synced;

    // Write side logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= 0;
        end else if (winc && !wfull) begin
            mem[wptr[ASIZE-1:0]] <= wdata;
            wptr <= wptr + 1;
        end
    end

    // Read side logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= 0;
        end else if (rinc && !rempty) begin
            rptr <= rptr + 1;
        end
    end

    // Gray code conversion for pointers
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr_gray <= 0;
        end else begin
            wptr_gray <= (wptr >> 1) ^ wptr;
        end
    end

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr_gray <= 0;
        end else begin
            rptr_gray <= (rptr >> 1) ^ rptr;
        end
    end

    // Synchronize gray pointers across clock domains
    reg [AWIDTH-1:0] wptr_gray_synced_reg1, wptr_gray_synced_reg2;
    reg [AWIDTH-1:0] rptr_gray_synced_reg1, rptr_gray_synced_reg2;

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_synced_reg1 <= 0;
            wptr_gray_synced_reg2 <= 0;
        end else begin
            wptr_gray_synced_reg1 <= wptr_gray;
            wptr_gray_synced_reg2 <= wptr_gray_synced_reg1;
        end
    end

    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_synced_reg1 <= 0;
            rptr_gray_synced_reg2 <= 0;
        end else begin
            rptr_gray_synced_reg1 <= rptr_gray;
            rptr_gray_synced_reg2 <= rptr_gray_synced_reg1;
        end
    end

    assign wptr_gray_synced = wptr_gray_synced_reg2;
    assign rptr_gray_synced = rptr_gray_synced_reg2;

    // Binary conversion of synchronized gray pointers
    reg [AWIDTH-1:0] wptr_synced, rptr_synced;

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_synced <= 0;
        end else begin
            wptr_synced <= gray_to_bin(wptr_gray_synced);
        end
    end

    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_synced <= 0;
        end else begin
            rptr_synced <= gray_to_bin(rptr_gray_synced);
        end
    end

    // Gray to binary conversion function
    function [AWIDTH-1:0] gray_to_bin;
        input [AWIDTH-1:0] gray;
        integer i;
        begin
            gray_to_bin = gray[AWIDTH-1];
            for (i = AWIDTH-2; i >= 0; i = i - 1) begin
                gray_to_bin[i] = gray_to_bin[i+1] ^ gray[i];
            end
        end
    endfunction

    // Full and empty flags
    assign wfull = (wptr_synced[AWIDTH-1:0] == {rptr_gray_synced[AWIDTH-2:0], ~rptr_gray_synced[AWIDTH-1]});
    assign rempty = (rptr_synced[AWIDTH-1:0] == {wptr_gray_synced[AWIDTH-2:0], ~wptr_gray_synced[AWIDTH-1]});

    // Almost full and almost empty flags
    assign awfull = (wptr_synced[AWIDTH-1:0] == {rptr_gray_synced[AWIDTH-2:0], ~rptr_gray_synced[AWIDTH-1]} + 1);
    assign arempty = (rptr_synced[AWIDTH-1:0] == {wptr_gray_synced[AWIDTH-2:0], ~wptr_gray_synced[AWIDTH-1]} - 1);

    // Read data output
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rdata <= 0;
        end else if (rinc && !rempty) begin
            rdata <= mem[rptr[ASIZE-1:0]];
        end
    end

endmodule
