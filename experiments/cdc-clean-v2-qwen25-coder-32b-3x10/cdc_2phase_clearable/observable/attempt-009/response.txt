module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source side signals
  logic [WIDTH-1:0] src_data_sync;
  logic src_valid_sync;
  logic src_ready_sync;
  logic src_clear_sync;
  logic src_clear_pending;

  // Destination side signals
  logic [WIDTH-1:0] dst_data_reg;
  logic dst_valid_reg;
  logic dst_ready_reg;
  logic dst_clear_sync;
  logic dst_clear_pending;

  // Synchronization registers for clear signals
  logic [SYNC_STAGES-1:0] src_clear_sync_chain;
  logic [SYNC_STAGES-1:0] dst_clear_sync_chain;

  // Synchronization for clear signals
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_clear_sync_chain <= '0;
    end else begin
      src_clear_sync_chain <= {src_clear_sync_chain[SYNC_STAGES-2:0], src_clear_i};
    end
  end

  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_clear_sync_chain <= '0;
    end else begin
      dst_clear_sync_chain <= {dst_clear_sync_chain[SYNC_STAGES-2:0], dst_clear_i};
    end
  end

  assign src_clear_sync = src_clear_sync_chain[SYNC_STAGES-1];
  assign dst_clear_sync = dst_clear_sync_chain[SYNC_STAGES-1];

  // Source side logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_clear_pending <= 0;
      src_ready_sync <= 0;
    end else begin
      if (src_clear_i) begin
        src_clear_pending <= 1;
        src_ready_sync <= 0;
      end else if (src_clear_sync) begin
        src_clear_pending <= 0;
        src_ready_sync <= 0;
      end else if (src_valid_i && src_ready_o) begin
        src_ready_sync <= 1;
      end else if (src_ready_sync && dst_ready_reg) begin
        src_ready_sync <= 0;
      end
    end
  end

  // Destination side logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_clear_pending <= 0;
      dst_valid_reg <= 0;
    end else begin
      if (dst_clear_i) begin
        dst_clear_pending <= 1;
        dst_valid_reg <= 0;
      end else if (dst_clear_sync) begin
        dst_clear_pending <= 0;
        dst_valid_reg <= 0;
      end else if (src_valid_sync && !dst_valid_reg) begin
        dst_data_reg <= src_data_sync;
        dst_valid_reg <= 1;
      end else if (dst_valid_reg && dst_ready_i) begin
        dst_valid_reg <= 0;
      end
    end
  end

  // Cross-clock domain synchronization for valid and data
  logic [SYNC_STAGES-1:0] src_valid_sync_chain;
  logic [SYNC_STAGES-1:0] [WIDTH-1:0] src_data_sync_chain;

  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_valid_sync_chain <= '0;
      src_data_sync_chain <= '0;
    end else begin
      src_valid_sync_chain <= {src_valid_sync_chain[SYNC_STAGES-2:0], src_valid_i};
      src_data_sync_chain <= {src_data_sync_chain[SYNC_STAGES-2:0], src_data_i};
    end
  end

  assign src_valid_sync = src_valid_sync_chain[SYNC_STAGES-1];
  assign src_data_sync = src_data_sync_chain[SYNC_STAGES-1];

  // Cross-clock domain synchronization for ready
  logic [SYNC_STAGES-1:0] dst_ready_sync_chain;

  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_ready_sync_chain <= '0;
    end else begin
      dst_ready_sync_chain <= {dst_ready_sync_chain[SYNC_STAGES-2:0], dst_ready_i};
    end
  end

  assign dst_ready_sync = dst_ready_sync_chain[SYNC_STAGES-1];

  // Outputs
  assign src_ready_o = ~src_clear_pending && ~src_valid_sync && dst_ready_sync;
  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;
  assign src_clear_pending_o = src_clear_pending;
  assign dst_clear_pending_o = dst_clear_pending;

endmodule
