module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Internal signals
  logic [WIDTH-1:0] src_data_sync;
  logic src_valid_sync;
  logic src_ready_sync;
  logic dst_valid_sync;
  logic dst_ready_sync;
  logic src_clear_sync;
  logic dst_clear_sync;
  logic src_clear_pending;
  logic dst_clear_pending;

  // Synchronizers for clear signals
  logic [SYNC_STAGES-1:0] src_clear_sync_chain;
  logic [SYNC_STAGES-1:0] dst_clear_sync_chain;

  // Synchronizers for valid and ready signals
  logic [SYNC_STAGES-1:0] src_valid_sync_chain;
  logic [SYNC_STAGES-1:0] dst_ready_sync_chain;

  // Synchronizers for data
  logic [WIDTH-1:0] src_data_sync_chain [SYNC_STAGES-1:0];

  // Source side logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_clear_sync_chain <= '0;
      src_valid_sync_chain <= '0;
      src_data_sync_chain  <= '{default: '0};
    end else begin
      src_clear_sync_chain <= {src_clear_sync_chain[SYNC_STAGES-2:0], src_clear_i};
      src_valid_sync_chain <= {src_valid_sync_chain[SYNC_STAGES-2:0], src_valid_i};
      src_data_sync_chain  <= {src_data_sync_chain[SYNC_STAGES-2:0], src_data_i};
    end
  end

  assign src_clear_sync = src_clear_sync_chain[SYNC_STAGES-1];
  assign src_valid_sync = src_valid_sync_chain[SYNC_STAGES-1];
  assign src_data_sync  = src_data_sync_chain[SYNC_STAGES-1];

  // Destination side logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_clear_sync_chain <= '0;
      dst_ready_sync_chain <= '0;
    end else begin
      dst_clear_sync_chain <= {dst_clear_sync_chain[SYNC_STAGES-2:0], dst_clear_i};
      dst_ready_sync_chain <= {dst_ready_sync_chain[SYNC_STAGES-2:0], dst_ready_i};
    end
  end

  assign dst_clear_sync = dst_clear_sync_chain[SYNC_STAGES-1];
  assign dst_ready_sync = dst_ready_sync_chain[SYNC_STAGES-1];

  // Clear pending signals
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_clear_pending <= 1'b0;
    end else if (src_clear_i) begin
      src_clear_pending <= 1'b1;
    end else if (dst_clear_sync) begin
      src_clear_pending <= 1'b0;
    end
  end

  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_clear_pending <= 1'b0;
    end else if (dst_clear_i) begin
      dst_clear_pending <= 1'b1;
    end else if (src_clear_sync) begin
      dst_clear_pending <= 1'b0;
    end
  end

  // Main handshake logic
  logic src_handshake;
  logic dst_handshake;

  assign src_handshake = src_valid_sync & dst_ready_sync & ~src_clear_sync & ~dst_clear_sync;
  assign dst_handshake = dst_valid_sync & src_ready_sync & ~src_clear_sync & ~dst_clear_sync;

  // Source side ready and valid
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ready_o <= 1'b0;
      src_valid_sync <= 1'b0;
    end else begin
      src_ready_o <= ~src_valid_sync | dst_handshake;
      src_valid_sync <= src_valid_sync & ~dst_handshake | src_handshake;
    end
  end

  // Destination side ready and valid
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_ready_o <= 1'b0;
      dst_valid_sync <= 1'b0;
    end else begin
      dst_ready_o <= ~dst_valid_sync | src_handshake;
      dst_valid_sync <= dst_valid_sync & ~src_handshake | dst_handshake;
    end
  end

  // Data transfer
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_o <= '0;
    end else if (dst_handshake) begin
      dst_data_o <= src_data_sync;
    end
  end

  // Clear pending outputs
  assign src_clear_pending_o = src_clear_pending;
  assign dst_clear_pending_o = dst_clear_pending;

endmodule
