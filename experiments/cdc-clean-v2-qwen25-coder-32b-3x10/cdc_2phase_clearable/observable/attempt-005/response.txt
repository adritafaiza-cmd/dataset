module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source side signals
  logic [WIDTH-1:0] src_data_ff;
  logic src_valid_ff;
  logic src_ready_ff;
  logic src_clear_ff;
  logic src_clear_pending_ff;

  // Destination side signals
  logic [WIDTH-1:0] dst_data_ff;
  logic dst_valid_ff;
  logic dst_ready_ff;
  logic dst_clear_ff;
  logic dst_clear_pending_ff;

  // Synchronization signals
  logic [SYNC_STAGES-1:0] src_clear_sync;
  logic [SYNC_STAGES-1:0] dst_clear_sync;

  // Source side logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_ff <= '0;
      src_valid_ff <= '0;
      src_ready_ff <= '0;
      src_clear_ff <= '0;
      src_clear_pending_ff <= '0;
    end else begin
      // Handle clear request
      if (src_clear_i) begin
        src_clear_ff <= '1;
        src_clear_pending_ff <= '1;
      end else if (src_clear_sync[SYNC_STAGES-1]) begin
        src_clear_ff <= '0;
        src_clear_pending_ff <= '0;
      end

      // Handle data transfer
      if (src_ready_ff && src_valid_i) begin
        src_data_ff <= src_data_i;
        src_valid_ff <= '0;
      end else if (src_ready_i && !src_valid_ff) begin
        src_data_ff <= src_data_i;
        src_valid_ff <= '1;
      end

      // Update ready signal
      src_ready_ff <= dst_ready_ff;
    end
  end

  // Destination side logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_ff <= '0;
      dst_valid_ff <= '0;
      dst_ready_ff <= '0;
      dst_clear_ff <= '0;
      dst_clear_pending_ff <= '0;
    end else begin
      // Handle clear request
      if (dst_clear_i) begin
        dst_clear_ff <= '1;
        dst_clear_pending_ff <= '1;
      end else if (dst_clear_sync[SYNC_STAGES-1]) begin
        dst_clear_ff <= '0;
        dst_clear_pending_ff <= '0;
      end

      // Handle data transfer
      if (dst_ready_i && dst_valid_ff) begin
        dst_valid_ff <= '0;
      end else if (src_valid_ff && !dst_valid_ff) begin
        dst_data_ff <= src_data_ff;
        dst_valid_ff <= '1;
      end

      // Update ready signal
      dst_ready_ff <= !dst_valid_ff || dst_ready_i;
    end
  end

  // Synchronization logic for clear signals
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_clear_sync <= '0;
    end else begin
      src_clear_sync <= {src_clear_sync[SYNC_STAGES-2:0], dst_clear_i};
    end
  end

  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_clear_sync <= '0;
    end else begin
      dst_clear_sync <= {dst_clear_sync[SYNC_STAGES-2:0], src_clear_i};
    end
  end

  // Output assignments
  assign src_ready_o = src_ready_ff;
  assign src_clear_pending_o = src_clear_pending_ff;
  assign dst_data_o = dst_data_ff;
  assign dst_valid_o = dst_valid_ff;
  assign dst_clear_pending_o = dst_clear_pending_ff;

endmodule
