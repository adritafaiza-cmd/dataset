module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain signals
  logic src_clear_synced;
  logic src_clear_req;
  logic src_clear_ack;
  logic src_clear_done;
  logic src_clear_active;
  logic src_valid_synced;
  logic src_ready;
  logic [WIDTH-1:0] src_data_synced;

  // Destination domain signals
  logic dst_clear_synced;
  logic dst_clear_req;
  logic dst_clear_ack;
  logic dst_clear_done;
  logic dst_clear_active;
  logic dst_valid;
  logic [WIDTH-1:0] dst_data;

  // Synchronization flops for clear signals
  logic [SYNC_STAGES-1:0] src_clear_sync;
  logic [SYNC_STAGES-1:0] dst_clear_sync;

  // Synchronization flops for valid and data signals
  logic [SYNC_STAGES-1:0] src_valid_sync;
  logic [WIDTH-1:0] src_data_sync [SYNC_STAGES];

  // Source domain logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_clear_sync <= '0;
      src_valid_sync <= '0;
      src_data_sync <= '{default: '0};
    end else begin
      src_clear_sync <= {src_clear_sync[SYNC_STAGES-2:0], src_clear_i};
      src_valid_sync <= {src_valid_sync[SYNC_STAGES-2:0], src_valid_i};
      src_data_sync <= {src_data_sync[SYNC_STAGES-2:0], src_data_i};
    end
  end

  assign src_clear_synced = src_clear_sync[SYNC_STAGES-1];
  assign src_valid_synced = src_valid_sync[SYNC_STAGES-1];
  assign src_data_synced = src_data_sync[SYNC_STAGES-1];

  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_clear_req <= 0;
      src_clear_ack <= 0;
      src_clear_done <= 0;
      src_clear_active <= 0;
    end else begin
      if (src_clear_i && !src_clear_active) begin
        src_clear_req <= 1;
        src_clear_active <= 1;
      end else if (src_clear_ack) begin
        src_clear_req <= 0;
        src_clear_done <= 1;
      end else if (src_clear_done && dst_clear_done) begin
        src_clear_done <= 0;
        src_clear_active <= 0;
      end
    end
  end

  assign src_clear_pending_o = src_clear_active;

  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ready <= 0;
    end else begin
      if (src_clear_active) begin
        src_ready <= 0;
      end else if (dst_ready_i && dst_valid) begin
        src_ready <= 1;
      end else if (src_valid_synced && src_ready) begin
        src_ready <= 0;
      end
    end
  end

  assign src_ready_o = src_ready;

  // Destination domain logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_clear_sync <= '0;
    end else begin
      dst_clear_sync <= {dst_clear_sync[SYNC_STAGES-2:0], dst_clear_i};
    end
  end

  assign dst_clear_synced = dst_clear_sync[SYNC_STAGES-1];

  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_clear_req <= 0;
      dst_clear_ack <= 0;
      dst_clear_done <= 0;
      dst_clear_active <= 0;
    end else begin
      if (dst_clear_i && !dst_clear_active) begin
        dst_clear_req <= 1;
        dst_clear_active <= 1;
      end else if (dst_clear_ack) begin
        dst_clear_req <= 0;
        dst_clear_done <= 1;
      end else if (dst_clear_done && src_clear_done) begin
        dst_clear_done <= 0;
        dst_clear_active <= 0;
      end
    end
  end

  assign dst_clear_pending_o = dst_clear_active;

  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid <= 0;
      dst_data <= '0;
    end else begin
      if (src_valid_synced && src_ready) begin
        dst_valid <= 1;
        dst_data <= src_data_synced;
      end else if (dst_ready_i && dst_valid) begin
        dst_valid <= 0;
      end
    end
  end

  assign dst_valid_o = dst_valid;
  assign dst_data_o = dst_data;

  // Clear handshake
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_clear_ack <= 0;
    end else begin
      src_clear_ack <= dst_clear_req;
    end
  end

  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_clear_ack <= 0;
    end else begin
      dst_clear_ack <= src_clear_req;
    end
  end

endmodule
