module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Internal signals
  logic [WIDTH-1:0] sync_data;
  logic sync_valid;
  logic sync_ready;

  // Synchronization logic for valid signal
  logic [SYNC_STAGES-1:0] sync_valid_reg;
  logic [SYNC_STAGES-1:0] sync_ready_reg;

  // Clear signals
  logic src_clear_reg;
  logic dst_clear_reg;
  logic src_clear_sync;
  logic dst_clear_sync;

  // Clear pending signals
  logic src_clear_pending;
  logic dst_clear_pending;

  // Source domain logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_clear_reg <= 0;
      src_clear_pending <= 0;
    end else begin
      src_clear_reg <= src_clear_i;
      if (src_clear_i) begin
        src_clear_pending <= 1;
      end else if (!src_clear_pending_o) begin
        src_clear_pending <= 0;
      end
    end
  end

  // Destination domain logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_clear_reg <= 0;
      dst_clear_pending <= 0;
    end else begin
      dst_clear_reg <= dst_clear_i;
      if (dst_clear_i) begin
        dst_clear_pending <= 1;
      end else if (!dst_clear_pending_o) begin
        dst_clear_pending <= 0;
      end
    end
  end

  // Synchronize clear signals
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_clear_sync <= 0;
    end else begin
      dst_clear_sync <= dst_clear_reg;
    end
  end

  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_clear_sync <= 0;
    end else begin
      src_clear_sync <= src_clear_reg;
    end
  end

  // Synchronize valid and ready signals
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      sync_valid_reg <= 0;
    end else begin
      sync_valid_reg <= {sync_valid_reg[SYNC_STAGES-2:0], src_valid_i & ~src_clear_reg};
    end
  end

  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      sync_ready_reg <= 0;
    end else begin
      sync_ready_reg <= {sync_ready_reg[SYNC_STAGES-2:0], dst_ready_i & ~dst_clear_reg};
    end
  end

  // Assign synchronized signals
  assign sync_valid = sync_valid_reg[SYNC_STAGES-1];
  assign sync_ready = sync_ready_reg[SYNC_STAGES-1];

  // Data synchronization
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      sync_data <= 0;
    end else if (src_valid_i & sync_ready & ~src_clear_reg) begin
      sync_data <= src_data_i;
    end
  end

  // Output assignments
  assign src_ready_o = sync_ready & ~src_clear_reg;
  assign src_clear_pending_o = src_clear_pending;

  assign dst_data_o = sync_data;
  assign dst_valid_o = sync_valid & ~dst_clear_reg;
  assign dst_clear_pending_o = dst_clear_pending;

endmodule
