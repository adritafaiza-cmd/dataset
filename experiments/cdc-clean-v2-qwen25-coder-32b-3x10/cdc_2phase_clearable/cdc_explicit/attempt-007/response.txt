module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain signals
  logic src_clear_sync;
  logic src_clear_req;
  logic src_clear_ack;
  logic src_clear_pending;
  logic src_valid_sync;
  logic src_valid_req;
  logic src_valid_ack;
  logic [WIDTH-1:0] src_data_sync;

  // Destination domain signals
  logic dst_clear_sync;
  logic dst_clear_req;
  logic dst_clear_ack;
  logic dst_clear_pending;
  logic dst_valid_sync;
  logic dst_valid_req;
  logic dst_valid_ack;
  logic [WIDTH-1:0] dst_data_sync;

  // Synchronization for clear signal from source to destination
  logic [SYNC_STAGES-1:0] src_clear_sync_reg;
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_clear_sync_reg <= '0;
    end else begin
      src_clear_sync_reg <= {src_clear_sync_reg[SYNC_STAGES-2:0], src_clear_i};
    end
  end
  assign src_clear_sync = src_clear_sync_reg[SYNC_STAGES-1];

  // Synchronization for clear signal from destination to source
  logic [SYNC_STAGES-1:0] dst_clear_sync_reg;
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_clear_sync_reg <= '0;
    end else begin
      dst_clear_sync_reg <= {dst_clear_sync_reg[SYNC_STAGES-2:0], dst_clear_i};
    end
  end
  assign dst_clear_sync = dst_clear_sync_reg[SYNC_STAGES-1];

  // Synchronization for valid signal from source to destination
  logic [SYNC_STAGES-1:0] src_valid_sync_reg;
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_valid_sync_reg <= '0;
    end else begin
      src_valid_sync_reg <= {src_valid_sync_reg[SYNC_STAGES-2:0], src_valid_i};
    end
  end
  assign src_valid_sync = src_valid_sync_reg[SYNC_STAGES-1];

  // Synchronization for data signal from source to destination
  logic [WIDTH-1:0] src_data_sync_reg [SYNC_STAGES-1:0];
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_data_sync_reg <= '{default: '0};
    end else if (src_valid_i) begin
      src_data_sync_reg[0] <= src_data_i;
      for (int i = 1; i < SYNC_STAGES; i++) begin
        src_data_sync_reg[i] <= src_data_sync_reg[i-1];
      end
    end
  end
  assign src_data_sync = src_data_sync_reg[SYNC_STAGES-1];

  // Source domain clear logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni || (CLEAR_ON_ASYNC_RESET && !src_rst_ni)) begin
      src_clear_req <= 0;
      src_clear_ack <= 0;
      src_clear_pending <= 0;
    end else begin
      if (src_clear_i && !src_clear_pending) begin
        src_clear_req <= 1;
        src_clear_pending <= 1;
      end else if (src_clear_ack) begin
        src_clear_req <= 0;
        src_clear_pending <= 0;
      end
    end
  end

  // Destination domain clear logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni || (CLEAR_ON_ASYNC_RESET && !dst_rst_ni)) begin
      dst_clear_req <= 0;
      dst_clear_ack <= 0;
      dst_clear_pending <= 0;
    end else begin
      if (dst_clear_i && !dst_clear_pending) begin
        dst_clear_req <= 1;
        dst_clear_pending <= 1;
      end else if (dst_clear_ack) begin
        dst_clear_req <= 0;
        dst_clear_pending <= 0;
      end
    end
  end

  // Source domain handshake logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_valid_req <= 0;
      src_valid_ack <= 0;
    end else begin
      if (src_valid_i && !src_clear_pending) begin
        src_valid_req <= 1;
      end else if (src_valid_ack) begin
        src_valid_req <= 0;
      end
    end
  end

  // Destination domain handshake logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid_req <= 0;
      dst_valid_ack <= 0;
    end else begin
      if (src_valid_sync && !dst_clear_pending) begin
        dst_valid_req <= 1;
      end else if (dst_valid_ack) begin
        dst_valid_req <= 0;
      end
    end
  end

  // Clear request and acknowledge handshake
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_clear_ack <= 0;
    end else begin
      src_clear_ack <= src_clear_req;
    end
  end

  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_clear_ack <= 0;
    end else begin
      dst_clear_ack <= dst_clear_req;
    end
  end

  // Valid request and acknowledge handshake
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_valid_ack <= 0;
    end else begin
      src_valid_ack <= dst_valid_req && dst_ready_i;
    end
  end

  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_valid_ack <= 0;
    end else begin
      dst_valid_ack <= src_valid_req;
    end
  end

  // Output assignments
  assign src_clear_pending_o = src_clear_pending;
  assign src_ready_o = !src_valid_sync || (src_valid_ack && dst_ready_i);
  assign dst_clear_pending_o = dst_clear_pending;
  assign dst_data_o = src_data_sync;
  assign dst_valid_o = src_valid_sync && !dst_clear_pending;

endmodule
