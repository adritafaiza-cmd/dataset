module apbxclk #(
		parameter	C_APB_ADDR_WIDTH = 12,
		parameter	C_APB_DATA_WIDTH = 32,
		parameter [0:0]	OPT_REGISTERED = 1'b0,
		localparam	AW = C_APB_ADDR_WIDTH,
		localparam	DW = C_APB_DATA_WIDTH
	) (
		input	wire			S_APB_PCLK, S_PRESETn,
		input	wire			S_APB_PSEL,
		input	wire			S_APB_PENABLE,
		output	reg			S_APB_PREADY,
		input	wire	[AW-1:0]	S_APB_PADDR,
		input	wire			S_APB_PWRITE,
		input	wire	[DW-1:0]	S_APB_PWDATA,
		input	wire	[DW/8-1:0]	S_APB_PWSTRB,
		input	wire	[2:0]		S_APB_PPROT,
		output	wire	[DW-1:0]	S_APB_PRDATA,
		output	wire			S_APB_PSLVERR,
		input	wire			M_APB_PCLK,
		output	reg			M_PRESETn,
		output	reg			M_APB_PSEL,
		output	reg			M_APB_PENABLE,
		input	wire			M_APB_PREADY,
		output	wire	[AW-1:0]	M_APB_PADDR,
		output	wire			M_APB_PWRITE,
		output	wire	[DW-1:0]	M_APB_PWDATA,
		output	wire	[DW/8-1:0]	M_APB_PWSTRB,
		output	wire	[2:0]		M_APB_PPROT,
		input	wire	[DW-1:0]	M_APB_PRDATA,
		input	wire			M_APB_PSLVERR
	);

reg [AW-1:0] s_addr_sync;
reg s_write_sync;
reg [DW-1:0] s_wdata_sync;
reg [DW/8-1:0] s_wstrb_sync;
reg [2:0] s_prot_sync;
reg s_req_sync;
reg s_req_sync_d;

reg [AW-1:0] m_addr;
reg m_write;
reg [DW-1:0] m_wdata;
reg [DW/8-1:0] m_wstrb;
reg [2:0] m_prot;
reg m_req;
reg m_req_d;

reg [DW-1:0] s_rdata;
reg s_slverr;

// Synchronizers for request fields from S_APB to M_APB
always @(posedge M_APB_PCLK or negedge S_PRESETn) begin
	if (!S_PRESETn) begin
		s_addr_sync <= 0;
		s_write_sync <= 0;
		s_wdata_sync <= 0;
		s_wstrb_sync <= 0;
		s_prot_sync <= 0;
		s_req_sync <= 0;
	end else begin
		s_addr_sync <= S_APB_PADDR;
		s_write_sync <= S_APB_PWRITE;
		s_wdata_sync <= S_APB_PWDATA;
		s_wstrb_sync <= S_APB_PWSTRB;
		s_prot_sync <= S_APB_PPROT;
		s_req_sync <= S_APB_PSEL & S_APB_PENABLE;
	end
end

// Synchronizers for response fields from M_APB to S_APB
always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
	if (!S_PRESETn) begin
		s_rdata <= 0;
		s_slverr <= 0;
	end else begin
		s_rdata <= M_APB_PRDATA;
		s_slverr <= M_APB_PSLVERR;
	end
end

// Detect request edge in master clock domain
always @(posedge M_APB_PCLK or negedge S_PRESETn) begin
	if (!S_PRESETn) begin
		s_req_sync_d <= 0;
	end else begin
		s_req_sync_d <= s_req_sync;
	end
end

// Detect request edge in slave clock domain
always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
	if (!S_PRESETn) begin
		m_req_d <= 0;
	end else begin
		m_req_d <= m_req;
	end
end

// Master side logic
always @(posedge M_APB_PCLK or negedge S_PRESETn) begin
	if (!S_PRESETn) begin
		M_PRESETn <= 0;
		M_APB_PSEL <= 0;
		M_APB_PENABLE <= 0;
		m_addr <= 0;
		m_write <= 0;
		m_wdata <= 0;
		m_wstrb <= 0;
		m_prot <= 0;
		m_req <= 0;
	end else begin
		M_PRESETn <= S_PRESETn;
		if (s_req_sync & ~s_req_sync_d) begin
			m_addr <= s_addr_sync;
			m_write <= s_write_sync;
			m_wdata <= s_wdata_sync;
			m_wstrb <= s_wstrb_sync;
			m_prot <= s_prot_sync;
			m_req <= 1;
		end else if (M_APB_PREADY) begin
			m_req <= 0;
		end
		M_APB_PSEL <= m_req;
		M_APB_PENABLE <= m_req;
	end
end

// Slave side logic
always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
	if (!S_PRESETn) begin
		S_APB_PREADY <= 0;
	end else begin
		if (m_req_d & ~m_req & M_APB_PREADY) begin
			S_APB_PREADY <= 1;
		end else begin
			S_APB_PREADY <= 0;
		end
	end
end

assign S_APB_PRDATA = s_rdata;
assign S_APB_PSLVERR = s_slverr;
assign M_APB_PADDR = m_addr;
assign M_APB_PWRITE = m_write;
assign M_APB_PWDATA = m_wdata;
assign M_APB_PWSTRB = m_wstrb;
assign M_APB_PPROT = m_prot;

endmodule
