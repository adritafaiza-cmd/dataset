module apbxclk #(
        parameter C_APB_ADDR_WIDTH = 12,
        parameter C_APB_DATA_WIDTH = 32,
        parameter [0:0] OPT_REGISTERED = 1'b0,
        localparam AW = C_APB_ADDR_WIDTH,
        localparam DW = C_APB_DATA_WIDTH
    ) (
        input wire S_APB_PCLK, S_PRESETn,
        input wire S_APB_PSEL,
        input wire S_APB_PENABLE,
        output reg S_APB_PREADY,
        input wire [AW-1:0] S_APB_PADDR,
        input wire S_APB_PWRITE,
        input wire [DW-1:0] S_APB_PWDATA,
        input wire [DW/8-1:0] S_APB_PWSTRB,
        input wire [2:0] S_APB_PPROT,
        output wire [DW-1:0] S_APB_PRDATA,
        output wire S_APB_PSLVERR,
        input wire M_APB_PCLK,
        output reg M_PRESETn,
        output reg M_APB_PSEL,
        output reg M_APB_PENABLE,
        input wire M_APB_PREADY,
        output wire [AW-1:0] M_APB_PADDR,
        output wire M_APB_PWRITE,
        output wire [DW-1:0] M_APB_PWDATA,
        output wire [DW/8-1:0] M_APB_PWSTRB,
        output wire [2:0] M_APB_PPROT,
        input wire [DW-1:0] M_APB_PRDATA,
        input wire M_APB_PSLVERR
    );

    // Internal signals
    reg [AW-1:0] addr_reg;
    reg [DW-1:0] wdata_reg;
    reg [DW/8-1:0] wstrb_reg;
    reg [2:0] prot_reg;
    reg write_reg;
    reg request_valid;
    reg request_accepted;

    // Source side logic
    always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
        if (!S_PRESETn) begin
            S_APB_PREADY <= 1'b0;
            request_valid <= 1'b0;
            request_accepted <= 1'b0;
        end else begin
            if (S_APB_PSEL && S_APB_PENABLE && !request_valid) begin
                request_valid <= 1'b1;
                addr_reg <= S_APB_PADDR;
                wdata_reg <= S_APB_PWDATA;
                wstrb_reg <= S_APB_PWSTRB;
                prot_reg <= S_APB_PPROT;
                write_reg <= S_APB_PWRITE;
            end

            if (request_accepted && M_APB_PREADY) begin
                S_APB_PREADY <= 1'b1;
                request_valid <= 1'b0;
                request_accepted <= 1'b0;
            end else begin
                S_APB_PREADY <= 1'b0;
            end
        end
    end

    // Destination side logic
    always @(posedge M_APB_PCLK or negedge S_PRESETn) begin
        if (!S_PRESETn) begin
            M_PRESETn <= 1'b0;
            M_APB_PSEL <= 1'b0;
            M_APB_PENABLE <= 1'b0;
        end else begin
            M_PRESETn <= 1'b1;

            if (request_valid && !request_accepted) begin
                M_APB_PSEL <= 1'b1;
                M_APB_PENABLE <= 1'b1;
                request_accepted <= 1'b1;
            end else if (M_APB_PREADY) begin
                M_APB_PSEL <= 1'b0;
                M_APB_PENABLE <= 1'b0;
            end
        end
    end

    // Data and control signals crossing
    generate
        if (OPT_REGISTERED) begin : registered_crossing
            reg [DW-1:0] prdata_reg;
            reg pslverr_reg;

            always @(posedge M_APB_PCLK or negedge S_PRESETn) begin
                if (!S_PRESETn) begin
                    prdata_reg <= 0;
                    pslverr_reg <= 0;
                end else if (M_APB_PREADY) begin
                    prdata_reg <= M_APB_PRDATA;
                    pslverr_reg <= M_APB_PSLVERR;
                end
            end

            assign S_APB_PRDATA = prdata_reg;
            assign S_APB_PSLVERR = pslverr_reg;

            always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
                if (!S_PRESETn) begin
                    M_APB_PADDR <= 0;
                    M_APB_PWDATA <= 0;
                    M_APB_PWSTRB <= 0;
                    M_APB_PPROT <= 0;
                    M_APB_PWRITE <= 0;
                end else if (request_valid && !request_accepted) begin
                    M_APB_PADDR <= addr_reg;
                    M_APB_PWDATA <= wdata_reg;
                    M_APB_PWSTRB <= wstrb_reg;
                    M_APB_PPROT <= prot_reg;
                    M_APB_PWRITE <= write_reg;
                end
            end
        end else begin : combinational_crossing
            assign S_APB_PRDATA = M_APB_PRDATA;
            assign S_APB_PSLVERR = M_APB_PSLVERR;

            assign M_APB_PADDR = addr_reg;
            assign M_APB_PWDATA = wdata_reg;
            assign M_APB_PWSTRB = wstrb_reg;
            assign M_APB_PPROT = prot_reg;
            assign M_APB_PWRITE = write_reg;
        end
    endgenerate

endmodule
