module apbxclk #(
		parameter	C_APB_ADDR_WIDTH = 12,
		parameter	C_APB_DATA_WIDTH = 32,
		parameter [0:0]	OPT_REGISTERED = 1'b0,
		localparam	AW = C_APB_ADDR_WIDTH,
		localparam	DW = C_APB_DATA_WIDTH
	) (
		input	wire			S_APB_PCLK, S_PRESETn,
		input	wire			S_APB_PSEL,
		input	wire			S_APB_PENABLE,
		output	reg			S_APB_PREADY,
		input	wire	[AW-1:0]	S_APB_PADDR,
		input	wire			S_APB_PWRITE,
		input	wire	[DW-1:0]	S_APB_PWDATA,
		input	wire	[DW/8-1:0]	S_APB_PWSTRB,
		input	wire	[2:0]		S_APB_PPROT,
		output	wire	[DW-1:0]	S_APB_PRDATA,
		output	wire			S_APB_PSLVERR,
		input	wire			M_APB_PCLK,
		output	reg			M_PRESETn,
		output	reg			M_APB_PSEL,
		output	reg			M_APB_PENABLE,
		input	wire			M_APB_PREADY,
		output	wire	[AW-1:0]	M_APB_PADDR,
		output	wire			M_APB_PWRITE,
		output	wire	[DW-1:0]	M_APB_PWDATA,
		output	wire	[DW/8-1:0]	M_APB_PWSTRB,
		output	wire	[2:0]		M_APB_PPROT,
		input	wire	[DW-1:0]	M_APB_PRDATA,
		input	wire			M_APB_PSLVERR
	);

	// Internal signals
	reg [AW-1:0]	s_apb_paddr_reg;
	reg			s_apb_pwrite_reg;
	reg [DW-1:0]	s_apb_pwdata_reg;
	reg [DW/8-1:0]	s_apb_pwstrb_reg;
	reg [2:0]		s_apb_pprot_reg;
	reg [DW-1:0]	m_apb_prdata_reg;
	reg			m_apb_pslverr_reg;
	reg			req_valid;
	reg			req_accepted;
	reg			resp_valid;

	// Source APB interface
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			S_APB_PREADY <= 1'b0;
			req_valid <= 1'b0;
			req_accepted <= 1'b0;
			resp_valid <= 1'b0;
		end else begin
			if (S_APB_PSEL && S_APB_PENABLE && !req_valid) begin
				req_valid <= 1'b1;
				s_apb_paddr_reg <= S_APB_PADDR;
				s_apb_pwrite_reg <= S_APB_PWRITE;
				s_apb_pwdata_reg <= S_APB_PWDATA;
				s_apb_pwstrb_reg <= S_APB_PWSTRB;
				s_apb_pprot_reg <= S_APB_PPROT;
			end

			if (req_accepted) begin
				req_valid <= 1'b0;
				req_accepted <= 1'b0;
			end

			if (resp_valid && M_APB_PREADY) begin
				resp_valid <= 1'b0;
				S_APB_PREADY <= 1'b1;
			end else begin
				S_APB_PREADY <= 1'b0;
			end
		end
	end

	// Destination APB interface
	always @(posedge M_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			M_PRESETn <= 1'b0;
			M_APB_PSEL <= 1'b0;
			M_APB_PENABLE <= 1'b0;
			resp_valid <= 1'b0;
		end else begin
			M_PRESETn <= 1'b1;

			if (req_valid && !req_accepted) begin
				M_APB_PSEL <= 1'b1;
				M_APB_PADDR <= s_apb_paddr_reg;
				M_APB_PWRITE <= s_apb_pwrite_reg;
				M_APB_PWDATA <= s_apb_pwdata_reg;
				M_APB_PWSTRB <= s_apb_pwstrb_reg;
				M_APB_PPROT <= s_apb_pprot_reg;
				req_accepted <= 1'b1;
			end

			if (M_APB_PENABLE && M_APB_PREADY) begin
				M_APB_PSEL <= 1'b0;
				M_APB_PENABLE <= 1'b0;
				m_apb_prdata_reg <= M_APB_PRDATA;
				m_apb_pslverr_reg <= M_APB_PSLVERR;
				resp_valid <= 1'b1;
			end else if (M_APB_PSEL) begin
				M_APB_PENABLE <= 1'b1;
			end
		end
	end

	// Output assignments
	assign S_APB_PRDATA = m_apb_prdata_reg;
	assign S_APB_PSLVERR = m_apb_pslverr_reg;
	assign M_APB_PADDR = OPT_REGISTERED ? s_apb_paddr_reg : S_APB_PADDR;
	assign M_APB_PWRITE = OPT_REGISTERED ? s_apb_pwrite_reg : S_APB_PWRITE;
	assign M_APB_PWDATA = OPT_REGISTERED ? s_apb_pwdata_reg : S_APB_PWDATA;
	assign M_APB_PWSTRB = OPT_REGISTERED ? s_apb_pwstrb_reg : S_APB_PWSTRB;
	assign M_APB_PPROT = OPT_REGISTERED ? s_apb_pprot_reg : S_APB_PPROT;

endmodule
