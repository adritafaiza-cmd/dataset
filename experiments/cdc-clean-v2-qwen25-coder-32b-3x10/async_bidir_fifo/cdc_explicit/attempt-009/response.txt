module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;

  // FIFO for A to B direction
  reg [DSIZE-1:0] mem_a2b [0:DEPTH-1];
  reg [ASIZE:0] wptr_a2b, rptr_a2b;
  reg [ASIZE:0] wptr_a2b_sync, rptr_a2b_sync;

  // FIFO for B to A direction
  reg [DSIZE-1:0] mem_b2a [0:DEPTH-1];
  reg [ASIZE:0] wptr_b2a, rptr_b2a;
  reg [ASIZE:0] wptr_b2a_sync, rptr_b2a_sync;

  // Synchronization for pointers
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      wptr_a2b_sync <= 0;
      rptr_a2b_sync <= 0;
    end else begin
      wptr_a2b_sync <= wptr_a2b;
      rptr_a2b_sync <= rptr_a2b;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      wptr_b2a_sync <= 0;
      rptr_b2a_sync <= 0;
    end else begin
      wptr_b2a_sync <= wptr_b2a;
      rptr_b2a_sync <= rptr_b2a;
    end
  end

  // Write and read logic for A to B direction
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      wptr_a2b <= 0;
      rptr_a2b <= 0;
    end else if (a_dir && !a_full) begin
      mem_a2b[wptr_a2b[ASIZE-1:0]] <= a_wdata;
      wptr_a2b <= wptr_a2b + 1;
    end else if (!a_dir && !a_empty) begin
      rptr_a2b <= rptr_a2b + 1;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      rptr_a2b <= 0;
    end else if (!b_dir && !b_empty) begin
      rptr_a2b <= rptr_a2b + 1;
    end
  end

  // Write and read logic for B to A direction
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      wptr_b2a <= 0;
      rptr_b2a <= 0;
    end else if (b_dir && !b_full) begin
      mem_b2a[wptr_b2a[ASIZE-1:0]] <= b_wdata;
      wptr_b2a <= wptr_b2a + 1;
    end else if (!b_dir && !b_empty) begin
      rptr_b2a <= rptr_b2a + 1;
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      rptr_b2a <= 0;
    end else if (!a_dir && !a_empty) begin
      rptr_b2a <= rptr_b2a + 1;
    end
  end

  // Full and empty flags for A to B direction
  assign a_full = (wptr_a2b[ASIZE:0] == {1'b1, rptr_a2b_sync[ASIZE-1:0]});
  assign a_afull = (wptr_a2b[ASIZE:0] == {rptr_a2b_sync[ASIZE-1:0], 1'b1});
  assign a_empty = (wptr_a2b_sync[ASIZE:0] == rptr_a2b[ASIZE:0]);
  assign a_aempty = (wptr_a2b_sync[ASIZE:0] == {rptr_a2b[ASIZE-1:0], 1'b0});

  // Full and empty flags for B to A direction
  assign b_full = (wptr_b2a[ASIZE:0] == {1'b1, rptr_b2a_sync[ASIZE-1:0]});
  assign b_afull = (wptr_b2a[ASIZE:0] == {rptr_b2a_sync[ASIZE-1:0], 1'b1});
  assign b_empty = (wptr_b2a_sync[ASIZE:0] == rptr_b2a[ASIZE:0]);
  assign b_aempty = (wptr_b2a_sync[ASIZE:0] == {rptr_b2a[ASIZE-1:0], 1'b0});

  // Read data output
  assign a_rdata = mem_a2b[rptr_a2b_sync[ASIZE-1:0]];
  assign b_rdata = mem_b2a[rptr_b2a_sync[ASIZE-1:0]];

endmodule
