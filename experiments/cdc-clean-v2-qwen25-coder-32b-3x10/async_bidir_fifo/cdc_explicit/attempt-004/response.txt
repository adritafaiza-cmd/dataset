module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

    localparam DEPTH = 1 << ASIZE;
    localparam AFULL_THRESHOLD = DEPTH - 2;

    // FIFO for A to B direction
    reg [DSIZE-1:0] mem_a2b [0:DEPTH-1];
    reg [ASIZE:0] wr_ptr_a2b, rd_ptr_a2b, wr_ptr_gray_a2b, rd_ptr_gray_a2b;
    reg [ASIZE:0] wr_ptr_gray_sync_a2b, rd_ptr_gray_sync_a2b;

    // FIFO for B to A direction
    reg [DSIZE-1:0] mem_b2a [0:DEPTH-1];
    reg [ASIZE:0] wr_ptr_b2a, rd_ptr_b2a, wr_ptr_gray_b2a, rd_ptr_gray_b2a;
    reg [ASIZE:0] wr_ptr_gray_sync_b2a, rd_ptr_gray_sync_b2a;

    // Binary to Gray conversion
    function [ASIZE:0] bin2gray;
        input [ASIZE:0] bin;
        bin2gray = bin ^ (bin >> 1);
    endfunction

    // Gray to Binary conversion
    function [ASIZE:0] gray2bin;
        input [ASIZE:0] gray;
        integer i;
        reg [ASIZE:0] bin;
        bin = gray;
        for (i = ASIZE-1; i > 0; i = i - 1)
            bin[i-1] = bin[i] ^ bin[i-1];
        gray2bin = bin;
    endfunction

    // A to B FIFO logic
    always @(posedge a_clk or negedge a_rst_n) begin
        if (!a_rst_n) begin
            wr_ptr_a2b <= 0;
            rd_ptr_a2b <= 0;
        end else begin
            if (a_dir && !a_full) begin
                mem_a2b[wr_ptr_a2b[ASIZE-1:0]] <= a_wdata;
                wr_ptr_a2b <= wr_ptr_a2b + 1;
            end
        end
    end

    always @(posedge b_clk or negedge b_rst_n) begin
        if (!b_rst_n) begin
            wr_ptr_gray_sync_a2b <= 0;
        end else begin
            wr_ptr_gray_sync_a2b <= bin2gray(wr_ptr_a2b);
        end
    end

    always @(posedge b_clk or negedge b_rst_n) begin
        if (!b_rst_n) begin
            rd_ptr_b2a <= 0;
        end else begin
            if (!b_dir && !b_empty) begin
                rd_ptr_b2a <= rd_ptr_b2a + 1;
            end
        end
    end

    always @(posedge b_clk or negedge b_rst_n) begin
        if (!b_rst_n) begin
            rd_ptr_gray_sync_a2b <= 0;
        end else begin
            rd_ptr_gray_sync_a2b <= bin2gray(rd_ptr_b2a);
        end
    end

    assign wr_ptr_gray_a2b = bin2gray(wr_ptr_a2b);
    assign rd_ptr_gray_a2b = bin2gray(rd_ptr_a2b);

    assign a_full = (wr_ptr_gray_a2b[ASIZE] != rd_ptr_gray_sync_a2b[ASIZE]) && (wr_ptr_gray_a2b[ASIZE-1:0] == rd_ptr_gray_sync_a2b[ASIZE-1:0]);
    assign a_afull = (wr_ptr_gray_a2b[ASIZE] == rd_ptr_gray_sync_a2b[ASIZE]) && (wr_ptr_gray_a2b[ASIZE-1:0] == rd_ptr_gray_sync_a2b[ASIZE-1:0] + AFULL_THRESHOLD);
    assign a_empty = (wr_ptr_gray_sync_a2b == rd_ptr_gray_a2b);
    assign a_aempty = (wr_ptr_gray_sync_a2b[ASIZE] != rd_ptr_gray_a2b[ASIZE]) && (wr_ptr_gray_sync_a2b[ASIZE-1:0] == rd_ptr_gray_a2b[ASIZE-1:0] + 1);

    assign a_rdata = mem_a2b[rd_ptr_a2b[ASIZE-1:0]];

    // B to A FIFO logic
    always @(posedge b_clk or negedge b_rst_n) begin
        if (!b_rst_n) begin
            wr_ptr_b2a <= 0;
            rd_ptr_b2a <= 0;
        end else begin
            if (b_dir && !b_full) begin
                mem_b2a[wr_ptr_b2a[ASIZE-1:0]] <= b_wdata;
                wr_ptr_b2a <= wr_ptr_b2a + 1;
            end
        end
    end

    always @(posedge a_clk or negedge a_rst_n) begin
        if (!a_rst_n) begin
            wr_ptr_gray_sync_b2a <= 0;
        end else begin
            wr_ptr_gray_sync_b2a <= bin2gray(wr_ptr_b2a);
        end
    end

    always @(posedge a_clk or negedge a_rst_n) begin
        if (!a_rst_n) begin
            rd_ptr_a2b <= 0;
        end else begin
            if (!a_dir && !a_empty) begin
                rd_ptr_a2b <= rd_ptr_a2b + 1;
            end
        end
    end

    always @(posedge a_clk or negedge a_rst_n) begin
        if (!a_rst_n) begin
            rd_ptr_gray_sync_b2a <= 0;
        end else begin
            rd_ptr_gray_sync_b2a <= bin2gray(rd_ptr_a2b);
        end
    end

    assign wr_ptr_gray_b2a = bin2gray(wr_ptr_b2a);
    assign rd_ptr_gray_b2a = bin2gray(rd_ptr_b2a);

    assign b_full = (wr_ptr_gray_b2a[ASIZE] != rd_ptr_gray_sync_b2a[ASIZE]) && (wr_ptr_gray_b2a[ASIZE-1:0] == rd_ptr_gray_sync_b2a[ASIZE-1:0]);
    assign b_afull = (wr_ptr_gray_b2a[ASIZE] == rd_ptr_gray_sync_b2a[ASIZE]) && (wr_ptr_gray_b2a[ASIZE-1:0] == rd_ptr_gray_sync_b2a[ASIZE-1:0] + AFULL_THRESHOLD);
    assign b_empty = (wr_ptr_gray_sync_b2a == rd_ptr_gray_b2a);
    assign b_aempty = (wr_ptr_gray_sync_b2a[ASIZE] != rd_ptr_gray_b2a[ASIZE]) && (wr_ptr_gray_sync_b2a[ASIZE-1:0] == rd_ptr_gray_b2a[ASIZE-1:0] + 1);

    assign b_rdata = mem_b2a[rd_ptr_b2a[ASIZE-1:0]];

endmodule
