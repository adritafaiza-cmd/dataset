module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

    localparam DEPTH = 1 << ASIZE;
    localparam AFULL_THRESHOLD = DEPTH - 2;
    localparam AEMPTY_THRESHOLD = 1;

    // FIFO for A to B direction
    reg [DSIZE-1:0] mem_a2b [0:DEPTH-1];
    reg [ASIZE:0] wptr_a2b, rptr_a2b, wptr_a2b_sync, rptr_a2b_sync;
    reg [ASIZE:0] wptr_a2b_gray, rptr_a2b_gray, wptr_a2b_gray_sync, rptr_a2b_gray_sync;

    // FIFO for B to A direction
    reg [DSIZE-1:0] mem_b2a [0:DEPTH-1];
    reg [ASIZE:0] wptr_b2a, rptr_b2a, wptr_b2a_sync, rptr_b2a_sync;
    reg [ASIZE:0] wptr_b2a_gray, rptr_b2a_gray, wptr_b2a_gray_sync, rptr_b2a_gray_sync;

    // Gray code conversion functions
    function [ASIZE:0] bin2gray;
        input [ASIZE:0] bin;
        bin2gray = (bin >> 1) ^ bin;
    endfunction

    function [ASIZE:0] gray2bin;
        input [ASIZE:0] gray;
        integer i;
        reg [ASIZE:0] bin;
        bin = gray;
        for (i = ASIZE; i > 0; i = i - 1)
            bin[i-1] = bin[i] ^ bin[i-1];
        return bin;
    endfunction

    // A to B FIFO write and read logic
    always @(posedge a_clk or negedge a_rst_n) begin
        if (!a_rst_n) begin
            wptr_a2b <= 0;
            rptr_a2b <= 0;
        end else begin
            if (a_dir && !a_full) begin
                mem_a2b[wptr_a2b[ASIZE-1:0]] <= a_wdata;
                wptr_a2b <= wptr_a2b + 1;
            end
        end
    end

    always @(posedge b_clk or negedge b_rst_n) begin
        if (!b_rst_n) begin
            wptr_a2b_gray_sync <= 0;
            rptr_a2b_gray_sync <= 0;
        end else begin
            wptr_a2b_gray_sync <= bin2gray(wptr_a2b);
            rptr_a2b_gray_sync <= rptr_a2b_gray;
        end
    end

    always @(posedge b_clk or negedge b_rst_n) begin
        if (!b_rst_n) begin
            rptr_a2b <= 0;
        end else begin
            if (!b_dir && !b_empty) begin
                rptr_a2b <= rptr_a2b + 1;
            end
        end
    end

    // B to A FIFO write and read logic
    always @(posedge b_clk or negedge b_rst_n) begin
        if (!b_rst_n) begin
            wptr_b2a <= 0;
            rptr_b2a <= 0;
        end else begin
            if (b_dir && !b_full) begin
                mem_b2a[wptr_b2a[ASIZE-1:0]] <= b_wdata;
                wptr_b2a <= wptr_b2a + 1;
            end
        end
    end

    always @(posedge a_clk or negedge a_rst_n) begin
        if (!a_rst_n) begin
            wptr_b2a_gray_sync <= 0;
            rptr_b2a_gray_sync <= 0;
        end else begin
            wptr_b2a_gray_sync <= bin2gray(wptr_b2a);
            rptr_b2a_gray_sync <= rptr_b2a_gray;
        end
    end

    always @(posedge a_clk or negedge a_rst_n) begin
        if (!a_rst_n) begin
            rptr_b2a <= 0;
        end else begin
            if (!a_dir && !a_empty) begin
                rptr_b2a <= rptr_b2a + 1;
            end
        end
    end

    // A side outputs
    assign a_rdata = mem_a2b[rptr_a2b[ASIZE-1:0]];
    assign a_full = (wptr_a2b_gray_sync[ASIZE:0] == {~rptr_a2b_gray_sync[ASIZE], rptr_a2b_gray_sync[ASIZE-1:0]});
    assign a_afull = (wptr_a2b_gray_sync[ASIZE:0] == {~rptr_a2b_gray_sync[ASIZE], rptr_a2b_gray_sync[ASIZE-1:0] + AFULL_THRESHOLD});
    assign a_empty = (wptr_a2b_gray_sync[ASIZE:0] == rptr_a2b_gray_sync[ASIZE:0]);
    assign a_aempty = (wptr_a2b_gray_sync[ASIZE:0] == {rptr_a2b_gray_sync[ASIZE], rptr_a2b_gray_sync[ASIZE-1:0] + AEMPTY_THRESHOLD});

    // B side outputs
    assign b_rdata = mem_b2a[rptr_b2a[ASIZE-1:0]];
    assign b_full = (wptr_b2a_gray_sync[ASIZE:0] == {~rptr_b2a_gray_sync[ASIZE], rptr_b2a_gray_sync[ASIZE-1:0]});
    assign b_afull = (wptr_b2a_gray_sync[ASIZE:0] == {~rptr_b2a_gray_sync[ASIZE], rptr_b2a_gray_sync[ASIZE-1:0] + AFULL_THRESHOLD});
    assign b_empty = (wptr_b2a_gray_sync[ASIZE:0] == rptr_b2a_gray_sync[ASIZE:0]);
    assign b_aempty = (wptr_b2a_gray_sync[ASIZE:0] == {rptr_b2a_gray_sync[ASIZE], rptr_b2a_gray_sync[ASIZE-1:0] + AEMPTY_THRESHOLD});

endmodule
