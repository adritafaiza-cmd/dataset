module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;
  localparam AWIDTH = ASIZE;

  reg [DSIZE-1:0] mem [0:DEPTH-1];

  // A-side signals
  reg [AWIDTH:0] a_wr_ptr, a_wr_ptr_gray, a_wr_ptr_gray_nxt;
  reg [AWIDTH:0] a_rd_ptr, a_rd_ptr_gray, a_rd_ptr_gray_nxt;
  reg [AWIDTH:0] a_wr_ptr_gray_sync, a_rd_ptr_gray_sync;

  // B-side signals
  reg [AWIDTH:0] b_wr_ptr, b_wr_ptr_gray, b_wr_ptr_gray_nxt;
  reg [AWIDTH:0] b_rd_ptr, b_rd_ptr_gray, b_rd_ptr_gray_nxt;
  reg [AWIDTH:0] b_wr_ptr_gray_sync, b_rd_ptr_gray_sync;

  // A-side logic
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= 0;
      a_wr_ptr_gray <= 0;
      a_wr_ptr_gray_nxt <= 0;
    end else begin
      a_wr_ptr_gray_nxt <= a_wr_ptr + 1;
      a_wr_ptr_gray <= a_wr_ptr_gray_nxt;
      if (a_winc && !a_full)
        a_wr_ptr <= a_wr_ptr + 1;
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_rd_ptr <= 0;
      a_rd_ptr_gray <= 0;
      a_rd_ptr_gray_nxt <= 0;
    end else begin
      a_rd_ptr_gray_nxt <= a_rd_ptr + 1;
      a_rd_ptr_gray <= a_rd_ptr_gray_nxt;
      if (a_rinc && !a_empty)
        a_rd_ptr <= a_rd_ptr + 1;
    end
  end

  // B-side logic
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= 0;
      b_wr_ptr_gray <= 0;
      b_wr_ptr_gray_nxt <= 0;
    end else begin
      b_wr_ptr_gray_nxt <= b_wr_ptr + 1;
      b_wr_ptr_gray <= b_wr_ptr_gray_nxt;
      if (b_winc && !b_full)
        b_wr_ptr <= b_wr_ptr + 1;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_rd_ptr <= 0;
      b_rd_ptr_gray <= 0;
      b_rd_ptr_gray_nxt <= 0;
    end else begin
      b_rd_ptr_gray_nxt <= b_rd_ptr + 1;
      b_rd_ptr_gray <= b_rd_ptr_gray_nxt;
      if (b_rinc && !b_empty)
        b_rd_ptr <= b_rd_ptr + 1;
    end
  end

  // Gray code synchronization
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      b_wr_ptr_gray_sync <= 0;
      b_rd_ptr_gray_sync <= 0;
    end else begin
      b_wr_ptr_gray_sync <= b_wr_ptr_gray;
      b_rd_ptr_gray_sync <= b_rd_ptr_gray;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      a_wr_ptr_gray_sync <= 0;
      a_rd_ptr_gray_sync <= 0;
    end else begin
      a_wr_ptr_gray_sync <= a_wr_ptr_gray;
      a_rd_ptr_gray_sync <= a_rd_ptr_gray;
    end
  end

  // Write and read data
  always @(posedge a_clk) begin
    if (a_winc && !a_full)
      mem[a_wr_ptr[AWIDTH-1:0]] <= a_wdata;
  end

  always @(posedge b_clk) begin
    if (b_winc && !b_full)
      mem[b_wr_ptr[AWIDTH-1:0]] <= b_wdata;
  end

  // Read data
  assign a_rdata = mem[a_rd_ptr[AWIDTH-1:0]];
  assign b_rdata = mem[b_rd_ptr[AWIDTH-1:0]];

  // Full and empty flags
  wire [AWIDTH:0] a_full_flag, a_empty_flag;
  wire [AWIDTH:0] b_full_flag, b_empty_flag;

  assign a_full_flag = a_wr_ptr_gray_nxt ^ a_rd_ptr_gray_sync;
  assign a_empty_flag = a_wr_ptr_gray_sync ^ a_rd_ptr_gray_nxt;
  assign b_full_flag = b_wr_ptr_gray_nxt ^ b_rd_ptr_gray_sync;
  assign b_empty_flag = b_wr_ptr_gray_sync ^ b_rd_ptr_gray_nxt;

  assign a_full = a_full_flag[AWIDTH];
  assign a_afull = a_full_flag[AWIDTH-1];
  assign a_empty = a_empty_flag[AWIDTH];
  assign a_aempty = a_empty_flag[AWIDTH-1];

  assign b_full = b_full_flag[AWIDTH];
  assign b_afull = b_full_flag[AWIDTH-1];
  assign b_empty = b_empty_flag[AWIDTH];
  assign b_aempty = b_empty_flag[AWIDTH-1];

endmodule
