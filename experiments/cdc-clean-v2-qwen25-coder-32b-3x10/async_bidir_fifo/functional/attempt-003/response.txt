module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam DEPTH = 1 << ASIZE;

  reg [DSIZE-1:0] mem [0:DEPTH-1];

  reg [ASIZE:0] a_wr_ptr, a_rd_ptr, a_wr_ptr_gray, a_rd_ptr_gray;
  reg [ASIZE:0] b_wr_ptr, b_rd_ptr, b_wr_ptr_gray, b_rd_ptr_gray;

  reg [DSIZE-1:0] a_rdata_reg, b_rdata_reg;

  // Gray code conversion functions
  function [ASIZE:0] bin_to_gray;
    input [ASIZE:0] bin;
    bin_to_gray = bin ^ (bin >> 1);
  endfunction

  function [ASIZE:0] gray_to_bin;
    input [ASIZE:0] gray;
    integer i;
    reg [ASIZE:0] bin;
    bin = gray;
    for (i = ASIZE; i > 0; i = i - 1)
      bin[i-1] = bin[i] ^ bin[i-1];
    gray_to_bin = bin;
  endfunction

  // A-side logic
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_wr_ptr <= 0;
      a_wr_ptr_gray <= 0;
      a_rd_ptr <= 0;
      a_rd_ptr_gray <= 0;
    end else begin
      if (a_dir) begin // Write
        if (!a_full) begin
          mem[a_wr_ptr[ASIZE-1:0]] <= a_wdata;
          a_wr_ptr <= a_wr_ptr + 1;
          a_wr_ptr_gray <= bin_to_gray(a_wr_ptr + 1);
        end
      end else begin // Read
        if (!a_empty) begin
          a_rd_ptr <= a_rd_ptr + 1;
          a_rd_ptr_gray <= bin_to_gray(a_rd_ptr + 1);
        end
      end
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_wr_ptr <= 0;
      b_wr_ptr_gray <= 0;
      b_rd_ptr <= 0;
      b_rd_ptr_gray <= 0;
    end else begin
      if (b_dir) begin // Write
        if (!b_full) begin
          mem[b_wr_ptr[ASIZE-1:0]] <= b_wdata;
          b_wr_ptr <= b_wr_ptr + 1;
          b_wr_ptr_gray <= bin_to_gray(b_wr_ptr + 1);
        end
      end else begin // Read
        if (!b_empty) begin
          b_rd_ptr <= b_rd_ptr + 1;
          b_rd_ptr_gray <= bin_to_gray(b_rd_ptr + 1);
        end
      end
    end
  end

  // A-side read data
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n)
      a_rdata_reg <= 0;
    else if (a_rinc && !a_empty)
      a_rdata_reg <= mem[a_rd_ptr[ASIZE-1:0]];
  end

  // B-side read data
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n)
      b_rdata_reg <= 0;
    else if (b_rinc && !b_empty)
      b_rdata_reg <= mem[b_rd_ptr[ASIZE-1:0]];
  end

  // A-side flags
  wire [ASIZE:0] a_wr_ptr_gray_sync, a_rd_ptr_gray_sync;
  sync_gray #(.WIDTH(ASIZE+1)) sync_a_wr_ptr_gray (.clk(a_clk), .rst_n(a_rst_n), .din(b_wr_ptr_gray), .dout(a_wr_ptr_gray_sync));
  sync_gray #(.WIDTH(ASIZE+1)) sync_a_rd_ptr_gray (.clk(a_clk), .rst_n(a_rst_n), .din(b_rd_ptr_gray), .dout(a_rd_ptr_gray_sync));

  assign a_full = (a_wr_ptr_gray_sync == bin_to_gray(a_wr_ptr + 1));
  assign a_afull = (a_wr_ptr_gray_sync == bin_to_gray(a_wr_ptr + DEPTH - 1));
  assign a_empty = (a_wr_ptr_gray == a_rd_ptr_gray_sync);
  assign a_aempty = (a_wr_ptr_gray == bin_to_gray(a_rd_ptr_sync + 1));

  // B-side flags
  wire [ASIZE:0] b_wr_ptr_gray_sync, b_rd_ptr_gray_sync;
  sync_gray #(.WIDTH(ASIZE+1)) sync_b_wr_ptr_gray (.clk(b_clk), .rst_n(b_rst_n), .din(a_wr_ptr_gray), .dout(b_wr_ptr_gray_sync));
  sync_gray #(.WIDTH(ASIZE+1)) sync_b_rd_ptr_gray (.clk(b_clk), .rst_n(b_rst_n), .din(a_rd_ptr_gray), .dout(b_rd_ptr_gray_sync));

  assign b_full = (b_wr_ptr_gray_sync == bin_to_gray(b_wr_ptr + 1));
  assign b_afull = (b_wr_ptr_gray_sync == bin_to_gray(b_wr_ptr + DEPTH - 1));
  assign b_empty = (b_wr_ptr_gray == b_rd_ptr_gray_sync);
  assign b_aempty = (b_wr_ptr_gray == bin_to_gray(b_rd_ptr_sync + 1));

  // Output assignments
  assign a_rdata = a_rdata_reg;
  assign b_rdata = b_rdata_reg;

endmodule

// Gray code synchronizer
module sync_gray
  #(
    parameter WIDTH = 5
    ) (
       input wire              clk,
       input wire              rst_n,
       input wire [WIDTH-1:0]  din,
       output reg [WIDTH-1:0]  dout
       );

  reg [WIDTH-1:0] sync_reg1, sync_reg2;

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      sync_reg1 <= 0;
      sync_reg2 <= 0;
      dout <= 0;
    end else begin
      sync_reg1 <= din;
      sync_reg2 <= sync_reg1;
      dout <= sync_reg2;
    end
  end

endmodule
