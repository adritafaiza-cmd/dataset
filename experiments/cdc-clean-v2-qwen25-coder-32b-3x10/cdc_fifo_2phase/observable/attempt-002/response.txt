module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    reg [WIDTH-1:0] mem [DEPTH-1:0];
    reg [ADDR_WIDTH:0] src_wr_ptr, src_wr_ptr_gray, src_wr_ptr_gray_sync;
    reg [ADDR_WIDTH:0] dst_rd_ptr, dst_rd_ptr_gray, dst_rd_ptr_gray_sync;
    reg [ADDR_WIDTH:0] src_wr_ptr_gray_synced;
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray_synced;

    // Gray code conversion functions
    function [ADDR_WIDTH:0] bin_to_gray;
        input [ADDR_WIDTH:0] bin;
        bin_to_gray = bin ^ (bin >> 1);
    endfunction

    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = ADDR_WIDTH; i > 0; i = i - 1) begin
            bin[i-1] = bin[i] ^ bin[i-1];
        end
        gray_to_bin = bin;
    endfunction

    // Source side logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= 0;
            src_wr_ptr_gray <= 0;
        end else if (src_valid_i && src_ready_o) begin
            mem[src_wr_ptr[ADDR_WIDTH-1:0]] <= src_data_i;
            src_wr_ptr <= src_wr_ptr + 1;
            src_wr_ptr_gray <= bin_to_gray(src_wr_ptr + 1);
        end
    end

    // Synchronize src_wr_ptr_gray to destination clock domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_gray_sync <= 0;
            src_wr_ptr_gray_synced <= 0;
        end else begin
            src_wr_ptr_gray_sync <= src_wr_ptr_gray;
            src_wr_ptr_gray_synced <= src_wr_ptr_gray_sync;
        end
    end

    // Destination side logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= 0;
            dst_rd_ptr_gray <= 0;
        end else if (dst_valid_o && dst_ready_i) begin
            dst_rd_ptr <= dst_rd_ptr + 1;
            dst_rd_ptr_gray <= bin_to_gray(dst_rd_ptr + 1);
        end
    end

    // Synchronize dst_rd_ptr_gray to source clock domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_gray_sync <= 0;
            dst_rd_ptr_gray_synced <= 0;
        end else begin
            dst_rd_ptr_gray_sync <= dst_rd_ptr_gray;
            dst_rd_ptr_gray_synced <= dst_rd_ptr_gray_sync;
        end
    end

    // Calculate full and empty conditions
    wire [ADDR_WIDTH:0] src_wr_ptr_gray_synced_bin = gray_to_bin(src_wr_ptr_gray_synced);
    wire [ADDR_WIDTH:0] dst_rd_ptr_gray_synced_bin = gray_to_bin(dst_rd_ptr_gray_synced);
    wire full = (src_wr_ptr_gray_synced_bin[ADDR_WIDTH] != dst_rd_ptr_gray_synced_bin[ADDR_WIDTH]) &&
                (src_wr_ptr_gray_synced_bin[ADDR_WIDTH-1:0] == dst_rd_ptr_gray_synced_bin[ADDR_WIDTH-1:0]);
    wire empty = (src_wr_ptr_gray_synced_bin == dst_rd_ptr_gray_synced_bin);

    // Ready and valid signals
    assign src_ready_o = !full;
    assign dst_valid_o = !empty;

    // Output data register
    reg [WIDTH-1:0] dst_data_reg;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_data_reg <= 0;
        end else if (dst_valid_o && dst_ready_i) begin
            dst_data_reg <= mem[dst_rd_ptr[ADDR_WIDTH-1:0]];
        end
    end

    assign dst_data_o = dst_data_reg;

endmodule
