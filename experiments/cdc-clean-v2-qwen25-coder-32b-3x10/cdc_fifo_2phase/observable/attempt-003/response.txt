module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 2**LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    reg [WIDTH-1:0] mem [DEPTH-1:0];
    reg [ADDR_WIDTH:0] src_wr_ptr, src_wr_ptr_gray, src_wr_ptr_gray_sync;
    reg [ADDR_WIDTH:0] dst_rd_ptr, dst_rd_ptr_gray, dst_rd_ptr_gray_sync;

    // Gray code conversion functions
    function [ADDR_WIDTH:0] bin_to_gray;
        input [ADDR_WIDTH:0] bin;
        bin_to_gray = bin ^ (bin >> 1);
    endfunction

    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = ADDR_WIDTH; i > 0; i = i - 1) begin
            bin[i-1] = bin[i] ^ bin[i-1];
        end
        gray_to_bin = bin;
    endfunction

    // Source clock domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= 0;
            src_wr_ptr_gray <= 0;
        end else if (src_ready_o && src_valid_i) begin
            src_wr_ptr <= src_wr_ptr + 1;
            src_wr_ptr_gray <= bin_to_gray(src_wr_ptr + 1);
        end
    end

    // Synchronize src_wr_ptr_gray to destination clock domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_gray_sync <= 0;
        end else begin
            src_wr_ptr_gray_sync <= src_wr_ptr_gray;
        end
    end

    // Destination clock domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= 0;
            dst_rd_ptr_gray <= 0;
        end else if (dst_valid_o && dst_ready_i) begin
            dst_rd_ptr <= dst_rd_ptr + 1;
            dst_rd_ptr_gray <= bin_to_gray(dst_rd_ptr + 1);
        end
    end

    // Synchronize dst_rd_ptr_gray to source clock domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_gray_sync <= 0;
        end else begin
            dst_rd_ptr_gray_sync <= dst_rd_ptr_gray;
        end
    end

    // Calculate src_ready_o
    assign src_ready_o = (gray_to_bin(src_wr_ptr_gray) + 1) != gray_to_bin(dst_rd_ptr_gray_sync);

    // Calculate dst_valid_o
    assign dst_valid_o = (gray_to_bin(src_wr_ptr_gray_sync) != gray_to_bin(dst_rd_ptr_gray));

    // Write to memory on source clock
    always @(posedge src_clk_i) begin
        if (src_ready_o && src_valid_i) begin
            mem[gray_to_bin(src_wr_ptr_gray)] <= src_data_i;
        end
    end

    // Read from memory on destination clock
    always @(posedge dst_clk_i) begin
        if (dst_valid_o && dst_ready_i) begin
            dst_data_o <= mem[gray_to_bin(dst_rd_ptr_gray)];
        end
    end

endmodule
