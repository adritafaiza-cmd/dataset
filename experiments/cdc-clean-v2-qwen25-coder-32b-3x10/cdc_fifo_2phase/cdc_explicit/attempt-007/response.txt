module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    reg [WIDTH-1:0] mem [DEPTH-1:0];
    reg [ADDR_WIDTH:0] src_wr_ptr, src_wr_ptr_gray, src_wr_ptr_gray_sync;
    reg [ADDR_WIDTH:0] dst_rd_ptr, dst_rd_ptr_gray, dst_rd_ptr_gray_sync;
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_sync;

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= 0;
        end else if (src_valid_i && src_ready_o) begin
            src_wr_ptr <= src_wr_ptr + 1;
        end
    end

    // Convert write pointer to Gray code
    assign src_wr_ptr_gray = (src_wr_ptr >> 1) ^ src_wr_ptr;

    // Synchronize Gray code write pointer to destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_gray_sync <= 0;
        end else begin
            src_wr_ptr_gray_sync <= src_wr_ptr_gray;
        end
    end

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= 0;
        end else if (dst_valid_o && dst_ready_i) begin
            dst_rd_ptr <= dst_rd_ptr + 1;
        end
    end

    // Convert read pointer to Gray code
    assign dst_rd_ptr_gray = (dst_rd_ptr >> 1) ^ dst_rd_ptr;

    // Synchronize Gray code read pointer to source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_gray_sync <= 0;
        end else begin
            dst_rd_ptr_gray_sync <= dst_rd_ptr_gray;
        end
    end

    // Synchronize Gray code write pointer to destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync <= 0;
        end else begin
            dst_wr_ptr_gray_sync <= src_wr_ptr_gray_sync;
        end
    end

    // Convert synchronized Gray code pointers back to binary
    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        begin
            gray_to_bin = gray[ADDR_WIDTH];
            for (i = ADDR_WIDTH - 1; i >= 0; i = i - 1) begin
                gray_to_bin[i] = gray_to_bin[i + 1] ^ gray[i];
            end
        end
    endfunction

    wire [ADDR_WIDTH:0] src_wr_ptr_sync = gray_to_bin(src_wr_ptr_gray_sync);
    wire [ADDR_WIDTH:0] dst_rd_ptr_sync = gray_to_bin(dst_rd_ptr_gray_sync);

    // FIFO full and empty conditions
    wire full = (src_wr_ptr_sync[ADDR_WIDTH] != dst_wr_ptr_gray_sync[ADDR_WIDTH]) &&
               (src_wr_ptr_sync[ADDR_WIDTH-1:0] == dst_wr_ptr_gray_sync[ADDR_WIDTH-1:0]);
    wire empty = (dst_rd_ptr_sync[ADDR_WIDTH] != dst_wr_ptr_gray_sync[ADDR_WIDTH]) &&
                (dst_rd_ptr_sync[ADDR_WIDTH-1:0] == dst_wr_ptr_gray_sync[ADDR_WIDTH-1:0]);

    // Source ready signal
    assign src_ready_o = !full;

    // Write data to FIFO
    always @(posedge src_clk_i) begin
        if (src_valid_i && src_ready_o) begin
            mem[src_wr_ptr[ADDR_WIDTH-1:0]] <= src_data_i;
        end
    end

    // Destination valid signal
    reg [1:0] dst_valid_reg;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_valid_reg <= 0;
        end else if (dst_valid_o && dst_ready_i) begin
            dst_valid_reg <= {dst_valid_reg[0], 1'b0};
        end else if (!empty) begin
            dst_valid_reg <= {dst_valid_reg[0], 1'b1};
        end
    end

    assign dst_valid_o = dst_valid_reg[1];

    // Read data from FIFO
    always @(posedge dst_clk_i) begin
        if (!empty && dst_valid_o && dst_ready_i) begin
            dst_data_o <= mem[dst_rd_ptr[ADDR_WIDTH-1:0]];
        end
    end

endmodule
