module apb_regs #(
  parameter int unsigned        NoApbRegs    = 32'd0,
  parameter int unsigned        ApbAddrWidth = 32'd0,
  parameter int unsigned        AddrOffset   = 32'd4,
  parameter int unsigned        ApbDataWidth = 32'd0,
  parameter int unsigned        RegDataWidth = 32'd0,
  parameter bit [NoApbRegs-1:0] ReadOnly     = 32'h0,
  parameter type                req_t        = logic,
  parameter type                resp_t       = logic,
  parameter type apb_addr_t                  = logic[ApbAddrWidth-1:0],
  parameter type reg_data_t                  = logic[RegDataWidth-1:0]
) (
  input  logic                      pclk_i,
  input  logic                      preset_ni,
  input  req_t                      req_i,
  output resp_t                     resp_o,
  input  apb_addr_t                 base_addr_i,
  input  reg_data_t [NoApbRegs-1:0] reg_init_i,
  output reg_data_t [NoApbRegs-1:0] reg_q_o
);

  // Internal signals
  logic [ApbAddrWidth-1:0] addr;
  logic [NoApbRegs-1:0]    reg_sel;
  logic [RegDataWidth-1:0] reg_data;
  logic [ApbDataWidth-1:0] write_data;
  logic [ApbDataWidth/8-1:0] pstrb;
  logic                      penable;
  logic                      psel;
  logic                      read;
  logic                      write;
  logic                      valid_access;

  // Decode the address
  assign addr = req_i.paddr - base_addr_i;
  assign reg_sel = (addr < NoApbRegs * AddrOffset) ? (1 << (addr / AddrOffset)) : '0;

  // Determine if the access is valid
  assign valid_access = (req_i.psel && req_i.penable && (addr < NoApbRegs * AddrOffset));

  // Determine if the access is a read or write
  assign read = valid_access && !req_i.pwrite;
  assign write = valid_access && req_i.pwrite;

  // Byte enable signals
  assign pstrb = req_i.pstrb;

  // Write data with byte enable
  always_comb begin
    write_data = '0;
    for (int i = 0; i < ApbDataWidth/8; i++) begin
      if (pstrb[i]) begin
        write_data[i*8 +: 8] = req_i.pwdata[i*8 +: 8];
      end
    end
  end

  // Register file
  reg_data_t [NoApbRegs-1:0] reg_file;

  always_ff @(posedge pclk_i or negedge preset_ni) begin
    if (!preset_ni) begin
      for (int i = 0; i < NoApbRegs; i++) begin
        if (ReadOnly[i]) begin
          reg_file[i] <= reg_init_i[i];
        end else begin
          reg_file[i] <= '0;
        end
      end
    end else begin
      for (int i = 0; i < NoApbRegs; i++) begin
        if (write && reg_sel[i] && !ReadOnly[i]) begin
          reg_file[i] <= (reg_file[i] & ~(pstrb << (i*8))) | (write_data & (pstrb << (i*8)));
        end
      end
    end
  end

  // Output the current register values
  always_comb begin
    reg_q_o = '0;
    for (int i = 0; i < NoApbRegs; i++) begin
      if (reg_sel[i]) begin
        reg_q_o[i] = ReadOnly[i] ? reg_init_i[i] : reg_file[i];
      end
    end
  end

  // APB response
  always_comb begin
    resp_o.prdata = '0;
    resp_o.pready = 1'b1;
    resp_o.pslverr = 1'b0;

    if (read) begin
      for (int i = 0; i < NoApbRegs; i++) begin
        if (reg_sel[i]) begin
          resp_o.prdata = ReadOnly[i] ? reg_init_i[i] : reg_file[i];
        end
      end
    end
  end

endmodule
