module apb_regs #(
  parameter int unsigned        NoApbRegs    = 32'd4,
  parameter int unsigned        ApbAddrWidth = 32'd32,
  parameter int unsigned        AddrOffset   = 32'd4,
  parameter int unsigned        ApbDataWidth = 32'd32,
  parameter int unsigned        RegDataWidth = 32'd32,
  parameter bit [NoApbRegs-1:0] ReadOnly     = 32'h0,
  parameter type                req_t        = logic,
  parameter type                resp_t       = logic,
  parameter type apb_addr_t                  = logic[ApbAddrWidth-1:0],
  parameter type reg_data_t                  = logic[RegDataWidth-1:0]
) (
  input  logic                      pclk_i,
  input  logic                      preset_ni,
  input  req_t                      req_i,
  output resp_t                     resp_o,
  input  apb_addr_t                 base_addr_i,
  input  reg_data_t [NoApbRegs-1:0] reg_init_i,
  output reg_data_t [NoApbRegs-1:0] reg_q_o
);

  // Internal register file
  reg_data_t [NoApbRegs-1:0] reg_file;

  // Internal signals
  logic [NoApbRegs-1:0] reg_addr_match;
  logic [ApbDataWidth/8-1:0] byte_strobe;
  logic [ApbDataWidth-1:0] write_data;
  logic [ApbDataWidth-1:0] read_data;
  logic write_enable;

  // Decode the address to find the matching register
  always_comb begin
    reg_addr_match = '0;
    for (int i = 0; i < NoApbRegs; i++) begin
      if (req_i.paddr[ApbAddrWidth-1:AddrOffset] == i) begin
        reg_addr_match[i] = 1'b1;
      end
    end
  end

  // Byte strobe handling
  always_comb begin
    byte_strobe = req_i.pstrb;
  end

  // Write data handling
  always_comb begin
    write_data = '0;
    for (int i = 0; i < ApbDataWidth/8; i++) begin
      if (byte_strobe[i]) begin
        write_data[i*8 +: 8] = req_i.pwdata[i*8 +: 8];
      end
    end
  end

  // Write enable handling
  always_comb begin
    write_enable = req_i.psel && req_i.penable && req_i.pwrite;
  end

  // Register file update logic
  always_ff @(posedge pclk_i or negedge preset_ni) begin
    if (!preset_ni) begin
      for (int i = 0; i < NoApbRegs; i++) begin
        if (!ReadOnly[i]) begin
          reg_file[i] <= '0;
        end else begin
          reg_file[i] <= reg_init_i[i];
        end
      end
    end else if (write_enable) begin
      for (int i = 0; i < NoApbRegs; i++) begin
        if (reg_addr_match[i] && !ReadOnly[i]) begin
          reg_file[i] <= write_data;
        end
      end
    end
  end

  // Read data handling
  always_comb begin
    read_data = '0;
    for (int i = 0; i < NoApbRegs; i++) begin
      if (reg_addr_match[i]) begin
        read_data = reg_file[i];
      end
    end
  end

  // Response handling
  always_comb begin
    resp_o.prdata = read_data;
    resp_o.pready = 1'b1;
    resp_o.pslverr = 1'b0;
  end

  // Output the register values
  assign reg_q_o = reg_file;

endmodule
