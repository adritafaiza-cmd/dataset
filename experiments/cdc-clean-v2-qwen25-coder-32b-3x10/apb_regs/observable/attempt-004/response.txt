module apb_regs #(
  parameter int unsigned        NoApbRegs    = 32'd0,
  parameter int unsigned        ApbAddrWidth = 32'd0,
  parameter int unsigned        AddrOffset   = 32'd4,
  parameter int unsigned        ApbDataWidth = 32'd0,
  parameter int unsigned        RegDataWidth = 32'd0,
  parameter bit [NoApbRegs-1:0] ReadOnly     = 32'h0,
  parameter type                req_t        = logic,
  parameter type                resp_t       = logic,
  parameter type apb_addr_t                  = logic[ApbAddrWidth-1:0],
  parameter type reg_data_t                  = logic[RegDataWidth-1:0]
) (
  input  logic                      pclk_i,
  input  logic                      preset_ni,
  input  req_t                      req_i,
  output resp_t                     resp_o,
  input  apb_addr_t                 base_addr_i,
  input  reg_data_t [NoApbRegs-1:0] reg_init_i,
  output reg_data_t [NoApbRegs-1:0] reg_q_o
);

  // Internal register file
  reg_data_t reg_file [NoApbRegs-1:0];

  // Internal signals
  logic [ApbAddrWidth-1:0] addr;
  logic [NoApbRegs-1:0]    reg_addr_match;
  logic [RegDataWidth-1:0] write_data;
  logic [RegDataWidth/8-1:0] byte_strobe;
  logic                      write_enable;

  // Decode address
  always_comb begin
    addr = req_i.paddr - base_addr_i;
    for (int i = 0; i < NoApbRegs; i++) begin
      reg_addr_match[i] = (addr[ApbAddrWidth-1:AddrOffset] == i);
    end
  end

  // Byte strobe generation
  always_comb begin
    byte_strobe = '0;
    if (req_i.pwrite) begin
      for (int i = 0; i < RegDataWidth/8; i++) begin
        byte_strobe[i] = req_i.pstrb[i];
      end
    end
  end

  // Write data generation
  always_comb begin
    write_data = '0;
    for (int i = 0; i < RegDataWidth/8; i++) begin
      if (byte_strobe[i]) begin
        write_data[i*8 +: 8] = req_i.pwdata[i*8 +: 8];
      end
    end
  end

  // Write enable generation
  always_comb begin
    write_enable = req_i.psel && req_i.penable && req_i.pwrite;
  end

  // Register file update
  always_ff @(posedge pclk_i or negedge preset_ni) begin
    if (!preset_ni) begin
      for (int i = 0; i < NoApbRegs; i++) begin
        if (ReadOnly[i]) begin
          reg_file[i] <= reg_init_i[i];
        end else begin
          reg_file[i] <= '0;
        end
      end
    end else if (write_enable) begin
      for (int i = 0; i < NoApbRegs; i++) begin
        if (reg_addr_match[i] && !ReadOnly[i]) begin
          reg_file[i] <= (reg_file[i] & ~({RegDataWidth{byte_strobe}})) | (write_data & {RegDataWidth{byte_strobe}});
        end
      end
    end
  end

  // Output register values
  always_comb begin
    reg_q_o = '0;
    for (int i = 0; i < NoApbRegs; i++) begin
      if (reg_addr_match[i]) begin
        reg_q_o[i] = ReadOnly[i] ? reg_init_i[i] : reg_file[i];
      end
    end
  end

  // Response generation
  always_comb begin
    resp_o.prdata = '0;
    resp_o.pready = 1'b1;
    resp_o.pslverr = 1'b0;
    for (int i = 0; i < NoApbRegs; i++) begin
      if (reg_addr_match[i]) begin
        resp_o.prdata = ReadOnly[i] ? reg_init_i[i] : reg_file[i];
      end
    end
  end

endmodule
