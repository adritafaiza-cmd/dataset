module apb_regs #(
  parameter int unsigned        NoApbRegs    = 32'd0,
  parameter int unsigned        ApbAddrWidth = 32'd0,
  parameter int unsigned        AddrOffset   = 32'd4,
  parameter int unsigned        ApbDataWidth = 32'd0,
  parameter int unsigned        RegDataWidth = 32'd0,
  parameter bit [NoApbRegs-1:0] ReadOnly     = 32'h0,
  parameter type                req_t        = logic,
  parameter type                resp_t       = logic,
  parameter type apb_addr_t                  = logic[ApbAddrWidth-1:0],
  parameter type reg_data_t                  = logic[RegDataWidth-1:0]
) (
  input  logic                      pclk_i,
  input  logic                      preset_ni,
  input  req_t                      req_i,
  output resp_t                     resp_o,
  input  apb_addr_t                 base_addr_i,
  input  reg_data_t [NoApbRegs-1:0] reg_init_i,
  output reg_data_t [NoApbRegs-1:0] reg_q_o
);

  // Internal register file
  reg_data_t [NoApbRegs-1:0] reg_file;

  // Internal signals
  logic [ApbAddrWidth-1:0] addr_offset;
  logic [NoApbRegs-1:0]    reg_addr_match;
  logic [NoApbRegs-1:0]    reg_write_enable;
  logic [RegDataWidth-1:0] reg_write_data;
  logic [RegDataWidth-1:0] reg_read_data;
  logic                    psel;
  logic                    penable;
  logic                    pwrite;
  logic [ApbDataWidth-1:0] pwdata;
  logic [ApbDataWidth/8-1:0] pstrb;

  // Decode the APB request
  assign psel    = req_i.psel;
  assign penable = req_i.penable;
  assign pwrite  = req_i.pwrite;
  assign pwdata  = req_i.pwdata;
  assign pstrb   = req_i.pstrb;

  // Calculate the address offset
  assign addr_offset = req_i.paddr - base_addr_i;

  // Generate address match signals
  genvar i;
  generate
    for (i = 0; i < NoApbRegs; i = i + 1) begin : gen_addr_match
      assign reg_addr_match[i] = (addr_offset == i * AddrOffset);
    end
  endgenerate

  // Generate write enable signals
  generate
    for (i = 0; i < NoApbRegs; i = i + 1) begin : gen_write_enable
      assign reg_write_enable[i] = psel & penable & pwrite & reg_addr_match[i] & ~ReadOnly[i];
    end
  endgenerate

  // Generate write data with byte enable
  always_comb begin
    reg_write_data = '0;
    for (int j = 0; j < ApbDataWidth/8; j = j + 1) begin
      if (pstrb[j]) begin
        reg_write_data[j*8 +: 8] = pwdata[j*8 +: 8];
      end
    end
  end

  // Register file update logic
  always_ff @(posedge pclk_i or negedge preset_ni) begin
    if (!preset_ni) begin
      for (int i = 0; i < NoApbRegs; i = i + 1) begin
        if (ReadOnly[i]) begin
          reg_file[i] <= reg_init_i[i];
        end else begin
          reg_file[i] <= '0;
        end
      end
    end else begin
      for (int i = 0; i < NoApbRegs; i = i + 1) begin
        if (reg_write_enable[i]) begin
          reg_file[i] <= reg_file[i] ^ ((reg_file[i] ^ reg_write_data) & {RegDataWidth{reg_write_enable[i]}});
        end
      end
    end
  end

  // Generate read data
  always_comb begin
    reg_read_data = '0;
    for (int i = 0; i < NoApbRegs; i = i + 1) begin
      if (reg_addr_match[i]) begin
        reg_read_data = ReadOnly[i] ? reg_init_i[i] : reg_file[i];
      end
    end
  end

  // Generate APB response
  always_ff @(posedge pclk_i or negedge preset_ni) begin
    if (!preset_ni) begin
      resp_o.prdata <= '0;
      resp_o.pready <= 1'b0;
      resp_o.pslverr <= 1'b0;
    end else begin
      if (psel & penable) begin
        resp_o.prdata <= reg_read_data;
        resp_o.pready <= 1'b1;
        resp_o.pslverr <= 1'b0;
      end else begin
        resp_o.pready <= 1'b0;
      end
    end
  end

  // Output current register values
  assign reg_q_o = reg_file;

endmodule
