module apb_cdc #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_pclk_i,
    input                      src_preset_ni,
    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,
    input  [ADDR_WIDTH-1:0]    src_paddr_i,
    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    // FIFO for APB requests
    reg [ADDR_WIDTH-1:0]       req_addr_fifo [0:(1<<LOG_DEPTH)-1];
    reg [DATA_WIDTH-1:0]       req_data_fifo [0:(1<<LOG_DEPTH)-1];
    reg [DATA_WIDTH/8-1:0]     req_strb_fifo [0:(1<<LOG_DEPTH)-1];
    reg [2:0]                  req_prot_fifo [0:(1<<LOG_DEPTH)-1];
    reg                        req_write_fifo [0:(1<<LOG_DEPTH)-1];
    reg [$clog2(1<<LOG_DEPTH):0] req_wr_ptr, req_rd_ptr;
    reg [$clog2(1<<LOG_DEPTH):0] req_full_cnt, req_empty_cnt;

    // FIFO for APB responses
    reg [DATA_WIDTH-1:0]       resp_data_fifo [0:(1<<LOG_DEPTH)-1];
    reg                        resp_err_fifo [0:(1<<LOG_DEPTH)-1];
    reg [$clog2(1<<LOG_DEPTH):0] resp_wr_ptr, resp_rd_ptr;
    reg [$clog2(1<<LOG_DEPTH):0] resp_full_cnt, resp_empty_cnt;

    // Source side logic
    reg                        src_req_valid, src_req_ready;
    reg                        src_resp_valid, src_resp_ready;

    // Destination side logic
    reg                        dst_req_valid, dst_req_ready;
    reg                        dst_resp_valid, dst_resp_ready;

    // Source side request FIFO
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            req_wr_ptr <= 0;
            req_full_cnt <= 0;
        end else if (src_req_valid && src_req_ready) begin
            req_addr_fifo[req_wr_ptr] <= src_paddr_i;
            req_data_fifo[req_wr_ptr] <= src_pwdata_i;
            req_strb_fifo[req_wr_ptr] <= src_pstrb_i;
            req_prot_fifo[req_wr_ptr] <= src_pprot_i;
            req_write_fifo[req_wr_ptr] <= src_pwrite_i;
            req_wr_ptr <= req_wr_ptr + 1;
            req_full_cnt <= req_full_cnt + 1;
        end else if (dst_req_ready) begin
            req_full_cnt <= req_full_cnt - 1;
        end
    end

    // Source side request FIFO read pointer
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            req_rd_ptr <= 0;
            req_empty_cnt <= (1<<LOG_DEPTH);
        end else if (dst_req_ready) begin
            req_rd_ptr <= req_rd_ptr + 1;
            req_empty_cnt <= req_empty_cnt - 1;
        end else if (src_req_valid && src_req_ready) begin
            req_empty_cnt <= req_empty_cnt + 1;
        end
    end

    // Source side request FIFO status
    assign src_req_ready = (req_full_cnt < (1<<LOG_DEPTH));
    assign dst_req_valid = (req_empty_cnt < (1<<LOG_DEPTH));

    // Source side request FIFO data
    assign dst_paddr_o = req_addr_fifo[req_rd_ptr];
    assign dst_pwdata_o = req_data_fifo[req_rd_ptr];
    assign dst_pstrb_o = req_strb_fifo[req_rd_ptr];
    assign dst_pprot_o = req_prot_fifo[req_rd_ptr];
    assign dst_pwrite_o = req_write_fifo[req_rd_ptr];
    assign dst_psel_o = dst_req_valid;
    assign dst_penable_o = dst_req_valid;

    // Destination side response FIFO
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            resp_wr_ptr <= 0;
            resp_full_cnt <= 0;
        end else if (dst_resp_valid && dst_resp_ready) begin
            resp_data_fifo[resp_wr_ptr] <= dst_prdata_i;
            resp_err_fifo[resp_wr_ptr] <= dst_pslverr_i;
            resp_wr_ptr <= resp_wr_ptr + 1;
            resp_full_cnt <= resp_full_cnt + 1;
        end else if (src_resp_ready) begin
            resp_full_cnt <= resp_full_cnt - 1;
        end
    end

    // Destination side response FIFO read pointer
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            resp_rd_ptr <= 0;
            resp_empty_cnt <= (1<<LOG_DEPTH);
        end else if (src_resp_ready) begin
            resp_rd_ptr <= resp_rd_ptr + 1;
            resp_empty_cnt <= resp_empty_cnt - 1;
        end else if (dst_resp_valid && dst_resp_ready) begin
            resp_empty_cnt <= resp_empty_cnt + 1;
        end
    end

    // Destination side response FIFO status
    assign dst_resp_ready = (resp_full_cnt < (1<<LOG_DEPTH));
    assign src_resp_valid = (resp_empty_cnt < (1<<LOG_DEPTH));

    // Destination side response FIFO data
    assign src_prdata_o = resp_data_fifo[resp_rd_ptr];
    assign src_pslverr_o = resp_err_fifo[resp_rd_ptr];

    // Source side ready signal
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_pready_o <= 0;
        end else begin
            src_pready_o <= src_req_ready;
        end
    end

endmodule
