module apb_cdc #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_pclk_i,
    input                      src_preset_ni,
    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,
    input  [ADDR_WIDTH-1:0]    src_paddr_i,
    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    // FIFO signals
    reg [ADDR_WIDTH-1:0]       fifo_addr [0:(1<<LOG_DEPTH)-1];
    reg [DATA_WIDTH-1:0]       fifo_wdata [0:(1<<LOG_DEPTH)-1];
    reg [DATA_WIDTH/8-1:0]     fifo_strb [0:(1<<LOG_DEPTH)-1];
    reg [2:0]                  fifo_prot [0:(1<<LOG_DEPTH)-1];
    reg                        fifo_write;
    reg                        fifo_read;
    reg [LOG_DEPTH:0]          fifo_wptr;
    reg [LOG_DEPTH:0]          fifo_rptr;
    reg [LOG_DEPTH:0]          fifo_count;

    // Source side signals
    reg                        src_req_valid;
    reg                        src_req_ready;
    reg                        src_req_accepted;
    reg [ADDR_WIDTH-1:0]       src_req_addr;
    reg [DATA_WIDTH-1:0]       src_req_wdata;
    reg [DATA_WIDTH/8-1:0]     src_req_strb;
    reg [2:0]                  src_req_prot;
    reg                        src_req_write;

    // Destination side signals
    reg                        dst_req_valid;
    reg                        dst_req_ready;
    reg [ADDR_WIDTH-1:0]       dst_req_addr;
    reg [DATA_WIDTH-1:0]       dst_req_wdata;
    reg [DATA_WIDTH/8-1:0]     dst_req_strb;
    reg [2:0]                  dst_req_prot;
    reg                        dst_req_write;

    // FIFO write logic
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            fifo_wptr <= 0;
            fifo_count <= 0;
        end else if (fifo_write && !fifo_full) begin
            fifo_addr[fifo_wptr[LOG_DEPTH-1:0]] <= src_paddr_i;
            fifo_wdata[fifo_wptr[LOG_DEPTH-1:0]] <= src_pwdata_i;
            fifo_strb[fifo_wptr[LOG_DEPTH-1:0]] <= src_pstrb_i;
            fifo_prot[fifo_wptr[LOG_DEPTH-1:0]] <= src_pprot_i;
            fifo_wptr <= fifo_wptr + 1;
            fifo_count <= fifo_count + 1;
        end
    end

    // FIFO read logic
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            fifo_rptr <= 0;
        end else if (fifo_read && !fifo_empty) begin
            fifo_rptr <= fifo_rptr + 1;
            fifo_count <= fifo_count - 1;
        end
    end

    // FIFO status
    wire fifo_full = (fifo_count == (1<<LOG_DEPTH));
    wire fifo_empty = (fifo_count == 0);

    // Source side request handling
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_req_valid <= 0;
            src_req_accepted <= 0;
        end else begin
            if (src_psel_i && src_penable_i && !src_req_valid) begin
                src_req_valid <= 1;
                src_req_addr <= src_paddr_i;
                src_req_wdata <= src_pwdata_i;
                src_req_strb <= src_pstrb_i;
                src_req_prot <= src_pprot_i;
                src_req_write <= src_pwrite_i;
            end else if (src_req_accepted) begin
                src_req_valid <= 0;
                src_req_accepted <= 0;
            end
        end
    end

    // Source side ready signal
    assign src_pready_o = !src_req_valid || src_req_accepted;

    // FIFO write enable
    assign fifo_write = src_req_valid && !src_req_accepted && !fifo_full;

    // FIFO read enable
    assign fifo_read = dst_req_ready && !fifo_empty;

    // Destination side request handling
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            dst_req_valid <= 0;
        end else begin
            if (fifo_read) begin
                dst_req_valid <= 1;
                dst_req_addr <= fifo_addr[fifo_rptr[LOG_DEPTH-1:0]];
                dst_req_wdata <= fifo_wdata[fifo_rptr[LOG_DEPTH-1:0]];
                dst_req_strb <= fifo_strb[fifo_rptr[LOG_DEPTH-1:0]];
                dst_req_prot <= fifo_prot[fifo_rptr[LOG_DEPTH-1:0]];
                dst_req_write <= fifo_wptr[LOG_DEPTH-1:0] == fifo_rptr[LOG_DEPTH-1:0];
            end else if (dst_pready_i) begin
                dst_req_valid <= 0;
            end
        end
    end

    // Destination side ready signal
    assign dst_req_ready = !dst_req_valid || dst_pready_i;

    // Destination side outputs
    assign dst_psel_o = dst_req_valid;
    assign dst_penable_o = dst_req_valid;
    assign dst_pwrite_o = dst_req_write;
    assign dst_paddr_o = dst_req_addr;
    assign dst_pwdata_o = dst_req_wdata;
    assign dst_pstrb_o = dst_req_strb;
    assign dst_pprot_o = dst_req_prot;

    // Source side response handling
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_prdata_o <= 0;
            src_pslverr_o <= 0;
        end else if (src_req_accepted && dst_pready_i) begin
            src_prdata_o <= dst_prdata_i;
            src_pslverr_o <= dst_pslverr_i;
        end
    end

    // Source side request acceptance
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_req_accepted <= 0;
        end else if (src_req_valid && fifo_write) begin
            src_req_accepted <= 1;
        end
    end

endmodule
