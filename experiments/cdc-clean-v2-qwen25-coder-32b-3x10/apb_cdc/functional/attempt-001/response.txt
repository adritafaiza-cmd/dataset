module apb_cdc #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_pclk_i,
    input                      src_preset_ni,
    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,
    input  [ADDR_WIDTH-1:0]    src_paddr_i,
    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    // FIFO signals
    reg [ADDR_WIDTH-1:0]       fifo_addr;
    reg [DATA_WIDTH-1:0]       fifo_wdata;
    reg [DATA_WIDTH/8-1:0]     fifo_strb;
    reg [2:0]                  fifo_prot;
    reg                        fifo_wren;
    reg                        fifo_rden;
    reg [DATA_WIDTH-1:0]       fifo_rdata;
    reg                        fifo_empty;
    reg                        fifo_full;

    // Source side signals
    reg                        src_req_valid;
    reg                        src_req_ready;
    reg                        src_req_accepted;
    reg                        src_req_done;

    // Destination side signals
    reg                        dst_req_valid;
    reg                        dst_req_ready;
    reg                        dst_req_accepted;
    reg                        dst_req_done;

    // FIFO instantiation
    fifo_sync #(
        .DATA_WIDTH (ADDR_WIDTH + DATA_WIDTH + DATA_WIDTH/8 + 3),
        .LOG_DEPTH  (LOG_DEPTH)
    ) fifo_inst (
        .src_clk_i  (src_pclk_i),
        .src_rst_ni (src_preset_ni),
        .src_wren_i (fifo_wren),
        .src_wdata_i({src_paddr_i, src_pwdata_i, src_pstrb_i, src_pprot_i}),
        .src_full_o (fifo_full),
        .dst_clk_i  (dst_pclk_i),
        .dst_rst_ni (dst_preset_ni),
        .dst_rden_i (fifo_rden),
        .dst_rdata_o({fifo_addr, fifo_wdata, fifo_strb, fifo_prot}),
        .dst_empty_o(fifo_empty)
    );

    // Source side logic
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_req_valid <= 0;
            src_req_accepted <= 0;
            src_req_done <= 0;
            fifo_wren <= 0;
        end else begin
            if (src_psel_i && src_penable_i && !src_req_valid) begin
                src_req_valid <= 1;
            end
            if (src_req_valid && !fifo_full) begin
                fifo_wren <= 1;
                src_req_accepted <= 1;
            end else begin
                fifo_wren <= 0;
            end
            if (src_req_accepted && fifo_full) begin
                src_req_valid <= 0;
            end
            if (src_req_accepted && dst_req_done) begin
                src_req_done <= 1;
            end
            if (src_req_done && src_pready_o) begin
                src_req_done <= 0;
            end
        end
    end

    assign src_pready_o = src_req_done;
    assign src_prdata_o = dst_prdata_i;
    assign src_pslverr_o = dst_pslverr_i;

    // Destination side logic
    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            dst_req_valid <= 0;
            dst_req_accepted <= 0;
            dst_req_done <= 0;
            fifo_rden <= 0;
        end else begin
            if (!fifo_empty && !dst_req_valid) begin
                fifo_rden <= 1;
                dst_req_valid <= 1;
            end else begin
                fifo_rden <= 0;
            end
            if (dst_req_valid && dst_pready_i) begin
                dst_req_accepted <= 1;
            end
            if (dst_req_accepted && dst_pready_i) begin
                dst_req_done <= 1;
            end
            if (dst_req_done) begin
                dst_req_valid <= 0;
                dst_req_accepted <= 0;
                dst_req_done <= 0;
            end
        end
    end

    assign dst_psel_o = dst_req_valid;
    assign dst_penable_o = dst_req_valid;
    assign dst_pwrite_o = src_pwrite_i;
    assign dst_paddr_o = fifo_addr;
    assign dst_pwdata_o = fifo_wdata;
    assign dst_pstrb_o = fifo_strb;
    assign dst_pprot_o = fifo_prot;

endmodule

// Synchronous FIFO module
module fifo_sync #(
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_clk_i,
    input                      src_rst_ni,
    input                      src_wren_i,
    input  [DATA_WIDTH-1:0]    src_wdata_i,
    output                     src_full_o,
    input                      dst_clk_i,
    input                      dst_rst_ni,
    input                      dst_rden_i,
    output [DATA_WIDTH-1:0]    dst_rdata_o,
    output                     dst_empty_o
);

    localparam DEPTH = 1 << LOG_DEPTH;
    reg [DATA_WIDTH-1:0]       mem [0:DEPTH-1];
    reg [$clog2(DEPTH):0]      src_wr_ptr;
    reg [$clog2(DEPTH):0]      src_rd_ptr;
    reg [$clog2(DEPTH):0]      dst_wr_ptr;
    reg [$clog2(DEPTH):0]      dst_rd_ptr;
    reg [$clog2(DEPTH):0]      src_wr_ptr_gray;
    reg [$clog2(DEPTH):0]      src_rd_ptr_gray;
    reg [$clog2(DEPTH):0]      dst_wr_ptr_gray;
    reg [$clog2(DEPTH):0]      dst_rd_ptr_gray;

    // Gray code conversion
    function [$clog2(DEPTH):0] bin2gray;
        input [$clog2(DEPTH):0] bin;
        bin2gray = bin ^ (bin >> 1);
    endfunction

    // Source side write pointer update
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= 0;
            src_wr_ptr_gray <= 0;
        end else if (src_wren_i && !src_full_o) begin
            src_wr_ptr <= src_wr_ptr + 1;
            src_wr_ptr_gray <= bin2gray(src_wr_ptr + 1);
        end
    end

    // Destination side read pointer update
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= 0;
            dst_rd_ptr_gray <= 0;
        end else if (dst_rden_i && !dst_empty_o) begin
            dst_rd_ptr <= dst_rd_ptr + 1;
            dst_rd_ptr_gray <= bin2gray(dst_rd_ptr + 1);
        end
    end

    // Memory write
    always @(posedge src_clk_i) begin
        if (src_wren_i && !src_full_o) begin
            mem[src_wr_ptr] <= src_wdata_i;
        end
    end

    // Memory read
    always @(posedge dst_clk_i) begin
        dst_rdata_o <= mem[dst_rd_ptr];
    end

    // Gray code to binary conversion for comparison
    function [$clog2(DEPTH):0] gray2bin;
        input [$clog2(DEPTH):0] gray;
        integer i;
        reg [$clog2(DEPTH):0] bin;
        begin
            bin = gray;
            for (i = 1; i <= $clog2(DEPTH); i = i + 1) begin
                bin = bin ^ (gray >> i);
            end
            gray2bin = bin;
        end
    endfunction

    // Full and empty flags
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_full_o <= 0;
        end else begin
            src_full_o <= (src_wr_ptr_gray == {~dst_rd_ptr_gray[$clog2(DEPTH)], dst_rd_ptr_gray[$clog2(DEPTH)-1:0]});
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_empty_o <= 1;
        end else begin
            dst_empty_o <= (dst_wr_ptr_gray == dst_rd_ptr_gray);
        end
    end

endmodule
