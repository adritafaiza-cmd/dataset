module cdc_4phase #(
  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain signals
  logic [WIDTH-1:0] src_data_reg;
  logic src_valid_reg;
  logic src_ready_reg;

  // Destination domain signals
  logic [WIDTH-1:0] dst_data_reg [3:0];
  logic dst_valid_reg [3:0];
  logic [1:0] phase;

  // Source domain logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_reg <= RESET_MSG;
      src_valid_reg <= SEND_RESET_MSG;
      src_ready_reg <= 1'b0;
    end else begin
      if (src_ready_reg && src_valid_i) begin
        src_data_reg <= src_data_i;
        src_valid_reg <= 1'b1;
      end else if (dst_valid_reg[3]) begin
        src_valid_reg <= 1'b0;
      end
      src_ready_reg <= DECOUPLED || !src_valid_reg;
    end
  end

  assign src_ready_o = src_ready_reg;

  // Cross-domain phase logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      phase <= 2'b00;
    end else begin
      phase <= phase + 1'b1;
    end
  end

  // Destination domain logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg <= '{default: RESET_MSG};
      dst_valid_reg <= '{default: SEND_RESET_MSG};
    end else begin
      // Shift data and valid signals through the pipeline
      dst_data_reg[0] <= src_data_reg;
      dst_valid_reg[0] <= src_valid_reg;
      dst_data_reg[1] <= dst_data_reg[0];
      dst_valid_reg[1] <= dst_valid_reg[0];
      dst_data_reg[2] <= dst_data_reg[1];
      dst_valid_reg[2] <= dst_valid_reg[1];
      dst_data_reg[3] <= dst_data_reg[2];
      dst_valid_reg[3] <= dst_valid_reg[2];

      // Clear valid signal when acknowledged
      if (dst_valid_reg[3] && dst_ready_i) begin
        dst_valid_reg[3] <= 1'b0;
      end
    end
  end

  assign dst_data_o = dst_data_reg[3];
  assign dst_valid_o = dst_valid_reg[3];

endmodule
