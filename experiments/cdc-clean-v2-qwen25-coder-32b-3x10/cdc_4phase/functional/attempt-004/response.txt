module cdc_4phase #(
  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain signals
  logic [WIDTH-1:0] src_data_sync [3:0];
  logic [1:0] src_phase;
  logic src_valid_sync;

  // Destination domain signals
  logic [WIDTH-1:0] dst_data_sync [3:0];
  logic [1:0] dst_phase;
  logic dst_valid_sync;

  // Source domain logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_phase <= 2'b00;
      src_valid_sync <= 1'b0;
    end else begin
      src_phase <= src_phase + 1'b1;
      if (src_valid_i && src_ready_o) begin
        src_data_sync[src_phase] <= src_data_i;
        src_valid_sync <= 1'b1;
      end else if (dst_valid_sync) begin
        src_valid_sync <= 1'b0;
      end
    end
  end

  // Destination domain logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_phase <= 2'b00;
      dst_valid_sync <= 1'b0;
      dst_data_o <= SEND_RESET_MSG ? RESET_MSG : {WIDTH{1'b0}};
      dst_valid_o <= SEND_RESET_MSG;
    end else begin
      dst_phase <= dst_phase + 1'b1;
      if (dst_valid_sync && dst_ready_i) begin
        dst_valid_sync <= 1'b0;
      end else if (src_valid_sync) begin
        dst_data_o <= src_data_sync[dst_phase];
        dst_valid_sync <= 1'b1;
      end
    end
  end

  // Source ready signal
  always_comb begin
    if (DECOUPLED) begin
      src_ready_o = ~src_valid_sync;
    end else begin
      src_ready_o = ~src_valid_sync && ~dst_valid_sync;
    end
  end

endmodule
