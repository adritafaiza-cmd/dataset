module cdc_4phase #(
  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain signals
  logic [WIDTH-1:0] src_data_sync;
  logic src_valid_sync;
  logic src_ready;

  // Destination domain signals
  logic [WIDTH-1:0] dst_data_reg [3:0];
  logic dst_valid_reg [3:0];
  logic dst_ready_sync;

  // Source domain to destination domain synchronization
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_sync <= SEND_RESET_MSG ? RESET_MSG : '0;
      src_valid_sync <= SEND_RESET_MSG;
    end else begin
      if (src_ready) begin
        src_data_sync <= src_data_i;
        src_valid_sync <= src_valid_i;
      end
    end
  end

  // Destination domain synchronization
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg[0] <= SEND_RESET_MSG ? RESET_MSG : '0;
      dst_valid_reg[0] <= SEND_RESET_MSG;
    end else begin
      dst_data_reg[0] <= src_data_sync;
      dst_valid_reg[0] <= src_valid_sync;
    end
  end

  // Shift register for data and valid
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg[1] <= SEND_RESET_MSG ? RESET_MSG : '0;
      dst_valid_reg[1] <= SEND_RESET_MSG;
      dst_data_reg[2] <= SEND_RESET_MSG ? RESET_MSG : '0;
      dst_valid_reg[2] <= SEND_RESET_MSG;
      dst_data_reg[3] <= SEND_RESET_MSG ? RESET_MSG : '0;
      dst_valid_reg[3] <= SEND_RESET_MSG;
    end else begin
      dst_data_reg[1] <= dst_data_reg[0];
      dst_valid_reg[1] <= dst_valid_reg[0];
      dst_data_reg[2] <= dst_data_reg[1];
      dst_valid_reg[2] <= dst_valid_reg[1];
      dst_data_reg[3] <= dst_data_reg[2];
      dst_valid_reg[3] <= dst_valid_reg[2];
    end
  end

  // Synchronize dst_ready_i to src_clk_i
  logic dst_ready_sync_reg [1:0];
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_ready_sync_reg[0] <= 1'b0;
      dst_ready_sync_reg[1] <= 1'b0;
    end else begin
      dst_ready_sync_reg[0] <= dst_ready_i;
      dst_ready_sync_reg[1] <= dst_ready_sync_reg[0];
    end
  end
  assign dst_ready_sync = dst_ready_sync_reg[1];

  // Source ready logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ready <= 1'b0;
    end else begin
      if (DECOUPLED) begin
        src_ready <= ~src_valid_sync | dst_ready_sync;
      end else begin
        src_ready <= ~src_valid_sync & dst_ready_sync;
      end
    end
  end
  assign src_ready_o = src_ready;

  // Destination valid and data output
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_o <= SEND_RESET_MSG ? RESET_MSG : '0;
      dst_valid_o <= SEND_RESET_MSG;
    end else begin
      if (dst_valid_reg[3] & dst_ready_i) begin
        dst_data_o <= dst_data_reg[3];
        dst_valid_o <= 1'b0;
      end else if (dst_ready_i) begin
        dst_data_o <= dst_data_reg[3];
        dst_valid_o <= dst_valid_reg[3];
      end
    end
  end

endmodule
