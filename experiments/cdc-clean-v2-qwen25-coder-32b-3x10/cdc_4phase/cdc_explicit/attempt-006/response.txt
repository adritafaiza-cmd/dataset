module cdc_4phase #(
  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain signals
  logic [WIDTH-1:0] src_data_sync;
  logic src_valid_sync;
  logic src_ready;

  // Destination domain signals
  logic [WIDTH-1:0] dst_data_reg [3:0];
  logic dst_valid_reg [3:0];
  logic dst_ready_sync;

  // Synchronization flops for src_valid and src_data
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_valid_sync <= 1'b0;
      src_data_sync <= RESET_MSG;
    end else begin
      src_valid_sync <= src_valid_i;
      src_data_sync <= src_data_i;
    end
  end

  // Synchronization flops for dst_ready
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_ready_sync <= 1'b0;
    end else begin
      dst_ready_sync <= dst_ready_i;
    end
  end

  // Shift register for data and valid signals
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg[0] <= SEND_RESET_MSG ? RESET_MSG : {WIDTH{1'b0}};
      dst_valid_reg[0] <= SEND_RESET_MSG;
      dst_data_reg[1] <= SEND_RESET_MSG ? RESET_MSG : {WIDTH{1'b0}};
      dst_valid_reg[1] <= SEND_RESET_MSG;
      dst_data_reg[2] <= SEND_RESET_MSG ? RESET_MSG : {WIDTH{1'b0}};
      dst_valid_reg[2] <= SEND_RESET_MSG;
      dst_data_reg[3] <= SEND_RESET_MSG ? RESET_MSG : {WIDTH{1'b0}};
      dst_valid_reg[3] <= SEND_RESET_MSG;
    end else begin
      dst_data_reg[0] <= src_data_sync;
      dst_valid_reg[0] <= src_valid_sync;
      dst_data_reg[1] <= dst_data_reg[0];
      dst_valid_reg[1] <= dst_valid_reg[0];
      dst_data_reg[2] <= dst_data_reg[1];
      dst_valid_reg[2] <= dst_valid_reg[1];
      dst_data_reg[3] <= dst_data_reg[2];
      dst_valid_reg[3] <= dst_valid_reg[2];
    end
  end

  // Output assignments
  assign dst_data_o = dst_data_reg[3];
  assign dst_valid_o = dst_valid_reg[3];

  // Source ready logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ready <= 1'b0;
    end else begin
      if (DECOUPLED) begin
        src_ready <= ~dst_valid_reg[0];
      end else begin
        src_ready <= ~dst_valid_reg[0] & dst_ready_sync;
      end
    end
  end

  assign src_ready_o = src_ready;

endmodule
