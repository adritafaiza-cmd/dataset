module axil_cdc #
(
    parameter DATA_WIDTH = 32,
    parameter ADDR_WIDTH = 32,
    parameter STRB_WIDTH = (DATA_WIDTH/8)
)
(
    /*
     * AXI lite slave interface
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [ADDR_WIDTH-1:0]  s_axil_awaddr,
    input  wire [2:0]             s_axil_awprot,
    input  wire                   s_axil_awvalid,
    output wire                   s_axil_awready,
    input  wire [DATA_WIDTH-1:0]  s_axil_wdata,
    input  wire [STRB_WIDTH-1:0]  s_axil_wstrb,
    input  wire                   s_axil_wvalid,
    output wire                   s_axil_wready,
    output wire [1:0]             s_axil_bresp,
    output wire                   s_axil_bvalid,
    input  wire                   s_axil_bready,
    input  wire [ADDR_WIDTH-1:0]  s_axil_araddr,
    input  wire [2:0]             s_axil_arprot,
    input  wire                   s_axil_arvalid,
    output wire                   s_axil_arready,
    output wire [DATA_WIDTH-1:0]  s_axil_rdata,
    output wire [1:0]             s_axil_rresp,
    output wire                   s_axil_rvalid,
    input  wire                   s_axil_rready,
    /*
     * AXI lite master interface
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [ADDR_WIDTH-1:0]  m_axil_awaddr,
    output wire [2:0]             m_axil_awprot,
    output wire                   m_axil_awvalid,
    input  wire                   m_axil_awready,
    output wire [DATA_WIDTH-1:0]  m_axil_wdata,
    output wire [STRB_WIDTH-1:0]  m_axil_wstrb,
    output wire                   m_axil_wvalid,
    input  wire                   m_axil_wready,
    input  wire [1:0]             m_axil_bresp,
    input  wire                   m_axil_bvalid,
    output wire                   m_axil_bready,
    output wire [ADDR_WIDTH-1:0]  m_axil_araddr,
    output wire [2:0]             m_axil_arprot,
    output wire                   m_axil_arvalid,
    input  wire                   m_axil_arready,
    input  wire [DATA_WIDTH-1:0]  m_axil_rdata,
    input  wire [1:0]             m_axil_rresp,
    input  wire                   m_axil_rvalid,
    output wire                   m_axil_rready
);

    // Slave to Master signals
    reg [ADDR_WIDTH-1:0] s_awaddr_r;
    reg [2:0] s_awprot_r;
    reg s_awvalid_r;
    reg [DATA_WIDTH-1:0] s_wdata_r;
    reg [STRB_WIDTH-1:0] s_wstrb_r;
    reg s_wvalid_r;
    reg [ADDR_WIDTH-1:0] s_araddr_r;
    reg [2:0] s_arprot_r;
    reg s_arvalid_r;

    // Master to Slave signals
    reg [1:0] m_bresp_r;
    reg m_bvalid_r;
    reg [DATA_WIDTH-1:0] m_rdata_r;
    reg [1:0] m_rresp_r;
    reg m_rvalid_r;

    // Cross-domain synchronization
    reg [1:0] s_awvalid_sync;
    reg [1:0] s_wvalid_sync;
    reg [1:0] s_arvalid_sync;
    reg [1:0] m_awready_sync;
    reg [1:0] m_wready_sync;
    reg [1:0] m_arready_sync;
    reg [1:0] m_bvalid_sync;
    reg [1:0] m_rvalid_sync;

    // Slave side
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_awvalid_r <= 0;
            s_wvalid_r <= 0;
            s_arvalid_r <= 0;
            s_axil_awready <= 0;
            s_axil_wready <= 0;
            s_axil_arready <= 0;
        end else begin
            // AW Channel
            if (s_axil_awvalid && !s_awvalid_r && !s_awvalid_sync[1]) begin
                s_awaddr_r <= s_axil_awaddr;
                s_awprot_r <= s_axil_awprot;
                s_awvalid_r <= 1;
            end else if (s_awvalid_sync[1] && m_axil_awready) begin
                s_awvalid_r <= 0;
            end
            s_axil_awready <= !s_awvalid_r || s_awvalid_sync[1] && m_axil_awready;

            // W Channel
            if (s_axil_wvalid && !s_wvalid_r && !s_wvalid_sync[1]) begin
                s_wdata_r <= s_axil_wdata;
                s_wstrb_r <= s_axil_wstrb;
                s_wvalid_r <= 1;
            end else if (s_wvalid_sync[1] && m_axil_wready) begin
                s_wvalid_r <= 0;
            end
            s_axil_wready <= !s_wvalid_r || s_wvalid_sync[1] && m_axil_wready;

            // AR Channel
            if (s_axil_arvalid && !s_arvalid_r && !s_arvalid_sync[1]) begin
                s_araddr_r <= s_axil_araddr;
                s_arprot_r <= s_axil_arprot;
                s_arvalid_r <= 1;
            end else if (s_arvalid_sync[1] && m_axil_arready) begin
                s_arvalid_r <= 0;
            end
            s_axil_arready <= !s_arvalid_r || s_arvalid_sync[1] && m_axil_arready;

            // B Channel
            if (m_bvalid_sync[1] && s_axil_bready) begin
                m_bvalid_r <= 0;
            end
            s_axil_bresp <= m_bresp_r;
            s_axil_bvalid <= m_bvalid_r;
            s_axil_bready <= !m_bvalid_r || m_bvalid_sync[1] && s_axil_bready;

            // R Channel
            if (m_rvalid_sync[1] && s_axil_rready) begin
                m_rvalid_r <= 0;
            end
            s_axil_rdata <= m_rdata_r;
            s_axil_rresp <= m_rresp_r;
            s_axil_rvalid <= m_rvalid_r;
            s_axil_rready <= !m_rvalid_r || m_rvalid_sync[1] && s_axil_rready;
        end
    end

    // Master side
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_axil_awaddr <= 0;
            m_axil_awprot <= 0;
            m_axil_awvalid <= 0;
            m_axil_wdata <= 0;
            m_axil_wstrb <= 0;
            m_axil_wvalid <= 0;
            m_axil_araddr <= 0;
            m_axil_arprot <= 0;
            m_axil_arvalid <= 0;
            m_axil_bready <= 0;
            m_axil_rready <= 0;
        end else begin
            // AW Channel
            if (s_awvalid_sync[1] && !m_axil_awvalid) begin
                m_axil_awaddr <= s_awaddr_r;
                m_axil_awprot <= s_awprot_r;
                m_axil_awvalid <= 1;
            end else if (m_axil_awready) begin
                m_axil_awvalid <= 0;
            end

            // W Channel
            if (s_wvalid_sync[1] && !m_axil_wvalid) begin
                m_axil_wdata <= s_wdata_r;
                m_axil_wstrb <= s_wstrb_r;
                m_axil_wvalid <= 1;
            end else if (m_axil_wready) begin
                m_axil_wvalid <= 0;
            end

            // AR Channel
            if (s_arvalid_sync[1] && !m_axil_arvalid) begin
                m_axil_araddr <= s_araddr_r;
                m_axil_arprot <= s_arprot_r;
                m_axil_arvalid <= 1;
            end else if (m_axil_arready) begin
                m_axil_arvalid <= 0;
            end

            // B Channel
            if (m_axil_bvalid && !m_bvalid_r && !m_bvalid_sync[1]) begin
                m_bresp_r <= m_axil_bresp;
                m_bvalid_r <= 1;
            end else if (m_bvalid_sync[1] && s_axil_bready) begin
                m_bvalid_r <= 0;
            end
            m_axil_bready <= !m_bvalid_r || m_bvalid_sync[1] && s_axil_bready;

            // R Channel
            if (m_axil_rvalid && !m_rvalid_r && !m_rvalid_sync[1]) begin
                m_rdata_r <= m_axil_rdata;
                m_rresp_r <= m_axil_rresp;
                m_rvalid_r <= 1;
            end else if (m_rvalid_sync[1] && s_axil_rready) begin
                m_rvalid_r <= 0;
            end
            m_axil_rready <= !m_rvalid_r || m_rvalid_sync[1] && s_axil_rready;
        end
    end

    // Cross-domain synchronization
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_awvalid_sync <= 0;
            s_wvalid_sync <= 0;
            s_arvalid_sync <= 0;
        end else begin
            s_awvalid_sync <= {s_awvalid_sync[0], s_awvalid_r};
            s_wvalid_sync <= {s_wvalid_sync[0], s_wvalid_r};
            s_arvalid_sync <= {s_arvalid_sync[0], s_arvalid_r};
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_awready_sync <= 0;
            m_wready_sync <= 0;
            m_arready_sync <= 0;
            m_bvalid_sync <= 0;
            m_rvalid_sync <= 0;
        end else begin
            m_awready_sync <= {m_awready_sync[0], m_axil_awready};
            m_wready_sync <= {m_wready_sync[0], m_axil_wready};
            m_arready_sync <= {m_arready_sync[0], m_axil_arready};
            m_bvalid_sync <= {m_bvalid_sync[0], m_axil_bvalid};
            m_rvalid_sync <= {m_rvalid_sync[0], m_axil_rvalid};
        end
    end

endmodule
