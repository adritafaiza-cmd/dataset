module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o,
    input  b_isolate_ack_i
);

    // Internal signals
    reg [SYNC_STAGES-1:0] a_clear_sync;
    reg [SYNC_STAGES-1:0] b_clear_sync;
    reg [SYNC_STAGES-1:0] a_isolate_sync;
    reg [SYNC_STAGES-1:0] b_isolate_sync;

    reg a_isolate_req;
    reg a_clear_req;
    reg a_isolate_done;
    reg a_clear_done;
    reg a_release_req;
    reg a_release_done;

    reg b_isolate_req;
    reg b_clear_req;
    reg b_isolate_done;
    reg b_clear_done;
    reg b_release_req;
    reg b_release_done;

    // Synchronize clear requests from B domain to A domain
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            a_clear_sync <= {a_clear_sync[SYNC_STAGES-2:0], b_clear_i};
        end
    end

    // Synchronize clear requests from A domain to B domain
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            b_clear_sync <= {b_clear_sync[SYNC_STAGES-2:0], a_clear_i};
        end
    end

    // Synchronize isolate acks from B domain to A domain
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_isolate_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            a_isolate_sync <= {a_isolate_sync[SYNC_STAGES-2:0], b_isolate_ack_i};
        end
    end

    // Synchronize isolate acks from A domain to B domain
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_isolate_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            b_isolate_sync <= {b_isolate_sync[SYNC_STAGES-2:0], a_isolate_ack_i};
        end
    end

    // A domain state machine
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_isolate_req <= 1'b0;
            a_clear_req <= 1'b0;
            a_isolate_done <= 1'b0;
            a_clear_done <= 1'b0;
            a_release_req <= 1'b0;
            a_release_done <= 1'b0;
        end else begin
            case ({a_isolate_req, a_isolate_done, a_clear_req, a_clear_done, a_release_req, a_release_done})
                6'b000000: begin // IDLE
                    if (a_clear_sync[SYNC_STAGES-1]) begin
                        a_isolate_req <= 1'b1;
                    end
                end
                6'b100000: begin // ISOLATE_REQUESTED
                    a_isolate_o <= 1'b1;
                    if (a_isolate_sync[SYNC_STAGES-1]) begin
                        a_isolate_req <= 1'b0;
                        a_isolate_done <= 1'b1;
                    end
                end
                6'b010000: begin // ISOLATE_DONE
                    a_clear_req <= 1'b1;
                end
                6'b011000: begin // CLEAR_REQUESTED
                    a_clear_o <= 1'b1;
                    if (a_clear_ack_i) begin
                        a_clear_req <= 1'b0;
                        a_clear_done <= 1'b1;
                    end
                end
                6'b011100: begin // CLEAR_DONE
                    a_release_req <= 1'b1;
                end
                6'b011110: begin // RELEASE_REQUESTED
                    a_isolate_o <= 1'b0;
                    a_clear_o <= 1'b0;
                    if (b_isolate_sync[SYNC_STAGES-1]) begin
                        a_release_req <= 1'b0;
                        a_release_done <= 1'b1;
                    end
                end
                6'b011111: begin // RELEASE_DONE
                    a_release_done <= 1'b0;
                end
                default: begin
                    a_isolate_req <= 1'b0;
                    a_clear_req <= 1'b0;
                    a_isolate_done <= 1'b0;
                    a_clear_done <= 1'b0;
                    a_release_req <= 1'b0;
                    a_release_done <= 1'b0;
                end
            endcase
        end
    end

    // B domain state machine
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_isolate_req <= 1'b0;
            b_clear_req <= 1'b0;
            b_isolate_done <= 1'b0;
            b_clear_done <= 1'b0;
            b_release_req <= 1'b0;
            b_release_done <= 1'b0;
        end else begin
            case ({b_isolate_req, b_isolate_done, b_clear_req, b_clear_done, b_release_req, b_release_done})
                6'b000000: begin // IDLE
                    if (b_clear_sync[SYNC_STAGES-1]) begin
                        b_isolate_req <= 1'b1;
                    end
                end
                6'b100000: begin // ISOLATE_REQUESTED
                    b_isolate_o <= 1'b1;
                    if (b_isolate_sync[SYNC_STAGES-1]) begin
                        b_isolate_req <= 1'b0;
                        b_isolate_done <= 1'b1;
                    end
                end
                6'b010000: begin // ISOLATE_DONE
                    b_clear_req <= 1'b1;
                end
                6'b011000: begin // CLEAR_REQUESTED
                    b_clear_o <= 1'b1;
                    if (b_clear_ack_i) begin
                        b_clear_req <= 1'b0;
                        b_clear_done <= 1'b1;
                    end
                end
                6'b011100: begin // CLEAR_DONE
                    b_release_req <= 1'b1;
                end
                6'b011110: begin // RELEASE_REQUESTED
                    b_isolate_o <= 1'b0;
                    b_clear_o <= 1'b0;
                    if (a_isolate_sync[SYNC_STAGES-1]) begin
                        b_release_req <= 1'b0;
                        b_release_done <= 1'b1;
                    end
                end
                6'b011111: begin // RELEASE_DONE
                    b_release_done <= 1'b0;
                end
                default: begin
                    b_isolate_req <= 1'b0;
                    b_clear_req <= 1'b0;
                    b_isolate_done <= 1'b0;
                    b_clear_done <= 1'b0;
                    b_release_req <= 1'b0;
                    b_release_done <= 1'b0;
                end
            endcase
        end
    end

endmodule
