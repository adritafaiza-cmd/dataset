module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o,
    input  b_isolate_ack_i
);

    // Internal signals
    reg [SYNC_STAGES-1:0] a_clear_sync;
    reg [SYNC_STAGES-1:0] b_clear_sync;
    reg [SYNC_STAGES-1:0] a_isolate_sync;
    reg [SYNC_STAGES-1:0] b_isolate_sync;

    reg a_isolate_req;
    reg b_isolate_req;
    reg a_clear_req;
    reg b_clear_req;

    reg a_isolate;
    reg a_clear;
    reg a_released;

    reg b_isolate;
    reg b_clear;
    reg b_released;

    // Synchronization flops for a_clear_i
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            a_clear_sync <= {a_clear_sync[SYNC_STAGES-2:0], a_clear_i};
        end
    end

    // Synchronization flops for b_clear_i
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            b_clear_sync <= {b_clear_sync[SYNC_STAGES-2:0], b_clear_i};
        end
    end

    // Synchronization flops for a_isolate_o
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            a_isolate_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            a_isolate_sync <= {a_isolate_sync[SYNC_STAGES-2:0], a_isolate_o};
        end
    end

    // Synchronization flops for b_isolate_o
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            b_isolate_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            b_isolate_sync <= {b_isolate_sync[SYNC_STAGES-2:0], b_isolate_o};
        end
    end

    // FSM for A side
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_isolate_req <= 1'b0;
            a_clear_req <= 1'b0;
            a_isolate <= 1'b0;
            a_clear <= 1'b0;
            a_released <= 1'b0;
        end else begin
            case ({a_isolate_req, a_clear_req, a_isolate, a_clear, a_released, a_isolate_ack_i, a_clear_ack_i, b_isolate_sync[SYNC_STAGES-1]})
                8'b00000000: begin // IDLE
                    if (a_clear_sync[SYNC_STAGES-1]) begin
                        a_isolate_req <= 1'b1;
                    end
                end
                8'b10000000: begin // ISOLATE_REQUESTED
                    a_isolate_o <= 1'b1;
                    if (a_isolate_ack_i) begin
                        a_isolate_req <= 1'b0;
                        a_clear_req <= 1'b1;
                    end
                end
                8'b01000000: begin // CLEAR_REQUESTED
                    a_clear_o <= 1'b1;
                    if (a_clear_ack_i) begin
                        a_clear_req <= 1'b0;
                    end
                end
                8'b00110000: begin // CLEAR_ACTIVE
                    if (b_isolate_sync[SYNC_STAGES-1]) begin
                        a_clear_o <= 1'b0;
                        a_released <= 1'b1;
                    end
                end
                8'b00101000: begin // RELEASE_REQUESTED
                    a_isolate_o <= 1'b0;
                    if (!a_isolate_o) begin
                        a_released <= 1'b0;
                    end
                end
                default: begin
                    // Stay in current state
                end
            endcase
        end
    end

    // FSM for B side
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_isolate_req <= 1'b0;
            b_clear_req <= 1'b0;
            b_isolate <= 1'b0;
            b_clear <= 1'b0;
            b_released <= 1'b0;
        end else begin
            case ({b_isolate_req, b_clear_req, b_isolate, b_clear, b_released, b_isolate_ack_i, b_clear_ack_i, a_isolate_sync[SYNC_STAGES-1]})
                8'b00000000: begin // IDLE
                    if (b_clear_sync[SYNC_STAGES-1]) begin
                        b_isolate_req <= 1'b1;
                    end
                end
                8'b10000000: begin // ISOLATE_REQUESTED
                    b_isolate_o <= 1'b1;
                    if (b_isolate_ack_i) begin
                        b_isolate_req <= 1'b0;
                        b_clear_req <= 1'b1;
                    end
                end
                8'b01000000: begin // CLEAR_REQUESTED
                    b_clear_o <= 1'b1;
                    if (b_clear_ack_i) begin
                        b_clear_req <= 1'b0;
                    end
                end
                8'b00110000: begin // CLEAR_ACTIVE
                    if (a_isolate_sync[SYNC_STAGES-1]) begin
                        b_clear_o <= 1'b0;
                        b_released <= 1'b1;
                    end
                end
                8'b00101000: begin // RELEASE_REQUESTED
                    b_isolate_o <= 1'b0;
                    if (!b_isolate_o) begin
                        b_released <= 1'b0;
                    end
                end
                default: begin
                    // Stay in current state
                end
            endcase
        end
    end

endmodule
