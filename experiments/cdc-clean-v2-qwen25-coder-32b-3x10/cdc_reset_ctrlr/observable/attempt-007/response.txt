module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o,
    input  b_isolate_ack_i
);

    // Internal signals
    reg [SYNC_STAGES-1:0] a_clear_sync;
    reg [SYNC_STAGES-1:0] b_clear_sync;
    reg [SYNC_STAGES-1:0] a_isolate_sync;
    reg [SYNC_STAGES-1:0] b_isolate_sync;

    reg a_isolate_req;
    reg a_clear_req;
    reg a_clear_done;
    reg a_isolate_done;
    reg a_release_req;
    reg a_release_done;

    reg b_isolate_req;
    reg b_clear_req;
    reg b_clear_done;
    reg b_isolate_done;
    reg b_release_req;
    reg b_release_done;

    // Synchronization flops for a_clear_i
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            a_clear_sync <= {a_clear_sync[SYNC_STAGES-2:0], a_clear_i};
        end
    end

    // Synchronization flops for b_clear_i
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            b_clear_sync <= {b_clear_sync[SYNC_STAGES-2:0], b_clear_i};
        end
    end

    // Synchronization flops for a_isolate_o
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            a_isolate_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            a_isolate_sync <= {a_isolate_sync[SYNC_STAGES-2:0], a_isolate_o};
        end
    end

    // Synchronization flops for b_isolate_o
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            b_isolate_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            b_isolate_sync <= {b_isolate_sync[SYNC_STAGES-2:0], b_isolate_o};
        end
    end

    // State machine for A side
    always @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni) begin
            a_isolate_req <= 1'b0;
            a_clear_req <= 1'b0;
            a_clear_done <= 1'b0;
            a_isolate_done <= 1'b0;
            a_release_req <= 1'b0;
            a_release_done <= 1'b0;
        end else begin
            case ({a_isolate_req, a_clear_req, a_clear_done, a_isolate_done, a_release_req, a_release_done})
                6'b000000: begin // Idle
                    if (a_clear_sync[SYNC_STAGES-1]) begin
                        a_isolate_req <= 1'b1;
                    end
                end
                6'b100000: begin // Isolate requested
                    if (a_isolate_ack_i) begin
                        a_isolate_req <= 1'b0;
                        a_clear_req <= 1'b1;
                    end
                end
                6'b010000: begin // Clear requested
                    if (b_isolate_sync[SYNC_STAGES-1]) begin
                        a_clear_req <= 1'b0;
                        a_clear_done <= 1'b1;
                    end
                end
                6'b001000: begin // Clear done
                    if (a_clear_ack_i) begin
                        a_clear_done <= 1'b0;
                        a_isolate_done <= 1'b1;
                    end
                end
                6'b000100: begin // Isolate done
                    if (b_clear_sync[SYNC_STAGES-1]) begin
                        a_isolate_done <= 1'b0;
                        a_release_req <= 1'b1;
                    end
                end
                6'b000010: begin // Release requested
                    if (b_isolate_ack_i) begin
                        a_release_req <= 1'b0;
                        a_release_done <= 1'b1;
                    end
                end
                6'b000001: begin // Release done
                    a_release_done <= 1'b0;
                end
                default: begin
                    a_isolate_req <= 1'b0;
                    a_clear_req <= 1'b0;
                    a_clear_done <= 1'b0;
                    a_isolate_done <= 1'b0;
                    a_release_req <= 1'b0;
                    a_release_done <= 1'b0;
                end
            endcase
        end
    end

    // State machine for B side
    always @(posedge b_clk_i or negedge b_rst_ni) begin
        if (!b_rst_ni) begin
            b_isolate_req <= 1'b0;
            b_clear_req <= 1'b0;
            b_clear_done <= 1'b0;
            b_isolate_done <= 1'b0;
            b_release_req <= 1'b0;
            b_release_done <= 1'b0;
        end else begin
            case ({b_isolate_req, b_clear_req, b_clear_done, b_isolate_done, b_release_req, b_release_done})
                6'b000000: begin // Idle
                    if (b_clear_sync[SYNC_STAGES-1]) begin
                        b_isolate_req <= 1'b1;
                    end
                end
                6'b100000: begin // Isolate requested
                    if (b_isolate_ack_i) begin
                        b_isolate_req <= 1'b0;
                        b_clear_req <= 1'b1;
                    end
                end
                6'b010000: begin // Clear requested
                    if (a_isolate_sync[SYNC_STAGES-1]) begin
                        b_clear_req <= 1'b0;
                        b_clear_done <= 1'b1;
                    end
                end
                6'b001000: begin // Clear done
                    if (b_clear_ack_i) begin
                        b_clear_done <= 1'b0;
                        b_isolate_done <= 1'b1;
                    end
                end
                6'b000100: begin // Isolate done
                    if (a_clear_sync[SYNC_STAGES-1]) begin
                        b_isolate_done <= 1'b0;
                        b_release_req <= 1'b1;
                    end
                end
                6'b000010: begin // Release requested
                    if (a_isolate_ack_i) begin
                        b_release_req <= 1'b0;
                        b_release_done <= 1'b1;
                    end
                end
                6'b000001: begin // Release done
                    b_release_done <= 1'b0;
                end
                default: begin
                    b_isolate_req <= 1'b0;
                    b_clear_req <= 1'b0;
                    b_clear_done <= 1'b0;
                    b_isolate_done <= 1'b0;
                    b_release_req <= 1'b0;
                    b_release_done <= 1'b0;
                end
            endcase
        end
    end

    // Output assignments for A side
    assign a_isolate_o = a_isolate_req;
    assign a_clear_o = a_clear_req & ~a_clear_done;

    // Output assignments for B side
    assign b_isolate_o = b_isolate_req;
    assign b_clear_o = b_clear_req & ~b_clear_done;

endmodule
