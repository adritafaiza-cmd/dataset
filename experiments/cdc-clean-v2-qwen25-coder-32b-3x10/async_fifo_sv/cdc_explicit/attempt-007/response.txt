module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 2**ADDR_WIDTH;
    localparam ADDR_WIDTH_PLUS1 = ADDR_WIDTH + 1;

    // Write and read pointers
    reg [ADDR_WIDTH_PLUS1:0] wptr, rptr;
    reg [ADDR_WIDTH_PLUS1:0] wptr_gray, rptr_gray;
    reg [ADDR_WIDTH_PLUS1:0] wptr_gray_sync [SYNC_STAGES-1:0];
    reg [ADDR_WIDTH_PLUS1:0] rptr_gray_sync [SYNC_STAGES-1:0];

    // Memory array
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];

    // Binary to Gray conversion
    function [ADDR_WIDTH_PLUS1:0] bin_to_gray;
        input [ADDR_WIDTH_PLUS1:0] bin;
        bin_to_gray = bin ^ (bin >> 1);
    endfunction

    // Gray to Binary conversion
    function [ADDR_WIDTH_PLUS1:0] gray_to_bin;
        input [ADDR_WIDTH_PLUS1:0] gray;
        integer i;
        reg [ADDR_WIDTH_PLUS1:0] bin;
        bin = gray;
        for (i = ADDR_WIDTH_PLUS1-1; i > 0; i = i - 1) begin
            bin[i-1] = bin[i] ^ bin[i-1];
        end
        gray_to_bin = bin;
    endfunction

    // Write domain logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= 0;
            wptr_gray <= 0;
        end else if (winc && !wfull) begin
            wptr <= wptr + 1;
            wptr_gray <= bin_to_gray(wptr + 1);
        end
    end

    // Read domain logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= 0;
            rptr_gray <= 0;
        end else if (rinc && !rempty) begin
            rptr <= rptr + 1;
            rptr_gray <= bin_to_gray(rptr + 1);
        end
    end

    // Synchronize write pointer to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_sync <= '{default: 0};
        end else begin
            wptr_gray_sync[0] <= wptr_gray;
            for (int i = 1; i < SYNC_STAGES; i = i + 1) begin
                wptr_gray_sync[i] <= wptr_gray_sync[i-1];
            end
        end
    end

    // Synchronize read pointer to write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_sync <= '{default: 0};
        end else begin
            rptr_gray_sync[0] <= rptr_gray;
            for (int i = 1; i < SYNC_STAGES; i = i + 1) begin
                rptr_gray_sync[i] <= rptr_gray_sync[i-1];
            end
        end
    end

    // Calculate full and empty flags
    wire [ADDR_WIDTH_PLUS1:0] wptr_sync, rptr_sync;
    assign wptr_sync = gray_to_bin(wptr_gray_sync[SYNC_STAGES-1]);
    assign rptr_sync = gray_to_bin(rptr_gray_sync[SYNC_STAGES-1]);

    assign wfull = (wptr_sync[ADDR_WIDTH:0] == {rptr_sync[ADDR_WIDTH], ~rptr_sync[ADDR_WIDTH-1:0]});
    assign rempty = (wptr_gray_sync[SYNC_STAGES-1] == rptr_gray);

    // Write data to memory
    always @(posedge wclk) begin
        if (winc && !wfull) begin
            mem[wptr[ADDR_WIDTH-1:0]] <= wdata;
        end
    end

    // Read data from memory
    always @(posedge rclk) begin
        if (rinc && !rempty) begin
            rdata <= mem[rptr[ADDR_WIDTH-1:0]];
        end
    end

    // Debug outputs
    assign waddr = wptr;
    assign raddr = rptr;

endmodule
