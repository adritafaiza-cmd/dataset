module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 2**ADDR_WIDTH;
    localparam ADDR_WIDTH_PLUS1 = ADDR_WIDTH + 1;

    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];

    reg [ADDR_WIDTH_PLUS1:0] wptr_gray, wptr_gray_next, wptr_gray_reg;
    reg [ADDR_WIDTH_PLUS1:0] rptr_gray, rptr_gray_next, rptr_gray_reg;

    reg [ADDR_WIDTH_PLUS1:0] wptr_bin, wptr_bin_next;
    reg [ADDR_WIDTH_PLUS1:0] rptr_bin, rptr_bin_next;

    reg [ADDR_WIDTH_PLUS1:0] wptr_gray_sync [0:SYNC_STAGES-1];
    reg [ADDR_WIDTH_PLUS1:0] rptr_gray_sync [0:SYNC_STAGES-1];

    // Write domain logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr_gray <= 0;
            wptr_gray_reg <= 0;
        end else begin
            wptr_gray <= wptr_gray_next;
            wptr_gray_reg <= wptr_gray_next;
        end
    end

    always @(*) begin
        wptr_bin_next = wptr_gray ^ (wptr_gray >> 1);
        if (winc && !wfull)
            wptr_gray_next = wptr_gray + 1;
        else
            wptr_gray_next = wptr_gray;
    end

    assign waddr = wptr_bin_next[ADDR_WIDTH:0];

    // Read domain logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr_gray <= 0;
            rptr_gray_reg <= 0;
        end else begin
            rptr_gray <= rptr_gray_next;
            rptr_gray_reg <= rptr_gray_next;
        end
    end

    always @(*) begin
        rptr_bin_next = rptr_gray ^ (rptr_gray >> 1);
        if (rinc && !rempty)
            rptr_gray_next = rptr_gray + 1;
        else
            rptr_gray_next = rptr_gray;
    end

    assign raddr = rptr_bin_next[ADDR_WIDTH:0];

    // Memory write
    always @(posedge wclk) begin
        if (winc && !wfull)
            mem[wptr_bin_next[ADDR_WIDTH-1:0]] <= wdata;
    end

    // Memory read
    always @(posedge rclk) begin
        if (rinc && !rempty)
            rdata <= mem[rptr_bin_next[ADDR_WIDTH-1:0]];
    end

    // Gray code synchronization
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            for (int i = 0; i < SYNC_STAGES; i = i + 1)
                wptr_gray_sync[i] <= 0;
        end else begin
            wptr_gray_sync[0] <= rptr_gray_reg;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                wptr_gray_sync[i] <= wptr_gray_sync[i-1];
        end
    end

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            for (int i = 0; i < SYNC_STAGES; i = i + 1)
                rptr_gray_sync[i] <= 0;
        end else begin
            rptr_gray_sync[0] <= wptr_gray_reg;
            for (int i = 1; i < SYNC_STAGES; i = i + 1)
                rptr_gray_sync[i] <= rptr_gray_sync[i-1];
        end
    end

    // Full and empty flags
    assign wptr_bin = wptr_gray ^ (wptr_gray >> 1);
    assign rptr_bin = rptr_gray_sync[SYNC_STAGES-1] ^ (rptr_gray_sync[SYNC_STAGES-1] >> 1);

    assign wfull = (wptr_bin[ADDR_WIDTH_PLUS1:0] == {~rptr_bin[ADDR_WIDTH_PLUS1], rptr_bin[ADDR_WIDTH:0]});
    assign rempty = (wptr_bin[ADDR_WIDTH_PLUS1:0] == rptr_bin[ADDR_WIDTH_PLUS1:0]);

endmodule
