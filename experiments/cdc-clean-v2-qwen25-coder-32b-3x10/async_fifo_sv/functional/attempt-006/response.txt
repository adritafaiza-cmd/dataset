module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 2**ADDR_WIDTH;
    localparam ADDR_MSB = ADDR_WIDTH;

    // Write and read pointers
    reg [ADDR_MSB:0] wptr, rptr;
    reg [ADDR_MSB:0] wptr_gray, rptr_gray;
    reg [ADDR_MSB:0] rptr_gray_sync [SYNC_STAGES-1:0];

    // Memory array
    reg [DATA_WIDTH-1:0] mem [DEPTH-1:0];

    // Gray code conversion functions
    function [ADDR_MSB:0] bin_to_gray;
        input [ADDR_MSB:0] bin;
        bin_to_gray = (bin >> 1) ^ bin;
    endfunction

    function [ADDR_MSB:0] gray_to_bin;
        input [ADDR_MSB:0] gray;
        integer i;
        reg [ADDR_MSB:0] bin;
        bin = gray;
        for (i = ADDR_MSB-1; i >= 0; i = i - 1) begin
            bin[i] = bin[i+1] ^ gray[i];
        end
        gray_to_bin = bin;
    endfunction

    // Write domain logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= 0;
            wptr_gray <= 0;
        end else if (winc && !wfull) begin
            wptr <= wptr + 1;
            wptr_gray <= bin_to_gray(wptr + 1);
        end
    end

    // Read domain logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= 0;
            rptr_gray_sync <= '{default: 0};
        end else begin
            // Synchronize write pointer gray code to read domain
            rptr_gray_sync[0] <= wptr_gray;
            for (int i = 1; i < SYNC_STAGES; i = i + 1) begin
                rptr_gray_sync[i] <= rptr_gray_sync[i-1];
            end

            // Update read pointer
            if (rinc && !rempty) begin
                rptr <= rptr + 1;
            end
        end
    end

    // Calculate full and empty flags
    wire [ADDR_MSB:0] wptr_gray_sync = rptr_gray_sync[SYNC_STAGES-1];
    wire [ADDR_MSB:0] wptr_bin_sync = gray_to_bin(wptr_gray_sync);
    wire [ADDR_MSB:0] rptr_bin_sync = gray_to_bin(rptr_gray);

    assign wfull = (wptr_bin_sync[ADDR_MSB:ADDR_MSB-ADDR_WIDTH] == rptr_bin_sync[ADDR_MSB:ADDR_MSB-ADDR_WIDTH]) && (wptr_bin_sync[ADDR_MSB-ADDR_WIDTH-1:0] == rptr_bin_sync[ADDR_MSB-ADDR_WIDTH-1:0] + 1);
    assign rempty = (wptr_bin_sync == rptr_bin_sync);

    // Write data to memory
    always @(posedge wclk) begin
        if (winc && !wfull) begin
            mem[wptr[ADDR_MSB-1:0]] <= wdata;
        end
    end

    // Read data from memory
    assign rdata = mem[rptr[ADDR_MSB-1:0]];

    // Debug outputs
    assign waddr = wptr;
    assign raddr = rptr;

endmodule
