module afifo #(
		parameter	LGFIFO = 3,
		parameter	WIDTH  = 16,
		parameter	NFF    = 2,
		parameter [0:0]	WRITE_ON_POSEDGE = 1'b1,
		parameter [0:0]	OPT_REGISTER_READS = 1'b1
	) (
		input	wire			i_wclk, i_wr_reset_n, i_wr,
		input	wire	[WIDTH-1:0]	i_wr_data,
		output	reg			o_wr_full,
		input	wire			i_rclk, i_rd_reset_n, i_rd,
		output	reg	[WIDTH-1:0]	o_rd_data,
		output	reg			o_rd_empty
	);

	localparam DEPTH = 2**LGFIFO;
	localparam ADDR_WIDTH = LGFIFO;

	reg [WIDTH-1:0] mem [0:DEPTH-1];
	reg [ADDR_WIDTH:0] wr_ptr, rd_ptr;
	reg [ADDR_WIDTH:0] wr_ptr_gray, rd_ptr_gray;
	reg [ADDR_WIDTH:0] wr_ptr_gray_sync, rd_ptr_gray_sync;

	// Write side logic
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			wr_ptr <= 0;
			o_wr_full <= 1'b0;
		end else begin
			if (i_wr && !o_wr_full) begin
				mem[wr_ptr[ADDR_WIDTH-1:0]] <= i_wr_data;
				wr_ptr <= wr_ptr + 1;
			end
			o_wr_full <= (wr_ptr_gray_sync[ADDR_WIDTH] != rd_ptr_gray[ADDR_WIDTH]) && (wr_ptr_gray_sync[ADDR_WIDTH-1:0] == rd_ptr_gray[ADDR_WIDTH-1:0]);
		end
	end

	// Convert wr_ptr to gray code
	always @(wr_ptr) begin
		wr_ptr_gray = (wr_ptr >> 1) ^ wr_ptr;
	end

	// Synchronize wr_ptr_gray to read domain
	generate
		genvar i;
		for (i = 0; i < NFF; i = i + 1) begin : sync_wr_ptr_gray
			reg [ADDR_WIDTH:0] ff;
			if (i == 0) begin
				always @(posedge i_rclk or negedge i_rd_reset_n) begin
					if (!i_rd_reset_n) begin
						ff <= 0;
					end else begin
						ff <= wr_ptr_gray;
					end
				end
			end else begin
				always @(posedge i_rclk or negedge i_rd_reset_n) begin
					if (!i_rd_reset_n) begin
						ff <= 0;
					end else begin
						ff <= sync_wr_ptr_gray[i-1].ff;
					end
				end
			end
			assign wr_ptr_gray_sync = sync_wr_ptr_gray[NFF-1].ff;
		end
	endgenerate

	// Read side logic
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			rd_ptr <= 0;
			o_rd_empty <= 1'b1;
		end else begin
			if (i_rd && !o_rd_empty) begin
				rd_ptr <= rd_ptr + 1;
			end
			o_rd_empty <= (wr_ptr_gray[ADDR_WIDTH] == rd_ptr_gray_sync[ADDR_WIDTH]) && (wr_ptr_gray[ADDR_WIDTH-1:0] == rd_ptr_gray_sync[ADDR_WIDTH-1:0]);
		end
	end

	// Convert rd_ptr to gray code
	always @(rd_ptr) begin
		rd_ptr_gray = (rd_ptr >> 1) ^ rd_ptr;
	end

	// Synchronize rd_ptr_gray to write domain
	generate
		genvar j;
		for (j = 0; j < NFF; j = j + 1) begin : sync_rd_ptr_gray
			reg [ADDR_WIDTH:0] ff;
			if (j == 0) begin
				always @(posedge i_wclk or negedge i_wr_reset_n) begin
					if (!i_wr_reset_n) begin
						ff <= 0;
					end else begin
						ff <= rd_ptr_gray;
					end
				end
			end else begin
				always @(posedge i_wclk or negedge i_wr_reset_n) begin
					if (!i_wr_reset_n) begin
						ff <= 0;
					end else begin
						ff <= sync_rd_ptr_gray[j-1].ff;
					end
				end
			end
			assign rd_ptr_gray_sync = sync_rd_ptr_gray[NFF-1].ff;
		end
	endgenerate

	// Read data output
	reg [WIDTH-1:0] rd_data_reg;

	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			rd_data_reg <= 0;
		end else if (i_rd && !o_rd_empty) begin
			rd_data_reg <= mem[rd_ptr[ADDR_WIDTH-1:0]];
		end
	end

	// Output selection based on OPT_REGISTER_READS
	assign o_rd_data = OPT_REGISTER_READS ? rd_data_reg : mem[rd_ptr[ADDR_WIDTH-1:0]];

endmodule
