module afifo #(
		parameter	LGFIFO = 3,
		parameter	WIDTH  = 16,
		parameter	NFF    = 2,
		parameter [0:0]	WRITE_ON_POSEDGE = 1'b1,
		parameter [0:0]	OPT_REGISTER_READS = 1'b1
	) (
		input	wire			i_wclk, i_wr_reset_n, i_wr,
		input	wire	[WIDTH-1:0]	i_wr_data,
		output	reg			o_wr_full,
		input	wire			i_rclk, i_rd_reset_n, i_rd,
		output	reg	[WIDTH-1:0]	o_rd_data,
		output	reg			o_rd_empty
	);

	localparam DEPTH = 2**LGFIFO;
	localparam ADDR_WIDTH = LGFIFO;

	reg [WIDTH-1:0] mem [0:DEPTH-1];
	reg [ADDR_WIDTH:0] wr_ptr, rd_ptr;
	reg [ADDR_WIDTH:0] wr_ptr_gray, rd_ptr_gray;
	reg [ADDR_WIDTH:0] wr_ptr_gray_sync, rd_ptr_gray_sync;

	// Write Pointer Logic
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			wr_ptr <= 0;
			o_wr_full <= 1'b0;
		end else if (i_wr && !o_wr_full) begin
			mem[wr_ptr[ADDR_WIDTH-1:0]] <= i_wr_data;
			wr_ptr <= wr_ptr + 1;
		end
	end

	// Read Pointer Logic
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			rd_ptr <= 0;
			o_rd_empty <= 1'b1;
		end else if (i_rd && !o_rd_empty) begin
			rd_ptr <= rd_ptr + 1;
		end
	end

	// Gray Code Conversion
	assign wr_ptr_gray = (wr_ptr >> 1) ^ wr_ptr;
	assign rd_ptr_gray = (rd_ptr >> 1) ^ rd_ptr;

	// Gray Code Synchronization
	sync_gray_code sync_wr (
		.clk(i_rclk),
		.reset_n(i_rd_reset_n),
		.async_in(wr_ptr_gray),
		.sync_out(wr_ptr_gray_sync)
	);

	sync_gray_code sync_rd (
		.clk(i_wclk),
		.reset_n(i_wr_reset_n),
		.async_in(rd_ptr_gray),
		.sync_out(rd_ptr_gray_sync)
	);

	// Full and Empty Flags
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			o_wr_full <= 1'b0;
		end else begin
			o_wr_full <= (wr_ptr_gray_sync[ADDR_WIDTH] != wr_ptr_gray[ADDR_WIDTH]) &&
					(wr_ptr_gray_sync[ADDR_WIDTH-1:0] == wr_ptr[ADDR_WIDTH-1:0]);
		end
	end

	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			o_rd_empty <= 1'b1;
		end else begin
			o_rd_empty <= (rd_ptr_gray_sync[ADDR_WIDTH] != rd_ptr_gray[ADDR_WIDTH]) &&
					(rd_ptr_gray_sync[ADDR_WIDTH-1:0] == rd_ptr[ADDR_WIDTH-1:0]);
		end
	end

	// Read Data Output
	reg [WIDTH-1:0] rd_data_reg;

	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			rd_data_reg <= 0;
		end else if (i_rd && !o_rd_empty) begin
			rd_data_reg <= mem[rd_ptr[ADDR_WIDTH-1:0]];
		end
	end

	assign o_rd_data = OPT_REGISTER_READS ? rd_data_reg : mem[rd_ptr[ADDR_WIDTH-1:0]];

endmodule

module sync_gray_code (
		input	wire			clk,
		input	wire			reset_n,
		input	wire [ADDR_WIDTH:0]	async_in,
		output	reg [ADDR_WIDTH:0]	sync_out
	);

	reg [ADDR_WIDTH:0] sync_reg1, sync_reg2;

	always @(posedge clk or negedge reset_n) begin
		if (!reset_n) begin
			sync_reg1 <= 0;
			sync_reg2 <= 0;
		end else begin
			sync_reg1 <= async_in;
			sync_reg2 <= sync_reg1;
		end
	end

	assign sync_out = sync_reg2;

endmodule
