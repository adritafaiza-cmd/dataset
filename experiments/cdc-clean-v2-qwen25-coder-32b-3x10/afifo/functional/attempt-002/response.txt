module afifo #(
		parameter	LGFIFO = 3,
		parameter	WIDTH  = 16,
		parameter	NFF    = 2,
		parameter [0:0]	WRITE_ON_POSEDGE = 1'b1,
		parameter [0:0]	OPT_REGISTER_READS = 1'b1
	) (
		input	wire			i_wclk, i_wr_reset_n, i_wr,
		input	wire	[WIDTH-1:0]	i_wr_data,
		output	reg			o_wr_full,
		input	wire			i_rclk, i_rd_reset_n, i_rd,
		output	reg	[WIDTH-1:0]	o_rd_data,
		output	reg			o_rd_empty
	);

	localparam DEPTH = 1 << LGFIFO;
	localparam ADDR_WIDTH = LGFIFO;

	reg [WIDTH-1:0] mem [0:DEPTH-1];
	reg [ADDR_WIDTH:0] wr_ptr, rd_ptr;
	reg [ADDR_WIDTH:0] wr_ptr_gray, rd_ptr_gray;
	reg [ADDR_WIDTH:0] wr_ptr_gray_sync, rd_ptr_gray_sync;

	// Write Pointer Logic
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			wr_ptr <= 0;
			o_wr_full <= 0;
		end else begin
			if (i_wr && !o_wr_full) begin
				if (WRITE_ON_POSEDGE) begin
					mem[wr_ptr[ADDR_WIDTH-1:0]] <= i_wr_data;
					wr_ptr <= wr_ptr + 1;
				end else begin
					mem[wr_ptr[ADDR_WIDTH-1:0]] <= i_wr_data;
				end
			end
			if (!WRITE_ON_POSEDGE) begin
				wr_ptr <= wr_ptr + 1;
			end
			wr_ptr_gray <= (wr_ptr >> 1) ^ wr_ptr;
		end
	end

	// Read Pointer Logic
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			rd_ptr <= 0;
			o_rd_empty <= 1;
		end else begin
			if (i_rd && !o_rd_empty) begin
				rd_ptr <= rd_ptr + 1;
			end
			rd_ptr_gray <= (rd_ptr >> 1) ^ rd_ptr;
		end
	end

	// Gray Code Synchronization
	generate
		genvar i;
		for (i = 0; i < NFF; i = i + 1) begin : sync_wr_ptr_gray
			reg [ADDR_WIDTH:0] wr_ptr_gray_sync_stage;
			always @(posedge i_rclk or negedge i_rd_reset_n) begin
				if (!i_rd_reset_n) begin
					wr_ptr_gray_sync_stage <= 0;
				end else begin
					wr_ptr_gray_sync_stage <= (i == 0) ? wr_ptr_gray : wr_ptr_gray_sync_stage;
				end
			end
			assign wr_ptr_gray_sync = (i == NFF - 1) ? wr_ptr_gray_sync_stage : wr_ptr_gray_sync;
		end
	endgenerate

	generate
		genvar j;
		for (j = 0; j < NFF; j = j + 1) begin : sync_rd_ptr_gray
			reg [ADDR_WIDTH:0] rd_ptr_gray_sync_stage;
			always @(posedge i_wclk or negedge i_wr_reset_n) begin
				if (!i_wr_reset_n) begin
					rd_ptr_gray_sync_stage <= 0;
				end else begin
					rd_ptr_gray_sync_stage <= (j == 0) ? rd_ptr_gray : rd_ptr_gray_sync_stage;
				end
			end
			assign rd_ptr_gray_sync = (j == NFF - 1) ? rd_ptr_gray_sync_stage : rd_ptr_gray_sync;
		end
	endgenerate

	// Full and Empty Flags
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			o_wr_full <= 0;
		end else begin
			o_wr_full <= (wr_ptr_gray_sync[ADDR_WIDTH] != wr_ptr_gray_sync[ADDR_WIDTH-1]) && (wr_ptr_gray_sync[ADDR_WIDTH-1:0] == rd_ptr_gray[ADDR_WIDTH-1:0]);
		end
	end

	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			o_rd_empty <= 1;
		end else begin
			o_rd_empty <= (wr_ptr_gray[ADDR_WIDTH] == wr_ptr_gray[ADDR_WIDTH-1]) && (wr_ptr_gray[ADDR_WIDTH-1:0] == rd_ptr_gray_sync[ADDR_WIDTH-1:0]);
		end
	end

	// Read Data Output
	reg [WIDTH-1:0] rd_data_reg;
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			rd_data_reg <= 0;
			o_rd_data <= 0;
		end else begin
			if (i_rd && !o_rd_empty) begin
				rd_data_reg <= mem[rd_ptr[ADDR_WIDTH-1:0]];
			end
			if (OPT_REGISTER_READS) begin
				o_rd_data <= rd_data_reg;
			end else begin
				o_rd_data <= mem[rd_ptr[ADDR_WIDTH-1:0]];
			end
		end
	end

endmodule
