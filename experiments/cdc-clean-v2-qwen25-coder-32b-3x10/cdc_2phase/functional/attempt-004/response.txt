module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain signals
  logic src_ready;
  logic [WIDTH-1:0] src_data_ff;
  logic src_valid_ff;

  // Destination domain signals
  logic dst_valid;
  logic [WIDTH-1:0] dst_data_ff;

  // Synchronization flops for crossing from source to destination domain
  logic [WIDTH-1:0] sync_data_ff [2:0];
  logic sync_valid_ff [2:0];

  // Source domain logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ready <= 1'b0;
      src_data_ff <= '0;
      src_valid_ff <= 1'b0;
    end else begin
      src_ready <= ~src_valid_ff | (src_valid_ff & dst_valid);
      if (src_ready & src_valid_i) begin
        src_data_ff <= src_data_i;
        src_valid_ff <= 1'b1;
      end else if (dst_valid) begin
        src_valid_ff <= 1'b0;
      end
    end
  end

  // Cross-domain synchronization
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      sync_data_ff[0] <= '0;
      sync_data_ff[1] <= '0;
      sync_data_ff[2] <= '0;
      sync_valid_ff[0] <= 1'b0;
      sync_valid_ff[1] <= 1'b0;
      sync_valid_ff[2] <= 1'b0;
    end else begin
      sync_data_ff[0] <= src_data_ff;
      sync_data_ff[1] <= sync_data_ff[0];
      sync_data_ff[2] <= sync_data_ff[1];
      sync_valid_ff[0] <= src_valid_ff;
      sync_valid_ff[1] <= sync_valid_ff[0];
      sync_valid_ff[2] <= sync_valid_ff[1];
    end
  end

  // Destination domain logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid <= 1'b0;
      dst_data_o <= '0;
      dst_valid_o <= 1'b0;
    end else begin
      if (sync_valid_ff[2] & ~dst_valid) begin
        dst_data_o <= sync_data_ff[2];
        dst_valid <= 1'b1;
      end else if (dst_valid & dst_ready_i) begin
        dst_valid <= 1'b0;
      end
      dst_valid_o <= dst_valid;
    end
  end

  // Assign output signals
  assign src_ready_o = src_ready;

endmodule
