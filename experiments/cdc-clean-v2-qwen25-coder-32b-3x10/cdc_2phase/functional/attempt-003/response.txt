module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain signals
  logic src_ready;
  logic [WIDTH-1:0] src_data_ff;
  logic src_valid_ff;

  // Destination domain signals
  logic dst_valid;
  logic [WIDTH-1:0] dst_data_ff;

  // Synchronization flops for src_valid and src_data
  logic [1:0] src_valid_sync;
  logic [WIDTH-1:0] src_data_sync;

  // Source domain logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ready <= 1'b0;
      src_data_ff <= '0;
      src_valid_ff <= 1'b0;
    end else begin
      // Accept new data if ready and valid
      if (src_ready && src_valid_i) begin
        src_data_ff <= src_data_i;
        src_valid_ff <= 1'b1;
      end else if (dst_ready_i && dst_valid) begin
        // Clear valid flag once data is acknowledged
        src_valid_ff <= 1'b0;
      end
      // Generate ready signal based on whether data is already in transit
      src_ready <= !src_valid_ff;
    end
  end

  // Synchronize src_valid and src_data to destination clock domain
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_valid_sync <= '0;
      src_data_sync <= '0;
    end else begin
      src_valid_sync <= {src_valid_sync[0], src_valid_ff};
      src_data_sync <= src_data_ff;
    end
  end

  // Destination domain logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid <= 1'b0;
      dst_data_o <= '0;
    end else begin
      // Accept new data if not already valid
      if (!dst_valid && src_valid_sync[1]) begin
        dst_data_o <= src_data_sync;
        dst_valid <= 1'b1;
      end else if (dst_valid && dst_ready_i) begin
        // Clear valid flag once data is acknowledged
        dst_valid <= 1'b0;
      end
    end
  end

  // Assign outputs
  assign src_ready_o = src_ready;
  assign dst_valid_o = dst_valid;

endmodule
