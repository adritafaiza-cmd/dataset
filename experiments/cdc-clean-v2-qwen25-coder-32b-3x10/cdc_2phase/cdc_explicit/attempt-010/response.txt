module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain signals
  logic src_ready;
  logic [WIDTH-1:0] src_data_ff;
  logic src_valid_ff;

  // Destination domain signals
  logic [WIDTH-1:0] dst_data_ff0, dst_data_ff1;
  logic dst_valid_ff0, dst_valid_ff1;

  // Source domain logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ready <= 1'b0;
      src_data_ff <= '0;
      src_valid_ff <= 1'b0;
    end else begin
      src_ready <= ~src_valid_ff;
      if (src_valid_i && src_ready) begin
        src_data_ff <= src_data_i;
        src_valid_ff <= 1'b1;
      end else if (dst_valid_ff1 && dst_ready_i) begin
        src_valid_ff <= 1'b0;
      end
    end
  end

  // Destination domain logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_ff0 <= '0;
      dst_data_ff1 <= '0;
      dst_valid_ff0 <= 1'b0;
      dst_valid_ff1 <= 1'b0;
    end else begin
      dst_data_ff0 <= src_data_ff;
      dst_valid_ff0 <= src_valid_ff;
      dst_data_ff1 <= dst_data_ff0;
      dst_valid_ff1 <= dst_valid_ff0;
    end
  end

  // Output assignments
  assign src_ready_o = src_ready;
  assign dst_data_o = dst_data_ff1;
  assign dst_valid_o = dst_valid_ff1;

endmodule
