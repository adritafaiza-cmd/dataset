module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source side signals
  logic [WIDTH-1:0] src_data_sync;
  logic src_valid_sync;
  logic src_ready;

  // Destination side signals
  logic [WIDTH-1:0] dst_data_reg;
  logic dst_valid_reg;

  // Synchronizers for crossing src_clk_i to dst_clk_i
  logic [WIDTH-1:0] src_data_meta;
  logic src_valid_meta;

  // Synchronizers for crossing dst_clk_i to src_clk_i
  logic dst_ready_meta;

  // Source side synchronization
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_meta <= '0;
      src_valid_meta <= '0;
    end else begin
      src_data_meta <= src_data_i;
      src_valid_meta <= src_valid_i;
    end
  end

  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_data_sync <= '0;
      src_valid_sync <= '0;
    end else begin
      src_data_sync <= src_data_meta;
      src_valid_sync <= src_valid_meta;
    end
  end

  // Destination side synchronization
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_ready_meta <= '0;
    end else begin
      dst_ready_meta <= dst_ready_i;
    end
  end

  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ready <= '0;
    end else begin
      src_ready <= dst_ready_meta;
    end
  end

  // Source side ready signal
  assign src_ready_o = src_ready;

  // Destination side logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg <= '0;
      dst_valid_reg <= '0;
    end else if (src_valid_sync && !dst_valid_reg) begin
      dst_data_reg <= src_data_sync;
      dst_valid_reg <= '1;
    end else if (dst_valid_reg && dst_ready_i) begin
      dst_valid_reg <= '0;
    end
  end

  // Destination side outputs
  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

endmodule
