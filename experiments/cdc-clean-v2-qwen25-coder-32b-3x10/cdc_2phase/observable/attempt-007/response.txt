module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Intermediate storage for crossing clock domains
  logic [WIDTH-1:0] src_data_sync;
  logic src_valid_sync;
  logic dst_valid_sync;

  // Synchronization flops for src_valid_i to dst_clk_i
  logic src_valid_meta;
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_valid_meta <= 1'b0;
      src_valid_sync <= 1'b0;
    end else begin
      src_valid_meta <= src_valid_i;
      src_valid_sync <= src_valid_meta;
    end
  end

  // Synchronization flops for dst_ready_i to src_clk_i
  logic dst_ready_meta;
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      dst_ready_meta <= 1'b0;
      dst_valid_sync <= 1'b0;
    end else begin
      dst_ready_meta <= dst_ready_i;
      dst_valid_sync <= dst_ready_meta;
    end
  end

  // Data synchronization
  logic [WIDTH-1:0] src_data_meta;
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      src_data_meta <= '0;
      src_data_sync <= '0;
    end else if (src_valid_i) begin
      src_data_meta <= src_data_i;
      src_data_sync <= src_data_meta;
    end
  end

  // Source side ready signal
  assign src_ready_o = !src_valid_sync;

  // Destination side data and valid signals
  assign dst_data_o = src_data_sync;
  assign dst_valid_o = src_valid_sync & !dst_valid_sync;

endmodule
