module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source side signals
  logic [WIDTH-1:0] src_data_sync;
  logic src_valid_sync;
  logic src_ready;

  // Destination side signals
  logic [WIDTH-1:0] dst_data_reg;
  logic dst_valid_reg;

  // Synchronization flops for src_valid_i
  logic src_valid_meta;
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_valid_meta <= 1'b0;
      src_valid_sync <= 1'b0;
    end else begin
      src_valid_meta <= src_valid_i;
      src_valid_sync <= src_valid_meta;
    end
  end

  // Synchronization flops for src_data_i
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_sync <= '0;
    end else if (src_valid_i && src_ready) begin
      src_data_sync <= src_data_i;
    end
  end

  // Source side ready signal
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_ready <= 1'b0;
    end else begin
      src_ready <= !src_valid_sync || (dst_ready_i && dst_valid_reg);
    end
  end

  assign src_ready_o = src_ready;

  // Synchronization flops for dst_valid_reg
  logic dst_valid_meta;
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid_meta <= 1'b0;
      dst_valid_reg <= 1'b0;
    end else begin
      dst_valid_meta <= src_valid_sync;
      dst_valid_reg <= dst_valid_meta;
    end
  end

  // Synchronization flops for dst_data_sync
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg <= '0;
    end else if (src_valid_sync && !dst_valid_reg) begin
      dst_data_reg <= src_data_sync;
    end
  end

  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

endmodule
