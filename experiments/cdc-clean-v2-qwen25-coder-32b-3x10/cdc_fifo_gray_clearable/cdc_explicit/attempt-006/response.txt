module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    // Source domain signals
    reg [ADDR_WIDTH:0] src_wr_ptr_gray;
    reg [ADDR_WIDTH:0] src_wr_ptr_bin;
    reg [ADDR_WIDTH:0] src_rd_ptr_gray_sync;
    reg [ADDR_WIDTH:0] src_clear_pending;
    reg [WIDTH-1:0]    mem [DEPTH-1:0];

    // Destination domain signals
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray;
    reg [ADDR_WIDTH:0] dst_rd_ptr_bin;
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_sync;
    reg [ADDR_WIDTH:0] dst_clear_pending;
    reg [WIDTH-1:0]    dst_data_reg;
    reg                dst_valid_reg;

    // Binary to Gray conversion
    function [ADDR_WIDTH:0] bin_to_gray;
        input [ADDR_WIDTH:0] bin;
        bin_to_gray = bin ^ (bin >> 1);
    endfunction

    // Gray to Binary conversion
    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = 1; i <= ADDR_WIDTH; i = i + 1) begin
            bin = bin ^ (gray >> i);
        end
        gray_to_bin = bin;
    endfunction

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr_gray <= 0;
            src_wr_ptr_bin <= 0;
            src_clear_pending <= 0;
        end else begin
            if (src_clear_i) begin
                src_wr_ptr_gray <= 0;
                src_wr_ptr_bin <= 0;
                src_clear_pending <= 1;
            end else if (src_ready_o && src_valid_i) begin
                src_wr_ptr_bin <= src_wr_ptr_bin + 1;
                src_wr_ptr_gray <= bin_to_gray(src_wr_ptr_bin + 1);
                mem[src_wr_ptr_bin % DEPTH] <= src_data_i;
            end
        end
    end

    // Synchronize read pointer to source domain
    reg [ADDR_WIDTH:0] src_rd_ptr_gray_sync_reg [SYNC_STAGES-1:0];
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rd_ptr_gray_sync_reg <= {SYNC_STAGES{0}};
        end else begin
            src_rd_ptr_gray_sync_reg[0] <= dst_rd_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                src_rd_ptr_gray_sync_reg[i] <= src_rd_ptr_gray_sync_reg[i-1];
            end
        end
    end
    assign src_rd_ptr_gray_sync = src_rd_ptr_gray_sync_reg[SYNC_STAGES-1];

    // Source ready logic
    assign src_ready_o = (src_wr_ptr_bin + 1 != gray_to_bin(src_rd_ptr_gray_sync));

    // Source clear pending logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_pending <= 0;
        end else if (src_clear_i) begin
            src_clear_pending <= 1;
        end else if (src_wr_ptr_bin == gray_to_bin(src_rd_ptr_gray_sync)) begin
            src_clear_pending <= 0;
        end
    end
    assign src_clear_pending_o = src_clear_pending;

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr_gray <= 0;
            dst_rd_ptr_bin <= 0;
            dst_clear_pending <= 0;
            dst_valid_reg <= 0;
        end else begin
            if (dst_clear_i) begin
                dst_rd_ptr_gray <= 0;
                dst_rd_ptr_bin <= 0;
                dst_clear_pending <= 1;
                dst_valid_reg <= 0;
            end else if (dst_valid_reg && dst_ready_i) begin
                dst_rd_ptr_bin <= dst_rd_ptr_bin + 1;
                dst_rd_ptr_gray <= bin_to_gray(dst_rd_ptr_bin + 1);
                dst_valid_reg <= 0;
            end else if (!dst_valid_reg && (dst_wr_ptr_bin != dst_rd_ptr_bin)) begin
                dst_data_reg <= mem[dst_rd_ptr_bin % DEPTH];
                dst_valid_reg <= 1;
            end
        end
    end

    // Synchronize write pointer to destination domain
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_sync_reg [SYNC_STAGES-1:0];
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync_reg <= {SYNC_STAGES{0}};
        end else begin
            dst_wr_ptr_gray_sync_reg[0] <= src_wr_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                dst_wr_ptr_gray_sync_reg[i] <= dst_wr_ptr_gray_sync_reg[i-1];
            end
        end
    end
    assign dst_wr_ptr_gray_sync = dst_wr_ptr_gray_sync_reg[SYNC_STAGES-1];

    // Destination clear pending logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_pending <= 0;
        end else if (dst_clear_i) begin
            dst_clear_pending <= 1;
        end else if (dst_wr_ptr_bin == dst_rd_ptr_bin) begin
            dst_clear_pending <= 0;
        end
    end
    assign dst_clear_pending_o = dst_clear_pending;

    // Destination data and valid outputs
    assign dst_data_o = dst_data_reg;
    assign dst_valid_o = dst_valid_reg;

endmodule
