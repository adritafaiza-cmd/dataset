module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    // Source domain signals
    reg [ADDR_WIDTH:0] src_wr_ptr_gray;
    reg [ADDR_WIDTH:0] src_wr_ptr_bin;
    reg [ADDR_WIDTH:0] src_rd_ptr_gray_sync;
    reg [WIDTH-1:0] mem [DEPTH-1:0];
    reg src_clear_pending;
    reg src_ready;

    // Destination domain signals
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray;
    reg [ADDR_WIDTH:0] dst_rd_ptr_bin;
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_sync;
    reg [WIDTH-1:0] dst_data_reg [1:0];
    reg dst_clear_pending;
    reg dst_valid;
    reg [1:0] dst_valid_reg;

    // Gray code conversion functions
    function [ADDR_WIDTH:0] bin2gray;
        input [ADDR_WIDTH:0] bin;
        bin2gray = bin ^ (bin >> 1);
    endfunction

    function [ADDR_WIDTH:0] gray2bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = 1; i <= ADDR_WIDTH; i = i + 1) begin
            bin = bin ^ (gray >> i);
        end
        gray2bin = bin;
    endfunction

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr_gray <= 0;
            src_wr_ptr_bin <= 0;
            src_clear_pending <= 0;
        end else begin
            if (src_clear_i) begin
                src_wr_ptr_gray <= 0;
                src_wr_ptr_bin <= 0;
                src_clear_pending <= 1;
            end else if (src_ready && src_valid_i) begin
                src_wr_ptr_bin <= src_wr_ptr_bin + 1;
                src_wr_ptr_gray <= bin2gray(src_wr_ptr_bin + 1);
            end
            if (src_clear_pending && (src_wr_ptr_gray == src_rd_ptr_gray_sync)) begin
                src_clear_pending <= 0;
            end
        end
    end

    // Source domain ready signal
    always @(*) begin
        src_ready = 1;
        if (bin2gray(src_wr_ptr_bin + 1) == src_rd_ptr_gray_sync) begin
            src_ready = 0;
        end
    end

    // Source domain memory write
    always @(posedge src_clk_i) begin
        if (src_ready && src_valid_i) begin
            mem[src_wr_ptr_bin % DEPTH] <= src_data_i;
        end
    end

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr_gray <= 0;
            dst_rd_ptr_bin <= 0;
            dst_clear_pending <= 0;
            dst_valid_reg <= 0;
        end else begin
            if (dst_clear_i) begin
                dst_rd_ptr_gray <= 0;
                dst_rd_ptr_bin <= 0;
                dst_clear_pending <= 1;
                dst_valid_reg <= 0;
            end else if (dst_valid && dst_ready_i) begin
                dst_rd_ptr_bin <= dst_rd_ptr_bin + 1;
                dst_rd_ptr_gray <= bin2gray(dst_rd_ptr_bin + 1);
                dst_valid_reg <= {dst_valid_reg[0], 0};
            end else if (dst_valid && !dst_ready_i) begin
                dst_valid_reg <= {dst_valid_reg[0], dst_valid_reg[1]};
            end
            if (dst_clear_pending && (dst_wr_ptr_gray_sync == dst_rd_ptr_gray)) begin
                dst_clear_pending <= 0;
            end
        end
    end

    // Destination domain valid signal
    always @(*) begin
        dst_valid = dst_valid_reg[1];
    end

    // Destination domain memory read
    always @(posedge dst_clk_i) begin
        if (dst_valid && dst_ready_i) begin
            dst_data_reg[0] <= mem[dst_rd_ptr_bin % DEPTH];
        end
        dst_data_reg[1] <= dst_data_reg[0];
    end

    // Synchronization of pointers
    genvar i;
    generate
        for (i = 0; i < SYNC_STAGES; i = i + 1) begin : sync_stages
            always @(posedge dst_clk_i or negedge dst_rst_ni) begin
                if (!dst_rst_ni) begin
                    src_rd_ptr_gray_sync[i] <= 0;
                end else begin
                    src_rd_ptr_gray_sync[i] <= (i == 0) ? src_wr_ptr_gray : src_rd_ptr_gray_sync[i-1];
                end
            end

            always @(posedge src_clk_i or negedge src_rst_ni) begin
                if (!src_rst_ni) begin
                    dst_wr_ptr_gray_sync[i] <= 0;
                end else begin
                    dst_wr_ptr_gray_sync[i] <= (i == 0) ? dst_rd_ptr_gray : dst_wr_ptr_gray_sync[i-1];
                end
            end
        end
    endgenerate

    // Output assignments
    assign src_ready_o = src_ready;
    assign src_clear_pending_o = src_clear_pending;
    assign dst_data_o = dst_data_reg[1];
    assign dst_valid_o = dst_valid;
    assign dst_clear_pending_o = dst_clear_pending;

endmodule
