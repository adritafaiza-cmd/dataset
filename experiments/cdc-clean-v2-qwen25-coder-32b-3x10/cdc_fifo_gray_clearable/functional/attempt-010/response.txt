module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    reg [ADDR_WIDTH:0] src_wr_ptr_gray, src_wr_ptr_bin, src_wr_ptr_bin_next;
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray, dst_rd_ptr_bin, dst_rd_ptr_bin_next;
    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg src_clear_pending, dst_clear_pending;

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr_gray <= 0;
            src_wr_ptr_bin <= 0;
            src_clear_pending <= 0;
        end else begin
            if (src_clear_i) begin
                src_wr_ptr_gray <= 0;
                src_wr_ptr_bin <= 0;
                src_clear_pending <= 1;
            end else if (src_ready_o && src_valid_i) begin
                src_wr_ptr_bin <= src_wr_ptr_bin_next;
                src_wr_ptr_gray <= src_wr_ptr_bin_next ^ (src_wr_ptr_bin_next >> 1);
            end
            if (src_clear_pending && !dst_clear_pending) begin
                src_clear_pending <= 0;
            end
        end
    end

    assign src_wr_ptr_bin_next = src_wr_ptr_bin + 1;

    assign src_ready_o = ~((src_wr_ptr_bin_next[ADDR_WIDTH-1:0] == dst_rd_ptr_bin[ADDR_WIDTH-1:0]) && (src_wr_ptr_bin_next[ADDR_WIDTH] != dst_rd_ptr_bin[ADDR_WIDTH]));

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            mem[0] <= 0;
        end else if (src_ready_o && src_valid_i) begin
            mem[src_wr_ptr_bin[ADDR_WIDTH-1:0]] <= src_data_i;
        end
    end

    assign src_clear_pending_o = src_clear_pending;

    // Destination domain logic
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray_sync [0:SYNC_STAGES-1];

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr_gray_sync[0] <= 0;
        end else begin
            dst_rd_ptr_gray_sync[0] <= src_wr_ptr_gray;
        end
    end

    genvar i;
    generate
        for (i = 1; i < SYNC_STAGES; i = i + 1) begin : gen_sync_stages
            always @(posedge dst_clk_i or negedge dst_rst_ni) begin
                if (!dst_rst_ni) begin
                    dst_rd_ptr_gray_sync[i] <= 0;
                end else begin
                    dst_rd_ptr_gray_sync[i] <= dst_rd_ptr_gray_sync[i-1];
                end
            end
        end
    endgenerate

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr_bin <= 0;
            dst_clear_pending <= 0;
        end else begin
            if (dst_clear_i) begin
                dst_rd_ptr_bin <= 0;
                dst_clear_pending <= 1;
            end else if (dst_valid_o && dst_ready_i) begin
                dst_rd_ptr_bin <= dst_rd_ptr_bin_next;
            end
            if (dst_clear_pending && !src_clear_pending) begin
                dst_clear_pending <= 0;
            end
        end
    end

    assign dst_rd_ptr_bin_next = dst_rd_ptr_bin + 1;

    assign dst_valid_o = ~((dst_rd_ptr_bin_next[ADDR_WIDTH-1:0] == src_wr_ptr_bin[ADDR_WIDTH-1:0]) && (dst_rd_ptr_bin_next[ADDR_WIDTH] != src_wr_ptr_bin[ADDR_WIDTH]));

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_data_o <= 0;
        end else if (dst_valid_o && dst_ready_i) begin
            dst_data_o <= mem[dst_rd_ptr_bin[ADDR_WIDTH-1:0]];
        end
    end

    assign dst_clear_pending_o = dst_clear_pending;

endmodule
