module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    reg [ADDR_WIDTH:0] src_wr_ptr_gray, src_wr_ptr_bin;
    reg [ADDR_WIDTH:0] src_rd_ptr_gray_synced;
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_synced;
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray, dst_rd_ptr_bin;

    reg [WIDTH-1:0] mem [DEPTH-1:0];

    reg src_clear_pending;
    reg dst_clear_pending;

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr_gray <= 0;
            src_wr_ptr_bin <= 0;
            src_clear_pending <= 0;
        end else begin
            if (src_clear_i) begin
                src_wr_ptr_gray <= 0;
                src_wr_ptr_bin <= 0;
                src_clear_pending <= 1;
            end else if (src_ready_o && src_valid_i) begin
                src_wr_ptr_bin <= src_wr_ptr_bin + 1;
                src_wr_ptr_gray <= src_wr_ptr_bin ^ (src_wr_ptr_bin + 1);
            end

            if (src_clear_pending && (src_wr_ptr_gray == src_rd_ptr_gray_synced)) begin
                src_clear_pending <= 0;
            end
        end
    end

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr_gray <= 0;
            dst_rd_ptr_bin <= 0;
            dst_clear_pending <= 0;
        end else begin
            if (dst_clear_i) begin
                dst_rd_ptr_gray <= 0;
                dst_rd_ptr_bin <= 0;
                dst_clear_pending <= 1;
            end else if (dst_valid_o && dst_ready_i) begin
                dst_rd_ptr_bin <= dst_rd_ptr_bin + 1;
                dst_rd_ptr_gray <= dst_rd_ptr_bin ^ (dst_rd_ptr_bin + 1);
            end

            if (dst_clear_pending && (dst_wr_ptr_gray_synced == dst_rd_ptr_gray)) begin
                dst_clear_pending <= 0;
            end
        end
    end

    // Gray code synchronization
    reg [ADDR_WIDTH:0] src_wr_ptr_gray_synced [SYNC_STAGES-1:0];
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray_synced [SYNC_STAGES-1:0];

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_gray_synced <= {SYNC_STAGES{1'b0}};
        end else begin
            src_wr_ptr_gray_synced[0] <= src_wr_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                src_wr_ptr_gray_synced[i] <= src_wr_ptr_gray_synced[i-1];
            end
        end
    end

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_gray_synced <= {SYNC_STAGES{1'b0}};
        end else begin
            dst_rd_ptr_gray_synced[0] <= dst_rd_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                dst_rd_ptr_gray_synced[i] <= dst_rd_ptr_gray_synced[i-1];
            end
        end
    end

    assign src_wr_ptr_gray_synced = src_wr_ptr_gray_synced[SYNC_STAGES-1];
    assign dst_rd_ptr_gray_synced = dst_rd_ptr_gray_synced[SYNC_STAGES-1];

    // Source ready and destination valid
    assign src_ready_o = (src_wr_ptr_bin + 1 != dst_rd_ptr_gray_synced);
    assign dst_valid_o = (src_wr_ptr_gray_synced != dst_rd_ptr_bin);

    // Destination data output
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_data_o <= 0;
        end else if (dst_valid_o && dst_ready_i) begin
            dst_data_o <= mem[dst_rd_ptr_bin];
        end
    end

    // Memory write
    always @(posedge src_clk_i) begin
        if (src_ready_o && src_valid_i) begin
            mem[src_wr_ptr_bin] <= src_data_i;
        end
    end

    // Clear pending outputs
    assign src_clear_pending_o = src_clear_pending;
    assign dst_clear_pending_o = dst_clear_pending;

endmodule
