module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    // Source domain signals
    reg [ADDR_WIDTH:0] src_wr_ptr_gray;
    reg [ADDR_WIDTH:0] src_wr_ptr_bin;
    reg [ADDR_WIDTH:0] src_rd_ptr_gray_sync;
    reg [WIDTH-1:0] mem [DEPTH-1:0];
    reg src_clear_pending;
    reg src_ready;

    // Destination domain signals
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray;
    reg [ADDR_WIDTH:0] dst_rd_ptr_bin;
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_sync;
    reg dst_clear_pending;
    reg dst_valid;
    reg [WIDTH-1:0] dst_data;

    // Gray code conversion functions
    function [ADDR_WIDTH:0] bin_to_gray;
        input [ADDR_WIDTH:0] bin;
        bin_to_gray = bin ^ (bin >> 1);
    endfunction

    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = ADDR_WIDTH-1; i > 0; i = i - 1) begin
            bin[i-1] = bin[i] ^ bin[i-1];
        end
        gray_to_bin = bin;
    endfunction

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr_gray <= 0;
            src_wr_ptr_bin <= 0;
            src_clear_pending <= 0;
        end else begin
            if (src_clear_i) begin
                src_wr_ptr_gray <= 0;
                src_wr_ptr_bin <= 0;
                src_clear_pending <= 1;
            end else if (src_ready && src_valid_i) begin
                src_wr_ptr_bin <= src_wr_ptr_bin + 1;
                src_wr_ptr_gray <= bin_to_gray(src_wr_ptr_bin + 1);
            end
            if (src_clear_pending && (src_wr_ptr_gray == src_rd_ptr_gray_sync)) begin
                src_clear_pending <= 0;
            end
        end
    end

    // Source domain ready signal
    always @(*) begin
        src_ready = (src_wr_ptr_gray + 1 != src_rd_ptr_gray_sync);
    end

    // Source domain memory write
    always @(posedge src_clk_i) begin
        if (src_ready && src_valid_i) begin
            mem[src_wr_ptr_bin[ADDR_WIDTH-1:0]] <= src_data_i;
        end
    end

    // Source domain clear pending output
    assign src_clear_pending_o = src_clear_pending;

    // Source domain ready output
    assign src_ready_o = src_ready;

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr_gray <= 0;
            dst_rd_ptr_bin <= 0;
            dst_clear_pending <= 0;
        end else begin
            if (dst_clear_i) begin
                dst_rd_ptr_gray <= 0;
                dst_rd_ptr_bin <= 0;
                dst_clear_pending <= 1;
            end else if (dst_valid && dst_ready_i) begin
                dst_rd_ptr_bin <= dst_rd_ptr_bin + 1;
                dst_rd_ptr_gray <= bin_to_gray(dst_rd_ptr_bin + 1);
            end
            if (dst_clear_pending && (dst_wr_ptr_gray_sync == dst_rd_ptr_gray)) begin
                dst_clear_pending <= 0;
            end
        end
    end

    // Destination domain valid signal
    always @(*) begin
        dst_valid = (dst_wr_ptr_gray_sync != dst_rd_ptr_gray);
    end

    // Destination domain memory read
    always @(posedge dst_clk_i) begin
        if (dst_valid && dst_ready_i) begin
            dst_data <= mem[dst_rd_ptr_bin[ADDR_WIDTH-1:0]];
        end
    end

    // Destination domain clear pending output
    assign dst_clear_pending_o = dst_clear_pending;

    // Destination domain valid output
    assign dst_valid_o = dst_valid;

    // Destination domain data output
    assign dst_data_o = dst_data;

    // Gray code synchronization for source to destination
    genvar i;
    generate
        for (i = 0; i <= ADDR_WIDTH; i = i + 1) begin : sync_src_to_dst
            reg [SYNC_STAGES-1:0] sync_reg;
            always @(posedge dst_clk_i or negedge dst_rst_ni) begin
                if (!dst_rst_ni) begin
                    sync_reg <= 0;
                end else begin
                    sync_reg <= {sync_reg[SYNC_STAGES-2:0], src_wr_ptr_gray[i]};
                end
            end
            assign dst_wr_ptr_gray_sync[i] = sync_reg[SYNC_STAGES-1];
        end
    endgenerate

    // Gray code synchronization for destination to source
    generate
        for (i = 0; i <= ADDR_WIDTH; i = i + 1) begin : sync_dst_to_src
            reg [SYNC_STAGES-1:0] sync_reg;
            always @(posedge src_clk_i or negedge src_rst_ni) begin
                if (!src_rst_ni) begin
                    sync_reg <= 0;
                end else begin
                    sync_reg <= {sync_reg[SYNC_STAGES-2:0], dst_rd_ptr_gray[i]};
                end
            end
            assign src_rd_ptr_gray_sync[i] = sync_reg[SYNC_STAGES-1];
        end
    endgenerate

endmodule
