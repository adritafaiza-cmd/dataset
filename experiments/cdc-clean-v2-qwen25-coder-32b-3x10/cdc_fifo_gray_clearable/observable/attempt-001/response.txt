module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    reg [ADDR_WIDTH:0] src_wr_ptr_gray, src_wr_ptr_gray_nxt;
    reg [ADDR_WIDTH:0] src_rd_ptr_gray, src_rd_ptr_gray_nxt;
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray, dst_wr_ptr_gray_nxt;
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray, dst_rd_ptr_gray_nxt;

    reg [ADDR_WIDTH:0] src_wr_ptr_bin, src_wr_ptr_bin_nxt;
    reg [ADDR_WIDTH:0] src_rd_ptr_bin, src_rd_ptr_bin_nxt;
    reg [ADDR_WIDTH:0] dst_wr_ptr_bin, dst_wr_ptr_bin_nxt;
    reg [ADDR_WIDTH:0] dst_rd_ptr_bin, dst_rd_ptr_bin_nxt;

    reg [WIDTH-1:0] mem [DEPTH-1:0];

    reg src_clear_pending, dst_clear_pending;
    reg src_clear_req, dst_clear_req;

    // Source side logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr_gray <= 0;
            src_wr_ptr_bin <= 0;
            src_clear_pending <= 0;
        end else begin
            src_wr_ptr_gray <= src_wr_ptr_gray_nxt;
            src_wr_ptr_bin <= src_wr_ptr_bin_nxt;
            src_clear_pending <= src_clear_req;
        end
    end

    always @(*) begin
        src_wr_ptr_gray_nxt = src_wr_ptr_gray;
        src_wr_ptr_bin_nxt = src_wr_ptr_bin;
        src_clear_req = src_clear_i;

        if (src_valid_i && src_ready_o) begin
            src_wr_ptr_bin_nxt = src_wr_ptr_bin + 1;
            src_wr_ptr_gray_nxt = src_wr_ptr_bin_nxt ^ (src_wr_ptr_bin_nxt >> 1);
        end

        if (src_clear_i) begin
            src_wr_ptr_gray_nxt = 0;
            src_wr_ptr_bin_nxt = 0;
        end
    end

    assign src_ready_o = !(src_wr_ptr_gray_nxt[ADDR_WIDTH:ADDR_WIDTH-1] == src_rd_ptr_gray[ADDR_WIDTH:ADDR_WIDTH-1] &&
                          src_wr_ptr_gray_nxt[ADDR_WIDTH-1:0] == src_rd_ptr_gray[ADDR_WIDTH-1:0]);

    // Destination side logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr_gray <= 0;
            dst_rd_ptr_bin <= 0;
            dst_clear_pending <= 0;
        end else begin
            dst_rd_ptr_gray <= dst_rd_ptr_gray_nxt;
            dst_rd_ptr_bin <= dst_rd_ptr_bin_nxt;
            dst_clear_pending <= dst_clear_req;
        end
    end

    always @(*) begin
        dst_rd_ptr_gray_nxt = dst_rd_ptr_gray;
        dst_rd_ptr_bin_nxt = dst_rd_ptr_bin;
        dst_clear_req = dst_clear_i;

        if (dst_valid_o && dst_ready_i) begin
            dst_rd_ptr_bin_nxt = dst_rd_ptr_bin + 1;
            dst_rd_ptr_gray_nxt = dst_rd_ptr_bin_nxt ^ (dst_rd_ptr_bin_nxt >> 1);
        end

        if (dst_clear_i) begin
            dst_rd_ptr_gray_nxt = 0;
            dst_rd_ptr_bin_nxt = 0;
        end
    end

    assign dst_valid_o = !(dst_wr_ptr_gray == dst_rd_ptr_gray_nxt[ADDR_WIDTH:ADDR_WIDTH-1] &&
                          dst_wr_ptr_gray == dst_rd_ptr_gray_nxt[ADDR_WIDTH-1:0]);

    // Cross-domain synchronization for write pointer
    reg [ADDR_WIDTH:0] src_wr_ptr_gray_sync [SYNC_STAGES-1:0];

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            for (integer i = 0; i < SYNC_STAGES; i = i + 1) begin
                src_wr_ptr_gray_sync[i] <= 0;
            end
        end else begin
            src_wr_ptr_gray_sync[0] <= src_wr_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                src_wr_ptr_gray_sync[i] <= src_wr_ptr_gray_sync[i-1];
            end
        end
    end

    assign dst_wr_ptr_gray = src_wr_ptr_gray_sync[SYNC_STAGES-1];

    // Cross-domain synchronization for read pointer
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray_sync [SYNC_STAGES-1:0];

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            for (integer i = 0; i < SYNC_STAGES; i = i + 1) begin
                dst_rd_ptr_gray_sync[i] <= 0;
            end
        end else begin
            dst_rd_ptr_gray_sync[0] <= dst_rd_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                dst_rd_ptr_gray_sync[i] <= dst_rd_ptr_gray_sync[i-1];
            end
        end
    end

    assign src_rd_ptr_gray = dst_rd_ptr_gray_sync[SYNC_STAGES-1];

    // Memory write logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            for (integer i = 0; i < DEPTH; i = i + 1) begin
                mem[i] <= 0;
            end
        end else if (src_valid_i && src_ready_o) begin
            mem[src_wr_ptr_bin[ADDR_WIDTH-1:0]] <= src_data_i;
        end
    end

    // Memory read logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_data_o <= 0;
        end else if (dst_valid_o && dst_ready_i) begin
            dst_data_o <= mem[dst_rd_ptr_bin[ADDR_WIDTH-1:0]];
        end
    end

    assign src_clear_pending_o = src_clear_pending;
    assign dst_clear_pending_o = dst_clear_pending;

endmodule
