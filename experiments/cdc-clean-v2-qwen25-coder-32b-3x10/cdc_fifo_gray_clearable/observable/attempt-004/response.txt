```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    reg [ADDR_WIDTH:0] src_wr_ptr_gray, src_wr_ptr_gray_nxt;
    reg [ADDR_WIDTH:0] src_rd_ptr_gray, src_rd_ptr_gray_nxt;
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray, dst_wr_ptr_gray_nxt;
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray, dst_rd_ptr_gray_nxt;

    reg [WIDTH-1:0] mem [0:DEPTH-1];

    reg src_clear_reg, src_clear_reg_nxt;
    reg dst_clear_reg, dst_clear_reg_nxt;

    // Binary pointers for internal use
    reg [ADDR_WIDTH:0] src_wr_ptr_bin, src_wr_ptr_bin_nxt;
    reg [ADDR_WIDTH:0] src_rd_ptr_bin, src_rd_ptr_bin_nxt;
    reg [ADDR_WIDTH:0] dst_wr_ptr_bin, dst_wr_ptr_bin_nxt;
    reg [ADDR_WIDTH:0] dst_rd_ptr_bin, dst_rd_ptr_bin_nxt;

    // Synchronized clear signals
    reg [SYNC_STAGES-1:0] src_clear_sync;
    reg [SYNC_STAGES-1:0] dst_clear_sync;

    // Synchronized read pointers
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray_sync [SYNC_STAGES-1:0];
    reg [ADDR_WIDTH:0] src_rd_ptr_gray_sync [SYNC_STAGES-1:0];

    // Binary conversion
    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        begin
            gray_to_bin = gray;
            for (i = ADDR_WIDTH; i > 0; i = i - 1) begin
                gray_to_bin[i-1] = gray_to_bin[i] ^ gray_to_bin[i-1];
            end
        end
    endfunction

    // Binary to Gray conversion
    function [ADDR_WIDTH:0] bin_to_gray;
        input [ADDR_WIDTH:0] bin;
        begin
            bin_to_gray = bin ^ (bin >> 1);
        end
    endfunction

    // Source side logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr_gray <= 0;
            src_wr_ptr_bin <= 0;
            src_clear_reg <= 0;
        end else begin
            src_wr_ptr_gray <= src_wr_ptr_gray_nxt;
            src_wr_ptr_bin <= src_wr_ptr_bin_nxt;
            src_clear_reg <= src_clear_reg_nxt;
        end
    end

    always @(*) begin
        src_wr_ptr_gray_nxt = src_wr_ptr_gray;
        src_wr_ptr_bin_nxt = src_wr_ptr_bin;
        src_clear_reg_nxt = src_clear_reg;
        src_ready_o = 1'b0;

        if (src_clear_i) begin
            src_wr_ptr_gray_nxt = 0;
            src_wr_ptr_bin_nxt = 0;
            src_clear_reg_nxt = 1'b1;
        end else if (src_valid_i && !src_clear_reg) begin
            src_wr_ptr_bin_nxt = src_wr_ptr_bin + 1;
            src_wr_ptr_gray_nxt = bin_to_gray(src_wr_ptr_bin_nxt);
            src_ready_o = 1'b1;
        end
    end

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_pending_o <= 0;
        end else begin
            src_clear_pending_o <= src_clear_reg;
        end
    end

    // Destination side logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr_gray <= 0;
            dst_rd_ptr_bin <= 0;
            dst_clear_reg <= 0;
        end else begin
            dst_rd_ptr_gray <= dst_rd_ptr_gray_nxt;
            dst_rd_ptr_bin <= dst_rd_ptr_bin_nxt;
            dst_clear_reg <= dst_clear_reg_nxt;
        end
    end

    always @(*) begin
        dst_rd_ptr_gray_nxt = dst_rd_ptr_gray;
        dst_rd_ptr_bin_nxt = dst_rd_ptr_bin;
        dst_clear_reg_nxt = dst_clear_reg;
        dst_valid_o = 1'b0;

        if (dst_clear_i) begin
            dst_rd_ptr_gray_nxt = 0;
            dst_rd_ptr_bin_nxt = 0;
            dst_clear_reg_nxt = 1'b1;
        end else if (dst_ready_i && dst_valid_o && !dst_clear_reg) begin
            dst_rd_ptr_bin_nxt = dst_rd_ptr_bin + 1;
            dst_rd_ptr_gray_nxt = bin_to_gray(dst_rd_ptr_bin_nxt);
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_pending_o <= 0;
        end else begin
            dst_clear_pending_o <= dst_clear_reg;
        end
    end

    // Memory write
    always @(posedge src_clk_i) begin
        if (src_valid_i && src_ready_o && !src_clear_reg) begin
            mem[src_wr_ptr_bin[ADDR_WIDTH-1:0]] <= src_data_i;
        end
    end

    // Memory read
    always @(posedge dst_clk_i) begin
        if (dst_valid_o && dst_ready_i && !dst_clear_reg) begin
            dst_data_o <= mem[dst_rd_ptr_bin[ADDR_WIDTH-1:0]];
        end
    end

    // Synchronization of clear signals
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            src_clear_sync <= {src_clear_sync[SYNC_STAGES-2:0], src_clear_i};
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            dst_clear_sync <= {dst_clear_sync[SYNC_STAGES-2:0], dst_clear_i};
        end
    end

    // Synchronization of read pointers
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_gray_sync <= {SYNC_STAGES{ADDR_WIDTH+1{1'b0}}};
        end else begin
            dst_rd_ptr_gray_sync <= {dst_rd_ptr_gray_sync[SYNC_STAGES-2:0], dst_rd_ptr_gray};
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_rd_ptr_gray_sync <= {SYNC_STAGES{ADDR_WIDTH+1{1'b0}}};
        end else begin
            src_rd_ptr_gray_sync <= {src_rd_ptr_gray_sync[SYNC_STAGES-2:0], src_rd_ptr_gray};
        end
    end

    // Source side valid and ready
    assign src_ready_o = (src_wr_ptr_gray_nxt != (dst_rd_ptr_gray_sync[SYNC_STAGES-1] + 1));

    // Destination side valid and ready
    assign dst_valid_o = (dst_wr_ptr_gray_sync[SYNC_STAGES-1] != (dst_rd_ptr_gray_nxt + 1));

endmodule
