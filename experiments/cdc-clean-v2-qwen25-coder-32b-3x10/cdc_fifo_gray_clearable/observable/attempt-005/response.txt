module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    reg [ADDR_WIDTH:0] src_wr_ptr_gray, src_wr_ptr_gray_nxt;
    reg [ADDR_WIDTH:0] src_rd_ptr_gray, src_rd_ptr_gray_sync;
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_sync, dst_rd_ptr_gray;
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_nxt;
    reg [WIDTH-1:0] mem [DEPTH-1:0];

    reg src_clear_reg, dst_clear_reg;
    reg src_clear_sync, dst_clear_sync;

    // Gray code conversion functions
    function [ADDR_WIDTH:0] bin_to_gray;
        input [ADDR_WIDTH:0] bin;
        bin_to_gray = bin ^ (bin >> 1);
    endfunction

    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = ADDR_WIDTH-1; i > 0; i = i - 1) begin
            bin[i-1] = bin[i] ^ bin[i-1];
        end
        gray_to_bin = bin;
    endfunction

    // Source side logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr_gray <= 0;
            src_clear_reg <= 0;
        end else begin
            if (src_clear_i) begin
                src_wr_ptr_gray <= 0;
                src_clear_reg <= 1;
            end else if (src_valid_i && src_ready_o) begin
                src_wr_ptr_gray <= src_wr_ptr_gray_nxt;
            end
        end
    end

    assign src_wr_ptr_gray_nxt = bin_to_gray(gray_to_bin(src_wr_ptr_gray) + 1);

    // Source side clear synchronization
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_sync <= 0;
        end else begin
            src_clear_sync <= src_clear_reg;
        end
    end

    assign src_clear_pending_o = src_clear_reg;

    // Source side ready signal
    assign src_ready_o = (src_wr_ptr_gray_nxt != src_rd_ptr_gray_sync);

    // Source side write operation
    always @(posedge src_clk_i) begin
        if (src_valid_i && src_ready_o) begin
            mem[gray_to_bin(src_wr_ptr_gray)[ADDR_WIDTH-1:0]] <= src_data_i;
        end
    end

    // Destination side logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr_gray <= 0;
            dst_clear_reg <= 0;
        end else begin
            if (dst_clear_i) begin
                dst_rd_ptr_gray <= 0;
                dst_clear_reg <= 1;
            end else if (dst_valid_o && dst_ready_i) begin
                dst_rd_ptr_gray <= dst_wr_ptr_gray_sync;
            end
        end
    end

    // Destination side clear synchronization
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_sync <= 0;
        end else begin
            dst_clear_sync <= dst_clear_reg;
        end
    end

    assign dst_clear_pending_o = dst_clear_reg;

    // Gray code synchronization for source read pointer
    genvar i;
    generate
        for (i = 0; i < SYNC_STAGES; i = i + 1) begin : sync_src_rd_ptr_gray
            reg [ADDR_WIDTH:0] sync_stage;
            if (i == 0) begin
                always @(posedge dst_clk_i or negedge dst_rst_ni) begin
                    if (!dst_rst_ni) begin
                        sync_stage <= 0;
                    end else begin
                        sync_stage <= src_wr_ptr_gray;
                    end
                end
            end else begin
                always @(posedge dst_clk_i or negedge dst_rst_ni) begin
                    if (!dst_rst_ni) begin
                        sync_stage <= 0;
                    end else begin
                        sync_stage <= sync_src_rd_ptr_gray[i-1].sync_stage;
                    end
                end
            end
            if (i == SYNC_STAGES - 1) begin
                assign src_rd_ptr_gray_sync = sync_stage;
            end
        end
    endgenerate

    // Gray code synchronization for destination write pointer
    generate
        for (i = 0; i < SYNC_STAGES; i = i + 1) begin : sync_dst_wr_ptr_gray
            reg [ADDR_WIDTH:0] sync_stage;
            if (i == 0) begin
                always @(posedge src_clk_i or negedge src_rst_ni) begin
                    if (!src_rst_ni) begin
                        sync_stage <= 0;
                    end else begin
                        sync_stage <= dst_rd_ptr_gray;
                    end
                end
            end else begin
                always @(posedge src_clk_i or negedge src_rst_ni) begin
                    if (!src_rst_ni) begin
                        sync_stage <= 0;
                    end else begin
                        sync_stage <= sync_dst_wr_ptr_gray[i-1].sync_stage;
                    end
                end
            end
            if (i == SYNC_STAGES - 1) begin
                assign dst_wr_ptr_gray_sync = sync_stage;
            end
        end
    endgenerate

    // Destination side valid signal
    assign dst_valid_o = (dst_wr_ptr_gray_sync != dst_rd_ptr_gray);

    // Destination side read operation
    always @(posedge dst_clk_i) begin
        if (dst_valid_o && dst_ready_i) begin
            dst_data_o <= mem[gray_to_bin(dst_rd_ptr_gray)[ADDR_WIDTH-1:0]];
        end
    end

endmodule
