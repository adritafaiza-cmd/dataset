module wbxclk #(
		parameter	AW=32,
				DW=32,
				LGFIFO = 5
	) (
		input	wire			i_wb_clk, i_reset,
		input	wire			i_wb_cyc, i_wb_stb, i_wb_we,
		input	wire	[(AW-1):0]	i_wb_addr,
		input	wire	[(DW-1):0]	i_wb_data,
		input	wire	[(DW/8-1):0]	i_wb_sel,
		output	wire			o_wb_stall,
		output	reg			o_wb_ack,
		output	reg	[(DW-1):0]	o_wb_data,
		output	reg			o_wb_err,
		input	wire			i_xclk_clk,
		output	reg			o_xclk_cyc,
		output	reg			o_xclk_stb,
		output	reg			o_xclk_we,
		output	reg	[(AW-1):0]	o_xclk_addr,
		output	reg	[(DW-1):0]	o_xclk_data,
		output	reg	[(DW/8-1):0]	o_xclk_sel,
		input	wire			i_xclk_stall,
		input	wire			i_xclk_ack,
		input	wire	[(DW-1):0]	i_xclk_data,
		input	wire			i_xclk_err
	);

	// FIFO for Wishbone to XCLK domain
	reg [(AW-1):0] fifo_addr [0:(1<<LGFIFO)-1];
	reg [(DW-1):0] fifo_data [0:(1<<LGFIFO)-1];
	reg [(DW/8-1):0] fifo_sel [0:(1<<LGFIFO)-1];
	reg fifo_we;
	reg fifo_re;
	reg [(LGFIFO-1):0] fifo_wptr;
	reg [(LGFIFO-1):0] fifo_rptr;
	reg [(LGFIFO):0] fifo_count;

	// FIFO write logic
	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset) begin
			fifo_wptr <= 0;
			fifo_count <= 0;
		end else if (fifo_we && !fifo_re) begin
			fifo_addr[fifo_wptr] <= i_wb_addr;
			fifo_data[fifo_wptr] <= i_wb_data;
			fifo_sel[fifo_wptr] <= i_wb_sel;
			fifo_wptr <= fifo_wptr + 1;
			fifo_count <= fifo_count + 1;
		end else if (!fifo_we && fifo_re) begin
			fifo_rptr <= fifo_rptr + 1;
			fifo_count <= fifo_count - 1;
		end else if (fifo_we && fifo_re) begin
			fifo_addr[fifo_wptr] <= i_wb_addr;
			fifo_data[fifo_wptr] <= i_wb_data;
			fifo_sel[fifo_wptr] <= i_wb_sel;
			fifo_wptr <= fifo_wptr + 1;
			fifo_rptr <= fifo_rptr + 1;
		end
	end

	// FIFO read logic
	always @(posedge i_xclk_clk or posedge i_reset) begin
		if (i_reset) begin
			o_xclk_cyc <= 0;
			o_xclk_stb <= 0;
			o_xclk_we <= 0;
			o_xclk_addr <= 0;
			o_xclk_data <= 0;
			o_xclk_sel <= 0;
		end else if (fifo_re) begin
			o_xclk_cyc <= 1;
			o_xclk_stb <= 1;
			o_xclk_we <= fifo_we;
			o_xclk_addr <= fifo_addr[fifo_rptr];
			o_xclk_data <= fifo_data[fifo_rptr];
			o_xclk_sel <= fifo_sel[fifo_rptr];
		end else begin
			o_xclk_stb <= 0;
		end
	end

	// FIFO control logic
	always @(posedge i_wb_clk or posedge i_reset) begin
		if (i_reset) begin
			fifo_we <= 0;
			o_wb_stall <= 0;
		end else begin
			fifo_we <= i_wb_cyc && i_wb_stb && (fifo_count < (1<<LGFIFO));
			o_wb_stall <= !(fifo_count < (1<<LGFIFO));
		end
	end

	always @(posedge i_xclk_clk or posedge i_reset) begin
		if (i_reset) begin
			fifo_re <= 0;
			o_wb_ack <= 0;
			o_wb_err <= 0;
			o_wb_data <= 0;
		end else begin
			fifo_re <= o_xclk_stb && !i_xclk_stall && i_xclk_ack;
			o_wb_ack <= i_xclk_ack;
			o_wb_err <= i_xclk_err;
			o_wb_data <= i_xclk_data;
		end
	end

endmodule
