module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    // Source domain signals
    reg [ADDR_WIDTH:0] src_wr_ptr_gray;
    reg [ADDR_WIDTH:0] src_wr_ptr_bin;
    reg [ADDR_WIDTH:0] src_rd_ptr_gray_sync;

    // Destination domain signals
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray;
    reg [ADDR_WIDTH:0] dst_rd_ptr_bin;
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_sync;

    // Memory
    reg [WIDTH-1:0] mem [DEPTH-1:0];

    // Gray code conversion functions
    function [ADDR_WIDTH:0] bin_to_gray;
        input [ADDR_WIDTH:0] bin;
        bin_to_gray = bin ^ (bin >> 1);
    endfunction

    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = ADDR_WIDTH-1; i > 0; i = i - 1) begin
            bin[i-1] = bin[i] ^ bin[i-1];
        end
        gray_to_bin = bin;
    endfunction

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr_gray <= 0;
            src_wr_ptr_bin <= 0;
        end else if (src_valid_i && src_ready_o) begin
            src_wr_ptr_bin <= src_wr_ptr_bin + 1;
            src_wr_ptr_gray <= bin_to_gray(src_wr_ptr_bin + 1);
        end
    end

    // Synchronize read pointer to source domain
    reg [ADDR_WIDTH:0] src_rd_ptr_gray_sync_reg [SYNC_STAGES-1:0];
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rd_ptr_gray_sync_reg <= {SYNC_STAGES{1'b0}};
        end else begin
            src_rd_ptr_gray_sync_reg[0] <= dst_rd_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                src_rd_ptr_gray_sync_reg[i] <= src_rd_ptr_gray_sync_reg[i-1];
            end
        end
    end
    assign src_rd_ptr_gray_sync = src_rd_ptr_gray_sync_reg[SYNC_STAGES-1];

    // Source ready signal
    assign src_ready_o = (src_wr_ptr_gray != src_rd_ptr_gray_sync);

    // Write data to memory
    always @(posedge src_clk_i) begin
        if (src_valid_i && src_ready_o) begin
            mem[src_wr_ptr_bin[ADDR_WIDTH-1:0]] <= src_data_i;
        end
    end

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr_gray <= 0;
            dst_rd_ptr_bin <= 0;
        end else if (dst_valid_o && dst_ready_i) begin
            dst_rd_ptr_bin <= dst_rd_ptr_bin + 1;
            dst_rd_ptr_gray <= bin_to_gray(dst_rd_ptr_bin + 1);
        end
    end

    // Synchronize write pointer to destination domain
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_sync_reg [SYNC_STAGES-1:0];
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_wr_ptr_gray_sync_reg <= {SYNC_STAGES{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync_reg[0] <= src_wr_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                dst_wr_ptr_gray_sync_reg[i] <= dst_wr_ptr_gray_sync_reg[i-1];
            end
        end
    end
    assign dst_wr_ptr_gray_sync = dst_wr_ptr_gray_sync_reg[SYNC_STAGES-1];

    // Read data from memory
    reg [WIDTH-1:0] dst_data_reg [1:0];
    reg [1:0] dst_valid_reg;

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_data_reg <= {2{WIDTH{1'b0}}};
            dst_valid_reg <= 2'b0;
        end else begin
            dst_data_reg[0] <= mem[dst_rd_ptr_bin[ADDR_WIDTH-1:0]];
            dst_valid_reg[0] <= (dst_wr_ptr_gray_sync != dst_rd_ptr_gray);
            dst_data_reg[1] <= dst_data_reg[0];
            dst_valid_reg[1] <= dst_valid_reg[0];
        end
    end

    assign dst_data_o = dst_data_reg[1];
    assign dst_valid_o = dst_valid_reg[1];

endmodule
