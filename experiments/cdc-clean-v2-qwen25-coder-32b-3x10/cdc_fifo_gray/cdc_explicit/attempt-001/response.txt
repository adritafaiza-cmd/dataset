module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    // Source domain signals
    reg [ADDR_WIDTH:0] src_wr_ptr_gray;
    reg [ADDR_WIDTH:0] src_wr_ptr_bin;
    reg [ADDR_WIDTH:0] src_rd_ptr_gray_synced;
    reg [WIDTH-1:0]    mem [0:DEPTH-1];

    // Destination domain signals
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray;
    reg [ADDR_WIDTH:0] dst_rd_ptr_bin;
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_synced;
    reg [WIDTH-1:0]    dst_data_reg [0:SYNC_STAGES-1];
    reg                dst_valid_reg [0:SYNC_STAGES-1];

    // Binary to Gray conversion
    function [ADDR_WIDTH:0] bin_to_gray;
        input [ADDR_WIDTH:0] bin;
        bin_to_gray = bin ^ (bin >> 1);
    endfunction

    // Gray to Binary conversion
    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = ADDR_WIDTH-1; i > 0; i = i - 1) begin
            bin[i-1] = bin[i] ^ bin[i-1];
        end
        gray_to_bin = bin;
    endfunction

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr_gray <= 0;
            src_wr_ptr_bin <= 0;
        end else if (src_valid_i && src_ready_o) begin
            src_wr_ptr_bin <= src_wr_ptr_bin + 1;
            src_wr_ptr_gray <= bin_to_gray(src_wr_ptr_bin + 1);
            mem[src_wr_ptr_bin % DEPTH] <= src_data_i;
        end
    end

    // Synchronize read pointer to destination domain
    reg [ADDR_WIDTH:0] src_rd_ptr_gray_sync [0:SYNC_STAGES-1];
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_rd_ptr_gray_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            src_rd_ptr_gray_sync[0] <= src_wr_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                src_rd_ptr_gray_sync[i] <= src_rd_ptr_gray_sync[i-1];
            end
        end
    end

    assign src_rd_ptr_gray_synced = src_rd_ptr_gray_sync[SYNC_STAGES-1];
    assign src_rd_ptr_bin = gray_to_bin(src_rd_ptr_gray_synced);

    // Source ready signal
    assign src_ready_o = (src_wr_ptr_bin != src_rd_ptr_bin + DEPTH);

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr_gray <= 0;
            dst_rd_ptr_bin <= 0;
            dst_data_reg <= {SYNC_STAGES{WIDTH{1'b0}}};
            dst_valid_reg <= {SYNC_STAGES{1'b0}};
        end else if (dst_valid_reg[SYNC_STAGES-2] && dst_ready_i) begin
            dst_rd_ptr_bin <= dst_rd_ptr_bin + 1;
            dst_rd_ptr_gray <= bin_to_gray(dst_rd_ptr_bin + 1);
        end

        // Shift data and valid through synchronization stages
        if (dst_valid_reg[SYNC_STAGES-2] && dst_ready_i) begin
            dst_data_reg[SYNC_STAGES-1] <= dst_data_reg[SYNC_STAGES-2];
            dst_valid_reg[SYNC_STAGES-1] <= dst_valid_reg[SYNC_STAGES-2];
        end

        if (dst_wr_ptr_gray_synced != dst_rd_ptr_gray) begin
            dst_data_reg[0] <= mem[dst_wr_ptr_bin % DEPTH];
            dst_valid_reg[0] <= 1;
        end else begin
            dst_valid_reg[0] <= 0;
        end

        for (integer i = 1; i < SYNC_STAGES-1; i = i + 1) begin
            dst_data_reg[i] <= dst_data_reg[i-1];
            dst_valid_reg[i] <= dst_valid_reg[i-1];
        end
    end

    // Synchronize write pointer to source domain
    reg [ADDR_WIDTH:0] dst_wr_ptr_gray_sync [0:SYNC_STAGES-1];
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_wr_ptr_gray_sync <= {SYNC_STAGES{1'b0}};
        end else begin
            dst_wr_ptr_gray_sync[0] <= dst_rd_ptr_gray;
            for (integer i = 1; i < SYNC_STAGES; i = i + 1) begin
                dst_wr_ptr_gray_sync[i] <= dst_wr_ptr_gray_sync[i-1];
            end
        end
    end

    assign dst_wr_ptr_gray_synced = dst_wr_ptr_gray_sync[SYNC_STAGES-1];
    assign dst_wr_ptr_bin = gray_to_bin(dst_wr_ptr_gray_synced);

    // Destination valid and data signals
    assign dst_valid_o = dst_valid_reg[SYNC_STAGES-1];
    assign dst_data_o = dst_data_reg[SYNC_STAGES-1];

endmodule
