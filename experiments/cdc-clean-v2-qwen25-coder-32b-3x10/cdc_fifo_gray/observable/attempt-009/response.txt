module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam ADDR_WIDTH = LOG_DEPTH;

    reg [WIDTH-1:0] mem [DEPTH-1:0];

    // Source side pointers
    reg [ADDR_WIDTH:0] src_wr_ptr;
    reg [ADDR_WIDTH:0] src_wr_ptr_gray;
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray_sync;

    // Destination side pointers
    reg [ADDR_WIDTH:0] dst_rd_ptr;
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray;
    reg [ADDR_WIDTH:0] src_wr_ptr_gray_sync;

    // Gray code conversion functions
    function [ADDR_WIDTH:0] bin_to_gray;
        input [ADDR_WIDTH:0] bin;
        bin_to_gray = bin ^ (bin >> 1);
    endfunction

    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = ADDR_WIDTH; i > 0; i = i - 1) begin
            bin[i-1] = bin[i] ^ bin[i-1];
        end
        gray_to_bin = bin;
    endfunction

    // Source side logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_wr_ptr <= 0;
            src_wr_ptr_gray <= 0;
        end else if (src_valid_i && src_ready_o) begin
            mem[src_wr_ptr[ADDR_WIDTH-1:0]] <= src_data_i;
            src_wr_ptr <= src_wr_ptr + 1;
            src_wr_ptr_gray <= bin_to_gray(src_wr_ptr + 1);
        end
    end

    // Synchronize destination read pointer to source clock domain
    reg [ADDR_WIDTH:0] dst_rd_ptr_gray_sync_reg [SYNC_STAGES-1:0];
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_rd_ptr_gray_sync_reg <= {SYNC_STAGES{0}};
        end else begin
            dst_rd_ptr_gray_sync_reg[0] <= dst_rd_ptr_gray;
            for (integer i = 0; i < SYNC_STAGES-1; i = i + 1) begin
                dst_rd_ptr_gray_sync_reg[i+1] <= dst_rd_ptr_gray_sync_reg[i];
            end
        end
    end
    assign dst_rd_ptr_gray_sync = dst_rd_ptr_gray_sync_reg[SYNC_STAGES-1];

    // Source ready signal
    assign src_ready_o = (src_wr_ptr_gray + 1 != dst_rd_ptr_gray_sync);

    // Destination side logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rd_ptr <= 0;
            dst_rd_ptr_gray <= 0;
        end else if (dst_valid_o && dst_ready_i) begin
            dst_rd_ptr <= dst_rd_ptr + 1;
            dst_rd_ptr_gray <= bin_to_gray(dst_rd_ptr + 1);
        end
    end

    // Synchronize source write pointer to destination clock domain
    reg [ADDR_WIDTH:0] src_wr_ptr_gray_sync_reg [SYNC_STAGES-1:0];
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_wr_ptr_gray_sync_reg <= {SYNC_STAGES{0}};
        end else begin
            src_wr_ptr_gray_sync_reg[0] <= src_wr_ptr_gray;
            for (integer i = 0; i < SYNC_STAGES-1; i = i + 1) begin
                src_wr_ptr_gray_sync_reg[i+1] <= src_wr_ptr_gray_sync_reg[i];
            end
        end
    end
    assign src_wr_ptr_gray_sync = src_wr_ptr_gray_sync_reg[SYNC_STAGES-1];

    // Destination valid signal
    assign dst_valid_o = (src_wr_ptr_gray_sync != dst_rd_ptr_gray);

    // Destination data output
    assign dst_data_o = mem[dst_rd_ptr[ADDR_WIDTH-1:0]];

endmodule
