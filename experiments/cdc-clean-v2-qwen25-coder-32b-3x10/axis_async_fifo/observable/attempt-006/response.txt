module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   s_pause_req,
    output wire                   s_pause_ack,
    input  wire                   m_pause_req,
    output wire                   m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] s_status_depth,
    output wire [$clog2(DEPTH):0] s_status_depth_commit,
    output wire                   s_status_overflow,
    output wire                   s_status_bad_frame,
    output wire                   s_status_good_frame,
    output wire [$clog2(DEPTH):0] m_status_depth,
    output wire [$clog2(DEPTH):0] m_status_depth_commit,
    output wire                   m_status_overflow,
    output wire                   m_status_bad_frame,
    output wire                   m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);

    reg [ADDR_WIDTH:0] s_wr_ptr, s_wr_ptr_gray, s_wr_ptr_gray_sync;
    reg [ADDR_WIDTH:0] s_rd_ptr, s_rd_ptr_gray, s_rd_ptr_gray_sync;
    reg [ADDR_WIDTH:0] m_wr_ptr, m_wr_ptr_gray, m_wr_ptr_gray_sync;
    reg [ADDR_WIDTH:0] m_rd_ptr, m_rd_ptr_gray, m_rd_ptr_gray_sync;

    reg [DATA_WIDTH-1:0] mem_data [0:DEPTH-1];
    reg [KEEP_WIDTH-1:0] mem_keep [0:DEPTH-1];
    reg [USER_WIDTH-1:0] mem_user [0:DEPTH-1];
    reg [ID_WIDTH-1:0] mem_id [0:DEPTH-1];
    reg [DEST_WIDTH-1:0] mem_dest [0:DEPTH-1];
    reg mem_last [0:DEPTH-1];

    reg s_axis_tready_reg;
    reg m_axis_tvalid_reg;
    reg s_pause_ack_reg;
    reg m_pause_ack_reg;

    reg s_status_overflow_reg;
    reg s_status_bad_frame_reg;
    reg s_status_good_frame_reg;
    reg m_status_overflow_reg;
    reg m_status_bad_frame_reg;
    reg m_status_good_frame_reg;

    reg [ADDR_WIDTH:0] s_status_depth_reg;
    reg [ADDR_WIDTH:0] s_status_depth_commit_reg;
    reg [ADDR_WIDTH:0] m_status_depth_reg;
    reg [ADDR_WIDTH:0] m_status_depth_commit_reg;

    // Binary to Gray conversion
    function [ADDR_WIDTH:0] bin_to_gray;
        input [ADDR_WIDTH:0] bin;
        bin_to_gray = bin ^ (bin >> 1);
    endfunction

    // Gray to Binary conversion
    function [ADDR_WIDTH:0] gray_to_bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = ADDR_WIDTH-1; i >= 0; i = i - 1)
            bin[i] = bin[i] ^ bin[i+1];
        gray_to_bin = bin;
    endfunction

    // Synchronous FIFO Write Logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= 0;
            s_wr_ptr_gray <= 0;
            s_wr_ptr_gray_sync <= 0;
            s_axis_tready_reg <= 0;
            s_status_depth_reg <= 0;
            s_status_depth_commit_reg <= 0;
            s_status_overflow_reg <= 0;
            s_status_bad_frame_reg <= 0;
            s_status_good_frame_reg <= 0;
        end else begin
            reg [ADDR_WIDTH:0] next_wr_ptr;
            reg [ADDR_WIDTH:0] next_wr_ptr_gray;
            reg [ADDR_WIDTH:0] next_wr_ptr_gray_sync;
            reg [ADDR_WIDTH:0] next_depth;
            reg [ADDR_WIDTH:0] next_depth_commit;
            reg next_axis_tready;
            reg next_status_overflow;
            reg next_status_bad_frame;
            reg next_status_good_frame;

            next_wr_ptr = s_wr_ptr;
            next_wr_ptr_gray = s_wr_ptr_gray;
            next_wr_ptr_gray_sync = s_wr_ptr_gray_sync;
            next_depth = s_status_depth_reg;
            next_depth_commit = s_status_depth_commit_reg;
            next_axis_tready = s_axis_tready_reg;
            next_status_overflow = s_status_overflow_reg;
            next_status_bad_frame = s_status_bad_frame_reg;
            next_status_good_frame = s_status_good_frame_reg;

            if (s_axis_tvalid && s_axis_tready_reg) begin
                mem_data[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tdata;
                mem_keep[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tkeep;
                mem_user[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tuser;
                mem_id[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tid;
                mem_dest[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tdest;
                mem_last[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tlast;
                next_wr_ptr = s_wr_ptr + 1;
                next_wr_ptr_gray = bin_to_gray(next_wr_ptr);
                next_depth = next_depth + 1;
                if (s_axis_tlast) begin
                    next_depth_commit = next_depth;
                    if (next_depth_commit > DEPTH) begin
                        next_status_overflow = 1;
                    end else begin
                        next_status_good_frame = 1;
                    end
                end
            end

            if (next_depth > DEPTH) begin
                next_axis_tready = 0;
                next_status_overflow = 1;
            end else if (DROP_WHEN_FULL && next_depth == DEPTH) begin
                next_axis_tready = 0;
                next_status_overflow = 1;
            end else if (MARK_WHEN_FULL && next_depth == DEPTH) begin
                next_axis_tready = 0;
                next_status_overflow = 1;
            end else begin
                next_axis_tready = 1;
                next_status_overflow = 0;
            end

            s_wr_ptr <= next_wr_ptr;
            s_wr_ptr_gray <= next_wr_ptr_gray;
            s_wr_ptr_gray_sync <= next_wr_ptr_gray;
            s_axis_tready_reg <= next_axis_tready;
            s_status_depth_reg <= next_depth;
            s_status_depth_commit_reg <= next_depth_commit;
            s_status_overflow_reg <= next_status_overflow;
            s_status_bad_frame_reg <= next_status_bad_frame;
            s_status_good_frame_reg <= next_status_good_frame;
        end
    end

    // Synchronous FIFO Read Logic
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr <= 0;
            m_rd_ptr_gray <= 0;
            m_rd_ptr_gray_sync <= 0;
            m_axis_tvalid_reg <= 0;
            m_status_depth_reg <= 0;
            m_status_depth_commit_reg <= 0;
            m_status_overflow_reg <= 0;
            m_status_bad_frame_reg <= 0;
            m_status_good_frame_reg <= 0;
        end else begin
            reg [ADDR_WIDTH:0] next_rd_ptr;
            reg [ADDR_WIDTH:0] next_rd_ptr_gray;
            reg [ADDR_WIDTH:0] next_rd_ptr_gray_sync;
            reg [ADDR_WIDTH:0] next_depth;
            reg [ADDR_WIDTH:0] next_depth_commit;
            reg next_axis_tvalid;
            reg next_status_overflow;
            reg next_status_bad_frame;
            reg next_status_good_frame;

            next_rd_ptr = m_rd_ptr;
            next_rd_ptr_gray = m_rd_ptr_gray;
            next_rd_ptr_gray_sync = m_rd_ptr_gray_sync;
            next_depth = m_status_depth_reg;
            next_depth_commit = m_status_depth_commit_reg;
            next_axis_tvalid = m_axis_tvalid_reg;
            next_status_overflow = m_status_overflow_reg;
            next_status_bad_frame = m_status_bad_frame_reg;
            next_status_good_frame = m_status_good_frame_reg;

            if (m_axis_tvalid_reg && m_axis_tready) begin
                next_rd_ptr = m_rd_ptr + 1;
                next_rd_ptr_gray = bin_to_gray(next_rd_ptr);
                next_depth = next_depth - 1;
                if (mem_last[m_rd_ptr[ADDR_WIDTH-1:0]]) begin
                    next_depth_commit = next_depth;
                    if (next_depth_commit > DEPTH) begin
                        next_status_overflow = 1;
                    end else begin
                        next_status_good_frame = 1;
                    end
                end
            end

            if (next_depth == 0) begin
                next_axis_tvalid = 0;
            end else begin
                next_axis_tvalid = 1;
            end

            m_rd_ptr <= next_rd_ptr;
            m_rd_ptr_gray <= next_rd_ptr_gray;
            m_rd_ptr_gray_sync <= next_rd_ptr_gray;
            m_axis_tvalid_reg <= next_axis_tvalid;
            m_status_depth_reg <= next_depth;
            m_status_depth_commit_reg <= next_depth_commit;
            m_status_overflow_reg <= next_status_overflow;
            m_status_bad_frame_reg <= next_status_bad_frame;
            m_status_good_frame_reg <= next_status_good_frame;
        end
    end

    // Cross-domain synchronization for write pointer
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_wr_ptr_gray_sync <= 0;
        end else begin
            s_wr_ptr_gray_sync <= s_wr_ptr_gray;
        end
    end

    // Cross-domain synchronization for read pointer
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_rd_ptr_gray_sync <= 0;
        end else begin
            m_rd_ptr_gray_sync <= m_rd_ptr_gray;
        end
    end

    // Cross-domain FIFO depth calculation
    wire [ADDR_WIDTH:0] s_depth;
    wire [ADDR_WIDTH:0] m_depth;

    assign s_depth = gray_to_bin(s_wr_ptr_gray_sync) - gray_to_bin(s_rd_ptr_gray);
    assign m_depth = gray_to_bin(m_wr_ptr_gray) - gray_to_bin(m_rd_ptr_gray_sync);

    // Output FIFO logic (if enabled)
    reg [ADDR_WIDTH:0] m_wr_ptr_out, m_wr_ptr_out_gray, m_wr_ptr_out_gray_sync;
    reg [ADDR_WIDTH:0] m_rd_ptr_out, m_rd_ptr_out_gray, m_rd_ptr_out_gray_sync;

    if (OUTPUT_FIFO_ENABLE) begin
        always @(posedge m_clk or posedge m_rst) begin
            if (m_rst) begin
                m_wr_ptr_out <= 0;
                m_wr_ptr_out_gray <= 0;
                m_wr_ptr_out_gray_sync <= 0;
                m_rd_ptr_out <= 0;
                m_rd_ptr_out_gray <= 0;
                m_rd_ptr_out_gray_sync <= 0;
            end else begin
                reg [ADDR_WIDTH:0] next_wr_ptr_out;
                reg [ADDR_WIDTH:0] next_wr_ptr_out_gray;
                reg [ADDR_WIDTH:0] next_wr_ptr_out_gray_sync;
                reg [ADDR_WIDTH:0] next_rd_ptr_out;
                reg [ADDR_WIDTH:0] next_rd_ptr_out_gray;
                reg [ADDR_WIDTH:0] next_rd_ptr_out_gray_sync;

                next_wr_ptr_out = m_wr_ptr_out;
                next_wr_ptr_out_gray = m_wr_ptr_out_gray;
                next_wr_ptr_out_gray_sync = m_wr_ptr_out_gray_sync;
                next_rd_ptr_out = m_rd_ptr_out;
                next_rd_ptr_out_gray = m_rd_ptr_out_gray;
                next_rd_ptr_out_gray_sync = m_rd_ptr_out_gray_sync;

                if (m_axis_tvalid_reg && m_axis_tready) begin
                    mem_data[m_wr_ptr_out[ADDR_WIDTH-1:0]] <= mem_data[m_rd_ptr[ADDR_WIDTH-1:0]];
                    mem_keep[m_wr_ptr_out[ADDR_WIDTH-1:0]] <= mem_keep[m_rd_ptr[ADDR_WIDTH-1:0]];
                    mem_user[m_wr_ptr_out[ADDR_WIDTH-1:0]] <= mem_user[m_rd_ptr[ADDR_WIDTH-1:0]];
                    mem_id[m_wr_ptr_out[ADDR_WIDTH-1:0]] <= mem_id[m_rd_ptr[ADDR_WIDTH-1:0]];
                    mem_dest[m_wr_ptr_out[ADDR_WIDTH-1:0]] <= mem_dest[m_rd_ptr[ADDR_WIDTH-1:0]];
                    mem_last[m_wr_ptr_out[ADDR_WIDTH-1:0]] <= mem_last[m_rd_ptr[ADDR_WIDTH-1:0]];
                    next_wr_ptr_out = m_wr_ptr_out + 1;
                    next_wr_ptr_out_gray = bin_to_gray(next_wr_ptr_out);
                end

                if (m_axis_tready) begin
                    next_rd_ptr_out = m_rd_ptr_out + 1;
                    next_rd_ptr_out_gray = bin_to_gray(next_rd_ptr_out);
                end

                m_wr_ptr_out <= next_wr_ptr_out;
                m_wr_ptr_out_gray <= next_wr_ptr_out_gray;
                m_wr_ptr_out_gray_sync <= next_wr_ptr_out_gray;
                m_rd_ptr_out <= next_rd_ptr_out;
                m_rd_ptr_out_gray <= next_rd_ptr_out_gray;
                m_rd_ptr_out_gray_sync <= next_rd_ptr_out_gray;
            end
        end

        // Cross-domain synchronization for output FIFO write pointer
        always @(posedge m_clk or posedge m_rst) begin
            if (m_rst) begin
                m_wr_ptr_out_gray_sync <= 0;
            end else begin
                m_wr_ptr_out_gray_sync <= m_wr_ptr_out_gray;
            end
        end

        // Cross-domain synchronization for output FIFO read pointer
        always @(posedge m_clk or posedge m_rst) begin
            if (m_rst) begin
                m_rd_ptr_out_gray_sync <= 0;
            end else begin
                m_rd_ptr_out_gray_sync <= m_rd_ptr_out_gray;
            end
        end

        // Output FIFO depth calculation
        wire [ADDR_WIDTH:0] m_depth_out;

        assign m_depth_out = gray_to_bin(m_wr_ptr_out_gray_sync) - gray_to_bin(m_rd_ptr_out_gray_sync);

        // Output FIFO read logic
        always @(posedge m_clk or posedge m_rst) begin
            if (m_rst) begin
                m_axis_tdata <= 0;
                m_axis_tkeep <= 0;
                m_axis_tlast <= 0;
                m_axis_tid <= 0;
                m_axis_tdest <= 0;
                m_axis_tuser <= 0;
            end else begin
                if (m_axis_tready) begin
                    m_axis_tdata <= mem_data[m_rd_ptr_out[ADDR_WIDTH-1:0]];
                    m_axis_tkeep <= mem_keep[m_rd_ptr_out[ADDR_WIDTH-1:0]];
                    m_axis_tlast <= mem_last[m_rd_ptr_out[ADDR_WIDTH-1:0]];
                    m_axis_tid <= mem_id[m_rd_ptr_out[ADDR_WIDTH-1:0]];
                    m_axis_tdest <= mem_dest[m_rd_ptr_out[ADDR_WIDTH-1:0]];
                    m_axis_tuser <= mem_user[m_rd_ptr_out[ADDR_WIDTH-1:0]];
                end
            end
        end
    end else begin
        // Directly output from main FIFO
        always @(posedge m_clk or posedge m_rst) begin
            if (m_rst) begin
                m_axis_tdata <= 0;
                m_axis_tkeep <= 0;
                m_axis_tlast <= 0;
                m_axis_tid <= 0;
                m_axis_tdest <= 0;
                m_axis_tuser <= 0;
            end else begin
                if (m_axis_tvalid_reg) begin
                    m_axis_tdata <= mem_data[m_rd_ptr[ADDR_WIDTH-1:0]];
                    m_axis_tkeep <= mem_keep[m_rd_ptr[ADDR_WIDTH-1:0]];
                    m_axis_tlast <= mem_last[m_rd_ptr[ADDR_WIDTH-1:0]];
                    m_axis_tid <= mem_id[m_rd_ptr[ADDR_WIDTH-1:0]];
                    m_axis_tdest <= mem_dest[m_rd_ptr[ADDR_WIDTH-1:0]];
                    m_axis_tuser <= mem_user[m_rd_ptr[ADDR_WIDTH-1:0]];
                end
            end
        end
    end

    // Pause logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_pause_ack_reg <= 0;
        end else begin
            if (s_pause_req) begin
                s_pause_ack_reg <= 1;
            end else begin
                s_pause_ack_reg <= 0;
            end
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_pause_ack_reg <= 0;
        end else begin
            if (m_pause_req) begin
                m_pause_ack_reg <= 1;
            end else begin
                m_pause_ack_reg <= 0;
            end
        end
    end

    // Status outputs
    assign s_axis_tready = s_axis_tready_reg;
    assign m_axis_tvalid = m_axis_tvalid_reg;
    assign s_pause_ack = s_pause_ack_reg;
    assign m_pause_ack = m_pause_ack_reg;
    assign s_status_depth = s_status_depth_reg;
    assign s_status_depth_commit = s_status_depth_commit_reg;
    assign s_status_overflow = s_status_overflow_reg;
    assign s_status_bad_frame = s_status_bad_frame_reg;
    assign s_status_good_frame = s_status_good_frame_reg;
    assign m_status_depth = m_status_depth_reg;
    assign m_status_depth_commit = m_status_depth_commit_reg;
    assign m_status_overflow = m_status_overflow_reg;
    assign m_status_bad_frame = m_status_bad_frame_reg;
    assign m_status_good_frame = m_status_good_frame_reg;

endmodule
