module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   s_pause_req,
    output wire                   s_pause_ack,
    input  wire                   m_pause_req,
    output wire                   m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] s_status_depth,
    output wire [$clog2(DEPTH):0] s_status_depth_commit,
    output wire                   s_status_overflow,
    output wire                   s_status_bad_frame,
    output wire                   s_status_good_frame,
    output wire [$clog2(DEPTH):0] m_status_depth,
    output wire [$clog2(DEPTH):0] m_status_depth_commit,
    output wire                   m_status_overflow,
    output wire                   m_status_bad_frame,
    output wire                   m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);

    // FIFO signals
    reg [ADDR_WIDTH:0] s_wr_ptr, s_wr_ptr_gray, s_wr_ptr_gray_sync, s_wr_ptr_sync;
    reg [ADDR_WIDTH:0] m_rd_ptr, m_rd_ptr_gray, m_rd_ptr_gray_sync, m_rd_ptr_sync;
    reg [ADDR_WIDTH:0] s_depth, m_depth;

    // FIFO memory
    reg [DATA_WIDTH-1:0] mem_data [DEPTH-1:0];
    reg [KEEP_WIDTH-1:0] mem_keep [DEPTH-1:0];
    reg [LAST_ENABLE:0]  mem_last [DEPTH-1:0];
    reg [ID_ENABLE:0]    mem_id [DEPTH-1:0];
    reg [DEST_ENABLE:0]  mem_dest [DEPTH-1:0];
    reg [USER_ENABLE:0]  mem_user [DEPTH-1:0];

    // Flags
    reg s_full, s_empty, m_full, m_empty;
    reg s_overflow, s_bad_frame, s_good_frame;
    reg m_overflow, m_bad_frame, m_good_frame;

    // Pause control
    reg s_pause, m_pause;

    // Status signals
    reg [$clog2(DEPTH):0] s_depth_commit, m_depth_commit;

    // Gray code conversion
    function [ADDR_WIDTH:0] bin2gray;
        input [ADDR_WIDTH:0] bin;
        bin2gray = bin ^ (bin >> 1);
    endfunction

    // Gray code conversion
    function [ADDR_WIDTH:0] gray2bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = 1; i <= ADDR_WIDTH; i = i + 1) begin
            bin = bin ^ (gray >> i);
        end
        gray2bin = bin;
    endfunction

    // Synchronous FIFO write logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= 0;
            s_wr_ptr_gray <= 0;
            s_wr_ptr_gray_sync <= 0;
            s_wr_ptr_sync <= 0;
            s_depth <= 0;
            s_depth_commit <= 0;
            s_full <= 0;
            s_empty <= 1;
            s_overflow <= 0;
            s_bad_frame <= 0;
            s_good_frame <= 0;
        end else begin
            s_wr_ptr_gray <= bin2gray(s_wr_ptr + s_axis_tvalid & ~s_full & ~s_pause);
            s_wr_ptr <= s_wr_ptr + s_axis_tvalid & ~s_full & ~s_pause;

            if (s_axis_tvalid & ~s_full & ~s_pause) begin
                mem_data[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tdata;
                mem_keep[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tkeep;
                mem_last[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tlast;
                mem_id[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tid;
                mem_dest[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tdest;
                mem_user[s_wr_ptr[ADDR_WIDTH-1:0]] <= s_axis_tuser;
            end

            s_depth <= s_depth + s_axis_tvalid & ~s_full & ~s_pause - m_axis_tready & ~m_empty & ~m_pause;
            s_depth_commit <= s_depth_commit + s_axis_tvalid & ~s_full & ~s_pause & s_axis_tlast - m_axis_tready & ~m_empty & ~m_pause & m_axis_tlast;

            s_full <= (s_wr_ptr_gray_sync == {1'b1, {ADDR_WIDTH{1'b0}}});
            s_empty <= (s_wr_ptr_gray_sync == s_rd_ptr_gray_sync);

            if (s_axis_tvalid & s_full & ~s_pause) begin
                s_overflow <= 1;
            end else if (s_axis_tready) begin
                s_overflow <= 0;
            end

            if (s_axis_tvalid & s_axis_tlast & ~s_pause) begin
                if (s_axis_tuser & USER_BAD_FRAME_MASK == USER_BAD_FRAME_VALUE) begin
                    s_bad_frame <= 1;
                end else begin
                    s_good_frame <= 1;
                end
            end else if (s_axis_tready) begin
                s_bad_frame <= 0;
                s_good_frame <= 0;
            end
        end
    end

    // Synchronous FIFO read logic
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr <= 0;
            m_rd_ptr_gray <= 0;
            m_rd_ptr_gray_sync <= 0;
            m_rd_ptr_sync <= 0;
            m_depth <= 0;
            m_depth_commit <= 0;
            m_full <= 0;
            m_empty <= 1;
            m_overflow <= 0;
            m_bad_frame <= 0;
            m_good_frame <= 0;
        end else begin
            m_rd_ptr_gray <= bin2gray(m_rd_ptr + m_axis_tready & ~m_empty & ~m_pause);
            m_rd_ptr <= m_rd_ptr + m_axis_tready & ~m_empty & ~m_pause;

            m_depth <= m_depth + s_axis_tvalid & ~s_full & ~s_pause - m_axis_tready & ~m_empty & ~m_pause;
            m_depth_commit <= m_depth_commit + s_axis_tvalid & ~s_full & ~s_pause & s_axis_tlast - m_axis_tready & ~m_empty & ~m_pause & m_axis_tlast;

            m_full <= (m_wr_ptr_gray_sync == {1'b1, {ADDR_WIDTH{1'b0}}});
            m_empty <= (m_wr_ptr_gray_sync == m_rd_ptr_gray_sync);

            if (m_axis_tvalid & m_full & ~m_pause) begin
                m_overflow <= 1;
            end else if (m_axis_tready) begin
                m_overflow <= 0;
            end

            if (m_axis_tvalid & m_axis_tlast & ~m_pause) begin
                if (m_axis_tuser & USER_BAD_FRAME_MASK == USER_BAD_FRAME_VALUE) begin
                    m_bad_frame <= 1;
                end else begin
                    m_good_frame <= 1;
                end
            end else if (m_axis_tready) begin
                m_bad_frame <= 0;
                m_good_frame <= 0;
            end
        end
    end

    // Cross-domain synchronization for write pointer
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_wr_ptr_gray_sync <= 0;
            s_wr_ptr_sync <= 0;
        end else begin
            s_wr_ptr_gray_sync <= s_wr_ptr_gray;
            s_wr_ptr_sync <= gray2bin(s_wr_ptr_gray_sync);
        end
    end

    // Cross-domain synchronization for read pointer
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_rd_ptr_gray_sync <= 0;
            m_rd_ptr_sync <= 0;
        end else begin
            m_rd_ptr_gray_sync <= m_rd_ptr_gray;
            m_rd_ptr_sync <= gray2bin(m_rd_ptr_gray_sync);
        end
    end

    // AXI-Stream output signals
    assign m_axis_tdata = mem_data[m_rd_ptr_sync[ADDR_WIDTH-1:0]];
    assign m_axis_tkeep = mem_keep[m_rd_ptr_sync[ADDR_WIDTH-1:0]];
    assign m_axis_tlast = mem_last[m_rd_ptr_sync[ADDR_WIDTH-1:0]];
    assign m_axis_tid = mem_id[m_rd_ptr_sync[ADDR_WIDTH-1:0]];
    assign m_axis_tdest = mem_dest[m_rd_ptr_sync[ADDR_WIDTH-1:0]];
    assign m_axis_tuser = mem_user[m_rd_ptr_sync[ADDR_WIDTH-1:0]];
    assign m_axis_tvalid = ~m_empty & ~m_pause;

    // AXI-Stream input signals
    assign s_axis_tready = ~s_full & ~s_pause;

    // Pause control
    assign s_pause = PAUSE_ENABLE ? s_pause_req : 0;
    assign m_pause = PAUSE_ENABLE ? m_pause_req : 0;
    assign s_pause_ack = s_pause;
    assign m_pause_ack = m_pause;

    // Status signals
    assign s_status_depth = s_depth;
    assign s_status_depth_commit = s_depth_commit;
    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;
    assign m_status_depth = m_depth;
    assign m_status_depth_commit = m_depth_commit;
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;

endmodule
