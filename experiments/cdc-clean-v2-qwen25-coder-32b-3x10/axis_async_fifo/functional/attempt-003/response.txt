module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   s_pause_req,
    output wire                   s_pause_ack,
    input  wire                   m_pause_req,
    output wire                   m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] s_status_depth,
    output wire [$clog2(DEPTH):0] s_status_depth_commit,
    output wire                   s_status_overflow,
    output wire                   s_status_bad_frame,
    output wire                   s_status_good_frame,
    output wire [$clog2(DEPTH):0] m_status_depth,
    output wire [$clog2(DEPTH):0] m_status_depth_commit,
    output wire                   m_status_overflow,
    output wire                   m_status_bad_frame,
    output wire                   m_status_good_frame
);

    localparam ADDR_WIDTH = $clog2(DEPTH);

    reg [ADDR_WIDTH:0] wptr_s, wptr_m;
    reg [ADDR_WIDTH:0] rptr_s, rptr_m;
    reg [ADDR_WIDTH:0] wptr_gray_s, wptr_gray_m;
    reg [ADDR_WIDTH:0] rptr_gray_s, rptr_gray_m;

    reg [DATA_WIDTH-1:0] mem_data [DEPTH-1:0];
    reg [KEEP_WIDTH-1:0] mem_keep [DEPTH-1:0];
    reg [LAST_ENABLE:0]  mem_last [DEPTH-1:0];
    reg [ID_ENABLE:0]    mem_id [DEPTH-1:0];
    reg [DEST_ENABLE:0]  mem_dest [DEPTH-1:0];
    reg [USER_ENABLE:0]  mem_user [DEPTH-1:0];

    reg [ADDR_WIDTH:0] depth_s, depth_m;
    reg [ADDR_WIDTH:0] depth_commit_s, depth_commit_m;
    reg overflow_s, overflow_m;
    reg bad_frame_s, bad_frame_m;
    reg good_frame_s, good_frame_m;

    reg pause_s, pause_m;
    reg pause_ack_s, pause_ack_m;

    // Gray code conversion
    function [ADDR_WIDTH:0] bin2gray;
        input [ADDR_WIDTH:0] bin;
        bin2gray = bin ^ (bin >> 1);
    endfunction

    // Binary code conversion
    function [ADDR_WIDTH:0] gray2bin;
        input [ADDR_WIDTH:0] gray;
        integer i;
        reg [ADDR_WIDTH:0] bin;
        bin = gray;
        for (i = ADDR_WIDTH-1; i >= 0; i = i - 1) begin
            bin[i] = bin[i] ^ bin[i+1];
        end
        gray2bin = bin;
    endfunction

    // Write pointer logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            wptr_s <= 0;
            wptr_gray_s <= 0;
        end else if (s_axis_tvalid && s_axis_tready) begin
            wptr_s <= wptr_s + 1;
            wptr_gray_s <= bin2gray(wptr_s + 1);
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            wptr_m <= 0;
            wptr_gray_m <= 0;
        end else if (m_axis_tvalid && m_axis_tready) begin
            wptr_m <= wptr_m + 1;
            wptr_gray_m <= bin2gray(wptr_m + 1);
        end
    end

    // Read pointer logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            rptr_s <= 0;
            rptr_gray_s <= 0;
        end else if (m_axis_tvalid && m_axis_tready) begin
            rptr_s <= rptr_s + 1;
            rptr_gray_s <= bin2gray(rptr_s + 1);
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            rptr_m <= 0;
            rptr_gray_m <= 0;
        end else if (s_axis_tvalid && s_axis_tready) begin
            rptr_m <= rptr_m + 1;
            rptr_gray_m <= bin2gray(rptr_m + 1);
        end
    end

    // Memory write logic
    always @(posedge s_clk) begin
        if (s_axis_tvalid && s_axis_tready) begin
            mem_data[gray2bin(wptr_gray_s)] <= s_axis_tdata;
            mem_keep[gray2bin(wptr_gray_s)] <= s_axis_tkeep;
            mem_last[gray2bin(wptr_gray_s)] <= s_axis_tlast;
            mem_id[gray2bin(wptr_gray_s)] <= s_axis_tid;
            mem_dest[gray2bin(wptr_gray_s)] <= s_axis_tdest;
            mem_user[gray2bin(wptr_gray_s)] <= s_axis_tuser;
        end
    end

    // Memory read logic
    always @(posedge m_clk) begin
        if (m_axis_tvalid && m_axis_tready) begin
            m_axis_tdata <= mem_data[gray2bin(rptr_gray_m)];
            m_axis_tkeep <= mem_keep[gray2bin(rptr_gray_m)];
            m_axis_tlast <= mem_last[gray2bin(rptr_gray_m)];
            m_axis_tid <= mem_id[gray2bin(rptr_gray_m)];
            m_axis_tdest <= mem_dest[gray2bin(rptr_gray_m)];
            m_axis_tuser <= mem_user[gray2bin(rptr_gray_m)];
        end
    end

    // Depth calculation
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            depth_s <= 0;
            depth_commit_s <= 0;
            overflow_s <= 0;
            bad_frame_s <= 0;
            good_frame_s <= 0;
        end else begin
            depth_s <= wptr_s - rptr_s;
            depth_commit_s <= depth_s;
            overflow_s <= (depth_s >= DEPTH);
            bad_frame_s <= (USER_ENABLE && (mem_user[gray2bin(wptr_gray_s)] & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE);
            good_frame_s <= (USER_ENABLE && (mem_user[gray2bin(wptr_gray_s)] & USER_BAD_FRAME_MASK) != USER_BAD_FRAME_VALUE);
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            depth_m <= 0;
            depth_commit_m <= 0;
            overflow_m <= 0;
            bad_frame_m <= 0;
            good_frame_m <= 0;
        end else begin
            depth_m <= wptr_m - rptr_m;
            depth_commit_m <= depth_m;
            overflow_m <= (depth_m >= DEPTH);
            bad_frame_m <= (USER_ENABLE && (mem_user[gray2bin(wptr_gray_m)] & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE);
            good_frame_m <= (USER_ENABLE && (mem_user[gray2bin(wptr_gray_m)] & USER_BAD_FRAME_MASK) != USER_BAD_FRAME_VALUE);
        end
    end

    // Handshake logic
    assign s_axis_tready = (!overflow_s && !pause_s) || MARK_WHEN_FULL;
    assign m_axis_tvalid = (depth_m > 0) && !pause_m;

    // Pause logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            pause_s <= 0;
            pause_ack_s <= 0;
        end else begin
            pause_s <= s_pause_req;
            pause_ack_s <= s_pause_req;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            pause_m <= 0;
            pause_ack_m <= 0;
        end else begin
            pause_m <= m_pause_req;
            pause_ack_m <= m_pause_req;
        end
    end

    // Status outputs
    assign s_status_depth = depth_s;
    assign s_status_depth_commit = depth_commit_s;
    assign s_status_overflow = overflow_s;
    assign s_status_bad_frame = bad_frame_s;
    assign s_status_good_frame = good_frame_s;

    assign m_status_depth = depth_m;
    assign m_status_depth_commit = depth_commit_m;
    assign m_status_overflow = overflow_m;
    assign m_status_bad_frame = bad_frame_m;
    assign m_status_good_frame = good_frame_m;

    // Pause ack outputs
    assign s_pause_ack = pause_ack_s;
    assign m_pause_ack = pause_ack_m;

endmodule
