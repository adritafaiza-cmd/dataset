module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    // Internal signals
    reg [$clog2(DEPTH):0] s_wr_ptr, s_wr_ptr_gray, s_wr_ptr_gray_sync, s_wr_ptr_gray_sync2;
    reg [$clog2(DEPTH):0] s_rd_ptr, s_rd_ptr_gray, s_rd_ptr_gray_sync, s_rd_ptr_gray_sync2;
    reg [$clog2(DEPTH):0] m_wr_ptr, m_wr_ptr_gray, m_wr_ptr_gray_sync, m_wr_ptr_gray_sync2;
    reg [$clog2(DEPTH):0] m_rd_ptr, m_rd_ptr_gray, m_rd_ptr_gray_sync, m_rd_ptr_gray_sync2;

    reg [S_DATA_WIDTH-1:0] s_fifo [DEPTH-1:0];
    reg [S_KEEP_WIDTH-1:0] s_keep_fifo [DEPTH-1:0];
    reg [ID_WIDTH-1:0] s_tid_fifo [DEPTH-1:0];
    reg [DEST_WIDTH-1:0] s_tdest_fifo [DEPTH-1:0];
    reg [USER_WIDTH-1:0] s_tuser_fifo [DEPTH-1:0];
    reg s_tlast_fifo [DEPTH-1:0];

    reg [M_DATA_WIDTH-1:0] m_fifo [DEPTH-1:0];
    reg [M_KEEP_WIDTH-1:0] m_keep_fifo [DEPTH-1:0];
    reg [ID_WIDTH-1:0] m_tid_fifo [DEPTH-1:0];
    reg [DEST_WIDTH-1:0] m_tdest_fifo [DEPTH-1:0];
    reg [USER_WIDTH-1:0] m_tuser_fifo [DEPTH-1:0];
    reg m_tlast_fifo [DEPTH-1:0];

    reg s_axis_tready_int;
    reg m_axis_tvalid_int;
    reg s_pause_ack_int;
    reg m_pause_ack_int;

    reg [$clog2(DEPTH):0] s_depth, s_depth_commit;
    reg [$clog2(DEPTH):0] m_depth, m_depth_commit;
    reg s_overflow, s_bad_frame, s_good_frame;
    reg m_overflow, m_bad_frame, m_good_frame;

    // Binary to Gray conversion
    function [$clog2(DEPTH):0] bin_to_gray;
        input [$clog2(DEPTH):0] bin;
        bin_to_gray = bin ^ (bin >> 1);
    endfunction

    // Gray to Binary conversion
    function [$clog2(DEPTH):0] gray_to_bin;
        input [$clog2(DEPTH):0] gray;
        integer i;
        reg [$clog2(DEPTH):0] bin;
        bin = gray;
        for (i = $clog2(DEPTH) - 1; i >= 0; i = i - 1)
            bin[i] = bin[i+1] ^ gray[i];
        gray_to_bin = bin;
    endfunction

    // Source clock domain
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= 0;
            s_wr_ptr_gray <= 0;
            s_wr_ptr_gray_sync <= 0;
            s_wr_ptr_gray_sync2 <= 0;
            s_axis_tready_int <= 0;
            s_depth <= 0;
            s_depth_commit <= 0;
            s_overflow <= 0;
            s_bad_frame <= 0;
            s_good_frame <= 0;
        end else begin
            s_wr_ptr_gray <= bin_to_gray(s_wr_ptr);
            s_wr_ptr_gray_sync <= s_wr_ptr_gray_sync2;
            s_wr_ptr_gray_sync2 <= s_wr_ptr_gray;

            if (s_axis_tvalid && s_axis_tready_int) begin
                s_fifo[s_wr_ptr] <= s_axis_tdata;
                s_keep_fifo[s_wr_ptr] <= s_axis_tkeep;
                s_tid_fifo[s_wr_ptr] <= s_axis_tid;
                s_tdest_fifo[s_wr_ptr] <= s_axis_tdest;
                s_tuser_fifo[s_wr_ptr] <= s_axis_tuser;
                s_tlast_fifo[s_wr_ptr] <= s_axis_tlast;
                s_wr_ptr <= s_wr_ptr + 1;
                s_depth <= s_depth + 1;
                if (s_axis_tlast) begin
                    s_depth_commit <= s_depth_commit + 1;
                    if (s_axis_tuser & USER_BAD_FRAME_MASK == USER_BAD_FRAME_VALUE)
                        s_bad_frame <= 1;
                    else
                        s_good_frame <= 1;
                end
            end

            if (s_depth == DEPTH)
                s_overflow <= 1;

            if (s_axis_tlast)
                s_axis_tready_int <= 1;
            else if (s_depth == DEPTH - 1)
                s_axis_tready_int <= 0;
            else
                s_axis_tready_int <= 1;

            if (s_good_frame && s_axis_tlast)
                s_good_frame <= 0;

            if (s_bad_frame && s_axis_tlast)
                s_bad_frame <= 0;

            if (s_overflow && s_axis_tlast)
                s_overflow <= 0;
        end
    end

    // Source to Master CDC
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_rd_ptr_gray_sync <= 0;
            s_rd_ptr_gray_sync2 <= 0;
        end else begin
            s_rd_ptr_gray_sync <= s_wr_ptr_gray_sync;
            s_rd_ptr_gray_sync2 <= s_wr_ptr_gray_sync2;
        end
    end

    assign s_rd_ptr_gray = s_rd_ptr_gray_sync2;
    assign s_rd_ptr = gray_to_bin(s_rd_ptr_gray);

    // Master clock domain
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_wr_ptr <= 0;
            m_wr_ptr_gray <= 0;
            m_wr_ptr_gray_sync <= 0;
            m_wr_ptr_gray_sync2 <= 0;
            m_axis_tvalid_int <= 0;
            m_depth <= 0;
            m_depth_commit <= 0;
            m_overflow <= 0;
            m_bad_frame <= 0;
            m_good_frame <= 0;
        end else begin
            m_wr_ptr_gray <= bin_to_gray(m_wr_ptr);
            m_wr_ptr_gray_sync <= m_wr_ptr_gray_sync2;
            m_wr_ptr_gray_sync2 <= m_wr_ptr_gray;

            if (m_axis_tvalid_int && m_axis_tready) begin
                m_wr_ptr <= m_wr_ptr + 1;
                m_depth <= m_depth - 1;
                if (m_tlast_fifo[m_rd_ptr])
                    m_depth_commit <= m_depth_commit - 1;
            end

            if (s_depth > m_depth && m_axis_tready) begin
                m_fifo[m_wr_ptr] <= s_fifo[s_rd_ptr];
                m_keep_fifo[m_wr_ptr] <= s_keep_fifo[s_rd_ptr];
                m_tid_fifo[m_wr_ptr] <= s_tid_fifo[s_rd_ptr];
                m_tdest_fifo[m_wr_ptr] <= s_tdest_fifo[s_rd_ptr];
                m_tuser_fifo[m_wr_ptr] <= s_tuser_fifo[s_rd_ptr];
                m_tlast_fifo[m_wr_ptr] <= s_tlast_fifo[s_rd_ptr];
                m_wr_ptr <= m_wr_ptr + 1;
                m_depth <= m_depth + 1;
                if (s_tlast_fifo[s_rd_ptr]) begin
                    m_depth_commit <= m_depth_commit + 1;
                    if (s_tuser_fifo[s_rd_ptr] & USER_BAD_FRAME_MASK == USER_BAD_FRAME_VALUE)
                        m_bad_frame <= 1;
                    else
                        m_good_frame <= 1;
                end
                s_rd_ptr <= s_rd_ptr + 1;
            end

            if (m_depth == DEPTH)
                m_overflow <= 1;

            if (m_depth > 0)
                m_axis_tvalid_int <= 1;
            else
                m_axis_tvalid_int <= 0;

            if (m_good_frame && m_tlast_fifo[m_rd_ptr])
                m_good_frame <= 0;

            if (m_bad_frame && m_tlast_fifo[m_rd_ptr])
                m_bad_frame <= 0;

            if (m_overflow && m_tlast_fifo[m_rd_ptr])
                m_overflow <= 0;
        end
    end

    // Master to Source CDC
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_rd_ptr_gray_sync <= 0;
            m_rd_ptr_gray_sync2 <= 0;
        end else begin
            m_rd_ptr_gray_sync <= m_wr_ptr_gray_sync;
            m_rd_ptr_gray_sync2 <= m_wr_ptr_gray_sync2;
        end
    end

    assign m_rd_ptr_gray = m_rd_ptr_gray_sync2;
    assign m_rd_ptr = gray_to_bin(m_rd_ptr_gray);

    // Width conversion
    reg [M_DATA_WIDTH-1:0] m_axis_tdata_int;
    reg [M_KEEP_WIDTH-1:0] m_axis_tkeep_int;
    reg m_axis_tlast_int;
    reg [ID_WIDTH-1:0] m_axis_tid_int;
    reg [DEST_WIDTH-1:0] m_axis_tdest_int;
    reg [USER_WIDTH-1:0] m_axis_tuser_int;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_axis_tdata_int <= 0;
            m_axis_tkeep_int <= 0;
            m_axis_tlast_int <= 0;
            m_axis_tid_int <= 0;
            m_axis_tdest_int <= 0;
            m_axis_tuser_int <= 0;
        end else begin
            if (m_axis_tvalid_int && m_axis_tready) begin
                m_axis_tdata_int <= m_fifo[m_rd_ptr];
                m_axis_tkeep_int <= m_keep_fifo[m_rd_ptr];
                m_axis_tlast_int <= m_tlast_fifo[m_rd_ptr];
                m_axis_tid_int <= m_tid_fifo[m_rd_ptr];
                m_axis_tdest_int <= m_tdest_fifo[m_rd_ptr];
                m_axis_tuser_int <= m_tuser_fifo[m_rd_ptr];
            end
        end
    end

    // Pause signals
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst)
            s_pause_ack_int <= 0;
        else
            s_pause_ack_int <= s_pause_req;
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst)
            m_pause_ack_int <= 0;
        else
            m_pause_ack_int <= m_pause_req;
    end

    // Status signals
    assign s_status_depth = s_depth;
    assign s_status_depth_commit = s_depth_commit;
    assign s_status_overflow = s_overflow;
    assign s_status_bad_frame = s_bad_frame;
    assign s_status_good_frame = s_good_frame;

    assign m_status_depth = m_depth;
    assign m_status_depth_commit = m_depth_commit;
    assign m_status_overflow = m_overflow;
    assign m_status_bad_frame = m_bad_frame;
    assign m_status_good_frame = m_good_frame;

    // Output assignments
    assign s_axis_tready = s_axis_tready_int;
    assign m_axis_tdata = m_axis_tdata_int;
    assign m_axis_tkeep = m_axis_tkeep_int;
    assign m_axis_tvalid = m_axis_tvalid_int;
    assign m_axis_tlast = m_axis_tlast_int;
    assign m_axis_tid = m_axis_tid_int;
    assign m_axis_tdest = m_axis_tdest_int;
    assign m_axis_tuser = m_axis_tuser_int;
    assign s_pause_ack = s_pause_ack_int;
    assign m_pause_ack = m_pause_ack_int;

endmodule
