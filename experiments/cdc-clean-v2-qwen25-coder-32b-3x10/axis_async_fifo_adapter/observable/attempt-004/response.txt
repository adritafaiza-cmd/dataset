module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    // Internal signals
    reg [S_DATA_WIDTH-1:0] s_axis_tdata_reg;
    reg [S_KEEP_WIDTH-1:0] s_axis_tkeep_reg;
    reg                    s_axis_tvalid_reg;
    reg                    s_axis_tlast_reg;
    reg [ID_WIDTH-1:0]     s_axis_tid_reg;
    reg [DEST_WIDTH-1:0]   s_axis_tdest_reg;
    reg [USER_WIDTH-1:0]   s_axis_tuser_reg;

    reg [M_DATA_WIDTH-1:0] m_axis_tdata_reg;
    reg [M_KEEP_WIDTH-1:0] m_axis_tkeep_reg;
    reg                    m_axis_tvalid_reg;
    reg                    m_axis_tlast_reg;
    reg [ID_WIDTH-1:0]     m_axis_tid_reg;
    reg [DEST_WIDTH-1:0]   m_axis_tdest_reg;
    reg [USER_WIDTH-1:0]   m_axis_tuser_reg;

    reg [$clog2(DEPTH):0]  s_wr_ptr;
    reg [$clog2(DEPTH):0]  s_wr_ptr_gray;
    reg [$clog2(DEPTH):0]  m_rd_ptr;
    reg [$clog2(DEPTH):0]  m_rd_ptr_gray;

    reg [$clog2(DEPTH):0]  s_depth;
    reg [$clog2(DEPTH):0]  m_depth;

    reg                    s_pause_ack_reg;
    reg                    m_pause_ack_reg;

    reg                    s_status_overflow_reg;
    reg                    s_status_bad_frame_reg;
    reg                    s_status_good_frame_reg;
    reg                    m_status_overflow_reg;
    reg                    m_status_bad_frame_reg;
    reg                    m_status_good_frame_reg;

    reg [$clog2(DEPTH):0]  s_depth_commit;
    reg [$clog2(DEPTH):0]  m_depth_commit;

    // FIFO memory
    reg [S_DATA_WIDTH-1:0] fifo_mem [DEPTH-1:0];
    reg [S_KEEP_WIDTH-1:0] fifo_keep [DEPTH-1:0];
    reg                    fifo_last [DEPTH-1:0];
    reg [ID_WIDTH-1:0]     fifo_tid [DEPTH-1:0];
    reg [DEST_WIDTH-1:0]   fifo_tdest [DEPTH-1:0];
    reg [USER_WIDTH-1:0]   fifo_tuser [DEPTH-1:0];

    // Gray code conversion
    function [$clog2(DEPTH):0] bin2gray;
        input [$clog2(DEPTH):0] bin;
        bin2gray = (bin >> 1) ^ bin;
    endfunction

    // Binary code conversion
    function [$clog2(DEPTH):0] gray2bin;
        input [$clog2(DEPTH):0] gray;
        integer i;
        reg [$clog2(DEPTH):0] bin;
        bin = gray;
        for (i = $clog2(DEPTH); i > 0; i = i - 1) begin
            bin[i-1] = bin[i] ^ bin[i-1];
        end
        gray2bin = bin;
    endfunction

    // Write side logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_wr_ptr <= 0;
            s_wr_ptr_gray <= 0;
            s_axis_tready <= 0;
            s_depth <= 0;
            s_depth_commit <= 0;
            s_status_overflow_reg <= 0;
            s_status_bad_frame_reg <= 0;
            s_status_good_frame_reg <= 0;
        end else begin
            reg [$clog2(DEPTH):0] next_wr_ptr;
            reg [$clog2(DEPTH):0] next_wr_ptr_gray;
            reg [$clog2(DEPTH):0] next_depth;
            reg [$clog2(DEPTH):0] next_depth_commit;
            reg s_axis_tready_next;

            next_wr_ptr = s_wr_ptr + 1;
            next_wr_ptr_gray = bin2gray(next_wr_ptr);
            next_depth = s_depth + (s_axis_tvalid & s_axis_tready);
            next_depth_commit = s_depth_commit + (s_axis_tvalid & s_axis_tready & s_axis_tlast);
            s_axis_tready_next = (next_depth < DEPTH) & ~s_pause_req;

            if (s_axis_tvalid & s_axis_tready) begin
                fifo_mem[s_wr_ptr] <= s_axis_tdata;
                fifo_keep[s_wr_ptr] <= s_axis_tkeep;
                fifo_last[s_wr_ptr] <= s_axis_tlast;
                fifo_tid[s_wr_ptr] <= s_axis_tid;
                fifo_tdest[s_wr_ptr] <= s_axis_tdest;
                fifo_tuser[s_wr_ptr] <= s_axis_tuser;
            end

            if (next_depth >= DEPTH) begin
                s_status_overflow_reg <= 1;
            end else begin
                s_status_overflow_reg <= 0;
            end

            if (s_axis_tvalid & s_axis_tready & s_axis_tlast) begin
                if (DROP_BAD_FRAME && (s_axis_tuser & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                    s_status_bad_frame_reg <= 1;
                end else begin
                    s_status_good_frame_reg <= 1;
                end
            end else begin
                s_status_bad_frame_reg <= 0;
                s_status_good_frame_reg <= 0;
            end

            s_wr_ptr <= next_wr_ptr;
            s_wr_ptr_gray <= next_wr_ptr_gray;
            s_axis_tready <= s_axis_tready_next;
            s_depth <= next_depth;
            s_depth_commit <= next_depth_commit;
        end
    end

    // Read side logic
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rd_ptr <= 0;
            m_rd_ptr_gray <= 0;
            m_axis_tvalid_reg <= 0;
            m_axis_tlast_reg <= 0;
            m_depth <= 0;
            m_depth_commit <= 0;
            m_status_overflow_reg <= 0;
            m_status_bad_frame_reg <= 0;
            m_status_good_frame_reg <= 0;
        end else begin
            reg [$clog2(DEPTH):0] next_rd_ptr;
            reg [$clog2(DEPTH):0] next_rd_ptr_gray;
            reg [$clog2(DEPTH):0] next_depth;
            reg [$clog2(DEPTH):0] next_depth_commit;
            reg m_axis_tvalid_next;

            next_rd_ptr = m_rd_ptr + (m_axis_tvalid_reg & m_axis_tready);
            next_rd_ptr_gray = bin2gray(next_rd_ptr);
            next_depth = m_depth - (m_axis_tvalid_reg & m_axis_tready);
            next_depth_commit = m_depth_commit - (m_axis_tvalid_reg & m_axis_tready & m_axis_tlast_reg);
            m_axis_tvalid_next = (m_depth > 0) & ~m_pause_req;

            if (m_axis_tvalid_reg & m_axis_tready) begin
                if (m_axis_tlast_reg) begin
                    if (DROP_BAD_FRAME && (m_axis_tuser_reg & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE) begin
                        m_status_bad_frame_reg <= 1;
                    end else begin
                        m_status_good_frame_reg <= 1;
                    end
                end else begin
                    m_status_bad_frame_reg <= 0;
                    m_status_good_frame_reg <= 0;
                end
            end

            if (m_axis_tvalid_reg & m_axis_tready) begin
                m_axis_tdata_reg <= fifo_mem[m_rd_ptr];
                m_axis_tkeep_reg <= fifo_keep[m_rd_ptr];
                m_axis_tlast_reg <= fifo_last[m_rd_ptr];
                m_axis_tid_reg <= fifo_tid[m_rd_ptr];
                m_axis_tdest_reg <= fifo_tdest[m_rd_ptr];
                m_axis_tuser_reg <= fifo_tuser[m_rd_ptr];
            end

            m_rd_ptr <= next_rd_ptr;
            m_rd_ptr_gray <= next_rd_ptr_gray;
            m_axis_tvalid_reg <= m_axis_tvalid_next;
            m_depth <= next_depth;
            m_depth_commit <= next_depth_commit;
        end
    end

    // Cross-domain synchronization
    reg [$clog2(DEPTH):0] s_wr_ptr_gray_sync;
    reg [$clog2(DEPTH):0] m_rd_ptr_gray_sync;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            s_wr_ptr_gray_sync <= 0;
        end else begin
            s_wr_ptr_gray_sync <= s_wr_ptr_gray;
        end
    end

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            m_rd_ptr_gray_sync <= 0;
        end else begin
            m_rd_ptr_gray_sync <= m_rd_ptr_gray;
        end
    end

    // Depth calculation
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_depth <= 0;
        end else begin
            s_depth <= gray2bin(s_wr_ptr_gray) - gray2bin(m_rd_ptr_gray_sync);
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_depth <= 0;
        end else begin
            m_depth <= gray2bin(s_wr_ptr_gray_sync) - gray2bin(m_rd_ptr_gray);
        end
    end

    // Output assignments
    assign s_axis_tready = s_axis_tready_reg;
    assign m_axis_tdata = m_axis_tdata_reg;
    assign m_axis_tkeep = m_axis_tkeep_reg;
    assign m_axis_tvalid = m_axis_tvalid_reg;
    assign m_axis_tlast = m_axis_tlast_reg;
    assign m_axis_tid = m_axis_tid_reg;
    assign m_axis_tdest = m_axis_tdest_reg;
    assign m_axis_tuser = m_axis_tuser_reg;

    // Pause signals
    assign s_pause_ack = s_pause_ack_reg;
    assign m_pause_ack = m_pause_ack_reg;

    // Status signals
    assign s_status_depth = s_depth;
    assign s_status_depth_commit = s_depth_commit;
    assign s_status_overflow = s_status_overflow_reg;
    assign s_status_bad_frame = s_status_bad_frame_reg;
    assign s_status_good_frame = s_status_good_frame_reg;
    assign m_status_depth = m_depth;
    assign m_status_depth_commit = m_depth_commit;
    assign m_status_overflow = m_status_overflow_reg;
    assign m_status_bad_frame = m_status_bad_frame_reg;
    assign m_status_good_frame = m_status_good_frame_reg;

endmodule
