module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

    // Internal signals
    reg [N-1:0]        shift_reg;
    reg [N-1:0]        received_data;
    reg [3:0]          bit_cnt;
    reg                shift_en;
    reg                shift_done;
    reg                spi_ssel;
    reg                spi_sck;
    reg                spi_mosi;
    reg                di_req;
    reg                wr_ack;
    reg                do_valid;
    reg [N-1:0]        do_data;

    // Synchronizers for reset and control signals
    reg                rst_sclk_sync1;
    reg                rst_sclk_sync2;
    reg                rst_pclk_sync1;
    reg                rst_pclk_sync2;
    reg                wren_sclk_sync1;
    reg                wren_sclk_sync2;

    // Reset synchronization for sclk domain
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            rst_sclk_sync1 <= 1'b1;
            rst_sclk_sync2 <= 1'b1;
        end else begin
            rst_sclk_sync1 <= 1'b0;
            rst_sclk_sync2 <= rst_sclk_sync1;
        end
    end

    // Reset synchronization for pclk domain
    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            rst_pclk_sync1 <= 1'b1;
            rst_pclk_sync2 <= 1'b1;
        end else begin
            rst_pclk_sync1 <= 1'b0;
            rst_pclk_sync2 <= rst_pclk_sync1;
        end
    end

    // wren synchronization to sclk domain
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            wren_sclk_sync1 <= 1'b0;
            wren_sclk_sync2 <= 1'b0;
        end else begin
            wren_sclk_sync1 <= wren_i;
            wren_sclk_sync2 <= wren_sclk_sync1;
        end
    end

    // SPI clock generation
    reg [1:0] sclk_div_cnt;
    always @(posedge sclk_i or posedge rst_sclk_sync2) begin
        if (rst_sclk_sync2) begin
            sclk_div_cnt <= 2'b00;
            spi_sck <= 1'b0;
        end else begin
            if (sclk_div_cnt == SPI_2X_CLK_DIV - 1) begin
                sclk_div_cnt <= 2'b00;
                spi_sck <= ~spi_sck;
            end else begin
                sclk_div_cnt <= sclk_div_cnt + 1;
            end
        end
    end

    // SPI master state machine
    typedef enum reg [2:0] {
        IDLE,
        LOAD_DATA,
        SHIFT,
        STORE_DATA
    } state_t;

    state_t state, next_state;

    always @(posedge sclk_i or posedge rst_sclk_sync2) begin
        if (rst_sclk_sync2)
            state <= IDLE;
        else
            state <= next_state;
    end

    always @(*) begin
        next_state = state;
        shift_en = 1'b0;
        shift_done = 1'b0;
        di_req = 1'b0;
        wr_ack = 1'b0;
        do_valid = 1'b0;

        case (state)
            IDLE: begin
                if (wren_sclk_sync2) begin
                    next_state = LOAD_DATA;
                end
            end
            LOAD_DATA: begin
                next_state = SHIFT;
                shift_en = 1'b1;
                wr_ack = 1'b1;
            end
            SHIFT: begin
                if (bit_cnt == N - 1) begin
                    next_state = STORE_DATA;
                    shift_done = 1'b1;
                end else begin
                    shift_en = 1'b1;
                end
            end
            STORE_DATA: begin
                next_state = IDLE;
                do_valid = 1'b1;
                di_req = 1'b1;
            end
            default: begin
                next_state = IDLE;
            end
        endcase
    end

    // Bit counter
    always @(posedge sclk_i or posedge rst_sclk_sync2) begin
        if (rst_sclk_sync2)
            bit_cnt <= 4'b0000;
        else if (shift_en) begin
            if (shift_done)
                bit_cnt <= 4'b0000;
            else
                bit_cnt <= bit_cnt + 1;
        end
    end

    // Shift register for transmit and receive
    always @(posedge sclk_i or posedge rst_sclk_sync2) begin
        if (rst_sclk_sync2) begin
            shift_reg <= {N{1'b0}};
            received_data <= {N{1'b0}};
        end else begin
            if (shift_en) begin
                if (spi_sck) begin
                    shift_reg <= {shift_reg[N-2:0], spi_miso_i};
                    received_data <= {received_data[N-2:0], spi_mosi};
                end
            end else if (state == LOAD_DATA) begin
                shift_reg <= di_i;
            end
        end
    end

    // Output assignments
    always @(posedge pclk_i or posedge rst_pclk_sync2) begin
        if (rst_pclk_sync2) begin
            spi_ssel_o <= 1'b1;
            spi_sck_o <= 1'b0;
            spi_mosi_o <= 1'b0;
            di_req_o <= 1'b0;
            wr_ack_o <= 1'b0;
            do_valid_o <= 1'b0;
            do_o <= {N{1'b0}};
        end else begin
            spi_ssel_o <= ~spi_ssel;
            spi_sck_o <= spi_sck;
            spi_mosi_o <= shift_reg[N-1];
            di_req_o <= di_req;
            wr_ack_o <= wr_ack;
            do_valid_o <= do_valid;
            do_o <= received_data;
        end
    end

    // SPI select control
    always @(posedge sclk_i or posedge rst_sclk_sync2) begin
        if (rst_sclk_sync2)
            spi_ssel <= 1'b1;
        else begin
            if (state == LOAD_DATA)
                spi_ssel <= 1'b0;
            else if (state == IDLE)
                spi_ssel <= 1'b1;
        end
    end

endmodule
