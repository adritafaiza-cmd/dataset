module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

    // Internal signals
    reg [N-1:0]        di_sync;
    reg                wren_sync;
    reg [N-1:0]        shift_reg;
    reg [3:0]          bit_cnt;
    reg                spi_busy;
    reg                spi_start;
    reg [1:0]          sclk_div_cnt;
    reg [1:0]          pclk_div_cnt;

    // Synchronize di_i and wren_i to pclk_i domain
    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            di_sync <= 0;
            wren_sync <= 0;
        end else begin
            di_sync <= di_i;
            wren_sync <= wren_i;
        end
    end

    // SPI control logic
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            spi_ssel_o <= 1;
            spi_sck_o <= 0;
            spi_busy <= 0;
            spi_start <= 0;
            bit_cnt <= 0;
            shift_reg <= 0;
            sclk_div_cnt <= 0;
        end else begin
            if (!spi_busy && spi_start) begin
                spi_busy <= 1;
                spi_ssel_o <= 0;
                shift_reg <= di_sync;
                bit_cnt <= N-1;
                spi_sck_o <= 0;
                sclk_div_cnt <= 0;
            end else if (spi_busy) begin
                if (sclk_div_cnt == SPI_2X_CLK_DIV - 1) begin
                    spi_sck_o <= ~spi_sck_o;
                    if (spi_sck_o) begin
                        // Shift out on rising edge
                        spi_mosi_o <= shift_reg[N-1];
                        shift_reg <= {shift_reg[N-2:0], 1'b0};
                    end else begin
                        // Shift in on falling edge
                        do_o <= {do_o[N-2:0], spi_miso_i};
                    end
                    if (bit_cnt == 0) begin
                        spi_busy <= 0;
                        spi_ssel_o <= 1;
                    end else begin
                        bit_cnt <= bit_cnt - 1;
                    end
                end
                sclk_div_cnt <= sclk_div_cnt + 1;
            end
        end
    end

    // PCLK domain control logic
    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            di_req_o <= 1;
            wr_ack_o <= 0;
            do_valid_o <= 0;
            spi_start <= 0;
            pclk_div_cnt <= 0;
        end else begin
            if (!spi_busy && wren_sync) begin
                spi_start <= 1;
                wr_ack_o <= 1;
                di_req_o <= 0;
            end else begin
                spi_start <= 0;
                wr_ack_o <= 0;
            end

            if (!spi_busy && pclk_div_cnt == 1) begin
                di_req_o <= 1;
                do_valid_o <= 1;
            end else begin
                do_valid_o <= 0;
            end

            pclk_div_cnt <= pclk_div_cnt + 1;
        end
    end

endmodule
