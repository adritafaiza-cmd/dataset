module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

    reg [N-1:0] shift_reg;
    reg [3:0]   bit_cnt;
    reg         sclk_div_cnt;
    reg         sclk_edge;
    reg         pclk_edge;
    reg         start_transfer;
    reg         transfer_in_progress;

    // SCLK edge detection
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            sclk_div_cnt <= 0;
            sclk_edge <= 0;
        end else begin
            if (sclk_div_cnt == SPI_2X_CLK_DIV - 1) begin
                sclk_div_cnt <= 0;
                sclk_edge <= ~sclk_edge;
            end else begin
                sclk_div_cnt <= sclk_div_cnt + 1;
            end
        end
    end

    // PCLK edge detection
    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            pclk_edge <= 0;
        end else begin
            pclk_edge <= ~pclk_edge;
        end
    end

    // Control logic
    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            spi_ssel_o <= 0;
            spi_sck_o <= 0;
            di_req_o <= 1;
            wr_ack_o <= 0;
            do_valid_o <= 0;
            start_transfer <= 0;
            transfer_in_progress <= 0;
            bit_cnt <= 0;
        end else begin
            wr_ack_o <= 0;
            do_valid_o <= 0;

            if (wren_i && di_req_o) begin
                start_transfer <= 1;
                di_req_o <= 0;
            end else begin
                start_transfer <= 0;
            end

            if (start_transfer && !transfer_in_progress) begin
                spi_ssel_o <= 1;
                shift_reg <= di_i;
                bit_cnt <= N - 1;
                transfer_in_progress <= 1;
            end

            if (transfer_in_progress) begin
                if (sclk_edge) begin
                    spi_sck_o <= ~spi_sck_o;
                    if (spi_sck_o) begin
                        spi_mosi_o <= shift_reg[bit_cnt];
                        if (bit_cnt == 0) begin
                            transfer_in_progress <= 0;
                            spi_ssel_o <= 0;
                            do_o <= shift_reg;
                            do_valid_o <= 1;
                            di_req_o <= 1;
                        end else begin
                            bit_cnt <= bit_cnt - 1;
                        end
                    end
                end
            end

            if (start_transfer) begin
                wr_ack_o <= 1;
            end
        end
    end

endmodule
