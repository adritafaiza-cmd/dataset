module fifo #(
		parameter WIDTH = 8,
		parameter DEPTH = 16
	) (
		input	wire				clk,
		input	wire				rst,
		input	wire [WIDTH-1 : 0]		din,
		input	wire				wr_en,
		input	wire				rd_en,
		output	wire [WIDTH-1 : 0]		dout,
		output	wire				full,
		output	wire				empty,
		output	wire				valid
	);

	reg [$clog2(DEPTH)-1 : 0]	wr_ptr, rd_ptr;
	reg [$clog2(DEPTH)-1 : 0]	count;
	reg [WIDTH-1 : 0]		mem [0 : DEPTH-1];

	always @(posedge clk or posedge rst) begin
		if (rst) begin
			wr_ptr <= 0;
			rd_ptr <= 0;
			count <= 0;
		end else begin
			if (wr_en & !full) begin
				mem[wr_ptr] <= din;
				wr_ptr <= wr_ptr + 1;
				count <= count + 1;
			end
			if (rd_en & !empty) begin
				rd_ptr <= rd_ptr + 1;
				count <= count - 1;
			end
		end
	end

	assign dout = mem[rd_ptr];
	assign full = (count == DEPTH);
	assign empty = (count == 0);
	assign valid = !empty;

endmodule
