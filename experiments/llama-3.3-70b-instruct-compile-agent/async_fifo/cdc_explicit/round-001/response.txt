```verilog
module async_fifo #(
  parameter DSIZE = 8,
  parameter ASIZE = 4,
  parameter FALLTHROUGH = "TRUE"
)(
  input  wire             wclk,
  input  wire             wrst_n,
  input  wire             winc,
  input  wire [DSIZE-1:0] wdata,
  output wire             wfull,
  output wire             awfull,

  input  wire             rclk,
  input  wire             rrst_n,
  input  wire             rinc,
  output wire [DSIZE-1:0] rdata,
  output wire             rempty,
  output wire             arempty
);

  reg [ASIZE-1:0] wptr, rptr;
  reg [ASIZE-1:0] wptr_gray, rptr_gray;
  reg [ASIZE-1:0] wptr_sync, rptr_sync;
  reg [ASIZE-1:0] wptr_sync_gray, rptr_sync_gray;
  reg [DSIZE-1:0] mem [2**ASIZE-1:0];
  reg [DSIZE-1:0] rdata_reg;

  assign wfull = (wptr_gray == {~rptr_sync_gray[ASIZE-2:0], rptr_sync_gray[ASIZE-1]} && wptr[ASIZE-1] == rptr_sync[ASIZE-1]);
  assign awfull = (wptr_gray == {~rptr_sync_gray[ASIZE-2:0], rptr_sync_gray[ASIZE-1]} && wptr[ASIZE-1] != rptr_sync[ASIZE-1]);
  assign rempty = (rptr_gray == wptr_sync_gray);
  assign arempty = (rptr_gray == {~wptr_sync_gray[ASIZE-2:0], wptr_sync_gray[ASIZE-1]});

  always @(posedge wclk or negedge wrst_n) begin
    if (~wrst_n) begin
      wptr <= 0;
      wptr_gray <= 0;
    end else if (winc && ~wfull) begin
      mem[wptr] <= wdata;
      wptr <= wptr + 1;
      wptr_gray <= (wptr + 1) ^ ((wptr + 1) >> 1);
    end
  end

  always @(posedge rclk or negedge rrst_n) begin
    if (~rrst_n) begin
      rptr <= 0;
      rptr_gray <= 0;
      rdata_reg <= 0;
    end else if (rinc && ~rempty) begin
      if (FALLTHROUGH == "TRUE") begin
        rdata_reg <= mem[rptr];
      end
      rptr <= rptr + 1;
      rptr_gray <= (rptr + 1) ^ ((rptr + 1) >> 1);
    end
  end

  always @(posedge wclk) begin
    wptr_sync <= wptr;
    wptr_sync_gray <= wptr_gray;
  end

  always @(posedge rclk) begin
    rptr_sync <= rptr;
    rptr_sync_gray <= rptr_gray;
  end

  assign rdata = (FALLTHROUGH == "TRUE") ? mem[rptr] : rdata_reg;

endmodule
```
