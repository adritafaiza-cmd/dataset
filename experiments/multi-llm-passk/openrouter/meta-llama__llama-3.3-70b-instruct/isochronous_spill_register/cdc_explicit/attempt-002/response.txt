```verilog
module isochronous_spill_register #(
    parameter WIDTH = 8
)(
    input              src_clk_i,
    input              src_rst_ni,
    input              src_valid_i,
    output             src_ready_o,
    input  [WIDTH-1:0] src_data_i,
    input              dst_clk_i,
    input              dst_rst_ni,
    output             dst_valid_o,
    input              dst_ready_i,
    output [WIDTH-1:0] dst_data_o
);

reg [WIDTH-1:0] buffer_reg [1:0];
reg [1:0] buffer_valid_reg;
reg [1:0] buffer_ready_reg;

always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (~src_rst_ni) begin
        buffer_valid_reg[0] <= 1'b0;
        buffer_valid_reg[1] <= 1'b0;
        buffer_ready_reg[0] <= 1'b1;
        buffer_ready_reg[1] <= 1'b1;
    end else if (src_valid_i && buffer_ready_reg[0]) begin
        buffer_reg[0] <= src_data_i;
        buffer_valid_reg[0] <= 1'b1;
        buffer_ready_reg[0] <= 1'b0;
    end else if (~src_valid_i && buffer_valid_reg[0] && buffer_ready_reg[1]) begin
        buffer_valid_reg[0] <= 1'b0;
        buffer_ready_reg[0] <= 1'b1;
    end
end

always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (~dst_rst_ni) begin
        buffer_valid_reg[1] <= 1'b0;
        buffer_ready_reg[1] <= 1'b1;
    end else if (buffer_valid_reg[0] && buffer_ready_reg[1] && dst_ready_i) begin
        buffer_reg[1] <= buffer_reg[0];
        buffer_valid_reg[1] <= 1'b1;
        buffer_ready_reg[1] <= 1'b0;
    end else if (~buffer_valid_reg[0] && buffer_valid_reg[1] && ~dst_ready_i) begin
        buffer_valid_reg[1] <= 1'b0;
        buffer_ready_reg[1] <= 1'b1;
    end
end

assign src_ready_o = buffer_ready_reg[0];
assign dst_valid_o = buffer_valid_reg[1];
assign dst_data_o = buffer_reg[1];

endmodule
```
