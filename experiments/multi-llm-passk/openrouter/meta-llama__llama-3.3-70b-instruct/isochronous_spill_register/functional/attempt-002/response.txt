```verilog
module isochronous_spill_register #(
    parameter WIDTH = 8
)(
    input              src_clk_i,
    input              src_rst_ni,
    input              src_valid_i,
    output             src_ready_o,
    input  [WIDTH-1:0] src_data_i,
    input              dst_clk_i,
    input              dst_rst_ni,
    output             dst_valid_o,
    input              dst_ready_i,
    output [WIDTH-1:0] dst_data_o
);

reg [WIDTH-1:0] buffer_reg1;
reg [WIDTH-1:0] buffer_reg2;
reg             buffer_valid1;
reg             buffer_valid2;

always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
        buffer_valid1 <= 1'b0;
        buffer_reg1 <= {WIDTH{1'b0}};
    end else if (src_valid_i &&!buffer_valid1) begin
        buffer_valid1 <= 1'b1;
        buffer_reg1 <= src_data_i;
    end else if (src_valid_i && buffer_valid1) begin
        // do nothing
    end
end

always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
        buffer_valid2 <= 1'b0;
        buffer_reg2 <= {WIDTH{1'b0}};
        dst_valid_o <= 1'b0;
        dst_data_o <= {WIDTH{1'b0}};
    end else if (buffer_valid1 &&!buffer_valid2 && dst_ready_i) begin
        buffer_valid2 <= 1'b1;
        buffer_reg2 <= buffer_reg1;
        buffer_valid1 <= 1'b0;
        dst_valid_o <= 1'b1;
        dst_data_o <= buffer_reg1;
    end else if (buffer_valid2 && dst_ready_i) begin
        buffer_valid2 <= 1'b0;
        dst_valid_o <= 1'b0;
        dst_data_o <= {WIDTH{1'b0}};
    end else if (!buffer_valid1 &&!buffer_valid2) begin
        dst_valid_o <= 1'b0;
        dst_data_o <= {WIDTH{1'b0}};
    end
end

assign src_ready_o =!buffer_valid1;

endmodule
```
