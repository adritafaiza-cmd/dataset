```verilog
module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  reg [DSIZE-1:0] mem [2**ASIZE-1:0];
  reg [ASIZE-1:0] wptr_a;
  reg [ASIZE-1:0] rptr_a;
  reg [ASIZE-1:0] wptr_b;
  reg [ASIZE-1:0] rptr_b;
  reg [ASIZE-1:0] wptr_a_gray;
  reg [ASIZE-1:0] rptr_a_gray;
  reg [ASIZE-1:0] wptr_b_gray;
  reg [ASIZE-1:0] rptr_b_gray;
  reg [ASIZE-1:0] wptr_a_sync;
  reg [ASIZE-1:0] rptr_a_sync;
  reg [ASIZE-1:0] wptr_b_sync;
  reg [ASIZE-1:0] rptr_b_sync;
  reg [DSIZE-1:0] rdata_a;
  reg [DSIZE-1:0] rdata_b;
  reg a_full_int;
  reg a_afull_int;
  reg a_empty_int;
  reg a_aempty_int;
  reg b_full_int;
  reg b_afull_int;
  reg b_empty_int;
  reg b_aempty_int;

  assign a_full = a_full_int;
  assign a_afull = a_afull_int;
  assign a_empty = a_empty_int;
  assign a_aempty = a_aempty_int;
  assign b_full = b_full_int;
  assign b_afull = b_afull_int;
  assign b_empty = b_empty_int;
  assign b_aempty = b_aempty_int;

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      wptr_a <= 0;
      rptr_a <= 0;
      wptr_a_gray <= 0;
      rptr_a_sync <= 0;
      a_full_int <= 0;
      a_afull_int <= 0;
      a_empty_int <= 1;
      a_aempty_int <= 1;
    end else begin
      if (a_dir && a_winc && !a_full_int) begin
        mem[wptr_a] <= a_wdata;
        wptr_a <= wptr_a + 1;
        wptr_a_gray <= gray_encode(wptr_a);
        if (wptr_a == (2**ASIZE-1)) begin
          wptr_a <= 0;
        end
        if (wptr_a == (rptr_a + 2**ASIZE - 1)) begin
          a_full_int <= 1;
        end else if (wptr_a == (rptr_a + 2**(ASIZE-1) - 1)) begin
          a_afull_int <= 1;
        end else begin
          a_afull_int <= 0;
        end
      end
      if (!a_dir && a_rinc && !a_empty_int) begin
        rptr_a <= rptr_a + 1;
        if (rptr_a == (2**ASIZE-1)) begin
          rptr_a <= 0;
        end
        if (rptr_a == wptr_a) begin
          a_empty_int <= 1;
        end else if (rptr_a == (wptr_a - 2**(ASIZE-1))) begin
          a_aempty_int <= 1;
        end else begin
          a_aempty_int <= 0;
        end
      end
      if (a_dir && a_winc && a_full_int) begin
        a_full_int <= 0;
      end
      if (!a_dir && a_rinc && a_empty_int) begin
        a_empty_int <= 0;
      end
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      wptr_b <= 0;
      rptr_b <= 0;
      wptr_b_gray <= 0;
      rptr_b_sync <= 0;
      b_full_int <= 0;
      b_afull_int <= 0;
      b_empty_int <= 1;
      b_aempty_int <= 1;
    end else begin
      if (b_dir && b_winc && !b_full_int) begin
        mem[wptr_b] <= b_wdata;
        wptr_b <= wptr_b + 1;
        wptr_b_gray <= gray_encode(wptr_b);
        if (wptr_b == (2**ASIZE-1)) begin
          wptr_b <= 0;
        end
        if (wptr_b == (rptr_b + 2**ASIZE - 1)) begin
          b_full_int <= 1;
        end else if (wptr_b == (rptr_b + 2**(ASIZE-1) - 1)) begin
          b_afull_int <= 1;
        end else begin
          b_afull_int <= 0;
        end
      end
      if (!b_dir && b_rinc && !b_empty_int) begin
        rptr_b <= rptr_b + 1;
        if (rptr_b == (2**ASIZE-1)) begin
          rptr_b <= 0;
        end
        if (rptr_b == wptr_b) begin
          b_empty_int <= 1;
        end else if (rptr_b == (wptr_b - 2**(ASIZE-1))) begin
          b_aempty_int <= 1;
        end else begin
          b_aempty_int <= 0;
        end
      end
      if (b_dir && b_winc && b_full_int) begin
        b_full_int <= 0;
      end
      if (!b_dir && b_rinc && b_empty_int) begin
        b_empty_int <= 0;
      end
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      rptr_a_sync <= 0;
    end else begin
      rptr_a_sync <= rptr_a;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      rptr_b_sync <= 0;
    end else begin
      rptr_b_sync <= rptr_b;
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      wptr_b_sync <= 0;
    end else begin
      wptr_b_sync <= wptr_b_gray;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      wptr_a_sync <= 0;
    end else begin
      wptr_a_sync <= wptr_a_gray;
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      rdata_a <= 0;
    end else if (!a_dir && a_rinc && !a_empty_int) begin
      rdata_a <= mem[rptr_a_sync];
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      rdata_b <= 0;
    end else if (!b_dir && b_rinc && !b_empty_int) begin
      rdata_b <= mem[rptr_b_sync];
    end
  end

  assign a_rdata = (FALLTHROUGH == "TRUE") ? (a_empty_int ? 0 : mem[rptr_a_sync]) : rdata_a;
  assign b_rdata = (FALLTHROUGH == "TRUE") ? (b_empty_int ? 0 : mem[rptr_b_sync]) : rdata_b;

  function [ASIZE-1:0] gray_encode;
    input [ASIZE-1:0] bin;
    reg [ASIZE-1:0] gray;
    begin
      gray = bin ^ (bin >> 1);
      gray_encode = gray;
    end
  endfunction

endmodule
```
