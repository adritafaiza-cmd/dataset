```verilog
module data_sync #(
    parameter STAGES = 2,
    parameter DWIDTH = 8
)(
    input                  clk,
    input                  rstn,
    input  [DWIDTH-1:0]    din,
    input                  dready_i,
    output reg [DWIDTH-1:0] dout,
    output reg             dready_o
);
    reg [DWIDTH-1:0] sync_reg [STAGES-1:0];
    reg [STAGES-1:0] dready_reg;

    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            for (int i = 0; i < STAGES; i++) begin
                sync_reg[i] <= {DWIDTH{1'b0}};
                if (i < STAGES) begin
                    dready_reg[i] <= 1'b0;
                end
            end
            dout <= {DWIDTH{1'b0}};
            dready_o <= 1'b0;
        end else begin
            sync_reg[0] <= din;
            for (int i = 1; i < STAGES; i++) begin
                sync_reg[i] <= sync_reg[i-1];
            end
            dready_reg[0] <= dready_i;
            for (int i = 1; i < STAGES; i++) begin
                dready_reg[i] <= dready_reg[i-1];
            end
            if (dready_reg[STAGES-1]) begin
                dout <= sync_reg[STAGES-1];
                dready_o <= 1'b1;
            end else begin
                dready_o <= 1'b0;
            end
        end
    end
endmodule
```
