module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    reg [WIDTH-1:0]         fifo_mem [2**LOG_DEPTH-1:0];
    reg [LOG_DEPTH-1:0]    wr_ptr;
    reg [LOG_DEPTH-1:0]    rd_ptr;
    reg [LOG_DEPTH-1:0]    wr_ptr_gray;
    reg [LOG_DEPTH-1:0]    rd_ptr_gray;
    reg [LOG_DEPTH:0]      count;
    reg                    src_ready;
    reg                    dst_valid;
    reg [WIDTH-1:0]        dst_data;

    // Gray encoding
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            wr_ptr_gray <= '0;
        end else begin
            wr_ptr_gray <= wr_ptr;
            wr_ptr_gray[LOG_DEPTH-1] <= wr_ptr[LOG_DEPTH-1] ^ wr_ptr[LOG_DEPTH-2];
            for (int i = LOG_DEPTH-2; i > 0; i--) begin
                wr_ptr_gray[i] <= wr_ptr[i] ^ wr_ptr[i-1];
            end
            wr_ptr_gray[0] <= wr_ptr[0];
        end
    end

    // Gray decoding
    reg [LOG_DEPTH-1:0]    rd_ptr_gray_sync;
    reg [LOG_DEPTH-1:0]    rd_ptr_sync;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            rd_ptr_gray_sync <= '0;
            rd_ptr_sync <= '0;
        end else begin
            rd_ptr_gray_sync <= wr_ptr_gray;
            rd_ptr_sync[LOG_DEPTH-1] <= rd_ptr_gray_sync[LOG_DEPTH-1];
            for (int i = LOG_DEPTH-2; i >= 0; i--) begin
                rd_ptr_sync[i] <= rd_ptr_gray_sync[i] ^ rd_ptr_sync[i+1];
            end
        end
    end

    // FIFO write logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            wr_ptr <= '0;
            count <= '0;
            src_ready <= 1'b1;
        end else if (src_valid_i && src_ready) begin
            fifo_mem[wr_ptr] <= src_data_i;
            if (wr_ptr == 2**LOG_DEPTH-1) begin
                wr_ptr <= '0;
            end else begin
                wr_ptr <= wr_ptr + 1;
            end
            if (count == 2**LOG_DEPTH-1) begin
                src_ready <= 1'b0;
            end else begin
                count <= count + 1;
            end
        end else if (!src_valid_i && src_ready && count == 2**LOG_DEPTH) begin
            src_ready <= 1'b0;
        end else if (!src_valid_i &&!src_ready && count == 0) begin
            src_ready <= 1'b1;
        end
    end

    // FIFO read logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            rd_ptr <= '0;
            count <= '0;
            dst_valid <= 1'b0;
            dst_data <= '0;
        end else if (dst_ready_i && dst_valid) begin
            if (rd_ptr == 2**LOG_DEPTH-1) begin
                rd_ptr <= '0;
            end else begin
                rd_ptr <= rd_ptr + 1;
            end
            count <= count - 1;
            dst_valid <= (count > 1);
        end else if (dst_ready_i &&!dst_valid && count > 0) begin
            dst_data <= fifo_mem[rd_ptr];
            dst_valid <= 1'b1;
            if (rd_ptr == 2**LOG_DEPTH-1) begin
                rd_ptr <= '0;
            end else begin
                rd_ptr <= rd_ptr + 1;
            end
            count <= count - 1;
        end else if (!dst_ready_i && dst_valid && count == 1) begin
            dst_valid <= 1'b0;
        end
    end

    assign src_ready_o = src_ready;
    assign dst_data_o = dst_data;
    assign dst_valid_o = dst_valid;

endmodule
