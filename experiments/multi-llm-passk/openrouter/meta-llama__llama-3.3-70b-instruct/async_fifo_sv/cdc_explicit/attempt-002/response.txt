```systemverilog
module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam DEPTH = 1 << ADDR_WIDTH;

    reg [ADDR_WIDTH-1:0] wptr;
    reg [ADDR_WIDTH-1:0] rptr;
    reg [ADDR_WIDTH-1:0] wptr_sync;
    reg [ADDR_WIDTH-1:0] rptr_sync;
    reg [ADDR_WIDTH:0]   wcnt;
    reg [ADDR_WIDTH:0]   rcnt;
    reg [DATA_WIDTH-1:0] data [DEPTH-1:0];

    assign wfull = wcnt == DEPTH;
    assign rempty = rcnt == 0;
    assign waddr = {wcnt, wptr};
    assign raddr = {rcnt, rptr};

    always @(posedge wclk or negedge wrst_n) begin
        if (~wrst_n) begin
            wptr <= 0;
            wcnt <= 0;
        end else if (winc && ~wfull) begin
            data[wptr] <= wdata;
            wptr <= wptr + 1;
            wcnt <= wcnt + 1;
        end
    end

    always @(posedge rclk or negedge rrst_n) begin
        if (~rrst_n) begin
            rptr <= 0;
            rcnt <= 0;
        end else if (rinc && ~rempty) begin
            rdata <= data[rptr];
            rptr <= rptr + 1;
            rcnt <= rcnt - 1;
        end
    end

    // Synchronize wptr to read clock domain
    reg [ADDR_WIDTH-1:0] wptr_sync_reg [SYNC_STAGES-1:0];
    always @(posedge rclk) begin
        wptr_sync_reg[0] <= wptr;
        for (int i = 1; i < SYNC_STAGES; i++) begin
            wptr_sync_reg[i] <= wptr_sync_reg[i-1];
        end
        wptr_sync <= wptr_sync_reg[SYNC_STAGES-1];
    end

    // Synchronize rptr to write clock domain
    reg [ADDR_WIDTH-1:0] rptr_sync_reg [SYNC_STAGES-1:0];
    always @(posedge wclk) begin
        rptr_sync_reg[0] <= rptr;
        for (int i = 1; i < SYNC_STAGES; i++) begin
            rptr_sync_reg[i] <= rptr_sync_reg[i-1];
        end
        rptr_sync <= rptr_sync_reg[SYNC_STAGES-1];
    end

    // Update rcnt
    always @(posedge wclk) begin
        if (~wrst_n) begin
            rcnt <= 0;
        end else if (winc && ~wfull) begin
            rcnt <= rcnt + 1;
        end else if (rinc && ~rempty && wptr_sync == rptr_sync) begin
            rcnt <= rcnt - 1;
        end
    end

    // Update wcnt
    always @(posedge rclk) begin
        if (~rrst_n) begin
            wcnt <= 0;
        end else if (rinc && ~rempty) begin
            wcnt <= wcnt - 1;
        end else if (winc && ~wfull && wptr_sync == rptr_sync) begin
            wcnt <= wcnt + 1;
        end
    end

endmodule
```
