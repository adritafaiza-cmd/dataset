```verilog
module apb_cdc #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_pclk_i,
    input                      src_preset_ni,
    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,
    input  [ADDR_WIDTH-1:0]    src_paddr_i,
    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    reg [ADDR_WIDTH-1:0]      fifo_addr;
    reg [DATA_WIDTH-1:0]      fifo_wdata;
    reg [DATA_WIDTH/8-1:0]    fifo_wstrb;
    reg [2:0]                 fifo_wprot;
    reg                       fifo_write;
    reg                       fifo_full;
    reg                       fifo_empty;
    reg                       src_req;
    reg                       dst_req;
    reg [DATA_WIDTH-1:0]      dst_rdata;
    reg                       dst_err;
    reg                       src_pready_int;
    reg                       dst_psel_int;
    reg                       dst_penable_int;
    reg                       dst_pwrite_int;

    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            src_req <= 1'b0;
            src_pready_int <= 1'b0;
        end else if (src_psel_i && src_penable_i && !fifo_full) begin
            src_req <= 1'b1;
            fifo_addr <= src_paddr_i;
            fifo_wdata <= src_pwdata_i;
            fifo_wstrb <= src_pstrb_i;
            fifo_wprot <= src_pprot_i;
            fifo_write <= 1'b1;
        end else if (src_psel_i && src_penable_i && fifo_full) begin
            src_pready_int <= 1'b0;
        end else if (dst_pready_i && dst_req) begin
            src_pready_int <= 1'b1;
            src_req <= 1'b0;
        end
    end

    always @(posedge dst_pclk_i or negedge dst_preset_ni) begin
        if (!dst_preset_ni) begin
            dst_req <= 1'b0;
            dst_psel_int <= 1'b0;
            dst_penable_int <= 1'b0;
            dst_pwrite_int <= 1'b0;
        end else if (fifo_write && !fifo_empty) begin
            dst_req <= 1'b1;
            dst_psel_int <= 1'b1;
            dst_penable_int <= 1'b1;
            dst_pwrite_int <= fifo_write;
            dst_paddr_o <= fifo_addr;
            dst_pwdata_o <= fifo_wdata;
            dst_pstrb_o <= fifo_wstrb;
            dst_pprot_o <= fifo_wprot;
            fifo_write <= 1'b0;
        end else if (dst_pready_i && dst_req) begin
            dst_req <= 1'b0;
            dst_psel_int <= 1'b0;
            dst_penable_int <= 1'b0;
            dst_pwrite_int <= 1'b0;
            dst_rdata <= dst_prdata_i;
            dst_err <= dst_pslverr_i;
        end
    end

    assign src_pready_o = src_pready_int;
    assign src_prdata_o = dst_rdata;
    assign src_pslverr_o = dst_err;
    assign dst_psel_o = dst_psel_int;
    assign dst_penable_o = dst_penable_int;
    assign dst_pwrite_o = dst_pwrite_int;

    // FIFO implementation
    reg [LOG_DEPTH-1:0]       fifo_ptr;
    reg [LOG_DEPTH-1:0]       fifo_cnt;
    reg                       fifo_empty_int;
    reg                       fifo_full_int;

    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (!src_preset_ni) begin
            fifo_ptr <= '0;
            fifo_cnt <= '0;
            fifo_empty_int <= 1'b1;
            fifo_full_int <= 1'b0;
        end else if (fifo_write && !fifo_full_int) begin
            fifo_ptr <= fifo_ptr + 1'b1;
            fifo_cnt <= fifo_cnt + 1'b1;
            fifo_empty_int <= 1'b0;
            if (fifo_cnt == (1 << LOG_DEPTH) - 1) begin
                fifo_full_int <= 1'b1;
            end
        end else if (dst_pready_i && dst_req) begin
            fifo_cnt <= fifo_cnt - 1'b1;
            fifo_empty_int <= (fifo_cnt == 0) ? 1'b1 : 1'b0;
            fifo_full_int <= 1'b0;
        end
    end

    assign fifo_empty = fifo_empty_int;
    assign fifo_full = fifo_full_int;

endmodule
```
