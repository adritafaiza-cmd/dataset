module edge_propagator (
  input  logic clk_tx_i,
  input  logic rstn_tx_i,
  input  logic edge_i,
  input  logic clk_rx_i,
  input  logic rstn_rx_i,
  output logic edge_o
);

logic [1:0] sync_edge;
logic       edge_synced;

always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
  if (!rstn_rx_i) begin
    sync_edge <= 2'b00;
  end else begin
    sync_edge <= {sync_edge[0], edge_synced};
  end
end

always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
  if (!rstn_tx_i) begin
    edge_synced <= 1'b0;
  end else begin
    edge_synced <= edge_i;
  end
end

always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
  if (!rstn_rx_i) begin
    edge_o <= 1'b0;
  end else begin
    edge_o <= sync_edge[1] && !sync_edge[0];
  end
end

endmodule
