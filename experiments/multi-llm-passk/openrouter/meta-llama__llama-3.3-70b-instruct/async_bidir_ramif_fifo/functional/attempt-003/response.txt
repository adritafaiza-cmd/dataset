```verilog
module async_bidir_ramif_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "FALSE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir,
       output wire             o_ram_a_clk,
       output wire [DSIZE-1:0] o_ram_a_wdata,
       input wire [DSIZE-1:0]  i_ram_a_rdata,
       output wire [ASIZE-1:0] o_ram_a_addr,
       output wire             o_ram_a_rinc,
       output wire             o_ram_a_winc,
       output wire             o_ram_b_clk,
       output wire [DSIZE-1:0] o_ram_b_wdata,
       input wire [DSIZE-1:0]  i_ram_b_rdata,
       output wire [ASIZE-1:0] o_ram_b_addr,
       output wire             o_ram_b_rinc,
       output wire             o_ram_b_winc
       );

   reg [ASIZE-1:0] a_ptr;
   reg [ASIZE-1:0] b_ptr;
   reg [ASIZE-1:0] a_cnt;
   reg [ASIZE-1:0] b_cnt;
   reg [DSIZE-1:0] a_rdata_reg;
   reg [DSIZE-1:0] b_rdata_reg;

   assign o_ram_a_clk = a_clk;
   assign o_ram_b_clk = b_clk;

   assign o_ram_a_wdata = a_dir ? a_wdata : i_ram_b_rdata;
   assign o_ram_b_wdata = b_dir ? b_wdata : i_ram_a_rdata;

   assign o_ram_a_addr = a_dir ? a_ptr : b_ptr;
   assign o_ram_b_addr = b_dir ? b_ptr : a_ptr;

   assign o_ram_a_rinc = a_dir && a_rinc;
   assign o_ram_a_winc = a_dir && a_winc;
   assign o_ram_b_rinc = b_dir && b_rinc;
   assign o_ram_b_winc = b_dir && b_winc;

   always @(posedge a_clk or negedge a_rst_n) begin
      if (!a_rst_n) begin
         a_ptr <= 0;
         a_cnt <= 0;
         a_rdata_reg <= 0;
      end else begin
         if (a_winc && !a_rinc) begin
            a_cnt <= a_cnt + 1;
         end else if (!a_winc && a_rinc) begin
            a_cnt <= a_cnt - 1;
         end
         if (a_rinc) begin
            a_ptr <= a_ptr + 1;
            a_rdata_reg <= i_ram_a_rdata;
         end
      end
   end

   always @(posedge b_clk or negedge b_rst_n) begin
      if (!b_rst_n) begin
         b_ptr <= 0;
         b_cnt <= 0;
         b_rdata_reg <= 0;
      end else begin
         if (b_winc && !b_rinc) begin
            b_cnt <= b_cnt + 1;
         end else if (!b_winc && b_rinc) begin
            b_cnt <= b_cnt - 1;
         end
         if (b_rinc) begin
            b_ptr <= b_ptr + 1;
            b_rdata_reg <= i_ram_b_rdata;
         end
      end
   end

   assign a_rdata = a_rdata_reg;
   assign b_rdata = b_rdata_reg;

   assign a_full = (a_cnt == (1 << ASIZE) - 1);
   assign a_afull = (a_cnt == (1 << ASIZE) - 2);
   assign a_empty = (a_cnt == 0);
   assign a_aempty = (a_cnt == 1);

   assign b_full = (b_cnt == (1 << ASIZE) - 1);
   assign b_afull = (b_cnt == (1 << ASIZE) - 2);
   assign b_empty = (b_cnt == 0);
   assign b_aempty = (b_cnt == 1);

endmodule
```
