```verilog
module async_bidir_ramif_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "FALSE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir,
       output wire             o_ram_a_clk,
       output wire [DSIZE-1:0] o_ram_a_wdata,
       input wire [DSIZE-1:0]  i_ram_a_rdata,
       output wire [ASIZE-1:0] o_ram_a_addr,
       output wire             o_ram_a_rinc,
       output wire             o_ram_a_winc,
       output wire             o_ram_b_clk,
       output wire [DSIZE-1:0] o_ram_b_wdata,
       input wire [DSIZE-1:0]  i_ram_b_rdata,
       output wire [ASIZE-1:0] o_ram_b_addr,
       output wire             o_ram_b_rinc,
       output wire             o_ram_b_winc
       );

  reg [ASIZE-1:0] a_ptr;
  reg [ASIZE-1:0] a_ptr_gray;
  reg [ASIZE-1:0] b_ptr;
  reg [ASIZE-1:0] b_ptr_gray;
  reg [ASIZE-1:0] a_ptr_sync;
  reg [ASIZE-1:0] b_ptr_sync;
  reg [ASIZE-1:0] a_ptr_sync_gray;
  reg [ASIZE-1:0] b_ptr_sync_gray;
  reg [DSIZE-1:0] a_rdata_reg;
  reg [DSIZE-1:0] b_rdata_reg;
  reg a_empty_reg;
  reg a_full_reg;
  reg b_empty_reg;
  reg b_full_reg;
  reg a_afull_reg;
  reg b_afull_reg;
  reg a_aempty_reg;
  reg b_aempty_reg;

  assign o_ram_a_clk = a_clk;
  assign o_ram_b_clk = b_clk;

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_ptr <= 0;
      a_ptr_gray <= 0;
    end else begin
      if (a_winc &&!a_full_reg) begin
        a_ptr <= a_ptr + 1;
      end
      if (a_rinc &&!a_empty_reg) begin
        a_ptr_gray <= a_ptr_gray + 1;
      end
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_ptr <= 0;
      b_ptr_gray <= 0;
    end else begin
      if (b_winc &&!b_full_reg) begin
        b_ptr <= b_ptr + 1;
      end
      if (b_rinc &&!b_empty_reg) begin
        b_ptr_gray <= b_ptr_gray + 1;
      end
    end
  end

  // CDC for a_ptr
  sync_cdc #(.DSIZE(ASIZE)) a_ptr_sync_cdc (
   .i_clk(a_clk),
   .i_rst_n(a_rst_n),
   .i_data(a_ptr),
   .o_data(a_ptr_sync)
  );

  // CDC for b_ptr
  sync_cdc #(.DSIZE(ASIZE)) b_ptr_sync_cdc (
   .i_clk(b_clk),
   .i_rst_n(b_rst_n),
   .i_data(b_ptr),
   .o_data(b_ptr_sync)
  );

  // CDC for a_ptr_gray
  sync_cdc #(.DSIZE(ASIZE)) a_ptr_sync_gray_cdc (
   .i_clk(a_clk),
   .i_rst_n(a_rst_n),
   .i_data(a_ptr_gray),
   .o_data(a_ptr_sync_gray)
  );

  // CDC for b_ptr_gray
  sync_cdc #(.DSIZE(ASIZE)) b_ptr_sync_gray_cdc (
   .i_clk(b_clk),
   .i_rst_n(b_rst_n),
   .i_data(b_ptr_gray),
   .o_data(b_ptr_sync_gray)
  );

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_empty_reg <= 1;
      a_full_reg <= 0;
      a_afull_reg <= 0;
      a_aempty_reg <= 1;
    end else begin
      if (a_winc &&!a_full_reg) begin
        if (a_ptr == {ASIZE{1'b1}}) begin
          a_full_reg <= 1;
        end
      end
      if (a_rinc &&!a_empty_reg) begin
        if (a_ptr_gray == 0) begin
          a_empty_reg <= 1;
        end
        if (a_ptr_gray == {ASIZE{1'b1}} - 1) begin
          a_afull_reg <= 0;
        end
      end
      if (a_winc && a_rinc && a_ptr == a_ptr_gray) begin
        a_empty_reg <= 0;
        a_full_reg <= 0;
      end
      if (a_winc &&!a_rinc && a_ptr == {ASIZE{1'b1}}) begin
        a_afull_reg <= 1;
      end
      if (a_rinc &&!a_winc && a_ptr_gray == 0) begin
        a_aempty_reg <= 0;
      end
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_empty_reg <= 1;
      b_full_reg <= 0;
      b_afull_reg <= 0;
      b_aempty_reg <= 1;
    end else begin
      if (b_winc &&!b_full_reg) begin
        if (b_ptr == {ASIZE{1'b1}}) begin
          b_full_reg <= 1;
        end
      end
      if (b_rinc &&!b_empty_reg) begin
        if (b_ptr_gray == 0) begin
          b_empty_reg <= 1;
        end
        if (b_ptr_gray == {ASIZE{1'b1}} - 1) begin
          b_afull_reg <= 0;
        end
      end
      if (b_winc && b_rinc && b_ptr == b_ptr_gray) begin
        b_empty_reg <= 0;
        b_full_reg <= 0;
      end
      if (b_winc &&!b_rinc && b_ptr == {ASIZE{1'b1}}) begin
        b_afull_reg <= 1;
      end
      if (b_rinc &&!b_winc && b_ptr_gray == 0) begin
        b_aempty_reg <= 0;
      end
    end
  end

  assign a_rdata = a_rdata_reg;
  assign b_rdata = b_rdata_reg;
  assign a_empty = a_empty_reg;
  assign a_full = a_full_reg;
  assign a_afull = a_afull_reg;
  assign a_aempty = a_aempty_reg;
  assign b_empty = b_empty_reg;
  assign b_full = b_full_reg;
  assign b_afull = b_afull_reg;
  assign b_aempty = b_aempty_reg;

  assign o_ram_a_addr = a_ptr;
  assign o_ram_b_addr = b_ptr;

  assign o_ram_a_wdata = a_wdata;
  assign o_ram_b_wdata = b_wdata;

  assign o_ram_a_winc = a_winc;
  assign o_ram_b_winc = b_winc;

  assign o_ram_a_rinc = a_rinc;
  assign o_ram_b_rinc = b_rinc;

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_rdata_reg <= 0;
    end else if (a_rinc &&!a_empty_reg) begin
      a_rdata_reg <= i_ram_a_rdata;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_rdata_reg <= 0;
    end else if (b_rinc &&!b_empty_reg) begin
      b_rdata_reg <= i_ram_b_rdata;
    end
  end

endmodule

module sync_cdc
  #(
    parameter DSIZE = 8
    ) (
       input wire              i_clk,
       input wire              i_rst_n,
       input wire [DSIZE-1:0]  i_data,
       output wire [DSIZE-1:0] o_data
       );

  reg [DSIZE-1:0] data_reg;
  reg [DSIZE-1:0] data_reg_sync;

  always @(posedge i_clk or negedge i_rst_n) begin
    if (!i_rst_n) begin
      data_reg <= 0;
      data_reg_sync <= 0;
    end else begin
      data_reg <= i_data;
      data_reg_sync <= data_reg;
    end
  end

  assign o_data = data_reg_sync;

endmodule
```
