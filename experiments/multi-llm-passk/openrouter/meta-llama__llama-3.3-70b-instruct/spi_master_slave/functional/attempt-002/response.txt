```verilog
module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

reg [N-1:0] shift_reg;
reg [N-1:0] capture_reg;
reg [3:0] sclk_div;
reg [3:0] sclk_cnt;
reg spi_sck_en;
reg spi_sck_rst;
reg spi_ssel_en;
reg spi_ssel_rst;
reg [1:0] state;
reg [1:0] next_state;

always @(posedge sclk_i or posedge rst_i)
begin
    if (rst_i)
    begin
        spi_sck_en <= 1'b0;
        spi_sck_rst <= 1'b1;
        spi_ssel_en <= 1'b0;
        spi_ssel_rst <= 1'b1;
        sclk_div <= 4'b0;
        sclk_cnt <= 4'b0;
        state <= 2'b00;
        next_state <= 2'b00;
    end
    else
    begin
        spi_sck_en <= spi_sck_rst;
        spi_sck_rst <= 1'b0;
        spi_ssel_en <= spi_ssel_rst;
        spi_ssel_rst <= 1'b0;
        sclk_div <= sclk_div + 1'b1;
        if (sclk_div == SPI_2X_CLK_DIV)
        begin
            sclk_div <= 4'b0;
            sclk_cnt <= sclk_cnt + 1'b1;
        end
        if (sclk_cnt == 4'b1)
        begin
            sclk_cnt <= 4'b0;
            state <= next_state;
        end
    end
end

always @(posedge pclk_i or posedge rst_i)
begin
    if (rst_i)
    begin
        spi_ssel_o <= 1'b1;
        spi_sck_o <= 1'b0;
        di_req_o <= 1'b0;
        wr_ack_o <= 1'b0;
        do_valid_o <= 1'b0;
        do_o <= {N{1'b0}};
        shift_reg <= {N{1'b0}};
        capture_reg <= {N{1'b0}};
    end
    else
    begin
        spi_ssel_o <= spi_ssel_en? 1'b0 : 1'b1;
        spi_sck_o <= spi_sck_en? ~spi_sck_o : 1'b0;
        di_req_o <= (state == 2'b00)? 1'b1 : 1'b0;
        wr_ack_o <= (state == 2'b01 && wren_i)? 1'b1 : 1'b0;
        do_valid_o <= (state == 2'b10)? 1'b1 : 1'b0;
        if (wren_i && state == 2'b00)
        begin
            shift_reg <= di_i;
        end
        if (state == 2'b01)
        begin
            shift_reg <= {shift_reg[N-2:0], 1'b0};
        end
        if (state == 2'b10)
        begin
            capture_reg <= {capture_reg[N-2:0], spi_miso_i};
            do_o <= capture_reg;
        end
    end
end

assign spi_mosi_o = shift_reg[N-1];

always @(*)
begin
    case (state)
        2'b00:
        begin
            if (wren_i)
            begin
                next_state <= 2'b01;
            end
            else
            begin
                next_state <= 2'b00;
            end
        end
        2'b01:
        begin
            if (sclk_cnt == 4'b1)
            begin
                if (shift_reg[N-1] == 1'b0)
                begin
                    next_state <= 2'b10;
                end
                else
                begin
                    next_state <= 2'b01;
                end
            end
            else
            begin
                next_state <= 2'b01;
            end
        end
        2'b10:
        begin
            next_state <= 2'b00;
        end
        default:
        begin
            next_state <= 2'b00;
        end
    endcase
end

always @(*)
begin
    case (state)
        2'b00:
        begin
            spi_sck_rst <= 1'b1;
            spi_ssel_rst <= 1'b0;
        end
        2'b01:
        begin
            spi_sck_rst <= 1'b0;
            spi_ssel_rst <= 1'b1;
        end
        2'b10:
        begin
            spi_sck_rst <= 1'b1;
            spi_ssel_rst <= 1'b0;
        end
        default:
        begin
            spi_sck_rst <= 1'b1;
            spi_ssel_rst <= 1'b0;
        end
    endcase
end

endmodule
```
