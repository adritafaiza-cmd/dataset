```verilog
module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

reg [N-1:0] shift_reg;
reg [N-1:0] capture_reg;
reg [3:0] sclk_cnt;
reg [3:0] pclk_cnt;
reg spi_sck_r;
reg spi_sck_f;
reg spi_ssel_r;
reg spi_ssel_f;
reg [N-1:0] di_sync;
reg wren_sync;
reg wr_ack_sync;
reg di_req_sync;
reg [N-1:0] do_sync;
reg do_valid_sync;

always @(posedge sclk_i or posedge rst_i) begin
    if (rst_i) begin
        spi_sck_r <= 1'b0;
        spi_sck_f <= 1'b0;
        spi_ssel_r <= 1'b1;
        spi_ssel_f <= 1'b1;
        sclk_cnt <= 4'b0;
    end else begin
        spi_sck_r <= ~spi_sck_f;
        spi_sck_f <= spi_sck_r;
        if (sclk_cnt == SPI_2X_CLK_DIV) begin
            sclk_cnt <= 4'b0;
            spi_ssel_r <= spi_ssel_f;
            spi_ssel_f <= spi_ssel_r;
        end else begin
            sclk_cnt <= sclk_cnt + 1'b1;
        end
    end
end

always @(posedge pclk_i or posedge rst_i) begin
    if (rst_i) begin
        di_req_o <= 1'b0;
        wr_ack_o <= 1'b0;
        do_valid_o <= 1'b0;
        do_o <= {N{1'b0}};
        pclk_cnt <= 4'b0;
    end else begin
        di_req_o <= di_req_sync;
        wr_ack_o <= wr_ack_sync;
        do_valid_o <= do_valid_sync;
        do_o <= do_sync;
        if (pclk_cnt == 4'b1111) begin
            pclk_cnt <= 4'b0;
            wr_ack_sync <= wren_sync;
            di_req_sync <= ~wren_sync;
        end else begin
            pclk_cnt <= pclk_cnt + 1'b1;
        end
    end
end

always @(posedge sclk_i or posedge rst_i) begin
    if (rst_i) begin
        spi_ssel_o <= 1'b1;
        spi_sck_o <= 1'b0;
        shift_reg <= {N{1'b0}};
    end else begin
        spi_ssel_o <= spi_ssel_r;
        spi_sck_o <= spi_sck_r;
        if (spi_ssel_r == 1'b0) begin
            shift_reg <= {di_sync, shift_reg[N-1:1]};
        end
    end
end

always @(posedge pclk_i or posedge rst_i) begin
    if (rst_i) begin
        di_sync <= {N{1'b0}};
        wren_sync <= 1'b0;
    end else begin
        di_sync <= di_i;
        wren_sync <= wren_i;
    end
end

always @(posedge sclk_i or posedge rst_i) begin
    if (rst_i) begin
        capture_reg <= {N{1'b0}};
    end else begin
        capture_reg <= {spi_miso_i, capture_reg[N-1:1]};
    end
end

always @(posedge pclk_i or posedge rst_i) begin
    if (rst_i) begin
        do_sync <= {N{1'b0}};
        do_valid_sync <= 1'b0;
    end else begin
        do_sync <= capture_reg;
        do_valid_sync <= spi_ssel_r;
    end
end

assign spi_mosi_o = shift_reg[N-1];

endmodule
```
