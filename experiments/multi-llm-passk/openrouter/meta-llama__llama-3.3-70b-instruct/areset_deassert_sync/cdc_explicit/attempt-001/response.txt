```verilog
module areset_deassert_sync #(
    parameter CHAINS = 2,
    parameter RST_POL = 1'b1
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);

reg [CHAINS-1:0] rst_chain;
reg sync_rst_int;

always @(posedge clk or negedge clk) begin
    if (!clk) begin
        if (RST_POL == 1'b1) begin
            sync_rst_int <= async_rst_i;
        end else begin
            sync_rst_int <= ~async_rst_i;
        end
    end else begin
        if (RST_POL == 1'b1) begin
            sync_rst_int <= (async_rst_i || rst_chain[CHAINS-1]);
        end else begin
            sync_rst_int <= (~async_rst_i && ~rst_chain[CHAINS-1]);
        end
    end
end

always @(posedge clk) begin
    if (RST_POL == 1'b1) begin
        rst_chain[0] <= async_rst_i;
        for (int i = 1; i < CHAINS; i++) begin
            rst_chain[i] <= rst_chain[i-1];
        end
    end else begin
        rst_chain[0] <= ~async_rst_i;
        for (int i = 1; i < CHAINS; i++) begin
            rst_chain[i] <= rst_chain[i-1];
        end
    end
end

assign sync_rst_o = sync_rst_int;

endmodule
```
