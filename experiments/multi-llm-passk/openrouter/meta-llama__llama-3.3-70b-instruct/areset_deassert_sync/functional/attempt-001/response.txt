```verilog
module areset_deassert_sync #(
    parameter CHAINS = 2,
    parameter RST_POL = 1'b1
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);
    reg [CHAINS-1:0] sync_rst;
    assign sync_rst_o = (RST_POL == 1'b1)? sync_rst[CHAINS-1] : ~sync_rst[CHAINS-1];

    always @(posedge clk or posedge async_rst_i) begin
        if (async_rst_i == RST_POL) begin
            sync_rst <= {CHAINS{1'b1}};
        end else begin
            sync_rst <= {1'b0, sync_rst[CHAINS-1:1]};
        end
    end
endmodule
```
