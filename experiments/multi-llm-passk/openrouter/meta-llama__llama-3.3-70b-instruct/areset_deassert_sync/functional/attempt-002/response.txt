```verilog
module areset_deassert_sync #(
    parameter CHAINS = 2,
    parameter RST_POL = 1'b1
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);

reg [CHAINS-1:0] rst_chain;
reg sync_rst;

always @(posedge clk or negedge clk) begin
    if (~clk) begin
        if (async_rst_i == RST_POL) begin
            sync_rst <= 1'b1;
        end
    end else begin
        if (async_rst_i != RST_POL) begin
            if (rst_chain == {CHAINS{1'b0}}) begin
                sync_rst <= 1'b0;
            end
        end
        rst_chain <= {rst_chain[CHAINS-2:0], async_rst_i == RST_POL};
    end
end

assign sync_rst_o = sync_rst;

endmodule
```
