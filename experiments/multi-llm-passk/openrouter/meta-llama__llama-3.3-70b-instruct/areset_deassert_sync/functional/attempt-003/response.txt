```verilog
module areset_deassert_sync #(
    parameter CHAINS = 2,
    parameter RST_POL = 1'b1
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);

reg [CHAINS-1:0] rst_chain;
reg sync_rst;

always @(posedge clk or negedge clk) begin
    if (!clk) begin
        sync_rst <= 1'b0;
        rst_chain <= {CHAINS{1'b0}};
    end else if (async_rst_i == RST_POL) begin
        sync_rst <= 1'b1;
        rst_chain <= {CHAINS{1'b1}};
    end else begin
        sync_rst <= rst_chain[CHAINS-1];
        rst_chain <= {rst_chain[CHAINS-2:0], 1'b0};
    end
end

assign sync_rst_o = sync_rst;

endmodule
```
