```verilog
module arbiter #
(
    parameter PORTS = 4,
    parameter ARB_TYPE_ROUND_ROBIN = 0,
    parameter ARB_BLOCK = 0,
    parameter ARB_BLOCK_ACK = 1,
    parameter ARB_LSB_HIGH_PRIORITY = 0
)
(
    input  wire                     clk,
    input  wire                     rst,
    input  wire [PORTS-1:0]         request,
    input  wire [PORTS-1:0]         acknowledge,
    output wire [PORTS-1:0]         grant,
    output wire                     grant_valid,
    output wire [$clog2(PORTS)-1:0] grant_encoded
);

reg [PORTS-1:0] grant_reg;
reg [$clog2(PORTS)-1:0] grant_encoded_reg;
reg grant_valid_reg;
reg [PORTS-1:0] priority;
reg [PORTS-1:0] priority_mask;

integer i;

always @(posedge clk or posedge rst) begin
    if (rst) begin
        grant_reg <= {PORTS{1'b0}};
        grant_encoded_reg <= {($clog2(PORTS)){1'b0}};
        grant_valid_reg <= 1'b0;
        priority <= {PORTS{1'b0}};
        priority_mask <= {PORTS{1'b1}};
    end else begin
        if (ARB_TYPE_ROUND_ROBIN) begin
            if (grant_valid_reg && (ARB_BLOCK_ACK ? acknowledge[grant_encoded_reg] : request[grant_encoded_reg] == 1'b0)) begin
                priority <= {priority[PORTS-2:0], priority[PORTS-1]};
            end
        end

        grant_valid_reg <= 1'b0;
        for (i = 0; i < PORTS; i = i + 1) begin
            if (request[i] && priority_mask[i]) begin
                if (ARB_LSB_HIGH_PRIORITY) begin
                    if (i == PORTS - 1) begin
                        grant_reg <= 1 << i;
                        grant_encoded_reg <= i;
                        grant_valid_reg <= 1'b1;
                    end else if (priority[i] == 1'b1) begin
                        grant_reg <= 1 << i;
                        grant_encoded_reg <= i;
                        grant_valid_reg <= 1'b1;
                    end
                end else begin
                    if (priority[i] == 1'b1) begin
                        grant_reg <= 1 << i;
                        grant_encoded_reg <= i;
                        grant_valid_reg <= 1'b1;
                    end else if (i == PORTS - 1) begin
                        grant_reg <= 1 << i;
                        grant_encoded_reg <= i;
                        grant_valid_reg <= 1'b1;
                    end
                end
            end
        end

        if (ARB_BLOCK && grant_valid_reg) begin
            if (ARB_BLOCK_ACK) begin
                priority_mask <= ~acknowledge;
            end else begin
                priority_mask <= ~request;
            end
        end else begin
            priority_mask <= {PORTS{1'b1}};
        end
    end
end

assign grant = grant_reg;
assign grant_valid = grant_valid_reg;
assign grant_encoded = grant_encoded_reg;

endmodule
```
