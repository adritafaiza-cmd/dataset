```systemverilog
module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

logic rst_sync_n;
logic rst_sync_n_d;
logic init_sync_n;
logic init_sync_n_d;

always_ff @(posedge clk_i or negedge rst_ni) begin
    if (!rst_ni) begin
        rst_sync_n <= 1'b0;
        rst_sync_n_d <= 1'b0;
        init_sync_n <= 1'b0;
        init_sync_n_d <= 1'b0;
    end else begin
        if (test_mode_i) begin
            rst_sync_n <= 1'b1;
            rst_sync_n_d <= 1'b1;
        end else begin
            rst_sync_n <= rst_ni;
            rst_sync_n_d <= rst_sync_n;
        end
        init_sync_n <= rst_sync_n_d;
        init_sync_n_d <= init_sync_n;
    end
end

assign rst_no = ~rst_sync_n_d;
assign init_no = init_sync_n_d;

endmodule
```
