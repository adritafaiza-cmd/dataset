```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    reg [WIDTH-1:0] fifo [2**LOG_DEPTH-1:0];
    reg [LOG_DEPTH-1:0] wr_ptr, rd_ptr;
    reg [LOG_DEPTH-1:0] gray_wr_ptr, gray_rd_ptr;
    reg [LOG_DEPTH-1:0] sync_wr_ptr [SYNC_STAGES-1:0];
    reg [LOG_DEPTH-1:0] sync_rd_ptr [SYNC_STAGES-1:0];
    reg [WIDTH-1:0] sync_data [SYNC_STAGES-1:0];
    reg src_clear_pending, dst_clear_pending;
    reg src_empty, dst_empty;
    reg src_full, dst_full;

    assign src_clear_pending_o = src_clear_pending;
    assign dst_clear_pending_o = dst_clear_pending;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_pending <= 1'b0;
            src_ready_o <= 1'b0;
            wr_ptr <= '0;
            gray_wr_ptr <= '0;
        end else if (src_clear_i) begin
            src_clear_pending <= 1'b1;
            src_ready_o <= 1'b0;
            wr_ptr <= '0;
            gray_wr_ptr <= '0;
        end else if (src_valid_i &&!src_full) begin
            src_ready_o <= 1'b1;
            fifo[wr_ptr] <= src_data_i;
            wr_ptr <= wr_ptr + 1'b1;
            gray_wr_ptr <= gray_wr_ptr + 1'b1;
        end else if (!src_valid_i) begin
            src_ready_o <= 1'b0;
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_pending <= 1'b0;
            dst_valid_o <= 1'b0;
            rd_ptr <= '0;
            gray_rd_ptr <= '0;
        end else if (dst_clear_i) begin
            dst_clear_pending <= 1'b1;
            dst_valid_o <= 1'b0;
            rd_ptr <= '0;
            gray_rd_ptr <= '0;
        end else if (dst_ready_i &&!dst_empty) begin
            dst_valid_o <= 1'b1;
            dst_data_o <= fifo[rd_ptr];
            rd_ptr <= rd_ptr + 1'b1;
            gray_rd_ptr <= gray_rd_ptr + 1'b1;
        end else if (!dst_ready_i) begin
            dst_valid_o <= 1'b0;
        end
    end

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            sync_wr_ptr[0] <= '0;
        end else if (src_clear_i) begin
            sync_wr_ptr[0] <= '0;
        end else begin
            sync_wr_ptr[0] <= gray_wr_ptr;
        end
    end

    genvar i;
    generate
        for (i = 1; i < SYNC_STAGES; i++) begin
            always @(posedge dst_clk_i or negedge dst_rst_ni) begin
                if (!dst_rst_ni) begin
                    sync_wr_ptr[i] <= '0;
                end else if (dst_clear_i) begin
                    sync_wr_ptr[i] <= '0;
                end else begin
                    sync_wr_ptr[i] <= sync_wr_ptr[i-1];
                end
            end
        end
    endgenerate

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            sync_rd_ptr[0] <= '0;
        end else if (dst_clear_i) begin
            sync_rd_ptr[0] <= '0;
        end else begin
            sync_rd_ptr[0] <= gray_rd_ptr;
        end
    end

    generate
        for (i = 1; i < SYNC_STAGES; i++) begin
            always @(posedge src_clk_i or negedge src_rst_ni) begin
                if (!src_rst_ni) begin
                    sync_rd_ptr[i] <= '0;
                end else if (src_clear_i) begin
                    sync_rd_ptr[i] <= '0;
                end else begin
                    sync_rd_ptr[i] <= sync_rd_ptr[i-1];
                end
            end
        end
    endgenerate

    assign src_empty = (wr_ptr == rd_ptr);
    assign dst_empty = (sync_wr_ptr[SYNC_STAGES-1] == sync_rd_ptr[SYNC_STAGES-1]);
    assign src_full = (wr_ptr == (rd_ptr + (1 << LOG_DEPTH)));
    assign dst_full = (sync_wr_ptr[SYNC_STAGES-1] == (sync_rd_ptr[SYNC_STAGES-1] + (1 << LOG_DEPTH)));

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_clear_pending <= 1'b0;
        end else if (src_clear_i) begin
            src_clear_pending <= 1'b1;
        end else if (src_empty) begin
            src_clear_pending <= 1'b0;
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_clear_pending <= 1'b0;
        end else if (dst_clear_i) begin
            dst_clear_pending <= 1'b1;
        end else if (dst_empty) begin
            dst_clear_pending <= 1'b0;
        end
    end

endmodule
```
