```verilog
module cdc_fifo_gray_clearable #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input                  src_clear_i,
    output                 src_clear_pending_o,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    input                  dst_clear_i,
    output                 dst_clear_pending_o,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    reg [WIDTH-1:0]         fifo [2**LOG_DEPTH-1:0];
    reg [LOG_DEPTH-1:0]    wr_ptr;
    reg [LOG_DEPTH-1:0]    rd_ptr;
    reg [LOG_DEPTH:0]      count;
    reg                    src_clear_pending;
    reg                    dst_clear_pending;
    reg                    src_ready;
    reg                    dst_valid;
    reg [WIDTH-1:0]         dst_data;

    wire                   src_clear;
    wire                   dst_clear;

    assign src_clear = src_clear_i;
    assign dst_clear = dst_clear_i;

    assign src_clear_pending_o = src_clear_pending;
    assign dst_clear_pending_o = dst_clear_pending;
    assign src_ready_o = src_ready;
    assign dst_data_o = dst_data;
    assign dst_valid_o = dst_valid;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (~src_rst_ni) begin
            src_clear_pending <= 1'b0;
            src_ready <= 1'b0;
            wr_ptr <= 'h0;
            count <= 'h0;
        end else if (src_clear) begin
            src_clear_pending <= 1'b1;
            src_ready <= 1'b0;
            wr_ptr <= 'h0;
            count <= 'h0;
        end else if (src_valid_i && src_ready) begin
            fifo[wr_ptr] <= src_data_i;
            wr_ptr <= wr_ptr + 1'b1;
            count <= count + 1'b1;
            src_ready <= (count == (2**LOG_DEPTH - 1)) ? 1'b0 : 1'b1;
        end else if (~src_valid_i && src_ready) begin
            src_ready <= (count == (2**LOG_DEPTH - 1)) ? 1'b0 : 1'b1;
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (~dst_rst_ni) begin
            dst_clear_pending <= 1'b0;
            dst_valid <= 1'b0;
            rd_ptr <= 'h0;
            count <= 'h0;
        end else if (dst_clear) begin
            dst_clear_pending <= 1'b1;
            dst_valid <= 1'b0;
            rd_ptr <= 'h0;
            count <= 'h0;
        end else if (dst_ready_i && dst_valid) begin
            rd_ptr <= rd_ptr + 1'b1;
            count <= count - 1'b1;
            dst_valid <= (count == 1'b1) ? 1'b0 : 1'b1;
        end else if (~dst_ready_i && dst_valid) begin
            dst_valid <= (count == 1'b1) ? 1'b0 : 1'b1;
        end
    end

    always @(posedge dst_clk_i) begin
        if (dst_valid) begin
            dst_data <= fifo[rd_ptr];
        end
    end

    always @(posedge src_clk_i) begin
        if (~src_clear) begin
            src_clear_pending <= 1'b0;
        end
    end

    always @(posedge dst_clk_i) begin
        if (~dst_clear) begin
            dst_clear_pending <= 1'b0;
        end
    end

    // CDC
    reg [SYNC_STAGES-1:0] src_valid_sync;
    reg [SYNC_STAGES-1:0] src_clear_sync;
    reg [SYNC_STAGES-1:0] dst_ready_sync;
    reg [SYNC_STAGES-1:0] dst_clear_sync;

    always @(posedge src_clk_i) begin
        src_valid_sync[0] <= src_valid_i;
        src_clear_sync[0] <= src_clear_i;
        for (int i = 1; i < SYNC_STAGES; i++) begin
            src_valid_sync[i] <= src_valid_sync[i-1];
            src_clear_sync[i] <= src_clear_sync[i-1];
        end
    end

    always @(posedge dst_clk_i) begin
        dst_ready_sync[0] <= dst_ready_i;
        dst_clear_sync[0] <= dst_clear_i;
        for (int i = 1; i < SYNC_STAGES; i++) begin
            dst_ready_sync[i] <= dst_ready_sync[i-1];
            dst_clear_sync[i] <= dst_clear_sync[i-1];
        end
    end

    assign src_ready = (count < (2**LOG_DEPTH - 1)) && ~src_clear_pending;
    assign dst_valid = (count > 0) && ~dst_clear_pending;

endmodule
```
