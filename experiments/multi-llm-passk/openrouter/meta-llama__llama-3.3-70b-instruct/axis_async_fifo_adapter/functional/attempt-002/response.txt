module axis_async_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = 0,
    parameter KEEP_WIDTH = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1
)
(
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [DATA_WIDTH-1:0]    s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]    s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [DATA_WIDTH-1:0]    m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]    m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    output wire                     empty,
    output wire                     full,
    output wire                     overflow,
    output wire                     bad_frame,
    output wire                     good_frame
);

reg [DATA_WIDTH-1:0] fifo_data[DEPTH-1:0];
reg [KEEP_WIDTH-1:0] fifo_keep[DEPTH-1:0];
reg [ID_WIDTH-1:0] fifo_id[DEPTH-1:0];
reg [DEST_WIDTH-1:0] fifo_dest[DEPTH-1:0];
reg [USER_WIDTH-1:0] fifo_user[DEPTH-1:0];
reg [DEPTH-1:0] fifo_count;
reg fifo_empty_reg;
reg fifo_full_reg;
reg fifo_overflow_reg;
reg fifo_bad_frame_reg;
reg fifo_good_frame_reg;

integer i;

always @(posedge s_clk)
begin
    if (s_rst)
    begin
        fifo_count <= 0;
        fifo_empty_reg <= 1'b1;
        fifo_full_reg <= 1'b0;
        fifo_overflow_reg <= 1'b0;
        fifo_bad_frame_reg <= 1'b0;
        fifo_good_frame_reg <= 1'b0;
    end
    else
    begin
        if (s_axis_tvalid &&!fifo_full_reg)
        begin
            fifo_data[fifo_count] <= s_axis_tdata;
            fifo_keep[fifo_count] <= s_axis_tkeep;
            fifo_id[fifo_count] <= s_axis_tid;
            fifo_dest[fifo_count] <= s_axis_tdest;
            fifo_user[fifo_count] <= s_axis_tuser;
            fifo_count <= fifo_count + 1;
            if (fifo_count == DEPTH - 1)
            begin
                fifo_full_reg <= 1'b1;
            end
            else
            begin
                fifo_full_reg <= 1'b0;
            end
            if (s_axis_tlast)
            begin
                fifo_good_frame_reg <= 1'b1;
            end
        end
        else if (s_axis_tvalid && fifo_full_reg)
        begin
            fifo_overflow_reg <= 1'b1;
        end
    end
end

always @(posedge m_clk)
begin
    if (m_rst)
    begin
        fifo_empty_reg <= 1'b1;
        fifo_full_reg <= 1'b0;
        fifo_overflow_reg <= 1'b0;
        fifo_bad_frame_reg <= 1'b0;
        fifo_good_frame_reg <= 1'b0;
    end
    else
    begin
        if (m_axis_tready &&!fifo_empty_reg)
        begin
            fifo_count <= fifo_count - 1;
            if (fifo_count == 0)
            begin
                fifo_empty_reg <= 1'b1;
            end
            else
            begin
                fifo_empty_reg <= 1'b0;
            end
            if (fifo_user[0] == {USER_WIDTH{1'b1}})
            begin
                fifo_bad_frame_reg <= 1'b1;
            end
        end
    end
end

assign s_axis_tready =!fifo_full_reg;
assign m_axis_tdata = fifo_data[0];
assign m_axis_tkeep = fifo_keep[0];
assign m_axis_tvalid =!fifo_empty_reg;
assign m_axis_tlast = fifo_user[0] == {USER_WIDTH{1'b1}};
assign m_axis_tid = fifo_id[0];
assign m_axis_tdest = fifo_dest[0];
assign m_axis_tuser = fifo_user[0];
assign empty = fifo_empty_reg;
assign full = fifo_full_reg;
assign overflow = fifo_overflow_reg;
assign bad_frame = fifo_bad_frame_reg;
assign good_frame = fifo_good_frame_reg;

endmodule

    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = 0,

)
(

    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    output wire                     empty,
    output wire                     full
);

reg [S_DATA_WIDTH-1:0] adapter_data;
reg [S_KEEP_WIDTH-1:0] adapter_keep;
reg adapter_valid;
reg adapter_ready;
reg adapter_last;
reg [ID_WIDTH-1:0] adapter_id;
reg [DEST_WIDTH-1:0] adapter_dest;
reg [USER_WIDTH-1:0] adapter_user;
reg adapter_empty;
reg adapter_full;

always @(posedge s_clk)
begin
    if (s_rst)
    begin
        adapter_empty <= 1'b1;
        adapter_full <= 1'b0;
    end
    else
    begin
        if (s_axis_tvalid &&!adapter_full)
        begin
            adapter_data <= s_axis_tdata;
            adapter_keep <= s_axis_tkeep;
            adapter_id <= s_axis_tid;
            adapter_dest <= s_axis_tdest;
            adapter_user <= s_axis_tuser;
            adapter_valid <= 1'b1;
            adapter_last <= s_axis_tlast;
            if (adapter_valid && m_axis_tready)
            begin
                adapter_valid <= 1'b0;
            end
        end
        else if (s_axis_tvalid && adapter_full)
        begin
            // Handle overflow
        end
    end
end

assign s_axis_tready =!adapter_full;
assign m_axis_tdata = adapter_data;
assign m_axis_tkeep = adapter_keep;
assign m_axis_tvalid = adapter_valid;
assign adapter_ready = m_axis_tready;
assign m_axis_tlast = adapter_last;
assign m_axis_tid = adapter_id;
assign m_axis_tdest = adapter_dest;
assign m_axis_tuser = adapter_user;
assign empty = adapter_empty;
assign full = adapter_full;

endmodule
