```verilog
module pulse_sync #(
    parameter STAGES = 2
)(
    input  clk_a,
    input  rstn_a,
    input  clk_b,
    input  rstn_b,
    input  pulseA_i,
    output pulseB_o,
    output busy_o
);

reg [STAGES-1:0] sync_reg;
reg busy_reg;
reg pulseB_reg;

always @(posedge clk_a or negedge rstn_a) begin
    if (!rstn_a) begin
        sync_reg <= {STAGES{1'b0}};
        busy_reg <= 1'b0;
    end else if (pulseA_i &&!busy_reg) begin
        sync_reg <= {1'b1, sync_reg[STAGES-1:1]};
        busy_reg <= 1'b1;
    end else if (!pulseA_i && busy_reg) begin
        sync_reg <= {1'b0, sync_reg[STAGES-1:1]};
    end
end

always @(posedge clk_b or negedge rstn_b) begin
    if (!rstn_b) begin
        pulseB_reg <= 1'b0;
    end else if (sync_reg[0]) begin
        pulseB_reg <= 1'b1;
    end else begin
        pulseB_reg <= 1'b0;
    end
end

assign pulseB_o = pulseB_reg;
assign busy_o = busy_reg;

endmodule
```
