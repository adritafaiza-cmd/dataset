```verilog
module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;

    reg [WIDTH-1:0]            fifo [DEPTH-1:0];
    reg [$clog2(DEPTH)-1:0]    wr_ptr;
    reg [$clog2(DEPTH)-1:0]    rd_ptr;
    reg [$clog2(DEPTH)-1:0]    gray_wr_ptr;
    reg [$clog2(DEPTH)-1:0]    gray_rd_ptr;
    reg                        full;
    reg                        empty;
    reg                        src_ready;
    reg                        dst_valid;
    reg [WIDTH-1:0]            dst_data;

    assign src_ready_o = src_ready;
    assign dst_data_o = dst_data;
    assign dst_valid_o = dst_valid;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (~src_rst_ni) begin
            wr_ptr <= 0;
            gray_wr_ptr <= 0;
            full <= 0;
            src_ready <= 1;
        end else if (src_valid_i && src_ready) begin
            fifo[wr_ptr] <= src_data_i;
            if (wr_ptr == (DEPTH - 1)) begin
                wr_ptr <= 0;
            end else begin
                wr_ptr <= wr_ptr + 1;
            end
            gray_wr_ptr <= (wr_ptr >> 1) ^ wr_ptr;
            if (wr_ptr == rd_ptr) begin
                full <= 1;
                src_ready <= 0;
            end else begin
                full <= 0;
                src_ready <= 1;
            end
        end
    end

    reg [SYNC_STAGES-1:0] gray_wr_ptr_sync;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (~dst_rst_ni) begin
            gray_wr_ptr_sync <= 0;
        end else begin
            gray_wr_ptr_sync[0] <= gray_wr_ptr;
            for (int i = 1; i < SYNC_STAGES; i++) begin
                gray_wr_ptr_sync[i] <= gray_wr_ptr_sync[i-1];
            end
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (~dst_rst_ni) begin
            gray_rd_ptr <= 0;
            empty <= 1;
            dst_valid <= 0;
        end else if (dst_ready_i && ~empty) begin
            dst_data <= fifo[gray_rd_ptr];
            if (gray_rd_ptr == (DEPTH - 1)) begin
                gray_rd_ptr <= 0;
            end else begin
                gray_rd_ptr <= gray_rd_ptr + 1;
            end
            if (gray_rd_ptr == gray_wr_ptr_sync[SYNC_STAGES-1]) begin
                empty <= 1;
                dst_valid <= 0;
            end else begin
                empty <= 0;
                dst_valid <= 1;
            end
        end
    end

    reg [SYNC_STAGES-1:0] gray_rd_ptr_sync;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (~src_rst_ni) begin
            gray_rd_ptr_sync <= 0;
        end else begin
            gray_rd_ptr_sync[0] <= gray_rd_ptr;
            for (int i = 1; i < SYNC_STAGES; i++) begin
                gray_rd_ptr_sync[i] <= gray_rd_ptr_sync[i-1];
            end
        end
    end

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (~src_rst_ni) begin
            rd_ptr <= 0;
        end else if (src_ready && ~full) begin
            if (rd_ptr == (DEPTH - 1)) begin
                rd_ptr <= 0;
            end else begin
                rd_ptr <= rd_ptr + 1;
            end
            if (rd_ptr == gray_rd_ptr_sync[SYNC_STAGES-1]) begin
                full <= 1;
                src_ready <= 0;
            end else begin
                full <= 0;
                src_ready <= 1;
            end
        end
    end

endmodule
```
