module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;

    reg [WIDTH-1:0] mem [DEPTH-1:0];
    reg [LOG_DEPTH-1:0] src_ptr;
    reg [LOG_DEPTH-1:0] dst_ptr;
    reg [LOG_DEPTH-1:0] src_ptr_gray;
    reg [LOG_DEPTH-1:0] dst_ptr_gray;
    reg [WIDTH-1:0] data_out;
    reg valid_out;
    reg src_ready;
    reg [SYNC_STAGES-1:0] src_valid_sync;
    reg [SYNC_STAGES-1:0] dst_ready_sync;

    assign src_ready_o = src_ready;
    assign dst_data_o = data_out;
    assign dst_valid_o = valid_out;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (~src_rst_ni) begin
            src_ptr <= 0;
            src_ptr_gray <= 0;
            src_valid_sync <= 0;
            src_ready <= 1;
        end else begin
            if (src_valid_i && src_ready) begin
                mem[src_ptr] <= src_data_i;
                src_ptr <= src_ptr + 1;
                src_ptr_gray <= gray_encode(src_ptr);
            end
            src_valid_sync[0] <= src_valid_i;
            for (int i = 1; i < SYNC_STAGES; i++) begin
                src_valid_sync[i] <= src_valid_sync[i-1];
            end
            if (src_valid_i && ~src_ready) begin
                src_ready <= 0;
            end else if (~src_valid_i && src_ready) begin
                src_ready <= 1;
            end
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (~dst_rst_ni) begin
            dst_ptr <= 0;
            dst_ptr_gray <= 0;
            dst_ready_sync <= 0;
            valid_out <= 0;
            data_out <= 0;
        end else begin
            dst_ready_sync[0] <= dst_ready_i;
            for (int i = 1; i < SYNC_STAGES; i++) begin
                dst_ready_sync[i] <= dst_ready_sync[i-1];
            end
            if (dst_ready_i && valid_out) begin
                dst_ptr <= dst_ptr + 1;
                dst_ptr_gray <= gray_encode(dst_ptr);
                valid_out <= 0;
            end
            if (dst_ptr_gray != src_ptr_gray) begin
                valid_out <= 1;
                data_out <= mem[dst_ptr];
            end
        end
    end

    function [LOG_DEPTH-1:0] gray_encode;
        input [LOG_DEPTH-1:0] bin;
        begin
            gray_encode = bin ^ (bin >> 1);
        end
    endfunction

endmodule
