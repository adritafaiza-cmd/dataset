```systemverilog
module cdc_4phase #(
  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  logic src_rst_ni,
  input  logic src_clk_i,
  input  logic [WIDTH-1:0] src_data_i,
  input  logic src_valid_i,
  output logic src_ready_o,
  input  logic dst_rst_ni,
  input  logic dst_clk_i,
  output logic [WIDTH-1:0] dst_data_o,
  output logic dst_valid_o,
  input  logic dst_ready_i
);

  logic [WIDTH-1:0] src_data_reg;
  logic src_valid_reg;
  logic src_ready_reg;
  logic [1:0] src_handshake;
  logic [1:0] dst_handshake;

  logic [WIDTH-1:0] dst_data_reg;
  logic dst_valid_reg;
  logic dst_ready_reg;

  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_reg <= '0;
      src_valid_reg <= 1'b0;
      src_ready_reg <= 1'b0;
      src_handshake <= 2'b00;
    end else begin
      if (src_valid_i && src_ready_o) begin
        src_data_reg <= src_data_i;
        src_valid_reg <= 1'b1;
        src_handshake <= 2'b01;
      end else if (src_handshake == 2'b01) begin
        src_handshake <= 2'b10;
      end else if (src_handshake == 2'b10) begin
        src_handshake <= 2'b00;
        src_valid_reg <= 1'b0;
      end
      src_ready_reg <= (src_handshake == 2'b00) && (dst_handshake == 2'b00);
    end
  end

  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg <= '0;
      dst_valid_reg <= 1'b0;
      dst_ready_reg <= 1'b0;
      dst_handshake <= 2'b00;
      if (SEND_RESET_MSG) begin
        dst_data_reg <= RESET_MSG;
        dst_valid_reg <= 1'b1;
      end
    end else begin
      if (dst_valid_reg && dst_ready_i) begin
        dst_valid_reg <= 1'b0;
        dst_handshake <= 2'b00;
      end else if (dst_handshake == 2'b00 && src_valid_reg) begin
        dst_data_reg <= src_data_reg;
        dst_valid_reg <= 1'b1;
        dst_handshake <= 2'b01;
      end else if (dst_handshake == 2'b01) begin
        dst_handshake <= 2'b10;
      end
      dst_ready_reg <= (dst_handshake == 2'b00);
    end
  end

  assign src_ready_o = src_ready_reg;
  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

endmodule
```
