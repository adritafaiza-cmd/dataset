`include "axis_async_fifo_adapter.vh"

module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    /*
     * AXI input
     */
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    /*
     * AXI output
     */
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    /*
     * Pause
     */
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam S_DATA_BYTES = (S_DATA_WIDTH+7)/8;
    localparam M_DATA_BYTES = (M_DATA_WIDTH+7)/8;

    localparam S_KEEP_WIDTH_ = (S_KEEP_ENABLE ? S_KEEP_WIDTH : 1);
    localparam M_KEEP_WIDTH_ = (M_KEEP_ENABLE ? M_KEEP_WIDTH : 1);

    localparam ID_WIDTH_ = (ID_ENABLE ? ID_WIDTH : 1);
    localparam DEST_WIDTH_ = (DEST_ENABLE ? DEST_WIDTH : 1);
    localparam USER_WIDTH_ = (USER_ENABLE ? USER_WIDTH : 1);

    localparam S_DATA_WIDTH_ = (S_DATA_WIDTH>0 ? S_DATA_WIDTH : 1);
    localparam M_DATA_WIDTH_ = (M_DATA_WIDTH>0 ? M_DATA_WIDTH : 1);

    localparam S_DATA_BYTES_ = (S_DATA_WIDTH_>0 ? (S_DATA_WIDTH_+7)/8 : 1);
    localparam M_DATA_BYTES_ = (M_DATA_WIDTH_>0 ? (M_DATA_WIDTH_+7)/8 : 1);

    localparam S_DATA_BYTES_MASK = (S_DATA_BYTES_==1 ? 1'b0 : (S_DATA_BYTES_-1));
    localparam M_DATA_BYTES_MASK = (M_DATA_BYTES_==1 ? 1'b0 : (M_DATA_BYTES_-1));

    localparam S_DATA_BYTES_LOG2 = $clog2(S_DATA_BYTES_);
    localparam M_DATA_BYTES_LOG2 = $clog2(M_DATA_BYTES_);

    localparam S_DATA_BYTES_LOG2_ = (S_DATA_BYTES_LOG2>0 ? S_DATA_BYTES_LOG2 : 1);
    localparam M_DATA_BYTES_LOG2_ = (M_DATA_BYTES_LOG2>0 ? M_DATA_BYTES_LOG2 : 1);

    localparam S_DATA_BYTES_MASK_ = (S_DATA_BYTES_LOG2_>0 ? (2**S_DATA_BYTES_LOG2_)-1 : 1'b0);
    localparam M_DATA_BYTES_MASK_ = (M_DATA_BYTES_LOG2_>0 ? (2**M_DATA_BYTES_LOG2_)-1 : 1'b0);

    localparam S_DATA_BYTES_MASK__ = (S_DATA_BYTES_MASK_>0 ? S_DATA_BYTES_MASK_ : 1'b0);
    localparam M_DATA_BYTES_MASK__ = (M_DATA_BYTES_MASK_>0 ? M_DATA_BYTES_MASK_ : 1'b0);

    localparam S_DATA_BYTES_MASK___ = (S_DATA_BYTES_MASK__>0 ? S_DATA_BYTES_MASK__ : 1'b0);
    localparam M_DATA_BYTES_MASK___ = (M_DATA_BYTES_MASK__>0 ? M_DATA_BYTES_MASK__ : 1'b0);

    localparam S_DATA_BYTES_MASK____ = (S_DATA_BYTES_MASK___>0 ? S_DATA_BYTES_MASK___ : 1'b0);
    localparam M_DATA_BYTES_MASK____ = (M_DATA_BYTES_MASK___>0 ? M_DATA_BYTES_MASK___ : 1'b0);

    localparam S_DATA_BYTES_MASK_____ = (S_DATA_BYTES_MASK____>0 ? S_DATA_BYTES_MASK____ : 1'b0);
    localparam M_DATA_BYTES_MASK_____ = (M_DATA_BYTES_MASK____>0 ? M_DATA_BYTES_MASK____ : 1'b0);

    localparam S_DATA_BYTES_MASK______ = (S_DATA_BYTES_MASK_____>0 ? S_DATA_BYTES_MASK_____ : 1'b0);
    localparam M_DATA_BYTES_MASK______ = (M_DATA_BYTES_MASK_____>0 ? M_DATA_BYTES_MASK_____ : 1'b0);

    localparam S_DATA_BYTES_MASK_______ = (S_DATA_BYTES_MASK______>0 ? S_DATA_BYTES_MASK______ : 1'b0);
    localparam M_DATA_BYTES_MASK_______ = (M_DATA_BYTES_MASK______>0 ? M_DATA_BYTES_MASK______ : 1'b0);
