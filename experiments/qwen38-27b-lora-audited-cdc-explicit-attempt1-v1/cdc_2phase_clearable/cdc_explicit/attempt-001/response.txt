// =============================================================================
// cdc_2phase_clearable
// =============================================================================
// A clearable two-phase valid/ready channel between asynchronous clocks.
//
// Architecture:
//   - Two-phase handshake (req/ack) with a single outstanding item.
//   - Multi-bit data is held in a source-domain register and sampled by the
//     destination only when the synchronized request is observed to rise.
//     This guarantees coherent data transfer without a Gray-coded FIFO.
//   - Clear is coordinated: both sides must acknowledge before the channel
//     returns to idle.  In-flight data is dropped.
//   - Reset and clear must not create a destination beat.
// =============================================================================

`default_nettype none

module cdc_2phase_clearable #(
  parameter WIDTH = 1,
  parameter SYNC_STAGES = 3,
  parameter CLEAR_ON_ASYNC_RESET = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  src_clear_i,
  output src_clear_pending_o,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,

  input  dst_rst_ni,
  input  dst_clk_i,
  input  dst_clear_i,
  output dst_clear_pending_o,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // ===========================================================================
  // Source domain
  // ===========================================================================

  // Handshake state
  reg req;
  reg ack_s;

  // Data register (held stable while req is high)
  reg [WIDTH-1:0] data_s;

  // Clear coordination
  reg clear_pending_s;
  reg clear_ack_s;

  // Synchronized destination signals
  wire ack_d;
  wire clear_ack_d;

  // ===========================================================================
  // Destination domain
  // ===========================================================================

  // Handshake state
  reg ack_d;
  reg req_s;

  // Data register (sampled on req rising edge)
  reg [WIDTH-1:0] data_d;

  // Clear coordination
  reg clear_pending_d;
  reg clear_ack_d;

  // Synchronized source signals
  wire req_s_sync;
  wire clear_ack_s;

  // ===========================================================================
  // Source domain logic
  // ===========================================================================

  // Synchronize destination ack and clear_ack into source domain
  (* ASYNC_REG = "TRUE" *)
  reg [SYNC_STAGES-1:0] ack_d_sync;
  (* ASYNC_REG = "TRUE" *)
  reg [SYNC_STAGES-1:0] clear_ack_d_sync;

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      ack_d_sync <= {SYNC_STAGES{1'b0}};
      clear_ack_d_sync <= {SYNC_STAGES{1'b0}};
    end else begin
      ack_d_sync <= {ack_d_sync[SYNC_STAGES-2:0], ack_d};
      clear_ack_d_sync <= {clear_ack_d_sync[SYNC_STAGES-2:0], clear_ack_d};
    end
  end

  assign ack_d = ack_d_sync[SYNC_STAGES-1];
  assign clear_ack_d = clear_ack_d_sync[SYNC_STAGES-1];

  // Source ready: channel is idle (no outstanding item) and not in clear
  assign src_ready_o = !req && !clear_pending_s;

  // Source clear pending: high until both sides have acknowledged the clear
  assign src_clear_pending_o = clear_pending_s;

  // Source domain state machine
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      req <= 1'b0;
      ack_s <= 1'b0;
      data_s <= {WIDTH{1'b0}};
      clear_pending_s <= 1'b0;
      clear_ack_s <= 1'b0;
    end else begin
      // Handle clear request
      if (src_clear_i && !clear_pending_s) begin
        clear_pending_s <= 1'b1;
        clear_ack_s <= 1'b0;
        req <= 1'b0;
        ack_s <= 1'b0;
      end else if (clear_pending_s) begin
        // Wait for destination to acknowledge the clear
        if (clear_ack_d) begin
          clear_pending_s <= 1'b0;
          clear_ack_s <= 1'b0;
        end else if (clear_ack_s) begin
          clear_ack_s <= 1'b0;
        end
      end else begin
        // Normal operation
        if (!req && src_valid_i) begin
          req <= 1'b1;
          data_s <= src_data_i;
        end else if (req && ack_d) begin
          req <= 1'b0;
          ack_s <= 1'b0;
        end
      end
    end
  end

  // ===========================================================================
  // Destination domain logic
  // ===========================================================================

  // Synchronize source req and clear_ack into destination domain
  (* ASYNC_REG = "TRUE" *)
  reg [SYNC_STAGES-1:0] req_s_sync;
  (* ASYNC_REG = "TRUE" *)
  reg [SYNC_STAGES-1:0] clear_ack_s_sync;

  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      req_s_sync <= {SYNC_STAGES{1'b0}};
      clear_ack_s_sync <= {SYNC_STAGES{1'b0}};
    end else begin
      req_s_sync <= {req_s_sync[SYNC_STAGES-2:0], req};
      clear_ack_s_sync <= {clear_ack_s_sync[SYNC_STAGES-2:0], clear_ack_s};
    end
  end

  assign req_s = req_s_sync[SYNC_STAGES-1];
  assign clear_ack_s = clear_ack_s_sync[SYNC_STAGES-1];

  // Destination valid: data is available and not in clear
  assign dst_valid_o = req_s && !clear_pending_d;

  // Destination data
  assign dst_data_o = data_d;

  // Destination clear pending: high until both sides have acknowledged the clear
  assign dst_clear_pending_o = clear_pending_d;

  // Destination domain state machine
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      ack_d <= 1'b0;
      req_s <= 1'b0;
      data_d <= {WIDTH{1'b0}};
      clear_pending_d <= 1'b0;
      clear_ack_d <= 1'b0;
    end else begin
      // Handle clear request
      if (dst_clear_i && !clear_pending_d) begin
        clear_pending_d <= 1'b1;
        clear_ack_d <= 1'b0;
        ack_d <= 1'b0;
        req_s <= 1'b0;
      end else if (clear_pending_d) begin
        // Wait for source to acknowledge the clear
        if (clear_ack_s) begin
          clear_pending_d <= 1'b0;
          clear_ack_d <= 1'b0;
        end else if (clear_ack_d) begin
          clear_ack_d <= 1'b0;
        end
      end else begin
        // Normal operation
        if (!req_s && req_s_sync[SYNC_STAGES-2]) begin
          // Detect rising edge of req to sample data
          req_s <= 1'b1;
          data_d <= data_s;
          ack_d <= 1'b1;
        end else if (req_s && dst_ready_i) begin
          req_s <= 1'b0;
          ack_d <= 1'b0;
        end
      end
    end
  end

endmodule

`default_nettype wire<|im_end|>
