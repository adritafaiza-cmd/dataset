// =============================================================================
// SPI Master with Parallel-Side Word Interface
// =============================================================================
//
// This module implements an SPI master that transfers N-bit words. It
// features a parallel-side interface for data input/output and a handshake
// mechanism for flow control.
//
// SPI Mode:
//   - Mode 0: CPOL=0, CPHA=0 (SCK idle low, data sampled on rising edge)
//   - MSB first
//
// Parallel Interface:
//   - di_i: Input data (N bits)
//   - wren_i: Write enable (active high)
//   - wr_ack_o: Write acknowledge (active high, one cycle)
//   - do_o: Output data (N bits)
//   - do_valid_o: Output data valid (active high, one cycle)
//   - di_req_o: Data input request (active high, one cycle)
//
// Clock Domains:
//   - sclk_i: SPI clock domain (for SPI signal generation)
//   - pclk_i: Parallel interface clock domain (for handshake)
//
// =============================================================================

`timescale 1ns / 1ps

module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,

    // SPI outputs
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,

    // SPI input
    input              spi_miso_i,

    // Parallel interface
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,

    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

    // =========================================================================
    // Reset Synchronization (for safe reset deassertion)
    // =========================================================================

    // Synchronize reset for sclk domain
    reg [1:0] rst_sync_sclk;
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i)
            rst_sync_sclk <= 2'b11;
        else
            rst_sync_sclk <= {rst_sync_sclk[0], 1'b0};
    end
    wire rst_sclk = rst_sync_sclk[1];

    // Synchronize reset for pclk domain
    reg [1:0] rst_sync_pclk;
    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i)
            rst_sync_pclk <= 2'b11;
        else
            rst_sync_pclk <= {rst_sync_pclk[0], 1'b0};
    end
    wire rst_pclk = rst_sync_pclk[1];

    // =========================================================================
    // SPI State Machine (sclk domain)
    // =========================================================================

    localparam S_IDLE    = 2'd0;
    localparam S_START   = 2'd1;
    localparam S_SHIFT   = 2'd2;
    localparam S_FINISH  = 2'd3;

    reg [1:0]  state;
    reg [N-1:0] tx_shift;
    reg [N-1:0] rx_shift;
    reg [N-1:0] bit_count;

    // Data transfer signals
    reg         tx_data_valid;
    reg [N-1:0] tx_data;
    reg         rx_data_valid;
    reg [N-1:0] rx_data;

    // =========================================================================
    // SPI Clock Generation (sclk domain)
    // =========================================================================

    reg [SPI_2X_CLK_DIV-1:0] clk_div;
    reg clk_phase;

    always @(posedge sclk_i or posedge rst_sclk) begin
        if (rst_sclk) begin
            clk_div   <= {SPI_2X_CLK_DIV{1'b0}};
            clk_phase <= 1'b0;
        end else begin
            if (clk_div == SPI_2X_CLK_DIV - 1) begin
                clk_div   <= {SPI_2X_CLK_DIV{1'b0}};
                clk_phase <= ~clk_phase;
            end else begin
                clk_div <= clk_div + 1'b1;
            end
        end
    end

    // =========================================================================
    // SPI State Machine (sclk domain)
    // =========================================================================

    always @(posedge sclk_i or posedge rst_sclk) begin
        if (rst_sclk) begin
            state         <= S_IDLE;
            spi_ssel_o    <= 1'b1;  // Deselected (active low)
            spi_sck_o     <= 1'b0;
            tx_shift      <= {N{1'b0}};
            rx_shift      <= {N{1'b0}};
            bit_count     <= {N{1'b0}};
            tx_data_valid <= 1'b0;
            tx_data       <= {N{1'b0}};
            rx_data_valid <= 1'b0;
            rx_data       <= {N{1'b0}};
        end else begin
            case (state)
                S_IDLE: begin
                    spi_ssel_o <= 1'b1;
                    spi_sck_o  <= 1'b0;
                    if (tx_data_valid) begin
                        state      <= S_START;
                        tx_shift   <= tx_data;
                        rx_shift   <= {N{1'b0}};
                        bit_count  <= {N{1'b0}};
                        tx_data_valid <= 1'b0;
                    end
                end

                S_START: begin
                    spi_ssel_o <= 1'b0;  // Select slave
                    spi_sck_o  <= 1'b0;
                    state      <= S_SHIFT;
                end

                S_SHIFT: begin
                    if (clk_phase) begin
                        // On rising edge: shift out MSB, shift in MISO
                        spi_sck_o  <= 1'b1;
                        tx_shift   <= {tx_shift[N-2:0], 1'b0};
                        rx_shift   <= {spi_miso_i, rx_shift[N-1:1]};
                        bit_count  <= bit_count + 1'b1;

                        if (bit_count == N - 1) begin
                            state <= S_FINISH;
                        end
                    end else begin
                        // On falling edge: prepare next bit
                        spi_sck_o <= 1'b0;
                        tx_shift  <= {tx_shift[N-2:0], 1'b0};
                    end
                end

                S_FINISH: begin
                    spi_ssel_o <= 1'b1;  // Deselect slave
                    spi_sck_o  <= 1'b0;
                    rx_data       <= rx_shift;
                    rx_data_valid <= 1'b1;
                    state         <= S_IDLE;
                end

                default: state <= S_IDLE;
            endcase
        end
    end

    // MOSI output (MSB of shift register)
    assign spi_mosi_o = tx_shift[N-1];

    // =========================================================================
    // Parallel Interface (pclk domain)
    // =========================================================================

    // Handshake signals
    reg         di_req_int;
    reg         wr_ack_int;
    reg         do_valid_int;
    reg [N-1:0] do_int;

    // Data transfer to SPI (pclk -> sclk)
    reg         tx_req;
    reg [N-1:0] tx_data_pclk;

    // Data from SPI (sclk -> pclk)
    reg         rx_valid_pclk;
    reg [N-1:0] rx_data_pclk;

    // =========================================================================
    // CDC: pclk -> sclk (Data transfer request)
    // =========================================================================

    // Use a simple handshake protocol for CDC
    reg         tx_req_sync;
    reg [N-1:0] tx_data_sync;

    // In pclk domain: send data
    always @(posedge pclk_i or posedge rst_pclk) begin
        if (rst_pclk) begin
            tx_req       <= 1'b0;
            tx_data_pclk <= {N{1'b0}};
            di_req_int   <= 1'b0;
            wr_ack_int   <= 1'b0;
            do_valid_int <= 1'b0;
            do_int       <= {N{1'b0}};
        end else begin
            // Request data when SPI is ready
            if (tx_req_sync == 1'b0 && tx_data_valid == 1'b0) begin
                di_req_int <= 1'b1;
            end else begin
                di_req_int <= 1'b0;
            end

            // Accept data when requested
            if (di_req_int && wren_i) begin
                tx_req       <= 1'b1;
                tx_data_pclk <= di_i;
                wr_ack_int   <= 1'b1;
            end else begin
                wr_ack_int <= 1'b0;
            end

            // Present received data
            if (rx_valid_pclk) begin
                do_valid_int <= 1'b1;
                do_int       <= rx_data_pclk;
            end else begin
                do_valid_int <= 1'b0;
            end
        end
    end

    // In sclk domain: receive data
    always @(posedge sclk_i or posedge rst_sclk) begin
        if (rst_sclk) begin
            tx_req_sync  <= 1'b0;
            tx_data_sync <= {N{1'b0}};
        end else begin
            if (tx_req && tx_req_sync == 1'b0) begin
                tx_req_sync  <= 1'b1;
                tx_data_sync <= tx_data_pclk;
                tx_data      <= tx_data_pclk;
                tx_data_valid <= 1'b1;
            end else if (tx_data_valid == 1'b0) begin
                tx_req_sync <= 1'b0;
            end
        end
    end

    // =========================================================================
    // CDC: sclk -> pclk (Data received)
    // =========================================================================

    // Use a simple handshake protocol for CDC
    reg         rx_valid_sync;
    reg [N-1:0] rx_data_sync;

    // In sclk domain: send data
    always @(posedge sclk_i or posedge rst_sclk) begin
        if (rst_sclk) begin
            rx_valid_sync <= 1'b0;
            rx_data_sync  <= {N{1'b0}};
        end else begin
            if (rx_data_valid && rx_valid_sync == 1'b0) begin
                rx_valid_sync <= 1'b1;
                rx_data_sync  <= rx_data;
                rx_data_valid <= 1'b0;
            end else if (rx_valid_pclk == 1'b0) begin
                rx_valid_sync <= 1'b0;
            end
        end
    end

    // In pclk domain: receive data
    always @(posedge pclk_i or posedge rst_pclk) begin
        if (rst_pclk) begin
            rx_valid_pclk <= 1'b0;
            rx_data_pclk  <= {N{1'b0}};
        end else begin
            if (rx_valid_sync && rx_valid_pclk == 1'b0) begin
                rx_valid_pclk <= 1'b1;
                rx_data_pclk  <= rx_data_sync;
            end else if (do_valid_int == 1'b0) begin
                rx_valid_pclk <= 1'b0;
            end
        end
    end

    // =========================================================================
    // Output assignments
    // =========================================================================

    assign di_req_o   = di_req_int;
    assign wr_ack_o   = wr_ack_int;
    assign do_valid_o = do_valid_int;
    assign do_o       = do_int;

endmodule<|im_end|>
