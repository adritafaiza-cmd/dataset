module sync #(
    parameter int unsigned STAGES = 2,
    parameter bit ResetValue = 1'b0
) (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic serial_i,

                    end
                end
            end else begin

                    end else begin
                        sync_reg[i] <= sync_reg[i-1];
