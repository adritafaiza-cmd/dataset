                    {dst_pwrite_reg, dst_paddr_reg, dst_pwdata_reg, dst_pstrb_reg, dst_pprot_reg} <= req_fifo_dout;

                    dst_penable_reg <= 0;
                    dst_state <= DST_SETUP;
                end
            end
            DST_SETUP: begin
                // Setup phase: psel=1, penable=0
                // Next state: ACCESS
                dst_penable_reg <= 1;
                dst_state <= DST_ACCESS;
            end
            DST_ACCESS: begin
                if (dst_pready_i) begin
                    // Transfer done
                    resp_prdata_reg <= dst_prdata_i;
                    resp_pslverr_reg <= dst_pslverr_i;
                    // Write to response FIFO
                    resp_fifo_wr_en <= 1;
                    dst_psel_reg <= 0;
                    dst_penable_reg <= 0;
                    dst_state <= DST_IDLE;
                end
            end
        endcase
    end
