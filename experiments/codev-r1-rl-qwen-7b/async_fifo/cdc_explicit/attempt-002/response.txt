    parameter FALLTHROUGH = "TRUE"

    input wire [DSIZE-1:0] wdata,
    output wire wfull,
    output wire awfull,
    input wire rclk,
    input wire rrst_n,
    input wire rinc,
    output wire [DSIZE-1:0] rdata,
    output wire rempty,
    output wire arempty
);

function [ASIZE:0] bin2gray;

    bin2gray = bin ^ (bin >> 1);
end

        rptr_gray_sync1 <= 0;
        rptr_gray_sync2 <= 0;
    end else begin
        rptr_gray_sync1 <= rptr_gray;
        rptr_gray_sync2 <= rptr_gray_sync1;
    end
end

always @(posedge rclk or negedge rrst_n) begin
    if (!rrst_n) begin
        wptr_gray_sync1 <= 0;
        wptr_gray_sync2 <= 0;
    end else begin
        wptr_gray_sync1 <= wptr_gray;
        wptr_gray_sync2 <= wptr_gray_sync1;
    end
end

    if (!wrst_n) begin
        wptr_bin <= 0;
    end else begin

        end
    end
