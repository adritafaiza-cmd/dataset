module async_fifo #(
    parameter DSIZE = 8,
    parameter ASIZE = 4,
    parameter FALLTHROUGH = "TRUE"
) (
    input wire wclk,
    input wire wrst_n,
    input wire winc,
    input wire [DSIZE-1:0] wdata,
    output wire wfull,
    output wire awfull,
    input wire rclk,
    input wire rrst_n,
    input wire rinc,
    output wire [DSIZE-1:0] rdata,
    output wire rempty,
    output wire arempty
);

    // Local parameters
    localparam FIFO_DEPTH = 1 << ASIZE;

    // Write domain signals
    reg [ASIZE:0] wptr_bin; // binary pointer (ASIZE+1 bits)
    wire [ASIZE:0] wgray; // Gray code of write pointer

    // Read domain signals
    reg [ASIZE:0] rptr_bin; // binary pointer (ASIZE+1 bits)
    wire [ASIZE:0] rgray; // Gray code of read pointer

    // Synchronization registers for read pointer in write domain
    reg [ASIZE:0] rgray_sync1, rgray_sync2;

    // Synchronization registers for write pointer in read domain
    reg [ASIZE:0] wgray_sync1, wgray_sync2;

    // Convert Gray to binary in write domain
    wire [ASIZE:0] rptr_bin_sync;

    // Convert Gray to binary in read domain
    wire [ASIZE:0] wptr_bin_sync;

    // Memory array
    reg [DSIZE-1:0] mem [0:FIFO_DEPTH-1];

    // Write pointer logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr_bin <= 0;
        end else begin
            if (winc && !wfull) begin
                wptr_bin <= wptr_bin + 1;
            end
        end
    end

    // Convert binary to Gray code for write pointer
    assign wgray = (wptr_bin >> 1) ^ wptr_bin;

    // Synchronize read pointer to write domain
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rgray_sync1 <= 0;
            rgray_sync2 <= 0;
        end else begin
            rgray_sync1 <= rgray;
            rgray_sync2 <= rgray_sync1;
        end
    end

    // Convert synchronized read pointer (Gray) to binary
    function [ASIZE:0] gray2bin;
        input [ASIZE:0] gray;
        integer i;
        begin
            gray2bin[ASIZE] = gray[ASIZE];
            for (i = ASIZE-1; i >=0; i = i-1)
                gray2bin[i] = gray2bin[i+1] ^ gray[i];
        end
    endfunction

    assign rptr_bin_sync = gray2bin(rgray_sync2);

    // Full condition: (wptr_bin - rptr_bin_sync) == FIFO_DEPTH
    wire full;
    assign full = ( (wptr_bin - rptr_bin_sync) == FIFO_DEPTH );

    // Almost full: >= FIFO_DEPTH -1
    wire awfull;
    assign awfull = ( (wptr_bin - rptr_bin_sync) >= (FIFO_DEPTH -1) );

    // Read pointer logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr_bin <= 0;
        end else begin
            if (rinc && !rempty) begin
                rptr_bin <= rptr_bin + 1;
            end
        end
    end

    // Convert binary to Gray code for read pointer
    assign rgray = (rptr_bin >> 1) ^ rptr_bin;

    // Synchronize write pointer to read domain
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wgray_sync1 <= 0;
            wgray_sync2 <= 0;
        end else begin
            wgray_sync1 <= wgray;
            wgray_sync2 <= wgray_sync1;
        end
    end

    // Convert synchronized write pointer (Gray) to binary
    assign wptr_bin_sync = gray2bin(wgray_sync2);

    // Empty condition: pointers equal
    wire empty;
    assign empty = (wptr_bin_sync == rptr_bin);

    // Almost empty: <= 1 entry left
    wire arempty;
    assign arempty = ( (wptr_bin_sync - rptr_bin) <= 1 );

    // Write data to memory
    always @(posedge wclk) begin
        if (winc && !wfull) begin
            mem[wptr_bin[ASIZE-1:0]] <= wdata;
        end
    end

    // Read data from memory
    generate
        if (FALLTHROUGH == "TRUE") begin
            // FWFT mode: data is available immediately
            assign rdata = mem[rptr_bin[ASIZE-1:0]];
        end else begin
            // Non-FWFT mode: data is registered
            reg [DSIZE-1:0] rdata_reg;
            always @(posedge rclk) begin
                if (rinc && !rempty) begin
                    rdata_reg <= mem[rptr_bin[ASIZE-1:0]];
                end
            end
            assign rdata = rdata_reg;
        end
    endgenerate

    // Assign outputs
    assign wfull = full;
    assign awfull = awfull;
    assign rempty = empty;
    assign arempty = arempty;
