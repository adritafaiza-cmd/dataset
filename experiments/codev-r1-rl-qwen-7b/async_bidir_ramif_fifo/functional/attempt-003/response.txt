    wire a_to_b_full = (a_to_b_wptr[ASIZE] != a_to_b_rptr[ASIZE]) &&

    // Almost full/empty
    wire [ASIZE:0] a_to_b_diff = a_to_b_wptr - a_to_b_rptr;
