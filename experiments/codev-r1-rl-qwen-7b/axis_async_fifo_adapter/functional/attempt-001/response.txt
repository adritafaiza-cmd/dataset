// On m_clk:

      buffer_count <= 0;

      m_valid <= 0;

      // ... other resets

   end else begin

      if (m_axis_tready && m_valid) begin

         // Acknowledge, reset buffer.

         m_valid <= 0;

         buffer_count <= 0;

      end else if (!empty) begin

         // Read from FIFO.

         buffer[buffer_count] <= fifo[rptr];

         buffer_count <= buffer_count + 1;

         if (buffer_count == RATIO - 1) begin

             // Now buffer is full.

             m_data <= {buffer[0], buffer[1], ...}; // depends on RATIO.

             m_valid <= 1;

             rptr <= rptr + 1;

             buffer_count <= 0;

         end

      end

   end

end
