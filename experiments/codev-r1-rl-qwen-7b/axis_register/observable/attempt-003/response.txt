            // So the code can proceed to assign the KEEP signals regardless, but when KEEP_ENABLE is 0, the code can assign 0.

                            skid_last <= 0;
                            skid_id <= 0;

                                    dest_reg <= DEST_ENABLE ? s_axis_tdest : {DEST_WIDTH{1'b0}};
                                    user_reg <= USER_ENABLE ? s_axis_tuser : {USER_WIDTH{1'b0}};

                                    skid_last <= LAST_ENABLE ? s_axis_tlast : 1'b0;
                                    skid_id <= ID_ENABLE ? s_axis_tid : {ID_WIDTH{1'b0}};
                                    skid_dest <= DEST_ENABLE ? s_axis_tdest : {DEST_WIDTH{1'b0}};
                                    skid_user <= USER_ENABLE ? s_axis_tuser : {USER_WIDTH{1'b0}};

                                end

                                data_reg <= skid_data;
                                keep_reg <= skid_keep;
                                last_reg <= skid_last;
                                id_reg <= skid_id;
                                dest_reg <= skid_dest;
                                user_reg <= skid_user;

                    assign m_axis_tid = id_reg;
