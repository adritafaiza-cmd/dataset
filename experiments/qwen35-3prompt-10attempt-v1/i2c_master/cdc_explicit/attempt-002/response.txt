                    tx_last = s_axis_data_tlast;
                    data_valid = 1'b0;
                    data_ready = 1'b0;
                    sm_next_state = 4'b0101; // DATA_TX_BYTE

                    bus_control_int = 1'b0;
                    bus_active_int = 0;

                sda_dir = 1'b1;
                scl_high = 1'b1;

                    scl_high = 1'b0;
                    scl_cnt = 0;

                            sda_high = 1'b0;
                        end else begin
                            // Next bit is 1, release SDA
                            sda_high = 1'b1;

                            // More bytes to send

                        end

                sda_high = 1'b1;

                    scl_high = 1'b0;
                    scl_cnt = 0;

                        ack_received = 1'b0;
                        missed_ack_int = 1'b1;
                    end
                    sm_next_state = 4'b0100; // DATA_TX_START (loop for more bytes)

                bus_control_int = 1'b1;

        endcase

            sda_cnt <= 0;

            sda_high <= 1'b0;
            scl_dir <= 1'b0;
            sda_dir <= 1'b0;
            scl_val <= 1'b0;
            sda_val <= 1'b0;
            ack_received <= 1'b0;

            bus_active_int <= 1'b0;

                            m_axis_data_tlast <= 1'b1;
                            m_axis_data_tdata <= rx_data_reg;
                            sm_next_state <= 4'b0000; // IDLE
                        end else begin
                            sm_next_state <= 4'b0000; // IDLE
                        end
                    end
                    
                4'b0012: // STOP_WAIT
                    scl_dir <= 1'b1;
                    sda_dir <= 1'b1;
                    scl_high <= 1'b1;
                    sda_high <= 1'b1;
                    if (scl_cnt >= SCL_PERIOD && scl_high) begin
                        scl_high <= 1'b0;
                        scl_cnt <= 0;
                    end
                    if (scl_cnt >= SCL_PERIOD && !scl_high) begin
                        sda_high <= 1'b1;
                        sda_cnt <= 0;
                        if (sda_cnt >= SDA_PERIOD && sda_high) begin
                            sm_next_state <= 4'b0000; // IDLE
                        end
                    end
                    
                default:
                    scl_dir <= 1'b0;
                    sda_dir <= 1'b0;
                    scl_high <= 1'b0;
                    sda_high <= 1'b0;
                    scl_cnt <= 0;
                    sda_cnt <= 0;
            endcase
        end
    end

    // ------------------------------------------------------------------------
    // Handshake Logic for Data
    // ------------------------------------------------------------------------
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            data_valid <= 1'b0;
            data_ready <= 1'b1;
            rx_valid <= 1'b0;
            rx_last <= 1'b0;
            m_axis_data_tvalid <= 1'b0;
            m_axis_data_tlast <= 1'b0;
            m_axis_data_tdata <= 0;
        end else begin
            // Data TX Handshake
            if (sm_state == 4'b0104) begin // DATA_RX_START
                if (data_valid && data_ready) begin
                    data_valid <= 1'b0;
                    data_ready <= 1'b0;
                end
                if (!data_valid && data_ready) begin
                    data_ready <= 1'b1;
                end
            end
            
            // Received Data Logic
            if (sm_state == 4'b0104) begin // ACK_CHECK_RX
                if (ack_received) begin
                    rx_valid <= 1'b1;
                    rx_last <= 1'b1;
                    m_axis_data_tvalid <= 1'b1;
                    m_axis_data_tlast <= 1'b1;
                    m_axis_data_tdata <= rx_data_reg;
                end else begin
                    rx_valid <= 1'b0;
                    m_axis_data_tvalid <= 1'b0;
                    m_axis_data_tlast <= 1'b0;
                    m_axis_data_tdata <= 0;
                end
            end
            
            // Output Data Handshake
            if (m_axis_data_tvalid && !m_axis_data_tready) begin
                m_axis_data_tvalid <= 1'b0;
                m_axis_data_tlast <= 1'b0;
                m_axis_data_tdata <= 0;
            end
            if (m_axis_data_tready && m_axis_data_tvalid) begin
                m_axis_data_tvalid <= 1'b0;
                m_axis_data_tlast <= 1'b0;
                m_axis_data_tdata <= 0;
            end
        end
    end

endmodule
