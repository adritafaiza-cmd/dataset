    wire [1:0]            m_bresp, m_rresp;

            s_wvalid_cdc <= 1'b0;

            else if (s_wvalid && s_wready)
                s_awvalid_cdc <= 1'b0;

                s_awvalid_cdc <= s_awvalid;
            
            if (s_wvalid && s_wready)
                s_wvalid_cdc <= 1'b1;
            else
                s_wvalid_cdc <= s_wvalid;
        end

            m_awprot_cdc  <= 3'd0;
            m_wdata_cdc   <= {DATA_WIDTH{1'b0}};
            m_wstrb_cdc   <= {STRB_WIDTH{1'b0}};
        end else begin
            m_awvalid_cdc <= s_awvalid_cdc;
            m_wvalid_cdc  <= s_wvalid_cdc;
            m_awaddr_cdc  <= s_awaddr_cdc;
            m_awprot_cdc  <= s_awprot_cdc;
            m_wdata_cdc   <= s_wdata_cdc;
            m_wstrb_cdc   <= s_wstrb_cdc;
        end
    end

    // ------------------------------------------------------------------
    // CDC Logic: Slave -> Master (Read Path)
    // ------------------------------------------------------------------
    always @(posedge s_clk or posedge s_rst) begin

            s_araddr_cdc <= {ADDR_WIDTH{1'b0}};
            s_arprot_cdc <= 3'd0;
            s_arvalid_cdc <= 1'b0;
            s_rdata_cdc <= {DATA_WIDTH{1'b0}};
            s_rresp_cdc <= 2'd0;
            s_rvalid_cdc <= 1'b0;
        end else begin
            s_araddr_cdc <= s_araddr;
            s_arprot_cdc <= s_arprot;
            if (s_arvalid && s_arready)
                s_arvalid_cdc <= 1'b1;
            else if (s_rvalid && s_rready)
                s_arvalid_cdc <= 1'b0;
            else
                s_arvalid_cdc <= s_arvalid;
            
            if (s_rvalid && s_rready) begin
                s_rdata_cdc <= s_rdata;
                s_rresp_cdc <= s_rresp;
                s_rvalid_cdc <= 1'b1;
            end else begin
                s_rdata_cdc <= s_rdata;
                s_rresp_cdc <= s_rresp;
                s_rvalid_cdc <= s_rvalid;
            end
        end
    end

    always @(posedge m_clk or posedge m_rst) begin

            m_arvalid_cdc <= 1'b0;
            m_rvalid_cdc  <= 1'b0;
            m_araddr_cdc  <= {ADDR_WIDTH{1'b0}};
            m_arprot_cdc  <= 3'd0;
            m_rdata_cdc   <= {DATA_WIDTH{1'b0}};
            m_rresp_cdc   <= 2'd0;
        end else begin
            m_arvalid_cdc <= s_arvalid_cdc;
            m_rvalid_cdc  <= s_rvalid_cdc;
            m_araddr_cdc  <= s_araddr_cdc;
            m_arprot_cdc  <= s_arprot_cdc;
            m_rdata_cdc   <= s_rdata_cdc;
            m_rresp_cdc   <= s_rresp_cdc;
        end
    end

    // ------------------------------------------------------------------
    // CDC Logic: Master -> Slave (Write Response Path)
    // ------------------------------------------------------------------
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_bresp_cdc <= 2'd0;
            s_bvalid_cdc <= 1'b0;
        end else begin
            if (m_bvalid && m_bready)
                s_bvalid_cdc <= 1'b1;
            else
                s_bvalid_cdc <= m_bvalid;
            s_bresp_cdc <= m_bresp;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_bready <= 1'b0;
        end else begin
            m_bready <= s_bvalid_cdc;
        end
    end

    // ------------------------------------------------------------------
    // CDC Logic: Master -> Slave (Read Response Path)
    // ------------------------------------------------------------------
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_rdata_cdc <= {DATA_WIDTH{1'b0}};
            s_rresp_cdc <= 2'd0;
            s_rvalid_cdc <= 1'b0;
        end else begin
            if (m_rvalid && m_rready)
                s_rvalid_cdc <= 1'b1;
            else
                s_rvalid_cdc <= m_rvalid;
            s_rdata_cdc <= m_rdata;
            s_rresp_cdc <= m_rresp;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_rready <= 1'b0;

            m_rready <= s_rvalid_cdc;
        end
    end

                end
