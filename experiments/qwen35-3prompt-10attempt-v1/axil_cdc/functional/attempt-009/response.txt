module axil_cdc #
(
    parameter DATA_WIDTH = 32,
    parameter ADDR_WIDTH = 32,
    parameter STRB_WIDTH = (DATA_WIDTH/8)
)
(
    /*
     * AXI lite slave interface
     */
    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [ADDR_WIDTH-1:0]  s_axil_awaddr,
    input  wire [2:0]             s_axil_awprot,
    input  wire                   s_axil_awvalid,
    output wire                   s_axil_awready,
    input  wire [DATA_WIDTH-1:0]  s_axil_wdata,
    input  wire [STRB_WIDTH-1:0]  s_axil_wstrb,
    input  wire                   s_axil_wvalid,
    output wire                   s_axil_wready,
    output wire [1:0]             s_axil_bresp,
    output wire                   s_axil_bvalid,
    input  wire                   s_axil_bready,
    input  wire [ADDR_WIDTH-1:0]  s_axil_araddr,
    input  wire [2:0]             s_axil_arprot,
    input  wire                   s_axil_arvalid,
    output wire                   s_axil_arready,
    output wire [DATA_WIDTH-1:0]  s_axil_rdata,
    output wire [1:0]             s_axil_rresp,
    output wire                   s_axil_rvalid,
    input  wire                   s_axil_rready,
    /*
     * AXI lite master interface
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [ADDR_WIDTH-1:0]  m_axil_awaddr,
    output wire [2:0]             m_axil_awprot,
    output wire                   m_axil_awvalid,
    input  wire                   m_axil_awready,
    output wire [DATA_WIDTH-1:0]  m_axil_wdata,
    output wire [STRB_WIDTH-1:0]  m_axil_wstrb,
    output wire                   m_axil_wvalid,
    input  wire                   m_axil_wready,
    input  wire [1:0]             m_axil_bresp,
    input  wire                   m_axil_bvalid,
    output wire                   m_axil_bready,
    output wire [ADDR_WIDTH-1:0]  m_axil_araddr,
    output wire [2:0]             m_axil_arprot,
    output wire                   m_axil_arvalid,
    input  wire                   m_axil_arready,
    input  wire [DATA_WIDTH-1:0]  m_axil_rdata,
    input  wire [1:0]             m_axil_rresp,
    input  wire                   m_axil_rvalid,
    output wire                   m_axil_rready
);

    // ------------------------------------------------------------------
    // Internal Signals
    // ------------------------------------------------------------------
    wire s_aw_valid, s_w_valid, s_ar_valid, s_r_valid;
    wire s_aw_ready, s_w_ready, s_ar_ready, s_r_ready;
    wire s_b_valid, s_r_valid_out;
    
    wire m_aw_valid, m_w_valid, m_ar_valid, m_r_valid;
    wire m_aw_ready, m_w_ready, m_ar_ready, m_r_ready;
    wire m_b_valid, m_r_valid_in;
    
    reg [ADDR_WIDTH-1:0] s_aw_addr, s_ar_addr;
    reg [2:0] s_aw_prot, s_ar_prot;
    reg [DATA_WIDTH-1:0] s_w_data, s_w_strb;
    reg [DATA_WIDTH-1:0] s_r_data;
    reg [1:0] s_b_resp, s_r_resp;
    
    reg s_aw_done, s_w_done, s_ar_done, s_r_done;
    reg s_b_done, s_r_done_out;
    
    reg m_aw_addr, m_ar_addr;
    reg [2:0] m_aw_prot, m_ar_prot;
    reg [DATA_WIDTH-1:0] m_w_data, m_w_strb;
    reg [DATA_WIDTH-1:0] m_r_data;
    reg [1:0] m_b_resp, m_r_resp;
    
    reg m_aw_done, m_w_done, m_ar_done, m_r_done;
    reg m_b_done, m_r_done_in;

    // ------------------------------------------------------------------
    // CDC Logic: Single-bit asynchronous FIFOs for Handshakes
    // ------------------------------------------------------------------
    
    // Write Channel (s -> m)
    // We need to capture s_w_valid and s_w_ready to ensure data is captured once.
    // Since we only need one outstanding write, we can use a simple handshake 
    // with a register to hold the data until the master is ready.
    
    // Read Channel (s -> m)
    // Similar logic for read.
    
    // Response Channel (m -> s)
    // Capture m_b_valid and m_b_ready.
    
    // Read Data Channel (m -> s)
    // Capture m_r_valid and m_r_ready.

    // ------------------------------------------------------------------
    // Write Channel Logic (Slave Write -> Master Write)
    // ------------------------------------------------------------------
    
    // Capture s_w_valid and s_w_ready to form a valid transaction window
    // We use a simple handshake mechanism where we wait for both valid and ready
    // before forwarding, but we must ensure we don't lose data if the other side
    // is slow. Since we allow 1 outstanding, we can buffer the data.
    
    reg s_w_valid_reg;
    reg s_w_ready_reg;
    reg s_w_valid_and_ready;
    
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_w_valid_reg <= 1'b0;
            s_w_ready_reg <= 1'b0;
            s_w_valid_and_ready <= 1'b0;
        end else begin
            s_w_valid_reg <= s_axil_wvalid;
            s_w_ready_reg <= s_axil_wready;
            s_w_valid_and_ready <= (s_w_valid_reg && s_w_ready_reg);
        end
    end

    // Capture s_w_data and s_w_strb when valid and ready are asserted
    reg [DATA_WIDTH-1:0] s_w_data_buf;
    reg [STRB_WIDTH-1:0] s_w_strb_buf;
    reg s_w_data_valid;
    
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_w_data_valid <= 1'b0;
            s_w_data_buf <= 1'b0;
            s_w_strb_buf <= 1'b0;
        end else if (s_w_valid_and_ready) begin
            s_w_data_valid <= 1'b1;
            s_w_data_buf <= s_axil_wdata;
            s_w_strb_buf <= s_axil_wstrb;
        end else begin
            s_w_data_valid <= 1'b0;
        end
    end

    // Forward to Master
    // Master side needs to be ready. If not, we keep the data valid on slave side
    // until master is ready.
    
    reg m_w_valid_reg;
    reg m_w_ready_reg;
    reg m_w_valid_and_ready;
    
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_w_valid_reg <= 1'b0;
            m_w_ready_reg <= 1'b0;
            m_w_valid_and_ready <= 1'b0;
        end else begin
            m_w_valid_reg <= m_axil_wvalid;
            m_w_ready_reg <= m_axil_wready;
            m_w_valid_and_ready <= (m_w_valid_reg && m_w_ready_reg);
        end
    end

    // Capture m_w_data and m_w_strb when valid and ready are asserted
    reg [DATA_WIDTH-1:0] m_w_data_buf;
    reg [STRB_WIDTH-1:0] m_w_strb_buf;
    reg m_w_data_valid;
    
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_w_data_valid <= 1'b0;
            m_w_data_buf <= 1'b0;
            m_w_strb_buf <= 1'b0;
        end else if (m_w_valid_and_ready) begin
            m_w_data_valid <= 1'b1;
            m_w_data_buf <= m_axil_wdata;
            m_w_strb_buf <= m_axil_wstrb;
        end else begin
            m_w_data_valid <= 1'b0;
        end
    end

    // Handshake logic for Write Data
    // Slave sends data when s_w_valid_and_ready is true AND m_w_data_valid is true
    // Master sends data when m_w_valid_and_ready is true AND s_w_data_valid is true
    
    assign s_axil_wready = (s_w_valid_reg && s_w_ready_reg) && m_w_data_valid;
    assign m_axil_wready = (m_w_valid_reg && m_w_ready_reg) && s_w_data_valid;
    
    assign m_axil_wdata = m_w_data_valid ? m_w_data_buf : 16'h0;
    assign m_axil_wstrb = m_w_data_valid ? m_w_strb_buf : 16'h0;
    assign m_axil_wvalid = m_w_valid_and_ready && s_w_data_valid;
    
    assign s_axil_wdata = s_w_data_valid ? s_w_data_buf : 16'h0;
    assign s_axil_wstrb = s_w_data_valid ? s_w_strb_buf : 16'h0;
    assign s_axil_wready = s_w_valid_and_ready && m_w_data_valid;

    // ------------------------------------------------------------------
    // Write Response Channel (Master Response -> Slave Response)
    // ------------------------------------------------------------------
    
    // Capture m_b_valid and m_b_ready
    reg m_b_valid_reg;
    reg m_b_ready_reg;
    reg m_b_valid_and_ready;
    
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_b_valid_reg <= 1'b0;
            m_b_ready_reg <= 1'b0;
            m_b_valid_and_ready <= 1'b0;
        end else begin
            m_b_valid_reg <= m_axil_bvalid;
            m_b_ready_reg <= m_axil_bready;
            m_b_valid_and_ready <= (m_b_valid_reg && m_b_ready_reg);
        end
    end

    // Capture m_b_resp when valid and ready are asserted
    reg [1:0] m_b_resp_buf;
    reg m_b_resp_valid;
    
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_b_resp_valid <= 1'b0;
            m_b_resp_buf <= 1'b0;
        end else if (m_b_valid_and_ready) begin
            m_b_resp_valid <= 1'b1;
            m_b_resp_buf <= m_axil_bresp;
        end else begin
            m_b_resp_valid <= 1'b0;
        end
    end

    // Forward to Slave
    reg s_b_valid_reg;
    reg s_b_ready_reg;
    reg s_b_valid_and_ready;
    
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_b_valid_reg <= 1'b0;
            s_b_ready_reg <= 1'b0;
            s_b_valid_and_ready <= 1'b0;
        end else begin
            s_b_valid_reg <= s_axil_bvalid;
            s_b_ready_reg <= s_axil_bready;
            s_b_valid_and_ready <= (s_b_valid_reg && s_b_ready_reg);
        end
    end

    // Capture s_b_resp when valid and ready are asserted
    reg [1:0] s_b_resp_buf;
    reg s_b_resp_valid;
    
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_b_resp_valid <= 1'b0;
            s_b_resp_buf <= 1'b0;
        end else if (s_b_valid_and_ready) begin
            s_b_resp_valid <= 1'b1;
            s_b_resp_buf <= s_axil_bresp;
        end else begin
            s_b_resp_valid <= 1'b0;
        end
    end

    // Handshake logic for Write Response
    assign m_axil_bready = m_b_valid_and_ready && s_b_resp_valid;
    assign s_axil_bready = s_b_valid_and_ready && m_b_resp_valid;
    
    assign m_axil_bresp = m_b_resp_valid ? m_b_resp_buf : 1'b0;
    assign s_axil_bresp = s_b_resp_valid ? s_b_resp_buf : 1'b0;
    
    assign m_axil_bvalid = m_b_valid_and_ready && s_b_resp_valid;
    assign s_axil_bvalid = s_b_valid_and_ready && m_b_resp_valid;

    // ------------------------------------------------------------------
    // Read Channel Logic (Slave Read -> Master Read)
    // ------------------------------------------------------------------
    
    // Capture s_ar_valid and s_ar_ready
    reg s_ar_valid_reg;
    reg s_ar_ready_reg;
    reg s_ar_valid_and_ready;
    
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_ar_valid_reg <= 1'b0;
            s_ar_ready_reg <= 1'b0;
            s_ar_valid_and_ready <= 1'b0;
        end else begin
            s_ar_valid_reg <= s_axil_arvalid;
            s_ar_ready_reg <= s_axil_arready;
            s_ar_valid_and_ready <= (s_ar_valid_reg && s_ar_ready_reg);
        end
    end

    // Capture s_ar_addr and s_ar_prot when valid and ready are asserted
    reg [ADDR_WIDTH-1:0] s_ar_addr_buf;
    reg [2:0] s_ar_prot_buf;
    reg s_ar_addr_valid;
    
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_ar_addr_valid <= 1'b0;
            s_ar_addr_buf <= 1'b0;
            s_ar_prot_buf <= 1'b0;
        end else if (s_ar_valid_and_ready) begin
            s_ar_addr_valid <= 1'b1;
            s_ar_addr_buf <= s_axil_araddr;
            s_ar_prot_buf <= s_axil_arprot;
        end else begin
            s_ar_addr_valid <= 1'b0;
        end
    end

    // Forward to Master
    reg m_ar_valid_reg;
    reg m_ar_ready_reg;
    reg m_ar_valid_and_ready;
    
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_ar_valid_reg <= 1'b0;
            m_ar_ready_reg <= 1'b0;
            m_ar_valid_and_ready <= 1'b0;
        end else begin
            m_ar_valid_reg <= m_axil_arvalid;
            m_ar_ready_reg <= m_axil_arready;
            m_ar_valid_and_ready <= (m_ar_valid_reg && m_ar_ready_reg);
        end
    end

    // Capture m_ar_addr and m_ar_prot when valid and ready are asserted
    reg [ADDR_WIDTH-1:0] m_ar_addr_buf;
    reg [2:0] m_ar_prot_buf;
    reg m_ar_addr_valid;
    
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_ar_addr_valid <= 1'b0;
            m_ar_addr_buf <= 1'b0;
            m_ar_prot_buf <= 1'b0;
        end else if (m_ar_valid_and_ready) begin
            m_ar_addr_valid <= 1'b1;
            m_ar_addr_buf <= m_axil_araddr;
            m_ar_prot_buf <= m_axil_arprot;
        end else begin
            m_ar_addr_valid <= 1'b0;
        end
    end

    // Handshake logic for Read Address
    assign s_axil_arready = s_ar_valid_and_ready && m_ar_addr_valid;
    assign m_axil_arready = m_ar_valid_and_ready && s_ar_addr_valid;
    
    assign m_axil_araddr = m_ar_addr_valid ? m_ar_addr_buf : 1'b0;
    assign m_axil_arprot = m_ar_addr_valid ? m_ar_prot_buf : 1'b0;
    assign m_axil_arvalid = m_ar_valid_and_ready && s_ar_addr_valid;
    
    assign s_axil_araddr = s_ar_addr_valid ? s_ar_addr_buf : 1'b0;
    assign s_axil_arprot = s_ar_addr_valid ? s_ar_prot_buf : 1'b0;
    assign s_axil_arready = s_ar_valid_and_ready && m_ar_addr_valid;

    // ------------------------------------------------------------------
    // Read Data Channel (Master Read Data -> Slave Read Data)
    // ------------------------------------------------------------------
    
    // Capture m_r_valid and m_r_ready
    reg m_r_valid_reg;
    reg m_r_ready_reg;
    reg m_r_valid_and_ready;
    
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_r_valid_reg <= 1'b0;
            m_r_ready_reg <= 1'b0;
            m_r_valid_and_ready <= 1'b0;
        end else begin
            m_r_valid_reg <= m_axil_rvalid;
            m_r_ready_reg <= m_axil_rready;
            m_r_valid_and_ready <= (m_r_valid_reg && m_r_ready_reg);
        end
    end

    // Capture m_r_data and m_r_resp when valid and ready are asserted
    reg [DATA_WIDTH-1:0] m_r_data_buf;
    reg [1:0] m_r_resp_buf;
    reg m_r_data_valid;
    
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_r_data_valid <= 1'b0;
            m_r_data_buf <= 1'b0;
            m_r_resp_buf <= 1'b0;
        end else if (m_r_valid_and_ready) begin
            m_r_data_valid <= 1'b1;
            m_r_data_buf <= m_axil_rdata;
            m_r_resp_buf <= m_axil_rresp;
        end else begin
            m_r_data_valid <= 1'b0;
        end
    end

    // Forward to Slave
    reg s_r_valid_reg;
    reg s_r_ready_reg;
    reg s_r_valid_and_ready;
    
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_r_valid_reg <= 1'b0;
            s_r_ready_reg <= 1'b0;
            s_r_valid_and_ready <= 1'b0;
        end else begin
            s_r_valid_reg <= s_axil_rvalid;
            s_r_ready_reg <= s_axil_rready;
            s_r_valid_and_ready <= (s_r_valid_reg && s_r_ready_reg);
        end
    end

    // Capture s_r_data and s_r_resp when valid and ready are asserted
    reg [DATA_WIDTH-1:0] s_r_data_buf;
    reg [1:0] s_r_resp_buf;
    reg s_r_data_valid;
    
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_r_data_valid <= 1'b0;
            s_r_data_buf <= 1'b0;
            s_r_resp_buf <= 1'b0;
        end else if (s_r_valid_and_ready) begin
            s_r_data_valid <= 1'b1;
            s_r_data_buf <= s_axil_rdata;
            s_r_resp_buf <= s_axil_rresp;
        end else begin
            s_r_data_valid <= 1'b0;
        end
    end

    // Handshake logic for Read Data
    assign m_axil_rready = m_r_valid_and_ready && s_r_data_valid;
    assign s_axil_rready = s_r_valid_and_ready && m_r_data_valid;
    
    assign m_axil_rdata = m_r_data_valid ? m_r_data_buf : 16'h0;
    assign s_axil_rdata = s_r_data_valid ? s_r_data_buf : 16'h0;
    
    assign m_axil_rresp = m_r_data_valid ? m_r_resp_buf : 1'b0;
    assign s_axil_rresp = s_r_data_valid ? s_r_resp_buf : 1'b0;
    
    assign m_axil_rvalid = m_r_valid_and_ready && s_r_data_valid;
    assign s_axil_rvalid = s_r_valid_and_ready && m_r_data_valid;

endmodule
