module axil_cdc #
(
    parameter DATA_WIDTH = 32,
    parameter ADDR_WIDTH = 32,
    parameter STRB_WIDTH = (DATA_WIDTH/8)
)
(

     * AXI lite slave interface

    input  wire                   s_clk,
    input  wire                   s_rst,
    input  wire [ADDR_WIDTH-1:0]  s_axil_awaddr,
    input  wire [2:0]             s_axil_awprot,
    input  wire                   s_axil_awvalid,
    output wire                   s_axil_awready,
    input  wire [DATA_WIDTH-1:0]  s_axil_wdata,
    input  wire [STRB_WIDTH-1:0]  s_axil_wstrb,
    input  wire                   s_axil_wvalid,
    output wire                   s_axil_wready,
    output wire [1:0]             s_axil_bresp,
    output wire                   s_axil_bvalid,
    input  wire                   s_axil_bready,
    input  wire [ADDR_WIDTH-1:0]  s_axil_araddr,
    input  wire [2:0]             s_axil_arprot,
    input  wire                   s_axil_arvalid,
    output wire                   s_axil_arready,
    output wire [DATA_WIDTH-1:0]  s_axil_rdata,
    output wire [1:0]             s_axil_rresp,
    output wire                   s_axil_rvalid,
    input  wire                   s_axil_rready,
    /*
     * AXI lite master interface
     */
    input  wire                   m_clk,
    input  wire                   m_rst,
    output wire [ADDR_WIDTH-1:0]  m_axil_awaddr,
    output wire [2:0]             m_axil_awprot,
    output wire                   m_axil_awvalid,
    input  wire                   m_axil_awready,
    output wire [DATA_WIDTH-1:0]  m_axil_wdata,
    output wire [STRB_WIDTH-1:0]  m_axil_wstrb,
    output wire                   m_axil_wvalid,
    input  wire                   m_axil_wready,
    input  wire [1:0]             m_axil_bresp,
    input  wire                   m_axil_bvalid,
    output wire                   m_axil_bready,
    output wire [ADDR_WIDTH-1:0]  m_axil_araddr,
    output wire [2:0]             m_axil_arprot,
    output wire                   m_axil_arvalid,
    input  wire                   m_axil_arready,
    input  wire [DATA_WIDTH-1:0]  m_axil_rdata,
    input  wire [1:0]             m_axil_rresp,
    input  wire                   m_axil_rvalid,
    output wire                   m_axil_rready
);

    // ------------------------------------------------------------------
    // Internal Signals
    // ------------------------------------------------------------------
    wire s_aw_valid, s_w_valid, s_b_valid, s_ar_valid, s_r_valid;
    wire s_aw_ready, s_w_ready, s_b_ready, s_ar_ready, s_r_ready;
    wire m_aw_valid, m_w_valid, m_b_valid, m_ar_valid, m_r_valid;
    wire m_aw_ready, m_w_ready, m_b_ready, m_ar_ready, m_r_ready;

    // State Registers for Write Channel (Slave -> Master)
    reg s_aw_state, s_w_state, s_b_state;
    reg [ADDR_WIDTH-1:0] s_aw_addr;
    reg [2:0] s_aw_prot;
    reg [DATA_WIDTH-1:0] s_w_data;
    reg [STRB_WIDTH-1:0] s_w_strb;
    reg s_w_valid_reg;
    reg s_b_valid_reg;
    reg [1:0] s_b_resp;

    // State Registers for Read Channel (Slave -> Master)
    reg s_ar_state, s_r_state;
    reg [ADDR_WIDTH-1:0] s_ar_addr;
    reg [2:0] s_ar_prot;
    reg s_r_valid_reg;
    reg [DATA_WIDTH-1:0] s_r_data;
    reg [1:0] s_r_resp;

    // State Registers for Master -> Slave (Write)
    reg m_aw_state, m_w_state, m_b_state;
    reg [ADDR_WIDTH-1:0] m_aw_addr;
    reg [2:0] m_aw_prot;
    reg [DATA_WIDTH-1:0] m_w_data;
    reg [STRB_WIDTH-1:0] m_w_strb;
    reg m_w_valid_reg;
    reg m_b_valid_reg;
    reg [1:0] m_b_resp;

    // State Registers for Master -> Slave (Read)
    reg m_ar_state, m_r_state;
    reg [ADDR_WIDTH-1:0] m_ar_addr;
    reg [2:0] m_ar_prot;
    reg m_r_valid_reg;
    reg [DATA_WIDTH-1:0] m_r_data;
    reg [1:0] m_r_resp;

    // CDC Synchronizers (2-stage)
    wire s_aw_valid_sync0, s_aw_valid_sync1;
    wire s_w_valid_sync0, s_w_valid_sync1;
    wire s_b_valid_sync0, s_b_valid_sync1;
    wire s_ar_valid_sync0, s_ar_valid_sync1;
    wire s_r_valid_sync0, s_r_valid_sync1;

    wire m_aw_valid_sync0, m_aw_valid_sync1;
    wire m_w_valid_sync0, m_w_valid_sync1;
    wire m_b_valid_sync0, m_b_valid_sync1;
    wire m_ar_valid_sync0, m_ar_valid_sync1;
    wire m_r_valid_sync0, m_r_valid_sync1;

    // ------------------------------------------------------------------
    // Synchronizer Logic
    // ------------------------------------------------------------------
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_aw_valid_sync0 <= 1'b0;
            s_w_valid_sync0  <= 1'b0;
            s_b_valid_sync0  <= 1'b0;
            s_ar_valid_sync0 <= 1'b0;
            s_r_valid_sync0  <= 1'b0;
        end else begin
            s_aw_valid_sync0 <= s_axil_awvalid;
            s_w_valid_sync0  <= s_axil_wvalid;
            s_b_valid_sync0  <= s_axil_bvalid;
            s_ar_valid_sync0 <= s_axil_arvalid;
            s_r_valid_sync0  <= s_axil_rvalid;
        end
    end

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_aw_valid_sync1 <= 1'b0;
            s_w_valid_sync1  <= 1'b0;
            s_b_valid_sync1  <= 1'b0;
            s_ar_valid_sync1 <= 1'b0;
            s_r_valid_sync1  <= 1'b0;
        end else begin
            s_aw_valid_sync1 <= s_aw_valid_sync0;
            s_w_valid_sync1  <= s_w_valid_sync0;
            s_b_valid_sync1  <= s_b_valid_sync0;
            s_ar_valid_sync1 <= s_ar_valid_sync0;
            s_r_valid_sync1  <= s_r_valid_sync0;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_aw_valid_sync0 <= 1'b0;
            m_w_valid_sync0  <= 1'b0;
            m_b_valid_sync0  <= 1'b0;
            m_ar_valid_sync0 <= 1'b0;
            m_r_valid_sync0  <= 1'b0;
        end else begin
            m_aw_valid_sync0 <= m_axil_awvalid;
            m_w_valid_sync0  <= m_axil_wvalid;
            m_b_valid_sync0  <= m_axil_bvalid;
            m_ar_valid_sync0 <= m_axil_arvalid;
            m_r_valid_sync0  <= m_axil_rvalid;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_aw_valid_sync1 <= 1'b0;
            m_w_valid_sync1  <= 1'b0;
            m_b_valid_sync1  <= 1'b0;
            m_ar_valid_sync1 <= 1'b0;
            m_r_valid_sync1  <= 1'b0;
        end else begin
            m_aw_valid_sync1 <= m_aw_valid_sync0;
            m_w_valid_sync1  <= m_w_valid_sync0;
            m_b_valid_sync1  <= m_b_valid_sync0;
            m_ar_valid_sync1 <= m_ar_valid_sync0;
            m_r_valid_sync1  <= m_r_valid_sync0;
        end
    end

    // ------------------------------------------------------------------
    // Slave Side State Machine (s_clk domain)
    // ------------------------------------------------------------------
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_aw_state <= 1'b0;
            s_w_state  <= 1'b0;
            s_b_state  <= 1'b0;
            s_ar_state <= 1'b0;
            s_r_state  <= 1'b0;
            s_aw_addr  <= 1'b0;
            s_aw_prot  <= 1'b0;
            s_w_data   <= 1'b0;
            s_w_strb   <= 1'b0;
            s_w_valid_reg <= 1'b0;
            s_b_valid_reg <= 1'b0;
            s_b_resp    <= 2'b00;
            s_ar_addr   <= 1'b0;
            s_ar_prot   <= 1'b0;
            s_r_valid_reg <= 1'b0;
            s_r_data    <= 1'b0;
            s_r_resp    <= 2'b00;
        end else begin
            // Write Address
            if (s_aw_valid_sync1 && s_aw_ready) begin
                s_aw_state <= 1'b1;
                s_aw_addr  <= s_axil_awaddr;
                s_aw_prot  <= s_axil_awprot;
            end else if (s_aw_state) begin
                s_aw_state <= 1'b0;
            end

            // Write Data
            if (s_w_valid_sync1 && s_w_ready) begin
                s_w_state <= 1'b1;
                s_w_data  <= s_axil_wdata;
                s_w_strb  <= s_axil_wstrb;
                s_w_valid_reg <= 1'b1;
            end else if (s_w_state) begin
                s_w_state <= 1'b0;
                s_w_valid_reg <= 1'b0;
            end

            // Write Response
            if (s_b_valid_sync1 && s_b_ready) begin
                s_b_state <= 1'b1;
                s_b_resp  <= s_axil_bresp;
                s_b_valid_reg <= 1'b1;
            end else if (s_b_state) begin
                s_b_state <= 1'b0;
                s_b_valid_reg <= 1'b0;
            end

            // Read Address
            if (s_ar_valid_sync1 && s_ar_ready) begin
                s_ar_state <= 1'b1;
                s_ar_addr  <= s_axil_araddr;
                s_ar_prot  <= s_axil_arprot;
            end else if (s_ar_state) begin
                s_ar_state <= 1'b0;
            end

            // Read Data
            if (s_r_valid_sync1 && s_r_ready) begin
                s_r_state <= 1'b1;
                s_r_data  <= s_axil_rdata;
                s_r_resp  <= s_axil_rresp;
                s_r_valid_reg <= 1'b1;
            end else if (s_r_state) begin
                s_r_state <= 1'b0;
                s_r_valid_reg <= 1'b0;
            end
        end
    end

    // ------------------------------------------------------------------
    // Master Side State Machine (m_clk domain)
    // ------------------------------------------------------------------
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_aw_state <= 1'b0;
            m_w_state  <= 1'b0;
            m_b_state  <= 1'b0;
            m_ar_state <= 1'b0;
            m_r_state  <= 1'b0;
            m_aw_addr  <= 1'b0;
            m_aw_prot  <= 1'b0;
            m_w_data   <= 1'b0;
            m_w_strb   <= 1'b0;
            m_w_valid_reg <= 1'b0;
            m_b_valid_reg <= 1'b0;
            m_b_resp    <= 2'b00;
            m_ar_addr   <= 1'b0;
            m_ar_prot   <= 1'b0;
            m_r_valid_reg <= 1'b0;
            m_r_data    <= 1'b0;
            m_r_resp    <= 2'b00;
        end else begin
            // Write Address
            if (m_aw_valid_sync1 && m_aw_ready) begin
                m_aw_state <= 1'b1;
                m_aw_addr  <= m_axil_awaddr;
                m_aw_prot  <= m_axil_awprot;
            end else if (m_aw_state) begin
                m_aw_state <= 1'b0;
            end

            // Write Data
            if (m_w_valid_sync1 && m_w_ready) begin
                m_w_state <= 1'b1;
                m_w_data  <= m_axil_wdata;
                m_w_strb  <= m_axil_wstrb;
                m_w_valid_reg <= 1'b1;
            end else if (m_w_state) begin
                m_w_state <= 1'b0;
                m_w_valid_reg <= 1'b0;
            end

            // Write Response
            if (m_b_valid_sync1 && m_b_ready) begin
                m_b_state <= 1'b1;
                m_b_resp  <= m_axil_bresp;
                m_b_valid_reg <= 1'b1;
            end else if (m_b_state) begin
                m_b_state <= 1'b0;
                m_b_valid_reg <= 1'b0;
            end

            // Read Address
            if (m_ar_valid_sync1 && m_ar_ready) begin
                m_ar_state <= 1'b1;
                m_ar_addr  <= m_axil_araddr;
                m_ar_prot  <= m_axil_arprot;
            end else if (m_ar_state) begin
                m_ar_state <= 1'b0;
            end

            // Read Data
            if (m_r_valid_sync1 && m_r_ready) begin
                m_r_state <= 1'b1;
                m_r_data  <= m_axil_rdata;
                m_r_resp  <= m_axil_rresp;
                m_r_valid_reg <= 1'b1;
            end else if (m_r_state) begin
                m_r_state <= 1'b0;
                m_r_valid_reg <= 1'b0;
            end
        end
    end

    // ------------------------------------------------------------------
    // Output Logic (Slave Side)
    // ------------------------------------------------------------------
    assign s_axil_awready = (s_aw_state == 1'b0);
    assign s_axil_wready  = (s_w_state == 1'b0);
    assign s_axil_bresp   = s_b_resp;
    assign s_axil_bvalid  = s_b_valid_reg;
    assign s_axil_arready = (s_ar_state == 1'b0);
    assign s_axil_rdata   = s_r_data;
    assign s_axil_rresp   = s_r_resp;
    assign s_axil_rvalid  = s_r_valid_reg;

    // ------------------------------------------------------------------
    // Output Logic (Master Side)
    // ------------------------------------------------------------------
    assign m_axil_awaddr  = m_aw_addr;
    assign m_axil_awprot  = m_aw_prot;
    assign m_axil_awvalid = m_aw_state;
    assign m_axil_wdata   = m_w_data;
    assign m_axil_wstrb   = m_w_strb;
    assign m_axil_wvalid  = m_w_state;
    assign m_axil_bresp   = m_b_resp;
    assign m_axil_bvalid  = m_b_valid_reg;
    assign m_axil_araddr  = m_ar_addr;
    assign m_axil_arprot  = m_ar_prot;
    assign m_axil_arvalid = m_ar_state;
    assign m_axil_rdata   = m_r_data;
    assign m_axil_rresp   = m_r_resp;
    assign m_axil_rvalid  = m_r_valid_reg;

endmodule
