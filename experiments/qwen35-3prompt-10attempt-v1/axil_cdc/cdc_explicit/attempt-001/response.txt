            s_valid_w_sync2 <= 1'b0;
            s_valid_ar_sync1 <= 1'b0;
            s_valid_ar_sync2 <= 1'b0;
            s_valid_b_sync1 <= 1'b0;
            s_valid_b_sync2 <= 1'b0;
