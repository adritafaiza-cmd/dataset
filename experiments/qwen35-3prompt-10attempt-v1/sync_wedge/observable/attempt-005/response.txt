module sync_wedge #(
    parameter STAGES = 2
)(
    input  clk_i,
    input  rst_ni,
    input  en_i,
    input  serial_i,
    output r_edge_o,
    output f_edge_o,
    output serial_o
);

    // Synchronous register array for the FIFO-like structure
    // We use a shift register approach to synchronize the bit
    reg [STAGES-1:0] sync_reg;
    
    // Edge detection registers in the destination clock domain
    reg r_edge_det;
    reg f_edge_det;
    
    // Output registers for the edge flags
    reg r_edge_o_reg;
    reg f_edge_o_reg;
    
    // Internal signal for the synchronized bit before edge detection logic
    wire sync_bit;

    // Synchronize the input bit through STAGES flip-flops
    // First stage captures the input on the first clock edge
    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            sync_reg <= {STAGES{1'b0}};
        end else begin
            sync_reg <= {sync_reg[STAGES-2:0], serial_i};
        end
    end

    // The last stage of the shift register holds the synchronized bit
    assign sync_bit = sync_reg[STAGES-1];

    // Edge detection logic in the destination clock domain
    // Rising edge: current bit is 1, previous bit was 0
    // Falling edge: current bit is 0, previous bit was 1
    // We use the second-to-last stage to represent the "previous" bit relative to the current sync_bit
    assign r_edge_det = (sync_bit == 1'b1) && (sync_reg[STAGES-2] == 1'b0);
    assign f_edge_det = (sync_bit == 1'b0) && (sync_reg[STAGES-2] == 1'b1);

    // Output registers for edge detection
    // These are gated by en_i to ensure outputs only change when enabled
    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            r_edge_o_reg <= 1'b0;
            f_edge_o_reg <= 1'b0;
        end else if (en_i) begin
            r_edge_o_reg <= r_edge_det;
            f_edge_o_reg <= f_edge_det;
        end
    end

    // Assign outputs
    assign r_edge_o = r_edge_o_reg;
    assign f_edge_o = f_edge_o_reg;
    // The synchronized bit is the output
    assign serial_o = sync_bit;

endmodule
