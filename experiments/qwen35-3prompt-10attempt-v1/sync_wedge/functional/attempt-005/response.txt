module sync_wedge #(
    parameter STAGES = 2
)(
    input  wire clk_i,
    input  wire rst_ni,
    input  wire en_i,
    input  wire serial_i,
    output wire r_edge_o,
    output wire f_edge_o,
    output wire serial_o
);

    // Synchronized data output
    reg serial_sync;

    // Edge detection flags
    reg r_edge_flag;
    reg f_edge_flag;

    // Synchronizer chain for serial_i
    genvar i;
    reg [STAGES-1:0] sync_chain;

    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            sync_chain <= {STAGES{1'b0}};
        else begin
            sync_chain <= {sync_chain[STAGES-2:0], serial_i};
        end
    end

    // Assign synchronized output
    assign serial_o = sync_chain[STAGES-1];

    // Edge detection logic on synchronized data
    // Rising edge: current is 1, previous was 0
    // Falling edge: current is 0, previous was 1
    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            r_edge_flag <= 1'b0;
            f_edge_flag <= 1'b0;
        end else begin
            // Detect rising edge
            if (sync_chain[STAGES-1] && !sync_chain[STAGES-2])
                r_edge_flag <= 1'b1;
            else
                r_edge_flag <= 1'b0;

            // Detect falling edge
            if (!sync_chain[STAGES-1] && sync_chain[STAGES-2])
                f_edge_flag <= 1'b1;
            else
                f_edge_flag <= 1'b0;
        end
    end

    // Output edge pulses gated by enable
    assign r_edge_o = en_i ? r_edge_flag : 1'b0;
    assign f_edge_o = en_i ? f_edge_flag : 1'b0;

endmodule<|im_end|>
