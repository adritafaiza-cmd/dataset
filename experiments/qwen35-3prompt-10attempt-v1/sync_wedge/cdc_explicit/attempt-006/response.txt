module sync_wedge #(
    parameter STAGES = 2
)(
    input  wire clk_i,
    input  wire rst_ni,
    input  wire en_i,
    input  wire serial_i,
    output wire r_edge_o,
    output wire f_edge_o,
    output wire serial_o
);

    // Synchronize the asynchronous input bit
    wire [STAGES-1:0] sync_serial;
    genvar i;
    generate
        for (i = 0; i < STAGES; i = i + 1) begin : sync_stage
            if (i == 0) begin : first_sync
                always @(posedge clk_i or negedge rst_ni) begin
                    if (!rst_ni)
                        sync_serial[i] <= 1'b0;
                    else
                        sync_serial[i] <= serial_i;
                end
            end else begin : subsequent_sync
                always @(posedge clk_i or negedge rst_ni) begin
                    if (!rst_ni)
                        sync_serial[i] <= 1'b0;
                    else
                        sync_serial[i] <= sync_serial[i-1];
                end
            end
        end
    endgenerate

    // Detect edges on the synchronized signal
    wire sync_rising;
    wire sync_falling;

    assign sync_rising = sync_serial[STAGES-1] & ~sync_serial[STAGES-2];
    assign sync_falling = ~sync_serial[STAGES-1] & sync_serial[STAGES-2];

    // Generate single-cycle edge pulses
    reg r_edge_pulse;
    reg f_edge_pulse;

    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            r_edge_pulse <= 1'b0;
            f_edge_pulse <= 1'b0;
        end else begin
            r_edge_pulse <= sync_rising;
            f_edge_pulse <= sync_falling;
        end
    end

    // Assign outputs
    assign r_edge_o = r_edge_pulse;
    assign f_edge_o = f_edge_pulse;
    assign serial_o = sync_serial[STAGES-1];

endmodule<|im_end|>
