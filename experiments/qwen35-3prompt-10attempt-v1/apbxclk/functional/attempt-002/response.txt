```verilog
module apbxclk #(
	parameter	C_APB_ADDR_WIDTH = 12,
	parameter	C_APB_DATA_WIDTH = 32,
	parameter [0:0]	OPT_REGISTERED = 1'b0,
	localparam	AW = C_APB_ADDR_WIDTH,
	localparam	DW = C_APB_DATA_WIDTH
) (
	input	wire			S_APB_PCLK, S_PRESETn,
	input	wire			S_APB_PSEL,
	input	wire			S_APB_PENABLE,
	output	reg			S_APB_PREADY,
	input	wire	[AW-1:0]	S_APB_PADDR,
	input	wire			S_APB_PWRITE,
	input	wire	[DW-1:0]	S_APB_PWDATA,
	input	wire	[DW/8-1:0]	S_APB_PWSTRB,
	input	wire	[2:0]		S_APB_PPROT,
	output	wire	[DW-1:0]	S_APB_PRDATA,
	output	wire			S_APB_PSLVERR,
	input	wire			M_APB_PCLK,
	output	reg			M_PRESETn,
	output	reg			M_APB_PSEL,
	output	reg			M_APB_PENABLE,
	input	wire			M_APB_PREADY,
	output	wire	[AW-1:0]	M_APB_PADDR,
	output	wire			M_APB_PWRITE,
	output	wire	[DW-1:0]	M_APB_PWDATA,
	output	wire	[DW/8-1:0]	M_APB_PWSTRB,
	output	wire	[2:0]		M_APB_PPROT,
	input	wire	[DW-1:0]	M_APB_PRDATA,
	input	wire			M_APB_PSLVERR
);

	// Internal signals for state machine and data buffering
	reg [AW-1:0]		i_addr;
	reg			i_write;
	reg [DW-1:0]		i_wdata;
	reg [DW/8-1:0]		i_wstrb;
	reg [2:0]			i_prot;
	reg			i_sel;
	reg			i_enable;
	reg			i_ready;
	reg			i_err;
	reg			i_data;

	// State machine states
	localparam IDLE = 2'b00;
	localparam REQ = 2'b01;
	localparam ACC = 2'b10;

	reg [1:0]		state;

	// Cross-domain registers for payload and response
	// When OPT_REGISTERED is 0, these are combinational outputs driven by the state machine
	// When OPT_REGISTERED is 1, these are registered in the destination clock domain
	reg [AW-1:0]		o_addr_reg;
	reg			o_write_reg;
	reg [DW-1:0]		o_wdata_reg;
	reg [DW/8-1:0]		o_wstrb_reg;
	reg [2:0]			o_prot_reg;
	reg			o_sel_reg;
	reg			o_enable_reg;
	reg			o_ready_reg;
	reg			o_err_reg;
	reg			o_data_reg;

	// Synchronizer for M_APB_PREADY to detect completion in S domain
	reg [1:0]		sync_ready;

	// Synchronizer for M_APB_PSLVERR to detect error in S domain
	reg [1:0]		sync_err;

	// Synchronizer for M_APB_PRDATA to capture data in S domain
	reg [DW-1:0]		sync_data;

	// Synchronizer for M_APB_PCLK to generate a pulse for capturing data (if needed)
	// Actually, we just need to capture data on the rising edge of M_APB_PREADY in S domain
	// But since M_APB_PRDATA is valid when M_APB_PREADY is high, we need to sync the ready signal
	// and capture the data at that moment.

	// Synchronizer for M_APB_PREADY
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			sync_ready <= 2'b00;
		else
			sync_ready <= {sync_ready[0], M_APB_PREADY};
	end

	// Synchronizer for M_APB_PSLVERR
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			sync_err <= 2'b00;
		else
			sync_err <= {sync_err[0], M_APB_PSLVERR};
	end

	// Synchronizer for M_APB_PRDATA
	// We capture the data when the second bit of sync_ready becomes 1
	// This ensures we capture the data stable in S domain
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			sync_data <= {DW{1'b0}};
		else if (sync_ready[1])
			sync_data <= M_APB_PRDATA;
	end

	// State machine logic in S domain
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			state <= IDLE;
		else
			case (state)
				IDLE:
					if (S_APB_PSEL && S_APB_PENABLE)
						state <= REQ;
					else
						state <= IDLE;

				REQ:
					// Wait for M domain to acknowledge
					if (sync_ready[1])
						state <= ACC;
					else
						state <= REQ;

				ACC:
					// Wait for M domain to complete
					if (sync_ready[1])
						state <= IDLE;
					else
						state <= ACC;
			endcase
	end

	// Output logic for S domain
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			S_APB_PREADY <= 1'b0;
			S_APB_PRDATA <= {DW{1'b0}};
			S_APB_PSLVERR <= 1'b0;
		end
		else begin
			case (state)
				IDLE: begin
					S_APB_PREADY <= 1'b0;
					S_APB_PRDATA <= {DW{1'b0}};
					S_APB_PSLVERR <= 1'b0;
				end
				REQ: begin
					S_APB_PREADY <= 1'b0;
					S_APB_PRDATA <= {DW{1'b0}};
					S_APB_PSLVERR <= 1'b0;
				end
				ACC: begin
					S_APB_PREADY <= 1'b0;
					S_APB_PRDATA <= sync_data;
					S_APB_PSLVERR <= sync_err[1];
				end
			endcase
		end
	end

	// Output logic for M domain
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			M_APB_PSEL <= 1'b0;
			M_APB_PENABLE <= 1'b0;
			M_APB_PADDR <= {AW{1'b0}};
			M_APB_PWRITE <= 1'b0;
			M_APB_PWDATA <= {DW{1'b0}};
			M_APB_PWSTRB <= {DW/8{1'b0}};
			M_APB_PPROT <= 3'b000;
		end
		else begin
			// Drive M domain signals based on current state and buffered data
			M_APB_PSEL <= o_sel_reg;
			M_APB_PENABLE <= o_enable_reg;
			M_APB_PADDR <= o_addr_reg;
			M_APB_PWRITE <= o_write_reg;
			M_APB_PWDATA <= o_wdata_reg;
			M_APB_PWSTRB <= o_wstrb_reg;
			M_APB_PPROT <= o_prot_reg;
		end
	end

	// Register M domain outputs when OPT_REGISTERED is set
	// This ensures that the data is stable across the clock domain boundary
	// and captured correctly in the S domain synchronizers
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			o_addr_reg <= {AW{1'b0}};
			o_write_reg <= 1'b0;
			o_wdata_reg <= {DW{1'b0}};
			o_wstrb_reg <= {DW/8{1'b0}};
			o_prot_reg <= 3'b000;
			o_sel_reg <= 1'b0;
			o_enable_reg <= 1'b0;
			o_ready_reg <= 1'b0;
			o_err_reg <= 1'b0;
			o_data_reg <= {DW{1'b0}};
		end
		else begin
			o_addr_reg <= M_APB_PADDR;
			o_write_reg <= M_APB_PWRITE;
			o_wdata_reg <= M_APB_PWDATA;
			o_wstrb_reg <= M_APB_PWSTRB;
			o_prot_reg <= M_APB_PPROT;
			o_sel_reg <= M_APB_PSEL;
			o_enable_reg <= M_APB_PENABLE;
			o_ready_reg <= M_APB_PREADY;
			o_err_reg <= M_APB_PSLVERR;
			o_data_reg <= M_APB_PRDATA;
		end
	end

	// When OPT_REGISTERED is 0, drive M domain outputs directly from S domain inputs
	// This avoids unnecessary registers and keeps the behavior simple
	always @(*) begin
		if (OPT_REGISTERED == 1'b1) begin
			M_APB_PADDR <= o_addr_reg;
			M_APB_PWRITE <= o_write_reg;
			M_APB_PWDATA <= o_wdata_reg;
			M_APB_PWSTRB <= o_wstrb_reg;
			M_APB_PPROT <= o_prot_reg;
			M_APB_PSEL <= o_sel_reg;
			M_APB_PENABLE <= o_enable_reg;
		end
		else begin
			M_APB_PADDR <= S_APB_PADDR;
			M_APB_PWRITE <= S_APB_PWRITE;
			M_APB_PWDATA <= S_APB_PWDATA;
			M_APB_PWSTRB <= S_APB_PWSTRB;
			M_APB_PPROT <= S_APB_PPROT;
			M_APB_PSEL <= S_APB_PSEL;
			M_APB_PENABLE <= S_APB_PENABLE;
		end
	end

	// Generate M domain reset
	always @(posedge M_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			M_PRESETn <= 1'b0;
		else
			M_PRESETn <= 1'b1;
	end

	// Generate S domain ready signal
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			S_APB_PREADY <= 1'b0;
		else
			S_APB_PREADY <= (state == ACC) ? 1'b1 : 1'b0;
	end

	// Generate S domain error signal
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			S_APB_PSLVERR <= 1'b0;
		else
			S_APB_PSLVERR <= sync_err[1];
	end

	// Generate S domain data signal
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn)
			S_APB_PRDATA <= {DW{1'b0}};
		else
			S_APB_PRDATA <= sync_data;
	end

endmodule
