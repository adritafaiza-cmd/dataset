```verilog
`timescale 1ns/1ps

module apbxclk #(
	parameter	C_APB_ADDR_WIDTH = 12,
	parameter	C_APB_DATA_WIDTH = 32,
	parameter [0:0]	OPT_REGISTERED = 1'b0,
	localparam	AW = C_APB_ADDR_WIDTH,
	localparam	DW = C_APB_DATA_WIDTH
) (
	input	wire			S_APB_PCLK, S_PRESETn,
	input	wire			S_APB_PSEL,
	input	wire			S_APB_PENABLE,
	output	reg			S_APB_PREADY,
	input	wire	[AW-1:0]	S_APB_PADDR,
	input	wire			S_APB_PWRITE,
	input	wire	[DW-1:0]	S_APB_PWDATA,
	input	wire	[DW/8-1:0]	S_APB_PWSTRB,
	input	wire	[2:0]		S_APB_PPROT,
	output	wire	[DW-1:0]	S_APB_PRDATA,
	output	wire			S_APB_PSLVERR,
	input	wire			M_APB_PCLK,
	output	reg			M_PRESETn,
	output	reg			M_APB_PSEL,
	output	reg			M_APB_PENABLE,
	input	wire			M_APB_PREADY,
	output	wire	[AW-1:0]	M_APB_PADDR,
	output	wire			M_APB_PWRITE,
	output	wire	[DW-1:0]	M_APB_PWDATA,
	output	wire	[DW/8-1:0]	M_APB_PWSTRB,
	output	wire	[2:0]		M_APB_PPROT,
	input	wire	[DW-1:0]	M_APB_PRDATA,
	input	wire			M_APB_PSLVERR
);

	localparam	AW = C_APB_ADDR_WIDTH;
	localparam	DW = C_APB_DATA_WIDTH;
	localparam	WSTRB_WIDTH = DW/8;

	// State Machine for Master Side (Slave Interface)
	// States: IDLE, REQ_SENT, WAITING_RESP
	localparam	IDLE = 2'b00;
	localparam	REQ_SENT = 2'b01;
	localparam	WAITING_RESP = 2'b10;

	reg [1:0]	m_state;
	reg [AW-1:0]	m_addr;
	reg	m_write;
	reg [DW-1:0]	m_wdata;
	reg [WSTRB_WIDTH-1:0]	m_wstrb;
	reg [2:0]	m_prot;
	reg [DW-1:0]	m_rdata;
	reg	m_err;
	reg	m_req_valid;
	reg	m_req_ready;
	reg	m_resp_valid;
	reg	m_resp_ready;

	// State Machine for Slave Side (Master Interface)
	// States: IDLE, REQ_SENT, WAITING_RESP
	localparam	IDLE_S = 2'b00;
	localparam	REQ_SENT_S = 2'b01;
	localparam	WAITING_RESP_S = 2'b10;

	reg [1:0]	s_state;
	reg [AW-1:0]	s_addr;
	reg	s_write;
	reg [DW-1:0]	s_wdata;
	reg [WSTRB_WIDTH-1:0]	s_wstrb;
	reg [2:0]	s_prot;
	reg [DW-1:0]	s_rdata;
	reg	s_err;
	reg	s_req_valid;
	reg	s_req_ready;
	reg	s_resp_valid;
	reg	s_resp_ready;

	// Cross-domain registers for payload capture
	// Master side captures Slave requests
	reg [AW-1:0]	m_cross_addr;
	reg	m_cross_write;
	reg [DW-1:0]	m_cross_wdata;
	reg [WSTRB_WIDTH-1:0]	m_cross_wstrb;
	reg [2:0]	m_cross_prot;
	reg	m_cross_valid;
	reg	m_cross_ready;

	// Slave side captures Master requests
	reg [AW-1:0]	s_cross_addr;
	reg	s_cross_write;
	reg [DW-1:0]	s_cross_wdata;
	reg [WSTRB_WIDTH-1:0]	s_cross_wstrb;
	reg [2:0]	s_cross_prot;
	reg	s_cross_valid;
	reg	s_cross_ready;

	// Cross-domain registers for response capture
	// Master side captures Slave responses
	reg [DW-1:0]	m_cross_rdata;
	reg	m_cross_err;
	reg	m_cross_resp_valid;
	reg	m_cross_resp_ready;

	// Slave side captures Master responses
	reg [DW-1:0]	s_cross_rdata;
	reg	s_cross_err;
	reg	s_cross_resp_valid;
	reg	s_cross_resp_ready;

	// Clock Domain Crossing Logic for Request (Slave -> Master)
	// Capture Slave request on Master clock
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_cross_valid <= 1'b0;
			m_cross_ready <= 1'b0;
			m_cross_addr <= 12'd0;
			m_cross_write <= 1'b0;
			m_cross_wdata <= {DW{1'b0}};
			m_cross_wstrb <= {WSTRB_WIDTH{1'b0}};
			m_cross_prot <= 3'd0;
		end else begin
			if (s_cross_valid && !s_cross_ready) begin
				m_cross_valid <= 1'b1;
				m_cross_addr <= s_cross_addr;
				m_cross_write <= s_cross_write;
				m_cross_wdata <= s_cross_wdata;
				m_cross_wstrb <= s_cross_wstrb;
				m_cross_prot <= s_cross_prot;
			end else begin
				m_cross_valid <= 1'b0;
				m_cross_ready <= 1'b1;
			end
		end
	end

	// Clock Domain Crossing Logic for Response (Slave -> Master)
	// Capture Slave response on Master clock
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_cross_resp_valid <= 1'b0;
			m_cross_resp_ready <= 1'b0;
			m_cross_rdata <= {DW{1'b0}};
			m_cross_err <= 1'b0;
		end else begin
			if (s_cross_resp_valid && !s_cross_resp_ready) begin
				m_cross_resp_valid <= 1'b1;
				m_cross_rdata <= s_cross_rdata;
				m_cross_err <= s_cross_err;
			end else begin
				m_cross_resp_valid <= 1'b0;
				m_cross_resp_ready <= 1'b1;
			end
		end
	end

	// Clock Domain Crossing Logic for Request (Master -> Slave)
	// Capture Master request on Slave clock
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_cross_valid <= 1'b0;
			s_cross_ready <= 1'b0;
			s_cross_addr <= 12'd0;
			s_cross_write <= 1'b0;
			s_cross_wdata <= {DW{1'b0}};
			s_cross_wstrb <= {WSTRB_WIDTH{1'b0}};
			s_cross_prot <= 3'd0;
		end else begin
			if (m_cross_valid && !m_cross_ready) begin
				s_cross_valid <= 1'b1;
				s_cross_addr <= m_cross_addr;
				s_cross_write <= m_cross_write;
				s_cross_wdata <= m_cross_wdata;
				s_cross_wstrb <= m_cross_wstrb;
				s_cross_prot <= m_cross_prot;
			end else begin
				s_cross_valid <= 1'b0;
				s_cross_ready <= 1'b1;
			end
		end
	end

	// Clock Domain Crossing Logic for Response (Master -> Slave)
	// Capture Master response on Slave clock
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_cross_resp_valid <= 1'b0;
			s_cross_resp_ready <= 1'b0;
			s_cross_rdata <= {DW{1'b0}};
			s_cross_err <= 1'b0;
		end else begin
			if (m_cross_resp_valid && !m_cross_resp_ready) begin
				s_cross_resp_valid <= 1'b1;
				s_cross_rdata <= m_cross_rdata;
				s_cross_err <= m_cross_err;
			end else begin
				s_cross_resp_valid <= 1'b0;
				s_cross_resp_ready <= 1'b1;
			end
		end
	end

	// Master Side Logic (Driving Slave Interface)
	always @(posedge M_APB_PCLK or negedge M_PRESETn) begin
		if (!M_PRESETn) begin
			m_state <= IDLE;
			M_APB_PSEL <= 1'b0;
			M_APB_PENABLE <= 1'b0;
			M_APB_PADDR <= 12'd0;
			M_APB_PWRITE <= 1'b0;
			M_APB_PWDATA <= {DW{1'b0}};
			M_APB_PWSTRB <= {WSTRB_WIDTH{1'b0}};
			M_APB_PPROT <= 3'd0;
			M_APB_PREADY <= 1'b1;
			M_APB_PSLVERR <= 1'b0;
		end else begin
			// Reset logic
			if (!M_PRESETn) begin
				m_state <= IDLE;
				M_APB_PSEL <= 1'b0;
				M_APB_PENABLE <= 1'b0;
				M_APB_PADDR <= 12'd0;
				M_APB_PWRITE <= 1'b0;
				M_APB_PWDATA <= {DW{1'b0}};
				M_APB_PWSTRB <= {WSTRB_WIDTH{1'b0}};
				M_APB_PPROT <= 3'd0;
				M_APB_PREADY <= 1'b1;
				M_APB_PSLVERR <= 1'b0;
			end else begin
				case (m_state)
					IDLE: begin
						if (m_cross_valid && !m_cross_ready) begin
							m_state <= REQ_SENT;
							M_APB_PSEL <= 1'b1;
							M_APB_PENABLE <= 1'b1;
							M_APB_PADDR <= m_cross_addr;
							M_APB_PWRITE <= m_cross_write;
							M_APB_PWDATA <= m_cross_wdata;
							M_APB_PWSTRB <= m_cross_wstrb;
							M_APB_PPROT <= m_cross_prot;
							M_APB_PREADY <= 1'b0; // Wait for slave
							m_cross_ready <= 1'b1;
						end else begin
							M_APB_PSEL <= 1'b0;
							M_APB_PENABLE <= 1'b0;
							M_APB_PREADY <= 1'b1;
							m_cross_ready <= 1'b0;
						end
					end

					REQ_SENT: begin
						if (M_APB_PREADY) begin
							m_state <= WAITING_RESP;
							M_APB_PREADY <= 1'b0;
						end
					end

					WAITING_RESP: begin
						if (m_cross_resp_valid && !m_cross_resp_ready) begin
							M_APB_PSEL <= 1'b0;
							M_APB_PENABLE <= 1'b0;
							M_APB_PREADY <= 1'b1;
							m_cross_resp_ready <= 1'b1;
							m_state <= IDLE;
						end else if (M_APB_PREADY) begin
							m_state <= WAITING_RESP;
							M_APB_PREADY <= 1'b0;
						end
					end
				endcase
			end
		end
	end

	// Slave Side Logic (Driving Master Interface)
	always @(posedge S_APB_PCLK or negedge S_PRESETn) begin
		if (!S_PRESETn) begin
			s_state <= IDLE_S;
			S_APB_PREADY <= 1'b1;
			S_APB_PSLVERR <= 1'b0;
			S_APB_PRDATA <= {DW{1'b0}};
		end else begin
			// Reset logic
			if (!S_PRESETn) begin
				s_state <= IDLE_S;
				S_APB_PREADY <= 1'b1;
				S_APB_PSLVERR <= 1'b0;
				S_APB_PRDATA <= {DW{1'b0}};
			end else begin
				case (s_state)
					IDLE_S: begin
						if (s_cross_valid && !s_cross_ready) begin
							s_state <= REQ_SENT_S;
							S_APB_PREADY <= 1'b0;
							s_cross_ready <= 1'b1;
						end else begin
							S_APB_PREADY <= 1'b1;
							s_cross_ready <= 1'b0;
						end
					end

					REQ_SENT_S: begin
						if (S_APB_PREADY) begin
							s_state <= WAITING_RESP_S;
							S_APB_PREADY <= 1'b0;
						end
					end

					WAITING_RESP_S: begin
						if (s_cross_resp_valid && !s_cross_resp_ready) begin
							S_APB_PREADY <= 1'b1;
							S_APB_PRDATA <= s_cross_rdata;
							S_APB_PSLVERR <= s_cross_err;
							s_cross_resp_ready <= 1'b1;
							s_state <= IDLE_S;
						end else if (S_APB_PREADY) begin
							s_state <= WAITING_RESP_S;
							S_APB_PREADY <= 1'b0;
						end
					end
				endcase
			end
		end
	end

	// Output assignments for Slave Interface
	assign S_APB_PADDR = m_cross_valid ? m_cross_addr : 12'd0;
	assign S_APB_PWRITE = m_cross_valid ? m_cross_write : 1'b0;
	assign S_APB_PWDATA = m_cross_valid ? m_cross_wdata : {DW{1'b0}};
	assign S_APB_PWSTRB = m_cross_valid ? m_cross_wstrb : {WSTRB_WIDTH{1'b0}};
	assign S_APB_PPROT = m_cross_valid ? m_cross_prot : 3'd0;
	assign S_APB_PRDATA = m_cross_resp_valid ? m_cross_rdata : {DW{1'b0}};
	assign S_APB_PSLVERR = m_cross_resp_valid ? m_cross_err : 1'b0;

	// Output assignments for Master Interface
	assign M_APB_PADDR = s_cross_valid ? s_cross_addr : 12'd0;
	assign M_APB_PWRITE = s_cross_valid ? s_cross_write : 1'b0;
	assign M_APB_PWDATA = s_cross_valid ? s_cross_wdata : {DW{1'b0}};
	assign M_APB_PWSTRB = s_cross_valid ? s_cross_wstrb : {WSTRB_WIDTH{1'b0}};
	assign M_APB_PPROT = s_cross_valid ? s_cross_prot : 3'd0;
	assign M_APB_PRDATA = s_cross_resp_valid ? s_cross_rdata : {DW{1'b0}};
	assign M_APB_PSLVERR = s_cross_resp_valid ? s_cross_err : 1'b0;

endmodule
