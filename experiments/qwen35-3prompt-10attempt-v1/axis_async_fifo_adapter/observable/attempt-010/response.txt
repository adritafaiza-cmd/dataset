```verilog
`timescale 1ns/1ps

module axis_async_fifo_adapter #
(
    parameter DEPTH = 4096,
    parameter S_DATA_WIDTH = 8,
    parameter S_KEEP_ENABLE = (S_DATA_WIDTH>8),
    parameter S_KEEP_WIDTH = ((S_DATA_WIDTH+7)/8),
    parameter M_DATA_WIDTH = 8,
    parameter M_KEEP_ENABLE = (M_DATA_WIDTH>8),
    parameter M_KEEP_WIDTH = ((M_DATA_WIDTH+7)/8),
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    input  wire                     s_clk,
    input  wire                     s_rst,
    input  wire [S_DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [S_KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                     s_axis_tvalid,
    output wire                     s_axis_tready,
    input  wire                     s_axis_tlast,
    input  wire [ID_WIDTH-1:0]      s_axis_tid,
    input  wire [DEST_WIDTH-1:0]    s_axis_tdest,
    input  wire [USER_WIDTH-1:0]    s_axis_tuser,
    input  wire                     m_clk,
    input  wire                     m_rst,
    output wire [M_DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [M_KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                     m_axis_tvalid,
    input  wire                     m_axis_tready,
    output wire                     m_axis_tlast,
    output wire [ID_WIDTH-1:0]      m_axis_tid,
    output wire [DEST_WIDTH-1:0]    m_axis_tdest,
    output wire [USER_WIDTH-1:0]    m_axis_tuser,
    input  wire                     s_pause_req,
    output wire                     s_pause_ack,
    input  wire                     m_pause_req,
    output wire                     m_pause_ack,
    output wire [$clog2(DEPTH):0]   s_status_depth,
    output wire [$clog2(DEPTH):0]   s_status_depth_commit,
    output wire                     s_status_overflow,
    output wire                     s_status_bad_frame,
    output wire                     s_status_good_frame,
    output wire [$clog2(DEPTH):0]   m_status_depth,
    output wire [$clog2(DEPTH):0]   m_status_depth_commit,
    output wire                     m_status_overflow,
    output wire                     m_status_bad_frame,
    output wire                     m_status_good_frame
);

    localparam LOG2_DEPTH = $clog2(DEPTH);
    localparam MAX_DEPTH = 2**LOG2_DEPTH;

    // Internal signals for cross-clock domain
    wire s_rst_n = ~s_rst;
    wire m_rst_n = ~m_rst;

    // Cross-clock domain synchronization
    // S -> M (Read side)
    wire s_valid_synced;
    wire s_last_synced;
    wire s_tid_synced;
    wire s_tdest_synced;
    wire s_tuser_synced;
    wire s_tkeep_synced;
    wire s_tdata_synced;
    wire s_pause_req_synced;

    // M -> S (Write side)
    wire m_ready_synced;
    wire m_valid_synced;
    wire m_last_synced;
    wire m_tid_synced;
    wire m_tdest_synced;
    wire m_tuser_synced;
    wire m_tkeep_synced;
    wire m_tdata_synced;
    wire m_pause_req_synced;

    // Synchronizers (2-stage)
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_valid_synced <= 1'b0;
            s_last_synced <= 1'b0;
            s_tid_synced <= '{default: 0};
            s_tdest_synced <= '{default: 0};
            s_tuser_synced <= '{default: 0};
            s_tkeep_synced <= '{default: 0};
            s_tdata_synced <= '{default: 0};
            s_pause_req_synced <= 1'b0;
        end else begin
            s_valid_synced <= s_axis_tvalid;
            s_last_synced <= s_axis_tlast;
            s_tid_synced <= s_axis_tid;
            s_tdest_synced <= s_axis_tdest;
            s_tuser_synced <= s_axis_tuser;
            s_tkeep_synced <= s_axis_tkeep;
            s_tdata_synced <= s_axis_tdata;
            s_pause_req_synced <= s_pause_req;
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_ready_synced <= 1'b0;
            m_valid_synced <= 1'b0;
            m_last_synced <= 1'b0;
            m_tid_synced <= '{default: 0};
            m_tdest_synced <= '{default: 0};
            m_tuser_synced <= '{default: 0};
            m_tkeep_synced <= '{default: 0};
            m_tdata_synced <= '{default: 0};
            m_pause_req_synced <= 1'b0;
        end else begin
            m_ready_synced <= m_axis_tready;
            m_valid_synced <= m_axis_tvalid;
            m_last_synced <= m_axis_tlast;
            m_tid_synced <= m_axis_tid;
            m_tdest_synced <= m_axis_tdest;
            m_tuser_synced <= m_axis_tuser;
            m_tkeep_synced <= m_axis_tkeep;
            m_tdata_synced <= m_axis_tdata;
            m_pause_req_synced <= m_pause_req;
        end
    end

    // FIFO Depth Status
    assign s_status_depth = LOG2_DEPTH[LOG2_DEPTH-1:0]; // Simplified for constant depth
    assign s_status_depth_commit = LOG2_DEPTH[LOG2_DEPTH-1:0];
    assign m_status_depth = LOG2_DEPTH[LOG2_DEPTH-1:0];
    assign m_status_depth_commit = LOG2_DEPTH[LOG2_DEPTH-1:0];

    // Internal FIFO State
    reg [LOG2_DEPTH-1:0] write_ptr;
    reg [LOG2_DEPTH-1:0] read_ptr;
    reg [DEPTH-1:0] valid_bits;
    reg [DEPTH-1:0] data_storage;
    reg [DEPTH-1:0] keep_storage;
    reg [DEPTH-1:0] last_storage;
    reg [DEPTH-1:0] tid_storage;
    reg [DEPTH-1:0] dest_storage;
    reg [DEPTH-1:0] user_storage;
    reg [DEPTH-1:0] pause_storage;
    reg [DEPTH-1:0] bad_frame_storage;
    reg [DEPTH-1:0] oversize_frame_storage;

    // Pipeline registers for RAM_PIPELINE
    reg [DEPTH-1:0] pipeline_data;
    reg [DEPTH-1:0] pipeline_keep;
    reg [DEPTH-1:0] pipeline_last;
    reg [DEPTH-1:0] pipeline_tid;
    reg [DEPTH-1:0] pipeline_dest;
    reg [DEPTH-1:0] pipeline_user;
    reg [DEPTH-1:0] pipeline_pause;
    reg [DEPTH-1:0] pipeline_bad_frame;
    reg [DEPTH-1:0] pipeline_oversize_frame;

    // Frame boundaries tracking
    reg [DEPTH-1:0] frame_start_ptr;
    reg [DEPTH-1:0] frame_end_ptr;
    reg frame_in_progress;
    reg frame_oversized;
    reg frame_bad;

    // Write Logic (S Domain)
    reg s_write_enable;
    reg s_pause_ack_internal;
    reg s_bad_frame_flag;
    reg s_oversize_flag;

    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            write_ptr <= 0;
            read_ptr <= 0;
            valid_bits <= 0;
            data_storage <= '{default: 0};
            keep_storage <= '{default: 0};
            last_storage <= '{default: 0};
            tid_storage <= '{default: 0};
            dest_storage <= '{default: 0};
            user_storage <= '{default: 0};
            pause_storage <= '{default: 0};
            bad_frame_storage <= '{default: 0};
            oversize_frame_storage <= '{default: 0};
            frame_start_ptr <= 0;
            frame_end_ptr <= 0;
            frame_in_progress <= 0;
            frame_oversized <= 0;
            frame_bad <= 0;
            s_write_enable <= 0;
            s_pause_ack_internal <= 0;
            s_bad_frame_flag <= 0;
            s_oversize_flag <= 0;
        end else begin
            // Frame boundary detection
            if (s_axis_tvalid && s_axis_tlast) begin
                frame_in_progress <= 0;
                frame_oversized <= 0;
                frame_bad <= 0;
                frame_start_ptr <= write_ptr;
                frame_end_ptr <= write_ptr;
            end else if (s_axis_tvalid && !frame_in_progress) begin
                frame_in_progress <= 1;
                frame_start_ptr <= write_ptr;
            end

            // Check for oversized frame
            if (frame_in_progress && s_axis_tvalid) begin
                if (write_ptr == frame_start_ptr) begin
                    // Start of frame
                    frame_oversized <= 0;
                end else if (write_ptr == frame_end_ptr) begin
                    // End of frame
                    if (write_ptr == frame_start_ptr) begin
                        // Single byte frame
                        frame_oversized <= 0;
                    end else begin
                        // Multi-byte frame completed
                        frame_oversized <= 0;
                    end
                end else begin
                    // Middle of frame
                    frame_oversized <= 1;
                end
            end

            // Write enable logic
            s_write_enable <= 0;
            if (s_axis_tvalid && m_ready_synced) begin
                if (write_ptr == 0) begin
                    if (read_ptr == 0) begin
                        // FIFO empty, can write
                        s_write_enable <= 1;
                    end else begin
                        // FIFO not empty, but read_ptr != 0 means we have data
                        // Actually, if read_ptr != 0, we have data to read
                        // But we need to check if we are at the start of a frame
                        if (!frame_in_progress) begin
                            s_write_enable <= 1;
                        end
                    end
                end else begin
                    // Check if we are at the start of a frame
                    if (!frame_in_progress) begin
                        s_write_enable <= 1;
                    end
                end
            end

            // Pause logic
            if (s_pause_req) begin
                s_pause_ack_internal <= 1;
            end else begin
                s_pause_ack_internal <= 0;
            end

            // Bad frame logic
            if (s_axis_tvalid && s_axis_tlast) begin
                if (frame_oversized) begin
                    s_bad_frame_flag <= 1;
                end else if (frame_bad) begin
                    s_bad_frame_flag <= 1;
                end else begin
                    s_bad_frame_flag <= 0;
                end
            end

            // Oversize flag
            if (s_axis_tvalid && s_axis_tlast) begin
                if (frame_oversized) begin
                    s_oversize_flag <= 1;
                end else begin
                    s_oversize_flag <= 0;
                end
            end

            // Update pointers and storage
            if (s_write_enable) begin
                if (write_ptr == 0) begin
                    // First byte of frame
                    frame_start_ptr <= write_ptr;
                    frame_end_ptr <= write_ptr;
                end else if (write_ptr == frame_end_ptr) begin
                    // Last byte of frame
                    frame_end_ptr <= write_ptr;
                    frame_in_progress <= 0;
                end

                // Check for drop conditions
                if (DROP_WHEN_FULL && write_ptr == (MAX_DEPTH - 1)) begin
                    s_write_enable <= 0;
                end

                // Write data
                data_storage[write_ptr] <= s_tdata_synced;
                keep_storage[write_ptr] <= s_tkeep_synced;
                last_storage[write_ptr] <= s_last_synced;
                tid_storage[write_ptr] <= s_tid_synced;
                dest_storage[write_ptr] <= s_tdest_synced;
                user_storage[write_ptr] <= s_tuser_synced;
                pause_storage[write_ptr] <= s_pause_req_synced;
                bad_frame_storage[write_ptr] <= s_bad_frame_flag;
                oversize_frame_storage[write_ptr] <= s_oversize_flag;

                write_ptr <= write_ptr + 1;
            end
        end
    end

    // Read Logic (M Domain)
    reg m_read_enable;
    reg m_pause_ack_internal;
    reg m_bad_frame_flag;
    reg m_oversize_flag;

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            read_ptr <= 0;
            m_read_enable <= 0;
            m_pause_ack_internal <= 0;
            m_bad_frame_flag <= 0;
            m_oversize_flag <= 0;
        end else begin
            // Read enable logic
            m_read_enable <= 0;
            if (m_valid_synced && s_pause_ack_internal) begin
                // Check if we have data to read
                if (read_ptr != write_ptr) begin
                    m_read_enable <= 1;
                end
            end

            // Pause logic
            if (m_pause_req) begin
                m_pause_ack_internal <= 1;
            end else begin
                m_pause_ack_internal <= 0;
            end

            // Bad frame logic
            if (m_valid_synced && s_pause_ack_internal) begin
                if (read_ptr != write_ptr) begin
                    if (bad_frame_storage[read_ptr]) begin
                        m_bad_frame_flag <= 1;
                    end else begin
                        m_bad_frame_flag <= 0;
                    end
                end
            end

            // Oversize flag
            if (m_valid_synced && s_pause_ack_internal) begin
                if (read_ptr != write_ptr) begin
                    if (oversize_frame_storage[read_ptr]) begin
                        m_oversize_flag <= 1;
                    end else begin
                        m_oversize_flag <= 0;
                    end
                end
            end

            // Update pointers and storage
            if (m_read_enable) begin
                // Check for drop conditions
                if (DROP_WHEN_FULL && write_ptr == (MAX_DEPTH - 1)) begin
                    m_read_enable <= 0;
                end

                // Read data
                pipeline_data[read_ptr] <= data_storage[read_ptr];
                pipeline_keep[read_ptr] <= keep_storage[read_ptr];
                pipeline_last[read_ptr] <= last_storage[read_ptr];
                pipeline_tid[read_ptr] <= tid_storage[read_ptr];
                pipeline_dest[read_ptr] <= dest_storage[read_ptr];
                pipeline_user[read_ptr] <= user_storage[read_ptr];
                pipeline_pause[read_ptr] <= pause_storage[read_ptr];
                pipeline_bad_frame[read_ptr] <= bad_frame_storage[read_ptr];
                pipeline_oversize_frame[read_ptr] <= oversize_frame_storage[read_ptr];

                read_ptr <= read_ptr + 1;
            end
        end
    end

    // Output Logic (M Domain)
    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_axis_tvalid <= 1'b0;
            m_axis_tdata <= '{default: 0};
            m_axis_tkeep <= '{default: 0};
            m_axis_tlast <= 1'b0;
            m_axis_tid <= '{default: 0};
            m_axis_tdest <= '{default: 0};
            m_axis_tuser <= '{default: 0};
            m_axis_tready <= 1'b0;
            m_axis_tlast <= 1'b0;
            m_axis_tkeep <= '{default: 0};
            m_axis_tid <= '{default: 0};
            m_axis_tdest <= '{default: 0};
            m_axis_tuser <= '{default: 0};
            m_status_overflow <= 0;
            m_status_bad_frame <= 0;
            m_status_good_frame <= 0;
        end else begin
            // Check if we have data to read
            if (read_ptr != write_ptr) begin
                // Check for bad frame
                if (pipeline_bad_frame[read_ptr]) begin
                    if (DROP_BAD_FRAME) begin
                        // Drop bad frame
                        m_axis_tvalid <= 1'b0;
                        m_axis_tdata <= '{default: 0};
                        m_axis_tkeep <= '{default: 0};
                        m_axis_tlast <= 1'b0;
                        m_axis_tid <= '{default: 0};
                        m_axis_tdest <= '{default: 0};
                        m_axis_tuser <= '{default: 0};
                        m_status_bad_frame <= 1;
                        m_status_good_frame <= 0;
                    end else begin
                        // Mark bad frame
                        m_axis_tvalid <= 1'b0;
                        m_axis_tdata <= '{default: 0};
                        m_axis_tkeep <= '{default: 0};
                        m_axis_tlast <= 1'b0;
                        m_axis_tid <= '{default: 0};
                        m_axis_tdest <= '{default: 0};
                        m_axis_tuser <= '{default: 0};
                        m_status_bad_frame <= 1;
                        m_status_good_frame <= 0;
                    end
                end else begin
                    // Check for oversize frame
                    if (pipeline_oversize_frame[read_ptr]) begin
                        if (DROP_OVERSIZE_FRAME) begin
                            // Drop oversize frame
                            m_axis_tvalid <= 1'b0;
                            m_axis_tdata <= '{default: 0};
                            m_axis_tkeep <= '{default: 0};
                            m_axis_tlast <= 1'b0;
                            m_axis_tid <= '{default: 0};
                            m_axis_tdest <= '{default: 0};
                            m_axis_tuser <= '{default: 0};
                            m_status_overflow <= 1;
                            m_status_bad_frame <= 0;
                            m_status_good_frame <= 0;
                        end else begin
                            // Mark oversize frame
                            m_axis_tvalid <= 1'b0;
                            m_axis_tdata <= '{default: 0};
                            m_axis_tkeep <= '{default: 0};
                            m_axis_tlast <= 1'b0;
                            m_axis_tid <= '{default: 0};
                            m_axis_tdest <= '{default: 0};
                            m_axis_tuser <= '{default: 0};
                            m_status_overflow <= 1;
                            m_status_bad_frame <= 0;
                            m_status_good_frame <= 0;
                        end
                    end else begin
                        // Normal data
                        m_axis_tvalid <= 1'b1;
                        m_axis_tdata <= pipeline_data[read_ptr];
                        m_axis_tkeep <= pipeline_keep[read_ptr];
                        m_axis_tlast <= pipeline_last[read_ptr];
                        m_axis_tid <= pipeline_tid[read_ptr];
                        m_axis_tdest <= pipeline_dest[read_ptr];
                        m_axis_tuser <= pipeline_user[read_ptr];
                        m_status_overflow <= 0;
                        m_status_bad_frame <= 0;
                        m_status_good_frame <= 1;
                    end
                end
            end else begin
                // FIFO empty
                m_axis_tvalid <= 1'b0;
                m_axis_tdata <= '{default: 0};
                m_axis_tkeep <= '{default: 0};
                m_axis_tlast <= 1'b0;
                m_axis_tid <= '{default: 0};
                m_axis_tdest <= '{default: 0};
                m_axis_tuser <= '{default: 0};
                m_status_overflow <= 0;
                m_status_bad_frame <= 0;
                m_status_good_frame <= 0;
            end
        end
    end

    // Input Logic (S Domain)
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_axis_tready <= 1'b0;
            s_pause_ack <= 0;
            s_status_overflow <= 0;
            s_status_bad_frame <= 0;
            s_status_good_frame <= 0;
        end else begin
            // Check if we can write
            if (write_ptr != read_ptr) begin
                // Check for bad frame
                if (bad_frame_storage[write_ptr]) begin
                    if (DROP_BAD_FRAME) begin
                        // Drop bad frame
                        s_axis_tready <= 1'b0;
                        s_pause_ack <= 0;
                        s_status_bad_frame <= 1;
                        s_status_good_frame <= 0;
                    end else begin
                        // Mark bad frame
                        s_axis_tready <= 1'b0;
                        s_pause_ack <= 0;
                        s_status_bad_frame <= 1;
                        s_status_good_frame <= 0;
                    end
                end else begin
                    // Check for oversize frame
                    if (oversize_frame_storage[write_ptr]) begin
                        if (DROP_OVERSIZE_FRAME) begin
                            // Drop oversize frame
                            s_axis_tready <= 1'b0;
                            s_pause_ack <= 0;
                            s_status_overflow <= 1;
                            s_status_bad_frame <= 0;
                            s_status_good_frame <= 0;
                        end else begin
                            // Mark oversize frame
                            s_axis_tready <= 1'b0;
                            s_pause_ack <= 0;
                            s_status_overflow <= 1;
                            s_status_bad_frame <= 0;
                            s_status_good_frame <= 0;
                        end
                    end else begin
                        // Normal data
                        s_axis_tready <= 1'b1;
                        s_pause_ack <= 0;
                        s_status_overflow <= 0;
                        s_status_bad_frame <= 0;
                        s_status_good_frame <= 1;
                    end
                end
            end else begin
                // FIFO full
                if (DROP_WHEN_FULL) begin
                    s_axis_tready <= 1'b0;
                    s_pause_ack <= 0;
                    s_status_overflow <= 1;
                    s_status_bad_frame <= 0;
                    s_status_good_frame <= 0;
                end else begin
                    s_axis_tready <= 1'b0;
                    s_pause_ack <= 0;
                    s_status_overflow <= 1;
                    s_status_bad_frame <= 0;
                    s_status_good_frame <= 0;
                end
            end
        end
    end

    // Pause Logic
    always @(posedge s_clk or posedge s_rst) begin
        if (s_rst) begin
            s_pause_ack <= 0;
        end else begin
            if (s_pause_req) begin
                s_pause_ack <= 1;
            end else begin
                s_pause_ack <= 0;
            end
        end
    end

    always @(posedge m_clk or posedge m_rst) begin
        if (m_rst) begin
            m_pause_ack <= 0;
        end else begin
            if (m_pause_req) begin
                m_pause_ack <= 1;
            end else begin
                m_pause_ack <= 0;
            end
        end
    end

    // Status outputs
    assign s_status_depth = LOG2_DEPTH[LOG2_DEPTH-1:0];
    assign s_status_depth_commit = LOG2_DEPTH[LOG2_DEPTH-1:0];
    assign m_status_depth = LOG2_DEPTH[LOG2_DEPTH-1:0];
    assign m_status_depth_commit = LOG2_DEPTH[LOG2_DEPTH-1:0];

endmodule
