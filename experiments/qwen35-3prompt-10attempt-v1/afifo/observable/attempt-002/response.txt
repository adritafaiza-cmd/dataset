module afifo #(
	parameter LGFIFO = 3,
	parameter WIDTH  = 16,
	parameter NFF    = 2,
	parameter [0:0]  WRITE_ON_POSEDGE = 1'b1,
	parameter [0:0]  OPT_REGISTER_READS = 1'b1
) (
	input  wire            i_wclk, i_wr_reset_n, i_wr,
	input  wire [WIDTH-1:0] i_wr_data,
	output reg             o_wr_full,
	input  wire            i_rclk, i_rd_reset_n, i_rd,
	output reg [WIDTH-1:0] o_rd_data,
	output reg             o_rd_empty
);

	localparam DEPTH = 1 << LGFIFO;
	localparam FULL_THRESHOLD = DEPTH - NFF;
	localparam EMPTY_THRESHOLD = NFF;

	// Write side logic
	reg [DEPTH-1:0] wr_ptr, wr_gray;
	reg [DEPTH-1:0] rd_ptr, rd_gray;
	reg [WIDTH-1:0] mem [DEPTH-1:0];
	reg [LGFIFO-1:0] wr_cnt, rd_cnt;

	// Write pointer update
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			wr_ptr <= '0;
			wr_gray <= '0;
			wr_cnt <= '0;
		end else if (i_wr) begin
			if (wr_cnt < FULL_THRESHOLD) begin
				wr_cnt <= wr_cnt + 1'b1;
				wr_ptr <= wr_ptr + 1'b1;
				mem[wr_ptr] <= i_wr_data;
			end
		end
	end

	// Write pointer to Gray code
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n)
			wr_gray <= '0;
		else
			wr_gray <= wr_ptr ^ (wr_ptr >> (LGFIFO-1));
	end

	// Read pointer update
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			rd_ptr <= '0;
			rd_gray <= '0;
			rd_cnt <= '0;
		end else if (i_rd) begin
			if (rd_cnt < EMPTY_THRESHOLD) begin
				rd_cnt <= rd_cnt + 1'b1;
				rd_ptr <= rd_ptr + 1'b1;
			end
		end
	end

	// Read pointer to Gray code
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n)
			rd_gray <= '0;
		else
			rd_gray <= rd_ptr ^ (rd_ptr >> (LGFIFO-1));
	end

	// Output data selection
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			o_rd_data <= '0;
			o_rd_empty <= 1'b1;
		end else if (i_rd) begin
			if (rd_cnt < EMPTY_THRESHOLD) begin
				o_rd_data <= mem[rd_ptr];
				o_rd_empty <= 1'b0;
			end else begin
				o_rd_data <= '0;
				o_rd_empty <= 1'b1;
			end
		end
	end

	// Full flag logic
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n)
			o_wr_full <= 1'b0;
		else if (wr_cnt >= FULL_THRESHOLD)
			o_wr_full <= 1'b1;
		else
			o_wr_full <= 1'b0;
	end

	// Empty flag logic (asynchronous comparison with Gray codes)
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n)
			o_rd_empty <= 1'b1;
		else if (rd_gray == wr_gray)
			o_rd_empty <= 1'b1;
		else if (rd_gray > wr_gray)
			o_rd_empty <= 1'b1;
		else
			o_rd_empty <= 1'b0;
	end

endmodule
