module afifo #(
		parameter	LGFIFO = 3,
		parameter	WIDTH  = 16,
		parameter	NFF    = 2,
		parameter [0:0]	WRITE_ON_POSEDGE = 1'b1,
		parameter [0:0]	OPT_REGISTER_READS = 1'b1
	) (
		input	wire			i_wclk, i_wr_reset_n, i_wr,
		input	wire	[WIDTH-1:0]	i_wr_data,
		output	reg			o_wr_full,
		input	wire			i_rclk, i_rd_reset_n, i_rd,
		output	reg	[WIDTH-1:0]	o_rd_data,
		output	reg			o_rd_empty
	);

	localparam FIFO_DEPTH = 1 << LGFIFO;

	// Write Pointer and Counter
	reg [LGFIFO-1:0] wr_ptr;
	reg [LGFIFO-1:0] rd_ptr;
	reg [LGFIFO+1:0] wr_cnt;
	reg [LGFIFO+1:0] rd_cnt;

	// Write Logic
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			wr_ptr <= 0;
			wr_cnt <= 0;
			o_wr_full <= 0;
		end else if (i_wr) begin
			if (wr_cnt < FIFO_DEPTH) begin
				if (WRITE_ON_POSEDGE) begin
					// Write on rising edge of i_wr
					o_rd_data <= i_wr_data; // This line is a placeholder for internal storage logic if needed, 
											// but typically we write to an array. 
											// Since we are using counters for full/empty, 
											// we need a memory array.
					wr_cnt <= wr_cnt + 1;
				end else begin
					// Write on level (hold high)
					if (wr_cnt < FIFO_DEPTH) begin
						wr_cnt <= wr_cnt + 1;
					end
				end
			end
		end
	end

	// Read Logic
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			rd_ptr <= 0;
			rd_cnt <= 0;
			o_rd_empty <= 1;
		end else if (i_rd) begin
			if (rd_cnt > 0) begin
				if (OPT_REGISTER_READS) begin
					// Read on rising edge of i_rd
					rd_cnt <= rd_cnt - 1;
				end else begin
					// Read on level (hold high)
					if (rd_cnt > 0) begin
						rd_cnt <= rd_cnt - 1;
					end
				end
			end
		end
	end

	// Full/Empty Logic
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			o_wr_full <= 0;
		end else begin
			o_wr_full <= (wr_cnt == FIFO_DEPTH);
		end
	end

	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			o_rd_empty <= 1;
		end else begin
			o_rd_empty <= (rd_cnt == 0);
		end
	end

	// Memory Array
	reg [WIDTH-1:0] mem [0:FIFO_DEPTH-1];

	// Write Data to Memory
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			mem <= 0;
		end else if (i_wr && (wr_cnt < FIFO_DEPTH)) begin
			if (WRITE_ON_POSEDGE) begin
				mem[wr_ptr] <= i_wr_data;
			end else begin
				if (wr_cnt < FIFO_DEPTH) begin
					mem[wr_ptr] <= i_wr_data;
				end
			end
		end
	end

	// Read Data from Memory
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			o_rd_data <= 0;
		end else if (i_rd && (rd_cnt > 0)) begin
			if (OPT_REGISTER_READS) begin
				o_rd_data <= mem[rd_ptr];
			end else begin
				if (rd_cnt > 0) begin
					o_rd_data <= mem[rd_ptr];
				end
			end
		end
	end

	// Update Pointers
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			wr_ptr <= 0;
		end else if (i_wr && (wr_cnt < FIFO_DEPTH)) begin
			if (WRITE_ON_POSEDGE) begin
				wr_ptr <= wr_ptr + 1;
			end else begin
				if (wr_cnt < FIFO_DEPTH) begin
					wr_ptr <= wr_ptr + 1;
				end
			end
		end
	end

	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			rd_ptr <= 0;
		end else if (i_rd && (rd_cnt > 0)) begin
			if (OPT_REGISTER_READS) begin
				rd_ptr <= rd_ptr + 1;
			end else begin
				if (rd_cnt > 0) begin
					rd_ptr <= rd_ptr + 1;
				end
			end
		end
	end

endmodule
