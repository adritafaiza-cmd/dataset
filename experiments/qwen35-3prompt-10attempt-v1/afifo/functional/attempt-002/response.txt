`timescale 1ns/1ps

module afifo #(
	parameter LGFIFO = 3,
	parameter WIDTH  = 16,
	parameter NFF    = 2,
	parameter [0:0] WRITE_ON_POSEDGE = 1'b1,
	parameter [0:0] OPT_REGISTER_READS = 1'b1
) (
	input  wire i_wclk, i_wr_reset_n, i_wr,
	input  wire [WIDTH-1:0] i_wr_data,
	output reg o_wr_full,
	input  wire i_rclk, i_rd_reset_n, i_rd,
	output reg [WIDTH-1:0] o_rd_data,
	output reg o_rd_empty
);

	localparam FIFO_DEPTH = 1 << LGFIFO;
	localparam GRAY_WIDTH = LGFIFO;

	// Gray code generator for address
	reg [GRAY_WIDTH-1:0] gray_addr_w, gray_addr_r;
	reg [GRAY_WIDTH-1:0] next_gray_addr;

	// Write pointer logic
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n)
			gray_addr_w <= 0;
		else if (i_wr && !o_wr_full) begin
			if (WRITE_ON_POSEDGE)
				gray_addr_w <= next_gray_addr;
			else
				gray_addr_w <= next_gray_addr;
		end
	end

	// Read pointer logic
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n)
			gray_addr_r <= 0;
		else if (i_rd && !o_rd_empty) begin
			gray_addr_r <= next_gray_addr;
		end
	end

	// Generate next Gray code
	always @(*) begin
		next_gray_addr = gray_addr_w;
		if (WRITE_ON_POSEDGE)
			next_gray_addr = gray_addr_w;
		else
			next_gray_addr = gray_addr_w;
	end

	// Gray to binary conversion
	function [GRAY_WIDTH-1:0] gray2bin;
		input [GRAY_WIDTH-1:0] gray;
		reg [GRAY_WIDTH-1:0] bin;
		reg [GRAY_WIDTH-1:0] mask;
		reg carry;
		begin
			bin = 0;
			carry = 1'b1;
			mask = 1;
			for (integer i = GRAY_WIDTH-1; i >= 0; i = i - 1) begin
				bin[i] = gray[i] ^ carry;
				carry = carry ^ gray[i];
				mask = mask << 1;
			end
			gray2bin = bin;
		end
	endfunction

	// Binary to Gray conversion
	function [GRAY_WIDTH-1:0] bin2gray;
		input [GRAY_WIDTH-1:0] bin;
		reg [GRAY_WIDTH-1:0] gray;
		reg [GRAY_WIDTH-1:0] mask;
		reg carry;
		begin
			gray = 0;
			carry = 1'b1;
			mask = 1;
			for (integer i = GRAY_WIDTH-1; i >= 0; i = i - 1) begin
				gray[i] = bin[i] ^ carry;
				carry = carry ^ bin[i];
				mask = mask << 1;
			end
			bin2gray = gray;
		endfunction

	// Write pointer (binary)
	reg [LGFIFO-1:0] wr_ptr;
	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n)
			wr_ptr <= 0;
		else if (i_wr && !o_wr_full)
			wr_ptr <= wr_ptr + 1;
	end

	// Read pointer (binary)
	reg [LGFIFO-1:0] rd_ptr;
	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n)
			rd_ptr <= 0;
		else if (i_rd && !o_rd_empty)
			rd_ptr <= rd_ptr + 1;
	end

	// Full flag
	assign o_wr_full = (wr_ptr == (FIFO_DEPTH - 1)) && (rd_ptr == 0);

	// Empty flag
	assign o_rd_empty = (rd_ptr == 0) && (wr_ptr == 0);

	// FIFO memory
	reg [WIDTH-1:0] mem [0:FIFO_DEPTH-1];
	reg [LGFIFO-1:0] wr_ptr_gray, rd_ptr_gray;

	always @(posedge i_wclk or negedge i_wr_reset_n) begin
		if (!i_wr_reset_n) begin
			mem <= 0;
			wr_ptr_gray <= 0;
		end
		else if (i_wr && !o_wr_full) begin
			mem[wr_ptr_gray] <= i_wr_data;
			wr_ptr_gray <= next_gray_addr;
		end
	end

	always @(posedge i_rclk or negedge i_rd_reset_n) begin
		if (!i_rd_reset_n) begin
			o_rd_data <= 0;
			rd_ptr_gray <= 0;
		end
		else if (i_rd && !o_rd_empty) begin
			if (OPT_REGISTER_READS)
				o_rd_data <= mem[rd_ptr_gray];
			else
				o_rd_data <= mem[rd_ptr_gray];
			rd_ptr_gray <= next_gray_addr;
		end
	end

	// Full flag logic for write enable
	assign o_wr_full = (wr_ptr_gray == bin2gray(FIFO_DEPTH - 1)) && (rd_ptr_gray == 0);

	// Empty flag logic for read enable
	assign o_rd_empty = (rd_ptr_gray == 0) && (wr_ptr_gray == 0);

endmodule
