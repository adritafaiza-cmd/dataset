module apbslave #(
	parameter	C_APB_ADDR_WIDTH = 12,
	parameter	C_APB_DATA_WIDTH = 32,
	localparam	AW = C_APB_ADDR_WIDTH,
	localparam	DW = C_APB_DATA_WIDTH,
	localparam	APBLSB = $clog2(C_APB_DATA_WIDTH)-3
) (
	input	wire			PCLK, PRESETn,
	input	wire			PSEL,
	input	wire			PENABLE,
	output	reg			PREADY,
	input	wire	[AW-1:0]	PADDR,
	input	wire			PWRITE,
	input	wire	[DW-1:0]	PWDATA,
	input	wire	[DW/8-1:0]	PWSTRB,
	input	wire	[2:0]		PPROT,
	output	reg	[DW-1:0]	PRDATA,
	output	wire			PSLVERR
);

	reg	[AW-1:0]	ADDR;
	reg	[DW-1:0]	DATA;
	reg			STATE;
	reg			READ;
	reg			WRITE;
	reg			VALID;
	reg			STRB;

	always @(posedge PCLK or negedge PRESETn) begin
		if (!PRESETn) begin
			STATE <= 1'b0;
			ADDR  <= 1'b0;
			DATA  <= 1'b0;
			READ  <= 1'b0;
			WRITE <= 1'b0;
			VALID <= 1'b0;
			STRB  <= 1'b0;
			PREADY <= 1'b0;
			PRDATA <= 1'b0;
			PSLVERR <= 1'b0;
		end
		else begin
			if (PSEL && PENABLE) begin
				ADDR  <= PADDR;
				DATA  <= PWDATA;
				READ  <= ~PWRITE;
				WRITE <= PWRITE;
				VALID <= 1'b1;
				STRB  <= PWSTRB;
				PREADY <= 1'b1;
				PSLVERR <= 1'b0;
			end
			else begin
				VALID <= 1'b0;
				PREADY <= 1'b0;
			end
		end
	end

	always @(posedge PCLK or negedge PRESETn) begin
		if (!PRESETn) begin
			PRDATA <= 1'b0;
		end
		else begin
			if (VALID) begin
				if (WRITE) begin
					PRDATA <= DATA;
				end
				else begin
					PRDATA <= DATA;
				end
			end
			else begin
				PRDATA <= 1'b0;
			end
		end
	end

	assign PSLVERR = 1'b0;

endmodule<|im_end|>
