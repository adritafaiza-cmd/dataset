module apb_regs #(
  parameter int unsigned        NoApbRegs    = 32'd0,
  parameter int unsigned        ApbAddrWidth = 32'd0,
  parameter int unsigned        AddrOffset   = 32'd4,
  parameter int unsigned        ApbDataWidth = 32'd0,
  parameter int unsigned        RegDataWidth = 32'd0,
  parameter bit [NoApbRegs-1:0] ReadOnly     = 32'h0,
  parameter type                req_t        = logic,
  parameter type                resp_t       = logic,
  parameter type apb_addr_t                  = logic[ApbAddrWidth-1:0],
  parameter type reg_data_t                  = logic[RegDataWidth-1:0]
) (
  input  logic                      pclk_i,
  input  logic                      preset_ni,
  input  req_t                      req_i,
  output resp_t                     resp_o,
  input  apb_addr_t                 base_addr_i,
  input  reg_data_t [NoApbRegs-1:0] reg_init_i,
  output reg_data_t [NoApbRegs-1:0] reg_q_o
);

  // Internal types for APB protocol signals
  typedef logic [ApbAddrWidth-1:0] apb_addr_internal_t;
  typedef logic [ApbDataWidth-1:0] apb_data_internal_t;
  
  // APB Protocol Signals
  logic paddr_i;
  logic pwrite_i;
  logic pend_i;
  logic pvalid_i;
  logic pready_o;
  apb_data_internal_t pdata_i;
  apb_data_internal_t presp_o;

  // Internal register storage
  reg_data_t [NoApbRegs-1:0] reg_storage;

  // APB Master Interface Generation
  assign paddr_i = base_addr_i + (req_i.addr * AddrOffset);
  assign pwrite_i = req_i.write;
  assign pend_i = req_i.end;
  assign pvalid_i = req_i.valid;
  assign pdata_i = req_i.data;

  // APB Slave Interface Generation
  assign pready_o = req_i.ready;
  assign presp_o = resp_i;

  // APB Protocol Logic
  always @(posedge pclk_i or negedge preset_ni) begin
    if (!preset_ni) begin
      reg_storage <= '0;
      resp_o <= 1'b0;
    end else begin
      if (pend_i) begin
        if (pwrite_i) begin
          // Write operation
          if (paddr_i < NoApbRegs * AddrOffset) begin
            // Calculate register index
            int unsigned idx = paddr_i / AddrOffset;
            
            // Check if this register is read-only
            if (ReadOnly[idx]) begin
              // Ignore write for read-only registers
              // Do nothing
            end else begin
              // Update register value
              reg_storage[idx] <= reg_data_t'(pdata_i);
            end
          end
        end else begin
          // Read operation
          if (paddr_i < NoApbRegs * AddrOffset) begin
            // Calculate register index
            int unsigned idx = paddr_i / AddrOffset;
            
            // Drive response data
            presp_o <= reg_data_t'(reg_storage[idx]);
          end else begin
            // Invalid address
            presp_o <= '0;
          end
        end
      end
    end
  end

  // Output register array
  assign reg_q_o = reg_storage;

endmodule
