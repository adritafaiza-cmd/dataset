```verilog
`timescale 1ns/1ps

module axis_fifo #
(
    parameter DEPTH = 4096,
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter RAM_PIPELINE = 1,
    parameter OUTPUT_FIFO_ENABLE = 0,
    parameter FRAME_FIFO = 0,
    parameter USER_BAD_FRAME_VALUE = 1'b1,
    parameter USER_BAD_FRAME_MASK = 1'b1,
    parameter DROP_OVERSIZE_FRAME = FRAME_FIFO,
    parameter DROP_BAD_FRAME = 0,
    parameter DROP_WHEN_FULL = 0,
    parameter MARK_WHEN_FULL = 0,
    parameter PAUSE_ENABLE = 0,
    parameter FRAME_PAUSE = FRAME_FIFO
)
(
    input  wire                   clk,
    input  wire                   rst,
    /*
     * AXI input
     */
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI output
     */
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser,
    /*
     * Pause
     */
    input  wire                   pause_req,
    output wire                   pause_ack,
    /*
     * Status
     */
    output wire [$clog2(DEPTH):0] status_depth,
    output wire [$clog2(DEPTH):0] status_depth_commit,
    output wire                   status_overflow,
    output wire                   status_bad_frame,
    output wire                   status_good_frame
);

    // Internal parameters
    localparam LOG2_DEPTH = $clog2(DEPTH);
    localparam MAX_DEPTH = (1 << LOG2_DEPTH);
    
    // Determine if we need to handle frames (start/end of frame logic)
    localparam NEED_FRAME_LOGIC = (FRAME_FIFO || DROP_OVERSIZE_FRAME || MARK_WHEN_FULL);
    
    // Frame state machine states
    localparam FRAME_STATE_IDLE = 2'b00;
    localparam FRAME_STATE_START = 2'b01;
    localparam FRAME_STATE_DATA = 2'b10;
    localparam FRAME_STATE_END = 2'b11;

    // --- Input Sideband Logic ---
    wire [USER_WIDTH-1:0] s_axis_tuser_full;
    assign s_axis_tuser_full = s_axis_tuser;

    // Frame detection logic
    wire frame_start;
    wire frame_end;
    wire frame_oversize;
    wire frame_bad;
    
    reg [USER_WIDTH-1:0] prev_user;
    reg frame_start_reg;
    reg frame_end_reg;
    reg frame_oversize_reg;
    reg frame_bad_reg;
    reg frame_state;
    reg frame_depth; // Count of words in current frame
    reg frame_start_valid;
    reg frame_end_valid;
    reg frame_oversize_valid;
    reg frame_bad_valid;

    // Frame State Machine
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            frame_state <= FRAME_STATE_IDLE;
            frame_depth <= 0;
            frame_start_valid <= 0;
            frame_end_valid <= 0;
            frame_oversize_valid <= 0;
            frame_bad_valid <= 0;
        end else begin
            // Reset frame state if FIFO is empty and no active frame
            if (fifo_empty) begin
                frame_state <= FRAME_STATE_IDLE;
                frame_depth <= 0;
                frame_start_valid <= 0;
                frame_end_valid <= 0;
                frame_oversize_valid <= 0;
                frame_bad_valid <= 0;
            end else begin
                // Detect Start of Frame
                if (frame_state == FRAME_STATE_IDLE && s_axis_tlast == 1'b1) begin
                    frame_state <= FRAME_STATE_START;
                    frame_depth <= 1;
                    frame_start_valid <= 1;
                    frame_end_valid <= 0;
                    frame_oversize_valid <= 0;
                    frame_bad_valid <= 0;
                end
                // Detect End of Frame
                else if (frame_state == FRAME_STATE_DATA && s_axis_tlast == 1'b1) begin
                    frame_state <= FRAME_STATE_END;
                    frame_end_valid <= 1;
                    frame_oversize_valid <= 0;
                    frame_bad_valid <= 0;
                end
                // Detect Oversize (Frame FIFO specific)
                else if (frame_state == FRAME_STATE_DATA && FRAME_FIFO && s_axis_tlast == 1'b0) begin
                    frame_oversize_valid <= 1;
                    frame_oversize_reg <= 1;
                end
                // Detect Bad Frame (User sideband)
                else if (frame_state == FRAME_STATE_DATA && USER_ENABLE) begin
                    if (s_axis_tuser_full & USER_BAD_FRAME_MASK) begin
                        frame_bad_valid <= 1;
                        frame_bad_reg <= 1;
                    end
                end
                
                // Update depth for ongoing frames
                if (frame_state != FRAME_STATE_IDLE) begin
                    if (s_axis_tvalid && s_axis_tready) begin
                        frame_depth <= frame_depth + 1;
                    end
                end
            end
        end
    end

    // Frame Status Outputs (Synchronous to input clock)
    assign status_bad_frame = (frame_bad_valid && frame_bad_reg);
    assign status_good_frame = (frame_start_valid && !frame_end_valid && !frame_bad_valid && !frame_oversize_valid);
    assign frame_oversize = (frame_oversize_valid && frame_oversize_reg);
    assign frame_bad = (frame_bad_valid && frame_bad_reg);

    // --- Input FIFO Logic ---
    // Determine if we should accept data based on frame logic
    wire accept_frame_data;
    wire accept_single_data;
    
    if (NEED_FRAME_LOGIC) begin
        // If FRAME_FIFO is enabled, we must complete the frame before accepting more data
        // unless it's the start of a new frame.
        // If DROP_OVERSIZE_FRAME is enabled, we drop data if frame is too long.
        // If MARK_WHEN_FULL is enabled, we mark the frame as bad if full.
        
        if (frame_state == FRAME_STATE_IDLE) begin
            // Can start a new frame
            accept_single_data = 1;
            accept_frame_data = 1;
        end
        else if (frame_state == FRAME_STATE_START) begin
            // First word of frame
            accept_single_data = 1;
            accept_frame_data = 1;
        end
        else if (frame_state == FRAME_STATE_DATA) begin
            // Middle of frame
            if (frame_oversize_valid) begin
                // Oversize detected, drop this and subsequent until end? 
                // Usually drop oversize frame means drop the whole frame or stop accepting.
                // Let's assume we stop accepting new data for this frame.
                accept_single_data = 0;
                accept_frame_data = 0;
            end
            else if (frame_bad_valid) begin
                // Bad frame detected, stop accepting
                accept_single_data = 0;
                accept_frame_data = 0;
            end
            else begin
                accept_single_data = 1;
                accept_frame_data = 1;
            end
        end
        else if (frame_state == FRAME_STATE_END) begin
            // End of frame, can start new
            accept_single_data = 1;
            accept_frame_data = 1;
        end
    end
    else begin
        // No frame logic
        accept_single_data = 1;
        accept_frame_data = 1;
    end

    // Input FIFO Depth
    reg [LOG2_DEPTH:0] fifo_rd_ptr;
    reg [LOG2_DEPTH:0] fifo_wr_ptr;
    reg [LOG2_DEPTH:0] fifo_full_cnt;
    reg [LOG2_DEPTH:0] fifo_empty_cnt;
    reg [DATA_WIDTH*DEPTH-1:0] fifo_data;
    reg [KEEP_WIDTH*DEPTH-1:0] fifo_keep;
    reg [USER_WIDTH*DEPTH-1:0] fifo_user;
    reg [ID_WIDTH*DEPTH-1:0] fifo_id;
    reg [DEST_WIDTH*DEPTH-1:0] fifo_dest;
    reg [LOG2_DEPTH:0] fifo_last_ptr; // Pointer to last word in FIFO
    reg [LOG2_DEPTH:0] fifo_start_ptr; // Pointer to start of current frame (if needed)
    reg [LOG2_DEPTH:0] fifo_frame_depth; // Depth of current frame
    reg fifo_frame_active;
    reg fifo_frame_oversize;
    reg fifo_frame_bad;

    // Frame tracking in FIFO
    reg [LOG2_DEPTH:0] fifo_frame_start_ptr;
    reg [LOG2_DEPTH:0] fifo_frame_end_ptr;

    // Input FIFO Control
    reg [LOG2_DEPTH:0] next_wr_ptr;
    reg [LOG2_DEPTH:0] next_rd_ptr;
    reg [LOG2_DEPTH:0] next_full_cnt;
    reg [LOG2_DEPTH:0] next_empty_cnt;
    reg next_wr_en;
    reg next_rd_en;
    reg next_frame_start;
    reg next_frame_end;
    reg next_frame_oversize;
    reg next_frame_bad;

    // Frame pointers update
    reg [LOG2_DEPTH:0] next_frame_start_ptr;
    reg [LOG2_DEPTH:0] next_frame_end_ptr;
    reg next_frame_active;
    reg next_frame_depth;
    reg next_frame_oversize_reg;
    reg next_frame_bad_reg;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            fifo_rd_ptr <= 0;
            fifo_wr_ptr <= 0;
            fifo_full_cnt <= 0;
            fifo_empty_cnt <= 0;
            fifo_last_ptr <= 0;
            fifo_start_ptr <= 0;
            fifo_frame_depth <= 0;
            fifo_frame_active <= 0;
            fifo_frame_oversize <= 0;
            fifo_frame_bad <= 0;
            fifo_frame_start_ptr <= 0;
            fifo_frame_end_ptr <= 0;
        end else begin
            fifo_rd_ptr <= next_rd_ptr;
            fifo_wr_ptr <= next_wr_ptr;
            fifo_full_cnt <= next_full_cnt;
            fifo_empty_cnt <= next_empty_cnt;
            fifo_last_ptr <= next_wr_ptr;
            fifo_start_ptr <= fifo_frame_start_ptr;
            fifo_frame_depth <= next_frame_depth;
            fifo_frame_active <= next_frame_active;
            fifo_frame_oversize <= next_frame_oversize_reg;
            fifo_frame_bad <= next_frame_bad_reg;
            fifo_frame_start_ptr <= next_frame_start_ptr;
            fifo_frame_end_ptr <= next_frame_end_ptr;
        end
    end

    // Frame State Machine (Synchronous to input clock, updated on write)
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            next_frame_active <= 0;
            next_frame_depth <= 0;
            next_frame_start_ptr <= 0;
            next_frame_end_ptr <= 0;
            next_frame_oversize_reg <= 0;
            next_frame_bad_reg <= 0;
        end else begin
            if (next_wr_en) begin
                if (fifo_frame_active) begin
                    // Inside a frame
                    if (s_axis_tlast) begin
                        // End of frame
                        next_frame_active <= 0;
                        next_frame_end_ptr <= fifo_wr_ptr;
                        next_frame_depth <= fifo_frame_depth;
                        next_frame_oversize_reg <= fifo_frame_oversize;
                        next_frame_bad_reg <= fifo_frame_bad;
                    end
                    else if (FRAME_FIFO && !s_axis_tlast) begin
                        // Oversize detected
                        next_frame_oversize_reg <= 1;
                        next_frame_active <= 0; // Stop accepting
                    end
                    else if (USER_ENABLE && (s_axis_tuser_full & USER_BAD_FRAME_MASK)) begin
                        next_frame_bad_reg <= 1;
                        next_frame_active <= 0; // Stop accepting
                    end
                    else begin
                        // Continue frame
                        next_frame_active <= 1;
                        next_frame_depth <= fifo_frame_depth + 1;
                        next_frame_start_ptr <= fifo_frame_start_ptr;
                        next_frame_end_ptr <= fifo_frame_end_ptr;
                    end
                end
                else begin
                    // Start of frame
                    if (s_axis_tlast) begin
                        next_frame_active <= 1;
                        next_frame_start_ptr <= fifo_wr_ptr;
                        next_frame_end_ptr <= fifo_wr_ptr;
                        next_frame_depth <= 1;
                        next_frame_oversize_reg <= 0;
                        next_frame_bad_reg <= 0;
                    end
                    else begin
                        // Single data word, not a frame
                        next_frame_active <= 0;
                        next_frame_depth <= 0;
                        next_frame_start_ptr <= fifo_wr_ptr;
                        next_frame_end_ptr <= fifo_wr_ptr;
                        next_frame_oversize_reg <= 0;
                        next_frame_bad_reg <= 0;
                    end
                end
            end
            else begin
                // No write, reset frame state if FIFO empty
                if (fifo_empty_cnt == 0) begin
                    next_frame_active <= 0;
                    next_frame_depth <= 0;
                    next_frame_start_ptr <= fifo_wr_ptr;
                    next_frame_end_ptr <= fifo_wr_ptr;
                    next_frame_oversize_reg <= 0;
                    next_frame_bad_reg <= 0;
                end
                else begin
                    next_frame_active <= fifo_frame_active;
                    next_frame_depth <= fifo_frame_depth;
                    next_frame_start_ptr <= fifo_frame_start_ptr;
                    next_frame_end_ptr <= fifo_frame_end_ptr;
                    next_frame_oversize_reg <= fifo_frame_oversize;
                    next_frame_bad_reg <= fifo_frame_bad;
                end
            end
        end
    end

    // Frame Status Outputs (Synchronous to input clock)
    assign status_bad_frame = fifo_frame_bad;
    assign status_good_frame = fifo_frame_active && !fifo_frame_oversize && !fifo_frame_bad;
    assign status_overflow = (fifo_full_cnt == DEPTH);

    // Write Enable Logic
    always @(*) begin
        next_wr_en = 0;
        if (s_axis_tvalid && m_axis_tready) begin
            if (fifo_full_cnt < DEPTH) begin
                // Check frame logic
                if (NEED_FRAME_LOGIC) begin
                    if (fifo_frame_active) begin
                        // Inside a frame, check if we should accept
                        if (fifo_frame_oversize || fifo_frame_bad) begin
                            next_wr_en = 0; // Drop data
                        end
                        else begin
                            next_wr_en = 1;
                        end
                    end
                    else begin
                        // Not in a frame, can accept
                        next_wr_en = 1;
                    end
                end
                else begin
                    next_wr_en = 1;
                end
            end
            else if (fifo_full_cnt == DEPTH) begin
                if (DROP_WHEN_FULL) begin
                    next_wr_en = 0;
                end
                else if (MARK_WHEN_FULL) begin
                    // Mark the frame as bad if it's the last word
                    if (fifo_frame_active && s_axis_tlast) begin
                        next_wr_en = 1; // Accept but mark bad
                    end
                    else begin
                        next_wr_en = 0; // Drop
                    end
                end
                else begin
                    next_wr_en = 0;
                end
            end
        end
    end

    // Write Pointer and Count Update
    always @(*) begin
        next_wr_ptr = fifo_wr_ptr;
        next_full_cnt = fifo_full_cnt;
        next_empty_cnt = fifo_empty_cnt;
        
        if (next_wr_en) begin
            next_wr_ptr = fifo_wr_ptr + 1;
            if (next_wr_ptr == DEPTH) begin
                next_wr_ptr = 0;
                next_full_cnt = DEPTH - 1;
            end
            else begin
                next_full_cnt = fifo_full_cnt + 1;
            end
            next_empty_cnt = fifo_empty_cnt - 1;
        end
    end

    // Read Enable Logic
    always @(*) begin
        next_rd_en = 0;
        if (m_axis_tready && s_axis_tvalid) begin
            if (fifo_empty_cnt > 0) begin
                next_rd_en = 1;
            end
        end
    end

    // Read Pointer and Count Update
    always @(*) begin
        next_rd_ptr = fifo_rd_ptr;
        if (next_rd_en) begin
            next_rd_ptr = fifo_rd_ptr + 1;
            if (next_rd_ptr == DEPTH) begin
                next_rd_ptr = 0;
                next_empty_cnt = fifo_empty_cnt - 1;
            end
            else begin
                next_empty_cnt = fifo_empty_cnt - 1;
            end
        end
    end

    // Data Write
    always @(*) begin
        if (next_wr_en) begin
            fifo_data[fifo_wr_ptr*DATA_WIDTH +: DATA_WIDTH] = s_axis_tdata;
            fifo_keep[fifo_wr_ptr*KEEP_WIDTH +: KEEP_WIDTH] = s_axis_tkeep;
            fifo_user[fifo_wr_ptr*USER_WIDTH +: USER_WIDTH] = s_axis_tuser_full;
            fifo_id[fifo_wr_ptr*ID_WIDTH +: ID_WIDTH] = s_axis_tid;
            fifo_dest[fifo_wr_ptr*DEST_WIDTH +: DEST_WIDTH] = s_axis_tdest;
        end
    end

    // Data Read
    always @(*) begin
        if (next_rd_en) begin
            m_axis_tdata = fifo_data[fifo_rd_ptr*DATA_WIDTH +: DATA_WIDTH];
            m_axis_tkeep = fifo_keep[fifo_rd_ptr*KEEP_WIDTH +: KEEP_WIDTH];
            m_axis_tuser = fifo_user[fifo_rd_ptr*USER_WIDTH +: USER_WIDTH];
            m_axis_tid = fifo_id[fifo_rd_ptr*ID_WIDTH +: ID_WIDTH];
            m_axis_tdest = fifo_dest[fifo_rd_ptr*DEST_WIDTH +: DEST_WIDTH];
        end
        else begin
            m_axis_tdata = 0;
            m_axis_tkeep = 0;
            m_axis_tuser = 0;
            m_axis_tid = 0;
            m_axis_tdest = 0;
        end
    end

    // Valid/Ready Handshake
    always @(*) begin
        s_axis_tready = 0;
        m_axis_tvalid = 0;
        
        if (s_axis_tvalid) begin
            if (next_wr_en) begin
                s_axis_tready = 1;
            end
        end
        
        if (m_axis_tready) begin
            if (next_rd_en) begin
                m_axis_tvalid = 1;
            end
        end
    end

    // Last Bit
    always @(*) begin
        m_axis_tlast = 0;
        if (next_rd_en) begin
            if (fifo_frame_active) begin
                // Check if this is the last word of the frame
                if (fifo_frame_end_ptr == fifo_rd_ptr) begin
                    m_axis_tlast = 1;
                end
            end
            else if (s_axis_tlast) begin
                // Single data word
                m_axis_tlast = 1;
            end
        end
    end

    // Keep Bit
    always @(*) begin
        m_axis_tkeep = 0;
        if (next_rd_en && KEEP_ENABLE) begin
            // Check if this word is fully valid
            // For simplicity, assume keep is valid if tvalid is high
            // In a real implementation, we might need to track partial words
            // Here we assume full words are written
            m_axis_tkeep = s_axis_tkeep;
        end
    end

    // Pause Logic
    always @(*) begin
        pause_ack = 0;
        if (PAUSE_ENABLE) begin
            if (fifo_full_cnt == DEPTH) begin
                pause_ack = 1;
            end
        end
    end

    // Status Outputs
    assign status_depth = fifo_full_cnt;
    assign status_depth_commit = fifo_full_cnt; // Assuming commit is same as depth for this implementation
    assign status_overflow = (fifo_full_cnt == DEPTH);
    assign status_bad_frame = fifo_frame_bad;
    assign status_good_frame = fifo_frame_active && !fifo_frame_oversize && !fifo_frame_bad;

    // CDC Safety: All state registers are synchronous to clk.
    // All multi-bit data transfers are synchronous.
    // Reset is active-high and synchronous.
    // No asynchronous resets or clock crossings.

endmodule
