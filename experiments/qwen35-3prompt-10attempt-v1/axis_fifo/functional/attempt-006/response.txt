    output wire [ID_WIDTH-1:0]    m_axis_tid,

    localparam LOG2_DEPTH = $clog2(DEPTH);
    localparam MAX_DEPTH = (1 << LOG2_DEPTH);

    // Internal signals for pipeline stages
    wire [LOG2_DEPTH:0] wr_addr, wr_addr_next, rd_addr, rd_addr_next;
    wire [LOG2_DEPTH:0] wr_addr_commit, rd_addr_commit;

    wire [KEEP_WIDTH-1:0] tkeep_in, tkeep_out;

    wire [ID_WIDTH-1:0] tid_in, tid_out;
    wire [DEST_WIDTH-1:0] tdest_in, tdest_out;
    wire [USER_WIDTH-1:0] tuser_in, tuser_out;

    wire tuser_drop;

    wire tuser_pause_ack;

    // Stage 1: Write Address Generation & Data Capture

        if (rst) begin
            wr_addr <= 0;
            depth_reg <= 0;
        end else begin
            wr_addr <= wr_addr_next;
            depth_reg <= depth_reg + 1;
        end
    end

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            rd_addr <= 0;
            depth_commit_reg <= 0;
        end else begin
            rd_addr <= rd_addr_next;
            depth_commit_reg <= depth_commit_reg + 1;
        end
    end

    always @(*) begin
        wr_addr_next = wr_addr + 1;
        rd_addr_next = rd_addr + 1;
        wr_addr_commit = wr_addr;
        rd_addr_commit = rd_addr;

    // Input Data Capture
    assign tdata_in = s_axis_tdata;
    assign tkeep_in = s_axis_tkeep;
    assign tlast_in = s_axis_tlast;
    assign tid_in = s_axis_tid;
    assign tdest_in = s_axis_tdest;
    assign tuser_in = s_axis_tuser;
    assign tvalid_in = s_axis_tvalid;

    assign m_axis_tkeep = tkeep_out;

    assign m_axis_tid = tid_out;
    assign m_axis_tdest = tdest_out;
    assign m_axis_tuser = tuser_out;

    assign status_depth_commit = depth_commit_reg;

    // 2. If FRAME_FIFO is enabled, we track if we saw a tlast before the current one in the same burst.
    //    Since this is a single clock FIFO, "oversize" usually means multiple tlasts in one write burst.
    //    We will flag a frame as bad if tlast is asserted AND we are not at the start of a new frame logic.

    //    Let's implement a simple check: If tlast is high, mark the frame.
    //    If FRAME_FIFO is 1, we might need to detect if tlast was already seen in the current burst.

    //    We will set status_bad_frame if tlast is asserted but we are in a "bad" state or if specific mask logic applies.

    // We will set status_bad_frame if tlast is asserted AND (DROP_OVERSIZE_FRAME or FRAME_PAUSE) is active,
    // implying we are checking frame boundaries.

    // We need to know if the previous word in the burst had tlast.
    // Since this is a FIFO, we can't easily look back at the previous write unless we store it.

    wire prev_tlast;

    // RAM Address for tlast storage is same as data address
    // We use a separate RAM or reuse the data RAM? 
    // To keep it simple and single source, we assume the data RAM holds the payload.
    // We can store tlast in the same RAM if we align, but tlast is 1 bit.

    // but to avoid complexity, we will assume standard AXI: tlast is only valid on the last beat.
    // If FRAME_FIFO is enabled, we assume the user expects us to detect if tlast is asserted prematurely.
    // We will implement a simple check: if tlast is asserted, we accept it.
    // status_bad_frame will be driven by the USER_BAD_FRAME_MASK logic if tuser indicates a bad frame.
    
    // Re-evaluating "Bad Frame":
    // Often in Vivado IP, "Bad Frame" is detected if tlast is asserted but the frame is not complete or oversized.
    // Without burst length, we assume tlast is the truth.
    // We will set status_bad_frame = 1 if tlast is asserted AND we are in a mode that checks frame integrity (FRAME_FIFO).
    // But strictly, if tlast is asserted, the frame is complete.
    // Let's assume "Bad Frame" means: tlast is asserted, but we are not supposed to accept it (e.g., oversized).
    // We will implement a simple counter for burst length if FRAME_FIFO is enabled.
    
    // Burst Length Counter
    wire [LOG2_DEPTH:0] burst_len;
    wire [LOG2_DEPTH:0] burst_len_reg;
    wire burst_len_reset;
    
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            burst_len_reg <= 0;
        end else begin
            if (tvalid_in && s_axis_tkeep[0]) begin // First beat of burst
                burst_len_reg <= 1;
            end else if (tvalid_in) begin
                burst_len_reg <= burst_len_reg + 1;
            end else begin
                burst_len_reg <= burst_len_reg;
            end
        end
    end
    
    // Detect Oversize: If tlast is asserted and burst_len > 1, it's an oversize frame (multiple tlasts).
    // Or if tlast is asserted and it's not the last beat (which is impossible if burst_len counts correctly).
    // Actually, if tlast is asserted on beat N, and burst_len is N, it's valid.
    // If tlast is asserted on beat N, and burst_len > N, it's invalid (oversize).
    // But burst_len increments on every valid beat. So if tlast is asserted, burst_len should be equal to current beat index + 1.
    // We can't easily know the current beat index without a counter.
    // Let's simplify: status_bad_frame is driven by tuser_bad_frame logic primarily.
    // We will set status_bad_frame = 1 if tlast is asserted AND FRAME_FIFO is enabled AND we detect a potential issue.
    // For this implementation, we will assume tlast is always valid if asserted, unless tuser indicates otherwise.

    wire tuser_bad = (tuser_in & USER_BAD_FRAME_MASK) == USER_BAD_FRAME_VALUE;

    // If FRAME_FIFO is enabled, we might need to check if tlast was asserted previously in the same burst.
    // We will store the previous tlast in a register.

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            prev_tlast_reg <= 0;
        end else begin
            prev_tlast_reg <= tlast_in;
        end
    end

    wire bad_frame_frame_fifo = (FRAME_FIFO && tlast_in && prev_tlast_reg);
    assign status_bad_frame = bad_frame_frame_fifo || tuser_bad;

    // Write Enable
    wire we = (tvalid_in && tready_out) && !tuser_drop;

    // Read Enable
    wire re = (m_axis_tready && m_axis_tvalid) && !tuser_drop;

    // Depth Check
    wire full = (current_depth == MAX_DEPTH);
    wire empty = (current_depth == 0);
    assign status_overflow = full;

    assign pause_ack = (can_pause && !pause_active) || (pause_active && !can_pause); // Toggle on change? No, usually active high.

    // If pause_req is high, we set pause_ack high if we can handle it.
    assign pause_ack = (pause_req && can_pause);

    // We will assume the RAM stores tdata. tlast is stored in a separate small RAM or we just use the data RAM to store tlast in the LSB if width permits?
    // No, let's assume standard practice: Data RAM for data, Sideband RAM for control.
    // To avoid multiple RAM instantiations which might be vendor specific, we will implement a simple RAM array using registers if DEPTH is small,
    // but DEPTH is a parameter.
    // We will use a generic RAM instantiation for data and a small one for tlast.
    
    // However, the prompt says "Do not include... vendor primitives".
    // So we must implement the RAM logic ourselves using registers.
    // This is feasible for a single file.
    
    // RAM Implementation
    // We need to store: tdata, tkeep, tlast, tid, tdest, tuser.
    // Let's create a struct-like approach using parallel wires.
    
    // Write Address
    wire [LOG2_DEPTH:0] wr_addr_final = (RAM_PIPELINE) ? wr_addr_next : wr_addr;
    wire [LOG2_DEPTH:0] rd_addr_final = (RAM_PIPELINE) ? rd_addr_next : rd_addr;

    // Since we can't modify tuser after reading, we need to set it before writing?
    // No, we set it when reading.
    // If we are marking, we set the LSB of tuser_out to 1 (assuming USER_WIDTH >= 1).

    if (USER_ENABLE) begin

            // Let's assume LSB for marking.

    assign m_axis_tuser = tuser_out_final;

    // Ready if not full, or if full and we can drop/mark.

    wire ready_in = (!full) || (full && (DROP_WHEN_FULL || MARK_WHEN_FULL));

    wire ready_out = (!empty) || (empty && (DROP_WHEN_FULL || MARK_WHEN_FULL));
    assign m_axis_tready = ready_out;

    assign status_good_frame = (!status_bad_frame && !drop_frame);

    // No, if we drop on write, we don't store it.
