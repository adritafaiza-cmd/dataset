module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

    // Synchronize the asynchronous reset signal to the clk_i domain
    // Using a 2-stage synchronizer to prevent metastability
    logic rst_n_sync_1;
    logic rst_n_sync_2;

    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            rst_n_sync_1 <= 1'b1;
        end else begin
            rst_n_sync_1 <= rst_n_sync_1;
        end
    end

    always_ff @(posedge clk_i or negedge rst_n_sync_1) begin
        if (!rst_n_sync_1) begin
            rst_n_sync_2 <= 1'b1;
        end else begin
            rst_n_sync_2 <= rst_n_sync_2;
        end
    end

    // Generate the synchronized reset
    // If test_mode_i is high, bypass the synchronizer and use the raw input directly
    // This allows for faster reset release during testing if needed, though care must be taken
    assign rst_no = test_mode_i ? rst_ni : rst_n_sync_2;

    // Generate the initialization indication
    // The module is considered initialized when the synchronized reset is released
    // and the system has been stable for at least one clock cycle after reset release.
    // We use the inverted synchronized reset to indicate "not reset".
    // init_no is active low, meaning it is 0 when initialized.
    // It stays high (1) during reset and for one cycle after reset release to ensure stability.
    // Logic: If rst_n_sync_2 is high (reset released), we wait one cycle before asserting init_no.
    // However, a simpler and more robust interpretation for "initialization indication" 
    // in this context is often: "Are we currently in the reset state or the first cycle after reset?"
    // Let's define init_no as active-low. It should be 0 when the system is fully initialized.
    // It should be 1 during reset and potentially for one cycle after reset to allow state machines to settle.
    
    // Standard approach: 
    // 1. During reset (rst_n_sync_2 == 0), init_no = 1.
    // 2. One cycle after reset releases, init_no = 1 (to allow logic to settle).
    // 3. From the second cycle after reset releases, init_no = 0.
    
    logic rst_n_sync_3;
    
    always_ff @(posedge clk_i or negedge rst_n_sync_2) begin
        if (!rst_n_sync_2) begin
            rst_n_sync_3 <= 1'b1;
        end else begin
            rst_n_sync_3 <= rst_n_sync_3;
        end
    end

    // init_no is active low. 
    // It is 1 (inactive) during reset and for one cycle after reset release.
    // It becomes 0 (active) once the system has settled for two cycles after reset release.
    assign init_no = (rst_n_sync_2 || rst_n_sync_3) ? 1'b1 : 1'b0;

endmodule
