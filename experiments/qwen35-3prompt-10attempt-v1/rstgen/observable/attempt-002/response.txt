module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

    // Synchronize the asynchronous reset signal to the clk_i domain
    // Using a 2-stage synchronizer to prevent metastability
    logic rst_n_sync_1;
    logic rst_n_sync_2;

    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            rst_n_sync_1 <= 1'b1;
        end else begin
            rst_n_sync_1 <= rst_n_sync_1;
        end
    end

    always_ff @(posedge clk_i or negedge rst_n_sync_1) begin
        if (!rst_n_sync_1) begin
            rst_n_sync_2 <= 1'b1;
        end else begin
            rst_n_sync_2 <= rst_n_sync_2;
        end
    end

    // Generate the synchronized reset
    // If test_mode_i is high, bypass synchronization and pass rst_ni directly
    // Otherwise, use the synchronized version
    assign rst_no = test_mode_i ? rst_ni : rst_n_sync_2;

    // Generate initialization indication
    // init_no is active low when the system is in reset or transitioning out of reset
    // It stays low until the reset is fully released and stable
    assign init_no = rst_no;

endmodule
