module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

    // Synchronized reset signal (active low)
    logic rst_n_sync;
    // Internal flag to track reset release state
    logic rst_n_released;
    // Internal flag to track initialization phase
    logic init_done;

    // Synchronize the active-low reset input
    // Using a 2-stage synchronizer to ensure metastability is resolved
    // before generating the synchronized reset output.
    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            rst_n_sync <= 1'b1;
        end else begin
            rst_n_sync <= rst_n_sync;
        end
    end

    // Generate synchronized reset output
    // In test mode, the output follows the input directly (bypass)
    // In normal mode, the output is the synchronized signal
    assign rst_no = test_mode_i ? rst_ni : rst_n_sync;

    // Logic to generate init_no
    // init_no is active low during reset and for a short period after reset release
    // It indicates the system is still in the initialization phase.
    
    // State machine to track reset release and initialization duration
    // We use a counter or a delay mechanism. Since no specific width is given,
    // we use a simple delay based on clock cycles.
    // Let's define a constant for the initialization duration (e.g., 10 cycles)
    localparam int INIT_DURATION = 10;
    
    logic [3:0] init_counter;
    logic init_counter_en;

    // Counter to measure time after reset release
    always_ff @(posedge clk_i or negedge rst_n_sync) begin
        if (!rst_n_sync) begin
            init_counter <= 4'd0;
            init_counter_en <= 1'b0;
        end else begin
            init_counter_en <= 1'b1;
            if (init_counter_en) begin
                if (init_counter < INIT_DURATION) begin
                    init_counter <= init_counter + 1'b1;
                end else begin
                    init_counter <= 4'd0;
                    init_counter_en <= 1'b0;
                end
            end
        end
    end

    // Generate init_no
    // Active low when reset is active OR when initialization is in progress
    assign init_no = rst_n_sync || (init_counter_en && init_counter > 4'd0);

endmodule
