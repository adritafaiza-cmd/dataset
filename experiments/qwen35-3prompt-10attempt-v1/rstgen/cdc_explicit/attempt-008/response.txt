module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

    // Synchronized reset output
    logic rst_n_sync;

    // Initialization qualifier output
    logic init_n_sync;

    // Internal state for initialization sequence
    logic [1:0] init_state;

    // State machine for initialization sequence
    always_ff @(posedge clk_i or negedge rst_n_sync) begin
        if (!rst_n_sync) begin
            init_state <= 2'b00;
        end else begin
            case (init_state)
                2'b00: begin
                    if (test_mode_i)
                        init_state <= 2'b01;
                    else
                        init_state <= 2'b00;
                end
                2'b01: begin
                    if (test_mode_i)
                        init_state <= 2'b01;
                    else
                        init_state <= 2'b10;
                end
                2'b10: begin
                    if (test_mode_i)
                        init_state <= 2'b01;
                    else
                        init_state <= 2'b00;
                end
                default:
                    init_state <= 2'b00;
            endcase
        end
    end

    // Synchronize the active-low reset
    always_ff @(posedge clk_i or negedge rst_n_sync) begin
        if (!rst_n_sync)
            rst_n_sync <= 1'b0;
        else
            rst_n_sync <= rst_ni;
    end

    // Generate synchronized reset output
    assign rst_no = rst_n_sync;

    // Generate initialization qualifier
    assign init_no = (init_state == 2'b00) ? 1'b1 : 1'b0;

endmodule<|im_end|>
