module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

    // Synchronized reset output (active-low)
    // In test_mode, the raw reset is passed through directly to allow scan/test access.
    // Otherwise, a 2-stage synchronizer ensures the reset is safe for CDC/RDC analysis.
    logic rst_synced;

    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            rst_synced <= 1'b1;
        else if (test_mode_i)
            rst_synced <= rst_ni;
        else
            rst_synced <= rst_synced;
    end

    assign rst_no = rst_synced;

    // Initialization qualifier (active-low)
    // Asserted briefly after the synchronized reset is released.
    // This provides a clean window for logic initialization without violating CDC rules.
    logic init_synced;
    logic init_cnt;

    // Two-stage synchronizer for the reset release event to ensure CDC safety
    // before generating the init pulse.
    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            init_synced <= 1'b1;
        else if (test_mode_i)
            init_synced <= rst_ni;
        else
            init_synced <= init_synced;
    end

    // Generate a short pulse when the synchronized reset goes low (active-low logic)
    // Since rst_no is active-low, we look for the transition from 1 to 0.
    // We use a counter that resets on the active reset and counts up after release.
    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            init_cnt <= 0;
        else if (test_mode_i)
            init_cnt <= 0;
        else
            init_cnt <= init_cnt + 1'b1;
    end

    // init_no is asserted (low) for a short duration after reset release.
    // The pulse width is determined by the counter.
    // Assuming a 4-cycle pulse for initialization.
    assign init_no = (init_cnt < 4) ? 1'b0 : 1'b1;

endmodule<|im_end|>
