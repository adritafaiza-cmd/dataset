module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

    // Synchronization logic for the reset signal
    // When test_mode_i is high, the reset is passed through directly.
    // Otherwise, it is synchronized using a 2-stage synchronizer.
    logic rst_sync_1;
    logic rst_sync_2;

    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            rst_sync_1 <= 1'b0;
            rst_sync_2 <= 1'b0;
        end else begin
            rst_sync_1 <= rst_ni;
            rst_sync_2 <= rst_sync_1;
        end
    end

    // Generate the final synchronized reset output
    // In test mode, bypass synchronization to avoid metastability during scan
    assign rst_no = test_mode_i ? rst_ni : rst_sync_2;

    // Generate the initialization qualifier
    // This signal is active low and indicates the system is in the initial state
    // immediately after reset release. It stays low for one clock cycle after
    // rst_no goes high, then returns high.
    logic rst_no_sync_1;
    logic rst_no_sync_2;

    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            rst_no_sync_1 <= 1'b0;
            rst_no_sync_2 <= 1'b0;
        end else begin
            rst_no_sync_1 <= rst_no;
            rst_no_sync_2 <= rst_no_sync_1;
        end
    end

    // init_no is active low. It is asserted (0) when rst_no is high (reset released)
    // and de-asserted (1) after one clock cycle.
    // Logic: if rst_no is 1, init_no becomes 0. Else, init_no follows rst_no_sync_2.
    // However, the requirement is a "short initialization qualifier".
    // Standard behavior: init_no = 0 while rst_no == 1, then init_no = 1.
    assign init_no = rst_no ? 1'b0 : rst_no_sync_2;

endmodule<|im_end|>
