module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

    logic rst_sync;
    logic rst_sync_n;
    logic init_sync;

    // Synchronize the active-low reset input
    always_ff @(posedge clk_i) begin
        if (!test_mode_i) begin
            rst_sync <= rst_ni;
        end else begin
            rst_sync <= 1'b1;
        end
    end

    // Generate active-low synchronized reset
    assign rst_no = rst_sync;

    // Generate initialization qualifier (active-low)
    // Asserted for 2 clock cycles after reset release
    always_ff @(posedge clk_i) begin
        if (!test_mode_i) begin
            if (rst_sync) begin
                init_sync <= 1'b1;
            end else if (init_sync) begin
                init_sync <= 1'b0;
            end else begin
                init_sync <= 1'b0;
            end
        end else begin
            init_sync <= 1'b0;
        end
    end

    assign init_no = init_sync;

endmodule<|im_end|>
