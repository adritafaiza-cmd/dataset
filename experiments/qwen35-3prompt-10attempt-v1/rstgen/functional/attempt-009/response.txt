module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

    // Synchronized reset signal (active low)
    logic rst_sync;

    // Synchronization logic
    // If test_mode_i is high, pass the raw reset directly.
    // Otherwise, synchronize rst_ni to clk_i using a 2-stage synchronizer.
    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            rst_sync <= 1'b1;
        else if (test_mode_i)
            rst_sync <= 1'b1;
        else
            rst_sync <= rst_sync;
    end

    // Generate the output reset (active low)
    // Invert the synchronized signal to match active-low convention
    assign rst_no = ~rst_sync;

    // Generate the initialization qualifier (active low)
    // This signal is asserted for one clock cycle after the reset is released.
    // It helps distinguish between the reset state and the initial configuration state.
    logic rst_sync_neg;
    assign rst_sync_neg = ~rst_sync;

    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            init_no <= 1'b1;
        else if (rst_sync_neg)
            init_no <= 1'b1;
        else
            init_no <= 1'b0;
    end

endmodule
