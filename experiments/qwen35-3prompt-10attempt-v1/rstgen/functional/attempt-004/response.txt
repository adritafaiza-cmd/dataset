module rstgen (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic test_mode_i,
    output logic rst_no,
    output logic init_no
);

    logic rst_sync;
    logic rst_sync_n;
    logic init_pulse;

    // Synchronize the active-low reset
    always_ff @(posedge clk_i) begin
        if (test_mode_i) begin
            // Bypass synchronization for scan/test mode
            rst_sync_n <= rst_ni;
        end else begin
            rst_sync_n <= rst_ni;
        end
    end

    // Generate active-low synchronized reset
    assign rst_no = rst_sync_n;

    // Generate short initialization pulse after reset release
    always_ff @(posedge clk_i) begin
        if (test_mode_i) begin
            // In test mode, init pulse is disabled or held low
            init_pulse <= 1'b0;
        end else begin
            if (rst_sync_n) begin
                // Reset has been released
                init_pulse <= 1'b1;
            end else begin
                // Reset is active, hold init pulse low
                init_pulse <= 1'b0;
            end
        end
    end

    // Output active-low init qualifier (low during init pulse)
    assign init_no = ~init_pulse;

endmodule<|im_end|>
