module data_sync #(
    parameter STAGES = 2,
    parameter DWIDTH = 8
)(
    input                  clk,
    input                  rstn,
    input  [DWIDTH-1:0]    din,
    input                  dready_i,
    output reg [DWIDTH-1:0] dout,
    output reg             dready_o
);

    // Internal state registers for the FIFO pipeline
    // Using a simple shift-register style FIFO to ensure coherent multi-bit data
    reg [DWIDTH-1:0] fifo_data;
    reg [STAGES-1:0] fifo_valid;

    // Logic to handle the handshake and data capture
    // We use a counter to track how many stages are filled
    reg [STAGES-1:0] stage_count;

    // Always block for sequential logic
    always @(posedge clk or negedge rstn) begin
        if (!rstn) begin
            dout      <= 0;
            dready_o  <= 0;
            fifo_data <= 0;
            fifo_valid <= 0;
            stage_count <= 0;
        end else begin
            // Check if we can accept new data (not all stages full)
            if (stage_count < STAGES) begin
                // Capture input data
                fifo_data <= din;
                
                // Update valid mask: set the bit corresponding to the current stage
                // We shift the valid mask left as we fill stages
                fifo_valid <= {fifo_valid[STAGES-2:0], 1'b0};
                
                // Increment stage count
                stage_count <= stage_count + 1'b1;
            end
            
            // Output logic:
            // dout is valid only if the first stage is full (stage_count > 0)
            // We output the data from the oldest stage (fifo_data[0] conceptually)
            // However, since we are shifting data in a pipeline, we need to manage the shift carefully.
            // Alternative approach: Use a circular buffer or shift the data array.
            // Given STAGES is small, let's use a shift register approach for data and valid.
            
            // Revised approach for coherent multi-bit output:
            // We will maintain an array of data and a valid mask.
            // On write: shift data left, set LSB to new din, set LSB of valid mask to 1.
            // On read: shift data right, shift valid mask right, output MSB of data if valid MSB is 1.
            
            // Let's restructure the internal logic for clarity and correctness.
            // We need to shift data out when dready_i is high and we have data.
            
            // Actually, the standard way for a synchronizer with handshake is:
            // 1. When dready_i is high, we want to push data into the sync chain.
            // 2. The sync chain shifts data towards the output.
            // 3. When the chain is full, we stop accepting.
            
            // Let's implement a shift-register based FIFO.
            // fifo_data[i] holds the data for stage i.
            // fifo_valid[i] indicates if stage i has valid data.
            
            // Write path (triggered by dready_i being high and space available):
            // We shift existing data to the next stage and insert new data at stage 0.
            // But we must ensure we don't overwrite valid data if the consumer hasn't read it.
            // This requires a more complex state machine or a simple counter-based shift.
            
            // Simpler robust implementation:
            // Use a counter `write_ptr` and `read_ptr`.
            // But since we need to shift data on every clock cycle to synchronize,
            // a shift register is best.
            
            // Let's go with the shift register method:
            // On every clock edge:
            // 1. If dready_i is high and we have space, shift data left, insert din at LSB, set valid LSB.
            // 2. If dready_i is low (or full), just shift data left (if valid) to propagate.
            // Wait, if dready_i is low, we shouldn't shift new data in, but we should still shift existing data out?
            // No, the standard synchronizer works by:
            // - When source says "I have data" (dready_i), we capture it.
            // - The data moves through STAGES over STAGES clock cycles.
            // - Once it reaches the last stage, we present it on dout and pulse dready_o.
            
            // Correct Logic:
            // We maintain a shift register of depth STAGES.
            // We also maintain a valid flag for each stage.
            // However, to keep it simple and efficient:
            // We only shift data when dready_i is high? No, that would stall the output.
            // We must shift data continuously to move it towards the output.
            
            // Refined Logic:
            // 1. Shift register `data_reg` of width DWIDTH * STAGES? No, that's inefficient.
            // 2. Array of registers `stage_data[STAGES]`.
            // 3. Array of valid flags `stage_valid[STAGES]`.
            // 4. Write pointer `wp` and Read pointer `rp`.
            
            // Let's use pointers for clarity.
            reg [STAGES-1:0] write_idx;
            reg [STAGES-1:0] read_idx;
            
            // Data storage
            reg [DWIDTH-1:0] stage_data [STAGES-1:0];
            reg [STAGES-1:0] stage_valid [STAGES-1:0];
            
            // Initialize
            if (!rstn) begin
                write_idx <= 0;
                read_idx  <= 0;
                for (integer i = 0; i < STAGES; i = i + 1) begin
                    stage_data[i] <= 0;
                    stage_valid[i] <= 0;
                end
            end else begin
                // Check if we can write (not full)
                if (write_idx != read_idx) begin
                    // Shift data from previous stage to current stage if it exists
                    if (write_idx > 0) begin
                        stage_data[write_idx] <= stage_data[write_idx - 1];
                        stage_valid[write_idx] <= stage_valid[write_idx - 1];
                    end
                    
                    // If dready_i is high, we can accept new data
                    if (dready_i) begin
                        stage_data[write_idx] <= din;
                        stage_valid[write_idx] <= 1'b1;
                    end
                    
                    // Advance write pointer
                    write_idx <= write_idx + 1'b1;
                end
                
                // Read logic: if read_idx < write_idx and valid, output data
                // We need to handle the case where data is being shifted out.
                // Actually, the standard way is:
                // The data moves from stage 0 -> 1 -> ... -> STAGES-1.
                // When data reaches STAGES-1 and is valid, we output it.
                
                // Let's simplify the pointer logic to a pure shift register behavior.
                // We don't need separate read/write pointers if we just shift everything.
                // But we need to know when to stop shifting new data in.
                
                // Re-evaluating the simplest robust design:
                // Use a single shift register for data and a single shift register for valid.
                // But Verilog doesn't support shifting arrays easily in always blocks without loops.
                // Given STAGES is a parameter, a loop is fine.
                
                // Let's stick to the pointer approach but ensure data flows correctly.
                // If dready_i is high, we write to `write_idx`.
                // The data at `write_idx` comes from `write_idx - 1`.
                // The data at `write_idx - 1` comes from `write_idx - 2`, etc.
                // This effectively shifts data towards higher indices.
                // When `write_idx` reaches `STAGES`, we stop writing.
                // The output is taken from `STAGES-1` if `stage_valid[STAGES-1]` is 1.
                
                // Wait, if we use pointers, we need to handle the circular nature or just let it grow?
                // Since it's a synchronizer, we don't need a circular buffer if we assume the source stops when full.
                // But the source might keep sending. So we need a circular buffer or a full check.
                // Full check: write_idx == read_idx + STAGES (modulo arithmetic).
                // To avoid modulo complexity, let's assume STAGES is small and use a simple counter.
                // Actually, the standard synchronizer just shifts data.
                // If the source keeps sending, the FIFO fills up.
                // Once full, we stop accepting.
                
                // Let's implement a circular buffer logic.
                // write_idx and read_idx increment every cycle?
                // No, we only increment write_idx if dready_i is high.
                // We always increment read_idx? No, we only read when we are ready to output.
                // But the requirement is: "pulse dready_o in clk".
                // This implies we output one bit per clock when data is ready.
                
                // Correct Flow:
                // 1. Data enters at stage 0 when dready_i is high.
                // 2. Data shifts from stage i to i+1 every clock cycle.
                // 3. When data reaches stage STAGES-1, it is valid.
                // 4. We output it and pulse dready_o.
                // 5. We must ensure we don't overwrite valid data at stage 0 before it has shifted out.
                
                // Implementation:
                // We maintain `stage_data` and `stage_valid`.
                // We maintain `write_ptr` and `read_ptr`.
                // `write_ptr` increments only when dready_i is high AND FIFO is not full.
                // `read_ptr` increments every clock cycle? No.
                // The data shifts automatically.
                // So, `stage_data[i]` gets data from `stage_data[i-1]`.
                // This means we need to shift the whole array every cycle.
                
                // Let's try a different approach:
                // Use a single `reg [DWIDTH*STAGES-1:0] shift_reg`.
                // This is inefficient for large DWIDTH but correct.
                // Or, use an array and shift it.
                
                // Let's go with the array shift approach.
                // On every clock:
                // 1. Shift data from i-1 to i for all i.
                // 2. If dready_i is high and FIFO is not full, set stage 0 to din.
                // 3. Check if stage STAGES-1 is valid. If so, output and pulse dready_o.
                
                // How to detect "not full"?
                // We need a counter of valid items.
                // Let's add a `valid_count` register.
                
                reg [STAGES-1:0] stage_valid_reg [STAGES-1:0];
                reg [DWIDTH-1:0] stage_data_reg [STAGES-1:0];
                reg [STAGES-1:0] valid_count;
                
                if (!rstn) begin
                    valid_count <= 0;
                    for (integer i = 0; i < STAGES; i = i + 1) begin
                        stage_data_reg[i] <= 0;
                        stage_valid_reg[i] <= 0;
                    end
                end else begin
                    // Shift data and validity
                    for (integer i = 1; i < STAGES; i = i + 1) begin
                        stage_data_reg[i] <= stage_data_reg[i-1];
                        stage_valid_reg[i] <= stage_valid_reg[i-1];
                    end
                    
                    // Handle stage 0
                    // If valid_count > 0, we can accept new data
                    if (valid_count > 0) begin
                        if (dready_i) begin
                            stage_data_reg[0] <= din;
                            stage_valid_reg[0] <= 1'b1;
                            // valid_count doesn't change immediately, it changes after shift
                        end
                    end
                    
                    // Update valid_count based on stage 0
                    // If stage 0 was valid before shift, it moves to stage 1.
                    // If stage 0 was not valid, it stays 0.
                    // We need to know the state of stage 0 BEFORE the shift to update count correctly?
                    // Actually, simpler:
                    // valid_count = sum(stage_valid_reg).
                    // But we can't sum in combinational logic inside sequential block easily without loops.
                    // Let's just shift and then check.
                    
                    // Better logic:
                    // 1. Shift everything.
                    // 2. If dready_i is high, set stage 0.
                    // 3. Calculate valid_count.
                    
                    // Let's refine the loop order.
                    // We need to capture the old state of stage 0 to decide if we increment count?
                    // No, if we set stage 0 to 1, count increases.
                    // If we shift stage 0 to stage 1, count stays same.
                    
                    // Let's do this:
                    // 1. Shift data from 0 to STAGES-1.
                    // 2. If dready_i is high, set stage 0.
                    // 3. Calculate valid_count.
                    
                    // Wait, if we shift first, stage 0 becomes garbage (from stage -1 which is 0).
                    // Then we set stage 0.
                    // This works.
                    
                    // But we need to know if stage 0 was valid BEFORE the shift to update the count?
                    // No, the count is the number of valid bits currently in the pipeline.
                    // If we shift, the valid bit moves from 0 to 1. Count remains same.
                    // If we insert new data at 0, count increases by 1.
                    
                    // So:
                    // 1. Shift data from 0 to STAGES-1.
                    // 2. If dready_i is high, set stage 0.
                    // 3. Calculate valid_count.
                    
                    // However, we need to ensure we don't overflow.
                    // If valid_count == STAGES, we cannot accept new data.
                    
                    // Let's implement the shift and count logic.
                    
                    // Shift data
                    for (integer i = 1; i < STAGES; i = i + 1) begin
                        stage_data_reg[i] <= stage_data_reg[i-1];
                        stage_valid_reg[i] <= stage_valid_reg[i-1];
                    end
                    
                    // Insert new data if possible
                    if (valid_count < STAGES) begin
                        if (dready_i) begin
                            stage_data_reg[0] <= din;
                            stage_valid_reg[0] <= 1'b1;
                        end
                    end
                    
                    // Calculate valid_count
                    valid_count <= 0;
                    for (integer i = 0; i < STAGES; i = i + 1) begin
                        if (stage_valid_reg[i]) begin
                            valid_count <= valid_count + 1'b1;
                        end
                    end
                    
                    // Output logic
                    // dout is valid if stage_valid_reg[STAGES-1] is 1
                    if (stage_valid_reg[STAGES-1]) begin
                        dout <= stage_data_reg[STAGES-1];
                        dready_o <= 1'b1;
                    end else begin
                        dout <= 0; // Or keep previous? Usually 0 or Z. Let's use 0.
                        dready_o <= 1'b0;
                    end
                end
            end
        end
    end

endmodule<|im_end|>
