module areset_sync #(
    parameter STAGES = 2
)(
    input  wire clk,
    input  wire async_rst_i,
    output wire sync_rst_o
);

    reg [STAGES-1:0] sync_rst_reg;

    always @(posedge clk or posedge async_rst_i) begin
        if (async_rst_i)
            sync_rst_reg <= {STAGES{1'b1}};
        else
            sync_rst_reg <= {sync_rst_reg[STAGES-2:0], 1'b0};
    end

    assign sync_rst_o = sync_rst_reg[STAGES-1];

endmodule
