module areset_sync #(
    parameter STAGES = 2
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);

    reg [STAGES-1:0] rst_pipe;

    // Synchronize the asynchronous reset release (de-assertion)
    // The reset is asserted immediately on async_rst_i.
    // The release is delayed through STAGES flip-flops to ensure
    // it is stable and synchronized before being used as a system reset.
    always @(posedge clk or posedge async_rst_i) begin
        if (async_rst_i)
            rst_pipe <= {STAGES{1'b1}};
        else
            rst_pipe <= {rst_pipe[STAGES-2:0], 1'b1};
    end

    // The output reset is the inverse of the synchronized release signal.
    // When rst_pipe is all zeros (release complete), sync_rst_o de-asserts.
    assign sync_rst_o = rst_pipe[STAGES-1];

endmodule<|im_end|>
