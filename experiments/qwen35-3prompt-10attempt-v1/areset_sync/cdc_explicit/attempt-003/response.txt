module areset_sync #(
    parameter STAGES = 2
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);

    // Synchronizer chain for reset release
    // STAGES defines the depth of the synchronizer pipeline
    // Using 2 stages is standard for robust reset release synchronization
    wire [STAGES-1:0] rst_sync_chain;

    // First stage of synchronizer
    always @(posedge clk or posedge async_rst_i) begin
        if (async_rst_i)
            rst_sync_chain[0] <= 1'b1;
        else
            rst_sync_chain[0] <= rst_sync_chain[0];
    end

    // Subsequent stages of synchronizer
    genvar i;
    generate
        for (i = 1; i < STAGES; i = i + 1) begin : sync_stage
            always @(posedge clk or posedge rst_sync_chain[i-1]) begin
                if (rst_sync_chain[i-1])
                    rst_sync_chain[i] <= 1'b1;
                else
                    rst_sync_chain[i] <= rst_sync_chain[i];
            end
        end
    endgenerate

    // Final output is the last stage of the synchronizer chain
    assign sync_rst_o = rst_sync_chain[STAGES-1];

endmodule<|im_end|>
