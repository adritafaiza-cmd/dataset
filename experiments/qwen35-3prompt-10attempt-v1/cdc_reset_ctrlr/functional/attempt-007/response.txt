module cdc_reset_ctrlr #(
    parameter SYNC_STAGES = 2
)(
    input  a_clk_i,
    input  a_rst_ni,
    input  a_clear_i,
    output a_clear_o,
    input  a_clear_ack_i,
    output a_isolate_o,
    input  a_isolate_ack_i,
    input  b_clk_i,
    input  b_rst_ni,
    input  b_clear_i,
    output b_clear_o,
    input  b_clear_ack_i,
    output b_isolate_o
);

    // ------------------------------------------------------------------
    // State Machine Definition
    // ------------------------------------------------------------------
    // States:
    // IDLE       : Normal operation, no clear/isolate active.
    // REQ        : Request received, initiating handshake.
    // ISOLATE    : Isolate signals asserted, waiting for peer ack.
    // CLEAR      : Clear signals asserted, waiting for peer ack.
    // DONE       : Handshake complete, releasing signals.
    // ------------------------------------------------------------------
    typedef enum logic [1:0] {
        IDLE,
        REQ,
        ISOLATE,
        CLEAR,
        DONE
    } state_t;

    state_t current_state, next_state;

    // ------------------------------------------------------------------
    // Synchronizers for Asynchronous Inputs
    // ------------------------------------------------------------------
    // Synchronize a_clear_i
    logic [SYNC_STAGES-1:0] a_clear_sync;
    genvar i;
    generate
        for (i = 0; i < SYNC_STAGES; i = i + 1) begin : gen_a_clear_sync
            logic sync_stage;
            if (i == 0)
                sync_stage = a_clear_i;
            else
                sync_stage = a_clear_sync[i-1];
            always_ff @(posedge a_clk_i or negedge a_rst_ni) begin
                if (!a_rst_ni)
                    sync_stage <= 1'b0;
                else
                    sync_stage <= sync_stage;
            end
        end
    endgenerate

    // Synchronize b_clear_i
    logic [SYNC_STAGES-1:0] b_clear_sync;
    generate
        for (i = 0; i < SYNC_STAGES; i = i + 1) begin : gen_b_clear_sync
            logic sync_stage;
            if (i == 0)
                sync_stage = b_clear_i;
            else
                sync_stage = b_clear_sync[i-1];
            always_ff @(posedge b_clk_i or negedge b_rst_ni) begin
                if (!b_rst_ni)
                    sync_stage <= 1'b0;
                else
                    sync_stage <= sync_stage;
            end
        end
    endgenerate

    // ------------------------------------------------------------------
    // Synchronizers for Peer Ack Signals (Input to FSM)
    // ------------------------------------------------------------------
    // Synchronize a_isolate_ack_i
    logic [SYNC_STAGES-1:0] a_isolate_ack_sync;
    generate
        for (i = 0; i < SYNC_STAGES; i = i + 1) begin : gen_a_isolate_ack_sync
            logic sync_stage;
            if (i == 0)
                sync_stage = a_isolate_ack_i;
            else
                sync_stage = a_isolate_ack_sync[i-1];
            always_ff @(posedge a_clk_i or negedge a_rst_ni) begin
                if (!a_rst_ni)
                    sync_stage <= 1'b0;
                else
                    sync_stage <= sync_stage;
            end
        end
    endgenerate

    // Synchronize b_isolate_ack_i
    logic [SYNC_STAGES-1:0] b_isolate_ack_sync;
    generate
        for (i = 0; i < SYNC_STAGES; i = i + 1) begin : gen_b_isolate_ack_sync
            logic sync_stage;
            if (i == 0)
                sync_stage = b_isolate_ack_i;
            else
                sync_stage = b_isolate_ack_sync[i-1];
            always_ff @(posedge b_clk_i or negedge b_rst_ni) begin
                if (!b_rst_ni)
                    sync_stage <= 1'b0;
                else
                    sync_stage <= sync_stage;
            end
        end
    endgenerate

    // Synchronize a_clear_ack_i
    logic [SYNC_STAGES-1:0] a_clear_ack_sync;
    generate
        for (i = 0; i < SYNC_STAGES; i = i + 1) begin : gen_a_clear_ack_sync
            logic sync_stage;
            if (i == 0)
                sync_stage = a_clear_ack_i;
            else
                sync_stage = a_clear_ack_sync[i-1];
            always_ff @(posedge a_clk_i or negedge a_rst_ni) begin
                if (!a_rst_ni)
                    sync_stage <= 1'b0;
                else
                    sync_stage <= sync_stage;
            end
        end
    endgenerate

    // Synchronize b_clear_ack_i
    logic [SYNC_STAGES-1:0] b_clear_ack_sync;
    generate
        for (i = 0; i < SYNC_STAGES; i = i + 1) begin : gen_b_clear_ack_sync
            logic sync_stage;
            if (i == 0)
                sync_stage = b_clear_ack_i;
            else
                sync_stage = b_clear_ack_sync[i-1];
            always_ff @(posedge b_clk_i or negedge b_rst_ni) begin
                if (!b_rst_ni)
                    sync_stage <= 1'b0;
                else
                    sync_stage <= sync_stage;
            end
        end
    endgenerate

    // ------------------------------------------------------------------
    // State Machine Logic
    // ------------------------------------------------------------------
    always_ff @(posedge a_clk_i or negedge a_rst_ni) begin
        if (!a_rst_ni)
            current_state <= IDLE;
        else
            current_state <= next_state;
    end

    always_comb begin
        next_state = current_state;

        case (current_state)
            IDLE: begin
                // Check for request from either side
                if (a_clear_sync[0] || b_clear_sync[0])
                    next_state = REQ;
                else
                    next_state = IDLE;
            end

            REQ: begin
                // Move to Isolate phase
                next_state = ISOLATE;
            end

            ISOLATE: begin
                // Wait for both peer isolate acks
                if (a_isolate_ack_sync[0] && b_isolate_ack_sync[0])
                    next_state = CLEAR;
                else
                    next_state = ISOLATE;
            end

            CLEAR: begin
                // Wait for both peer clear acks
                if (a_clear_ack_sync[0] && b_clear_ack_sync[0])
                    next_state = DONE;
                else
                    next_state = CLEAR;
            end

            DONE: begin
                // Release all signals, return to Idle
                next_state = IDLE;
            end

            default:
                next_state = IDLE;
        endcase
    end

    // ------------------------------------------------------------------
    // Output Logic (Driven by State Machine)
    // ------------------------------------------------------------------
    // Isolate outputs are asserted in ISOLATE state
    assign a_isolate_o = (current_state == ISOLATE);
    assign b_isolate_o = (current_state == ISOLATE);

    // Clear outputs are asserted in CLEAR state
    assign a_clear_o = (current_state == CLEAR);
    assign b_clear_o = (current_state == CLEAR);

endmodule
