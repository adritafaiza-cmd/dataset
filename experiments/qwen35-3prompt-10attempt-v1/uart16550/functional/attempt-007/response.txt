```verilog
module uart_top (
    wb_clk_i,
    wb_rst_i, wb_adr_i, wb_dat_i, wb_dat_o, wb_we_i, wb_stb_i, wb_cyc_i, wb_sel_i, wb_ack_o,
    int_o,
    stx_pad_o, srx_pad_i,
    rts_pad_o, cts_pad_i, dtr_pad_o, dsr_pad_i, ri_pad_i, dcd_pad_i
);

    // Wishbone Interface
    input wb_clk_i;
    input wb_rst_i;
    input [15:0] wb_adr_i;
    input [7:0] wb_dat_i;
    output reg [7:0] wb_dat_o;
    input wb_we_i;
    input wb_stb_i;
    input wb_cyc_i;
    input [3:0] wb_sel_i;
    output wb_ack_o;

    // Modem Pins
    input cts_pad_i;
    input dsr_pad_i;
    input ri_pad_i;
    input dcd_pad_i;
    output stx_pad_o;
    output rts_pad_o;
    output dtr_pad_o;

    // Interrupt
    output int_o;

    // Internal Signals
    reg [7:0] wb_dat_o_reg;
    reg wb_ack_o_reg;
    reg int_o_reg;
    reg stx_pad_o_reg;
    reg rts_pad_o_reg;
    reg dtr_pad_o_reg;

    // FIFO and Shift Register
    reg [7:0] rx_fifo [0:15];
    reg [3:0] rx_fifo_cnt;
    reg rx_fifo_full;
    reg rx_fifo_empty;
    reg [7:0] shift_reg;
    reg shift_reg_valid;
    reg [7:0] tx_fifo [0:15];
    reg [3:0] tx_fifo_cnt;
    reg tx_fifo_full;
    reg tx_fifo_empty;
    reg tx_shift_reg;
    reg tx_shift_reg_valid;

    // Control Registers (Standard 16550 Map)
    // 0x00: Line Control
    // 0x01: Modem Control
    // 0x02: Interrupt Enable
    // 0x03: Interrupt Identification
    // 0x04: Line Status
    // 0x05: Modem Status
    // 0x06: Receive FIFO Data Register
    // 0x07: Transmit Holding Register
    // 0x08: IER (Interrupt Enable Register) - Overlapped with 0x02 in some maps, but standard is 0x02
    // 0x09: IIR (Interrupt Identification Register) - Overlapped with 0x03
    // 0x0A: FCR (FIFO Control Register)
    // 0x0B: EFR (Extended FIFO Register) - Optional, but we support basic FIFO
    // 0x0C: MSR (Modem Status Register) - Overlapped with 0x05
    // 0x0D: LCR (Line Control Register) - Overlapped with 0x00
    // 0x0E: LSR (Line Status Register) - Overlapped with 0x04
    // 0x0F: MCR (Modem Control Register) - Overlapped with 0x01

    // We will implement the registers at their standard addresses.
    // Address 0x00: LCR
    // Address 0x01: MCR
    // Address 0x02: IER
    // Address 0x03: IIR
    // Address 0x04: LSR
    // Address 0x05: MSR
    // Address 0x06: RHR
    // Address 0x07: THR
    // Address 0x08: FCR
    // Address 0x09: EFR (Optional, set to 0)
    // Address 0x0A: MSR (Overlap)
    // Address 0x0B: LCR (Overlap)
    // Address 0x0C: LSR (Overlap)
    // Address 0x0D: MCR (Overlap)
    // Address 0x0E: IER (Overlap)
    // Address 0x0F: IIR (Overlap)

    // Baud Rate Generator
    // Assuming 16 MHz clock for 9600 baud (16x oversampling)
    // Divider = 16000000 / (16 * 9600) = 104.166 -> 104
    // We will use a configurable divider logic based on a default divisor.
    // For simplicity in this implementation without external params, we assume a fixed divisor 
    // or a simple counter that can be adjusted. 
    // Let's implement a simple baud rate generator with a fixed divisor for 9600 baud @ 16MHz.
    // Divisor = 16MHz / (16 * 9600) = 104.
    // We'll use a counter that counts to 104.
    
    reg [15:0] baud_counter;
    reg baud_counter_enable;
    reg [7:0] baud_divisor; // 104 for 9600 @ 16MHz

    // Assign default baud divisor
    assign baud_divisor = 104;

    // State Machine for UART
    // States: IDLE, START, DATA0, DATA1, DATA2, DATA3, DATA4, DATA5, DATA6, DATA7, STOP
    // Since we are doing 8N1, we need 10 states.
    typedef enum reg [2:0] {
        UART_IDLE = 3'b000,
        UART_START = 3'b001,
        UART_DATA0 = 3'b010,
        UART_DATA1 = 3'b011,
        UART_DATA2 = 3'b100,
        UART_DATA3 = 3'b101,
        UART_DATA4 = 3'b110,
        UART_DATA5 = 3'b111,
        UART_DATA6 = 3'b000, // Reuse 000? No, let's use 3 bits for 10 states.
        UART_DATA7 = 3'b100,
        UART_DATA8 = 3'b101,
        UART_STOP = 3'b110
    } uart_state_t;

    uart_state_t uart_state;
    reg [7:0] uart_rx_data;
    reg uart_rx_valid;
    reg uart_tx_data;
    reg uart_tx_valid;

    // Baud Rate Counter
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            baud_counter <= 0;
            baud_counter_enable <= 0;
        end else begin
            if (baud_counter_enable) begin
                if (baud_counter == baud_divisor) begin
                    baud_counter <= 0;
                    baud_counter_enable <= 0;
                end else begin
                    baud_counter <= baud_counter + 1;
                end
            end
        end
    end

    // UART State Machine
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            uart_state <= UART_IDLE;
            uart_rx_data <= 0;
            uart_rx_valid <= 0;
            uart_tx_data <= 0;
            uart_tx_valid <= 0;
        end else begin
            case (uart_state)
                UART_IDLE: begin
                    if (baud_counter_enable) begin
                        uart_state <= UART_START;
                    end
                end
                UART_START: begin
                    uart_state <= UART_DATA0;
                end
                UART_DATA0: begin
                    uart_state <= UART_DATA1;
                end
                UART_DATA1: begin
                    uart_state <= UART_DATA2;
                end
                UART_DATA2: begin
                    uart_state <= UART_DATA3;
                end
                UART_DATA3: begin
                    uart_state <= UART_DATA4;
                end
                UART_DATA4: begin
                    uart_state <= UART_DATA5;
                end
                UART_DATA5: begin
                    uart_state <= UART_DATA6;
                end
                UART_DATA6: begin
                    uart_state <= UART_DATA7;
                end
                UART_DATA7: begin
                    uart_state <= UART_STOP;
                end
                UART_STOP: begin
                    uart_state <= UART_IDLE;
                end
            endcase
        end
    end

    // Shift Register Logic
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            shift_reg <= 0;
            shift_reg_valid <= 0;
        end else begin
            if (baud_counter_enable) begin
                if (uart_state == UART_DATA0) begin
                    shift_reg <= {7'b0, srx_pad_i};
                end else if (uart_state == UART_DATA1) begin
                    shift_reg <= {shift_reg[6:0], srx_pad_i};
                end else if (uart_state == UART_DATA2) begin
                    shift_reg <= {shift_reg[5:0], srx_pad_i};
                end else if (uart_state == UART_DATA3) begin
                    shift_reg <= {shift_reg[4:0], srx_pad_i};
                end else if (uart_state == UART_DATA4) begin
                    shift_reg <= {shift_reg[3:0], srx_pad_i};
                end else if (uart_state == UART_DATA5) begin
                    shift_reg <= {shift_reg[2:0], srx_pad_i};
                end else if (uart_state == UART_DATA6) begin
                    shift_reg <= {shift_reg[1:0], srx_pad_i};
                end else if (uart_state == UART_DATA7) begin
                    shift_reg <= {shift_reg[0], srx_pad_i};
                end
            end
        end
    end

    // RX FIFO Logic
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            rx_fifo_cnt <= 0;
            rx_fifo_full <= 0;
            rx_fifo_empty <= 1;
        end else begin
            if (shift_reg_valid && !rx_fifo_full) begin
                rx_fifo[rx_fifo_cnt] <= shift_reg;
                rx_fifo_cnt <= rx_fifo_cnt + 1;
                shift_reg_valid <= 0;
                if (rx_fifo_cnt == 15) begin
                    rx_fifo_full <= 1;
                end
            end
            if (rx_fifo_cnt == 0) begin
                rx_fifo_empty <= 1;
            end
        end
    end

    // TX FIFO Logic
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            tx_fifo_cnt <= 0;
            tx_fifo_full <= 0;
            tx_fifo_empty <= 1;
        end else begin
            if (wb_we_i && wb_stb_i && wb_cyc_i && wb_sel_i[3] && !tx_fifo_full) begin
                tx_fifo[tx_fifo_cnt] <= wb_dat_i;
                tx_fifo_cnt <= tx_fifo_cnt + 1;
                if (tx_fifo_cnt == 15) begin
                    tx_fifo_full <= 1;
                end
            end
            if (tx_fifo_cnt == 0) begin
                tx_fifo_empty <= 1;
            end
        end
    end

    // TX Shift Register Logic
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            tx_shift_reg <= 0;
            tx_shift_reg_valid <= 0;
        end else begin
            if (baud_counter_enable) begin
                if (uart_state == UART_DATA0) begin
                    tx_shift_reg <= tx_fifo[0];
                    tx_shift_reg_valid <= 1;
                end else if (uart_state == UART_DATA1) begin
                    tx_shift_reg <= {tx_shift_reg[6:0], tx_fifo[0][0]};
                end else if (uart_state == UART_DATA2) begin
                    tx_shift_reg <= {tx_shift_reg[5:0], tx_fifo[0][1]};
                end else if (uart_state == UART_DATA3) begin
                    tx_shift_reg <= {tx_shift_reg[4:0], tx_fifo[0][2]};
                end else if (uart_state == UART_DATA4) begin
                    tx_shift_reg <= {tx_shift_reg[3:0], tx_fifo[0][3]};
                end else if (uart_state == UART_DATA5) begin
                    tx_shift_reg <= {tx_shift_reg[2:0], tx_fifo[0][4]};
                end else if (uart_state == UART_DATA6) begin
                    tx_shift_reg <= {tx_shift_reg[1:0], tx_fifo[0][5]};
                end else if (uart_state == UART_DATA7) begin
                    tx_shift_reg <= {tx_shift_reg[0], tx_fifo[0][6]};
                end
            end
        end
    end

    // Output Data for Wishbone
    assign wb_dat_o = shift_reg_valid ? shift_reg : 8'b0;

    // Acknowledge
    assign wb_ack_o = wb_stb_i && wb_cyc_i && (wb_sel_i[3] || wb_sel_i[2] || wb_sel_i[1] || wb_sel_i[0]);

    // Interrupt Logic
    // INT = (IER & IIR)
    // IER: Bit 0 = RX Enable, Bit 1 = TX Enable
    // IIR: Bit 0 = Interrupt Pending (0 = No Interrupt, 1 = Interrupt Pending)
    // We will generate an interrupt if:
    // 1. RX FIFO is not empty and RX Enable is set
    // 2. TX FIFO is not full and TX Enable is set
    // 3. TX Shift Register is valid and TX Enable is set (for immediate TX)
    
    reg rx_enable;
    reg tx_enable;
    reg rx_interrupt;
    reg tx_interrupt;

    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            rx_enable <= 0;
            tx_enable <= 0;
            rx_interrupt <= 0;
            tx_interrupt <= 0;
        end else begin
            // Read LCR (0x00) to get DLAB (Bit 7) - ignored here as we use fixed baud
            // Read MCR (0x01) to get RTS, DTR, etc.
            // Read IER (0x02)
            if (wb_we_i && wb_stb_i && wb_cyc_i && wb_sel_i[1]) begin
                rx_enable <= wb_dat_i[0];
                tx_enable <= wb_dat_i[1];
            end

            // Check RX Interrupt
            if (rx_enable && !rx_fifo_empty) begin
                rx_interrupt <= 1;
            end else begin
                rx_interrupt <= 0;
            end

            // Check TX Interrupt
            if (tx_enable) begin
                if (tx_fifo_empty && !tx_shift_reg_valid) begin
                    tx_interrupt <= 0;
                end else if (tx_fifo_empty && tx_shift_reg_valid) begin
                    tx_interrupt <= 1;
                end else if (!tx_fifo_empty) begin
                    tx_interrupt <= 1;
                end else begin
                    tx_interrupt <= 0;
                end
            end else begin
                tx_interrupt <= 0;
            end
        end
    end

    assign int_o = rx_interrupt || tx_interrupt;

    // Modem Control Pins
    // RTS (Bit 1 of MCR), DTR (Bit 0 of MCR), OUT1 (Bit 2), OUT2 (Bit 3)
    // We will map MCR bits to outputs.
    // MCR: 0x01
    reg [7:0] mcr_reg;

    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            mcr_reg <= 0;
            rts_pad_o_reg <= 1;
            dtr_pad_o_reg <= 1;
        end else begin
            if (wb_we_i && wb_stb_i && wb_cyc_i && wb_sel_i[0]) begin
                mcr_reg <= wb_dat_i;
            end
            // RTS is inverted in 16550 (Active Low)
            assign rts_pad_o_reg = ~mcr_reg[1];
            // DTR is active high
            assign dtr_pad_o_reg = mcr_reg[0];
            // OUT1 and OUT2 are not connected to anything specific, default to 0
            // stx_pad_o is active high
            assign stx_pad_o_reg = tx_shift_reg_valid;
        end
    end

    // Modem Status Pins
    // CTS, DSR, RI, DCD are inputs
    // We need to generate MSR (0x05) and LSR (0x04)
    // MSR: Bit 0 = CTS, Bit 1 = DSR, Bit 2 = RI, Bit 3 = DCD
    // LSR: Bit 0 = RX Overrun, Bit 1 = RX Parity Error, Bit 2 = RX Framing Error, Bit 3 = RX Break, Bit 4 = RX Data Ready, Bit 5 = TX Holding Register Empty, Bit 6 = TX FIFO Empty, Bit 7 = Transmitter Holding Register Empty

    reg [7:0] lsr_reg;
    reg [7:0] msr_reg;

    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            lsr_reg <= 0;
            msr_reg <= 0;
        end else begin
            // Update LSR
            lsr_reg[0] <= 0; // RX Overrun
            lsr_reg[1] <= 0; // RX Parity Error
            lsr_reg[2] <= 0; // RX Framing Error
            lsr_reg[3] <= 0; // RX Break
            lsr_reg[4] <= rx_fifo_empty ? 0 : 1; // RX Data Ready
            lsr_reg[5] <= tx_fifo_empty ? 1 : 0; // TX Holding Register Empty
            lsr_reg[6] <= tx_fifo_empty ? 1 : 0; // TX FIFO Empty
            lsr_reg[7] <= tx_shift_reg_valid ? 1 : 0; // Transmitter Holding Register Empty

            // Update MSR
            msr_reg[0] <= cts_pad_i; // CTS
            msr_reg[1] <= dsr_pad_i; // DSR
            msr_reg[2] <= ri_pad_i; // RI
            msr_reg[3] <= dcd_pad_i; // DCD
            msr_reg[4:7] <= 4'b0000; // Other bits
        end
    end

    // Wishbone Read Logic
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            wb_dat_o <= 0;
            wb_ack_o <= 0;
        end else begin
            if (wb_stb_i && wb_cyc_i && wb_ack_o_reg == 0) begin
                case (wb_adr_i)
                    16'h00: wb_dat_o <= mcr_reg; // LCR (Overlap)
                    16'h01: wb_dat_o <= mcr_reg; // MCR
                    16'h02: wb_dat_o <= mcr_reg; // IER (Overlap)
                    16'h03: wb_dat_o <= mcr_reg; // IIR (Overlap)
                    16'h04: wb_dat_o <= lsr_reg; // LSR
                    16'h05: wb_dat_o <= msr_reg; // MSR
                    16'h06: wb_dat_o <= rx_fifo[0]; // RHR
                    16'h07: wb_dat_o <= tx_fifo[0]; // THR
                    16'h08: wb_dat_o <= mcr_reg; // FCR (Overlap)
                    16'h09: wb_dat_o <= mcr_reg; // EFR (Overlap)
                    16'h0A: wb_dat_o <= msr_reg; // MSR (Overlap)
                    16'h0B: wb_dat_o <= lsr_reg; // LCR (Overlap)
                    16'h0C: wb_dat_o <= lsr_reg; // LSR (Overlap)
                    16'h0D: wb_dat_o <= mcr_reg; // MCR (Overlap)
                    16'h0E: wb_dat_o <= mcr_reg; // IER (Overlap)
                    16'h0F: wb_dat_o <= mcr_reg; // IIR (Overlap)
                    default: wb_dat_o <= 0;
                endcase
            end
        end
    end

    // Wishbone Write Logic
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            wb_ack_o <= 0;
        end else begin
            if (wb_stb_i && wb_cyc_i && wb_ack_o_reg == 0) begin
                case (wb_adr_i)
                    16'h00: begin
                        // LCR: Bit 0 = DLAB
                        // We ignore DLAB here as we use fixed baud
                        mcr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h01: begin
                        mcr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h02: begin
                        mcr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h03: begin
                        mcr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h04: begin
                        lsr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h05: begin
                        msr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h06: begin
                        rx_fifo[0] <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h07: begin
                        tx_fifo[0] <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h08: begin
                        mcr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h09: begin
                        mcr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h0A: begin
                        msr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h0B: begin
                        lsr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h0C: begin
                        lsr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h0D: begin
                        mcr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h0E: begin
                        mcr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    16'h0F: begin
                        mcr_reg <= wb_dat_i;
                        wb_ack_o <= 1;
                    end
                    default: wb_ack_o <= 0;
                endcase
            end
        end
    end

    // Output Assignments
    assign wb_ack_o_reg = wb_ack_o;
    assign int_o_reg = int_o;
    assign stx_pad_o = stx_pad_o_reg;
    assign rts_pad_o = rts_pad_o_reg;
    assign dtr_pad_o = dtr_pad_o_reg;

endmodule
