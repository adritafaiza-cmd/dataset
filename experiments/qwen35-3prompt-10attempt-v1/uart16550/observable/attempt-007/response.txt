```verilog
module uart_top (
    wb_clk_i,
    wb_rst_i, wb_adr_i, wb_dat_i, wb_dat_o, wb_we_i, wb_stb_i, wb_cyc_i, wb_ack_o, wb_sel_i,
    int_o,
    stx_pad_o, srx_pad_i,
    rts_pad_o, cts_pad_i, dtr_pad_o, dsr_pad_i, ri_pad_i, dcd_pad_i
);

    // --------------------------------------------------------------------------
    // Parameters
    // --------------------------------------------------------------------------
    parameter WB_ADDR_WIDTH = 16;
    parameter WB_DATA_WIDTH = 32;
    parameter UART_FIFO_DEPTH = 16;
    parameter UART_BAUD_DIV = 16; // Simplified divisor for simulation; real design uses baud rate config
    
    // --------------------------------------------------------------------------
    // Internal Signals
    // --------------------------------------------------------------------------
    reg [WB_ADDR_WIDTH-1:0] wb_adr_reg;
    reg [WB_DATA_WIDTH-1:0] wb_dat_reg;
    reg wb_ack_reg;
    reg wb_stb_reg;
    reg wb_cyc_reg;
    reg wb_we_reg;
    reg [WB_DATA_WIDTH-1:0] wb_dat_in_reg;
    reg [WB_DATA_WIDTH-1:0] wb_dat_out_reg;
    reg wb_sel_reg;
    
    // UART State Machine
    reg [3:0] state;
    // State Encoding:
    // 0: IDLE
    // 1: START_BIT
    // 2: DATA_BIT_0
    // 3: DATA_BIT_1
    // 4: DATA_BIT_2
    // 5: DATA_BIT_3
    // 6: DATA_BIT_4
    // 7: DATA_BIT_5
    // 8: DATA_BIT_6
    // 9: DATA_BIT_7
    // 10: STOP_BIT
    // 11: RX_FIFO_FULL (Interrupt Pending)
    // 12: TX_FIFO_EMPTY (Interrupt Pending)
    // 13: TX_FIFO_NOT_EMPTY (Interrupt Pending)
    
    reg [4:0] tx_fifo_ptr;
    reg [4:0] rx_fifo_ptr;
    reg [UART_FIFO_DEPTH-1:0] tx_fifo;
    reg [UART_FIFO_DEPTH-1:0] rx_fifo;
    reg tx_fifo_empty;
    reg tx_fifo_full;
    reg rx_fifo_empty;
    reg rx_fifo_full;
    
    reg [7:0] data_in;
    reg [7:0] data_out;
    reg start_bit;
    reg stop_bit;
    reg tx_busy;
    reg rx_busy;
    
    reg [7:0] rx_data;
    reg rx_valid;
    reg tx_shift_reg;
    reg tx_shift_count;
    
    // Modem Control/Status Registers (Simplified mapping)
    // RTS, DTR, CTS, DSR, RI, DCD are handled via pads but internal logic tracks them
    reg rts_out;
    reg dtr_out;
    reg cts_in;
    reg dsr_in;
    reg ri_in;
    reg dcd_in;
    
    // Interrupt Logic
    reg int_pending;
    reg int_mask;
    
    // Timing
    reg clk_16x;
    reg [3:0] bit_counter;
    reg [7:0] shift_reg_rx;
    reg shift_reg_rx_valid;
    
    // --------------------------------------------------------------------------
    // Wishbone Interface Logic
    // --------------------------------------------------------------------------
    assign wb_ack_o = wb_ack_reg;
    assign wb_dat_o = wb_dat_out_reg;
    assign wb_stb_i = wb_stb_reg;
    assign wb_cyc_i = wb_cyc_reg;
    assign wb_we_i = wb_we_reg;
    assign wb_sel_i = wb_sel_reg;
    
    // Address Decoding
    // 0x00: Status Register (Read)
    // 0x01: Control Register (Write)
    // 0x02: Interrupt Enable Register (Write)
    // 0x03: Interrupt Mask Register (Write)
    // 0x04: Line Status Register (Read)
    // 0x05: IER (Read) - Alias for Interrupt Enable
    // 0x06: LSR (Read) - Alias for Line Status
    // 0x07: FIFO Control (Write)
    // 0x08: FIFO Status (Read)
    // 0x09: Modem Control (Read/Write)
    // 0x0A: Modem Status (Read)
    // 0x0B: Line Control (Write)
    // 0x0C: Baud Rate Generator (Read/Write)
    // 0x0D: Error Status (Read)
    // 0x0E: FIFO Threshold (Write)
    // 0x0F: FIFO Threshold (Read)
    
    // Simplified Address Map for this implementation:
    // 0x00: Status (Read)
    // 0x01: Control (Write)
    // 0x02: Interrupt Enable (Write)
    // 0x03: Interrupt Mask (Write)
    // 0x04: Line Status (Read)
    // 0x05: IER (Read)
    // 0x06: LSR (Read)
    // 0x07: FIFO Control (Write)
    // 0x08: FIFO Status (Read)
    // 0x09: Modem Control (Read/Write)
    // 0x0A: Modem Status (Read)
    // 0x0B: Line Control (Write)
    // 0x0C: Baud Rate Generator (Read/Write)
    // 0x0D: Error Status (Read)
    // 0x0E: FIFO Threshold (Write)
    // 0x0F: FIFO Threshold (Read)
    
    // --------------------------------------------------------------------------
    // Wishbone Read Logic
    // --------------------------------------------------------------------------
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            wb_dat_out_reg <= 32'b0;
            wb_ack_reg <= 1'b0;
        end else if (wb_stb_reg && wb_cyc_reg) begin
            case (wb_adr_reg)
                16'h00: wb_dat_out_reg <= {8'b0, status_reg}; // Status
                16'h01: wb_dat_out_reg <= {8'b0, control_reg}; // Control (Readback)
                16'h02: wb_dat_out_reg <= {8'b0, int_enable_reg}; // Interrupt Enable
                16'h03: wb_dat_out_reg <= {8'b0, int_mask_reg}; // Interrupt Mask
                16'h04: wb_dat_out_reg <= {8'b0, line_status_reg}; // Line Status
                16'h05: wb_dat_out_reg <= {8'b0, int_enable_reg}; // IER
                16'h06: wb_dat_out_reg <= {8'b0, line_status_reg}; // LSR
                16'h07: wb_dat_out_reg <= {8'b0, fifo_ctrl_reg}; // FIFO Control
                16'h08: wb_dat_out_reg <= {8'b0, fifo_status_reg}; // FIFO Status
                16'h09: wb_dat_out_reg <= {8'b0, modem_ctrl_reg}; // Modem Control
                16'h0A: wb_dat_out_reg <= {8'b0, modem_status_reg}; // Modem Status
                16'h0B: wb_dat_out_reg <= {8'b0, line_ctrl_reg}; // Line Control
                16'h0C: wb_dat_out_reg <= {8'b0, baud_gen_reg}; // Baud Rate Generator
                16'h0D: wb_dat_out_reg <= {8'b0, error_status_reg}; // Error Status
                16'h0E: wb_dat_out_reg <= {8'b0, fifo_thresh_reg}; // FIFO Threshold
                16'h0F: wb_dat_out_reg <= {8'b0, fifo_thresh_reg}; // FIFO Threshold
                default: wb_dat_out_reg <= 32'b0;
            endcase
            wb_ack_reg <= 1'b1;
        end
    end

    // --------------------------------------------------------------------------
    // Wishbone Write Logic
    // --------------------------------------------------------------------------
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            wb_dat_in_reg <= 32'b0;
        end else if (wb_stb_reg && wb_cyc_reg && wb_we_i) begin
            case (wb_adr_reg)
                16'h01: control_reg <= wb_dat_in_reg[7:0];
                16'h02: int_enable_reg <= wb_dat_in_reg[7:0];
                16'h03: int_mask_reg <= wb_dat_in_reg[7:0];
                16'h07: fifo_ctrl_reg <= wb_dat_in_reg[7:0];
                16'h09: modem_ctrl_reg <= wb_dat_in_reg[7:0];
                16'h0B: line_ctrl_reg <= wb_dat_in_reg[7:0];
                16'h0C: baud_gen_reg <= wb_dat_in_reg[7:0];
                16'h0E: fifo_thresh_reg <= wb_dat_in_reg[7:0];
                default: wb_dat_in_reg <= wb_dat_in_reg; // Hold
            endcase
        end
    end

    // --------------------------------------------------------------------------
    // UART State Machine & Logic
    // --------------------------------------------------------------------------
    always @(posedge wb_clk_i or posedge wb_rst_i) begin
        if (wb_rst_i) begin
            state <= 4'b0000; // IDLE
            tx_fifo_ptr <= 5'b0;
            rx_fifo_ptr <= 5'b0;
            tx_fifo <= 16'h0;
            rx_fifo <= 16'h0;
            tx_fifo_empty <= 1'b1;
            tx_fifo_full <= 1'b0;
            rx_fifo_empty <= 1'b1;
            rx_fifo_full <= 1'b0;
            data_in <= 8'b0;
            data_out <= 8'b0;
            start_bit <= 1'b0;
            stop_bit <= 1'b1;
            tx_busy <= 1'b0;
            rx_busy <= 1'b0;
            tx_shift_reg <= 8'b0;
            tx_shift_count <= 4'b0;
            shift_reg_rx <= 8'b0;
            shift_reg_rx_valid <= 1'b0;
            int_pending <= 1'b0;
            int_mask <= 1'b0;
            rts_out <= 1'b1;
            dtr_out <= 1'b1;
            cts_in <= 1'b0;
            dsr_in <= 1'b0;
            ri_in <= 1'b0;
            dcd_in <= 1'b0;
            bit_counter <= 4'b0;
        end else begin
            // State Machine
            case (state)
                4'b0000: begin // IDLE
                    if (start_bit) begin
                        state <= 4'b0001; // START_BIT
                        start_bit <= 1'b0;
                        stop_bit <= 1'b1;
                    end
                end
                4'b0001: begin // START_BIT
                    if (bit_counter == 4'b1111) begin
                        state <= 4'b0010; // DATA_BIT_0
                        bit_counter <= 4'b0;
                    end
                end
                4'b0010: begin // DATA_BIT_0
                    if (bit_counter == 4'b1111) begin
                        state <= 4'b0011; // DATA_BIT_1
                        bit_counter <= 4'b0;
                    end
                end
                4'b0011: begin // DATA_BIT_1
                    if (bit_counter == 4'b1111) begin
                        state <= 4'b0100; // DATA_BIT_2
                        bit_counter <= 4'b0;
                    end
                end
                4'b0100: begin // DATA_BIT_2
                    if (bit_counter == 4'b1111) begin
                        state <= 4'b0101; // DATA_BIT_3
                        bit_counter <= 4'b0;
                    end
                end
                4'b0101: begin // DATA_BIT_3
                    if (bit_counter == 4'b1111) begin
                        state <= 4'b0110; // DATA_BIT_4
                        bit_counter <= 4'b0;
                    end
                end
                4'b0110: begin // DATA_BIT_4
                    if (bit_counter == 4'b1111) begin
                        state <= 4'b0111; // DATA_BIT_5
                        bit_counter <= 4'b0;
                    end
                end
                4'b0111: begin // DATA_BIT_5
                    if (bit_counter == 4'b1111) begin
                        state <= 4'b1000; // DATA_BIT_6
                        bit_counter <= 4'b0;
                    end
                end
                4'b1000: begin // DATA_BIT_6
                    if (bit_counter == 4'b1111) begin
                        state <= 4'b1001; // DATA_BIT_7
                        bit_counter <= 4'b0;
                    end
                end
                4'b1001: begin // DATA_BIT_7
                    if (bit_counter == 4'b1111) begin
                        state <= 4'b1010; // STOP_BIT
                        bit_counter <= 4'b0;
                    end
                end
                4'b1010: begin // STOP_BIT
                    if (bit_counter == 4'b1111) begin
                        state <= 4'b0000; // IDLE
                        stop_bit <= 1'b1;
                        bit_counter <= 4'b0;
                    end
                end
                4'b1011: begin // RX_FIFO_FULL
                    if (rx_fifo_full == 1'b0) begin
                        state <= 4'b0000; // IDLE
                        rx_fifo_full <= 1'b0;
                    end
                end
                4'b1100: begin // TX_FIFO_EMPTY
                    if (tx_fifo_empty == 1'b1) begin
                        state <= 4'b0000; // IDLE
                        tx_fifo_empty <= 1'b0;
                    end
                end
                4'b1101: begin // TX_FIFO_NOT_EMPTY
                    if (tx_fifo_empty == 1'b1) begin
                        state <= 4'b0000; // IDLE
                        tx_fifo_empty <= 1'b0;
                    end
                end
                default: state <= 4'b0000;
            endcase
            
            // Shift Register Logic
            if (state == 4'b0001) begin // START_BIT
                shift_reg_rx <= 8'b1xxxxxxx; // Start bit is 0, but we shift in 0 first
                shift_reg_rx_valid <= 1'b0;
            end else if (state == 4'b0010) begin // DATA_BIT_0
                shift_reg_rx <= {shift_reg_rx[6:0], srx_pad_i};
                shift_reg_rx_valid <= 1'b1;
            end else if (state == 4'b0011) begin // DATA_BIT_1
                shift_reg_rx <= {shift_reg_rx[6:0], srx_pad_i};
            end else if (state == 4'b0100) begin // DATA_BIT_2
                shift_reg_rx <= {shift_reg_rx[6:0], srx_pad_i};
            end else if (state == 4'b0101) begin // DATA_BIT_3
                shift_reg_rx <= {shift_reg_rx[6:0], srx_pad_i};
            end else if (state == 4'b0110) begin // DATA_BIT_4
                shift_reg_rx <= {shift_reg_rx[6:0], srx_pad_i};
            end else if (state == 4'b0111) begin // DATA_BIT_5
                shift_reg_rx <= {shift_reg_rx[6:0], srx_pad_i};
            end else if (state == 4'b1000) begin // DATA_BIT_6
                shift_reg_rx <= {shift_reg_rx[6:0], srx_pad_i};
            end else if (state == 4'b1001) begin // DATA_BIT_7
                shift_reg_rx <= {shift_reg_rx[6:0], srx_pad_i};
            end else if (state == 4'b1010) begin // STOP_BIT
                shift_reg_rx <= {shift_reg_rx[6:0], srx_pad_i};
                if (shift_reg_rx_valid) begin
                    rx_data <= shift_reg_rx;
                    rx_fifo[rx_fifo_ptr] <= rx_data;
                    rx_fifo_ptr <= rx_fifo_ptr + 1;
                    rx_fifo_full <= 1'b1;
                    rx_fifo_empty <= 1'b0;
                    rx_valid <= 1'b1;
                    shift_reg_rx_valid <= 1'b0;
                end
            end
            
            // TX Shift Register Logic
            if (state == 4'b0001) begin // START_BIT
                tx_shift_reg <= 8'b0;
                tx_shift_count <= 4'b0;
            end else if (state == 4'b0010) begin // DATA_BIT_0
                tx_shift_reg <= {tx_shift_reg[6:0], tx_fifo[tx_fifo_ptr][0]};
                tx_shift_count <= 4'b1;
            end else if (state == 4'b0011) begin // DATA_BIT_1
                tx_shift_reg <= {tx_shift_reg[6:0], tx_fifo[tx_fifo_ptr][1]};
                tx_shift_count <= 4'b2;
            end else if (state == 4'b0100) begin // DATA_BIT_2
                tx_shift_reg <= {tx_shift_reg[6:0], tx_fifo[tx_fifo_ptr][2]};
                tx_shift_count <= 4'b3;
            end else if (state == 4'b0101) begin // DATA_BIT_3
                tx_shift_reg <= {tx_shift_reg[6:0], tx_fifo[tx_fifo_ptr][3]};
                tx_shift_count <= 4'b4;
            end else if (state == 4'b0110) begin // DATA_BIT_4
                tx_shift_reg <= {tx_shift_reg[6:0], tx_fifo[tx_fifo_ptr][4]};
                tx_shift_count <= 4'b5;
            end else if (state == 4'b0111) begin // DATA_BIT_5
                tx_shift_reg <= {tx_shift_reg[6:0], tx_fifo[tx_fifo_ptr][5]};
                tx_shift_count <= 4'b6;
            end else if (state == 4'b1000) begin // DATA_BIT_6
                tx_shift_reg <= {tx_shift_reg[6:0], tx_fifo[tx_fifo_ptr][6]};
                tx_shift_count <= 4'b7;
            end else if (state == 4'b1001) begin // DATA_BIT_7
                tx_shift_reg <= {tx_shift_reg[6:0], tx_fifo[tx_fifo_ptr][7]};
                tx_shift_count <= 4'b8;
            end else if (state == 4'b1010) begin // STOP_BIT
                tx_shift_reg <= {tx_shift_reg[6:0], 1'b1};
                tx_shift_count <= 4'b9;
            end
            
            // FIFO Management
            if (tx_fifo_ptr == 5'b0) begin
                tx_fifo_empty <= 1'b1;
                tx_fifo_full <= 1'b0;
            end else begin
                tx_fifo_empty <= 1'b0;
                tx_fifo_full <= 1'b1;
            end
            
            if (rx_fifo_ptr == 5'b0) begin
                rx_fifo_empty <= 1'b1;
                rx_fifo_full <= 1'b0;
            end else begin
                rx_fifo_empty <= 1'b0;
                rx_fifo_full <= 1'b1;
            end
            
            // Interrupt Logic
            int_pending <= 1'b0;
            if (int_enable_reg[0] && rx_fifo_full && !int_mask[0]) int_pending <= 1'b1;
            if (int_enable_reg[1] && tx_fifo_empty && !int_mask[1]) int_pending <= 1'b1;
            if (int_enable_reg[2] && tx_fifo_not_empty && !int_mask[2]) int_pending <= 1'b1;
            
            // Output Interrupt
            int_o <= int_pending;
            
            // Modem Control/Status
            rts_out <= !modem_ctrl_reg[0]; // RTS is active low
            dtr_out <= !modem_ctrl_reg[1]; // DTR is active low
            cts_in <= cts_pad_i;
            dsr_in <= dsr_pad_i;
            ri_in <= ri_pad_i;
            dcd_in <= dcd_pad_i;
            
            // Pad Outputs
            stx_pad_o <= 1'b1; // Active high
            srx_pad_i <= srx_pad_i; // Input
            rts_pad_o <= rts_out;
            cts_pad_i <= cts_in;
            dtr_pad_o <= dtr_out;
            dsr_pad_i <= dsr_in;
            ri_pad_i <= ri_in;
            dcd_pad_i <= dcd_in;
            
            // Register Values
            status_reg <= (rx_fifo_full ? 1'b1 : 1'b0) | (tx_fifo_empty ? 1'b2 : 1'b0) | (rx_fifo_empty ? 1'b4 : 1'b0) | (tx_fifo_full ? 1'b8 : 1'b0);
            control_reg <= modem_ctrl_reg;
            int_enable_reg <= int_enable_reg;
            int_mask_reg <= int_mask_reg;
            fifo_ctrl_reg <= fifo_ctrl_reg;
            fifo_status_reg <= (rx_fifo_full ? 1'b1 : 1'b0) | (tx_fifo_empty ? 1'b2 : 1'b0);
            modem_ctrl_reg <= modem_ctrl_reg;
            modem_status_reg <= (cts_in ? 1'b1 : 1'b0) | (dsr_in ? 1'b2 : 1'b0) | (ri_in ? 1'b4 : 1'b0) | (dcd_in ? 1'b8 : 1'b0);
            line_ctrl_reg <= line_ctrl_reg;
            baud_gen_reg <= baud_gen_reg;
            error_status_reg <= 8'b0;
            fifo_thresh_reg <= fifo_thresh_reg;
            
            // TX FIFO Not Empty Check
            tx_fifo_not_empty <= (tx_fifo_ptr != 5'b0);
            
            // TX Shift Count Check
            if (tx_shift_count == 4'b10) begin // After STOP_BIT
                if (tx_fifo_empty) begin
                    state <= 4'b0000; // IDLE
                end else begin
                    tx_fifo_ptr <= tx_fifo_ptr - 1;
                    state <= 4'b0001; // START_BIT
                end
            end
        end
    end

    // --------------------------------------------------------------------------
    // Register Definitions
    // --------------------------------------------------------------------------
    reg [7:0] status_reg;
    reg [7:0] control_reg;
    reg [7:0] int_enable_reg;
    reg [7:0] int_mask_reg;
    reg [7:0] fifo_ctrl_reg;
    reg [7:0] fifo_status_reg;
    reg [7:0] modem_ctrl_reg;
    reg [7:0] modem_status_reg;
    reg [7:0] line_ctrl_reg;
    reg [7:0] baud_gen_reg;
    reg [7:0] error_status_reg;
    reg [7:0] fifo_thresh_reg;
    reg tx_fifo_not_empty;

endmodule
