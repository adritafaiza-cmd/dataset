module sync_reset #
(
    parameter N = 2
)
(
    input  wire clk,
    input  wire rst,
    output wire out
);

    reg [N-1:0] sync_cnt;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            sync_cnt <= 0;
            out      <= 0;
        end else begin
            sync_cnt <= sync_cnt + 1;
            if (sync_cnt >= N) begin
                out <= 1;
            end else begin
                out <= 0;
            end
        end
    end

endmodule<|im_end|>
