module pulse_sync #(
    parameter STAGES = 2
)(
    input  wire clk_a,
    input  wire rstn_a,
    input  wire clk_b,
    input  wire rstn_b,
    input  wire pulseA_i,
    output wire pulseB_o,
    output wire busy_o
);

    // Internal signals
    wire [STAGES-1:0] busy_reg_a;
    wire [STAGES-1:0] busy_reg_b;
    wire [STAGES-1:0] pulse_reg_a;
    wire [STAGES-1:0] pulse_reg_b;

    // Clock Domain A: Capture incoming pulse and propagate busy signal
    // ------------------------------------------------------------------

    //    We need rstn_a in Domain B to know when to clear the busy flag in B.
    wire rstn_a_synced;
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            rstn_a_synced <= 1'b1;
        else
            rstn_a_synced <= rstn_a;
    end

    //    We capture the pulse on the first clock edge after reset.
    //    To ensure we don't miss a pulse if it arrives exactly at reset release,
    //    we use a simple edge detection or level capture with a guard.
    //    Here we use a level capture with a "valid" flag that clears on next clock.
    wire pulseA_valid;
    reg pulseA_valid;
    
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            pulseA_valid <= 1'b0;
        else if (pulseA_i)
            pulseA_valid <= 1'b1;
        else
            pulseA_valid <= 1'b0;
    end

    // 3. Busy State Machine in Domain A
    //    busy_reg_a tracks if a pulse is currently being transferred.
    //    It is set when pulseA_valid is high and cleared after STAGES clocks.
    reg [STAGES-1:0] busy_reg_a;
    
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            busy_reg_a <= {STAGES{1'b0}};
        else begin
            // Shift left: if any stage is busy, the next stage becomes busy
            // The MSB (STAGES-1) is the output busy signal for domain A logic
            busy_reg_a <= {busy_reg_a[STAGES-2:0], pulseA_valid};
        end
    end

    // 4. Pulse Propagation in Domain A
    //    We shift the pulse through STAGES flip-flops to create a delay window.
    //    If busy_reg_a[STAGES-1] is high, it means a pulse is in flight.
    //    We only allow new pulses if busy_reg_a[STAGES-1] is low.
    //    However, the requirement says "ignore additional pulseA_i while busy".
    //    This is handled by the pulseA_valid logic above (it only sets if pulseA_i is high).
    //    But we must ensure we don't start a new transfer if one is already in flight.
    //    Actually, the standard way is: if busy, ignore input. If not busy, capture and shift.
    //    Let's refine the capture logic to respect the "busy" state.
    
    reg [STAGES-1:0] pulse_reg_a;
    reg pulse_a_start;

    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            pulse_reg_a <= {STAGES{1'b0}};
        else begin

            // AND a new pulse is detected.
            pulse_a_start = (busy_reg_a[STAGES-1] == 1'b0) && pulseA_valid;
            
            // Shift register logic
            pulse_reg_a <= {pulse_reg_a[STAGES-2:0], pulse_a_start};
        end
    end

    // ------------------------------------------------------------------
    // Clock Domain B: Receive pulse and propagate busy signal back
    // ------------------------------------------------------------------

    // 1. Reset Synchronizer for rstn_b to Domain A (needed for busy_reg_a logic)
    //    We need rstn_b in Domain A to know when to clear the busy flag in A.
    wire rstn_b_synced;
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a)
            rstn_b_synced <= 1'b1;
        else
            rstn_b_synced <= rstn_b;
    end

    // 2. Pulse Capture in Domain B
    //    We capture the pulse from Domain A.
    wire pulseA_received;
    reg pulseA_received;
    
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            pulseA_received <= 1'b0;
        else if (pulse_reg_b[STAGES-1]) // Pulse arrived at the last stage
            pulseA_received <= 1'b1;
        else
            pulseA_received <= 1'b0;
    end

    // 3. Busy State Machine in Domain B
    //    busy_reg_b tracks if a pulse is currently being transferred.
    //    It is set when pulseA_received is high and cleared after STAGES clocks.
    reg [STAGES-1:0] busy_reg_b;
    
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            busy_reg_b <= {STAGES{1'b0}};
        else begin
            busy_reg_b <= {busy_reg_b[STAGES-2:0], pulseA_received};
        end
    end

    // 4. Pulse Propagation in Domain B
    reg [STAGES-1:0] pulse_reg_b;
    reg pulse_b_start;

    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b)
            pulse_reg_b <= {STAGES{1'b0}};
        else begin
            // Only start a new pulse if the previous one has cleared
            pulse_b_start = (busy_reg_b[STAGES-1] == 1'b0) && pulseA_received;
            
            pulse_reg_b <= {pulse_reg_b[STAGES-2:0], pulse_b_start};
        end
    end

    // ------------------------------------------------------------------
    // Output Generation
    // ------------------------------------------------------------------

    // Output pulseB_o: High when the pulse reaches the last stage in Domain B
    assign pulseB_o = pulse_reg_b[STAGES-1];

    // Output busy_o: High if either domain is busy
    // This ensures the external logic knows the system is occupied.
    assign busy_o = busy_reg_a[STAGES-1] | busy_reg_b[STAGES-1];

    // CDC Safety Checks (Implicit in design)

    //    - rstn_a -> rstn_a_synced (Domain B): 2-stage synchronizer logic implemented via sequential logic.

            rstn_b_synced_2 <= 1'b1;
        end
        else begin
            rstn_b_synced_1 <= rstn_b;
            rstn_b_synced_2 <= rstn_b_synced_1;
        end
    end

    reg pulseA_valid_new;
    reg [STAGES-1:0] busy_reg_a_new;
    reg [STAGES-1:0] pulse_reg_a_new;

            pulseA_valid_new <= 1'b0;
            busy_reg_a_new <= {STAGES{1'b0}};
            pulse_reg_a_new <= {STAGES{1'b0}};

            pulseA_valid_new <= pulseA_i;

            busy_reg_a_new <= {busy_reg_a_new[STAGES-2:0], pulseA_valid_new};
            
            // Pulse Shift Logic
            pulse_a_start_new = (busy_reg_a_new[STAGES-1] == 1'b0) && pulseA_valid_new;
            pulse_reg_a_new <= {pulse_reg_a_new[STAGES-2:0], pulse_a_start_new};
        end
    end

    // --- Revised Domain B Logic ---
    reg pulseA_received_new;
    reg [STAGES-1:0] busy_reg_b_new;
    reg [STAGES-1:0] pulse_reg_b_new;
    reg pulse_b_start_new;

    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b) begin
            pulseA_received_new <= 1'b0;
            busy_reg_b_new <= {STAGES{1'b0}};
            pulse_reg_b_new <= {STAGES{1'b0}};
        end
        else begin
            // Pulse Capture
            pulseA_received_new <= pulse_reg_b_new[STAGES-1];
            
            // Busy Logic
            busy_reg_b_new <= {busy_reg_b_new[STAGES-2:0], pulseA_received_new};

    wire rstn_a_synced_b2;
    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b) begin
            rstn_a_synced_b1 <= 1'b1;
            rstn_a_synced_b2 <= 1'b1;
        end
        else begin
            rstn_a_synced_b1 <= rstn_a;
            rstn_a_synced_b2 <= rstn_a_synced_b1;
        end
    end

    wire rstn_b_synced_a1;
    wire rstn_b_synced_a2;
    always @(posedge clk_a or negedge rstn_a) begin
        if (!rstn_a) begin
            rstn_b_synced_a1 <= 1'b1;
            rstn_b_synced_a2 <= 1'b1;
        end
        else begin
            rstn_b_synced_a1 <= rstn_b;
            rstn_b_synced_a2 <= rstn_b_synced_a1;
        end
    end

    assign pulseB_o = pulse_reg_b_new[STAGES-1];
    assign busy_o = busy_reg_a_new[STAGES-1] | busy_reg_b_new[STAGES-1];

    // We must re-integrate the synchronized resets into the main always blocks.
    // Since Verilog doesn't support multiple reset sources easily in one block without duplication,
    // we will create a helper module or just duplicate the logic cleanly.
    // Given the constraint of "one source file" and "no testbench", a clean single module is best.
    // We will refactor the module to use the synchronized resets explicitly.

    // Refactored Implementation below for clarity and correctness.
    
    // ------------------------------------------------------------------

    // ------------------------------------------------------------------

    wire rstn_a_synced_b1, rstn_a_synced_b2;
    wire rstn_b_synced_a1, rstn_b_synced_a2;

    // rstn_a -> Domain B (2-stage)

        if (!rstn_b) begin
            rstn_a_synced_b1 <= 1'b1;
            rstn_a_synced_b2 <= 1'b1;

            rstn_a_synced_b2 <= rstn_a_synced_b1;

    reg [STAGES-1:0] pulse_b;
    reg pulse_start_b;

    always @(posedge clk_b or negedge rstn_b) begin
        if (!rstn_b) begin
            pulseA_valid_b <= 1'b0;
            busy_b <= {STAGES{1'b0}};
            pulse_b <= {STAGES{1'b0}};
        end
        else begin
            pulseA_valid_b <= pulse_b[STAGES-1];
            busy_b <= {busy_b[STAGES-2:0], pulseA_valid_b};

    assign busy_o = busy_a[STAGES-1] | busy_b[STAGES-1];

    // The current implementation uses rstn_a and rstn_b directly in the always blocks.
    // This is safe because the reset is active in its own domain.

    // When rstn_a is released, busy_a clears. When rstn_b is released, busy_b clears.
    // This is correct behavior. The cross-domain reset syncs are only needed if we were
    // using a global reset signal or if the reset logic depended on the other domain's state.
    // Since the requirement is "Reset release must be safe", and we have local resets,

    // However, to be absolutely robust against metastability during reset release in the OTHER domain

    // Here, since we have independent resets, the design is safe as is.
