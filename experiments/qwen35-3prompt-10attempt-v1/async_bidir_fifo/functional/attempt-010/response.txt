```verilog
`timescale 1ns/1ps

module async_bidir_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "TRUE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir
       );

  localparam INT_ASIZE = ASIZE;
  localparam INT_DEPTH = 1 << ASIZE;
  localparam INT_MASK  = (1 << ASIZE) - 1;

  // Cross-clock pointers
  reg [INT_ASIZE-1:0] a_write_ptr;
  reg [INT_ASIZE-1:0] a_read_ptr;
  reg [INT_ASIZE-1:0] b_write_ptr;
  reg [INT_ASIZE-1:0] b_read_ptr;

  // Cross-clock data registers
  reg [DSIZE-1:0] a_write_data_reg;
  reg [DSIZE-1:0] b_write_data_reg;

  // Cross-clock pointers for almost-full/almost-empty detection
  // We use a separate pointer that lags by 1 to detect "almost full/empty"
  // relative to the other side's view.
  reg [INT_ASIZE-1:0] a_write_ptr_afull;
  reg [INT_ASIZE-1:0] a_read_ptr_aempty;
  reg [INT_ASIZE-1:0] b_write_ptr_afull;
  reg [INT_ASIZE-1:0] b_read_ptr_aempty;

  // Internal flags for almost-full/almost-empty logic
  reg a_almost_full;
  reg a_almost_empty;
  reg b_almost_full;
  reg b_almost_empty;

  // Write enable signals derived from direction
  wire a_write_en = (a_dir == 1'b1) ? a_winc : 1'b0;
  wire b_write_en = (b_dir == 1'b1) ? b_winc : 1'b0;

  // Read enable signals derived from direction
  wire a_read_en = (a_dir == 1'b0) ? a_rinc : 1'b0;
  wire b_read_en = (b_dir == 1'b0) ? b_rinc : 1'b0;

  // Full/Empty logic based on pointer comparison
  // Full: write_ptr == read_ptr + 1 (modulo depth)
  // Empty: write_ptr == read_ptr
  wire a_full_local = (a_write_ptr == (a_read_ptr + 1) & INT_MASK);
  wire a_empty_local = (a_write_ptr == a_read_ptr);
  wire b_full_local = (b_write_ptr == (b_read_ptr + 1) & INT_MASK);
  wire b_empty_local = (b_write_ptr == b_read_ptr);

  // Almost Full/Empty logic
  // Almost Full: write_ptr == read_ptr + 2 (modulo depth)
  // Almost Empty: write_ptr == read_ptr - 1 (modulo depth)
  wire a_almost_full_local = (a_write_ptr == (a_read_ptr + 2) & INT_MASK);
  wire a_almost_empty_local = (a_write_ptr == (a_read_ptr - 1 + INT_DEPTH) & INT_MASK);
  wire b_almost_full_local = (b_write_ptr == (b_read_ptr + 2) & INT_MASK);
  wire b_almost_empty_local = (b_write_ptr == (b_read_ptr - 1 + INT_DEPTH) & INT_MASK);

  // Almost full/empty flags are sampled by the opposite side
  // They are registered in the opposite clock domain to avoid metastability
  // and to provide a stable signal for the other side.
  reg a_afull_reg;
  reg a_aempty_reg;
  reg b_afull_reg;
  reg b_aempty_reg;

  // Data output registers
  reg [DSIZE-1:0] a_rdata_reg;
  reg [DSIZE-1:0] b_rdata_reg;

  // Write pointer update logic (A side)
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_write_ptr <= 0;
      a_read_ptr  <= 0;
      a_write_data_reg <= 0;
      a_write_ptr_afull <= 0;
      a_read_ptr_aempty <= 0;
    end else begin
      if (a_write_en) begin
        a_write_ptr <= a_write_ptr + 1;
        a_write_ptr_afull <= a_write_ptr + 1; // Lagging pointer for almost-full
        a_write_data_reg <= a_wdata;
      end else begin
        a_write_ptr_afull <= a_write_ptr;
      end
      if (a_read_en) begin
        a_read_ptr <= a_read_ptr + 1;
        a_read_ptr_aempty <= a_read_ptr + 1; // Lagging pointer for almost-empty
      end else begin
        a_read_ptr_aempty <= a_read_ptr;
      end
    end
  end

  // Write pointer update logic (B side)
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_write_ptr <= 0;
      b_read_ptr  <= 0;
      b_write_data_reg <= 0;
      b_write_ptr_afull <= 0;
      b_read_ptr_aempty <= 0;
    end else begin
      if (b_write_en) begin
        b_write_ptr <= b_write_ptr + 1;
        b_write_ptr_afull <= b_write_ptr + 1;
        b_write_data_reg <= b_wdata;
      end else begin
        b_write_ptr_afull <= b_write_ptr;
      end
      if (b_read_en) begin
        b_read_ptr <= b_read_ptr + 1;
        b_read_ptr_aempty <= b_read_ptr + 1;
      end else begin
        b_read_ptr_aempty <= b_read_ptr;
      end
    end
  end

  // Almost full/empty flag generation (A side)
  // These flags are generated in A clock domain but sampled in B clock domain
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_afull_reg <= 0;
      a_aempty_reg <= 1;
    end else begin
      a_afull_reg <= a_almost_full_local;
      a_aempty_reg <= a_almost_empty_local;
    end
  end

  // Almost full/empty flag generation (B side)
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_afull_reg <= 0;
      b_aempty_reg <= 1;
    end else begin
      b_afull_reg <= b_almost_full_local;
      b_aempty_reg <= b_almost_empty_local;
    end
  end

  // Read data output logic (A side)
  // Data is read from the cross-clock register when a_read_en is asserted
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_rdata_reg <= 0;
    end else if (a_read_en) begin
      // Read data from the B-side write register
      a_rdata_reg <= b_write_data_reg;
    end
  end

  // Read data output logic (B side)
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_rdata_reg <= 0;
    end else if (b_read_en) begin
      // Read data from the A-side write register
      b_rdata_reg <= a_write_data_reg;
    end
  end

  // Output assignments
  assign a_rdata = a_rdata_reg;
  assign b_rdata = b_rdata_reg;

  // Full/Empty flags (local to each side)
  assign a_full = a_full_local;
  assign a_empty = a_empty_local;
  assign b_full = b_full_local;
  assign b_empty = b_empty_local;

  // Almost Full/Empty flags (cross-domain)
  // These are sampled in the opposite clock domain
  assign a_afull = a_afull_reg;
  assign a_aempty = a_aempty_reg;
  assign b_afull = b_afull_reg;
  assign b_aempty = b_aempty_reg;

endmodule
