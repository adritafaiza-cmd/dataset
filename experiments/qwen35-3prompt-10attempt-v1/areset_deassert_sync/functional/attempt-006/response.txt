module areset_deassert_sync #(
    parameter CHAINS = 2,
    parameter RST_POL = 1'b1
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);

    reg [CHAINS-1:0] rst_pipe;

    // Stage 0: Immediate assertion handling
    // If RST_POL is 1, active high means assert when async_rst_i is 1.
    // If RST_POL is 0, active low means assert when async_rst_i is 0.
    // We invert the input if active-low to maintain a consistent "asserted" logic flow for the pipeline.
    assign rst_pipe[0] = RST_POL ? async_rst_i : ~async_rst_i;

    // Pipeline stages for release synchronization
    // Each stage synchronizes the deassertion (release) of the reset signal.
    genvar i;
    generate
        for (i = 1; i < CHAINS; i = i + 1) begin : rst_pipe_stages
            always @(posedge clk) begin
                if (rst_pipe[i-1]) begin
                    // If the previous stage is asserted, hold the assertion
                    rst_pipe[i] <= 1'b1;
                end else begin
                    // If the previous stage is deasserted, propagate deassertion
                    rst_pipe[i] <= 0;
                end
            end
        end
    endgenerate

    // Final output logic
    // Invert back if active-low to match the original polarity
    assign sync_rst_o = RST_POL ? rst_pipe[CHAINS-1] : ~rst_pipe[CHAINS-1];

endmodule<|im_end|>
