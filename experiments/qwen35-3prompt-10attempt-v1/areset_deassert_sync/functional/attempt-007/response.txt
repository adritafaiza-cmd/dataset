module areset_deassert_sync #(
    parameter CHAINS = 2,
    parameter RST_POL = 1'b1
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);

    reg [CHAINS-1:0] rst_pipe;

    // Stage 0: Capture the asynchronous reset signal
    // If RST_POL is 1, we capture active-high. If 0, we capture active-low.
    always @(posedge clk) begin
        if (RST_POL)
            rst_pipe[0] <= async_rst_i;
        else
            rst_pipe[0] <= ~async_rst_i;
    end

    // Stages 1 to CHAINS-1: Shift register to synchronize the deassertion
    // This ensures that the release of the reset is stable across clock domains
    // and observed by multiple clock edges before being propagated.
    genvar i;
    generate
        for (i = 1; i < CHAINS; i = i + 1) begin : rst_pipe_stages
            always @(posedge clk) begin
                rst_pipe[i] <= rst_pipe[i-1];
            end
        end
    endgenerate

    // Output logic:
    // If RST_POL is 1, output is active when the last stage is high.
    // If RST_POL is 0, output is active when the last stage is low.
    assign sync_rst_o = (RST_POL) ? rst_pipe[CHAINS-1] : ~rst_pipe[CHAINS-1];

endmodule<|im_end|>
