module areset_deassert_sync #(
    parameter CHAINS = 2,
    parameter RST_POL = 1'b1
)(
    input  clk,
    input  async_rst_i,
    output sync_rst_o
);

    reg [CHAINS-1:0] rst_pipe;

    // Stage 0: Immediate assertion handling
    // If RST_POL is 1, active high means assert when async_rst_i is 1.
    // If RST_POL is 0, active low means assert when async_rst_i is 0.
    // We invert the input if active-low to maintain a consistent "asserted" logic flow (1 = asserted).
    reg rst_asserted;
    assign rst_asserted = (RST_POL == 1'b1) ? async_rst_i : (~async_rst_i);

    // Pipeline stages for deassertion synchronization
    // On reset (asserted), the pipeline is cleared.
    // On release (deasserted), the signal propagates through the chain.
    // Only when the last stage is 0 (deasserted) do we output the synchronized reset.
    always @(posedge clk) begin
        if (rst_asserted) begin
            // Assertion detected: Clear the entire pipeline immediately
            rst_pipe <= {CHAINS{1'b0}};
        end else begin
            // Deassertion detected: Shift the pipeline
            // The first stage takes the current deassertion signal (which is 0 here)
            // Subsequent stages shift left
            rst_pipe <= {1'b0, rst_pipe[CHAINS-2:0]};
        end
    end

    // Output logic:
    // If the last stage of the pipeline is 0, it means the deassertion has been observed
    // for CHAINS clock cycles. We output 0 (inactive reset).
    // If the pipeline is still propagating (any 1s remain), we hold the previous state
    // or output 1 (active reset) to ensure safety during the transition.
    // However, the requirement states "sync_rst_o must follow assertion... immediately".
    // This implies if async_rst_i goes high, sync_rst_o goes high immediately.
    // If async_rst_i goes low, sync_rst_o stays high until the chain clears.
    
    // Logic refinement:
    // When rst_asserted is 1, sync_rst_o must be 1 immediately.
    // When rst_asserted is 0, sync_rst_o follows the pipeline.
    // The pipeline outputs 1 if any stage is 1, else 0.
    // Since we shift in 0s on deassertion, the pipeline will eventually become all 0s.
    // We need to ensure that while the chain is clearing, the output remains 1 (active reset).
    
    assign sync_rst_o = rst_asserted | rst_pipe[CHAINS-1];

endmodule<|im_end|>
