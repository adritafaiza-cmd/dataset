module sync #(
    parameter int unsigned STAGES = 2,
    parameter bit ResetValue = 1'b0
) (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic serial_i,
    output logic serial_o
);

    // Generate pipeline stages for CDC compliance
    // Using a synchronous chain where each stage captures the previous stage's output
    // on the destination clock. This ensures metastability is resolved before the next stage.
    genvar i;
    logic [STAGES-1:0] sync_chain;

    // Initialize the first stage of the chain
    // The first stage captures the asynchronous input serial_i
    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            sync_chain[0] <= ResetValue;
        else
            sync_chain[0] <= serial_i;
    end

    // Generate remaining stages
    generate
        for (i = 1; i < STAGES; i = i + 1) begin : stage_gen
            always_ff @(posedge clk_i or negedge rst_ni) begin
                if (!rst_ni)
                    sync_chain[i] <= ResetValue;
                else
                    sync_chain[i] <= sync_chain[i-1];
            end
        end
    endgenerate

    // Assign the final synchronized output
    assign serial_o = sync_chain[STAGES-1];

endmodule<|im_end|>
