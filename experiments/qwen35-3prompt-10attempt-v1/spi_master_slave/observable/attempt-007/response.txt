module spi_master #(
    parameter N = 8,
    parameter SPI_2X_CLK_DIV = 2
)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,
    output reg         spi_ssel_o,
    output reg         spi_sck_o,
    output             spi_mosi_o,
    input              spi_miso_i,
    output reg         di_req_o,
    input  [N-1:0]     di_i,
    input              wren_i,
    output reg         wr_ack_o,
    output reg         do_valid_o,
    output reg [N-1:0] do_o
);

    localparam SPI_CLK_DIV = SPI_2X_CLK_DIV;
    localparam MAX_DEPTH = 16; // Sufficient for typical SPI bursts

    // State Machine States
    localparam IDLE = 0;
    localparam PREPARE = 1;
    localprogram = 2;
    localparam TRANSFER = 3;
    localparam WAIT_ACK = 4;

    reg [2:0] state;
    reg [N-1:0] shift_reg;
    reg [N-1:0] rx_data;
    reg [N-1:0] tx_data;
    reg shift_en;
    reg bit_count;
    reg transfer_complete;
    reg ack_received;
    reg ssel_active;

    // Clock domain crossing registers for parallel interface
    reg [N-1:0] di_sync_reg;
    reg wren_sync_reg;
    reg di_req_sync_reg;
    reg wr_ack_sync_reg;
    reg do_valid_sync_reg;
    reg [N-1:0] do_sync_reg;

    // Synchronous logic for parallel interface
    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            di_sync_reg <= {N{1'b0}};
            wren_sync_reg <= 1'b0;
            di_req_sync_reg <= 1'b0;
            wr_ack_sync_reg <= 1'b0;
            do_valid_sync_reg <= 1'b0;
            do_sync_reg <= {N{1'b0}};
        end else begin
            di_sync_reg <= di_i;
            wren_sync_reg <= wren_i;
            di_req_sync_reg <= di_req_o;
            wr_ack_sync_reg <= wr_ack_o;
            do_valid_sync_reg <= do_valid_o;
            do_sync_reg <= do_o;
        end
    end

    // SPI Clock Domain Logic
    always @(posedge sclk_i or posedge rst_i) begin
        if (rst_i) begin
            state <= IDLE;
            shift_reg <= {N{1'b0}};
            rx_data <= {N{1'b0}};
            tx_data <= {N{1'b0}};
            shift_en <= 1'b0;
            bit_count <= 0;
            transfer_complete <= 1'b0;
            ack_received <= 1'b0;
            ssel_active <= 1'b0;
            spi_sck_o <= 1'b0;
            spi_mosi_o <= 1'b0;
            spi_ssel_o <= 1'b1;
            di_req_o <= 1'b0;
            wr_ack_o <= 1'b0;
            do_valid_o <= 1'b0;
            do_o <= {N{1'b0}};
        end else begin
            // State Machine
            case (state)
                IDLE: begin
                    if (wren_sync_reg && !di_req_sync_reg) begin
                        state <= PREPARE;
                        tx_data <= di_sync_reg;
                        shift_en <= 1'b1;
                        bit_count <= 0;
                        transfer_complete <= 1'b0;
                        ack_received <= 1'b0;
                        di_req_o <= 1'b1;
                        spi_ssel_o <= 1'b0;
                        spi_sck_o <= 1'b0;
                        spi_mosi_o <= 1'b0;
                    end
                end

                PREPARE: begin
                    if (wren_sync_reg && di_req_sync_reg) begin
                        state <= TRANSFER;
                        shift_en <= 1'b1;
                        bit_count <= 0;
                        transfer_complete <= 1'b0;
                        ack_received <= 1'b0;
                        di_req_o <= 1'b0;
                        spi_ssel_o <= 1'b0;
                        spi_sck_o <= 1'b0;
                        spi_mosi_o <= 1'b0;
                    end
                end

                TRANSFER: begin
                    if (shift_en) begin
                        if (bit_count < N) begin
                            // Shift out MSB first
                            spi_mosi_o <= tx_data[N-1-bit_count];
                            rx_data[N-1-bit_count] <= spi_miso_i;
                            shift_reg[N-1-bit_count] <= spi_miso_i;
                            bit_count <= bit_count + 1;
                            spi_sck_o <= 1'b1;
                        end else begin
                            // Transfer complete
                            shift_en <= 1'b0;
                            transfer_complete <= 1'b1;
                            spi_sck_o <= 1'b0;
                            if (bit_count == N) begin
                                state <= WAIT_ACK;
                            end
                        end
                    end else begin
                        spi_sck_o <= 1'b0;
                    end
                end

                WAIT_ACK: begin
                    if (ack_received) begin
                        state <= IDLE;
                        spi_ssel_o <= 1'b1;
                        di_req_o <= 1'b0;
                        wr_ack_o <= 1'b1;
                        do_valid_o <= 1'b1;
                        do_o <= shift_reg;
                    end
                end
            endcase
        end
    end

    // Clock domain crossing for SPI signals to parallel domain
    // Using a simple 2-stage synchronizer for safety
    reg [N-1:0] do_sync_stage1;
    reg do_valid_sync_stage1;
    reg do_valid_sync_stage2;
    reg wr_ack_sync_stage1;
    reg wr_ack_sync_stage2;
    reg di_req_sync_stage1;
    reg di_req_sync_stage2;

    always @(posedge pclk_i or posedge rst_i) begin
        if (rst_i) begin
            do_sync_stage1 <= {N{1'b0}};
            do_valid_sync_stage1 <= 1'b0;
            do_valid_sync_stage2 <= 1'b0;
            wr_ack_sync_stage1 <= 1'b0;
            wr_ack_sync_stage2 <= 1'b0;
            di_req_sync_stage1 <= 1'b0;
            di_req_sync_stage2 <= 1'b0;
        end else begin
            do_sync_stage1 <= do_o;
            do_valid_sync_stage1 <= do_valid_o;
            do_valid_sync_stage2 <= do_valid_sync_stage1;
            wr_ack_sync_stage1 <= wr_ack_o;
            wr_ack_sync_stage2 <= wr_ack_sync_stage1;
            di_req_sync_stage1 <= di_req_o;
            di_req_sync_stage2 <= di_req_sync_stage1;
        end
    end

    // Output assignments based on synchronized signals
    assign do_o = do_sync_stage1;
    assign do_valid_o = do_valid_sync_stage2;
    assign wr_ack_o = wr_ack_sync_stage2;
    assign di_req_o = di_req_sync_stage2;

endmodule
