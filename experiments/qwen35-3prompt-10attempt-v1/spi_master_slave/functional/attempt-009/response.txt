module spi_master #(

)(
    input              sclk_i,
    input              pclk_i,
    input              rst_i,

    output             spi_mosi_o,
    input              spi_miso_i,

    output reg [N-1:0] do_o

    reg [N-1:0] shift_reg_next;
    reg [N-1:0] do_data;

    localparam DESELECT = 2'b11;

    reg [N-1:0] rx_data;

            do_o <= 0;

            end

            end

                di_req_o <= 0;
                wr_ack_o <= 1; // Acknowledge write once data is latched and shifting starts
            end

                // Wait for shift to complete
            end
            else begin
                rx_ready <= 1;
                do_valid_o <= rx_valid;
                do_o <= rx_data;
            end
        end
    end

    // SCLK Domain: SPI Timing and Shifting
    // ========================================================================

        if (rst_i) begin
            state <= IDLE;
            shift_reg <= 0;
            shift_reg_next <= 0;
            do_data <= 0;
            do_valid_next <= 0;
            spi_sck_next <= 0;
            spi_ssel_next <= 0;
            spi_mosi_next <= 0;
            di_req_next <= 0;
            wr_ack_next <= 0;

        else begin
            // State Machine Logic
            case (state)
                IDLE: begin
                    if (tx_valid) begin
                        state <= SELECT;
                        shift_reg <= tx_data;
                        tx_valid <= 0;
                        tx_ready <= 1;
                        wr_ack_next <= 0; // Will be set in next cycle or handled via handshake
                    end
                    else begin
                        state <= IDLE;
                    end

                    // Drive SSEL low, prepare to shift
                    spi_ssel_next <= 1'b0;
                    spi_sck_next <= 1'b0;
                    spi_mosi_next <= 1'b0;
                    shift_reg_next <= shift_reg;
                    do_data <= shift_reg;
                    do_valid_next <= 0;
                    di_req_next <= 0;
                    wr_ack_next <= 1; // Acknowledge write completion
                    state <= SHIFT;

                    spi_sck_next <= 1'b1;

            wr_ack_o <= 0;
            do_valid_o <= 0;
            do_o <= 0;
        end

                    shift_count <= 0;
                    shift_done <= 0;
                    shift_count_next <= 0;
                    shift_done_next <= 0;
                    spi_ssel_o <= 1'b0;
                    spi_sck_o <= 1'b0;
                    spi_mosi_o <= 1'b0;

                    wr_ack_o <= 0;

                    spi_mosi_o <= 1'b0;

                        shift_done_next <= 0;
                    end
                end
            end
        end

    always @(posedge sclk_i or posedge rst_i) begin

            shift_reg <= 0;
        end

                shift_reg <= {spi_miso_i, shift_reg[N-1:1]};
            end
            else begin
                shift_reg <= shift_reg;
            end
        end
    end

    assign wr_ack_o = shift_done;
    assign do_valid_o = shift_done;
    assign do_o = shift_reg;

endmodule
