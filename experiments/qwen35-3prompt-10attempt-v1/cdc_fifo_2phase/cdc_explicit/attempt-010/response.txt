```verilog
`timescale 1ns/1ps

module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output reg [WIDTH-1:0] dst_data_o,
    output reg             dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam HALF_DEPTH = DEPTH / 2;

    // Source Domain Signals
    reg [LOG_DEPTH-1:0] src_ptr;
    reg [LOG_DEPTH-1:0] dst_ptr;
    reg [LOG_DEPTH-1:0] gray_src_ptr;
    reg [LOG_DEPTH-1:0] gray_dst_ptr;
    reg [WIDTH*DEPTH-1:0] mem;
    reg [LOG_DEPTH-1:0] empty_gray;
    reg [LOG_DEPTH-1:0] full_gray;
    reg src_empty;
    reg src_full;
    reg src_valid_reg;
    reg dst_valid_reg;

    // Destination Domain Signals
    reg [LOG_DEPTH-1:0] dst_ptr_reg;
    reg [LOG_DEPTH-1:0] dst_ptr_gray;
    reg [LOG_DEPTH-1:0] dst_ptr_gray_reg;
    reg [WIDTH*DEPTH-1:0] mem_reg;
    reg [LOG_DEPTH-1:0] empty_gray_reg;
    reg [LOG_DEPTH-1:0] full_gray_reg;
    reg dst_empty;
    reg dst_full;
    reg dst_valid_reg_reg;

    // ------------------------------------------------------------------
    // Source Domain Logic
    // ------------------------------------------------------------------

    // Reset Synchronizer for Source Reset (Active Low)
    reg src_rst_sync;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_rst_sync <= 1'b1;
        else
            src_rst_sync <= 1'b0;
    end

    // Pointer Logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_ptr <= '0;
            dst_ptr <= '0;
        end else begin
            src_ptr <= src_ptr + 1'b1;
            dst_ptr <= dst_ptr + 1'b1;
        end
    end

    // Gray Code Encoding for Source Pointers
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            gray_src_ptr <= '0;
        else
            gray_src_ptr <= {src_ptr[LOG_DEPTH-2:0], ~src_ptr[LOG_DEPTH-2] ^ src_ptr[LOG_DEPTH-1]};
    end

    // Gray Code Encoding for Destination Pointers
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            gray_dst_ptr <= '0;
        else
            gray_dst_ptr <= {dst_ptr[LOG_DEPTH-2:0], ~dst_ptr[LOG_DEPTH-2] ^ dst_ptr[LOG_DEPTH-1]};
    end

    // Empty/Full Status Calculation (Source Domain)
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_empty <= 1'b1;
            src_full <= 1'b0;
        end else begin
            src_empty <= (gray_src_ptr == gray_dst_ptr);
            src_full <= (gray_src_ptr == (gray_dst_ptr + HALF_DEPTH));
        end
    end

    // Write Enable & Data Storage
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            mem <= '0;
            src_valid_reg <= 1'b0;
        end else if (src_valid_i && !src_full) begin
            mem[src_ptr] <= src_data_i;
            src_valid_reg <= 1'b1;
        end else begin
            src_valid_reg <= 1'b0;
        end
    end

    // Source Ready Output
    assign src_ready_o = !src_full;

    // ------------------------------------------------------------------
    // Destination Domain Logic
    // ------------------------------------------------------------------

    // Reset Synchronizer for Destination Reset (Active Low)
    reg dst_rst_sync;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_rst_sync <= 1'b1;
        else
            dst_rst_sync <= 1'b0;
    end

    // Pointer Logic (Destination Domain)
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_ptr_reg <= '0;
            dst_ptr_gray <= '0;
        end else begin
            dst_ptr_reg <= dst_ptr_reg + 1'b1;
            // Update Gray code based on current binary pointer
            dst_ptr_gray <= {dst_ptr_reg[LOG_DEPTH-2:0], ~dst_ptr_reg[LOG_DEPTH-2] ^ dst_ptr_reg[LOG_DEPTH-1]};
        end
    end

    // Gray Code Encoding for Source Pointers (as seen from Dest)
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            empty_gray_reg <= '0;
        else
            empty_gray_reg <= {src_ptr[LOG_DEPTH-2:0], ~src_ptr[LOG_DEPTH-2] ^ src_ptr[LOG_DEPTH-1]};
    end

    // Gray Code Encoding for Full Status (as seen from Dest)
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            full_gray_reg <= '0;
        else
            full_gray_reg <= {dst_ptr_reg[LOG_DEPTH-2:0], ~dst_ptr_reg[LOG_DEPTH-2] ^ dst_ptr_reg[LOG_DEPTH-1]};
    end

    // Empty/Full Status Calculation (Destination Domain)
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_empty <= 1'b1;
            dst_full <= 1'b0;
        end else begin
            dst_empty <= (empty_gray_reg == dst_ptr_gray);
            dst_full <= (full_gray_reg == dst_ptr_gray);
        end
    end

    // Read Enable & Data Output
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_data_o <= '0;
            dst_valid_reg_reg <= 1'b0;
        end else if (dst_valid_reg_reg && !dst_full) begin
            dst_data_o <= mem_reg[dst_ptr_reg];
            dst_valid_reg_reg <= 1'b1;
        end else begin
            dst_valid_reg_reg <= 1'b0;
        end
    end

    // Read Pointer Update (only when data is consumed)
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_ptr_reg <= '0;
            mem_reg <= '0;
        end else if (dst_valid_reg_reg && !dst_full) begin
            dst_ptr_reg <= dst_ptr_reg + 1'b1;
            mem_reg[dst_ptr_reg] <= mem[dst_ptr_reg]; // Shift data to fill gap
        end else begin
            // No change to pointer or memory on idle/full
        end
    end

    // Destination Valid Output
    assign dst_valid_o = dst_valid_reg_reg;

    // ------------------------------------------------------------------
    // Memory Initialization
    // ------------------------------------------------------------------
    initial begin
        mem = '0;
        mem_reg = '0;
    end

endmodule
