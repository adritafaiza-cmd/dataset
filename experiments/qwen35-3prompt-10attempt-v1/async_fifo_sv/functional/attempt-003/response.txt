module async_fifo #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter SYNC_STAGES = 2
)(
    input  wire                    wclk,
    input  wire                    wrst_n,
    input  wire                    winc,
    input  wire [DATA_WIDTH-1:0]   wdata,
    output wire                    wfull,
    output wire [ADDR_WIDTH:0]     waddr,
    input  wire                    rclk,
    input  wire                    rrst_n,
    input  wire                    rinc,
    output wire [DATA_WIDTH-1:0]   rdata,
    output wire                    rempty,
    output wire [ADDR_WIDTH:0]     raddr
);

    localparam ADDR_MASK = (1 << ADDR_WIDTH) - 1;
    localparam FIFO_DEPTH = 1 << ADDR_WIDTH;

    // Internal pointers: [ADDR_WIDTH-1:0] for address, [ADDR_WIDTH:0] for full/empty logic
    reg [ADDR_WIDTH-1:0] wptr, rptr;
    reg [ADDR_WIDTH-1:0] wptr_sync, rptr_sync;
    reg [ADDR_WIDTH-1:0] wptr_gray, rptr_gray;
    reg [ADDR_WIDTH-1:0] wptr_gray_sync, rptr_gray_sync;
    reg [ADDR_WIDTH-1:0] wptr_gray_sync2, rptr_gray_sync2;

    // Full/Empty flags
    reg wfull;
    reg rempty;

    // Write Domain Logic
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            wptr <= '0;
            wfull <= 1'b0;
        end else begin
            if (winc && !wfull) begin
                wptr <= wptr + 1'b1;
            end
        end
    end

    // Gray Encoder for Write Pointer
    assign wptr_gray = wptr ^ (wptr >> 1);

    // Read Domain Logic
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            rptr <= '0;
            rempty <= 1'b1;
        end else begin
            if (rinc && !rempty) begin
                rptr <= rptr + 1'b1;
            end
        end
    end

    // Gray Encoder for Read Pointer
    assign rptr_gray = rptr ^ (rptr >> 1);

    // Cross-Clock Synchronization: Write Pointer (Read Domain)
    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_sync <= '0;
        end else begin
            wptr_gray_sync <= wptr_gray;
        end
    end

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wptr_gray_sync2 <= '0;
        end else begin
            wptr_gray_sync2 <= wptr_gray_sync;
        end
    end

    // Full Flag Generation (Read Domain)
    // Full when write pointer equals read pointer + 1 (considering wrap-around)
    assign wfull = (wptr_gray_sync2 == (rptr_gray_sync + 1'b1)) ? 1'b1 : 1'b0;

    // Cross-Clock Synchronization: Read Pointer (Write Domain)
    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_sync <= '0;
        end else begin
            rptr_gray_sync <= rptr_gray;
        end
    end

    always @(posedge wclk or negedge wrst_n) begin
        if (!wrst_n) begin
            rptr_gray_sync2 <= '0;
        end else begin
            rptr_gray_sync2 <= rptr_gray_sync;
        end
    end

    // Empty Flag Generation (Write Domain)
    // Empty when write pointer equals read pointer
    assign rempty = (wptr_gray_sync2 == rptr_gray_sync2) ? 1'b1 : 1'b0;

    // Address Outputs (Debug Pointers)
    assign waddr = wptr;
    assign raddr = rptr;

    // Data Output (Read Domain)
    // Use a register to hold the data from the write domain
    reg [DATA_WIDTH-1:0] wdata_sync;
    reg [DATA_WIDTH-1:0] wdata_sync2;

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wdata_sync <= '0;
        end else begin
            wdata_sync <= wdata;
        end
    end

    always @(posedge rclk or negedge rrst_n) begin
        if (!rrst_n) begin
            wdata_sync2 <= '0;
        end else begin
            wdata_sync2 <= wdata_sync;
        end
    end

    assign rdata = wdata_sync2;

endmodule
