module cdc_4phase #(
  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source Domain State Machine
  // States: IDLE, WAIT_DST_READY, WAIT_DST_VALID
  localparam [1:0] SRC_IDLE = 2'b00;
  localparam [1:0] SRC_WAIT_READY = 2'b01;
  localparam [1:0] SRC_WAIT_VALID = 2'b10;

  reg [1:0] src_state;
  reg [WIDTH-1:0] src_data_reg;
  reg src_valid_reg;
  reg src_ready_reg;

  // Destination Domain State Machine
  // States: IDLE, WAIT_SRC_VALID, WAIT_SRC_READY
  localparam [1:0] DST_IDLE = 2'b00;
  localparam [1:0] DST_WAIT_VALID = 2'b01;
  localparam [1:0] DST_WAIT_READY = 2'b10;

  reg [1:0] dst_state;
  reg [WIDTH-1:0] dst_data_reg;
  reg dst_valid_reg;
  reg dst_ready_reg;

  // Source Domain Logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_state <= SRC_IDLE;
      src_data_reg <= '0;
      src_valid_reg <= 1'b0;
      src_ready_reg <= 1'b1;
    end else begin
      case (src_state)
        SRC_IDLE: begin
          src_ready_reg <= 1'b1;
          if (src_valid_i) begin
            src_data_reg <= src_data_i;
            src_valid_reg <= 1'b1;
            src_state <= (DECOUPLED) ? SRC_WAIT_READY : SRC_WAIT_VALID;
          end
        end

        SRC_WAIT_READY: begin
          src_ready_reg <= 1'b1;
          if (dst_ready_i) begin
            src_state <= SRC_IDLE;
            src_valid_reg <= 1'b0;
          end else begin
            // Stay here waiting for dst_ready
          end
        end

        SRC_WAIT_VALID: begin
          src_ready_reg <= 1'b1;
          if (dst_valid_o) begin
            src_state <= SRC_IDLE;
            src_valid_reg <= 1'b0;
          end else begin
            // Stay here waiting for dst_valid
          end
        end
      endcase
    end
  end

  // Destination Domain Logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_state <= DST_IDLE;
      dst_data_reg <= '0;
      dst_valid_reg <= 1'b0;
      dst_ready_reg <= 1'b1;
    end else begin
      case (dst_state)
        DST_IDLE: begin
          dst_ready_reg <= 1'b1;
          if (src_valid_reg) begin
            dst_data_reg <= src_data_reg;
            dst_valid_reg <= 1'b1;
            dst_state <= (DECOUPLED) ? DST_WAIT_READY : DST_WAIT_VALID;
          end
        end

        DST_WAIT_READY: begin
          dst_ready_reg <= 1'b1;
          if (src_ready_reg) begin
            dst_state <= DST_IDLE;
            dst_valid_reg <= 1'b0;
          end else begin
            // Stay here waiting for src_ready
          end
        end

        DST_WAIT_VALID: begin
          dst_ready_reg <= 1'b1;
          if (src_valid_reg) begin
            dst_state <= DST_IDLE;
            dst_valid_reg <= 1'b0;
          end else begin
            // Stay here waiting for src_valid
          end
        end
      endcase
    end
  end

  // Output Assignments
  assign src_ready_o = src_ready_reg;
  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

endmodule<|im_end|>
