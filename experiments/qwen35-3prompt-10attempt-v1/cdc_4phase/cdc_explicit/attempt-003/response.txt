module cdc_4phase #(
  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // ---------------------------------------------------------------------------
  // Source Domain Logic
  // ---------------------------------------------------------------------------
  // State Machine States:
  // IDLE: Waiting for valid
  // GRAB: Valid asserted, waiting for ready (Phase 1)
  // WAIT: Ready asserted, waiting for valid to drop (Phase 2)
  // STABLE: Valid dropped, waiting for ready to drop (Phase 3)
  // RELEASE: Ready dropped, waiting for valid to rise again (Phase 4)

  typedef enum logic [1:0] {
    SRC_IDLE,
    SRC_GRAB,
    SRC_WAIT,
    SRC_STABLE,
    SRC_RELEASE
  } src_state_t;

  src_state_t src_state, src_state_next;
  logic       src_valid_reg;
  logic       src_ready_reg;
  logic       src_valid_d; // Delayed valid for stability check

  // Reset logic for source domain
  logic src_rst_n;
  assign src_rst_n = src_rst_ni;

  // State Register
  always_ff @(posedge src_clk_i or negedge src_rst_n) begin
    if (!src_rst_n) begin
      src_state <= SRC_IDLE;
      src_valid_reg <= 1'b0;
      src_ready_reg <= 1'b0;
    end else begin
      src_state <= src_state_next;
      src_valid_reg <= src_valid_d;
      src_ready_reg <= src_ready_reg; // Hold ready until state changes
    end
  end

  // State Next Logic
  always_comb begin
    src_state_next = src_state;
    src_ready_reg = 1'b0;
    src_valid_d = 1'b0;

    case (src_state)
      SRC_IDLE: begin
        if (src_valid_i) begin
          src_state_next = SRC_GRAB;
          src_ready_reg = 1'b1;
          src_valid_d = 1'b1;
        end
      end

      SRC_GRAB: begin
        if (src_ready_i) begin
          src_state_next = SRC_WAIT;
          src_valid_d = 1'b1;
        end
      end

      SRC_WAIT: begin
        if (!src_valid_i) begin
          src_state_next = SRC_STABLE;
          src_valid_d = 1'b0;
        end
      end

      SRC_STABLE: begin
        if (src_ready_i) begin
          src_state_next = SRC_RELEASE;
          src_valid_d = 1'b0;
        end
      end

      SRC_RELEASE: begin
        if (src_valid_i) begin
          src_state_next = SRC_GRAB;
          src_ready_reg = 1'b1;
          src_valid_d = 1'b1;
        end
      end

      default: begin
        src_state_next = SRC_IDLE;
      end
    endcase
  end

  // Output assignments for Source Domain
  assign src_ready_o = src_ready_reg;

  // ---------------------------------------------------------------------------
  // Destination Domain Logic
  // ---------------------------------------------------------------------------
  // State Machine States:
  // IDLE: Waiting for valid
  // GRAB: Valid asserted, waiting for ready (Phase 1)
  // WAIT: Ready asserted, waiting for valid to drop (Phase 2)
  // STABLE: Valid dropped, waiting for ready to drop (Phase 3)
  // RELEASE: Ready dropped, waiting for valid to rise again (Phase 4)

  typedef enum logic [1:0] {
    DST_IDLE,
    DST_GRAB,
    DST_WAIT,
    DST_STABLE,
    DST_RELEASE
  } dst_state_t;

  dst_state_t dst_state, dst_state_next;
  logic       dst_valid_reg;
  logic       dst_ready_reg;
  logic       dst_valid_d; // Delayed valid for stability check
  logic       dst_data_reg; // Delayed data for stability check
  logic       dst_reset_msg_flag; // Flag to track if reset message has been sent

  // Reset logic for destination domain
  logic dst_rst_n;
  assign dst_rst_n = dst_rst_ni;

  // State Register
  always_ff @(posedge dst_clk_i or negedge dst_rst_n) begin
    if (!dst_rst_n) begin
      dst_state <= DST_IDLE;
      dst_valid_reg <= 1'b0;
      dst_ready_reg <= 1'b0;
      dst_valid_d <= 1'b0;
      dst_data_reg <= RESET_MSG;
      dst_reset_msg_flag <= 0;
    end else begin
      dst_state <= dst_state_next;
      dst_valid_reg <= dst_valid_d;
      dst_ready_reg <= dst_ready_reg; // Hold ready until state changes
      dst_valid_d <= dst_valid_d;
      dst_data_reg <= dst_data_reg;
      dst_reset_msg_flag <= dst_reset_msg_flag;
    end
  end

  // State Next Logic
  always_comb begin
    dst_state_next = dst_state;
    dst_ready_reg = 1'b0;
    dst_valid_d = 1'b0;
    dst_data_reg = RESET_MSG; // Default to reset message on reset or idle if configured

    case (dst_state)
      DST_IDLE: begin
        if (dst_valid_i) begin
          dst_state_next = DST_GRAB;
          dst_ready_reg = 1'b1;
          dst_valid_d = 1'b1;
          dst_data_reg = src_data_i;
        end
      end

      DST_GRAB: begin
        if (dst_ready_i) begin
          dst_state_next = DST_WAIT;
          dst_valid_d = 1'b1;
          dst_data_reg = src_data_i;
        end
      end

      DST_WAIT: begin
        if (!dst_valid_i) begin
          dst_state_next = DST_STABLE;
          dst_valid_d = 1'b0;
        end
      end

      DST_STABLE: begin
        if (dst_ready_i) begin
          dst_state_next = DST_RELEASE;
          dst_valid_d = 1'b0;
        end
      end

      DST_RELEASE: begin
        if (dst_valid_i) begin
          dst_state_next = DST_GRAB;
          dst_ready_reg = 1'b1;
          dst_valid_d = 1'b1;
          dst_data_reg = src_data_i;
        end
      end

      default: begin
        dst_state_next = DST_IDLE;
      end
    endcase
  end

  // Output assignments for Destination Domain
  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

  // ---------------------------------------------------------------------------
  // Reset Message Handling
  // ---------------------------------------------------------------------------
  // If SEND_RESET_MSG is enabled, we need to ensure the reset message is sent
  // exactly once after reset release in the destination domain.
  // We use a flag to track if the reset message has been sent.
  
  always_ff @(posedge dst_clk_i or negedge dst_rst_n) begin
    if (!dst_rst_n) begin
      dst_reset_msg_flag <= 0;
    end else begin
      // If we are in DST_IDLE and the flag is not set, send the reset message
      if (dst_state == DST_IDLE && !dst_reset_msg_flag) begin
        dst_reset_msg_flag <= 1;
        dst_data_reg <= RESET_MSG;
      end
    end
  end

endmodule
