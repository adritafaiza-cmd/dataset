module cdc_4phase #(
  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Sender side (src_clk_i domain)
  // State machine for 4-phase handshake:
  // IDLE -> WAIT_VALID -> WAIT_READY -> IDLE
  // We also handle reset message propagation if enabled.
  
  localparam STATE_IDLE = 2'b00;
  localparam STATE_WAIT_VALID = 2'b01;
  localparam STATE_WAIT_READY = 2'b10;
  
  reg [1:0] src_state;
  reg [WIDTH-1:0] src_msg;
  reg src_valid_reg;
  reg src_ready_reg;
  
  // Receiver side (dst_clk_i domain)
  // State machine for 4-phase handshake:
  // IDLE -> WAIT_VALID -> WAIT_READY -> IDLE
  
  localparam DST_STATE_IDLE = 2'b00;
  localparam DST_STATE_WAIT_VALID = 2'b01;
  localparam DST_STATE_WAIT_READY = 2'b10;
  
  reg [1:0] dst_state;
  reg [WIDTH-1:0] dst_msg;
  reg dst_valid_reg;
  reg dst_ready_reg;
  
  // Internal signals for cross-domain communication
  // src_ready_o is driven by the sender based on receiver's state
  // dst_valid_o is driven by the receiver based on sender's state
  
  // Sender logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_state <= STATE_IDLE;
      src_msg <= RESET_MSG;
      src_valid_reg <= 1'b0;
      src_ready_reg <= 1'b1;
    end else begin
      case (src_state)
        STATE_IDLE: begin
          if (src_valid_i) begin
            src_state <= STATE_WAIT_VALID;
            src_msg <= src_data_i;
          end
          src_ready_reg <= 1'b1;
        end
        
        STATE_WAIT_VALID: begin
          if (src_valid_i) begin
            src_state <= STATE_WAIT_READY;
            src_msg <= src_data_i;
          end
          src_ready_reg <= 1'b0;
        end
        
        STATE_WAIT_READY: begin
          if (src_ready_reg) begin
            src_state <= STATE_IDLE;
            src_ready_reg <= 1'b1;
          end
        end
      endcase
    end
  end
  
  // Receiver logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_state <= DST_STATE_IDLE;
      dst_msg <= RESET_MSG;
      dst_valid_reg <= 1'b0;
      dst_ready_reg <= 1'b1;
    end else begin
      case (dst_state)
        DST_STATE_IDLE: begin
          if (dst_ready_i) begin
            dst_state <= DST_STATE_WAIT_VALID;
            dst_msg <= src_msg; // Capture the message from sender
          end
          dst_valid_reg <= 1'b0;
        end
        
        DST_STATE_WAIT_VALID: begin
          if (dst_ready_i) begin
            dst_state <= DST_STATE_WAIT_READY;
            dst_msg <= src_msg; // Capture the message from sender
          end
          dst_valid_reg <= 1'b0;
        end
        
        DST_STATE_WAIT_READY: begin
          if (dst_ready_i) begin
            dst_state <= DST_STATE_IDLE;
            dst_valid_reg <= 1'b1;
            dst_msg <= src_msg; // Capture the message from sender
          end
        end
      endcase
    end
  end
  
  // Output assignments
  assign src_ready_o = src_ready_reg;
  assign dst_valid_o = dst_valid_reg;
  assign dst_data_o = dst_msg;
  
  // Handle reset message propagation if enabled
  if (SEND_RESET_MSG) begin
    // When in reset state, we want to send the reset message
    // This is handled by the state machine logic above
    // The reset message is captured when the receiver is in IDLE state
    // and the sender sends the reset message
  end
  
endmodule
