module edge_propagator (
  input  logic clk_tx_i,
  input  logic rstn_tx_i,
  input  logic edge_i,
  input  logic clk_rx_i,
  input  logic rstn_rx_i,
  output logic edge_o
);

  // Synchronizer for the input edge signal to the RX clock domain
  // We use a 2-stage synchronizer to filter out metastability
  logic sync_edge_1;
  logic sync_edge_2;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i)
      sync_edge_1 <= 1'b0;
    else
      sync_edge_1 <= edge_i;
  end

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i)
      sync_edge_2 <= 1'b0;
    else
      sync_edge_2 <= sync_edge_1;
  end

  // Edge detection in the RX domain
  // We detect a rising edge on the synchronized signal
  logic prev_sync_edge;
  logic edge_detected;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      prev_sync_edge <= 1'b0;
      edge_detected  <= 1'b0;
    end else begin
      prev_sync_edge <= sync_edge_2;
      // Detect transition from 0 to 1
      edge_detected  <= (sync_edge_2 && !prev_sync_edge);
    end
  end

  // Output the detected edge directly to the RX domain output
  // Since the detection happens in the RX domain, no further CDC is needed for edge_o
  assign edge_o = edge_detected;

endmodule
