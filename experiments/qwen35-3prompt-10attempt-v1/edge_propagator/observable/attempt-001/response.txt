module edge_propagator (
  input  logic clk_tx_i,
  input  logic rstn_tx_i,
  input  logic edge_i,
  input  logic clk_rx_i,
  input  logic rstn_rx_i,
  output logic edge_o
);

  // Synchronizer for the input event from the transmit clock domain
  // Uses a 2-stage synchronizer to filter out metastability
  logic sync_edge_1;
  logic sync_edge_2;

  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i) begin
      sync_edge_1 <= 1'b0;
      sync_edge_2 <= 1'b0;
    end else begin
      sync_edge_1 <= edge_i;
      sync_edge_2 <= sync_edge_1;
    end
  end

  // Pulse generator to create a single-cycle pulse on the transmit side
  // This ensures the event is captured as a single event without duplication
  logic tx_pulse;

  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i) begin
      tx_pulse <= 1'b0;
    end else begin
      if (sync_edge_2 && !sync_edge_1) begin
        tx_pulse <= 1'b1;
      end else begin
        tx_pulse <= 1'b0;
      end
    end
  end

  // Clock domain crossing from transmit to receive
  // Synchronize the pulse signal to the receive clock domain
  logic rx_sync_pulse_1;
  logic rx_sync_pulse_2;

  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      rx_sync_pulse_1 <= 1'b0;
      rx_sync_pulse_2 <= 1'b0;
    end else begin
      rx_sync_pulse_1 <= tx_pulse;
      rx_sync_pulse_2 <= rx_sync_pulse_1;
    end
  end

  // Generate the output pulse on the receive side
  // Ensure the output is a single-cycle pulse and reset properly
  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      edge_o <= 1'b0;
    end else begin
      if (rx_sync_pulse_2 && !rx_sync_pulse_1) begin
        edge_o <= 1'b1;
      end else begin
        edge_o <= 1'b0;
      end
    end
  end

endmodule
