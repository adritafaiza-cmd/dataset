module edge_propagator (
  input  logic clk_tx_i,
  input  logic rstn_tx_i,
  input  logic edge_i,
  input  logic clk_rx_i,
  input  logic rstn_rx_i,
  output logic edge_o
);

  // State to track the last value of edge_i in the TX domain
  logic tx_edge_last;
  
  // State to track the last value of edge_o in the RX domain
  logic rx_edge_last;

  // TX Domain: Detect edges on edge_i
  // We sample edge_i and detect transitions.
  // On a rising edge of edge_i (or falling if we want to catch both, 
  // but typically "edge" implies rising unless specified otherwise. 
  // However, to catch any pulse/edge reliably without double counting, 
  // we usually detect the transition from 0->1 and then wait for 1->0 to reset the flag).
  // To ensure we catch a single pulse regardless of width, we detect the rising edge.
  // If the requirement is to catch both rising and falling edges as separate events, 
  // the logic would differ. Given "pulse or edge", detecting the start of the event (rising) 
  // and ensuring we don't re-trigger until the signal returns to low is the standard approach 
  // for single-shot propagation.
  
  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i) begin
      tx_edge_last <= 1'b0;
    end else begin
      // Detect rising edge
      if (edge_i && !tx_edge_last) begin
        tx_edge_last <= 1'b1;
      end else begin
        // If edge_i goes low, we are ready for the next edge
        tx_edge_last <= 1'b0;
      end
    end
  end

  // RX Domain: Propagate the detected edge to the output
  // We need to latch the 'tx_edge_last' signal across the clock domain boundary.
  // Since we only care about the transition, we can use a simple flip-flop chain 
  // or a single register if we are careful about metastability and timing.
  // A single register is sufficient if we assume the TX domain logic stabilizes 
  // before the RX clock samples it, or if we use a multi-stage synchronizer.
  // For robustness against asynchronous clocks, a 2-stage synchronizer is standard practice.
  
  logic sync_edge_1;
  logic sync_edge_2;

  // Synchronizer for the TX edge detection signal
  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      sync_edge_1 <= 1'b0;
      sync_edge_2 <= 1'b0;
    end else begin
      sync_edge_1 <= tx_edge_last;
      sync_edge_2 <= sync_edge_1;
    end
  end

  // Generate the output pulse
  // We detect the rising edge of the synchronized signal.
  // Since sync_edge_2 is the stable version of the TX edge, we detect its transition.
  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      edge_o <= 1'b0;
    end else begin
      // Detect rising edge of the synchronized signal
      if (sync_edge_2 && !sync_edge_1) begin
        edge_o <= 1'b1;
      end else begin
        edge_o <= 1'b0;
      end
    end
  end

endmodule
