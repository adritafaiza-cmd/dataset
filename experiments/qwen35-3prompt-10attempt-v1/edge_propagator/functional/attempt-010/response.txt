module edge_propagator (
  input  logic clk_tx_i,
  input  logic rstn_tx_i,
  input  logic edge_i,
  input  logic clk_rx_i,
  input  logic rstn_rx_i,
  output logic edge_o
);

  // State to track the last value of edge_i sampled in the TX domain
  logic tx_edge_prev;
  
  // State to track the last value of edge_o sampled in the RX domain
  logic rx_edge_prev;

  // TX Domain: Sample the input edge and detect transitions
  always_ff @(posedge clk_tx_i or negedge rstn_tx_i) begin
    if (!rstn_tx_i) begin
      tx_edge_prev <= 1'b0;
    end else begin
      tx_edge_prev <= edge_i;
    end
  end

  // RX Domain: Detect transitions from the TX domain and propagate a single pulse
  always_ff @(posedge clk_rx_i or negedge rstn_rx_i) begin
    if (!rstn_rx_i) begin
      rx_edge_prev <= 1'b0;
      edge_o       <= 1'b0;
    end else begin
      // Detect a rising edge in the TX domain
      if (tx_edge_prev == 1'b0 && edge_i == 1'b1) begin
        // Rising edge detected in TX: assert pulse in RX
        edge_o <= 1'b1;
        // Update RX state to remember we just sent a pulse
        rx_edge_prev <= 1'b1;
      end else if (tx_edge_prev == 1'b1 && edge_i == 1'b0) begin
        // Falling edge detected in TX: de-assert pulse in RX
        edge_o <= 1'b0;
        // Update RX state to remember we just cleared the pulse
        rx_edge_prev <= 1'b0;
      end else begin
        // No edge detected in TX, maintain current RX state
        edge_o       <= edge_o;
        rx_edge_prev <= rx_edge_prev;
      end
    end
  end

endmodule
