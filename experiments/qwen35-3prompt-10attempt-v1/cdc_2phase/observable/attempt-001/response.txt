module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source side: Capture valid/ready handshake and data
  logic src_valid_reg;
  logic src_ready_reg;
  logic [WIDTH-1:0] src_data_reg;
  logic src_valid_prev;
  logic src_ready_prev;

  // Destination side: Capture valid/ready handshake and data
  logic dst_valid_reg;
  logic dst_ready_reg;
  logic [WIDTH-1:0] dst_data_reg;
  logic dst_valid_prev;
  logic dst_ready_prev;

  // Two-phase handshake state machine for source side
  // State 0: Idle
  // State 1: Waiting for dst_ready (data latched, valid asserted)
  logic [1:0] src_state;
  localparam SRC_IDLE = 2'b00;
  localparam SRC_WAIT = 2'b01;

  // Two-phase handshake state machine for destination side
  // State 0: Idle
  // State 1: Waiting for src_valid (data latched, ready asserted)
  logic [1:0] dst_state;
  localparam DST_IDLE = 2'b00;
  localparam DST_WAIT = 2'b01;

  // Source logic
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_valid_reg <= 1'b0;
      src_ready_reg <= 1'b1;
      src_data_reg  <= 1'b0;
      src_state     <= SRC_IDLE;
    end else begin
      // Capture previous values for two-phase detection
      src_valid_prev <= src_valid_reg;
      src_ready_prev <= src_ready_reg;

      case (src_state)
        SRC_IDLE: begin
          if (src_valid_i && dst_ready_reg) begin
            src_valid_reg <= 1'b1;
            src_data_reg  <= src_data_i;
            src_state     <= SRC_WAIT;
          end else begin
            src_valid_reg <= 1'b0;
            src_ready_reg <= 1'b1;
          end
        end
        SRC_WAIT: begin
          if (dst_ready_reg) begin
            src_valid_reg <= 1'b0;
            src_ready_reg <= 1'b1;
            src_state     <= SRC_IDLE;
          end else begin
            src_valid_reg <= 1'b1;
            src_ready_reg <= 1'b0;
          end
        end
      endcase
    end
  end

  // Destination logic
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid_reg <= 1'b0;
      dst_ready_reg <= 1'b1;
      dst_data_reg  <= 1'b0;
      dst_state     <= DST_IDLE;
    end else begin
      // Capture previous values for two-phase detection
      dst_valid_prev <= dst_valid_reg;
      dst_ready_prev <= dst_ready_reg;

      case (dst_state)
        DST_IDLE: begin
          if (src_valid_reg && dst_ready_i) begin
            dst_valid_reg <= 1'b1;
            dst_data_reg  <= src_data_reg;
            dst_state     <= DST_WAIT;
          end else begin
            dst_valid_reg <= 1'b0;
            dst_ready_reg <= 1'b1;
          end
        end
        DST_WAIT: begin
          if (src_valid_reg) begin
            dst_valid_reg <= 1'b0;
            dst_ready_reg <= 1'b0;
            dst_state     <= DST_IDLE;
          end else begin
            dst_valid_reg <= 1'b1;
            dst_ready_reg <= 1'b0;
          end
        end
      endcase
    end
  end

  // Output assignments
  assign src_ready_o = src_ready_reg;
  assign dst_valid_o = dst_valid_reg;
  assign dst_data_o  = dst_data_reg;

endmodule
