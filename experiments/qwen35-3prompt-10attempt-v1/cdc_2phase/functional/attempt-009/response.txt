module cdc_2phase #(
  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Source domain state
  logic src_state_idle, src_state_busy;
  logic src_data_reg;
  logic src_valid_reg;
  logic src_ready_reg;

  // Destination domain state
  logic dst_state_idle, dst_state_busy;
  logic dst_data_reg;
  logic dst_valid_reg;
  logic dst_ready_reg;

  // Generate synchronous logic for source domain
  always_ff @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_state_idle <= 1'b1;
      src_state_busy <= 1'b0;
      src_data_reg <= '0;
      src_valid_reg <= 1'b0;
      src_ready_reg <= 1'b1;
    end else begin
      src_state_idle <= ~src_state_busy;
      src_state_busy <= ~src_state_idle;

      if (src_state_idle) begin
        src_data_reg <= src_data_i;
        src_valid_reg <= src_valid_i;
        src_ready_reg <= 1'b1;
      end else begin
        src_data_reg <= '0;
        src_valid_reg <= 1'b0;
        src_ready_reg <= 1'b0;
      end
    end
  end

  // Generate synchronous logic for destination domain
  always_ff @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_state_idle <= 1'b1;
      dst_state_busy <= 1'b0;
      dst_data_reg <= '0;
      dst_valid_reg <= 1'b0;
      dst_ready_reg <= 1'b1;
    end else begin
      dst_state_idle <= ~dst_state_busy;
      dst_state_busy <= ~dst_state_idle;

      if (dst_state_idle) begin
        dst_data_reg <= src_data_reg;
        dst_valid_reg <= src_valid_reg;
        dst_ready_reg <= 1'b1;
      end else begin
        dst_data_reg <= '0;
        dst_valid_reg <= 1'b0;
        dst_ready_reg <= 1'b0;
      end
    end
  end

  // Assign output ports
  assign src_ready_o = src_ready_reg;
  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

endmodule
