            src_ready_o <= 1'b0;
          end else if (src_valid_i) begin

            src_ready_o <= 1'b0;

            src_ready_o <= 1'b1;

          // Revised Logic:
          // IDLE: Ready to accept.
          // WAIT_READY: Have data, waiting for dst_ready.
