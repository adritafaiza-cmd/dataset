```verilog
module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output reg [WIDTH-1:0] dst_data_o,
    output reg             dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam GRAY_WIDTH = $clog2(DEPTH);

    // Source Domain Signals
    reg [LOG_DEPTH-1:0] src_write_ptr;
    reg [LOG_DEPTH-1:0] src_read_ptr;
    reg [LOG_DEPTH-1:0] src_gray_ptr;
    reg src_write_en;
    reg src_full;
    reg [DEPTH-1:0] src_full_reg;

    // Destination Domain Signals
    reg [LOG_DEPTH-1:0] dst_read_ptr;
    reg [LOG_DEPTH-1:0] dst_write_ptr; // Used as read pointer in dest domain
    reg [LOG_DEPTH-1:0] dst_gray_ptr;
    reg dst_read_en;
    reg dst_empty;
    reg [DEPTH-1:0] dst_empty_reg;

    // Cross-domain pointers (Gray encoded for stability)
    reg [LOG_DEPTH-1:0] src_gray_write_ptr;
    reg [LOG_DEPTH-1:0] dst_gray_read_ptr;

    // Data Storage
    reg [WIDTH-1:0] mem [DEPTH-1:0];

    // Synchronizers for control signals
    reg [SYNC_STAGES-1:0] src_valid_sync;
    reg [SYNC_STAGES-1:0] dst_ready_sync;
    reg [SYNC_STAGES-1:0] src_rst_sync;
    reg [SYNC_STAGES-1:0] dst_rst_sync;

    // ---------------------------------------------------------
    // Source Domain Logic
    // ---------------------------------------------------------
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_write_ptr <= '0;
            src_read_ptr  <= '0;
            src_gray_ptr  <= '0;
            src_full      <= 1'b0;
            src_full_reg  <= '0;
            src_write_en  <= 1'b0;
        end else begin
            src_write_ptr <= src_write_ptr + src_write_en;
            src_read_ptr  <= src_read_ptr + dst_read_en;
            
            // Calculate Gray pointer from binary write pointer
            src_gray_ptr <= src_write_ptr ^ (src_write_ptr >> 1);
            
            // Full detection: (write_ptr - read_ptr) == DEPTH - 1
            // Using subtraction to avoid overflow issues with large depths
            if (src_write_en && !src_full) begin
                if (src_write_ptr == src_read_ptr) begin
                    // FIFO is full
                    src_full <= 1'b1;
                end else begin
                    src_full <= 1'b0;
                end
            end
            
            // Register full signal for stability
            src_full_reg <= src_full;
        end
    end

    // ---------------------------------------------------------
    // Destination Domain Logic
    // ---------------------------------------------------------
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_read_ptr <= '0;
            dst_write_ptr <= '0; // Alias for read pointer in dest domain
            dst_gray_ptr <= '0;
            dst_empty    <= 1'b1;
            dst_empty_reg <= '0;
            dst_read_en  <= 1'b0;
        end else begin
            dst_read_ptr <= dst_read_ptr + dst_read_en;
            dst_write_ptr <= dst_write_ptr + src_write_en; // Alias for read pointer in dest domain
            
            // Calculate Gray pointer from binary read pointer
            dst_gray_ptr <= dst_read_ptr ^ (dst_read_ptr >> 1);
            
            // Empty detection: (read_ptr - write_ptr) == 0
            if (dst_read_en && !dst_empty) begin
                if (dst_read_ptr == dst_write_ptr) begin
                    // FIFO is empty
                    dst_empty <= 1'b1;
                end else begin
                    dst_empty <= 1'b0;
                end
            end
            
            // Register empty signal for stability
            dst_empty_reg <= dst_empty;
        end
    end

    // ---------------------------------------------------------
    // Cross-Clock Pointers (Gray Encoding)
    // ---------------------------------------------------------
    // Source domain generates Gray pointer for destination
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_gray_write_ptr <= '0;
        else
            src_gray_write_ptr <= src_gray_ptr;
    end

    // Destination domain generates Gray pointer for source
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_gray_read_ptr <= '0;
        else
            dst_gray_read_ptr <= dst_gray_ptr;
    end

    // ---------------------------------------------------------
    // Synchronizers
    // ---------------------------------------------------------
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_valid_sync <= {SYNC_STAGES{1'b0}};
        else
            src_valid_sync <= {src_valid_sync[SYNC_STAGES-2:0], src_valid_i};
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_ready_sync <= {SYNC_STAGES{1'b0}};
        else
            dst_ready_sync <= {dst_ready_sync[SYNC_STAGES-2:0], dst_ready_i};
    end

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_rst_sync <= {SYNC_STAGES{1'b0}};
        else
            src_rst_sync <= {src_rst_sync[SYNC_STAGES-2:0], src_rst_ni};
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_rst_sync <= {SYNC_STAGES{1'b0}};
        else
            dst_rst_sync <= {dst_rst_sync[SYNC_STAGES-2:0], dst_rst_ni};
    end

    // ---------------------------------------------------------
    // Control Logic
    // ---------------------------------------------------------
    // Source Ready: Not full AND valid signal synchronized
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_ready_o <= 1'b0;
        else
            src_ready_o <= ~src_full_reg && src_valid_sync[SYNC_STAGES-1];
    end

    // Destination Valid: Not empty AND ready signal synchronized
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_valid_o <= 1'b0;
        else
            dst_valid_o <= ~dst_empty_reg && dst_ready_sync[SYNC_STAGES-1];
    end

    // ---------------------------------------------------------
    // Write Enable (Source Domain)
    // ---------------------------------------------------------
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_write_en <= 1'b0;
        else
            src_write_en <= src_valid_i && src_ready_o;
    end

    // ---------------------------------------------------------
    // Read Enable (Destination Domain)
    // ---------------------------------------------------------
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_read_en <= 1'b0;
        else
            dst_read_en <= dst_valid_o;
    end

    // ---------------------------------------------------------
    // Data Output (Destination Domain)
    // ---------------------------------------------------------
    // Read data from memory using the synchronized read pointer
    // The pointer is updated on dst_clk_i, so we use the current value of dst_read_ptr
    // to index into the memory.
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_data_o <= '0;
            dst_valid_o <= 1'b0;
        end else begin
            if (dst_valid_o) begin
                dst_data_o <= mem[dst_read_ptr];
            end else begin
                dst_data_o <= '0;
            end
        end
    end

    // ---------------------------------------------------------
    // Memory Write (Source Domain)
    // ---------------------------------------------------------
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            mem <= '0;
        end else if (src_write_en) begin
            mem[src_write_ptr] <= src_data_i;
        end
    end

endmodule
