```verilog
`timescale 1ns/1ps

module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output reg [WIDTH-1:0] dst_data_o,
    output reg             dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam GRAY_WIDTH = $clog2(DEPTH);

    // --- Source Side (Write) ---
    reg [LOG_DEPTH-1:0] write_ptr;
    reg [LOG_DEPTH-1:0] read_ptr;
    reg [WIDTH-1:0]     data_mem [DEPTH-1:0];
    reg                 write_en;
    reg                 read_en;
    reg                 full;
    reg                 empty;
    reg [LOG_DEPTH-1:0] write_gray;
    reg [LOG_DEPTH-1:0] read_gray;

    // --- Destination Side (Read) ---
    // We need to capture the write pointer from the source domain to determine if data is available
    // and to handle the "lag" requirement for dst_valid_o.
    wire [LOG_DEPTH-1:0] src_write_ptr_captured;
    wire [LOG_DEPTH-1:0] src_read_ptr_captured;
    wire                 src_full_captured;
    wire                 src_empty_captured;

    // --- CDC Logic: Gray Encoding for Pointers ---
    // Source Domain: Generate Gray encoded pointers
    assign write_gray = write_ptr ^ (write_ptr >> 1);
    assign read_gray  = read_ptr ^ (read_ptr >> 1);

    // Destination Domain: Decode Gray pointers back to binary
    // Since the pointers are captured asynchronously, we decode them here.
    // Note: For small depths, direct decoding is fine. For larger, a dedicated decoder might be used,
    // but standard binary-to-gray and gray-to-binary logic works for the pointer values themselves.
    // However, the standard approach for FIFOs is to keep pointers in binary on each side
    // and only cross the boundary using Gray encoding.
    // Here, we capture the Gray pointers from the other domain.
    
    // Capture Write Pointer (Gray) from Source to Dest
    reg [LOG_DEPTH-1:0] w_ptr_gray_stage1;
    reg [LOG_DEPTH-1:0] w_ptr_gray_stage2;
    reg [LOG_DEPTH-1:0] w_ptr_gray_stage3;
    reg [LOG_DEPTH-1:0] w_ptr_gray_stage4;
    
    // Capture Read Pointer (Gray) from Source to Dest
    reg [LOG_DEPTH-1:0] r_ptr_gray_stage1;
    reg [LOG_DEPTH-1:0] r_ptr_gray_stage2;
    reg [LOG_DEPTH-1:0] r_ptr_gray_stage3;
    reg [LOG_DEPTH-1:0] r_ptr_gray_stage4;

    // Capture Full/Empty flags from Source to Dest
    reg                 full_stage1;
    reg                 full_stage2;
    reg                 full_stage3;
    reg                 full_stage4;
    
    reg                 empty_stage1;
    reg                 empty_stage2;
    reg                 empty_stage3;
    reg                 empty_stage4;

    // --- CDC Flip-Flops for Pointers (Source -> Dest) ---
    // Write Pointer
    always @(posedge dst_clk_i or posedge dst_rst_ni) begin
        if (dst_rst_ni) begin
            w_ptr_gray_stage1 <= {LOG_DEPTH{1'b0}};
            w_ptr_gray_stage2 <= {LOG_DEPTH{1'b0}};
            w_ptr_gray_stage3 <= {LOG_DEPTH{1'b0}};
            w_ptr_gray_stage4 <= {LOG_DEPTH{1'b0}};
        end else begin
            w_ptr_gray_stage1 <= write_gray;
            w_ptr_gray_stage2 <= w_ptr_gray_stage1;
            w_ptr_gray_stage3 <= w_ptr_gray_stage2;
            w_ptr_gray_stage4 <= w_ptr_gray_stage3;
        end
    end

    // Read Pointer
    always @(posedge dst_clk_i or posedge dst_rst_ni) begin
        if (dst_rst_ni) begin
            r_ptr_gray_stage1 <= {LOG_DEPTH{1'b0}};
            r_ptr_gray_stage2 <= {LOG_DEPTH{1'b0}};
            r_ptr_gray_stage3 <= {LOG_DEPTH{1'b0}};
            r_ptr_gray_stage4 <= {LOG_DEPTH{1'b0}};
        end else begin
            r_ptr_gray_stage1 <= read_gray;
            r_ptr_gray_stage2 <= r_ptr_gray_stage1;
            r_ptr_gray_stage3 <= r_ptr_gray_stage2;
            r_ptr_gray_stage4 <= r_ptr_gray_stage3;
        end
    end

    // --- CDC Flip-Flops for Status (Source -> Dest) ---
    always @(posedge dst_clk_i or posedge dst_rst_ni) begin
        if (dst_rst_ni) begin
            full_stage1 <= 1'b0;
            full_stage2 <= 1'b0;
            full_stage3 <= 1'b0;
            full_stage4 <= 1'b0;
            empty_stage1 <= 1'b1;
            empty_stage2 <= 1'b1;
            empty_stage3 <= 1'b1;
            empty_stage4 <= 1'b1;
        end else begin
            full_stage1 <= full;
            full_stage2 <= full_stage1;
            full_stage3 <= full_stage2;
            full_stage4 <= full_stage3;
            empty_stage1 <= empty;
            empty_stage2 <= empty_stage1;
            empty_stage3 <= empty_stage2;
            empty_stage4 <= empty_stage3;
        end
    end

    // --- Gray to Binary Decoder for Captured Pointers ---
    // We need to convert the captured Gray pointers back to binary to compare with local read_ptr
    // and to determine the actual data to output.
    // Since the pointers are captured with SYNC_STAGES, we use the final stage (stage4) for logic.
    
    // Helper function to convert Gray to Binary
    // bin[0] = gray[0]
    // bin[i] = gray[i] ^ bin[i-1]
    
    // Decode Write Pointer (Captured)
    reg [LOG_DEPTH-1:0] w_ptr_bin;
    always @(*) begin
        w_ptr_bin = 0;
        for (integer i = 0; i < LOG_DEPTH; i = i + 1) begin
            if (i == 0)
                w_ptr_bin[i] = w_ptr_gray_stage4[i];
            else
                w_ptr_bin[i] = w_ptr_gray_stage4[i] ^ w_ptr_bin[i-1];
        end
    end

    // Decode Read Pointer (Captured)
    reg [LOG_DEPTH-1:0] r_ptr_bin;
    always @(*) begin
        r_ptr_bin = 0;
        for (integer i = 0; i < LOG_DEPTH; i = i + 1) begin
            if (i == 0)
                r_ptr_bin[i] = r_ptr_gray_stage4[i];
            else
                r_ptr_bin[i] = r_ptr_gray_stage4[i] ^ r_ptr_bin[i-1];
        end
    end

    // --- Source Domain Logic ---
    always @(posedge src_clk_i or posedge src_rst_ni) begin
        if (src_rst_ni) begin
            write_ptr <= {LOG_DEPTH{1'b0}};
            read_ptr  <= {LOG_DEPTH{1'b0}};
            full      <= 1'b0;
            empty     <= 1'b1;
            src_ready_o <= 1'b0;
        end else begin
            // Calculate Full/Empty
            // Full: (write_ptr + 1) == read_ptr
            // Empty: write_ptr == read_ptr
            if (write_ptr == read_ptr)
                empty <= 1'b1;
            else
                empty <= 1'b0;

            if (write_ptr == (DEPTH - 1) && read_ptr == 0)
                full <= 1'b1;
            else if (write_ptr == (DEPTH - 1) && read_ptr == DEPTH - 1)
                full <= 1'b0; // This case shouldn't happen if logic is correct, but for safety
            else
                full <= (write_ptr + 1) == read_ptr;

            // Write Enable
            write_en = src_valid_i && !full;
            
            // Update Pointers
            if (write_en)
                write_ptr <= write_ptr + 1;
            
            // Flow Control
            src_ready_o = !full;
        end
    end

    // --- Destination Domain Logic ---
    always @(posedge dst_clk_i or posedge dst_rst_ni) begin
        if (dst_rst_ni) begin
            dst_valid_o <= 1'b0;
            dst_data_o  <= {WIDTH{1'b0}};
            read_en     <= 1'b0;
        end else begin
            // Determine if we can read
            // We can read if the captured write pointer is ahead of the captured read pointer
            // AND the status flags indicate data is available.
            // To handle the "lag" requirement: dst_valid_o may fall for one clock after accepted beat.
            // This is naturally handled if we only assert valid when we are sure data is there.
            // However, the prompt says: "dst_valid_o may fall for one destination clock after an accepted beat even when another word is stored."
            // This implies we should be greedy: if there is data, assert valid.
            
            // Check if data is available based on captured pointers
            // Available if w_ptr_bin > r_ptr_bin
            // But we must also respect the full/empty flags captured from source to avoid glitches
            // during pointer transitions.
            
            // Logic: 
            // 1. If empty_captured is high, no data.
            // 2. If full_captured is high, data is available (unless we are at the very edge of wrap around which full handles).
            // 3. Pointer comparison: w_ptr_bin > r_ptr_bin.
            
            // A robust way to determine validity in the destination domain:
            // valid = !empty_captured && (w_ptr_bin > r_ptr_bin)
            // However, due to asynchronous nature, w_ptr_bin and r_ptr_bin might be momentarily inconsistent.
            // Using the captured full/empty flags is usually safer for the "ready" signal, 
            // but for "valid", we rely on the pointer comparison.
            
            // Let's use the captured status flags to drive the enable, and pointer comparison to drive valid?
            // No, standard design:
            // dst_valid_o = !empty_captured && (w_ptr_bin > r_ptr_bin)
            // But we need to ensure we don't output garbage when pointers are transitioning.
            // The "lag" behavior mentioned suggests we should assert valid as soon as data is present.
            
            // Refined Logic:
            // If the source says it's not empty (empty_captured == 0), and the captured write pointer
            // is strictly greater than the captured read pointer, then data is valid.
            
            // To prevent metastability on the comparison result, we can add a small delay or 
            // rely on the fact that pointers only change by 1.
            // Given SYNC_STAGES=2, the probability of mismatch is low, but we can add a safety check.
            
            // Let's implement the standard valid logic:
            // valid = (!empty_captured) && (w_ptr_bin > r_ptr_bin)
            // However, to strictly follow "dst_valid_o may fall for one destination clock after an accepted beat",
            // we should assert valid whenever data is available.
            
            // One nuance: If w_ptr_bin == r_ptr_bin, it means empty.
            // If w_ptr_bin > r_ptr_bin, it means full or partially full.
            
            // We will use the captured 'full' flag to gate the pointer comparison to avoid glitches
            // when the FIFO transitions from empty to full or vice versa.
            // Actually, the simplest robust method is:
            // dst_valid_o = !empty_captured && (w_ptr_bin > r_ptr_bin);
            // But we must ensure w_ptr_bin and r_ptr_bin are stable.
            // With 4 stages, they are quite stable.
            
            // Let's add a small debounce or just rely on the logic.
            // To handle the specific requirement about valid falling:
            // We assert valid if data is present. If the source accepts a beat, it increments write_ptr.
            // The destination sees the new write_ptr in the next cycle(s).
            
            // Implementation:
            if (empty_stage4) begin
                dst_valid_o <= 1'b0;
                dst_data_o  <= {WIDTH{1'b0}};
                read_en     <= 1'b0;
            end else begin
                // Check pointer difference
                // Since we are in binary domain now (decoded from gray), we can compare directly.
                // We need to be careful about the wrap-around.
                // w_ptr_bin > r_ptr_bin covers all cases where data exists.
                
                // To be extra safe against transient glitches in the decoded pointers:
                // We can check if the difference is at least 1.
                // If w_ptr_bin > r_ptr_bin, then valid.
                
                // However, there is a known issue: if w_ptr_bin and r_ptr_bin are decoded from 
                // asynchronously captured gray codes, they might momentarily be equal when they shouldn't be,
                // or one might be ahead.
                // A common fix is to use the captured 'full' flag to inhibit valid when transitioning to empty.
                
                // Let's use a slightly more conservative valid logic to ensure stability:
                // valid = (!empty_stage4) && (w_ptr_bin > r_ptr_bin)
                // But we also need to consider the 'full' flag. If full_stage4 is 1, we definitely have data.
                // If full_stage4 is 0, we might still have data if w_ptr > r_ptr.
                
                // Let's stick to the pointer comparison as the primary source of truth for 'valid',
                // gated by !empty_stage4.
                
                if (w_ptr_bin > r_ptr_bin) begin
                    dst_valid_o <= 1'b1;
                    dst_data_o  <= data_mem[r_ptr_bin];
                    read_en     <= 1'b1;
                end else begin
                    dst_valid_o <= 1'b0;
                    dst_data_o  <= {WIDTH{1'b0}};
                    read_en     <= 1'b0;
                end
            end
        end
    end

    // --- Read Pointer Update (Destination Domain) ---
    always @(posedge dst_clk_i or posedge dst_rst_ni) begin
        if (dst_rst_ni) begin
            read_ptr <= {LOG_DEPTH{1'b0}};
        end else begin
            if (read_en)
                read_ptr <= read_ptr + 1;
        end
    end

    // --- Data Memory Write (Source Domain) ---
    always @(posedge src_clk_i or posedge src_rst_ni) begin
        if (src_rst_ni) begin
            data_mem <= {DEPTH{1'b0}};
        end else if (write_en) begin
            data_mem[write_ptr] <= src_data_i;
        end
    end

endmodule
