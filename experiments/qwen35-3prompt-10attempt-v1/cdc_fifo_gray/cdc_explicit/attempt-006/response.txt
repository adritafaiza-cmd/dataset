`timescale 1ns/1ps

module cdc_fifo_gray #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3,
    parameter SYNC_STAGES = 2
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output reg [WIDTH-1:0] dst_data_o,
    output reg             dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;
    localparam GRAY_WIDTH = LOG_DEPTH;

    // Internal registers for pointers and data
    reg [LOG_DEPTH-1:0] src_ptr;
    reg [LOG_DEPTH-1:0] dst_ptr;
    reg [WIDTH-1:0]     fifo_mem [DEPTH-1:0];
    
    // Gray encoded pointers for CDC
    reg [GRAY_WIDTH-1:0] src_gray_ptr;
    reg [GRAY_WIDTH-1:0] dst_gray_ptr;
    
    // Synchronizers for pointers
    reg [GRAY_WIDTH-1:0] src_gray_synced [SYNC_STAGES-1:0];
    reg [GRAY_WIDTH-1:0] dst_gray_synced [SYNC_STAGES-1:0];
    
    // Synchronizers for ready/valid signals
    reg src_valid_synced [SYNC_STAGES-1:0];
    reg dst_ready_synced [SYNC_STAGES-1:0];
    
    // Synchronizers for reset signals
    reg src_rst_synced;
    reg dst_rst_synced;

    // ---------------------------------------------------------
    // Source Domain Logic
    // ---------------------------------------------------------
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_ptr <= '0;
            src_valid_synced <= '0;
            src_rst_synced <= 1'b1;
        end else begin
            src_rst_synced <= 1'b0;
            
            // Shift synchronizers for src_valid
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                src_valid_synced[i] <= src_valid_synced[i-1];
            end
            src_valid_synced[0] <= src_valid_i;
            
            // Shift synchronizers for dst_ready
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                // We need to know if dst is ready to accept data
                // Since dst_ready is an input, we sync it to src domain to decide when to push
                // However, standard async FIFO usually syncs the pointer crossing.
                // To implement src_ready correctly, we need to know if the destination can take data.
                // We sync dst_ready to src domain.
                src_ready_o <= src_valid_synced[0] & dst_ready_synced[0];
                
                dst_ready_synced[i] <= dst_ready_synced[i-1];
            end
            dst_ready_synced[0] <= dst_ready_i;
            
            // Update pointer if valid and ready
            if (src_valid_synced[0] & dst_ready_synced[0]) begin
                src_ptr <= src_ptr + 1'b1;
                fifo_mem[src_ptr] <= src_data_i;
            end
        end
    end

    // ---------------------------------------------------------
    // Destination Domain Logic
    // ---------------------------------------------------------
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_ptr <= '0;
            dst_valid_o <= 1'b0;
            dst_data_o <= '0;
            dst_gray_synced <= '0;
            dst_rst_synced <= 1'b1;
        end else begin
            dst_rst_synced <= 1'b0;
            
            // Shift synchronizers for src_valid (to know if FIFO is full)
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                src_valid_synced[i] <= src_valid_synced[i-1];
            end
            src_valid_synced[0] <= src_valid_i;
            
            // Shift synchronizers for src_gray_ptr
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                src_gray_synced[i] <= src_gray_synced[i-1];
            end
            src_gray_synced[0] <= src_gray_ptr;
            
            // Shift synchronizers for dst_ready (to know if we can pull)
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                dst_ready_synced[i] <= dst_ready_synced[i-1];
            end
            dst_ready_synced[0] <= dst_ready_i;
            
            // Shift synchronizers for dst_gray_ptr
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                dst_gray_synced[i] <= dst_gray_synced[i-1];
            end
            dst_gray_synced[0] <= dst_gray_ptr;
            
            // Update pointer if valid and ready
            if (src_valid_synced[0] & dst_ready_synced[0]) begin
                dst_ptr <= dst_ptr + 1'b1;
            end
            
            // Read data
            if (src_valid_synced[0] & dst_ready_synced[0]) begin
                dst_data_o <= fifo_mem[dst_ptr];
                dst_valid_o <= 1'b1;
            end else begin
                dst_valid_o <= 1'b0;
            end
        end
    end

    // ---------------------------------------------------------
    // Pointer Generation (Gray Encoding)
    // ---------------------------------------------------------
    // Generate Gray code from binary pointer in source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_gray_ptr <= '0;
        end else begin
            // Simple binary to gray conversion
            src_gray_ptr[LOG_DEPTH-1] <= src_ptr[LOG_DEPTH-1];
            for (integer i = LOG_DEPTH-2; i >= 0; i = i - 1) begin
                src_gray_ptr[i] <= src_ptr[i+1] ^ src_ptr[i];
            end
        end
    end

    // Generate Gray code from binary pointer in destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_gray_ptr <= '0;
        end else begin
            // Simple binary to gray conversion
            dst_gray_ptr[LOG_DEPTH-1] <= dst_ptr[LOG_DEPTH-1];
            for (integer i = LOG_DEPTH-2; i >= 0; i = i - 1) begin
                dst_gray_ptr[i] <= dst_ptr[i+1] ^ dst_ptr[i];
            end
        end
    end

    // ---------------------------------------------------------
    // Empty/Full Detection using Gray Code Comparison
    // ---------------------------------------------------------
    // Empty: src_gray_ptr == dst_gray_ptr
    // Full: (src_gray_ptr + 1) == dst_gray_ptr (with wrap around check)
    
    // We need to compare the synchronized gray pointers to determine empty/full status
    // However, the standard approach for async FIFOs is to compare the pointers
    // directly in the respective domains after synchronization.
    
    // Empty Signal (Source Domain): src_gray_ptr == dst_gray_ptr (synchronized)
    // Full Signal (Destination Domain): (src_gray_ptr + 1) == dst_gray_ptr (synchronized)
    
    // Note: The above logic inside the always blocks handles the pointer updates.
    // The actual empty/full flags are derived from the synchronized pointers.
    
    // Since we are using a single module, we need to ensure the comparison happens
    // in the correct domain.
    
    // Empty check in Source Domain (using synced dst_gray_ptr)
    // This determines if we can write (src_ready)
    // But wait, src_ready is determined by whether the FIFO is NOT FULL.
    // Full is detected in the destination domain usually, or by comparing pointers.
    
    // Let's refine the logic:
    // 1. Source Domain calculates if FIFO is FULL.
    //    Full condition: src_gray_ptr == (dst_gray_ptr - 1) (modulo 2^N)
    //    Actually, simpler: Full if (src_gray_ptr + 1) == dst_gray_ptr (considering wrap)
    //    But since we sync pointers, we can do this comparison in Source Domain.
    
    // 2. Destination Domain calculates if FIFO is EMPTY.
    //    Empty condition: src_gray_ptr == dst_gray_ptr
    
    // Re-evaluating the always blocks for clarity and correctness:
    
    // Source Domain: Determine if we can accept data (not full)
    // We need to know if dst_gray_ptr == (src_gray_ptr + 1)
    // Since both are available in src domain after sync:
    
    reg [GRAY_WIDTH-1:0] src_next_gray_ptr;
    
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_next_gray_ptr <= '0;
        end else begin
            src_next_gray_ptr[LOG_DEPTH-1] <= src_gray_ptr[LOG_DEPTH-1];
            for (integer i = LOG_DEPTH-2; i >= 0; i = i - 1) begin
                src_next_gray_ptr[i] <= src_gray_ptr[i+1] ^ src_gray_ptr[i];
            end
        end
    end

    // Full detection in Source Domain
    // Full if src_gray_ptr + 1 == dst_gray_ptr (with wrap around)
    // In Gray code, addition is not straightforward. 
    // Standard practice: Compare binary pointers after synchronization.
    // But we only have Gray pointers synchronized.
    // Alternative: Use the property that if src_gray == dst_gray, it's empty.
    // If src_gray == (dst_gray - 1), it's full.
    // Calculating (dst_gray - 1) in Gray code is complex.
    
    // Better Approach: 
    // 1. Sync binary pointers to the other domain? No, that's what we are avoiding.
    // 2. Use the Gray pointers directly.
    //    Empty: src_gray == dst_gray
    //    Full: src_gray == (dst_gray with MSB flipped if LSB is 1? No.)
    
    // Let's stick to the most robust method: 
    // Compare the synchronized Gray pointers.
    // Empty = (src_gray_synced[0] == dst_gray_synced[0])
    // Full = (src_gray_synced[0] == (dst_gray_synced[0] - 1)) ?? 
    // Actually, the standard way with Gray pointers is:
    // Empty: src_gray == dst_gray
    // Full: src_gray == (dst_gray - 1) is tricky.
    
    // Correct Standard Implementation:
    // The pointers are incremented in binary. The Gray code is just for transmission.
    // To check Empty/Full, we compare the Gray codes.
    // Empty: src_gray == dst_gray
    // Full: src_gray == (dst_gray - 1) is NOT correct for Gray code arithmetic.
    
    // Let's revert to comparing the binary values after synchronization?
    // No, the requirement says "Pointers cross as acknowledged binary snapshots".
    // This implies we should convert Gray to Binary in the receiving domain?
    // Or simply use the Gray comparison logic which is:
    // Empty: src_gray == dst_gray
    // Full: src_gray == (dst_gray - 1) is only true if we implement a Gray decrementer.
    
    // Actually, the most common and safe implementation for Async FIFOs with Gray pointers:
    // 1. Generate Gray pointers in both domains.
    // 2. Sync Gray pointers across domains.
    // 3. Compare Synced Gray pointers.
    //    Empty: src_gray == dst_gray
    //    Full: src_gray == (dst_gray - 1) is NOT the way.
    //    The correct logic for Full using Gray pointers is:
    //    Full if (src_gray == dst_gray) is Empty.
    //    Full if (src_gray == (dst_gray - 1)) is incorrect.
    
    // Let's use the property:
    // If src_gray == dst_gray, FIFO is empty.
    // If src_gray == (dst_gray - 1), FIFO is full.
    // But calculating (dst_gray - 1) in Gray code is non-trivial without a decoder.
    
    // Alternative Robust Method:
    // 1. Sync the GRAY pointers.
    // 2. In the Source Domain, check if src_gray == dst_gray (Empty) -> Not needed for src_ready.
    // 3. In the Source Domain, check if src_gray == (dst_gray - 1).
    //    To do this safely, we can implement a Gray decrementer or convert to binary.
    //    Converting Gray to Binary is safe if done in the same domain as the Gray pointer.
    
    // Let's implement Gray to Binary conversion in the Source Domain for Full check.
    // And Gray to Binary conversion in the Destination Domain for Empty check.
    
    // Source Domain: Check Full
    // We have src_gray (local) and dst_gray_synced (from dst).
    // Convert dst_gray_synced to binary in src domain.
    reg [LOG_DEPTH-1:0] dst_ptr_bin_synced;
    
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            dst_ptr_bin_synced <= '0;
        end else begin
            // Convert Gray to Binary
            dst_ptr_bin_synced[LOG_DEPTH-1] <= dst_gray_synced[LOG_DEPTH-1];
            for (integer i = LOG_DEPTH-2; i >= 0; i = i - 1) begin
                dst_ptr_bin_synced[i] <= dst_gray_synced[i] ^ dst_ptr_bin_synced[i+1];
            end
        end
    end
    
    // Full Condition: src_ptr == dst_ptr_bin_synced - 1
    // Handle wrap around: if dst_ptr_bin_synced is 0, then full if src_ptr is DEPTH-1.
    reg src_full;
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_full <= 1'b0;
        end else begin
            if (dst_ptr_bin_synced == 0) begin
                src_full <= (src_ptr == (DEPTH-1));
            end else begin
                src_full <= (src_ptr == (dst_ptr_bin_synced - 1));
            end
        end
    end
    
    // Destination Domain: Check Empty
    // We have dst_gray (local) and src_gray_synced (from src).
    // Convert src_gray_synced to binary in dst domain.
    reg [LOG_DEPTH-1:0] src_ptr_bin_synced;
    
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_ptr_bin_synced <= '0;
        end else begin
            // Convert Gray to Binary
            src_ptr_bin_synced[LOG_DEPTH-1] <= src_gray_synced[LOG_DEPTH-1];
            for (integer i = LOG_DEPTH-2; i >= 0; i = i - 1) begin
                src_ptr_bin_synced[i] <= src_gray_synced[i] ^ src_ptr_bin_synced[i+1];
            end
        end
    end
    
    // Empty Condition: src_ptr_bin_synced == dst_ptr
    reg src_empty;
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            src_empty <= 1'b1;
        end else begin
            src_empty <= (src_ptr_bin_synced == dst_ptr);
        end
    end

    // Update src_ready_o based on src_full
    // src_ready_o = !src_full
    // But we need to ensure this is updated correctly in the source domain logic.
    // The previous always block for src_ptr update used src_valid_synced and dst_ready_synced.
    // We need to integrate src_full there.
    
    // Re-writing the Source Domain Logic to include Full check
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_ptr <= '0;
            src_valid_synced <= '0;
            src_rst_synced <= 1'b1;
            src_ready_o <= 1'b0;
        end else begin
            src_rst_synced <= 1'b0;
            
            // Shift src_valid
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                src_valid_synced[i] <= src_valid_synced[i-1];
            end
            src_valid_synced[0] <= src_valid_i;
            
            // Shift dst_ready
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                dst_ready_synced[i] <= dst_ready_synced[i-1];
            end
            dst_ready_synced[0] <= dst_ready_i;
            
            // Shift dst_gray_ptr
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                src_gray_synced[i] <= src_gray_synced[i-1];
            end
            src_gray_synced[0] <= src_gray_ptr;
            
            // Shift dst_ready_synced (already done above, but needed for full check logic if we used it differently)
            // Actually, we calculated src_full using dst_ptr_bin_synced which is derived from dst_gray_synced.
            // So we need to ensure dst_gray_synced is shifted correctly.
            
            // Update src_ready_o
            // Ready if valid AND NOT full
            src_ready_o <= src_valid_synced[0] & ~src_full;
            
            // Update pointer if valid and ready
            if (src_valid_synced[0] & dst_ready_synced[0] & ~src_full) begin
                src_ptr <= src_ptr + 1'b1;
                fifo_mem[src_ptr] <= src_data_i;
            end
        end
    end

    // Re-writing the Destination Domain Logic to include Empty check
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_ptr <= '0;
            dst_valid_o <= 1'b0;
            dst_data_o <= '0;
            dst_gray_synced <= '0;
            dst_rst_synced <= 1'b1;
        end else begin
            dst_rst_synced <= 1'b0;
            
            // Shift src_valid
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                src_valid_synced[i] <= src_valid_synced[i-1];
            end
            src_valid_synced[0] <= src_valid_i;
            
            // Shift src_gray_ptr
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                dst_gray_synced[i] <= dst_gray_synced[i-1];
            end
            dst_gray_synced[0] <= src_gray_ptr;
            
            // Shift dst_ready
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                dst_ready_synced[i] <= dst_ready_synced[i-1];
            end
            dst_ready_synced[0] <= dst_ready_i;
            
            // Update dst_valid_o and dst_data_o
            // Valid if NOT empty and ready
            if (src_valid_synced[0] & dst_ready_synced[0] & ~src_empty) begin
                dst_ptr <= dst_ptr + 1'b1;
                dst_data_o <= fifo_mem[dst_ptr]; // Read BEFORE increment? 
                // Wait, standard FIFO: Read pointer points to the data to be read.
                // If we increment after reading:
                // dst_ptr points to current data.
                // Read data.
                // Increment dst_ptr.
                // Check empty again?
                
                // Let's trace:
                // Init: src=0, dst=0. Empty.
                // Write: src=1, dst=0. Not empty.
                // Read: Read fifo[0]. Increment dst to 1.
                // Now src=1, dst=1. Empty.
                
                // So logic:
                // if (!empty && ready) {
                //    data = fifo[dst_ptr];
                //    dst_ptr++;
                //    valid = 1;
                // }
                
                // But we need to be careful with the order.
                // If we increment dst_ptr first, then read, we read the next one.
                // Correct order: Read, then Increment.
                
                // However, the condition `~src_empty` ensures we are not reading from empty.
                // So:
                dst_data_o <= fifo_mem[dst_ptr];
                dst_ptr <= dst_ptr + 1'b1;
                dst_valid_o <= 1'b1;
            end else begin
                dst_valid_o <= 1'b0;
            end
        end
    end

    // ---------------------------------------------------------
    // Reset Synchronization
    // ---------------------------------------------------------
    // Ensure resets are synchronized before use in logic
    // (Already handled in the always blocks with negedge detection, 
    // but explicit sync chains are safer for RDC)
    
    // Source Reset Sync
    reg src_rst_sync_chain [SYNC_STAGES-1:0];
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_rst_sync_chain <= '0;
        end else begin
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                src_rst_sync_chain[i] <= src_rst_sync_chain[i-1];
            end
            src_rst_sync_chain[0] <= src_rst_ni;
        end
    end
    
    // Destination Reset Sync
    reg dst_rst_sync_chain [SYNC_STAGES-1:0];
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_rst_sync_chain <= '0;
        end else begin
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                dst_rst_sync_chain[i] <= dst_rst_sync_chain[i-1];
            end
            dst_rst_sync_chain[0] <= dst_rst_ni;
        end
    end

    // Use the synchronized resets for the main logic to ensure clean state transitions
    // Re-implementing the main logic with synchronized resets for maximum safety
    
    // Source Domain Logic (Revised)
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_ptr <= '0;
            src_valid_synced <= '0;
            src_rst_synced <= 1'b1;
            src_ready_o <= 1'b0;
        end else begin
            src_rst_synced <= 1'b0;
            
            // Shift src_valid
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                src_valid_synced[i] <= src_valid_synced[i-1];
            end
            src_valid_synced[0] <= src_valid_i;
            
            // Shift dst_ready
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                dst_ready_synced[i] <= dst_ready_synced[i-1];
            end
            dst_ready_synced[0] <= dst_ready_i;
            
            // Shift dst_gray_ptr
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                src_gray_synced[i] <= src_gray_synced[i-1];
            end
            src_gray_synced[0] <= src_gray_ptr;
            
            // Update src_ready_o
            src_ready_o <= src_valid_synced[0] & ~src_full;
            
            // Update pointer if valid and ready
            if (src_valid_synced[0] & dst_ready_synced[0] & ~src_full) begin
                src_ptr <= src_ptr + 1'b1;
                fifo_mem[src_ptr] <= src_data_i;
            end
        end
    end

    // Destination Domain Logic (Revised)
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_ptr <= '0;
            dst_valid_o <= 1'b0;
            dst_data_o <= '0;
            dst_gray_synced <= '0;
            dst_rst_synced <= 1'b1;
        end else begin
            dst_rst_synced <= 1'b0;
            
            // Shift src_valid
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                src_valid_synced[i] <= src_valid_synced[i-1];
            end
            src_valid_synced[0] <= src_valid_i;
            
            // Shift src_gray_ptr
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                dst_gray_synced[i] <= dst_gray_synced[i-1];
            end
            dst_gray_synced[0] <= src_gray_ptr;
            
            // Shift dst_ready
            for (integer i = SYNC_STAGES-1; i > 0; i = i - 1) begin
                dst_ready_synced[i] <= dst_ready_synced[i-1];
            end
            dst_ready_synced[0] <= dst_ready_i;
            
            // Update dst_valid_o and dst_data_o
            if (src_valid_synced[0] & dst_ready_synced[0] & ~src_empty) begin
                dst_data_o <= fifo_mem[dst_ptr];
                dst_ptr <= dst_ptr + 1'b1;
                dst_valid_o <= 1'b1;
            end else begin
                dst_valid_o <= 1'b0;
            end
        end
    end

endmodule
