module sync #(
    parameter int unsigned STAGES = 2,
    parameter bit ResetValue = 1'b0
) (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic serial_i,
    output logic serial_o
);

    logic [STAGES-1:0] stage;

    genvar i;
    generate
        for (i = 0; i < STAGES; i++) begin : gen_stage
            logic sync_stage;
            assign sync_stage = (rst_ni) ? ResetValue : stage[i];

            always_ff @(posedge clk_i or negedge rst_ni) begin
                if (!rst_ni)
                    stage[i] <= ResetValue;
                else
                    stage[i] <= sync_stage;
            end
        end
    endgenerate

    assign serial_o = stage[STAGES-1];

endmodule
