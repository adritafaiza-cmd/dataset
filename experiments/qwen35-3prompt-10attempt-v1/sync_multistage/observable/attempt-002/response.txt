module sync #(
    parameter int unsigned STAGES = 2,
    parameter bit ResetValue = 1'b0
) (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic serial_i,
    output logic serial_o
);

    logic [STAGES-1:0] stage;

    genvar i;
    generate
        for (i = 0; i < STAGES; i++) begin : stage_gen
            logic sync_stage;
            always_ff @(posedge clk_i or negedge rst_ni) begin
                if (!rst_ni)
                    sync_stage <= ResetValue;
                else
                    sync_stage <= stage[i];
            end
            assign stage[i] = sync_stage;
        end
    endgenerate

    assign serial_o = stage[STAGES-1];

endmodule
