module sync #(
    parameter int unsigned STAGES = 2,
    parameter bit ResetValue = 1'b0
) (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic serial_i,
    output logic serial_o
);

    // Generate pipeline stages for CDC compliance
    // Using a counter to generate enable signals for each stage
    // This ensures all flip-flops are clocked by the same domain (clk_i)
    // and reset is synchronized before being applied to the pipeline.

    logic [STAGES-1:0] stage_data;
    logic [STAGES-1:0] stage_en;
    logic rst_sync;

    // Synchronize the asynchronous reset to the destination clock domain
    // This is critical for RDC (Reset Domain Crossing) safety.
    // We use a 2-stage synchronizer for the reset signal itself.
    logic rst_sync_1;
    logic rst_sync_2;

    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            rst_sync_1 <= 1'b0;
        else
            rst_sync_1 <= rst_ni;
    end

    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            rst_sync_2 <= 1'b0;
        else
            rst_sync_2 <= rst_sync_1;
    end

    // Generate enable signals for the pipeline stages
    // Stage 0 is enabled when rst_sync_2 is low (during reset) or high (normal operation)
    // Actually, for a simple FIFO-like structure where we just pass data:
    // We need to ensure the first flip-flop captures on the first valid clock edge after reset.
    // A common pattern is to use the synchronized reset to gate the first stage,
    // but since serial_i is asynchronous, we must ensure the first stage only updates
    // when the reset is stable.
    
    // Strategy:
    // 1. Synchronize rst_ni to get rst_sync (active high, stable).
    // 2. The first stage (stage_data[0]) should capture serial_i only after reset is released.
    //    We can use ~rst_sync as the enable for the first stage, but we need to be careful.
    //    Better approach: Use a counter or a specific enable logic.
    //    Standard robust method: 
    //    - Stage 0: Captures serial_i when ~rst_sync is asserted (i.e., after reset release).
    //    - Stages 1 to STAGES-1: Always capture the previous stage's data on every clock edge.
    //    However, if we just chain them, stage 0 might capture a glitch if serial_i changes
    //    exactly at reset release. 
    //    To be absolutely safe against metastability at reset release, we often use a "reset release"
    //    signal that is one clock cycle wide, or simply rely on the fact that serial_i is held
    //    stable during reset. Assuming serial_i is stable during reset (common assumption),
    //    we can use the following logic:
    //    - Stage 0: Enabled by ~rst_sync.
    //    - Stages 1..N: Always enabled.
    
    // Let's refine the reset synchronization for the pipeline enables.
    // We need a signal that is 1 during reset and 0 after reset is stable.
    // We already have rst_sync (0 during reset, 1 after).
    // So ~rst_sync is 1 during reset, 0 after.
    // If we use ~rst_sync as enable for stage 0, it will capture serial_i on the first clock
    // edge after reset. This is safe provided serial_i is stable during reset.
    
    // Generate enables
    // Stage 0 enable: Active when reset is not active (rst_sync is 1)
    // Wait, if rst_sync is 1 (reset released), ~rst_sync is 0. That disables it.
    // We want it to capture AFTER reset is released.
    // Let's re-evaluate:
    // rst_ni = 0 -> rst_sync = 0.
    // rst_ni = 1 -> rst_sync eventually becomes 1.
    // We want stage 0 to capture serial_i when rst_ni is 1 (and stable).
    // So enable = rst_sync.
    // But what about the very first edge? If rst_ni goes high, rst_sync takes 2 cycles to go high.
    // Cycle 1: rst_ni=1, rst_sync=0. Stage 0 disabled.
    // Cycle 2: rst_ni=1, rst_sync=1. Stage 0 enabled. Captures serial_i.
    // This is safe.
    
    // For subsequent stages, they just pass the data through.
    
    // Re-assigning logic for clarity:
    // rst_sync is 1 when reset is inactive and stable.
    
    // Stage 0: Capture serial_i when rst_sync is high.
    // Stages 1 to STAGES-1: Capture previous stage data always.
    
    // However, there is a nuance: If STAGES=1, we just need one FF.
    // If STAGES > 1, we need a chain.
    
    // Let's implement the chain.
    
    // First, let's create the enable signal for the first stage properly.
    // We need to ensure that the first stage does not capture until reset is stable.
    // Using rst_sync directly as enable for stage 0 works if serial_i is stable during reset.
    
    // Assign enables
    // stage_en[0] = rst_sync;
    // stage_en[1:STAGES-1] = 1'b1;
    
    // But wait, if we use a counter approach to generate enables, it's more flexible and standard.
    // However, for a simple bit synchronizer, a direct enable chain is sufficient and efficient.
    
    // Let's stick to the direct enable chain for simplicity and clarity, ensuring rst_sync is used.
    
    // Note: In SystemVerilog, we can use a loop to generate the always blocks or assign statements.
    // Since STAGES is a parameter, we can't use a for-loop in procedural code directly without generate.
    // We will use generate blocks for the pipeline stages.
    
    genvar i;
    generate
        for (i = 0; i < STAGES; i = i + 1) begin : stage_pipeline
            if (i == 0) begin : stage_first
                // First stage: Enabled only after reset is stable
                always_ff @(posedge clk_i or negedge rst_ni) begin
                    if (!rst_ni)
                        stage_data[0] <= ResetValue;
                    else if (rst_sync)
                        stage_data[0] <= serial_i;
                    else
                        stage_data[0] <= stage_data[0]; // Hold
                end
            end else begin : stage_subsequent
                // Subsequent stages: Always capture previous stage data
                always_ff @(posedge clk_i or negedge rst_ni) begin
                    if (!rst_ni)
                        stage_data[i] <= ResetValue;
                    else
                        stage_data[i] <= stage_data[i-1];
                end
            end
        end
    endgenerate

    // Output assignment
    assign serial_o = stage_data[STAGES-1];

endmodule<|im_end|>
