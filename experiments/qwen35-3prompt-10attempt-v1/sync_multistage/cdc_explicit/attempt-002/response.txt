module sync #(
    parameter int unsigned STAGES = 2,
    parameter bit ResetValue = 1'b0
) (
    input  logic clk_i,
    input  logic rst_ni,
    input  logic serial_i,
    output logic serial_o
);

    // Internal clock for the destination domain (same as clk_i)
    logic clk_int;
    assign clk_int = clk_i;

    // Internal reset for the destination domain
    // We use a simple synchronizer for the reset signal to ensure RDC safety
    // before distributing it to the pipeline stages.
    logic rst_ni_synced;
    logic rst_ni_synced_2;

    // Reset Synchronizer (2-stage)
    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            rst_ni_synced <= 1'b1;
        else
            rst_ni_synced <= rst_ni_synced_2;
    end

    always_ff @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni)
            rst_ni_synced_2 <= 1'b1;
        else
            rst_ni_synced_2 <= rst_ni_synced;
    end

    // Pipeline stages for data synchronization
    // Each stage captures the data on the destination clock edge
    genvar i;
    logic [STAGES-1:0] sync_pipe;

    generate
        for (i = 0; i < STAGES; i = i + 1) begin : stage_gen
            always_ff @(posedge clk_int or negedge rst_ni) begin
                if (!rst_ni)
                    sync_pipe[i] <= ResetValue;
                else
                    sync_pipe[i] <= sync_pipe[i-1];
            end
        end
    endgenerate

    // Assign output from the last stage
    assign serial_o = sync_pipe[STAGES-1];

endmodule<|im_end|>
