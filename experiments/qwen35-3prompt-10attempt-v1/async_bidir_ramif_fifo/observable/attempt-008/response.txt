```verilog
`timescale 1ns/1ps

module async_bidir_ramif_fifo
  #(
    parameter DSIZE         = 8,
    parameter ASIZE         = 4,
    parameter FALLTHROUGH   = "FALSE"
    ) (
       input wire              a_clk,
       input wire              a_rst_n,
       input wire              a_winc,
       input wire [DSIZE-1:0]  a_wdata,
       input wire              a_rinc,
       output wire [DSIZE-1:0] a_rdata,
       output wire             a_full,
       output wire             a_afull,
       output wire             a_empty,
       output wire             a_aempty,
       input wire              a_dir,
       input wire              b_clk,
       input wire              b_rst_n,
       input wire              b_winc,
       input wire [DSIZE-1:0]  b_wdata,
       input wire              b_rinc,
       output wire [DSIZE-1:0] b_rdata,
       output wire             b_full,
       output wire             b_afull,
       output wire             b_empty,
       output wire             b_aempty,
       input wire              b_dir,
       output wire             o_ram_a_clk,
       output wire [DSIZE-1:0] o_ram_a_wdata,
       input wire [DSIZE-1:0]  i_ram_a_rdata,
       output wire [ASIZE-1:0] o_ram_a_addr,
       output wire             o_ram_a_rinc,
       output wire             o_ram_a_winc,
       output wire             o_ram_b_clk,
       output wire [DSIZE-1:0] o_ram_b_wdata,
       input wire [DSIZE-1:0]  i_ram_b_rdata,
       output wire [ASIZE-1:0] o_ram_b_addr,
       output wire             o_ram_b_rinc,
       output wire             o_ram_b_winc
       );

  // Internal pointers and counters
  reg [ASIZE-1:0] a_write_ptr, a_read_ptr;
  reg [ASIZE-1:0] b_write_ptr, b_read_ptr;
  
  // Cross-domain pointers (snapshots)
  reg [ASIZE-1:0] a_read_ptr_b, b_write_ptr_a;
  reg [ASIZE-1:0] b_read_ptr_a, a_write_ptr_b;

  // Status flags
  reg a_full_int, a_empty_int;
  reg b_full_int, b_empty_int;
  reg a_full_sync, a_empty_sync;
  reg b_full_sync, b_empty_sync;

  // Direction control
  reg a_dir_reg, b_dir_reg;

  // RAM Interface Signals
  wire [ASIZE-1:0] ram_addr_a, ram_addr_b;
  wire [DSIZE-1:0] ram_wdata_a, ram_wdata_b;
  wire ram_winc_a, ram_winc_b;
  wire ram_rinc_a, ram_rinc_b;
  wire [DSIZE-1:0] ram_rdata_a, ram_rdata_b;
  wire ram_clk_a, ram_clk_b;

  // Generate clock signals for RAM
  assign o_ram_a_clk = ram_clk_a;
  assign o_ram_b_clk = ram_clk_b;
  assign o_ram_a_wdata = ram_wdata_a;
  assign o_ram_b_wdata = ram_wdata_b;
  assign o_ram_a_addr = ram_addr_a;
  assign o_ram_b_addr = ram_addr_b;
  assign o_ram_a_rinc = ram_rinc_a;
  assign o_ram_b_rinc = ram_rinc_b;
  assign o_ram_a_winc = ram_winc_a;
  assign o_ram_b_winc = ram_winc_b;

  // RAM Read Data Inputs
  assign i_ram_a_rdata = ram_rdata_a;
  assign i_ram_b_rdata = ram_rdata_b;

  // Logic to determine direction and update pointers
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_write_ptr <= 0;
      a_read_ptr  <= 0;
      a_dir_reg   <= 0;
    end else begin
      a_write_ptr <= a_write_ptr;
      a_read_ptr  <= a_read_ptr;
      a_dir_reg   <= a_dir;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_write_ptr <= 0;
      b_read_ptr  <= 0;
      b_dir_reg   <= 0;
    end else begin
      b_write_ptr <= b_write_ptr;
      b_read_ptr  <= b_read_ptr;
      b_dir_reg   <= b_dir;
    end
  end

  // Cross-domain pointer updates
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_write_ptr_a <= 0;
      b_read_ptr_a  <= 0;
    end else begin
      if (b_rinc) b_read_ptr_a <= a_write_ptr;
      else if (b_winc) b_write_ptr_a <= a_read_ptr;
      else b_write_ptr_a <= b_write_ptr_a;
      b_read_ptr_a <= b_read_ptr_a;
    end
  end

  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_write_ptr_b <= 0;
      a_read_ptr_b  <= 0;
    end else begin
      if (a_rinc) a_read_ptr_b <= b_write_ptr;
      else if (a_winc) a_write_ptr_b <= b_read_ptr;
      else a_write_ptr_b <= a_write_ptr_b;
      a_read_ptr_b <= a_read_ptr_b;
    end
  end

  // Full/Empty logic for Side A
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_full_int <= 0;
      a_empty_int <= 1;
    end else begin
      if (a_dir_reg) begin
        // Writing to A (reading from B)
        if (a_write_ptr == a_read_ptr_b) a_full_int <= 1;
        else if (a_write_ptr == a_read_ptr_b - 1) a_empty_int <= 1;
        else begin
          a_full_int <= 0;
          a_empty_int <= 0;
        end
      end else begin
        // Reading from A (writing to B)
        if (a_read_ptr == a_write_ptr_b) a_empty_int <= 1;
        else if (a_read_ptr == a_write_ptr_b - 1) a_full_int <= 1;
        else begin
          a_full_int <= 0;
          a_empty_int <= 0;
        end
      end
    end
  end

  // Full/Empty logic for Side B
  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_full_int <= 0;
      b_empty_int <= 1;
    end else begin
      if (b_dir_reg) begin
        // Writing to B (reading from A)
        if (b_write_ptr == b_read_ptr_a) b_full_int <= 1;
        else if (b_write_ptr == b_read_ptr_a - 1) b_empty_int <= 1;
        else begin
          b_full_int <= 0;
          b_empty_int <= 0;
        end
      end else begin
        // Reading from B (writing to A)
        if (b_read_ptr == b_write_ptr_a) b_empty_int <= 1;
        else if (b_read_ptr == b_write_ptr_a - 1) b_full_int <= 1;
        else begin
          b_full_int <= 0;
          b_empty_int <= 0;
        end
      end
    end
  end

  // Synchronize Full/Empty signals across clock domains
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      a_full_sync <= 0;
      a_empty_sync <= 1;
    end else begin
      a_full_sync <= b_full_int;
      a_empty_sync <= b_empty_int;
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      b_full_sync <= 0;
      b_empty_sync <= 1;
    end else begin
      b_full_sync <= a_full_int;
      b_empty_sync <= a_empty_int;
    end
  end

  // Output assignments
  assign a_full = a_full_sync;
  assign a_empty = a_empty_sync;
  assign b_full = b_full_sync;
  assign b_empty = b_empty_sync;

  assign a_afull = a_full_int;
  assign a_aempty = a_empty_int;
  assign b_afull = b_full_int;
  assign b_aempty = b_empty_int;

  // Read Data Outputs
  assign a_rdata = (a_dir_reg) ? i_ram_a_rdata : 0;
  assign b_rdata = (b_dir_reg) ? i_ram_b_rdata : 0;

  // Write Data and Address Generation
  always @(posedge a_clk or negedge a_rst_n) begin
    if (!a_rst_n) begin
      ram_addr_a <= 0;
      ram_wdata_a <= 0;
      ram_winc_a <= 0;
      ram_rinc_a <= 0;
    end else begin
      if (a_dir_reg) begin
        // Writing to A (reading from B)
        if (a_winc) begin
          ram_addr_a <= a_write_ptr;
          ram_wdata_a <= a_wdata;
          ram_winc_a <= 1;
          ram_rinc_a <= 0;
        end else begin
          ram_addr_a <= a_write_ptr;
          ram_wdata_a <= 0;
          ram_winc_a <= 0;
          ram_rinc_a <= 0;
        end
      end else begin
        // Reading from A (writing to B)
        if (a_rinc) begin
          ram_addr_a <= a_read_ptr;
          ram_wdata_a <= 0;
          ram_winc_a <= 0;
          ram_rinc_a <= 1;
        end else begin
          ram_addr_a <= a_read_ptr;
          ram_wdata_a <= 0;
          ram_winc_a <= 0;
          ram_rinc_a <= 0;
        end
      end
    end
  end

  always @(posedge b_clk or negedge b_rst_n) begin
    if (!b_rst_n) begin
      ram_addr_b <= 0;
      ram_wdata_b <= 0;
      ram_winc_b <= 0;
      ram_rinc_b <= 0;
    end else begin
      if (b_dir_reg) begin
        // Writing to B (reading from A)
        if (b_winc) begin
          ram_addr_b <= b_write_ptr;
          ram_wdata_b <= b_wdata;
          ram_winc_b <= 1;
          ram_rinc_b <= 0;
        end else begin
          ram_addr_b <= b_write_ptr;
          ram_wdata_b <= 0;
          ram_winc_b <= 0;
          ram_rinc_b <= 0;
        end
      end else begin
        // Reading from B (writing to A)
        if (b_rinc) begin
          ram_addr_b <= b_read_ptr;
          ram_wdata_b <= 0;
          ram_winc_b <= 0;
          ram_rinc_b <= 1;
        end else begin
          ram_addr_b <= b_read_ptr;
          ram_wdata_b <= 0;
          ram_winc_b <= 0;
          ram_rinc_b <= 0;
        end
      end
    end
  end

endmodule
