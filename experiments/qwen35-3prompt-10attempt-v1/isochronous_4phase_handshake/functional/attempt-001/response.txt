module isochronous_4phase_handshake (
    input  src_clk_i,
    input  src_rst_ni,
    input  src_valid_i,
    output src_ready_o,
    input  dst_clk_i,
    input  dst_rst_ni,
    output dst_valid_o,
    input  dst_ready_i
);

    reg src_state;
    reg dst_state;
    reg src_valid_reg;
    reg dst_valid_reg;
    reg src_ready_reg;
    reg dst_ready_reg;

    // State encoding: 0=Idle, 1=SentValid, 2=SentReady, 3=SentValidAgain
    parameter IDLE = 2'b00;
    parameter SENT_VALID = 2'b01;
    parameter SENT_READY = 2'b10;
    parameter SENT_VALID_AGAIN = 2'b11;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_state <= IDLE;
            src_valid_reg <= 1'b0;
            src_ready_reg <= 1'b0;
        end else begin
            case (src_state)
                IDLE: begin
                    if (src_valid_i)
                        src_state <= SENT_VALID;
                    src_valid_reg <= 1'b0;
                    src_ready_reg <= 1'b0;
                end
                SENT_VALID: begin
                    if (src_valid_i)
                        src_state <= SENT_VALID;
                    else
                        src_state <= SENT_READY;
                    src_valid_reg <= 1'b1;
                    src_ready_reg <= 1'b0;
                end
                SENT_READY: begin
                    if (src_valid_i)
                        src_state <= SENT_VALID_AGAIN;
                    else
                        src_state <= IDLE;
                    src_valid_reg <= 1'b0;
                    src_ready_reg <= 1'b1;
                end
                SENT_VALID_AGAIN: begin
                    if (src_valid_i)
                        src_state <= SENT_VALID;
                    else
                        src_state <= IDLE;
                    src_valid_reg <= 1'b1;
                    src_ready_reg <= 1'b0;
                end
                default: begin
                    src_state <= IDLE;
                    src_valid_reg <= 1'b0;
                    src_ready_reg <= 1'b0;
                end
            endcase
        end
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_state <= IDLE;
            dst_valid_reg <= 1'b0;
            dst_ready_reg <= 1'b0;
        end else begin
            case (dst_state)
                IDLE: begin
                    if (dst_ready_i)
                        dst_state <- SENT_VALID;
                    dst_valid_reg <= 1'b0;
                    dst_ready_reg <= 1'b0;
                end
                SENT_VALID: begin
                    if (dst_ready_i)
                        dst_state <= SENT_READY;
                    else
                        dst_state <= IDLE;
                    dst_valid_reg <= 1'b1;
                    dst_ready_reg <= 1'b0;
                end
                SENT_READY: begin
                    if (dst_ready_i)
                        dst_state <= SENT_VALID_AGAIN;
                    else
                        dst_state <= IDLE;
                    dst_valid_reg <= 1'b0;
                    dst_ready_reg <= 1'b1;
                end
                SENT_VALID_AGAIN: begin
                    if (dst_ready_i)
                        dst_state <= SENT_VALID;
                    else
                        dst_state <= IDLE;
                    dst_valid_reg <= 1'b1;
                    dst_ready_reg <= 1'b0;
                end
                default: begin
                    dst_state <= IDLE;
                    dst_valid_reg <= 1'b0;
                    dst_ready_reg <= 1'b0;
                end
            endcase
        end
    end

    assign src_ready_o = src_ready_reg;
    assign dst_valid_o = dst_valid_reg;

endmodule<|im_end|>
