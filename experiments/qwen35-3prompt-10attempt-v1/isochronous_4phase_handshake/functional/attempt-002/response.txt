module isochronous_4phase_handshake (
    input  src_clk_i,
    input  src_rst_ni,
    input  src_valid_i,
    output src_ready_o,
    input  dst_clk_i,
    input  dst_rst_ni,
    output dst_valid_o,
    input  dst_ready_i
);

    reg src_state;
    reg dst_state;
    reg src_valid_reg;
    reg dst_valid_reg;
    reg dst_ready_reg;

    // State encoding: 0=Idle, 1=SentValid, 2=SentReady, 3=SentValidAgain
    parameter IDLE = 2'b00;
    parameter SENT_VALID = 2'b01;
    parameter SENT_READY = 2'b10;
    parameter SENT_VALID_AGAIN = 2'b11;

    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_state <= IDLE;
        else
            src_state <= src_state;
    end

    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_state <= IDLE;
        else
            dst_state <= dst_state;
    end

    // Synchronize src_valid_i to dst domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            src_valid_reg <= 1'b0;
        else
            src_valid_reg <= src_valid_i;
    end

    // Synchronize dst_ready_i to src domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            dst_ready_reg <= 1'b0;
        else
            dst_ready_reg <= dst_ready_i;
    end

    // Synchronize dst_valid_o to src domain for state machine logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            dst_valid_reg <= 1'b0;
        else
            dst_valid_reg <= dst_valid_o;
    end

    // Synchronize src_ready_o to dst domain for state machine logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            src_ready_reg <= 1'b0;
        else
            src_ready_reg <= src_ready_o;
    end

    // State machine for Source domain
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni)
            src_state <= IDLE;
        else
            case (src_state)
                IDLE: begin
                    if (src_valid_i)
                        src_state <= SENT_VALID;
                    else
                        src_state <= IDLE;
                end
                SENT_VALID: begin
                    if (dst_ready_reg)
                        src_state <= SENT_READY;
                    else
                        src_state <= SENT_VALID;
                end
                SENT_READY: begin
                    if (src_valid_i)
                        src_state <= SENT_VALID_AGAIN;
                    else
                        src_state <= SENT_READY;
                end
                SENT_VALID_AGAIN: begin
                    if (dst_ready_reg)
                        src_state <= IDLE;
                    else
                        src_state <= SENT_VALID_AGAIN;
                end
                default: src_state <= IDLE;
            endcase
    end

    // State machine for Destination domain
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni)
            dst_state <= IDLE;
        else
            case (dst_state)
                IDLE: begin
                    if (src_valid_reg)
                        dst_state <= SENT_VALID;
                    else
                        dst_state <= IDLE;
                end
                SENT_VALID: begin
                    if (src_ready_reg)
                        dst_state <= SENT_READY;
                    else
                        dst_state <= SENT_VALID;
                end
                SENT_READY: begin
                    if (dst_ready_i)
                        dst_state <= SENT_VALID_AGAIN;
                    else
                        dst_state <= SENT_READY;
                end
                SENT_VALID_AGAIN: begin
                    if (src_ready_reg)
                        dst_state <= IDLE;
                    else
                        dst_state <= SENT_VALID_AGAIN;
                end
                default: dst_state <= IDLE;
            endcase
    end

    // Output assignments
    assign src_ready_o = (src_state == SENT_READY) || (src_state == SENT_VALID_AGAIN);
    assign dst_valid_o = (dst_state == SENT_VALID) || (dst_state == SENT_VALID_AGAIN);

endmodule<|im_end|>
