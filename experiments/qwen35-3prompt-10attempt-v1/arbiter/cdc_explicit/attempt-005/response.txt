                    grant = 1'b1;
                    grant_valid = 1'b1;
                    grant_encoded = priority_idx;
                end else begin
                    sm_next_state = 0;
                    grant = 1'b0;
                    grant_valid = 1'b0;
                    grant_encoded = 0;
                end
            end
        end

        end else begin
            sm_state <= sm_next_state;
        end
    end

    // But in our logic, only one bit is ever set at a time.

    // Since only one bit is set, we can use $clog2 or a loop.
    // To be safe and parameterized:
    assign grant_encoded = 0;

        if (sm_state[k]) begin
            grant_encoded = k;
        end
    end
