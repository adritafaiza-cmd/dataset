module isochronous_spill_register #(
    parameter WIDTH = 8
)(
    input              src_clk_i,
    input              src_rst_ni,
    input              src_valid_i,
    output             src_ready_o,
    input  [WIDTH-1:0] src_data_i,
    input              dst_clk_i,
    input              dst_rst_ni,
    output             dst_valid_o,
    input              dst_ready_i,
    output reg [WIDTH-1:0] dst_data_o
);

    // Internal signals
    reg [WIDTH-1:0] src_data_reg;
    reg src_valid_reg;
    reg src_ready_reg;
    reg dst_valid_reg;
    reg dst_ready_reg;
    reg [WIDTH-1:0] dst_data_reg;

    // ---------------------------------------------------------
    // Source Domain Logic
    // ---------------------------------------------------------
    // Capture incoming data and valid signal on the source clock.
    // We hold the data until we can push it to the destination.
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_data_reg      <= '0;
            src_valid_reg     <= 1'b0;
            src_ready_reg     <= 1'b1;
        end else begin
            src_data_reg      <= src_data_i;
            src_valid_reg     <= src_valid_i;
            src_ready_reg     <= dst_ready_reg;
        end
    end

    // ---------------------------------------------------------
    // Destination Domain Logic
    // ---------------------------------------------------------
    // Capture the data and valid signal from the source domain.
    // We hold the data until the destination is ready.
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_data_reg      <= '0;
            dst_valid_reg     <= 1'b0;
            dst_ready_reg     <= 1'b1;
        end else begin
            dst_data_reg      <= src_data_reg;
            dst_valid_reg     <= src_valid_reg;
            dst_ready_reg     <= src_ready_reg;
        end
    end

    // ---------------------------------------------------------
    // Output Logic (CDC Safe)
    // ---------------------------------------------------------
    // Drive outputs based on the registered values in the respective domains.
    // This ensures no multi-bit data crossing occurs directly between clocks.
    assign src_ready_o = src_ready_reg;
    assign dst_valid_o = dst_valid_reg;
    assign dst_data_o  = dst_data_reg;

endmodule
