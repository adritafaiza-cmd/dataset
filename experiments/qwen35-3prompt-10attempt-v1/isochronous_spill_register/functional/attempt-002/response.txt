module isochronous_spill_register #(
    parameter WIDTH = 8
)(
    input              src_clk_i,
    input              src_rst_ni,
    input              src_valid_i,
    output             src_ready_o,
    input  [WIDTH-1:0] src_data_i,
    input              dst_clk_i,
    input              dst_rst_ni,
    output             dst_valid_o,
    input              dst_ready_i,
    output [WIDTH-1:0] dst_data_o
);

    // Source domain state
    reg src_valid_reg;
    reg [WIDTH-1:0] src_data_reg;
    reg src_valid_prev;

    // Destination domain state (2-deep buffer)
    reg [WIDTH-1:0] dst_data_buf [0:1];
    reg dst_valid_buf [0:1];
    reg dst_valid_prev;

    // Source domain logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_valid_reg <= 1'b0;
            src_data_reg <= 1'b0;
            src_valid_prev <= 1'b0;
        end else begin
            src_valid_prev <= src_valid_reg;
            src_valid_reg <= src_valid_i;
            src_data_reg <= src_data_i;
        end
    end

    // Destination domain logic
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_valid_buf[0] <= 1'b0;
            dst_valid_buf[1] <= 1'b0;
            dst_data_buf[0] <= 1'b0;
            dst_data_buf[1] <= 1'b0;
            dst_valid_prev <= 1'b0;
        end else begin
            dst_valid_prev <= dst_valid_buf[0];
            // Shift buffer: move index 1 to index 0, then load new data if available
            dst_valid_buf[0] <= dst_valid_buf[1];
            dst_data_buf[0] <= dst_data_buf[1];
            
            // Load new data if source is ready and destination has space
            if (src_valid_reg && !dst_valid_buf[1]) begin
                dst_valid_buf[1] <= 1'b1;
                dst_data_buf[1] <= src_data_reg;
            end
        end
    end

    // Output assignments
    assign src_ready_o = src_valid_prev && dst_ready_i && !dst_valid_buf[1];
    assign dst_valid_o = dst_valid_buf[0];
    assign dst_data_o = dst_data_buf[0];

endmodule<|im_end|>
