module isochronous_spill_register #(

)(
    input              src_clk_i,
    input              src_rst_ni,
    input              src_valid_i,
    output             src_ready_o,
    input  [WIDTH-1:0] src_data_i,
    input              dst_clk_i,
    input              dst_rst_ni,
    output             dst_valid_o,
    input              dst_ready_i,
    output reg [WIDTH-1:0] dst_data_o

    // 0: IDLE (Reset state, no data held)
    // 1: BUFFERED (Data received from src, waiting for dst)

    localparam BUFFERED = 1'b1;

    reg [WIDTH-1:0] data_reg;
    reg src_valid_reg;
    reg dst_valid_reg;

    // Synchronous logic for state and data registers
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            state_reg <= IDLE;
            data_reg <= 0;
            src_valid_reg <= 1'b0;
        end else begin
            state_reg <= state_reg;
            data_reg <= data_reg;
            src_valid_reg <= src_valid_i;
        end
    end

    // Synchronous logic for destination state and data registers
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            state_reg <= IDLE; // Note: We use the same state variable for simplicity in this 2-deep design

            // we need separate state machines or a single state machine driven by src

            // A 2-deep buffer typically means we can absorb 2 cycles of mismatch.

            // often implies a FIFO-like behavior or a specific handshake protocol.
            // Given the interface and "2-deep", let's implement a simple 2-element FIFO logic
            // or a specific state machine that holds data until ready.
            // Re-reading: "Buffer a short rate mismatch... Deliver accepted data once in order".
            // This implies a FIFO behavior. Let's implement a 2-deep FIFO.

        end

    // --- Revised Implementation: 2-Deep FIFO Logic ---
    
    // FIFO Pointer Logic (Gray code or simple binary for 2 depth)
    // Depth = 2. We need 2 bits for pointers.
    reg [1:0] write_ptr;
    reg [1:0] read_ptr;
    reg full;
    reg empty;
    reg [WIDTH-1:0] fifo_mem [0:1];

    // Write Pointer Logic (Src Domain)
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            write_ptr <= 2'b00;
            full <= 1'b0;
        end else begin
            if (src_valid_i && src_ready_o) begin
                if (!full) begin
                    write_ptr <= write_ptr + 1'b1;
                    full <= 1'b1;
                end
            end else begin
                // If not ready, we might still increment if we are just buffering?
                // No, standard FIFO: if not ready, we don't write.
                // But the requirement says "Buffer a short rate mismatch".
                // If dst is not ready, src must wait (src_ready_o = 0).
                // So write_ptr only increments on successful write.
                write_ptr <= write_ptr;

            end
        end

                    read_ptr <= read_ptr + 1'b1;
                    empty <= 1'b1;
                end

                read_ptr <= read_ptr;
                empty <= 1'b0;

    // Since it's a 2-deep buffer, let's use a simpler state machine approach

    // but a FIFO is the robust way to handle "rate mismatch" and "in order".

    reg buffer_valid;

                    buffer_data <= src_data_i;
                    buffer_valid <= 1'b1;

                    buffer_data <= src_data_i;
                    buffer_valid <= 1'b1;

                    // Still full, wait
                    depth_state <= 2'b10;
                end else if (depth_state == 2'b01) begin
                    // Still 1 item
                    depth_state <= 2'b01;
                end else begin
                    // Empty
                    depth_state <= 2'b00;

    // If valid && ready && count > 0 -> count--, output data.
    
    reg [1:0] count_reg;
    reg [WIDTH-1:0] stored_data;
    reg src_valid_sync;
    reg dst_valid_sync;

    // State Machine in SRC domain that tracks the number of items.
    // But the decrement happens on DST clock.
    // This creates a race condition if not handled.

    // Use a single counter for the depth, but update it on BOTH clocks? No.

    // Let's go with the standard "Gray Code" pointer approach or simply

    // Implementation Plan:
    // 1. Write Pointer (wr_ptr) in SRC domain.
    // 2. Read Pointer (rd_ptr) in DST domain.
    // 3. Memory array in SRC domain (since data comes from SRC).
    // 4. Synchronize "Full" flag from SRC to DST.
    // 5. Synchronize "Empty" flag from DST to SRC.
    
    // Wait, if rd_ptr is in DST domain, how does SRC know when to stop writing?
    // SRC needs to know if the buffer is Full.
    // DST needs to know if the buffer is Empty (to generate dst_valid).
    
    // Logic:
    // SRC Domain:
    //   - Update wr_ptr on posedge src_clk.
    //   - Calculate full_flag = (wr_ptr == rd_ptr + 2).
    //   - Send full_flag to DST domain (synchronizer).
    //   - Calculate empty_flag = (wr_ptr == rd_ptr).

    reg [1:0] wr_ptr_sync2;

    always @(posedge src_clk_i or negedge src_rst_ni) begin

            wr_ptr <= 2'b00;

            if (src_valid_i && src_ready_o) begin

            wr_ptr_sync1 <= wr_ptr;
            wr_ptr_sync2 <= wr_ptr_sync1;

                if (rd_ptr != 2'b00) begin // Not empty
                    rd_ptr <= rd_ptr + 1'b1;
                    dst_data_o <= mem[rd_ptr]; // Read from memory
                end
            end
            // Update sync registers
            wr_ptr_sync1 <= wr_ptr;
            wr_ptr_sync2 <= wr_ptr_sync1;
            rd_ptr_sync1 <= rd_ptr;
            rd_ptr_sync2 <= rd_ptr_sync1;
        end
    end

    // Calculate Empty flag in DST
    // Empty if rd_ptr == wr_ptr_sync2
    reg empty_flag_dst;
    assign empty_flag_dst = (rd_ptr == wr_ptr_sync2);

    // Calculate Full flag in DST (to stop reading if needed, though usually we just don't output valid)
    // Full if rd_ptr == wr_ptr_sync2 - 2
    // Or if wr_ptr_sync2 == rd_ptr + 2
    reg full_flag_dst;
    assign full_flag_dst = (wr_ptr_sync2 == 2'b10) && (rd_ptr == 2'b00);

    // --- Handshake Logic ---
    
    // Src Ready: !full_flag
    assign src_ready_o = ~full_flag;

    // Dst Valid: !empty_flag_dst
    assign dst_valid_o = ~empty_flag_dst;

    // Memory initialization
    // In Verilog, reg arrays are initialized to 0 by default if no initial block, 
    // but explicit initialization is good practice.
    // However, since we can't use initial blocks in module level easily without testbench context

    // The reset sets `rd_ptr` to 0. If `wr_ptr` is 0, it's empty.
    // So the first write will happen at index 0.

            mem[0] <= 0;
            mem[1] <= 0;
        end
    end

endmodule
