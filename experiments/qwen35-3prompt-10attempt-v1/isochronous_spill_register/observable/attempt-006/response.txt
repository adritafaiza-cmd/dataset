module isochronous_spill_register #(
    parameter WIDTH = 8
)(
    input              src_clk_i,
    input              src_rst_ni,
    input              src_valid_i,
    output             src_ready_o,
    input  [WIDTH-1:0] src_data_i,
    input              dst_clk_i,
    input              dst_rst_ni,
    output             dst_valid_o,
    input              dst_ready_i,
    output reg [WIDTH-1:0] dst_data_o
);

    // State encoding
    // IDLE: Buffer is empty
    // WAIT_SRC: Waiting for valid data from source
    // WAIT_DST: Waiting for ready from destination
    localparam IDLE = 2'b00;
    localparam WAIT_SRC = 2'b01;
    localparam WAIT_DST = 2'b10;

    reg [1:0] state;
    reg [WIDTH-1:0] buffer_data;
    reg src_valid_reg;
    reg src_ready_reg;
    reg dst_valid_reg;

    // Synchronous logic for source side signals
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            src_valid_reg <= 1'b0;
            src_ready_reg <= 1'b1;
        end else begin
            src_valid_reg <= src_valid_i;
            // Ready is asserted unless we are waiting for destination
            src_ready_reg <= (state == WAIT_SRC) ? 1'b0 : 1'b1;
        end
    end

    // Synchronous logic for destination side signals
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            dst_valid_reg <= 1'b0;
            dst_data_o <= 1'b0; // Drive 0 on reset
        end else begin
            dst_valid_reg <= (state == WAIT_DST) ? 1'b1 : 1'b0;
            // Drive buffer data when valid
            dst_data_o <= (state == WAIT_DST) ? buffer_data : 1'b0;
        end
    end

    // State machine logic
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            state <= IDLE;
        end else begin
            case (state)
                IDLE: begin
                    if (src_valid_i && src_ready_reg) begin
                        // Capture data and move to waiting for destination
                        buffer_data <= src_data_i;
                        state <= WAIT_DST;
                    end
                end
                WAIT_SRC: begin
                    // Stay here until destination is ready
                    // Note: In a 2-deep buffer context, this state effectively
                    // acts as the first stage holding data. If the requirement
                    // implies buffering multiple items, the state machine would
                    // need to track count. However, "2-deep valid/ready spill buffer"
                    // between related clocks often implies a handshake buffer with
                    // potential for 2 cycles of latency or a specific depth.
                    // Given the interface and "deliver accepted data once in order",
                    // a single FIFO element (1-deep) is the standard interpretation
                    // for a simple spill register unless a specific FIFO depth
                    // parameter is requested. 
                    // 
                    // Re-reading "2-deep": This usually implies a depth of 2.
                    // To implement a depth-2 FIFO between clocks, we need a counter
                    // and a pointer.
                    
                    // Let's implement a true 2-deep FIFO logic here.
                    // We will use a counter to track depth.
                    // Since we are in a single clock domain logic block for state,
                    // we need to handle the cross-clock FIFO pointer logic carefully.
                    // However, the prompt asks for a "spill buffer" which often implies
                    // a specific latency or small depth. 
                    // 
                    // Correction: A "2-deep" buffer implies a FIFO with capacity 2.
                    // We need to track the write pointer (src side) and read pointer (dst side).
                    // Since clocks are related, we can use synchronous pointers.
                    
                    // Let's refactor the state machine to support a 2-deep FIFO.
                    // States:
                    // 0: Empty
                    // 1: 1 item
                    // 2: 2 items (Full)
                    
                    // Actually, to keep it simple and robust for "related clocks",
                    // we can use a single counter for depth and pointers.
                    // But the prompt asks for a specific behavior: "Buffer a short rate mismatch".
                    // A 2-deep FIFO is the standard solution.
                    
                    // Let's switch to a counter-based approach for the 2-deep requirement.
                    // We will move the state logic into the counter logic.
                    
                    // Wait, the previous state machine was for a 1-deep buffer.
                    // To strictly follow "2-deep", I must implement a FIFO with depth 2.
                    // Let's restart the logic block below to implement the 2-deep FIFO correctly.
                    
                    // Placeholder to avoid compilation error in this thought block,
                    // actual implementation follows below.
                    state <= IDLE; 
                end
                default: state <= IDLE;
            endcase
        end
    end

    // --- 2-Deep FIFO Implementation ---
    
    // Pointers: 2 bits are sufficient for depth 2 (00, 01, 10)
    // We use a simple counter for write pointer and read pointer.
    // Since clocks are related, we can use synchronous pointers.
    
    reg [1:0] write_ptr;
    reg [1:0] read_ptr;
    reg [WIDTH-1:0] fifo_mem [0:1]; // Depth 2
    reg [1:0] full_mask; // Mask to indicate full status (11 = full)
    reg [1:0] empty_mask; // Mask to indicate empty status (00 = empty)

    // Write logic (Source Clock Domain)
    always @(posedge src_clk_i or negedge src_rst_ni) begin
        if (!src_rst_ni) begin
            write_ptr <= 2'b00;
            fifo_mem <= 2'b0;
            full_mask <= 2'b00;
        end else begin
            // Write enable: valid is high AND we are not full
            // Full condition: write_ptr == read_ptr + 2 (wrapping)
            // For depth 2:
            // 00 -> 01 -> 10 -> 00 (Full)
            // 00 -> 01 -> 10 -> 11 (Empty? No, 11 is usually full in some encodings)
            // Let's use standard circular buffer logic.
            
            // Check if full: (write_ptr + 1) == read_ptr (for depth 2, indices 0,1)
            // If write_ptr=0, next=1. If read_ptr=1, full.
            // If write_ptr=1, next=0. If read_ptr=0, full.
            
            if (src_valid_i && src_ready_reg) begin
                // Check full condition
                // If (write_ptr + 1) == read_ptr, then full
                // For depth 2:
                // Case 1: w=0, r=1 -> next w=1 != r(1) -> Not full? Wait.
                // Let's trace:
                // Init: w=0, r=0, empty.
                // Write 1: w=1, r=0. Data at [1].
                // Write 2: w=0, r=0. Data at [0]. Now w=0, r=0. Is it full?
                // If w=0 and r=0, it could be empty or full.
                // We need a separate full flag or check count.
                
                // Simpler approach for small depth:
                // Count items.
                // But let's stick to pointers for generality.
                // Depth 2:
                // Empty: w == r
                // Full: (w + 1) % 2 == r
                
                if (write_ptr == 2'b00 && read_ptr == 2'b01) begin
                    // Full
                    src_ready_reg <= 1'b0;
                end else if (write_ptr == 2'b01 && read_ptr == 2'b00) begin
                    // Full
                    src_ready_reg <= 1'b0;
                end else begin
                    // Not full
                    src_ready_reg <= 1'b1;
                    fifo_mem[write_ptr] <= src_data_i;
                    write_ptr <= write_ptr + 1'b1;
                    if (write_ptr == 2'b10) write_ptr <= 2'b00;
                end
            end else begin
                src_ready_reg <= 1'b1; // Ready if not full
            end
        end
    end

    // Read logic (Destination Clock Domain)
    always @(posedge dst_clk_i or negedge dst_rst_ni) begin
        if (!dst_rst_ni) begin
            read_ptr <= 2'b00;
            dst_valid_reg <= 1'b0;
            dst_data_o <= 1'b0;
        end else begin
            // Check if empty: w == r
            if (write_ptr == read_ptr) begin
                dst_valid_reg <= 1'b0;
                dst_data_o <= 1'b0;
            end else begin
                dst_valid_reg <= 1'b1;
                dst_data_o <= fifo_mem[read_ptr];
                // Advance read pointer
                read_ptr <= read_ptr + 1'b1;
                if (read_ptr == 2'b10) read_ptr <= 2'b00;
            end
        end
    end

    // Output assignments
    assign src_ready_o = src_ready_reg;
    assign dst_valid_o = dst_valid_reg;

endmodule
