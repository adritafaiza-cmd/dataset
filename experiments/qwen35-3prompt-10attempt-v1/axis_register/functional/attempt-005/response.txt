module axis_register #
(
    parameter DATA_WIDTH = 8,
    parameter KEEP_ENABLE = (DATA_WIDTH>8),
    parameter KEEP_WIDTH = ((DATA_WIDTH+7)/8),
    parameter LAST_ENABLE = 1,
    parameter ID_ENABLE = 0,
    parameter ID_WIDTH = 8,
    parameter DEST_ENABLE = 0,
    parameter DEST_WIDTH = 8,
    parameter USER_ENABLE = 1,
    parameter USER_WIDTH = 1,
    parameter REG_TYPE = 2
)
(
    input  wire                   clk,
    input  wire                   rst,
    /*
     * AXI Stream input
     */
    input  wire [DATA_WIDTH-1:0]  s_axis_tdata,
    input  wire [KEEP_WIDTH-1:0]  s_axis_tkeep,
    input  wire                   s_axis_tvalid,
    output wire                   s_axis_tready,
    input  wire                   s_axis_tlast,
    input  wire [ID_WIDTH-1:0]    s_axis_tid,
    input  wire [DEST_WIDTH-1:0]  s_axis_tdest,
    input  wire [USER_WIDTH-1:0]  s_axis_tuser,
    /*
     * AXI Stream output
     */
    output wire [DATA_WIDTH-1:0]  m_axis_tdata,
    output wire [KEEP_WIDTH-1:0]  m_axis_tkeep,
    output wire                   m_axis_tvalid,
    input  wire                   m_axis_tready,
    output wire                   m_axis_tlast,
    output wire [ID_WIDTH-1:0]    m_axis_tid,
    output wire [DEST_WIDTH-1:0]  m_axis_tdest,
    output wire [USER_WIDTH-1:0]  m_axis_tuser
);

    // Internal registers for state
    reg [DATA_WIDTH-1:0] reg_tdata;
    reg [KEEP_WIDTH-1:0] reg_tkeep;
    reg reg_tlast;
    reg [ID_WIDTH-1:0] reg_tid;
    reg [DEST_WIDTH-1:0] reg_tdest;
    reg [USER_WIDTH-1:0] reg_tuser;
    reg reg_valid;

    // Assignments for output ports based on register state
    assign m_axis_tdata = reg_tdata;
    assign m_axis_tkeep = reg_tkeep;
    assign m_axis_tlast = reg_tlast;
    assign m_axis_tid = reg_tid;
    assign m_axis_tdest = reg_tdest;
    assign m_axis_tuser = reg_tuser;
    assign m_axis_tvalid = reg_valid;

    // Handshake logic
    assign s_axis_tready = (REG_TYPE == 2) ? m_axis_tready : 1'b1;

    // Sequential logic
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            reg_tdata <= 0;
            reg_tkeep <= 0;
            reg_tlast <= 0;
            reg_tid <= 0;
            reg_tdest <= 0;
            reg_tuser <= 0;
            reg_valid <= 0;
        end
        else begin
            // Logic for REG_TYPE 0 (Wire) and 1 (Simple Buffer)
            if (REG_TYPE < 2) begin
                if (s_axis_tvalid && s_axis_tready) begin
                    reg_tdata <= s_axis_tdata;
                    reg_tkeep <= s_axis_tkeep;
                    reg_tlast <= s_axis_tlast;
                    reg_tid <= s_axis_tid;
                    reg_tdest <= s_axis_tdest;
                    reg_tuser <= s_axis_tuser;
                    reg_valid <= 1;
                end
                else if (s_axis_tvalid && !s_axis_tready) begin
                    // For wire/buffer, we cannot hold data if we don't have a ready signal
                    // effectively dropping the beat or stalling input. 
                    // Standard behavior for non-skid buffers on stall is to drop the beat.
                    reg_valid <= 0;
                end
            end
            // Logic for REG_TYPE 2 (Skid Buffer)
            else begin
                if (s_axis_tvalid && s_axis_tready) begin
                    reg_tdata <= s_axis_tdata;
                    reg_tkeep <= s_axis_tkeep;
                    reg_tlast <= s_axis_tlast;
                    reg_tid <= s_axis_tid;
                    reg_tdest <= s_axis_tdest;
                    reg_tuser <= s_axis_tuser;
                    reg_valid <= 1;
                end
                else if (s_axis_tvalid && !s_axis_tready) begin
                    // Skid buffer: hold the data until ready
                    // Do not update registers, keep current state
                end
                else if (!s_axis_tvalid && reg_valid) begin
                    // Input invalid but we have data, wait for ready
                    // Do nothing, hold state
                end
                else if (!s_axis_tvalid && !reg_valid) begin
                    // Both invalid, stay empty
                end
            end
        end
    end

endmodule
