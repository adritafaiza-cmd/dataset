
module cdc_2phase #(
  parameter WIDTH = 1
)(
  input                  src_rst_ni,
  input                  src_clk_i,
  input      [WIDTH-1:0] src_data_i,
  input                  src_valid_i,
  output                 src_ready_o,

  input                  dst_rst_ni,
  input                  dst_clk_i,
  output     [WIDTH-1:0] dst_data_o,
  output                 dst_valid_o,
  input                  dst_ready_i
);

  reg [WIDTH-1:0] data;
  reg valid;
  wire ready;

  always @(posedge src_clk_i) begin
    if (!src_rst_ni) begin
      valid <= 0;
    end else if (src_valid_i && ready) begin
      data <= src_data_i;
      valid <= 1;
    end
  end

  assign src_ready_o = ready;

  always @(posedge dst_clk_i) begin
    if (!dst_rst_ni) begin
      dst_valid_o <= 0;
    end else if (dst_ready_i && valid) begin
      dst_data_o <= data;
      dst_valid_o <= 1;
      valid <= 0;
    end
  end

  assign ready = !valid;

endmodule
