module async_fifo #(
  parameter DSIZE = 8,
  parameter ASIZE = 4,
  parameter FALLTHROUGH = "TRUE"
)(
  input  wire             wclk,
  input  wire             wrst_n,
  input  wire             winc,
  input  wire [DSIZE-1:0] wdata,
  output wire             wfull,
  output wire             awfull,

  input  wire             rclk,
  input  wire             rrst_n,
  input  wire             rinc,
  output wire [DSIZE-1:0] rdata,
  output wire             rempty,
  output wire             arempty
);

  localparam FSIZE = 1 << ASIZE;

  reg [DSIZE-1:0] mem [FSIZE-1:0];
  reg [ASIZE-1:0] wptr;
  reg [ASIZE-1:0] rptr;
  reg [ASIZE-1:0] wptr_next;
  reg [ASIZE-1:0] rptr_next;
  reg [ASIZE:0]   wcount;
  reg [ASIZE:0]   rcount;
  reg [ASIZE:0]   wcount_next;
  reg [ASIZE:0]   rcount_next;

  wire [ASIZE-1:0] wptr_next_wrap = wptr_next ^ {ASIZE{1'b1}};
  wire [ASIZE-1:0] rptr_next_wrap = rptr_next ^ {ASIZE{1'b1}};

  assign wfull = (wcount == FSIZE);
  assign awfull = (wcount_next == FSIZE);
  assign rempty = (rcount == 0);
  assign arempty = (rcount_next == 0);

  always @(posedge wclk or negedge wrst_n) begin
    if (~wrst_n) begin
      wptr <= 0;
      wcount <= 0;
    end else if (winc && ~wfull) begin
      mem[wptr] <= wdata;
      wptr <= wptr_next;
      wcount <= wcount_next;
    end
  end

  always @(posedge rclk or negedge rrst_n) begin
    if (~rrst_n) begin
      rptr <= 0;
      rcount <= 0;
    end else if (rinc && ~rempty) begin
      rdata <= mem[rptr];
      rptr <= rptr_next;
      rcount <= rcount_next;
    end
  end

  always @(*) begin
    wptr_next = wptr;
    wcount_next = wcount;
    if (winc && ~wfull) begin
      wptr_next = wptr_next_wrap + 1;
      wcount_next = wcount_next + 1;
    end
  end

  always @(*) begin
    rptr_next = rptr;
    rcount_next = rcount;
    if (rinc && ~rempty) begin
      rptr_next = rptr_next_wrap + 1;
      rcount_next = rcount_next - 1;
    end
  end

  assign rdata = (FALLTHROUGH == "TRUE") ? mem[rptr] : rdata;
