module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    localparam FSIZE = 2**ASIZE;

    reg [DSIZE-1:0] mem [0:FSIZE-1];
    reg [ASIZE-1:0] wptr;
    reg [ASIZE-1:0] rptr;
    reg [ASIZE:0]   wcount;
    reg [ASIZE:0]   rcount;

    assign wfull = (wcount == FSIZE);
    assign awfull = (wcount == FSIZE-1);
    assign rempty = (rcount == 0);
    assign arempty = (rcount == 1);

    always @(posedge wclk or negedge wrst_n) begin
        if (~wrst_n) begin
            wptr <= 0;
            wcount <= 0;
        end else if (winc && ~wfull) begin
            mem[wptr] <= wdata;
            wptr <= wptr + 1;
            wcount <= wcount + 1;
        end
    end

    always @(posedge rclk or negedge rrst_n) begin
        if (~rrst_n) begin
            rptr <= 0;
            rcount <= 0;
        end else if (rinc && ~rempty) begin
            rdata <= mem[rptr];
            rptr <= rptr + 1;
            rcount <= rcount - 1;
        end
    end

    always @(posedge wclk or negedge wrst_n) begin
        if (~wrst_n) begin
            wcount <= 0;
        end else if (winc && ~wfull) begin
            wcount <= wcount + 1;
        end else if (~winc && ~wfull) begin
            wcount <= wcount - 1;
        end
    end

    always @(posedge rclk or negedge rrst_n) begin
        if (~rrst_n) begin
            rcount <= 0;
        end else if (rinc && ~rempty) begin
            rcount <= rcount - 1;
        end else if (~rinc && ~rempty) begin
            rcount <= rcount + 1;
        end
    end
