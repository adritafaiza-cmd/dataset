module async_fifo
    #(
        parameter DSIZE = 8,
        parameter ASIZE = 4,
        parameter FALLTHROUGH = "TRUE"
    )(
        input  wire             wclk,
        input  wire             wrst_n,
        input  wire             winc,
        input  wire [DSIZE-1:0] wdata,
        output wire             wfull,
        output wire             awfull,
        input  wire             rclk,
        input  wire             rrst_n,
        input  wire             rinc,
        output wire [DSIZE-1:0] rdata,
        output wire             rempty,
        output wire             arempty
    );

    // Local parameters
    localparam DEPTH = 1 << ASIZE;

    // Internal signals
    reg [DSIZE-1:0] mem [0:DEPTH-1];
    reg [ASIZE-1:0] wptr;
    reg [ASIZE-1:0] rptr;
    reg [ASIZE:0]   wcnt;
    reg [ASIZE:0]   rcnt;
    wire [ASIZE:0]  wcnt_next;
    wire [ASIZE:0]  rcnt_next;
    wire            wfull_int;
    wire            awfull_int;
    wire            rempty_int;
    wire            arempty_int;

    // Write pointer
    always @(posedge wclk or negedge wrst_n) begin
        if (~wrst_n) begin
            wptr <= 0;
        end else if (winc && ~wfull_int) begin
            wptr <= wptr + 1;
        end
    end

    // Read pointer
    always @(posedge rclk or negedge rrst_n) begin
        if (~rrst_n) begin
            rptr <= 0;
        end else if (rinc && ~rempty_int) begin
            rptr <= rptr + 1;
        end
    end

    // Write counter
    assign wcnt_next = wcnt + winc;
    always @(posedge wclk or negedge wrst_n) begin
        if (~wrst_n) begin
            wcnt <= 0;
        end else if (winc && ~wfull_int) begin
            wcnt <= wcnt_next;
        end
    end

    // Read counter
    assign rcnt_next = rcnt + rinc;
    always @(posedge rclk or negedge rrst_n) begin
        if (~rrst_n) begin
            rcnt <= 0;
        end else if (rinc && ~rempty_int) begin
            rcnt <= rcnt_next;
        end
    end

    // Full and almost full
    assign wfull_int = (wcnt == DEPTH);
    assign awfull_int = (wcnt == DEPTH - 1);

    // Empty and almost empty
    assign rempty_int = (rcnt == 0);
    assign arempty_int = (rcnt == 1);

    // Write data
    always @(posedge wclk or negedge wrst_n) begin
        if (~wrst_n) begin
            mem <= 0;
        end else if (winc && ~wfull_int) begin
            mem[wptr] <= wdata;
        end
    end

    // Read data
    assign rdata = mem[rptr];

    // Output ports
    assign wfull = wfull_int;
    assign awfull = awfull_int;
    assign rempty = rempty_int;
    assign arempty = arempty_int;
