module cdc_fifo_2phase #(
    parameter WIDTH = 8,
    parameter LOG_DEPTH = 3
)(
    input                  src_rst_ni,
    input                  src_clk_i,
    input  [WIDTH-1:0]     src_data_i,
    input                  src_valid_i,
    output                 src_ready_o,
    input                  dst_rst_ni,
    input                  dst_clk_i,
    output [WIDTH-1:0]     dst_data_o,
    output                 dst_valid_o,
    input                  dst_ready_i
);

    localparam DEPTH = 1 << LOG_DEPTH;

    reg [WIDTH-1:0] fifo [DEPTH-1:0];
    reg [LOG_DEPTH-1:0] head;
    reg [LOG_DEPTH-1:0] tail;
    reg [LOG_DEPTH-1:0] count;

    wire [LOG_DEPTH-1:0] next_head = head + 1'b1;
    wire [LOG_DEPTH-1:0] next_tail = tail + 1'b1;

    wire empty = (count == 0);
    wire full = (count == DEPTH);

    always @(posedge src_clk_i) begin
        if (!src_rst_ni) begin
            head <= 0;
            count <= 0;
        end else if (src_valid_i && !full) begin
            fifo[head] <= src_data_i;
            head <= next_head;
            count <= count + 1'b1;
        end
    end

    always @(posedge dst_clk_i) begin
        if (!dst_rst_ni) begin
            tail <= 0;
            count <= 0;
        end else if (dst_ready_i && !empty) begin
            dst_data_o <= fifo[tail];
            tail <= next_tail;
            count <= count - 1'b1;
        end
    end

    assign src_ready_o = !full;
    assign dst_valid_o = !empty;
