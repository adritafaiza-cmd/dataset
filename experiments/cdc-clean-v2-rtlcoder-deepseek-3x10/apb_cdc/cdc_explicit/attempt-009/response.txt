The APB clock-domain bridge can be implemented using a handshake mechanism between the source and destination domains. The source domain sends a request to the destination domain, and the destination domain sends an acknowledgement back to the source domain. The source domain waits for the acknowledgement before sending the next request.

The following Verilog code implements the APB clock-domain bridge:

```

    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1
)(
    input                      src_pclk_i,
    input                      src_preset_ni,
    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,
    input  [ADDR_WIDTH-1:0]    src_paddr_i,
    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    // Internal signals
    reg [ADDR_WIDTH-1:0]    paddr_reg;
    reg [DATA_WIDTH-1:0]    pwdata_reg;
    reg [DATA_WIDTH/8-1:0]  pstrb_reg;
    reg [2:0]               pprot_reg;
    reg                     psel_reg;
    reg                     penable_reg;
    reg                     pwrite_reg;
    reg                     pready_reg;
    reg [DATA_WIDTH-1:0]    prdata_reg;
    reg                     pslverr_reg;

    // Reset logic
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (~src_preset_ni) begin
            paddr_reg <= 0;
            pwdata_reg <= 0;
            pstrb_reg <= 0;
            pprot_reg <= 0;
            psel_reg <= 0;
            penable_reg <= 0;
            pwrite_reg <= 0;
            pready_reg <= 0;
            prdata_reg <= 0;
            pslverr_reg <= 0;
        end else begin
            paddr_reg <= src_paddr_i;
            pwdata_reg <= src_pwdata_i;
            pstrb_reg <= src_pstrb_i;
            pprot_reg <= src_pprot_i;
            psel_reg <= src_psel_i;
            penable_reg <= src_penable_i;
            pwrite_reg <= src_pwrite_i;
            pready_reg <= src_pready_o;
            prdata_reg <= src_prdata_o;
            pslverr_reg <= src_pslverr_o;
        end
    end

    // Source domain handshake logic
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (~src_preset_ni) begin
            src_pready_o <= 0;
            src_prdata_o <= 0;
            src_pslverr_o <= 0;
        end else begin
            if (psel_reg && penable_reg) begin
                dst_psel_o <= 1;
                dst_penable_o <= 1;
                dst_pwrite_o <= pwrite_reg;
                dst_paddr_o <= paddr_reg;
                dst_pwdata_o <= pwdata_reg;
                dst_pstrb_o <= pstrb_reg;
                dst_pprot_o <= pprot_reg;
                if (dst_pready_i) begin
                    src_pready_o <= 1;
                    src_prdata_o <= dst_prdata_i;
                    src_pslverr_o <= dst_pslverr_i;
                    dst_psel_o <= 0;
                    dst_penable_o <= 0;
                    dst_pwrite_o <= 0;
                    dst_paddr_o <= 0;
                    dst_pwdata_o <= 0;
                    dst_pstrb_o <= 0;
                    dst_pprot_o <= 0;
                end
            end else begin
                dst_psel_o <= 0;
                dst_penable_o <= 0;
                dst_pwrite_o <= 0;
                dst_paddr_o <= 0;
                dst_pwdata_o <= 0;
                dst_pstrb_o <= 0;
                dst_pprot_o <= 0;
                src_pready_o <= 0;
                src_prdata_o <= 0;
                src_pslverr_o <= 0;
            end
        end
    end
