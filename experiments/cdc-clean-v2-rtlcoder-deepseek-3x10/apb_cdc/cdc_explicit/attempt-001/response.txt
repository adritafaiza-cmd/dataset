    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32,
    parameter LOG_DEPTH = 1

    input                      src_pclk_i,

    input                      src_psel_i,
    input                      src_penable_i,
    input                      src_pwrite_i,

    input  [DATA_WIDTH-1:0]    src_pwdata_i,
    input  [DATA_WIDTH/8-1:0]  src_pstrb_i,
    input  [2:0]               src_pprot_i,
    output                     src_pready_o,
    output [DATA_WIDTH-1:0]    src_prdata_o,
    output                     src_pslverr_o,
    input                      dst_pclk_i,
    input                      dst_preset_ni,
    output                     dst_psel_o,
    output                     dst_penable_o,
    output                     dst_pwrite_o,
    output [ADDR_WIDTH-1:0]    dst_paddr_o,
    output [DATA_WIDTH-1:0]    dst_pwdata_o,
    output [DATA_WIDTH/8-1:0]  dst_pstrb_o,
    output [2:0]               dst_pprot_o,
    input                      dst_pready_i,
    input  [DATA_WIDTH-1:0]    dst_prdata_i,
    input                      dst_pslverr_i
);

    // Internal signals
    reg [ADDR_WIDTH-1:0] addr_reg;
    reg [DATA_WIDTH-1:0] data_reg;
    reg [DATA_WIDTH/8-1:0] strb_reg;
    reg [2:0] prot_reg;
    reg sel_reg;
    reg enable_reg;
    reg write_reg;
    reg ready_reg;
    reg [DATA_WIDTH-1:0] prdata_reg;
    reg slverr_reg;
    reg [LOG_DEPTH-1:0] depth_reg;

    // Reset logic
    always @(posedge src_pclk_i or negedge src_preset_ni) begin
        if (~src_preset_ni) begin
            addr_reg <= 0;
            data_reg <= 0;
            strb_reg <= 0;
            prot_reg <= 0;
            sel_reg <= 0;
            enable_reg <= 0;
            write_reg <= 0;
            ready_reg <= 0;
            prdata_reg <= 0;
            slverr_reg <= 0;
            depth_reg <= 0;
        end else begin
            if (src_psel_i && ~sel_reg) begin
                addr_reg <= src_paddr_i;
                data_reg <= src_pwdata_i;
                strb_reg <= src_pstrb_i;
                prot_reg <= src_pprot_i;
                sel_reg <= 1;
                enable_reg <= src_penable_i;
                write_reg <= src_pwrite_i;
                ready_reg <= 0;
                prdata_reg <= 0;
                slverr_reg <= 0;
                depth_reg <= depth_reg + 1;
            end else if (enable_reg && ~ready_reg) begin
                ready_reg <= 1;
                prdata_reg <= dst_prdata_i;
                slverr_reg <= dst_pslverr_i;
            end else if (ready_reg && dst_pready_i) begin
                sel_reg <= 0;
                enable_reg <= 0;
                write_reg <= 0;
                ready_reg <= 0;
                depth_reg <= depth_reg - 1;
            end
        end
    end

    // Output logic
    assign src_pready_o = ready_reg;
    assign src_prdata_o = prdata_reg;
    assign src_pslverr_o = slverr_reg;
    assign dst_psel_o = sel_reg;
    assign dst_penable_o = enable_reg;
    assign dst_pwrite_o = write_reg;
    assign dst_paddr_o = addr_reg;
    assign dst_pwdata_o = data_reg;
    assign dst_pstrb_o = strb_reg;
    assign dst_pprot_o = prot_reg;
