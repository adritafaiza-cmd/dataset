            dst_fifo_empty <= 1;
        end else begin
            if (dst_penable_reg && dst_pready_reg) begin
                if (dst_fifo_head == dst_fifo_tail) begin
                    dst_fifo_full <= 1;
                end else begin
                    dst_fifo_full <= 0;
                end
                if (dst_fifo_head == dst_fifo_tail) begin
                    dst_fifo_empty <= 1;
                end else begin
                    dst_fifo_empty <= 0;
                end
            end
        end
    end

    // APB slave interface
    always @(posedge src_pclk_i) begin
        if (!src_preset_ni) begin
            src_pready_reg <= 0;
            src_prdata_o <= 0;
            src_pslverr_o <= 0;
        end else begin
            if (src_psel_i && src_penable_i) begin
                src_pready_reg <= 0;
                src_prdata_o <= 0;
                src_pslverr_o <= 0;
            end else if (src_psel_i && !src_penable_i) begin
                src_pready_reg <= 1;
                src_prdata_o <= dst_prdata_reg;
                src_pslverr_o <= dst_pslverr_reg;
            end else begin
                src_pready_reg <= 0;
                src_prdata_o <= 0;
                src_pslverr_o <= 0;
            end
        end
    end

    // APB master interface
    always @(posedge dst_pclk_i) begin
        if (!dst_preset_ni) begin
            dst_psel_o <= 0;
            dst_penable_o <= 0;
            dst_pwrite_o <= 0;
            dst_paddr_o <= 0;
            dst_pwdata_o <= 0;
            dst_pstrb_o <= 0;
            dst_pprot_o <= 0;
        end else begin
            if (dst_psel_reg && dst_penable_reg) begin
                dst_psel_o <= 1;
                dst_penable_o <= 1;
                dst_pwrite_o <= src_pwrite_reg;
                dst_paddr_o <= src_paddr_i;
                dst_pwdata_o <= src_pwdata_i;
                dst_pstrb_o <= src_pstrb_i;
                dst_pprot_o <= src_pprot_i;
            end else begin
                dst_psel_o <= 0;
                dst_penable_o <= 0;
                dst_pwrite_o <= 0;
                dst_paddr_o <= 0;
                dst_pwdata_o <= 0;
                dst_pstrb_o <= 0;
                dst_pprot_o <= 0;
            end
        end
    end
