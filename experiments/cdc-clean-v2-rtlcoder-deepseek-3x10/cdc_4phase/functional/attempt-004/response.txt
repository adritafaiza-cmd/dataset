  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Internal signals
  reg [WIDTH-1:0] src_data_reg;
  reg src_valid_reg;
  wire src_ready_wire;
  wire dst_valid_wire;
  reg [WIDTH-1:0] dst_data_reg;
  reg dst_valid_reg;
  wire dst_ready_wire;

  // Source domain
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_reg <= {WIDTH{1'b0}};
      src_valid_reg <= 1'b0;
    end else if (src_valid_i && src_ready_wire) begin
      src_data_reg <= src_data_i;
      src_valid_reg <= 1'b1;
    end else begin
      src_valid_reg <= 1'b0;
    end
  end

  assign src_ready_wire = dst_ready_i || (DECOUPLED && dst_valid_reg);
  assign src_ready_o = src_ready_wire;

  // Destination domain
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg <= {WIDTH{1'b0}};
      dst_valid_reg <= 1'b0;
    end else if (dst_ready_wire && src_valid_reg) begin
      dst_data_reg <= src_data_reg;
      dst_valid_reg <= 1'b1;
    end else begin
      dst_valid_reg <= 1'b0;
    end
  end

  assign dst_valid_wire = src_valid_reg && !dst_valid_reg;
  assign dst_valid_o = dst_valid_wire;
  assign dst_data_o = dst_data_reg;

  // Reset message
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_reg <= RESET_MSG;
      src_valid_reg <= SEND_RESET_MSG;
    end
  end

  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_data_reg <= RESET_MSG;
      dst_valid_reg <= SEND_RESET_MSG;
    end
  end
