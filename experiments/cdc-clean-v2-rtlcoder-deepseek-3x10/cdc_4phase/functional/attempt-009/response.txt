  parameter WIDTH = 1,
  parameter DECOUPLED = 1,
  parameter SEND_RESET_MSG = 0,
  parameter [WIDTH-1:0] RESET_MSG = {WIDTH{1'b0}}
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  // Internal signals
  reg [WIDTH-1:0] src_data_reg;
  reg src_valid_reg;
  reg src_ready_reg;
  reg [WIDTH-1:0] dst_data_reg;
  reg dst_valid_reg;
  reg dst_ready_reg;
  reg [1:0] src_phase_reg;
  reg [1:0] dst_phase_reg;

  // Assign output ports
  assign src_ready_o = src_ready_reg;
  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

  // Source phase
  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_phase_reg <= 2'b00;
      src_valid_reg <= 1'b0;
      src_ready_reg <= 1'b0;
      src_data_reg <= {WIDTH{1'b0}};
    end else begin
      case (src_phase_reg)
        2'b00: begin // Idle
          if (src_valid_i && src_ready_o) begin
            src_phase_reg <= 2'b01;
            src_valid_reg <= 1'b1;
            src_ready_reg <= 1'b0;
            src_data_reg <= src_data_i;
          end
        end
        2'b01: begin // Stage 1
          if (dst_ready_i) begin
            src_phase_reg <= 2'b10;
            src_valid_reg <= 1'b0;
            src_ready_reg <= 1'b1;
          end
        end
        2'b10: begin // Stage 2
          if (dst_ready_i) begin
            src_phase_reg <= 2'b11;
            src_valid_reg <= 1'b0;
            src_ready_reg <= 1'b1;
          end
        end
        2'b11: begin // Stage 3
          if (dst_ready_i) begin
            src_phase_reg <= 2'b00;
            src_valid_reg <= 1'b0;
            src_ready_reg <= 1'b1;
          end
        end
      endcase
    end
  end

  // Destination phase
  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_phase_reg <= 2'b00;
      dst_valid_reg <= 1'b0;
      dst_ready_reg <= 1'b0;
      dst_data_reg <= {WIDTH{1'b0}};
    end else begin
      case (dst_phase_reg)
        2'b00: begin // Idle
          if (src_valid_reg && src_ready_reg) begin
            dst_phase_reg <= 2'b01;
            dst_valid_reg <= 1'b1;
            dst_ready_reg <= 1'b0;
            dst_data_reg <= src_data_reg;

            dst_phase_reg <= 2'b10;
            dst_valid_reg <= 1'b1;
            dst_ready_reg <= 1'b0;
            dst_data_reg <= RESET_MSG;
          end
        end

            dst_phase_reg <= 2'b00;
            dst_valid_reg <= 1'b0;
            dst_ready_reg <= 1'b1;
          end
        end
      endcase
    end
  end
