			wr_ptr <= 0;
			wr_ptr_next <= 0;
			wr_ptr_next_next <= 0;
			wr_ptr_next_next_next <= 0;
			wr_sync <= 0;
		end else begin

			wr_ptr_next <= wr_ptr_next_next;

			wr_ptr_next_next_next <= wr_ptr_next_next_next_next;
			wr_sync <= {wr_sync[NFF-2:0], i_wr};
		end

				o_rd_data <= fifo[rd_ptr_next_next_next_next];
			end
		end
