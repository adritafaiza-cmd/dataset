The following Verilog code implements a pipelined Wishbone clock-domain bridge using a FIFO to buffer the data between the two clock domains. The FIFO is implemented using a shift register and a pointer to keep track of the head and tail of the FIFO.

				DW=32,
				LGFIFO = 5

		input	wire			i_wb_cyc, i_wb_stb, i_wb_we,
		input	wire	[(AW-1):0]	i_wb_addr,
		input	wire	[(DW-1):0]	i_wb_data,
		input	wire	[(DW/8-1):0]	i_wb_sel,
		output	wire			o_wb_stall,
		output	reg			o_wb_ack,
		output	reg	[(DW-1):0]	o_wb_data,
		output	reg			o_wb_err,

		output	reg			o_xclk_cyc,
		output	reg			o_xclk_stb,
		output	reg			o_xclk_we,
		output	reg	[(AW-1):0]	o_xclk_addr,
		output	reg	[(DW-1):0]	o_xclk_data,
		output	reg	[(DW/8-1):0]	o_xclk_sel,
		input	wire			i_xclk_stall,
		input	wire			i_xclk_ack,
		input	wire	[(DW-1):0]	i_xclk_data,
		input	wire			i_xclk_err
	);

	// FIFO parameters
	localparam FIFO_DEPTH = 1 << LGFIFO;
	localparam FIFO_ADDR_WIDTH = LGFIFO;

	// FIFO signals
	reg [DW-1:0] fifo [0:FIFO_DEPTH-1];
	reg [FIFO_ADDR_WIDTH-1:0] fifo_head;
	reg [FIFO_ADDR_WIDTH-1:0] fifo_tail;
	reg [FIFO_ADDR_WIDTH-1:0] fifo_next_head;
	reg [FIFO_ADDR_WIDTH-1:0] fifo_next_tail;
	reg [FIFO_ADDR_WIDTH-1:0] fifo_count;
	reg [FIFO_ADDR_WIDTH-1:0] fifo_count_next;

	// FIFO write logic
	always @(posedge i_wb_clk) begin
		if (i_reset) begin
			fifo_head <= 0;
			fifo_tail <= 0;
			fifo_count <= 0;
		end else begin
			if (i_wb_cyc && i_wb_stb) begin
				fifo[fifo_head] <= i_wb_data;
				fifo_head <= fifo_next_head;
				fifo_count_next <= fifo_count + 1;
			end
			fifo_count <= fifo_count_next;
		end
	end

	// FIFO read logic
	always @(posedge i_xclk_clk) begin
		if (i_reset) begin
			o_xclk_cyc <= 0;
			o_xclk_stb <= 0;
			o_xclk_we <= 0;
			o_xclk_addr <= 0;
			o_xclk_data <= 0;
			o_xclk_sel <= 0;
			o_xclk_err <= 0;
			fifo_tail <= 0;
			fifo_count <= 0;
		end else begin
			if (i_xclk_ack) begin
				o_xclk_ack <= 0;
				o_xclk_data <= fifo[fifo_tail];
				o_xclk_err <= i_xclk_err;
				fifo_tail <= fifo_next_tail;
				fifo_count_next <= fifo_count - 1;
			end
			if (i_wb_cyc && i_wb_stb) begin
				o_xclk_cyc <= 1;
				o_xclk_stb <= 1;
				o_xclk_we <= i_wb_we;
				o_xclk_addr <= i_wb_addr;
				o_xclk_data <= i_wb_data;
				o_xclk_sel <= i_wb_sel;
				o_xclk_err <= 0;
				fifo_next_head <= fifo_head + 1;
				fifo_next_tail <= fifo_tail + 1;
			end else begin
				o_xclk_cyc <= 0;
				o_xclk_stb <= 0;
				o_xclk_we <= 0;
				o_xclk_addr <= 0;
				o_xclk_data <= 0;
				o_xclk_sel <= 0;
				o_xclk_err <= 0;
				fifo_next_head <= fifo_head;
				fifo_next_tail <= fifo_tail;
			end
			fifo_count <= fifo_count_next;
		end
	end

	// Stall logic
	assign o_wb_stall = (fifo_count == FIFO_DEPTH);
	assign o_xclk_stall = (fifo_count == 0);

	// Ack logic
	always @(posedge i_wb_clk) begin
		if (i_reset) begin
			o_wb_ack <= 0;
			o_wb_data <= 0;
			o_wb_err <= 0;
		end else begin
			if (i_wb_cyc && i_wb_stb && !o_wb_stall) begin
				o_wb_ack <= 1;
				o_wb_data <= fifo[fifo_tail];
				o_wb_err <= i_xclk_err;
				fifo_tail <= fifo_tail + 1;
			end else begin
				o_wb_ack <= 0;
				o_wb_data <= 0;
				o_wb_err <= 0;
			end
		end
	end
