  parameter WIDTH = 1
)(
  input  src_rst_ni,
  input  src_clk_i,
  input  [WIDTH-1:0] src_data_i,
  input  src_valid_i,
  output src_ready_o,
  input  dst_rst_ni,
  input  dst_clk_i,
  output [WIDTH-1:0] dst_data_o,
  output dst_valid_o,
  input  dst_ready_i
);

  reg [WIDTH-1:0] src_data_reg;
  reg src_valid_reg;
  wire src_ready_wire;
  reg dst_valid_reg;
  wire dst_ready_wire;
  reg [WIDTH-1:0] dst_data_reg;

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (!src_rst_ni) begin
      src_data_reg <= {WIDTH{1'b0}};
      src_valid_reg <= 1'b0;
    end else if (src_valid_i && src_ready_wire) begin
      src_data_reg <= src_data_i;
      src_valid_reg <= 1'b1;
    end else if (dst_ready_wire) begin
      src_valid_reg <= 1'b0;
    end
  end

  assign src_ready_o = src_ready_wire;

  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (!dst_rst_ni) begin
      dst_valid_reg <= 1'b0;
      dst_data_reg <= {WIDTH{1'b0}};
    end else if (src_valid_reg && dst_ready_wire) begin
      dst_valid_reg <= 1'b1;
      dst_data_reg <= src_data_reg;
    end else if (dst_valid_reg && dst_ready_i) begin
      dst_valid_reg <= 1'b0;
    end
  end

  assign dst_data_o = dst_data_reg;
  assign dst_valid_o = dst_valid_reg;

  assign src_ready_wire = dst_ready_wire && !dst_valid_reg;
  assign dst_ready_wire = src_ready_wire && !src_valid_reg;
