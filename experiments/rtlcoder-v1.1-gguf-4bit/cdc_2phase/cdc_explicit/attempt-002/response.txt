
module cdc_2phase #(
  parameter WIDTH = 1
)(
  input                  src_rst_ni,
  input                  src_clk_i,
  input      [WIDTH-1:0] src_data_i,
  input                  src_valid_i,
  output                 src_ready_o,

  input                  dst_rst_ni,
  input                  dst_clk_i,
  output     [WIDTH-1:0] dst_data_o,
  output                 dst_valid_o,
  input                  dst_ready_i
);

  reg [WIDTH-1:0] src_data_r;
  reg [WIDTH-1:0] dst_data_r;
  reg [1:0] state;
  reg [1:0] next_state;

  always @(posedge src_clk_i or negedge src_rst_ni) begin
    if (~src_rst_ni) begin
      src_data_r <= 0;
      src_ready_o <= 1;
      state <= 0;
    end else begin
      case (state)
        0: begin
          if (src_valid_i && dst_ready_i) begin
            src_data_r <= src_data_i;
            src_ready_o <= 0;
            state <= 1;
          end
        end
        1: begin
          dst_data_o <= src_data_r;
          dst_valid_o <= 1;
          state <= 0;
        end
      endcase
    end
  end

  always @(posedge dst_clk_i or negedge dst_rst_ni) begin
    if (~dst_rst_ni) begin
      dst_data_o <= 0;
      dst_valid_o <= 0;
      state <= 0;
    end else begin
      case (state)
        0: begin
          if (src_valid_i && dst_ready_i) begin
            dst_data_o <= src_data_r;
            dst_valid_o <= 1;
            state <= 1;
          end
        end
        1: begin
          dst_valid_o <= 0;
          state <= 0;
        end
      endcase
    end
  end

endmodule
