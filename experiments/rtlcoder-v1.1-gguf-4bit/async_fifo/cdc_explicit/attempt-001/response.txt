
module async_fifo #(
  parameter DSIZE = 8,
  parameter ASIZE = 4,
  parameter FALLTHROUGH = "TRUE"
)(
  input  wire             wclk,
  input  wire             wrst_n,
  input  wire             winc,
  input  wire [DSIZE-1:0] wdata,
  output wire             wfull,
  output wire             awfull,

  input  wire             rclk,
  input  wire             rrst_n,
  input  wire             rinc,
  output wire [DSIZE-1:0] rdata,
  output wire             rempty,
  output wire             arempty
);

  // Define FIFO state
  parameter EMPTY = 2'b00;
  parameter HALF = 2'b01;
  parameter FULL = 2'b10;

  // Define FIFO registers
  reg [DSIZE-1:0] fifo [0:(2**ASIZE)-1];
  reg [ASIZE-1:0] head = EMPTY;
  reg [ASIZE-1:0] tail = EMPTY;
  reg [ASIZE-1:0] count = 0;

  // Define FIFO signals
  wire [ASIZE-1:0] next_head = head + 1;
  wire [ASIZE-1:0] next_tail = tail + 1;
  wire [ASIZE-1:0] next_count = count + 1;
  wire [ASIZE-1:0] prev_count = count - 1;
  wire [ASIZE-1:0] next_prev_count = count + 1;

  // Define FIFO state signals
  wire [1:0] state = {count[ASIZE-1], head[ASIZE-1]};
  wire [1:0] next_state = {next_count[ASIZE-1], next_head[ASIZE-1]};

  // Define FIFO write logic
  always @(posedge wclk, negedge wrst_n) begin
    if (~wrst_n) begin
      head <= EMPTY;
      tail <= EMPTY;
      count <= 0;
    end else if (winc && ~wfull) begin
      fifo[head] <= wdata;
      head <= next_head;
      count <= next_count;
    end
  end

  // Define FIFO read logic
  always @(posedge rclk, negedge rrst_n) begin
    if (~rrst_n) begin
      head <= EMPTY;
      tail <= EMPTY;
      count <= 0;
    end else if (rinc && ~rempty) begin
      rdata <= fifo[tail];
      tail <= next_tail;
      count <= next_prev_count;
    end
  end

  // Define FIFO output logic
  assign wfull = (count == (2**ASIZE));
  assign awfull = (count == (2**ASIZE - 1));
  assign rempty = (count == 0);
  assign arempty = (count == 1);

  // Define FIFO state logic
  always @(state, next_state) begin
    case (state)
      EMPTY: begin
        wfull <= 1'b1;
        rempty <= 1'b1;
      end
      HALF: begin
        wfull <= 1'b1;
        rempty <= 1'b0;
      end
      FULL: begin
        wfull <= 1'b1;
        rempty <= 1'b0;
      end
    endcase
  end

  // Define FIFO output logic for FALLTHROUGH
  assign rdata = (FALLTHROUGH) ? fifo[tail] : 0;

endmodule
