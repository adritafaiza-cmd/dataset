
module async_fifo #(
  parameter DSIZE = 8,
  parameter ASIZE = 4,
  parameter FALLTHROUGH = "TRUE"
)(
  input  wire             wclk,
  input  wire             wrst_n,
  input  wire             winc,
  input  wire [DSIZE-1:0] wdata,
  output wire             wfull,
  output wire             awfull,

  input  wire             rclk,
  input  wire             rrst_n,
  input  wire             rinc,
  output wire [DSIZE-1:0] rdata,
  output wire             rempty,
  output wire             arempty
);

  reg [ASIZE-1:0] wrptr;
  reg [ASIZE-1:0] rrptr;
  reg [ASIZE-1:0] rrptr_next;
  reg [DSIZE-1:0] fifo [2**ASIZE-1:0];
  reg [DSIZE-1:0] rdata_next;
  reg [ASIZE-1:0] count;
  reg [ASIZE-1:0] count_next;
  reg [ASIZE-1:0] count_next_r;
  reg [ASIZE-1:0] count_next_w;
  reg [ASIZE-1:0] count_next_r_next;
  reg [ASIZE-1:0] count_next_w_next;

  assign wfull = (count == 2**ASIZE);
  assign awfull = (count == 2**ASIZE - 1);
  assign rempty = (count == 0);
  assign arempty = (count == 1);

  always @(posedge wclk or negedge wrst_n) begin
    if (~wrst_n) begin
      wrptr <= 0;
      count <= 0;
    end else if (winc && ~wfull) begin
      fifo[wrptr] <= wdata;
      wrptr <= wrptr + 1;
      count <= count + 1;
    end
  end

  always @(posedge rclk or negedge rrst_n) begin
    if (~rrst_n) begin
      rrptr <= 0;
      rrptr_next <= 0;
      rrptr_next_next <= 0;
      count_next_r <= 0;
      count_next_w <= 0;
      count_next_w_next <= 0;
      rdata_next <= 0;
    end else if (rinc && ~rempty) begin
      rrptr_next <= rrptr + 1;
      rrptr <= rrptr_next;
      count_next_r <= count - 1;
      count_next_w <= count_next_r - 1;
      count_next_w_next <= count_next_w - 1;
      rdata_next <= fifo[rrptr];
    end
  end

  always @(posedge rclk or negedge rrst_n) begin
    if (~rrst_n) begin
      count_next_r_next <= 0;
    end else if (rinc && ~rempty) begin
      count_next_r_next <= count_next_r_next + 1;
    end
  end

  assign rdata = (FALLTHROUGH) ? rdata_next : rdata;

endmodule
