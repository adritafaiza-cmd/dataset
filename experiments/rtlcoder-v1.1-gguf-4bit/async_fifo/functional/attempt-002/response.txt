
module async_fifo #(
  parameter DSIZE = 8,
  parameter ASIZE = 4,
  parameter FALLTHROUGH = "TRUE"
)(
  input  wire             wclk,
  input  wire             wrst_n,
  input  wire             winc,
  input  wire [DSIZE-1:0] wdata,
  output wire             wfull,
  output wire             awfull,

  input  wire             rclk,
  input  wire             rrst_n,
  input  wire             rinc,
  output wire [DSIZE-1:0] rdata,
  output wire             rempty,
  output wire             arempty
);

  parameter WIDTH = 2**ASIZE;
  parameter DEPTH = WIDTH - 1;

  reg [DSIZE-1:0] fifo [DEPTH-1:0];
  reg [ASIZE-1:0] head = 0;
  reg [ASIZE-1:0] tail = 0;
  reg [ASIZE-1:0] count = 0;

  assign wfull = (count == WIDTH);
  assign awfull = (count == WIDTH - 1);
  assign rempty = (count == 0);
  assign arempty = (count == 1);

  always @(posedge wclk or negedge wrst_n) begin
    if (~wrst_n) begin
      head <= 0;
      tail <= 0;
      count <= 0;
    end else if (winc && ~wfull) begin
      fifo[head] <= wdata;
      head <= head + 1;
      count <= count + 1;
    end
  end

  always @(posedge rclk or negedge rrst_n) begin
    if (~rrst_n) begin
      head <= 0;
      tail <= 0;
      count <= 0;
    end else if (rinc && ~rempty) begin
      rdata <= fifo[tail];
      tail <= tail + 1;
      count <= count - 1;
    end
  end

  always @(posedge wclk or negedge wrst_n) begin
    if (~wrst_n) begin
      if (FALLTHROUGH == "TRUE") begin
        rdata <= fifo[tail];
      end else begin
        rdata <= 0;
      end
    end
  end

endmodule
